"""
Outgoing Tri‑Multiplicative update (TriMul) – fast Triton + PyTorch implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor  [B·N·N, dim]   (fp32→fp16).
2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single GEMM:
        X_norm @ W_fused.T → (R, 5·H)      with R = B·N·N.
3️⃣  Split the fused result, apply the optional pair‑wise mask,
    and evaluate the sigmoid gates (single Triton kernel).  Produces
    tensors L, R (both [R, H]) and G_out ([R, H]).
4️⃣  Batched half‑precision GEMM:
        out = (L @ Rᵀ)   →   [R, H]   (fp16)
5️⃣  Fuse the second LayerNorm, multiplication by the out‑gate,
    and the final projection back to `dim` into one Triton kernel.
    This eliminates an intermediate [R, H] write‑read and reduces
    memory traffic.
6️⃣  Reshape to [B, N, N, dim] and cast to float32.

All heavy arithmetic is performed in fp16 (Tensor‑Core friendly); the
few reductions needed for LayerNorm are accumulated in fp32 for
numerical stability.
"""

        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values (fp16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ, fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β, fp32
        out = norm * w[None, :] + b[None, :]                  # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const   *[rows, 5*H]   (fp16)
    left_ptr,       # mutable *[rows, H]   (fp16)
    right_ptr,      # mutable *[rows, H]   (fp16)
    out_gate_ptr,   # mutable *[rows, H]   (fp16)
    mask_ptr,       # const   *[rows]      (fp16) – ignored if HAS_MASK==0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load per‑row scalar mask (1.0 if no mask)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process hidden dimension in blocks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Row‑wise offset into the concatenated matrix
        row_off = offs_r[:, None] * (5 * H)               # [BLOCK_R,1]
        col_off = offs_h[None, :]                         # [1,BLOCK_H]

        #   0 ... H-1   → left_proj
        left_raw = tl.load(proj_ptr + row_off + col_off,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)
        #   H ... 2H-1 → right_proj
        right_raw = tl.load(proj_ptr + row_off + H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        #   2H ... 3H-1 → left_gate
        left_gate_raw = tl.load(proj_ptr + row_off + 2 * H + col_off,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        #   3H ... 4H-1 → right_gate
        right_gate_raw = tl.load(proj_ptr + row_off + 3 * H + col_off,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        #   4H ... 5H-1 → out_gate
        out_gate_raw = tl.load(proj_ptr + row_off + 4 * H + col_off,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        tl.store(left_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fused second LayerNorm + out‑gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_proj_kernel(
    inp_ptr,            # const   *[rows, H]   (fp16)  -- result of left@rightᵀ
    gate_ptr,           # const   *[rows, H]   (fp16)  -- out_gate
    out_ptr,            # mutable *[rows, dim] (fp16) -- final result
    proj_weight_ptr,    # const   *[dim, H]   (fp16)  -- to_out.weight
    ln_w_ptr,           # const   *[H]        (fp32)  -- to_out_norm.weight
    ln_b_ptr,           # const   *[H]        (fp32)  -- to_out_norm.bias
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    The kernel performs, for each row r:
        μ,σ²  ← mean / variance of inp[r,:]   (fp32)
        rstd  ← rsqrt(σ² + eps)
        y_h   ← ((inp[r,h] - μ) * rstd) * ln_w[h] + ln_b[h]   // LayerNorm
        y_h   ← y_h * gate[r,h]                               // out‑gate
        out[r,d] ← Σ_h y_h * proj_weight[d,h]                // final Linear
    All heavy work is in fp16, reductions in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1) Compute mean & variance of inp across H (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        vals = tl.load(inp_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2) Loop over output‑dimension blocks and accumulate dot‑products
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # accumulator for this output‑dim block
        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # ---- load raw tensors ----
            inp = tl.load(inp_ptr + offs_r[:, None] * H + offs_h[None, :],
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)        # [BR, BH]
            gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)       # [BR, BH]

            # ---- layer‑norm parameters (γ,β) ----
            gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
            beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # fp32

            # ---- LayerNorm + out‑gate ----
            norm = (inp - mean[:, None]) * rstd[:, None]       # fp32
            ln   = norm * gamma[None, :] + beta[None, :]       # fp32
            y    = ln * gate                                   # fp32

            # ---- weight slice (dim‑block × H‑block) ----
            w = tl.load(proj_weight_ptr + offs_d[:, None] * H + offs_h[None, :],
                        mask=mask_d[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)          # [BD, BH]

            # ---- accumulate dot product ----
            # y : [BR, BH]   w.T : [BH, BD]   →  [BR, BD]
            acc += tl.dot(y, w.T)

        # ---- store result for this block of output dimensions ----
        out_ptr_base = out_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr_base,
                 acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict with all Linear/LayerNorm parameters
                            (all tensors are float32, Linear layers have bias=False)
            - config_dict  : must contain "dim", "hidden_dim",
                             "nomask" (bool) and may contain other entries

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                      # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 512
    BLOCK_D = 32
    grid_ln1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],     # γ  (fp32)
        weights["norm.bias"],       # β  (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free temporary

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5 * hidden_dim)
    # --------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # [5*H, dim]  (fp16)

    proj = F.linear(x_norm, w_concat)   # [R, 5*H]  (fp16)
    del w_concat, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split + mask + sigmoid gates
    # --------------------------------------------------------------
    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_half = mask.to(torch.float16).reshape(R)  # flatten to [R]
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: left @ rightᵀ  (produces [R, H])
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)    # [B*H, N, N]
    right_mat = right.view(B * H, N, N)   # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, [B*H, N, N]
    del left, right, left_mat, right_mat

    out_flat = out_mat.view(R, H).contiguous()   # [R, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Fused LayerNorm (to_out_norm), out‑gate, and final projection
    # --------------------------------------------------------------
    final_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R_FNL = 64      # rows per program
    BLOCK_H_FNL = 32      # hidden‑dim block
    BLOCK_D_FNL = 64      # output‑dim block (fits in registers)

    grid_fnl = (triton.cdiv(R, BLOCK_R_FNL),)
    fused_norm_gate_proj_kernel[grid_fnl](
        out_flat,                                 # inp_ptr
        out_gate.view(R, H),                      # gate_ptr
        final_fp16,                               # out_ptr
        weights["to_out.weight"].to(torch.float16),  # proj_weight_ptr  [dim, H]
        weights["to_out_norm.weight"],            # ln_w_ptr (fp32)
        weights["to_out_norm.bias"],              # ln_b_ptr (fp32)
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R_FNL,
        BLOCK_H=BLOCK_H_FNL,
        BLOCK_D=BLOCK_D_FNL,
        num_warps=8,
    )
    del out_flat, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Reshape to [B, N, N, dim] and cast back to float32
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
