"""
Outgoing Tri‑Multiplicative Update (TriMul) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the input tensor.
2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single GEMM:  X_norm @ W_fusedᵀ → (R, 5·H)   (R = B·N·N).
3️⃣ Triton kernel splits the fused result, applies the optional pair‑wise mask,
   evaluates sigmoid gates and writes three FP16 tensors:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
5️⃣ A single Triton kernel fuses the second LayerNorm, the element‑wise out‑gate
   scaling and the final projection back to `dim`.  No intermediate tensors are
   materialised for this step.
6️⃣ Return the result in FP32.

The heavy kernels are written in Triton, while the two matrix‑multiplications are
delegated to cuBLAS (torch.bmm / torch.nn.functional.linear), which already
uses Tensor‑Cores on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  *[rows, dim]   (fp32)
    y_ptr,          # mutable* [rows, dim]  (fp16)
    weight_ptr,     # const  *[dim]        (fp32) γ
    bias_ptr,       # const  *[dim]        (fp32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ mean / variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized values: γ·(x‑μ)·rstd + β  (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ  (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β  (fp32)
        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split / mask / sigmoid‑gate kernel
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const  *[rows, 5*H]    (fp16)
    left_ptr,          # mutable* [rows, H]    (fp16)
    right_ptr,         # mutable* [rows, H]    (fp16)
    out_gate_ptr,      # mutable* [rows, H]    (fp16)
    mask_ptr,          # const  *[rows]       (fp16) – ignored if HAS_MASK==0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r ∈ [0, rows):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are performed in FP16 (sigmoid evaluated in FP32).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over hidden dim
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # address offset for the concatenated 5·H matrix
        row_off = offs_r[:, None] * (5 * H)                       # [R,1]
        col_off = offs_h[None, :]                                 # [1,H]

        # left projection
        left_raw = tl.load(proj_ptr + row_off + col_off,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)

        # right projection
        right_raw = tl.load(proj_ptr + row_off + H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)

        # left gate
        left_gate_raw = tl.load(proj_ptr + row_off + 2 * H + col_off,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)

        # right gate
        right_gate_raw = tl.load(proj_ptr + row_off + 3 * H + col_off,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # out gate
        out_gate_raw = tl.load(proj_ptr + row_off + 4 * H + col_off,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # sigmoid evaluated in fp32 for stability
        l_g = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_g = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_g = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask + gates
        left  = left_raw  * m * l_g
        right = right_raw * m * r_g

        # store
        tl.store(left_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 o_g,   mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out‑gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_ln_proj_kernel(
    inp_ptr,          # const  *[rows, H]     (fp16)   – output of left·rightᵀ
    gate_ptr,         # const  *[rows, H]     (fp16)   – out_gate
    norm_w_ptr,       # const  *[H]          (fp32)   – to_out_norm.weight
    norm_b_ptr,       # const  *[H]          (fp32)   – to_out_norm.bias
    proj_w_ptr,       # const  *[dim, H]     (fp16)   – to_out.weight
    out_ptr,          # mutable* [rows, dim] (fp16)
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    For each row r:
        1) Compute mean / variance of inp[r, :]  (FP32)
        2) Normalize, apply LN gain/bias (γ,β) (FP32)
        3) Multiply by out_gate[r, :] (FP32)
        4) Project to `dim` using proj_w (FP16) → final output (FP16)
    The kernel fuses the three steps, eliminating an intermediate buffer.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # 1) mean / variance over the hidden dimension H
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H
        vals = tl.load(inp_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) accumulate projection (fp32 accumulator)
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # raw value
        raw = tl.load(inp_ptr + offs_r[:, None] * H + offs_h[None, :],
                      mask=mask_r[:, None] & mask_h[None, :],
                      other=0.0).to(tl.float32)

        # layer‑norm: (x‑μ)·rstd
        norm = (raw - mean[:, None]) * rstd[:, None]       # fp32

        # LN weight & bias (γ,β)
        w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
        b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
        norm = norm * w_norm[None, :] + b_norm[None, :]                # fp32

        # out‑gate
        gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)                      # fp32
        scaled = norm * gate                                            # fp32

        # ------------------------------------------------------------------
        # 3) dot with projection weight for each output dim block
        #    weight layout: [dim, H] → we load a transposed tile (H, BLOCK_D)
        # ------------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
            mask_d = offs_d < dim

            # weight_T[h, d] = weight[d, h]
            w_T = tl.load(proj_w_ptr + offs_h[:, None] + offs_d[None, :] * H,
                          mask=mask_h[:, None] & mask_d[None, :],
                          other=0.0).to(tl.float32)                     # [BLOCK_H, BLOCK_D]

            # accumulate: (BLOCK_R, BLOCK_H) @ (BLOCK_H, BLOCK_D) → (BLOCK_R, BLOCK_D)
            acc += tl.dot(scaled, w_T)

    # ------------------------------------------------------------------
    # 4) store FP16 result
    # ------------------------------------------------------------------
    out_fp16 = acc.to(tl.float16)
    tl.store(out_ptr + offs_r[:, None] * dim + offs_d[None, :],
             out_fp16,
             mask=mask_r[:, None] & mask_d[None, :])

# Custom forward kernel – outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (x, mask, weights, cfg)
            x      : torch.Tensor   [B, N, N, dim]      dtype float32
            mask   : torch.Tensor   [B, N, N]          (bool/float) or None
            weights: dict of all linear / layer‑norm parameters (float32)
            cfg    : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free fp32 input

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·H outputs) – single GEMM
    # --------------------------------------------------------------
    # concatenate the five weight matrices (shape: [5*H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5*H, dim)

    proj = F.linear(x_norm, w_fused)   # (R, 5*H), bias=None
    del w_fused, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)  # flatten to [R]
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dim: left @ rightᵀ   (fp16)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] → permute → (B*H, N, N)
    left  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_mat  = left .view(B * H, N, N)   # [B*H, N, N]
    right_mat = right.view(B * H, N, N)   # [B*H, N, N]

    # out_mat[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N), fp16
    del left, right, left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate scaling and final projection
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] and flatten rows
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(R, H).contiguous()                               # [R, H]

    # allocate final output (fp16)  – will be cast back to fp32 later
    final_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_D = 32                               # output‑dim tile
    grid_final = (triton.cdiv(R, BLOCK_R),)
    final_fused_ln_proj_kernel[grid_final](
        out_flat,
        out_gate.view(R, H),
        weights["to_out_norm.weight"],   # γ (fp32)   size H
        weights["to_out_norm.bias"],     # β (fp32)
        weights["to_out.weight"].to(torch.float16),   # [dim, H] (fp16)
        final_fp16,
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 6️⃣ Cast to FP32 and reshape to the original 4‑D layout
    # --------------------------------------------------------------
    final_fp32 = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final_fp32
```
