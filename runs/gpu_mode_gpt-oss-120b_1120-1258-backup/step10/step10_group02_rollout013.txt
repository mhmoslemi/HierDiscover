"""
Outgoing Tri‑Multiplicative Update (TriMul) – fused LayerNorm + projection.
---------------------------------------------------------
1️⃣   Fuse the first LayerNorm (γ,β) with the five linear maps
     (left_proj, right_proj, left_gate, right_gate, out_gate) into
     a single Triton kernel:
        out = ((x-μ)·rstd) @ (γ·Wᵀ) + β·Wᵀ
     where W is the concatenation of the five weight matrices.
     This removes the intermediate fp32→fp16 copy and reduces the
     amount of weight traffic (the weight matrix is only ~0.5 MiB).

2️⃣   Split the fused result, apply the optional pair‑wise mask,
     evaluate the three σ‑gates and write three tensors:
        left  = proj_left   * mask * σ(proj_left_gate)
        right = proj_right  * mask * σ(proj_right_gate)
        out_gate = σ(proj_out_gate)

3️⃣   Batched half‑precision GEMM:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ      (H = hidden_dim)

4️⃣   Fuse the second LayerNorm with the element‑wise multiplication
     by the output gate (already in FP16).

5️⃣   Final linear projection back to `dim` (FP16 → FP32).

All heavy matrix multiplications use cuBLAS Tensor‑Core kernels;
the two Triton kernels handle the row‑wise normalisation and
the split‑gate logic, which are memory‑bound and benefit from the
tight fusions above.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Fuse first LayerNorm with the five linear projections
# ----------------------------------------------------------------------
@triton.jit
def ln_proj_kernel(
    x_ptr,          # const  *[rows, dim]   (fp32)
    out_ptr,        # mut    *[rows, out]   (fp16)
    w_ptr,          # const  *[dim, out]   (fp16)  = (γ·W)ᵀ
    b_ptr,          # const  *[out]        (fp16)  = β·Wᵀ
    rows: tl.int32,
    dim: tl.int32,
    out_feat: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,   # number of rows per program
    BLOCK_D: tl.constexpr,   # tile over the inner dimension
    BLOCK_O: tl.constexpr,   # tile over the output features
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1) Compute per‑row mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2) Compute projection:  ((x-μ)·rstd) @ (γ·Wᵀ) + β·Wᵀ
    #    Loop over output‑feature tiles.
    # --------------------------------------------------------------
    for o_off in range(0, tl.cdiv(out_feat, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)   # [BLOCK_O]
        mask_o = offs_o < out_feat

        # accumulator for a (BLOCK_R × BLOCK_O) tile
        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # -------- load x and normalise ----------
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)
            x_norm = (x - mean[:, None]) * rstd[:, None]      # fp32
            x_norm_f16 = x_norm.to(tl.float16)                # fp16 for dot

            # -------- load weight tile (dim → out) ----
            # w_ptr is (dim, out_feat) row‑major → stride = out_feat
            w = tl.load(w_ptr + offs_d[:, None] * out_feat + offs_o[None, :],
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float16)

            # -------- accumulate dot product ----------
            acc += tl.dot(x_norm_f16, w)   # (R,D)·(D,O) → (R,O)   fp32 accumulator

        # -------- add bias (β·Wᵀ) -----------------
        b = tl.load(b_ptr + offs_o, mask=mask_o, other=0.0).to(tl.float32)
        acc += b[None, :]               # broadcast over rows

        # -------- store ---------------------------------
        out_tile_ptr = out_ptr + offs_r[:, None] * out_feat + offs_o[None, :]
        tl.store(out_tile_ptr, acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_o[None, :])

# ----------------------------------------------------------------------
# 2️⃣  Split fused projection, apply mask, compute σ‑gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, O]      (fp16)
    left_ptr,          # mutable *[rows, H]      (fp16)
    right_ptr,         # mutable *[rows, H]      (fp16)
    out_gate_ptr,      # mutable *[rows, H]      (fp16)
    mask_ptr,          # const   *[rows]         (fp16)  (ignored if HAS_MASK==0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        row_off = offs_r[:, None] * (5 * H)                       # [R,1]
        col_off = offs_h[None, :]                                 # [1,H]

        # left projection
        left_raw = tl.load(proj_ptr + row_off + col_off,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)
        # right projection
        right_raw = tl.load(proj_ptr + row_off + H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        # left gate
        left_g_raw = tl.load(proj_ptr + row_off + 2 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        # right gate
        right_g_raw = tl.load(proj_ptr + row_off + 3 * H + col_off,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        # out gate
        out_g_raw = tl.load(proj_ptr + row_off + 4 * H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)

        # sigmoid (computed in fp32 for stability)
        left_gate = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate  = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣  Second LayerNorm + element‑wise multiplication by out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)   (out_gate)
    y_ptr,            # mutable *[rows, dim]   (fp16)   (result)
    weight_ptr,       # const   *[dim]        (fp32)   γ
    bias_ptr,         # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # normalise, apply γ/β, multiply by gate, store (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        norm = (x_val - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        norm = norm * w[None, :] + b[None, :]                     # fp32

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32

        out = norm * gate                                        # fp32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict of model parameters (all float32, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ---------------------- unpack ----------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows = B·N·N
    H = hidden_dim
    O = 5 * H                         # concatenated output dim of the fused projection

    # --------------------------------------------------------------
    # 1️⃣ Fuse LayerNorm + 5 linear maps (first projection)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (shape: [O, dim])
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [O, dim]

    # Scale by LayerNorm γ (norm.weight) and pre‑compute bias β·Wᵀ
    norm_weight = weights["norm.weight"]               # γ, shape [dim]
    norm_bias   = weights["norm.bias"]                 # β, shape [dim]

    # γ·W   (broadcast γ over rows)
    proj_weight_scaled = proj_weight * norm_weight.unsqueeze(0)   # [O, dim]
    # bias term β·Wᵀ  → shape [O]
    bias_proj_fp32 = torch.mv(proj_weight.float(), norm_bias.float())
    bias_proj = bias_proj_fp32.to(torch.float16)

    # Transpose for the kernel: (dim, O) and make contiguous
    proj_weight_T = proj_weight_scaled.t().contiguous()   # [dim, O]

    # Flatten input to [R, dim] (fp32)
    x_flat = x.view(R, dim).contiguous()

    # Allocate fused projection output
    proj = torch.empty(R, O, dtype=torch.float16, device=device)

    # launch kernel
    BLOCK_R = 128
    BLOCK_D = 32
    BLOCK_O = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    ln_proj_kernel[grid](
        x_flat,
        proj,
        proj_weight_T,
        bias_proj,
        R,
        dim,
        O,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )
    del x_flat, proj_weight, proj_weight_scaled, proj_weight_T, bias_proj

    # --------------------------------------------------------------
    # 2️⃣ Split, apply mask, sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        # mask is broadcasted over the hidden dimension
        mask_half = mask.to(torch.float16).reshape(R)   # [R]
        mask_ptr = mask_half

    BLOCK_R_SPLIT = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM: left @ rightᵀ   (fp16)
    # --------------------------------------------------------------
    # reshape to [B, H, N, N] for batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B*H, N, N]
    right_mat = right.view(B * H, N, N)  # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N), fp16
    del left, right, left_mat, right_mat

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm + multiplication by out_gate (fused)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2 = 256
    BLOCK_D2 = 32
    grid_ln2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid_ln2](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    # to_out.weight shape: [dim, hidden_dim]
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
