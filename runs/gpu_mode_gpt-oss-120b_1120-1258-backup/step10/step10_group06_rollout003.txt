"""
TriMul – outgoing version (AlphaFold‑3)

Algorithm (forward only)
-----------------------
1️⃣ Row‑wise LayerNorm of the input tensor (B·N·N × dim).  We use
   torch.nn.functional.layer_norm (highly tuned cuDNN kernel).

2️⃣ Fuse the five linear maps
       left_proj, right_proj,
       left_gate, right_gate, out_gate
   into a single GEMM:
       (R, dim) @ (5·H, dim).T → (R, 5·H)
   where  R = B·N·N  and  H = hidden_dim.
   The result is kept in fp16.

3️⃣ Split the fused result, apply an optional pair‑wise mask and the
   sigmoid gates.  All of this is done with regular PyTorch ops
   (vectorised, fp32‑sigmoid → fp16) – the kernel launch overhead is
   therefore avoided.

4️⃣ Reshape ``left`` and ``right`` to [B·H, N, N] and perform a batched
   matrix‑multiply (torch.bmm) → the “tri‑multiplicative update”.

5️⃣ Second LayerNorm + element‑wise multiply with ``out_gate``.
   This is the only custom Triton kernel that remains (``layernorm_mul``).

6️⃣ Final linear projection back to the original dimension (fp16 → fp32).

Only the second LayerNorm is written in Triton; the rest relies on
cuBLAS / cuDNN which already exploit Tensor‑Cores on an H100.
The implementation supports optional masks and works for any combination
of B, N, dim and hidden_dim required by the test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)   (out_gate)
    y_ptr,            # mutable *[rows, dim]   (float16)   (result)
    weight_ptr,       # const   *[dim]        (float32)   γ
    bias_ptr,         # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm (fp32) on the hidden tensor ``x`` and multiply the
    normalized result by ``gate`` (the out_gate).  The final value is stored
    as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)           # Σ x
        sum_sq  += tl.sum(vals * vals, axis=1)    # Σ x²
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalize, apply γ/β, multiply by gate, store (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]   # fp32

        # γ and β (float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        norm = norm * w[None, :] + b[None, :]   # fp32

        # gate (out_gate) – fp16 → fp32
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)

        out = norm * gate_vals                 # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------

    """

    """
    # --------------------------------------------------------------

    # --------------------------------------------------------------

    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                      # still float32
    x_norm = x_norm_fp32.half()            # fp16 for the following GEMM
    del x_flat, x_norm_fp32

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * H outputs)
    # --------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]

    # (R, dim) @ (5*H, dim).T → (R, 5*H)   fp16
    proj = F.linear(x_norm, w_fused)        # [R, 5*H]
    del w_fused, x_norm                     # free memory

    # reshape to split the five streams
    proj = proj.view(R, 5, H)               # [R, 5, H]

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask (if any) and evaluate sigmoid gates (torch)
    # --------------------------------------------------------------
    left_proj  = proj[:, 0, :]               # (R, H) fp16
    right_proj = proj[:, 1, :]               # (R, H) fp16
    left_gate  = torch.sigmoid(proj[:, 2, :].float()).half()
    right_gate = torch.sigmoid(proj[:, 3, :].float()).half()
    out_gate   = torch.sigmoid(proj[:, 4, :].float()).half()
    del proj

    if not nomask and mask is not None:
        # mask shape [B, N, N] → [R]  (0/1)
        mask_half = mask.to(torch.float16).reshape(R, 1)   # broadcast over H
        left_proj  = left_proj * mask_half
        right_proj = right_proj * mask_half
        # gates are *not* masked – they stay unchanged
    # apply the element‑wise gates
    left  = left_proj * left_gate
    right = right_proj * right_gate
    del left_proj, right_proj, left_gate, right_gate

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into the batch axis
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)   # [B·H, N, N]

    # out_mat[b·H] = left_mat[b·H] @ right_mat[b·H]^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # back to [B, N, N, H]  (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left, right, left_mat, right_mat, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # --------------------------------------------------------------
    out_flat   = out.view(R, H).contiguous()           # [R, H] fp16
    out_gate_f = out_gate.view(R, H).contiguous()     # [R, H] fp16

    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256        # rows per program (tune if needed)
    BLOCK_D = 32         # hidden dimension per iteration
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_mul_kernel[grid](
        out_flat,
        out_gate_f,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)  shape [H]
        weights["to_out_norm.bias"],     # β (float32)  shape [H]
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate_f, out

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to ``dim`` (fp16 → fp32)
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).float()   # fp32 output
    return final
