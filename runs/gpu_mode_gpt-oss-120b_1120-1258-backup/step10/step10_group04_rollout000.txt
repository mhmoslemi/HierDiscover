"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass
================================================================

The module implements the *outgoing* TriMul operator described in the
AlphaFold3 paper.  The forward pass consists of:

1️⃣  Row‑wise LayerNorm over the last dimension (FP32 → FP16).
2️⃣  A *fused* linear projection that computes the two value tensors
    (left/right) together with their three sigmoidal gates.
3️⃣  A Triton kernel that splits the fused tensor, applies the optional
    pairwise mask, and applies the sigmoid gates.
4️⃣  Batched GEMM (`torch.bmm`) for the true tri‑multiplicative update:
    `(left) @ (right)ᵀ`      – shape [B·H, N, N].
5️⃣  Row‑wise LayerNorm + out‑gate multiplication (fused in a Triton
    kernel).
6️⃣  Final linear projection back to the original `dim` (FP16 GEMM)
    and cast to FP32.

Only the two LayerNorm kernels and the split‑gate kernel are written in
Triton; the heavy matrix‑multiplications use cuBLAS via PyTorch which
already exploits Tensor‑Cores on the H100.

The implementation works for any configuration in the test matrix,
supports an optional binary mask, and returns a float32 tensor of shape
`[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # Compute mean & variance (fp32 accumulator)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Write normalized values:  y = γ·(x‑μ)·rstd + β   (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]          # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# 3️⃣ Split fused projection, apply optional pair‑wise mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)

    H: tl.int32,

    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):

    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]

    # --------------------------------------------------------------

        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H]

        right_raw_ptr   = proj_ptr + base + H

        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask & gates ----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ---- store results ----
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# 5️⃣ Row‑wise LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(

    y_ptr,          # mutable *[rows, dim]   (float16)

    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # Compute mean & variance in fp32
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, apply LN weight/bias and multiply by gate
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x and gate
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],

        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],

        norm = (x_vals - mean[:, None]) * rstd[:, None]       # fp32

        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals       # fp32
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])
