"""
Tri‑Multiplicative (outgoing) update – fast implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (torch.nn.functional.layer_norm – highly
   tuned CUDA kernel).  Cast the normalized tensor to fp16.

2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   half‑precision GEMM:
        X_norm @ W_fusedᵀ → (R, 5·H)    with R = B·N·N, H = hidden_dim.

3️⃣ Triton kernel `split_fuse_gate_kernel`:
        * splits the fused result into left, right and out_gate tensors,
        * applies the optional pair‑wise mask,
        * evaluates three sigmoid gates (fp32 for stability).

4️⃣ Batched matrix‑multiplication over the hidden dimension:
        left  = left .reshape(B, H, N, N)
        right = right.reshape(B, H, N, N)
        out   = bmm(left, rightᵀ)       # B·H independent GEMMs (Tensor‑Core)

5️⃣ Row‑wise LayerNorm on the result, multiply by `out_gate`
   (torch.nn.functional.layer_norm).

6️⃣ Final linear projection back to `dim` (half‑precision GEMM) and cast to
   float32 for the returned tensor.

Only the split‑/gate‑kernel is written in Triton, satisfying the “must use a
kernel” requirement while delegating the heavy linear algebra to cuBLAS /
torch‑native kernels for maximum performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left      = (proj_left  * mask) * sigmoid(proj_left_gate)
        right     = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate  = sigmoid(proj_out_gate)

    All arithmetic is performed in fp16; sigmoid is evaluated in fp32 for
    numerical stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load the (optional) mask and broadcast to the hidden dimension.
    # ------------------------------------------------------------------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in chunks.
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # layout in proj: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # offsets into proj

        # pointers to the five slices
        left_raw_ptr     = proj_ptr + base
        right_raw_ptr    = proj_ptr + base + H
        left_gate_ptr    = proj_ptr + base + 2 * H
        right_gate_ptr   = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        # load raw values
        left_raw   = tl.load(left_raw_ptr,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_g_raw = tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32, cast back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # store results
        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Main forward function
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all Linear/LayerNorm parameters
                         (float32 tensors; all Linear layers have bias=False)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    nomask     = cfg.get("nomask", True)

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                # total number of (b,i,j) triples
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch‑native, then cast to fp16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left, right, three gates) – one GEMM
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape [5*H, dim]) and cast to fp16.
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # Linear: (R, dim) @ (5*H, dim)ᵀ → (R, 5*H)
    proj = F.linear(x_norm.view(R, dim), w_fused)   # fp16
    del x_norm, w_fused

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask:
        # No mask – compile‑time flag disables mask loading.
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_ptr = mask.to(torch.float16).reshape(R)   # [R]

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # Reshape left/right to [B, H, N, N] for the batched bmm.
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # bmm: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back‑to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm + gate multiplication (torch‑native)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                          # fp16

    # Multiply by out_gate (broadcasted)
    out_gate_reshaped = out_gate.view(B, N, N, H)   # [B, N, N, H]
    out_norm = out_norm * out_gate_reshaped
    del out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]   fp16
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
