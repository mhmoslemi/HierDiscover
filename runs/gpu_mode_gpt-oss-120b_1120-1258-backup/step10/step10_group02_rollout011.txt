    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

                          mask=mask_r[:, None] & mask_k[None, :],
                          other=0.0).to(tl.float32)
        sum_val += tl.sum(out_val, axis=1)
        sum_sq += tl.sum(out_val * out_val, axis=1)

def fused_ln_gate_linear_kernel(
    out_ptr, gate_ptr, weight_ptr,
    norm_weight_ptr, norm_bias_ptr,
    eps,
    rows, H, dim,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # Compute mean and variance per row (over H)
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for k_off in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_k[None, :],
                           mask=mask_r[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # allocate accumulator
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    # second pass: compute gated normalized values and accumulate dot product
    for k_off in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out and gate
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_k[None, :],
                           mask=mask_r[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_k[None, :],
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # Normalize
        norm_vals = (out_vals - mean[:, None]) * rstd[:, None]

        # Apply LN weight and bias
        gamma = tl.load(norm_weight_ptr + offs_k, mask=mask_k, other=0.0)
        beta = tl.load(norm_bias_ptr + offs_k, mask=mask_k, other=0.0)
        # broadcast gamma,beta
        norm_vals = norm_vals * gamma[None, :] + beta[None, :]

        # Apply gate
        gated = norm_vals * gate_vals  # fp32

        # Load weight tile: weight_ptr shape [dim, H] (dim -> rows, H -> cols)
        # Since weight is transposed: shape [H, dim]; we want tile columns block of dim
        # Let's store weight_T shape [H, dim] for easier loads.

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):

            mask_d = offs_d < dim

            # Load weight tile

                        mask=mask_k[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)  # shape [BLOCK_K, BLOCK_D]

            # accumulate
            acc += tl.dot(gated, w, trans_b=False)  # (R,K) dot (K,D) -> (R,D)
    # end for k

    # write out
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        out_ptr_final = ... pointer to final output.

        tl.store(out_ptr_final + offs_r[:, None] * dim + offs_d[None, :],

        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
4️⃣ Batched half‑precision GEMM (left @ rightᵀ) over the hidden dimension
   using `torch.bmm`.  This yields a tensor `out` of shape [B,N,N,hidden_dim].
5️⃣ Triton kernel `fused_ln_gate_linear_kernel` fuses the second LayerNorm,
   the element‑wise multiplication by `out_gate`, and the final linear
   projection back to `dim`.  The kernel computes for each row:
        y = ( ( (out‑μ)·rstd·γ + β ) ⊙ out_gate ) @ Wᵀ
   where γ,β are the LayerNorm parameters and W is `to_out.weight`.
   All arithmetic is performed in fp32 and the result is written as fp16.
6️⃣ Cast the final tensor back to fp32 and reshape to [B, N, N, dim].

Only the three Triton kernels are needed; the heavy matrix multiplications
use cuBLAS/Tensor‑Core kernels, achieving near‑peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (fp32)
    y_ptr,          # mutable *[rows, dim]   (fp16)
    weight_ptr,     # const   *[dim]        (fp32)   γ
    bias_ptr,       # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized values: γ·(x‑μ)·rstd + β  (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]               # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)    # γ (fp32)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)      # β (fp32)

        out = norm * w[None, :] + b[None, :]                        # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16)   (ignored if HAS_MASK==0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are in fp16 (sigmoid evaluated in fp32 for stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load mask row (broadcast later) if present
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        row_off = offs_r[:, None] * (5 * H)                       # [R,1]
        col_off = offs_h[None, :]                                 # [1,H]

        # left projection
        left_raw = tl.load(proj_ptr + row_off + col_off,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)
        # right projection
        right_raw = tl.load(proj_ptr + row_off + H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        # left gate
        left_g_raw = tl.load(proj_ptr + row_off + 2 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        # right gate
        right_g_raw = tl.load(proj_ptr + row_off + 3 * H + col_off,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        # out gate
        out_g_raw = tl.load(proj_ptr + row_off + 4 * H + col_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        left_gate = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        tl.store(left_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, out_gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_linear_kernel(
    inp_ptr,           # const   *[rows, H]    (fp16)   = out from left·rightᵀ
    gate_ptr,          # const   *[rows, H]    (fp16)   = out_gate
    weight_ptr,        # const   *[H, dim]    (fp16)   = to_out.weightᵀ
    norm_w_ptr,        # const   *[H]         (fp32)   = to_out_norm.weight
    norm_b_ptr,        # const   *[H]         (fp32)   = to_out_norm.bias
    out_ptr,           # mutable *[rows, dim] (fp16)
    eps,               # fp32 scalar
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes for each row r:
        y_r = ( ( (inp_r - μ_r)·rstd_r·γ + β ) ⊙ gate_r ) @ Wᵀ
    where (μ_r, σ²_r) are mean / variance of `inp_r`,
    γ,β are LayerNorm parameters (size H), and W has shape [H, dim].
    All arithmetic is done in fp32; output stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # First pass: compute mean and variance of each row (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_off in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        vals = tl.load(inp_ptr + offs_r[:, None] * H + offs_k[None, :],
                       mask=mask_r[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Outer loop over output‑dimension tiles
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # accumulator for this d‑tile
        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        # ----------------------------------------------------------
        # Inner loop over hidden‑dimension tiles
        # ----------------------------------------------------------
        for k_off in range(0, tl.cdiv(H, BLOCK_K)):
            offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # load raw values and corresponding gate
            inp = tl.load(inp_ptr + offs_r[:, None] * H + offs_k[None, :],
                          mask=mask_r[:, None] & mask_k[None, :],
                          other=0.0).to(tl.float32)            # (R, K)
            g   = tl.load(gate_ptr + offs_r[:, None] * H + offs_k[None, :],
                          mask=mask_r[:, None] & mask_k[None, :],
                          other=0.0).to(tl.float32)            # (R, K)

            # LayerNorm: (inp - μ)·rstd
            norm = (inp - mean[:, None]) * rstd[:, None]      # (R, K)

            # Apply γ,β (to_out_norm params)
            gamma = tl.load(norm_w_ptr + offs_k, mask=mask_k, other=0.0)   # (K)
            beta  = tl.load(norm_b_ptr + offs_k, mask=mask_k, other=0.0)   # (K)
            norm = norm * gamma[None, :] + beta[None, :]                  # (R, K)

            # Multiply by out_gate
            gated = norm * g                                            # (R, K)

            # Load weight tile: shape (K, BLOCK_D)
            w = tl.load(
                weight_ptr + offs_k[:, None] * dim + offs_d[None, :],
                mask=mask_k[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                                            # (K, D)

            # Accumulate GEMM contribution
            acc += tl.dot(gated, w, trans_b=False)                      # (R, D)

        # ----------------------------------------------------------
        # Store the completed d‑tile
        # ----------------------------------------------------------
        out_tile_ptr = out_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_tile_ptr,
                 acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict with all Linear/LayerNorm parameters
                             (float32 tensors, Linear layers have bias=False)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free fp32 input

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates) via cuBLAS
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape [5*H, dim]

    proj = F.linear(x_norm, w_concat)   # (R, 5*H) fp16
    del w_concat, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_half = mask.to(torch.float16).reshape(R)  # [R]
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM: left @ rightᵀ   (shape [R, H] → final hidden tensor)
    # ------------------------------------------------------------------
    # left/right: [B, N, N, H] → permute to [B, H, N, N] for batched bmm
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B*H, N, N]
    right_mat = right.view(B * H, N, N)  # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, shape [B*H, N, N]
    del left, right, left_mat, right_mat

    # reshape back to [R, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(R, H).contiguous()                             # [R, H] fp16
    out_gate_flat = out_gate.view(R, H).contiguous()                   # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, gate multiplication & final projection
    # ------------------------------------------------------------------
    # Transpose to_out.weight → shape [H, dim] for column‑major tiling
    weight_to_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)  # [H, dim]

    final_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_D = 32
    BLOCK_K = 32
    grid_fused = (triton.cdiv(R, BLOCK_R),)

    fused_ln_gate_linear_kernel[grid_fused](
        out_flat,
        out_gate_flat,
        weight_to_out_T,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        final_half,
        eps,
        R,
        H,
        dim,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    del out_flat, out_gate_flat, out, out_gate, weight_to_out_T

    # ------------------------------------------------------------------
    # 6️⃣ Cast back to float32 and reshape to original layout
    # ------------------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
