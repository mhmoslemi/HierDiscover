"""
High‑performance Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – done with
   `torch.nn.functional.layer_norm` (cuDNN) because it is faster than a
   custom Triton kernel for the large number of rows.

2️⃣ Fuse the five Linear maps (left/right projections + three gates) into a
   single half‑precision GEMM:
        (R, dim) @ (5·H, dim).T → (R, 5·H)

3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused result, applies the
   optional pair‑wise mask, evaluates the three sigmoid gates and writes three
   tensors:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

4️⃣ Batched half‑precision GEMM (torch.bmm) over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ

5️⃣ Triton kernel `layernorm_mul_kernel` performs a second LayerNorm on the
   intermediate result and multiplies it element‑wise by the `out_gate`.

6️⃣ Final linear projection back to `dim` (half‑precision → float32).

Only the mask‑splitting and the second LayerNorm are written in Triton,
satisfying the “must‑use‑a‑kernel” requirement while keeping the
implementation fast on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < rows

    # load per‑row mask (if present)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [left, right, left_gate, right_gate, out_gate] each size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # offset into proj

        p_left       = proj_ptr + base
        p_right      = proj_ptr + base + H
        p_left_gate  = proj_ptr + base + 2 * H
        p_right_gate = proj_ptr + base + 3 * H
        p_out_gate   = proj_ptr + base + 4 * H

        left_raw       = tl.load(p_left,       mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw      = tl.load(p_right,      mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw  = tl.load(p_left_gate,  mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw = tl.load(p_right_gate, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw   = tl.load(p_out_gate,   mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst_left    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        dst_right   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        dst_outgate = out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(dst_left,    left,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_right,   right,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_outgate, out_gate,mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Triton kernel: second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)  (out_gate)
    y_ptr,            # mutable *[rows, dim]   (float16)  (result)
    weight_ptr,       # const   *[dim]        (float32)  γ
    bias_ptr,         # const   *[dim]        (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ normalize, apply γ/β, multiply with gate, store (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        norm = norm * w[None, :] + b[None, :]                     # fp32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)            # fp32

        out = norm * gate_vals                                   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use cuDNN for speed)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] fp32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)               # half‑precision for next GEMM
    del x_flat, x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + three gates → (R, 5*H)
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5*H, dim]

    proj = F.linear(x_norm, w_concat)                    # fp16, shape [R, 5*H]
    del w_concat, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split fused result, apply mask, evaluate sigmoid gates
    # ------------------------------------------------------------------
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_ptr = mask.to(torch.float16).reshape(R)   # flatten to [R]

    BLOCK_R = 256          # rows per program (tune for occupancy)
    BLOCK_H = 32           # hidden dim per iteration
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into the batch axis
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # GEMM: left @ rightᵀ → [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_flat   = out.view(R, H).contiguous()
    out_fused  = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program
    BLOCK_D = 128          # dim per iteration (covers dim ≤ 384 in ≤3 loops)
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32, [H])
        weights["to_out_norm.bias"],     # β (float32, [H])
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
