"""
TriMul (outgoing) – fused Triton implementation

Algorithm
---------
1️⃣ Row‑wise LayerNorm + fused linear projection (5·H outputs) in a single
   Triton kernel – eliminates the intermediate 𝑥_norm tensor.
2️⃣ Split the fused projection, apply the pair‑wise mask, evaluate three
   sigmoid gates (left/right/out) – Triton.
3️⃣ Batched half‑precision GEMM:  left ⟂ rightᵀ  (B·H independent matmuls).
   This is performed with torch.bmm (cuBLAS/Tensor‑Cores) – the dominant
   N³ work.
4️⃣ Second LayerNorm, gate multiplication and final linear projection back
   to `dim` are fused in a single Triton kernel, removing two extra kernels
   and one memory copy.
5️⃣ Cast the final result back to float32.

All kernels are written for Triton 3.3.1 and target an H100.  The code works
for any configuration in the test‑matrix and respects the optional mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm + fused projection
# ----------------------------------------------------------------------
@triton.jit
def norm_proj_kernel(
    x_ptr,                # const   *[rows, dim]      (float32)
    out_ptr,              # mutable *[rows, out_feat] (float16)
    w_ptr,                # const   *[dim, out_feat] (float16) – weightᵀ
    rows: tl.int32,
    dim: tl.int32,
    out_feat: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim   per tile
    BLOCK_O: tl.constexpr,   # out‑features per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: mean & variance (fp32) -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)            # [BLOCK_D]
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                       # [M,D]
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: projection (fp16) ----------------------------
    n_out_blocks = tl.cdiv(out_feat, BLOCK_O)

    for o_off in range(0, n_out_blocks):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)            # [BLOCK_O]
        mask_o = offs_o < out_feat
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
            mask_d = offs_d < dim

            # load raw input and normalize
            x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                 # [M,D]
            norm = (x - mean[:, None]) * rstd[:, None]            # fp32

            # weight block: shape [BLOCK_D, BLOCK_O]
            w = tl.load(w_ptr + offs_d[:, None] * out_feat + offs_o[None, :],
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)                 # fp32

            acc += tl.dot(norm, w)                                 # [M,O]

        # store the block
        out = acc.to(tl.float16)
        out_ptrs = out_ptr + offs_m[:, None] * out_feat + offs_o[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)              # [BLOCK_R]
    mask_r = offs_r < rows

    # mask scalar per row (broadcast later)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout per row: [L, R, Lg, Rg, Og] each H long
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]       # offset into proj

        # pointers to the 5 segments
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid (fp32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Fused second LayerNorm, gate multiplication and final linear
# ----------------------------------------------------------------------
@triton.jit
def final_norm_linear_kernel(
    x_ptr,                # const *[rows, dim]      (float16) – output of bmm
    gate_ptr,             # const *[rows, dim]      (float16) – out_gate
    y_ptr,                # mutable *[rows, out_dim] (float16) – final result
    norm_w_ptr,           # const *[dim]            (float32) – to_out_norm.weight
    norm_b_ptr,           # const *[dim]            (float32) – to_out_norm.bias
    w_out_ptr,            # const *[dim, out_dim]   (float16) – to_out.weightᵀ
    rows: tl.int32,
    dim: tl.int32,
    out_dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # hidden dim per tile
    BLOCK_O: tl.constexpr,   # output dim per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- compute mean & variance over the hidden dimension ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- loop over output‑dim blocks --------------------------------
    n_out_blocks = tl.cdiv(out_dim, BLOCK_O)

    for o_off in range(0, n_out_blocks):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)          # [BLOCK_O]
        mask_o = offs_o < out_dim
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
            mask_d = offs_d < dim

            # load raw hidden vector
            x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

            # ----- LayerNorm (γ/β) ------------------------------------
            gamma = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)   # float32
            beta  = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)   # float32
            x_norm = (x - mean[:, None]) * rstd[:, None]                # fp32
            x_norm = x_norm * gamma[None, :] + beta[None, :]             # fp32

            # ----- Gate multiplication ---------------------------------
            g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)
            gated = x_norm * g                                          # fp32

            # ----- Linear projection ------------------------------------
            w = tl.load(w_out_ptr + offs_d[:, None] * out_dim + offs_o[None, :],
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)                       # fp32
            acc += tl.dot(gated, w)                                     # [M,O]

        # store final block
        out_ptr = y_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
        tl.store(out_ptr, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])

# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, [B, N, N, dim]  (float32)
        * mask_tensor  : torch.Tensor, [B, N, N]    (bool/float) or None
        * weights_dict : dict with keys
              - "norm.weight", "norm.bias"
              - "left_proj.weight", "right_proj.weight"
              - "left_gate.weight", "right_gate.weight", "out_gate.weight"
              - "to_out_norm.weight", "to_out_norm.bias"
              - "to_out.weight"
        * config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # flattened rows for pair‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ fused LayerNorm + projection (produces 5·H features)
    # --------------------------------------------------------------
    # concatenate the five weight matrices (no bias terms)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    # we need the transposed version (dim, 5·H) for the tile‑wise matmul
    w_fused_T = w_fused.t().contiguous()                # [dim, 5·H]

    # flatten input and allocate output buffer for the fused projection
    x_flat = x.view(R, dim).contiguous()                # [R, dim] float32
    proj = torch.empty(R, 5 * H, dtype=torch.float16, device=device)

    BLOCK_M = 128      # rows per program
    BLOCK_D = 32       # dim tile
    BLOCK_O = 32       # output‑feature tile
    grid_proj = (triton.cdiv(R, BLOCK_M),)

    norm_proj_kernel[grid_proj](
        x_flat,
        proj,
        w_fused_T,
        R,
        dim,
        5 * H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )
    del x_flat, w_fused, w_fused_T

    # --------------------------------------------------------------
    # 2️⃣ split, apply mask, sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
    else:
        # mask is broadcastable to (B,N,N); flatten to rows
        mask_flat = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_flat
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 3️⃣ batched GEMM over the hidden dimension (half‑precision)
    # --------------------------------------------------------------
    # reshape to [B, H, N, N] for the bmm
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to flat shape (R, H) – still fp16
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).reshape(R, H).contiguous()
    del left, right, left_mat, right_mat, out_mat

    # --------------------------------------------------------------
    # 4️⃣ fused second LayerNorm, gate multiplication and final linear
    # --------------------------------------------------------------
    final = torch.empty(R, dim, dtype=torch.float16, device=device)

    # weight_T = to_out.weightᵀ  shape (H, dim)
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)

    grid_final = (triton.cdiv(R, BLOCK_M),)
    final_norm_linear_kernel[grid_final](
        out_flat,
        out_gate,
        final,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        w_out_T,
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )
    del out_flat, out_gate, w_out_T

    # --------------------------------------------------------------
    # cast back to float32 and reshape to [B, N, N, dim]
    # --------------------------------------------------------------
    final_fp32 = final.view(B, N, N, dim).to(torch.float32)
    return final_fp32
