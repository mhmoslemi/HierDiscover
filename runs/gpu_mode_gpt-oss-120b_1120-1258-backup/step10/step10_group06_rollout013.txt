def layernorm_linear_fused_kernel(
    x_ptr, weight_ptr, ln_weight_ptr, ln_bias_ptr,
    out_ptr,
    rows: tl.int32,
    dim: tl.int32,
    out_dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # rows offset
    mask_m = offs_m < rows

    # Phase 1: compute mean & var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # shape [BLOCK_D]
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    inv_std = tl.rsqrt(var + eps)

    # Phase 2: compute linear projections
    # accumulator array for outputs: [BLOCK_M, BLOCK_O]
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # normalize x using layernorm parameters
        ln_w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        ln_b = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        # load weight block (dim->out_dim) transposed orientation (BLOCK_D, BLOCK_O)
        w_ptrs = weight_ptr + offs_d[:, None] * out_dim + tl.arange(0, BLOCK_O)[None, :]  # shape [BLOCK_D, BLOCK_O]
        w = tl.load(w_ptrs,
                    mask=mask_d[:, None] & (tl.arange(0, BLOCK_O)[None, :] < out_dim),
                    other=0.0).to(tl.float16)
        w_fp32 = w.to(tl.float32)

        # accumulation
        acc += tl.dot(norm, w_fp32)  # (M,D) dot (D,O) -> (M,O)

    # after loops, store acc as fp16
    for o_off in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim
        out_ptrs = out_ptr + offs_m[:, None] * out_dim + offs_o[None, :]

                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])

    x_ptr,           # const *[rows, H]  (float16) post-gemm
    gate_ptr,        # const *[rows, H]  (float16)
    weight_ptr,      # const *[H, dim]  (float16) (transposed maybe)

    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # Compute mean,var per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

        sum_sq += tl.sum(vals * vals, axis=1)

    var = sum_sq / hidden - mean * mean

        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)

                        mask=mask_m[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)  # fp16 to fp32

                        mask=mask_h[:, None] & mask_d[None, :],

"""
High‑performance Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The implementation fuses the two LayerNorms with the
preceding/following Linear layers, removes the intermediate
(normalized) tensors and keeps the heavy matrix‑multiplications (the
triple‑product) on cuBLAS.  The forward pass follows exactly the
reference PyTorch code but executes the three LayerNorm‑Linear stages
in custom Triton kernels.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input + fused Linear that produces the
   five projected tensors (left, right, three gates) in a single kernel
   (no temporary normalised tensor).

2️⃣ `split_fuse_gate_kernel` splits the fused projection, multiplies by the
   optional pair‑wise mask, evaluates the three sigmoid gates and stores
   `left`, `right` and `out_gate`.

3️⃣ Batched half‑precision GEMM (`torch.bmm`) computes the outgoing
   multiplicative update:
       out[b,h] = left[b,h] @ right[b,h].T   →   shape [B·H, N, N].

4️⃣ `layernorm_gate_to_out_kernel` fuses the second LayerNorm, the
   element‑wise multiplication with `out_gate` and the final Linear
   projection back to the model dimension `dim`.

All kernels run in fp32 for the arithmetic‑heavy parts (mean/variance,
dot‑products) and store the results in fp16.  The final tensor is cast
back to fp32 before being returned.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm + fused Linear (5·H outputs)
# ----------------------------------------------------------------------
@triton.jit
def ln_linear_fuse_kernel(
    x_ptr,                # const  *[rows, dim]      (float32)
    w_ptr,                # const  *[dim, out_dim]   (float16)  – transposed weight
    ln_w_ptr,             # const  *[dim]            (float32)  – LayerNorm γ
    ln_b_ptr,             # const  *[dim]            (float32)  – LayerNorm β
    out_ptr,              # mutable*[rows, out_dim]  (float16)
    rows: tl.int32,
    dim: tl.int32,
    out_dim: tl.int32,    # = 5 * hidden_dim   (compile‑time constant)
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim per iteration
    BLOCK_O: tl.constexpr,   # output‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Phase 1 – compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                  # [M, D]
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Phase 2 – compute fused projection:
    #           out = ( (x‑μ)·rstd·γ + β ) @ Wᵀ
    # ------------------------------------------------------------------
    n_out_blocks = tl.cdiv(out_dim, BLOCK_O)

    for ob in range(0, n_out_blocks):
        offs_o = ob * BLOCK_O + tl.arange(0, BLOCK_O)          # [BLOCK_O]
        mask_o = offs_o < out_dim

        # accumulator for this output tile
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # ----- normalize the current chunk of x -----
            x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)              # [M, D]

            ln_w = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
            ln_b = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

            norm = ((x - mean[:, None]) * rstd[:, None]) * ln_w[None, :] + ln_b[None, :]  # [M,D]

            # ----- weight block (dim × out_tile) -----
            w = tl.load(w_ptr + offs_d[:, None] * out_dim + offs_o[None, :],
                        mask=mask_d[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float16)               # [D, O]
            w_fp32 = w.to(tl.float32)

            # ----- accumulate -----
            acc += tl.dot(norm, w_fp32)                         # [M, O]

        # store the tile
        out = acc.to(tl.float16)
        out_ptrs = out_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """
    left  = (proj_left  * mask) * sigmoid(proj_left_gate)
    right = (proj_right * mask) * sigmoid(proj_right_gate)
    out_gate = sigmoid(proj_out_gate)
    All ops are in fp16 (sigmoid evaluated in fp32).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ----- per‑row mask (scalar) -----
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ----- iterate over hidden dimension -----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)       # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        p_left       = proj_ptr + base
        p_right      = proj_ptr + base + H
        p_left_gate  = proj_ptr + base + 2 * H
        p_right_gate = proj_ptr + base + 3 * H
        p_out_gate   = proj_ptr + base + 4 * H

        # load raw projection values
        left_raw      = tl.load(p_left,      mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw     = tl.load(p_right,     mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(p_left_gate, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw= tl.load(p_right_gate,mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw  = tl.load(p_out_gate,  mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # sigmoid (fp32 for stability)
        lg = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        rg = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        og = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m * lg
        right = right_raw * m * rg

        # store
        tl.store(left_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 og,    mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + out_gate multiplication + final Linear
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_to_out_kernel(
    x_ptr,                # const  *[rows, H]      (float16) – output of the batch GEMM
    gate_ptr,             # const  *[rows, H]      (float16) – out_gate
    w_ptr,                # const  *[H, dim]      (float16) – transposed final weight
    ln_w_ptr,             # const  *[H]           (float32) – LayerNorm γ (to_out_norm)
    ln_b_ptr,             # const  *[H]           (float32) – LayerNorm β (to_out_norm)
    out_ptr,              # mutable*[rows, dim]   (float16)
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
    BLOCK_D: tl.constexpr,   # output dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Phase 1 – mean & variance of the GEMM output (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)       # [BLOCK_H]
        mask_h = offs_h < H
        vals = tl.load(x_ptr + offs_m[:, None] * H + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Phase 2 – LayerNorm → *out_gate* → final Linear (fp32 → fp16)
    # ------------------------------------------------------------------
    n_dim_blocks = tl.cdiv(dim, BLOCK_D)

    for d_off in range(0, n_dim_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim

        # accumulator for this output tile
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
            mask_h = offs_h < H

            # load raw GEMM result
            x = tl.load(x_ptr + offs_m[:, None] * H + offs_h[None, :],
                        mask=mask_m[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)

            # LayerNorm parameters
            ln_w = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            ln_b = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

            # normalized hidden vector
            norm = ((x - mean[:, None]) * rstd[:, None]) * ln_w[None, :] + ln_b[None, :]

            # gate
            g = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                        mask=mask_m[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)                     # fp16→fp32

            gated = norm * g                                           # [M, Hb]

            # weight block (H × dim)
            w = tl.load(w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float16)                     # [Hb, D]
            w_fp32 = w.to(tl.float32)

            # accumulate contribution of this hidden slice
            acc += tl.dot(gated, w_fp32)                               # [M, D]

        # store the completed output block
        out = acc.to(tl.float16)
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device
    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Fused LayerNorm + Linear producing the five projection tensors
    # ------------------------------------------------------------------
    # concatenate the five linear weights (each [H, dim]) and transpose
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).t().contiguous()                     # shape [dim, 5*H]

    OUT_DIM = 5 * H                                           # compile‑time constant for the kernel
    proj = torch.empty(R, OUT_DIM, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    BLOCK_O = 64
    grid_proj = (triton.cdiv(R, BLOCK_M),)
    ln_linear_fuse_kernel[grid_proj](
        x,
        w_concat,
        weights["norm.weight"],
        weights["norm.bias"],
        proj,
        R,
        dim,
        OUT_DIM,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )
    del w_concat

    # ------------------------------------------------------------------
    # 2️⃣ Split, mask and apply sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_half = mask.to(torch.float16).reshape(R)          # [R]
        has_mask = 1
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM: left @ rightᵀ  (out shape [B·H, N, N])
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into the batch dimension
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # half‑precision batched GEMM (Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out_gate multiplication + final Linear
    # ------------------------------------------------------------------
    out_flat   = out_mat.view(R, H).contiguous()                # [R, H]  fp16
    final_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    # final weight matrix (H × dim) – transposed for kernel
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    grid_final = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_to_out_kernel[grid_final](
        out_flat,
        out_gate.view(R, H),
        w_out_T,
        weights["to_out_norm.weight"],   # γ (float32, [H])
        weights["to_out_norm.bias"],     # β (float32, [H])
        final_half,
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out_mat, left, right, w_out_T

    # ------------------------------------------------------------------
    # 5️⃣ Cast back to fp32 and reshape to [B, N, N, dim]
    # ------------------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
