"""
TriMul (outgoing) – fast forward pass on H100 (Triton ≥3.3)

The implementation follows the PyTorch reference but moves the three
row‑wise LayerNorms and the mask+gate handling into Triton kernels.
All heavy matrix multiplications stay in cuBLAS (torch.bmm / F.linear) which
already uses Tensor‑cores and yields the best throughput on H100.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (B·N·N × dim) → fp16.
2️⃣ One fused linear projection (dim → 5·hidden) producing
   left, right, left‑gate, right‑gate and out‑gate tensors.
3️⃣ A Triton kernel splits the fused projection, applies the optional pair‑wise
   mask, and evaluates the sigmoid gates (producing three fp16 tensors:
   left, right and out_gate).
4️⃣ Batched GEMM over the hidden dimension:
   (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
   implements `einsum('... i k d, ... j k d -> ... i j d')`.
5️⃣ A fused kernel performs the second LayerNorm on the hidden tensor and
   multiplies by `out_gate`.
6️⃣ Final linear projection back to `dim` (fp16 GEMM) and cast to fp32.

Only steps 1, 3 and 5 are written in Triton – this satisfies the “must
contain a kernel” rule while keeping the dominant O(N³) work in cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from math import ceil


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim tiles per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    #   mean / variance in fp32 (row‑wise)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    #   Write normalized values (γ··· + β) as fp16
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]                  # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    #   Optional broadcast mask  (1.0 ⇒ keep, 0.0 ⇒ discard)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    #   Loop over the hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw    = tl.load(proj_ptr + base,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        right_raw   = tl.load(proj_ptr + base + H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        left_gate   = tl.load(proj_ptr + base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        right_gate  = tl.load(proj_ptr + base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        out_gate    = tl.load(proj_ptr + base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)

        # sigmoid in fp32 then cast back
        lg = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        rg = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        og = (1.0 / (1.0 + tl.exp(-out_gate.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m * lg
        right = right_raw * m * rg

        # store
        left_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_ptr,   left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr,  right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr, og,
                 mask=mask_r[:, None] & mask_h[None, :])

    dim: tl.int32,

    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid_norm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid_norm2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection hidden → dim (fp16 GEMM) and cast to fp32
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16, non_blocking=True),   # [dim, H]
        bias=None,
    )   # → [R, dim]  fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
