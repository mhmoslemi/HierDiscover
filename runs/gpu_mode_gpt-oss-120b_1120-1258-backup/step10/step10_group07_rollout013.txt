"""
Optimized outgoing TriMul forward pass.

Algorithm:
1. Row‑wise LayerNorm of the 4‑D input (B·N·N × dim) using the
   highly‑optimized torch.nn.functional.layer_norm (FP32→FP16).
2. Fuse the five linear maps (left/right projections + three gates) into a
   single GEMM: (R, dim) @ (5·H, dim)^T → (R, 5·H).  The weight matrix is
   pre‑concatenated and stored in FP16.
3. Triton kernel `split_fuse_gate_kernel` splits the fused result into
   left, right and out_gate tensors, applies the optional pairwise mask,
   evaluates sigmoid gates (in FP32 for numeric stability) and writes the
   three FP16 tensors.
4. Reshape left/right to (B, H, N, N) and perform a batched FP16 matrix‑
   multiplication `left @ rightᵀ` using cuBLAS/Tensor‑Cores.
5. Second LayerNorm (FP32) + element‑wise multiplication with `out_gate`
   using torch.nn.functional.layer_norm.
6. Final FP16 linear projection back to `dim` and cast to FP32.

Only the split‑and‑gate step is implemented as a custom Triton kernel;
all heavy GEMMs and LayerNorms use highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask and sigmoid gates
# -----------------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r (r == b·N·N + i·N + k):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are in fp16; sigmoid evaluated in fp32 for stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # Load per‑row mask (scalar broadcasted over H) if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]                         # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Process the hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # layout in proj: [left, right, left_gate, right_gate, out_gate] each size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)                       # [R, dim] (FP32)
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                               # FP32
    x_norm = x_norm_fp32.to(torch.float16)          # FP16, ready for GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection: left/right + three gates   (R, 5*H)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (each [H, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5*H, dim]

    # (R, dim) @ (5*H, dim)^T  → (R, 5*H)
    proj = F.linear(x_norm, w_concat)                  # FP16
    del w_concat, x_norm                               # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        # flatten mask to match the row order used above
        mask_flat = mask.to(torch.float16).reshape(R)

    # Triton launch configuration
    BLOCK_R = 256   # rows per program
    BLOCK_H = 64    # hidden per program (must divide H efficiently)
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (B·H independent N×N matmuls)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for torch.matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2)   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2) # [B, H, N, N]

    # (B·H, N, N) @ (B·H, N, N)^T   → (B·H, N, N)   (FP16, Tensor‑Core)
    out_mat = torch.matmul(left, right.transpose(-1, -2))

    # back to [B, N, N, H] for the second LayerNorm
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP32) + multiplication with out_gate
    # ------------------------------------------------------------------
    # LayerNorm in FP32 then gate in FP16
    out_norm_fp32 = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                   # FP32
    # apply gate (out_gate has shape [R, H] → reshape)
    out_gate_reshaped = out_gate.view(B, N, N, H)       # [B, N, N, H]
    gated = (out_norm_fp32 * out_gate_reshaped).to(torch.float16)

    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

    BLOCK_R = 256    # rows per program
    BLOCK_H = 64     # hidden tiles per program
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (B·H independent N×N matmuls)
    # --------------------------------------------------------------
    # reshape to [B, H, N, N] for torch.matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2)   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2)  # [B, H, N, N]

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.matmul(left, right.transpose(-1, -2))   # FP16

    # back to [B, N, N, H] for the second LayerNorm
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP32) + element‑wise multiplication with out_gate
    # --------------------------------------------------------------
    out_norm_fp32 = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                   # FP32

    # reshape out_gate to match `out`'s layout and apply
    out_gate_reshaped = out_gate.view(B, N, N, H)       # [B, N, N, H]
    gated = (out_norm_fp32 * out_gate_reshaped).to(torch.float16)

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (FP16 linear) and cast to FP32
    # --------------------------------------------------------------
    final_half = F.linear(
        gated,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
