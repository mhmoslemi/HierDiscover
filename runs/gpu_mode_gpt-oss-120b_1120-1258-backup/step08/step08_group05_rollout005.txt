"""
Optimized forward pass of the *outgoing* TriMul operator (AlphaFold‑3).

Key ideas
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (and later on the hidden tensor)
   – implemented in Triton (fp32 reduction → fp16 output).

2️⃣ Fuse **all five** linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single half‑precision GEMM.  This reduces
   memory traffic and eliminates extra kernel launches.

3️⃣ Apply the optional pair‑wise mask, sigmoid gates and the element‑wise
   products entirely in FP16.

4️⃣ The expensive N³ update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is carried out as a batched half‑precision matmul
   (B·H, N, N) @ (B·H, N, N)ᵀ using cuBLAS/Torch – the fastest path on
   H100.

5️⃣ Second LayerNorm on the hidden dimension (again Triton).

6️⃣ Apply the output gate, project back to `dim` with a final FP16 GEMM,
   and return a FP32 result.

The implementation works for any batch ≤ 2, any sequence length used in
the test suite, supports masked / un‑masked variants and runs entirely
on‑device (no host‑CPU loops).  The heavy linear work is performed in
FP16; only the two LayerNorm reductions use FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – γ
    bias_ptr,       # const   * [dim]      (float32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the dim axis in tiles of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer arithmetic: [row, col] → row*dim + col
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+ε)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16) with affine γ,β
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                 # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 64) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D fp32 tensor `x` of shape [rows, dim].
    Returns an fp16 tensor of the same shape.
    """
    assert x.is_contiguous()
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – may be
                         ignored if config["nomask"] == True
        - weights_dict : mapping of all Linear / LayerNorm parameters
                         (float32 tensors)
        - config_dict  : must contain "dim", "hidden_dim" and may contain
                         "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                                   # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # flatten to [R, dim] for the Triton kernel
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        eps=eps,
    )                                                       # fp16, [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fuse ALL five linear projections (left, right, left‑gate,
    #    right‑gate, out‑gate) into a single GEMM.
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate weight matrices → shape [5·H, dim]
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                     # fp16

    # (R, dim) @ (dim, 5·H)^T  → (R, 5·H)
    fused_out = F.linear(x_norm, fused_weight)               # fp16

    # Split the fused result
    left_proj_raw   = fused_out[:, 0:H]                       # [R, H]
    right_proj_raw  = fused_out[:, H:2*H]                     # [R, H]
    left_gate_raw   = fused_out[:, 2*H:3*H]                   # [R, H]
    right_gate_raw  = fused_out[:, 3*H:4*H]                   # [R, H]
    out_gate_raw    = fused_out[:, 4*H:5*H]                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gates + element‑wise product
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] (broadcast over hidden dim)
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched half‑precision GEMM)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring hidden dim to the second axis → [B, H, N, N] (contiguous)
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # batch dimension for cuBLAS = B * H
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)   # (N, N)ᵀ
    )                                 # fp16, shape [B, H, N, N]

    # back to original layout [B, N, N, H]
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
    )                                     # fp16, [R, H]
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),                 # [R, H]
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                 # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
