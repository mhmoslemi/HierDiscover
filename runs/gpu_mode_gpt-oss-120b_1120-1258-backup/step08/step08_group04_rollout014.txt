"""
custom_kernel – Outgoing TriMul (AlphaFold3)

Implements the forward pass of the “outgoing” TriMul operator
using a small Triton kernel for the two LayerNorms and a
fully‑fused PyTorch implementation for the remaining linear
operations.  Optimisations over the reference implementation:

*   A single fused GEMM (`torch.nn.functional.linear`) computes the
    four projections, four gates **and** the output‑gate in one
    mat‑mul (5·hidden_dim × dim weight matrix).  This removes an
    extra GEMM and one memory copy.
*   The costly `einsum('... i k d, ... j k d -> ... i j d')` replaces the
    explicit `permute → view → torch.bmm` sequence, eliminating the
    two large `contiguous` copies for the left/right tensors.
*   The row‑wise LayerNorm is written in Triton and processes the
    entire channel dimension in a single tile (`BLOCK_D`), reducing the
    loop overhead.
*   All heavy arithmetic is performed in FP16 (Tensor‑Core) while the
    LayerNorms keep the numerical stability of FP32 accumulators.
*   Optional binary masks are applied after the linear projections,
    avoiding unnecessary work when `nomask=True`.

The function expects:
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

and returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const   * [total_rows, dim]   (float32)
    y_ptr,                # mutable * [total_rows, dim]   (float16)
    weight_ptr,           # const   * [dim]               (float32)  γ
    bias_ptr,             # const   * [dim]               (float32)  β
    total_rows,           # int (runtime)
    dim,                  # int (runtime)
    eps,                  # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Input  : float32
    Output : float16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute per‑row mean and variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the channel dimension in tiles of size BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # address: row * dim + col
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)                                   # fp32 load
        sum_val += tl.sum(vals, axis=1)                    # sum over dim‑tile
        sum_sq  += tl.sum(vals * vals, axis=1)             # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)            # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  Write normalized values: γ·(x−μ)·rstd + β   (as fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters (float32)
        - config_dict  : dict with keys
                         * "dim"         (int)
                         * "hidden_dim" (int)
                         * "nomask"      (bool, optional)

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣  Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # default = no mask
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (Row‑wise) – Triton
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim]  (fp32)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)           # [R, dim] (fp16)

    # tuning: one tile processes the whole channel when dim ≤ 128,
    # otherwise fall back to 32‑wide tiles.
    BLOCK_D0 = 128 if dim % 128 == 0 else 32
    BLOCK_M = 256
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D0,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Fuse left/right projections, gates and the output‑gate
    # ------------------------------------------------------------------
    # Weight layout: [5·H, dim] (left_proj, right_proj, left_gate,
    # right_gate, out_gate) – all bias‑free
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [5·H, dim]

    # GEMM: [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]   (FP16 tensor‑core)
    fused = F.linear(x_norm_fp16, fused_weight)   # [R, 5·H]

    # Split the fused result
    left_raw       = fused[:, 0:H]                         # [R, H]
    right_raw      = fused[:, H:2*H]                       # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]                     # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]                     # [R, H]
    out_gate_raw   = fused[:, 4*H:5*H]                     # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣  Apply the optional binary mask (if present)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1]
        mask_fp16 = mask.to(dtype=torch.float16, device=device).reshape(total_rows, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣  Apply sigmoid gates (in‑place to save memory)
    # ------------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H]
    right = right_raw * right_gate        # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣  N³ multiplicative update – fused einsum (no extra copy)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] – still column‑major in the H dimension
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('bikd,bjkd->bijd', left, right)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()      # [R, H] (fp16)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_D1 = 128 if H % 128 == 0 else 32
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D1,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)            # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣  Apply the output‑gate (element‑wise) and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)      # [B, N, N, H] fp16

    # Final linear: [R, H] @ [dim, H]ᵀ → [R, dim]   (FP16)
    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device)   # [dim, H]
    )   # [R, dim] (fp16)

    # Cast back to FP32 as expected by the surrounding model
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
