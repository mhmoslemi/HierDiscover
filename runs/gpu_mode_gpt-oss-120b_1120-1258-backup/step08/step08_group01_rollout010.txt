@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_head = tl.program_id(2)  # encodes batch*H

    # Compute batch and head indices
    b = pid_head // H
    h = pid_head % H

    # Compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Ranges
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Strides for batch and head offset
    b_off = b * stride_lb
    h_off = h * stride_lh   # same for left/right/out since same layout

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile [BLOCK_M, BLOCK_K]
        left_ptrs = left_ptr + b_off + h_off \
                    + offs_i[:, None] * stride_li \
                    + offs_k[None, :] * stride_lk
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load right tile transposed [BLOCK_K, BLOCK_N]
        # we want left: (..., j, k, h), but we load as (k, j) order for dot
        right_ptrs = right_ptr + b_off + h_off \
                     + offs_k[:, None] * stride_rk \
                     + offs_j[None, :] * stride_ri
        b = tl.load(right_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # [BLOCK_M, BLOCK_N]

    # Write result
    out_ptrs = out_ptr + b_off + h_off \
               + offs_i[:, None] * stride_oi \
               + offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
```

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).  The implementation follows the reference PyTorch
module but fuses the heavy element‑wise work (LayerNorm, masking,
gating) in Triton while delegating the dense matrix multiplications to
cuBLAS Tensor‑Cores.  The N³ multiplicative update is also expressed as
a custom Triton kernel that performs a tiled matrix‑multiply for each
head and batch (B·H independent GEMMs of shape N×N).  Mixed‑precision
(fp16) is used throughout and the final result is cast back to fp32.

Key steps
---------
1️⃣ Row‑wise LayerNorm on the input (float‑32 → float‑16) – Triton.
2️⃣ One fused linear layer producing left/right projections and their
   gates (4·H) plus the output gate (1·H) → shape [R,5·H] where R=B·N·N.
3️⃣ Apply the optional mask, sigmoid gates and element‑wise weighting.
4️⃣ Custom Triton kernel computes   out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   (the “outgoing” multiplicative update) for all heads in parallel.
5️⃣ Second LayerNorm over the hidden dimension (H) – Triton.
6️⃣ Apply the output‑gate, final linear projection and cast to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # const  * [total_rows, dim]   (float32)
    y_ptr,      # mutable* [total_rows, dim]   (float16)
    w_ptr,      # const  * [dim]               (float32)  – γ
    b_ptr,      # const  * [dim]               (float32)  – β
    total_rows, # int (runtime)
    dim,        # int (runtime)
    eps,        # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    pid = tl.program_id(0)                # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- compute mean & variance (fp32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                              # 1/√(var+ε)

    # ----- write normalized output (fp16) -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ  fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β  fp32
        out = norm * w[None, :] + b[None, :]                # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: outgoing N³ multiplicative update
# Computes for each (b, h):
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,         # const  * [B, N, N, H]  (fp16)
    right_ptr,        # const  * [B, N, N, H]  (fp16)
    out_ptr,          # mutable* [B, N, N, H]  (fp16)
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides (same layout)
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile over i dimension
    pid_j = tl.program_id(1)          # tile over j dimension
    pid_head = tl.program_id(2)       # encodes batch*H (size B*H)

    # Decode batch and head indices
    b = pid_head // H
    h = pid_head % H

    # Starting offsets for the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Offsets that are constant for this program
    b_off = b * stride_lb
    h_off = h * stride_lh          # same for left/right/out (stride on last dim)

    # Accumulator in fp32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile: shape [BLOCK_M, BLOCK_K] ----
        left_ptrs = left_ptr + b_off + h_off \
                    + offs_i[:, None] * stride_li \
                    + offs_k[None, :] * stride_lk
        a = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- load right tile (transposed) : shape [BLOCK_K, BLOCK_N] ----
        # right is indexed as (b, j, k, h); we load (k, j) order so that
        # tl.dot receives [BK , BN].
        right_ptrs = right_ptr + b_off + h_off \
                     + offs_k[:, None] * stride_rk \
                     + offs_j[None, :] * stride_ri
        b = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)          # [BLOCK_M, BLOCK_N] accumulate

    # ---- write back result (fp16) ----
    out_ptrs = out_ptr + b_off + h_off \
               + offs_i[:, None] * stride_oi \
               + offs_j[None, :] * stride_oj
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all weight tensors (float32)
        - config_dict  : {"dim": int,
                          "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                                 # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm: [R, dim] (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()         # [R, dim] fp32
    ln1_out = torch.empty(R, dim, dtype=torch.float16, device=device)

    # block sizes for layer‑norm (covers dim ≤ 384 efficiently)
    LN_BLOCK_M = 256
    LN_BLOCK_D = 64
    grid_ln = (triton.cdiv(R, LN_BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=LN_BLOCK_M,
        BLOCK_D=LN_BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear producing left/right projections + 4 gates
    #    (4H for left/right + left/right gates, + H for out_gate)
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)     # [5·H, dim]

    fused = F.linear(ln1_out, fused_weight)      # [R, 5·H]  (fp16)

    # split fused result
    left_raw      = fused[:, 0:H]                       # [R, H]
    right_raw     = fused[:, H:2 * H]                   # [R, H]
    left_gate_raw = fused[:, 2 * H:3 * H]               # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]              # [R, H]
    out_gate_raw  = fused[:, 4 * H:5 * H]               # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any) and sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.reshape(R, 1).to(device=device, dtype=torch.float16)
        left_raw = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left  = (left_raw * left_gate).contiguous()    # [R, H] fp16
    right = (right_raw * right_gate).contiguous()  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Outgoing N³ multiplicative update (custom Triton kernel)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    out = torch.empty(B, N, N, H, dtype=torch.float16, device=device)

    # strides for the three tensors (contiguous layout => known strides)
    s_left = left.stride()   # (N*N*H, N*H, H, 1)
    s_right = right.stride()
    s_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid_mul = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    tri_mul_outgoing_kernel[grid_mul](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(R, LN_BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=LN_BLOCK_M,
        BLOCK_D=LN_BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast back to fp32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)           # fp16
    out_norm = out_norm * out_gate.view(B, N, N, H)

    final_fp16 = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )                                            # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
