"""
custom_kernel: high‑performance forward of the outgoing TriMul operator.

Algorithm (row‑wise kernels + native torch kernels, mixed FP16):
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the flattened [B·N·N, dim] input.
2️⃣  Fuse the four linear layers (left/right proj + left/right gates) into one
    FP16 GEMM:  X_norm @ (W_left;W_right;W_left_gate;W_right_gate)ᵀ → [R,4·H].
    Split the result, apply the binary mask (if any) and the three sigmoid gates.
3️⃣  Compute the output‑gate with a second independent FP16 GEMM.
4️⃣  Reshape left/right tensors to [B, H, N, N] and perform the cubic
    multiplicative update as a batched matrix multiplication:
        out[b,h] = left[b,h] @ right[b,h]ᵀ   (Tensor‑Core FP16).
5️⃣  Row‑wise LayerNorm over the hidden dimension (FP16).
6️⃣  Multiply by the output‑gate, apply the final linear projection and cast
    back to FP32.

All heavy linear algebra runs in FP16 for Tensor‑Core utilization.
Two Triton kernels (first and second LayerNorm) are the only hand‑written GPU
code.  The implementation works for:
    B ∈ {1,2}, N ∈ {128…1024}, dim ∈ {128,384,768}, hidden_dim = 128,
    with or without a binary mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Triton row‑wise LayerNorm: y = ((x-μ)/σ)·γ + β   (x fp32, y fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (fp32)
    y_ptr,          # mutable * [total_rows, dim]   (fp16)
    w_ptr,          # const   * [dim]                (fp32)   – γ
    b_ptr,          # const   * [dim]                (fp32)   – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1) Compute mean and variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) Write normalized result (fp16) = (x-mean)/σ * γ + β
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ   (fp32)

        # γ, β are fp32 – keep them in fp32 then cast the final result.
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]            # fp32

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of weight names to torch.Tensor (float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack and constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm: fp32 → fp16
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16,
                              device=device)

    BLOCK_M = 256      # rows per program (tuned for H100)
    BLOCK_D = 64       # columns per tile
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (fp32)
        weights["norm.bias"],        # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates into one GEMM (FP16)
    # ------------------------------------------------------------------
    #   fused_weight shape: [4*H, dim]
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # FP16 weight

    fused = F.linear(x_norm_fp16, fused_weight)   # [R, 4*H] FP16

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2*H]
    left_gate_raw  = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (independent GEMM)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Apply optional mask and sigmoid gates (in‑place where possible)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] → broadcast to [R, H]
        mask_fp16 = mask.reshape(R, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # sigmoid in‑place to keep memory line‑footprint low
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H] FP16
    right = right_raw * right_gate        # [R, H] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Cubic update: batched matrix multiplication over the sequence dim
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # permute to bring H into the batch dimension → [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch and heads → [B*H, N, N]
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matmul: (N×N) @ (N×N)ᵀ  →  [B*H, N, N]   (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()            # [R, H] fp16
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection and cast back to fp32
    # ------------------------------------------------------------------
    out_norm_fp16 = out_norm_fp16 * out_gate.view(R, H)   # element‑wise gating

    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] FP16

    # reshape to original layout and convert to float32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
