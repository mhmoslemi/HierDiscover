@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_m, stride_left_n,
    stride_right_batch, stride_right_m, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute batch index and tile indices
    batch_id = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    tile_id = pid % (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_i = tile_id // tl.cdiv(N, BLOCK_N)
    block_j = tile_id % tl.cdiv(N, BLOCK_N)

    i0 = block_i * BLOCK_M
    j0 = block_j * BLOCK_N

    # offsets within tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointer offsets compute linear address
        left_ptrs = left_ptr + batch_id * stride_left_batch \
            + offs_i[:, None] * stride_left_m \
            + offs_k[None, :] * stride_left_n

        right_ptrs = right_ptr + batch_id * stride_right_batch \
            + offs_k[:, None] * stride_right_m \
            + offs_j[None, :] * stride_right_n

        left_tile = tl.load(left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0)
        right_tile = tl.load(right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0)

        # convert to fp32
        left_tile = left_tile.to(tl.float32)
        right_tile = right_tile.to(tl.float32)
        acc += tl.dot(left_tile, right_tile)

    # write results
    out_ptrs = out_ptr + batch_id * stride_out_batch \
        + offs_i[:, None] * stride_out_m \
        + offs_j[None, :] * stride_out_n

    tl.store(out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – high‑performance Triton implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm of the 4‑D pair representation (FP32 → FP16) with a
    custom Triton kernel (two passes: mean/var → normalize).

2️⃣  Fuse the four linear maps (left_proj, right_proj, left_gate, right_gate)
    into a single GEMM (shape (4·H, dim)) – keep everything in FP16.

3️⃣  Compute the out‑gate linear map separately.

4️⃣  (Optional) apply the pairwise mask, then sigmoid the three gate tensors
    and multiply the projection tensors by their gates.

5️⃣  Cubic update  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
    Implemented as a tiled batched GEMM (B·H independent N×N matrix
    multiplications) in a custom Triton kernel, staying in FP16 /
    accumulating in FP32.

6️⃣  Second row‑wise LayerNorm over the hidden dimension H (again Triton).

7️⃣  Multiply by the out‑gate and a final linear projection back to the original
    feature dimension dim.  The result is cast to FP32 to match the reference.

Only the heavy‐weight LayerNorms and the cubic update are written in Triton;
all other work uses highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 -> FP16) – reusable for both norms
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,              # const  * [total_rows, dim]   (float32)
    y_ptr,              # mutable* [total_rows, dim]   (float16)
    gamma_ptr,          # const  * [dim]               (float32) – weight
    beta_ptr,           # const  * [dim]               (float32) – bias
    total_rows: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # ---------- Partial sums (FP32) ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- Write normalized values (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Cubic update (batched GEMM) – computes Σₖ left[i,k,h] * right[j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,   # const  * [B*H, N, N] (float16)
    right_ptr,  # const  * [B*H, N, N] (float16)
    out_ptr,    # mutable* [B*H, N, N] (float16)
    batch_dim,  # int32 = B*H
    N,          # int32
    stride_l_batch, stride_l_m, stride_l_n,
    stride_r_batch, stride_r_m, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Number of tiles per matrix
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # Decode program ID → (batch, tile_i, tile_j)
    batch_id = pid // (num_i * num_j)
    tile_id  = pid %  (num_i * num_j)
    tile_i   = tile_id // num_j
    tile_j   = tile_id % num_j

    # Starting indices of the current tile
    i0 = tile_i * BLOCK_M
    j0 = tile_j * BLOCK_N

    # Offsets inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  : (BLOCK_M, BLOCK_K) = left[batch, i, k]
        left_ptrs = left_ptr \
            + batch_id * stride_l_batch \
            + offs_i[:, None] * stride_l_m \
            + offs_k[None, :] * stride_l_n

        # right tile : (BLOCK_K, BLOCK_N) = right[batch, k, j]
        # (notice the transposition – we load as (k, j) directly)
        right_ptrs = right_ptr \
            + batch_id * stride_r_batch \
            + offs_k[:, None] * stride_r_m \
            + offs_j[None, :] * stride_r_n

        left_tile  = tl.load(left_ptrs,
                             mask=mask_i[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)        # (BLOCK_M, BLOCK_N)

    # Store the result (convert back to FP16)
    out_ptrs = out_ptr \
        + batch_id * stride_o_batch \
        + offs_i[:, None] * stride_o_m \
        + offs_j[None, :] * stride_o_n

    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model weight tensors (see keys in the function)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)

    BLOCK_R = 256   # rows per program (tuned for H100)
    BLOCK_D = 128   # covers dim ≤ 384
    grid_norm = (triton.cdiv(total_rows, BLOCK_R),)

    rowwise_layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ  (FP32)
        weights["norm.bias"],            # β  (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (4·H × dim)
    # --------------------------------------------------------------
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (4·H, dim)

    # Linear: [R, dim] @ [dim, 4·H]ᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm, w_4h)       # FP16

    # Slice the four tensors
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Out‑gate (separate linear map)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # [R, H] FP16

    # --------------------------------------------------------------
    # 3️⃣ Masking, sigmoid gates and gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] (broadcasted)
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # In‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate    # [R, H] FP16
    right = right_proj_raw * right_gate  # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Cubic update via custom Triton GEMM (B·H independent N×N matmuls)
    # --------------------------------------------------------------
    # Move H into the batch dimension for the GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat  = left_perm.view(B * H, N, N)   # [B·H, N, N] FP16
    right_flat = right_perm.view(B * H, N, N)  # [B·H, N, N] FP16
    out_flat   = torch.empty_like(left_flat)   # placeholder for result

    # Strides for the flattened tensors (int64 -> Python int)
    stride_l_batch, stride_l_m, stride_l_n = left_flat.stride()
    stride_r_batch, stride_r_m, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    # Tile sizes – chosen to keep each program within shared‑memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 16

    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    total_programs = (B * H) * num_i * num_j
    grid_mul = (total_programs,)

    tri_mul_kernel[grid_mul](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        stride_l_batch, stride_l_m, stride_l_n,
        stride_r_batch, stride_r_m, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H (row‑wise)
    # --------------------------------------------------------------
    out_flat2 = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat2, dtype=torch.float16)

    grid_norm2 = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_kernel[grid_norm2](
        out_flat2,
        out_norm,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        total_rows,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate               # [B, N, N, H] FP16

    final = F.linear(
        gated.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # [R, dim] FP16

    final = final.view(B, N, N, dim).to(torch.float32)   # cast to FP32
    return final
