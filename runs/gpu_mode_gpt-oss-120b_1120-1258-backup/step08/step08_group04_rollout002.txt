"""
custom_kernel – Outgoing TriMul (AlphaFold3)

Implements the forward pass of the “outgoing” TriMul operator used in AlphaFold3,
Chai, Protenix, … .  The implementation follows the reference PyTorch module but
optimizes the two LayerNorms with Triton and fuses the four projection Linear
layers (left/right + gates) into a single GEMM.  All heavy arithmetic is done
in fp16, while the final output is cast back to fp32.

Key ideas
---------
1️⃣ Row‑wise LayerNorm (input → fp16) in a single Triton kernel.
2️⃣ Fuse left_proj, right_proj, left_gate, right_gate into one matrix‑multiply:
      X_norm @ W_fusedᵀ → [R, 4·H]   (R = B·N·N).
3️⃣ Optional binary mask applied to the two raw projections.
4️⃣ Sigmoid on the three gate tensors (in‑place) and element‑wise gating.
5️⃣ Multiplicative update per hidden channel:
      out[b,h] = left[b,h] @ right[b,h]ᵀ   (batched fp16 GEMM, cuBLAS).
6️⃣ Second LayerNorm (hidden dimension) in the same Triton kernel.
7️⃣ Scale by the output‑gate and apply the final Linear projection to recover
   the original channel size.

The kernel works for any combination of batch size (1‑2), sequence length
(256‑1024), model dimension (128 or 384) and hidden dimension (128).  The mask
may be omitted (`nomask=True`) or supplied as a boolean/float tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)  γ
    bias_ptr,       # const   * [dim]               (float32)  β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / √(σ²+ε)) * γ + β
    Input  : fp32
    Output : fp16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of the input
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 → fp32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)            # 1 / √(var+ε)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values: γ·(x−μ)·rstd + β (as fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Helper: compute a fused linear for 4·H outputs (proj + gates)
# ----------------------------------------------------------------------
def fused_linear(x_fp16: torch.Tensor, w_fused: torch.Tensor) -> torch.Tensor:
    """
    Compute x @ w_fusedᵀ where:
        x    : [R, dim]   (fp16)
        w_fused : [4·H, dim] (fp16)
    Returns:
        out : [R, 4·H]   (fp16)
    """
    # torch.nn.functional.linear does the transpose internally
    return F.linear(x_fp16, w_fused)   # fp16



# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict with all Linear/LayerNorm parameters
            - config_dict  : dict with keys
                * "dim" (int) – model dimension
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # flatten to [R, dim] (contiguous view is enough)
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256   # rows per program (tuned for H100)
    BLOCK_D = 32    # cols per iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        w["norm.weight"],   # γ  (float32)
        w["norm.bias"],     # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM
    # ------------------------------------------------------------------
    # Build a weight matrix of shape [4·H, dim] (no bias, all fp16)
    w_fused = torch.cat(
        [
            w["left_proj.weight"],   # [H, dim]
            w["right_proj.weight"],  # [H, dim]
            w["left_gate.weight"],   # [H, dim]
            w["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [4·H, dim]

    fused = fused_linear(x_norm_fp16, w_fused)   # [R, 4·H]  fp16

    # Split the fused result
    left_raw       = fused[:, 0:H]                 # [R, H]
    right_raw      = fused[:, H:2*H]               # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]             # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]             # [R, H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        w["out_gate.weight"].to(dtype=torch.float16, device=device)  # [H, dim]
    )   # [R, H]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates and combine
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_fp16 = mask.reshape(total_rows, 1).to(dtype=torch.float16, device=device)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid for memory efficiency
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H]  fp16
    right = right_raw * right_gate    # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Multiplicative update (batched GEMM)
    #    out[b, h] = left[b, h] @ right[b, h]ᵀ
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → permute to [B, H, N, N] for batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch & heads → [B·H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Batched matrix‑multiply (FP16 Tensor‑Core GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    # Element‑wise gating (still fp16)
    out_norm = out_norm * out_gate.view(B, N, N, H)

    # Final Linear: [R, H] → [R, dim]   (fp16)
    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        w["to_out.weight"].to(dtype=torch.float16, device=device)   # [dim, H]
    )   # [R, dim]  fp16

    # Cast back to fp32 as required by the downstream model
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
