"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D pair representation (FP32 → FP16) via a
   custom Triton kernel.

2️⃣ Fuse the four linear maps (left_proj, right_proj, left_gate, right_gate)
   into a single GEMM: W₄h = [Wₗₚ; Wᵣₚ; Wₗg; Wᵣg]  (shape 4·H × dim).

3️⃣ Compute the separate out‑gate linear map.

4️⃣ (Optional) apply the pairwise mask, then sigmoid the three gate tensors
   and multiply projections by their gates.

5️⃣ Reshape tensors to (B, N, N, H) and perform the cubic update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   This is realised as a batched GEMM over the combined batch‑hidden
   dimension (B·H, N, N) → (B·H, N, N) using torch.bmm (Tensor‑Core
   friendly FP16).

6️⃣ Row‑wise LayerNorm over the hidden dimension H (again via Triton).

7️⃣ Multiply by the out‑gate and a final linear projection back to the
   original feature dimension dim.

All heavy linear work stays in FP16; the final result is cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,       # const  * [total_rows, dim] (float32)
    y_ptr,       # mutable* [total_rows, dim] (float16)
    w_ptr,       # const  * [dim] (float32) – gamma
    b_ptr,       # const  * [dim] (float32) – beta
    total_rows,  # int32
    dim,         # int32
    eps,         # float32
    BLOCK_R: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # column tile size
):
    """
    Row‑wise LayerNorm:
    y = ((x - μ) / √(σ² + eps)) * γ + β
    Input  : x (FP32)
    Output : y (FP16)
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    # row offsets, may be out‑of‑bounds → mask later
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean and variance (FP32) in a tiled fashion
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # sum over columns
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / √(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model weights (see keys below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape      # [B, N, N, dim]
    R = B * N * N                         # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – FP16 output
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()          # (R, dim) FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 256            # rows per program (tunable)
    BLOCK_D = 128            # column tile (covers dim ≤ 384)
    grid = (triton.cdiv(R, BLOCK_R),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # gamma  (FP32)
        weights["norm.bias"],            # beta   (FP32)
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear layers (4·H × dim)
    # --------------------------------------------------------------
    # concatenate weight matrices (no bias)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # (4·H, dim)

    # one GEMM → (R, 4·H)
    proj_gate = F.linear(x_norm, w_4h)   # FP16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # out‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: (B,N,N) → (R,1) in FP16
        mask_fp16 = mask_tensor.reshape(R, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # inplace sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate      # (R, H) FP16
    right = right_proj_raw * right_gate    # (R, H) FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)    # (B, N, N, H)
    right = right.view(B, N, N, H)  # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Cubic update (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    # bring H into the batch dimension: (B, H, N, N) → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N)

    # batched matrix multiplication: (i,k)·(j,k)^T → (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H (row‑wise)
    # --------------------------------------------------------------
    out_flat = out.reshape(R, H).contiguous()          # (R, H) FP32 → FP16 output
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_R),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # gamma (FP32)
        weights["to_out_norm.bias"],     # beta  (FP32)
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)  # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final linear projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate          # (B, N, N, H) FP16

    final = F.linear(
        gated.reshape(R, H),                     # (R, H)
        weights["to_out.weight"].to(torch.float16)  # (dim, H)
    )   # (R, dim) FP16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
