"""
Outgoing TriMul (AlphaFold‑3) – Fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the pair representation (FP32 → FP16) via a
   custom Triton kernel.
2️⃣ Fuse the four projection matrices (left_proj, right_proj, left_gate,
   right_gate) and the out‑gate matrix into a single GEMM of shape
   (5·H, dim).  Split the result into the five tensors that follow the
   reference implementation.
3️⃣ Apply the optional pairwise mask, sigmoid the three gate tensors and
   multiply the projection tensors by their gates.
4️⃣ Compute the cubic update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   using a single ``torch.einsum`` call (no explicit permutes /
   copies).  The whole operation stays in FP16.
5️⃣ Row‑wise LayerNorm over the hidden dimension H (again via Triton).
6️⃣ Multiply by the out‑gate and apply the final linear projection back
   to the original feature dimension dim.
7️⃣ Cast the result back to FP32 (the reference implementation’s dtype).

All heavy linear work stays in half‑precision, memory traffic is reduced
by fusing the linear maps and by avoiding costly ``permute``/``contiguous``
copies for the cubic update.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm kernel (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (float32)
    y_ptr,          # mutable* [total_rows, dim] (float16)
    w_ptr,          # const  * [dim] (float32) – γ
    b_ptr,          # const  * [dim] (float32) – β
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / √(σ²+eps)) * γ + β
    Input  : x – FP32
    Output : y – FP16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    # ---------- compute mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / √(var + eps)

    # ---------- write normalized values (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model weights (see keys below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_R = 256          # rows per program (tuned for B·N² ≤ 1M)
    BLOCK_D = 128          # columns per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_R),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],       # γ (FP32)
        weights["norm.bias"],         # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear maps (5·H × dim)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (left_proj, right_proj,
    # left_gate, right_gate, out_gate) → shape (5·H, dim)
    w_5h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],          # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                         # (5·H, dim)

    # Single GEMM: produces left_proj, right_proj, left_gate,
    # right_gate and out_gate in one pass.
    proj_all = F.linear(x_norm, w_5h)           # [R, 5·H] FP16

    # Split the fused output
    left_proj_raw  = proj_all[:, 0:H]                         # [R, H]
    right_proj_raw = proj_all[:, H:2*H]                       # [R, H]
    left_gate_raw  = proj_all[:, 2*H:3*H]                     # [R, H]
    right_gate_raw = proj_all[:, 3*H:4*H]                     # [R, H]
    out_gate_raw   = proj_all[:, 4*H:5*H]                     # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] FP16 (broadcast over hidden dim)
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # In‑place sigmoid for speed
    left_gate_raw  = left_gate_raw.sigmoid_()
    right_gate_raw = right_gate_raw.sigmoid_()
    out_gate_raw   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate_raw   # [R, H] FP16
    right = right_proj_raw * right_gate_raw # [R, H] FP16
    out_gate = out_gate_raw                 # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # ------------------------------------------------------------------
    left  = left .view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Cubic update via a single einsum (no extra permutes)
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Result shape: (B, N, N, H) – stays in FP16
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H (row‑wise)
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()   # [R, H] FP16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        total_rows,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)                # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final linear projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # [B, N, N, H] FP16

    final = F.linear(
        gated.view(total_rows, H),                     # [R, H]
        weights["to_out.weight"].to(torch.float16)    # (dim, H)
    )   # [R, dim] FP16

    final = final.view(B, N, N, dim).to(torch.float32)   # cast to FP32

    return final
