"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style). 

The implementation follows the reference PyTorch module but fuses the
expensive element‑wise work in Triton kernels while delegating the large
matrix multiplications to cuBLAS/Tensor‑Cores (via torch.nn.functional.linear
and torch.bmm).  

Main steps
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16) – Triton.
2️⃣ One fused linear layer producing the four projections/gates
   (left, right, left‑gate, right‑gate) in a single GEMM.
3️⃣ Separate linear for the output‑gate.
4️⃣ Optional mask application, sigmoid on the three gates and
   element‑wise gating.
5️⃣ Batched GEMM that computes the N³ multiplicative update:
   out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]  (half‑precision).
6️⃣ Second LayerNorm over the hidden dimension (H) – Triton.
7️⃣ Apply the output‑gate, a final linear projection back to `dim` and
   cast the result to float32.

All heavy element‑wise work (LayerNorm, optional mask, sigmoids) is done
in Triton; the dense matrix products use cuBLAS for maximum throughput.
Mixed‑precision (fp16) is used throughout except the final cast to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (fp32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a tile of the input
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                    # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β fp32
        out = norm * w[None, :] + b[None, :]                    # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict) where
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (float32)
        - config_dict  : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs & configuration
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim                  # hidden dimension

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm: [R, dim] (fp32 → fp16) – Triton
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()        # [R, dim] fp32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    # Choose tile sizes that comfortably cover the common ``dim`` (≤384)
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their two gates (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate the four weight matrices (each [H, dim])
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,                           # [4·H, dim]
    ).to(device=device, dtype=torch.float16)

    # Fused GEMM: [R, dim] @ [4·H, dim].T → [R, 4·H]  (fp16, no bias)
    fused = F.linear(ln1_out, w_proj)   # fp16

    # Split the fused result
    left_raw      = fused[:, 0:H]               # [R, H]
    right_raw     = fused[:, H:2 * H]           # [R, H]
    left_gate_raw = fused[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]      # [R, H]

    # Separate output‑gate GEMM
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise weighting
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] → [R, 1] (float16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid to keep memory pressure low
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate    # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)       # [B, N, N, H]
    right = right.view(B, N, N, H)      # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)

    # Move hidden dimension to the batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & heads → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (Tensor‑Core)
    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Restore layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H) – Triton
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast back to fp32
    # --------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)    # fp16
    out_norm = out_norm * out_gate         # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # float32 output
    return final
