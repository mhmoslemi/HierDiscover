"""
custom_kernel – Outgoing TriMul (AlphaFold3)

This implementation follows the “outgoing” TriMul operator described in the
AlphaFold3 paper.  The forward pass consists of

1️⃣ Row‑wise LayerNorm over the last channel (`dim`) – done with a Triton
   kernel (`layernorm_kernel`).  Input is FP32, output is FP16.

2️⃣ Fuse the four linear projections (left/right) and the three gating
   matrices (left_gate, right_gate, out_gate) into **one** FP16 GEMM:
      X_norm (R×dim) @ W_fusedᵀ  →  (R×5·H)   where R = B·N·N and H = hidden_dim.
   The result is split into:
      left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw.

3️⃣ (Optional) apply the binary mask to `left_raw` and `right_raw`.

4️⃣ Apply the three sigmoids and fuse them:
      left  = left_raw  * sigmoid(left_gate_raw)
      right = right_raw * sigmoid(right_gate_raw)
      out_gate = sigmoid(out_gate_raw)

5️⃣ N³ multiplicative update – a batched matrix multiply for each head:
      out[b,h] = left[b,h] @ right[b,h]ᵀ   (FP16 Tensor‑Core GEMM).

6️⃣ Row‑wise LayerNorm over the hidden dimension `H` – another Triton kernel.

7️⃣ Multiply by the output‑gate and perform the final linear projection
   (FP16 GEMM) back to `dim`.

All heavy arithmetic is in FP16, the two LayerNorms are performed in
Triton, and the only PyTorch operations are highly‑optimized GEMMs
(`torch.nn.functional.linear` and `torch.bmm`).  The final result is
returned as a FP32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 input → float16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32 or float16)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)  γ
    bias_ptr,         # const   * [dim]               (float32)  β
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Input   x : fp32 or fp16
    Output  y : fp16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a tile of the row
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # promote to fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)         # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values: γ·(x−μ)·rstd + β  (as fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # promote to fp32

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ  (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β  (fp32)

        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype=float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters
        - config_dict  : dict with keys:
                         * "dim" (int)
                         * "hidden_dim" (int)
                         * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """

    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True ⇒ no mask
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp16 output
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()      # [R, dim], fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256   # rows per program (tuned)
    BLOCK_D = 32    # cols per iteration (tuned)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse all linear projections and gating matrices (single GEMM)
    # --------------------------------------------------------------
    # Concatenate (left_proj, right_proj, left_gate, right_gate, out_gate)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [5·H, dim]

    # GEMM: (R, dim) @ (5·H, dim)ᵀ → (R, 5·H)   (FP16)
    fused = F.linear(x_norm_fp16, w_fused)     # fp16

    # Split the fused tensor
    left_raw       = fused[:,               0:H]          # [R, H]
    right_raw      = fused[:,           H:2*H]          # [R, H]
    left_gate_raw  = fused[:,       2*H:3*H]          # [R, H]
    right_gate_raw = fused[:,   3*H:4*H]          # [R, H]
    out_gate_raw   = fused[:, 4*H:5*H]          # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise combination
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask -> [R, 1] (fp16) and broadcast
        mask_fp16 = mask.to(dtype=torch.float16,
                            device=device).reshape(total_rows, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid for memory efficiency
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, H, N, N] for the batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse batch & heads: [B·H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H (row‑wise)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H] fp16
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # elementwise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device)   # [dim, H]
    )   # [R, dim] fp16

    # Cast back to FP32 as the model expects
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
