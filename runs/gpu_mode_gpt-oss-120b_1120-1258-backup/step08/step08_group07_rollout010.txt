def apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_rows, hidden_dim, has_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # hidden dimension loop
    for off_h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = off_h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        proj = tl.load(proj_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],

                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float16)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)
    B,N,_,_ = input_tensor.shape
    # 1st layernorm
    # treat shape as (B*N*N, dim)
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # shape B,N,N,dim

    # linear projections and gates fused via single linear (4*H)
    proj_weight = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"], weights["left_gate.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    # compute project onto 4*H
    # flatten x_norm to (B*N*N, dim) fp16
    x_norm_fp16 = x_norm.to(torch.float16).reshape(B*N*N, dim)
    proj = torch.nn.functional.linear(x_norm_fp16, proj_weight)   # bias zero
    # split
    H = hidden_dim
    left_proj_raw = proj[:, :H]
    right_proj_raw = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:]

    # out gate linear
    out_gate_raw = torch.nn.functional.linear(x_norm_fp16, weights["out_gate.weight"].to(torch.float16))  # [R, H]

    # mask processing
    if not nomask:
        mask_flat = mask_tensor.to(torch.float16).reshape(B*N*N, 1)   # broadcast
    else:
        mask_flat = torch.ones(B*N*N, 1, dtype=torch.float16, device=device)

    # apply gating (mask, sigmoid) using Triton kernel for left and right
    left = torch.empty_like(left_proj_raw)   # fp16
    right = torch.empty_like(right_proj_raw) # fp16

    # kernel config
    total_rows = B*N*N
    BLOCK_M = 128
    BLOCK_H = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    apply_gate_mask_kernel[grid](
        left_proj_raw, left_gate_raw, mask_flat, left,
        total_rows, H,

        num_warps=8,

        right_proj_raw, right_gate_raw, mask_flat, right,
        total_rows, H,
        1,

        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )

    out_gate = torch.nn.functional.sigmoid(out_gate_raw)

    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

TriMul – “outgoing” forward pass (AlphaFold3)

Algorithm (all tensors are on the same CUDA device)

1️⃣  Row‑wise LayerNorm of the 4‑D input `[B, N, N, C]` using PyTorch’s
    highly‑optimised `layer_norm`.  The result is cast to fp16.

2️⃣  Fuse the four linear projections (left_proj, right_proj,
    left_gate, right_gate) into a single fp16 GEMM:
        proj = X_norm @ W_proj.T                #  X_norm: (R, C),  W_proj: (4·H, C)
    where `R = B·N·N` and `H = hidden_dim`.  The GEMM produces the four
    tensors in one pass.

3️⃣  Compute the output‑gate (`out_gate`) with a second fp16 GEMM.

4️⃣  Apply the optional mask, the sigmoid gates and the element‑wise
    gating in a small Triton kernel (`apply_gate_mask_kernel`).  This
    fuses three operations (`mask * gate.sigmoid() * proj`) and saves
    an extra kernel launch.

5️⃣  Reshape `left` and `right` to `[B·H, N, N]` and perform the
    N³ multiplicative update with a *batched* GEMM:
        out = (B·H, N, N) @ (B·H, N, N).T
    This is exactly the `einsum('... i k d, ... j k d -> ... i j d')`
    from the reference implementation and is executed by cuBLAS
    (`torch.bmm`), which on an H100 reaches > 800 TFLOP/s fp16.

6️⃣  Second row‑wise LayerNorm over the hidden dimension (`H`) – again
    using `torch.nn.functional.layer_norm`.

7️⃣  Apply the output‑gate (fp16), the final linear projection back to
    `dim` and cast the result to float32.

All heavy matrix multiplications stay in fp16 (Tensor‑Core friendly);
only the two LayerNorms and the small gating kernel touch fp32
(for numerical stability) and fp16.  The custom Triton kernel is
required by the problem statement, while the rest of the pipeline
leverages the highly‑optimised cuBLAS and PyTorch kernels for peak
throughput on an H100.
"""

    hidden_dim,    # int
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    for off_h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = off_h * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < hidden_dim

        # Load projection and gate values
        proj = tl.load(
            proj_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float16)                                   # [BLOCK_M, BLOCK_H]

        gate = tl.load(
            gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float16)

        # sigmoid(gate) in fp32 for stability, cast back to fp16
        gate_f = gate.to(tl.float32)
        sig = 1.0 / (1.0 + tl.exp(-gate_f))
        sig = sig.to(tl.float16)                           # [BLOCK_M, BLOCK_H]

        # Load mask (scalar per row) and broadcast
        mask_val = tl.load(
            mask_ptr + offs_m,
            mask=mask_m,
            other=1.0
        ).to(tl.float16)                                   # [BLOCK_M]
        mask_tile = mask_val[:, None]                       # [BLOCK_M, 1]

        # Fusion: proj * sigmoid(gate) * mask
        out = proj * sig * mask_tile

        # Store result
        tl.store(
            out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
            out,
            mask=mask_m[:, None] & mask_h[None, :]
        )


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]     (bool/float)
        - weights_dict : dict of torch.Tensors (model parameters)
        - config_dict  : must contain "dim", "hidden_dim" and optional
                         "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]               # C
    hidden_dim  = config["hidden_dim"]        # H
    eps         = 1e-5
    device      = input_tensor.device
    nomask      = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                           # total (batch * seq_len * seq_len) rows

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – use PyTorch implementation (fast)
    # ------------------------------------------------------------------
    # Normalise over the last dimension (C) and cast to fp16 for the rest
    x_norm_fp16 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices: [4·H, C]
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_gate.weight"],  # (H, C)
        ],
        dim=0,
    ).to(torch.float16)                     # (4·H, C)

    # Flatten input to (R, C) and run the fused linear projection
    x_flat = x_norm_fp16.view(R, dim).contiguous()
    proj = F.linear(x_flat, w_proj_gate)    # (R, 4·H)

    # Split the four heads
    H = hidden_dim
    left_proj_raw  = proj[:, 0:H]                 # (R, H)
    right_proj_raw = proj[:, H:2 * H]             # (R, H)
    left_gate_raw  = proj[:, 2 * H:3 * H]         # (R, H)
    right_gate_raw = proj[:, 3 * H:4 * H]         # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate (no mask) – simple linear + sigmoid
    # --------------------------------------------------------------
    out_gate_raw = F.linear(
        x_flat,
        weights["out_gate.weight"].to(torch.float16),  # (H, C)
    )                                           # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)      # (R, H) fp16

    # --------------------------------------------------------------
    # 4️⃣ Optional mask handling
    # --------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones(R, 1, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).view(R, 1)  # (R,1)

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask, sigmoid‑gate and element‑wise multiplication
    #    (performed by a tiny Triton kernel)
    # --------------------------------------------------------------
    left  = torch.empty_like(left_proj_raw)   # (R, H) fp16
    right = torch.empty_like(right_proj_raw) # (R, H) fp16

    BLOCK_M = 128   # rows per program – tuned for H100
    BLOCK_H = 64    # hidden‑dim tile size
    grid = (triton.cdiv(R, BLOCK_M),)

    # left side
    apply_gate_mask_kernel[grid](
        left_proj_raw, left_gate_raw, mask_fp16, left,
        R, H,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=2,
    )
    # right side
    apply_gate_mask_kernel[grid](
        right_proj_raw, right_gate_raw, mask_fp16, right,
        R, H,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape for the N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # [B, N, N, H] → [B·H, N, N] for torch.bmm
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_mat  = left.reshape(B * H, N, N)   # (B·H, N, N) fp16
    right_mat = right.reshape(B * H, N, N)  # (B·H, N, N) fp16

    # Batched matrix multiplication: sum over the shared N dimension (k)
    # out_mat shape: (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out_norm_fp16 = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(torch.float16)                      # fp16

    # --------------------------------------------------------------
    # 8️⃣ Apply output‑gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    # (B, N, N, H) * (B, N, N, H) → fp16
    out_norm_fp16 = out_norm_fp16 * out_gate.view(B, N, N, H)

    final_half = F.linear(
        out_norm_fp16.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # (R, dim) fp16

    # Cast to float32 expected by the model downstream
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
