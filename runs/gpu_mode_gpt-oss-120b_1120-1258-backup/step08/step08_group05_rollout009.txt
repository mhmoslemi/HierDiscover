"""
custom_kernel (outgoing TriMul)

Fast forward implementation of the “outgoing” TriMul operator from AlphaFold‑3.
The algorithm follows the reference PyTorch module but moves the two
LayerNorms to Triton kernels and performs all heavy linear work in FP16.
Steps
-----
1️⃣ Row‑wise LayerNorm over the input tensor (float32 → fp16) – Triton.
2️⃣ Fuse the four projection/gate Linear layers (left, right, left‑gate,
   right‑gate) into a single GEMM (FP16) and split the result.
3️⃣ (Optional) apply the pair‑wise mask.
4️⃣ Compute the N³ update  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched half‑precision mat‑mul: reshape to (B·H, N, N) and call
   `torch.matmul`.  cuBLAS Tensor‑Core kernels handle the O(N³·H) work.
5️⃣ Second LayerNorm over the hidden dimension – Triton.
6️⃣ Apply the output‑gate, project back to the original channel dimension
   with a final FP16 GEMM, and cast the result to float32.

The kernel works for batch size ≤ 2, any sequence length in the test suite,
and both masked / un‑masked variants.  Only the forward pass is required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 -> fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – gamma
    bias_ptr,       # const   * [dim]      (float32) – beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute per‑row mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)           # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized output (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]              # fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def _layernorm_triton(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    block_m: int = 256,
    block_d: int = 32,
) -> torch.Tensor:
    """
    Row‑wise LayerNorm for a 2‑D fp32 tensor ``x`` of shape [rows, dim].
    Returns a fp16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored
                         if config["nomask"] is True
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and optional
                         "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        eps=eps,
    )                               # fp16, [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear layers (all FP16)
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection / gate weight matrices (each [H, dim])
    proj_gate_w = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                # [4·H, dim]

    # (R, dim) @ (dim, 4·H)ᵀ  → (R, 4·H)
    proj_gate = F.linear(x_norm, proj_gate_w)   # FP16

    # split the fused result
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # separate linear for the output‑gate (H, dim)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )                                                   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates and element‑wise products
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [R, 1] broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()          # [R, H]

    left = left_proj_raw * left_gate            # [R, H] FP16
    right = right_proj_raw * right_gate         # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched FP16 GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)                # [B, N, N, H]
    right = right.view(B, N, N, H)              # [B, N, N, H]

    # Bring the hidden dimension to the second axis for cuBLAS
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)
    )   # FP16, shape [B, H, N, N]

    # Back to the original layout
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
    )                                        # fp16, [R, H]
    out_norm = out_norm.view(B, N, N, H)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Output gate, final projection back to ``dim`` and cast to FP32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)    # FP16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
