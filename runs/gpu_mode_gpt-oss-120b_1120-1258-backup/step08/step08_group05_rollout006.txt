"""
TriMul (outgoing) – fast forward implementation.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} (float32) and computes

    out[b,i,j,c] = Σ_k  ( (x·W_left  + bias)·σ(x·W_lgate) )_{i,k,c}
                 * ( (x·W_right + bias)·σ(x·W_rgate) )_{j,k,c}
                 after a first LayerNorm over C,
                 an optional pair‑wise mask,
                 a second LayerNorm over the hidden dimension H,
                 and a final projection back to C.

Key implementation ideas
------------------------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) is performed by a custom Triton kernel.
2️⃣ The four linear projections (left, right, left‑gate, right‑gate) are fused
   into a single FP16 GEMM (weight matrix of shape (4·H, C)).
3️⃣ The optional mask is applied in FP16, then the sigmoid gates are computed
   in‑place.
4️⃣ The N³ multiplicative update is a batched half‑precision matrix‑multiply:
   for every head h we compute   L_h @ R_hᵀ   where L_h,R_h ∈ ℝ^{B·H×N×N}.
   This maps to cuBLAS Tensor‑Core kernels (torch.matmul).
5️⃣ The second LayerNorm (over H) is another Triton kernel.
6️⃣ Finally the output‑gate is applied and a FP16‑to‑FP32 projection yields
   the result.

All heavy arithmetic is in FP16; the two LayerNorm reductions stay in FP32.
The implementation works for any batch ≤ 2, any sequence length in the test
suite, and both masked / un‑masked variants.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# --------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,        # const   * [rows, dim] (float32)
    y_ptr,        # mutable * [rows, dim] (float16)
    weight_ptr,   # const   * [dim]      (float32) – γ
    bias_ptr,     # const   * [dim]      (float32) – β
    rows,         # int32   (runtime)
    dim,          # int32   (runtime)
    eps,          # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # number of rows per program
    BLOCK_D: tl.constexpr,   # number of columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a tile of shape [BLOCK_M, BLOCK_D] (FP32)
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)            # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)     # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1/√(var+ε)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized + affine output (FP16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D FP32 tensor ``x`` of shape [rows, dim].
    Returns an FP16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,          # enough for the small reduction kernels
    )
    return y


# --------------------------------------------------------------
# TriMul forward (outgoing) – the public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float). Ignored
                         if config["nomask"] == True.
        - weights_dict : mapping of all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim".
                         Optional key "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ
        weights["norm.bias"],     # β
        eps=eps,
    )                           # FP16 tensor [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections (left/right + gates)
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices (each [H, dim])
    proj_gate_w = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                  # [4·H, dim]

    # (R, dim) @ (dim, 4·H).T  → (R, 4·H)  (FP16)
    proj_gate = F.linear(x_norm, proj_gate_w)

    # Split the fused result
    left_proj_raw  = proj_gate[:, 0:H]                     # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]                 # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]             # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]             # [R, H]

    # Separate weight for output gate (H, dim)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )                                                   # [R, H] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] (FP16) for broadcasting over H
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid on the gate tensors
    left_gate = left_gate_raw.sigmoid_()          # [R, H] FP16
    right_gate = right_gate_raw.sigmoid_()        # [R, H] FP16
    out_gate = out_gate_raw.sigmoid_()            # [R, H] FP16

    # Gated projections
    left = left_proj_raw * left_gate              # [R, H] FP16
    right = right_proj_raw * right_gate           # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Bring H to the batch dimension: [B, H, N, N] → (B·H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)               # transpose the “k” axis
    )                                            # FP16

    # Back to [B, N, N, H]
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
    )                                   # FP16 [R, H]
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # FP16 element‑wise

    final_half = F.linear(
        out_norm.view(total_rows, H),                # [R, H]
        weights["to_out.weight"].to(torch.float16), # [dim, H]
    )                                                # FP16 [R, dim]

    # Return float32 as required by the spec
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
