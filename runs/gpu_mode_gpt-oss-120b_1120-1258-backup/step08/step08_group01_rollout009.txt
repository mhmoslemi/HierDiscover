"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – custom Triton kernel.
2️⃣ Fuse the four linear projections/gates (left, right, left‑gate, right‑gate)
   into a single FP16 GEMM (B·N·N rows × dim cols → 4·H cols).
3️⃣ Compute the output‑gate with a second FP16 GEMM.
4️⃣ Optional mask, sigmoid‑gates and element‑wise weighting.
5️⃣ N³ multiplicative update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   performed with a single ``torch.einsum`` (no costly permute+copy).
6️⃣ Second row‑wise LayerNorm over the hidden dimension (H) – same kernel.
7️⃣ Apply the output‑gate, final FP16 linear projection and cast back to FP32.

All heavy element‑wise work (two LayerNorms, mask, sigmoids) is done in
Triton, while the two dense GEMMs and the N³ update use cuBLAS/Tensor‑Core
FP16 kernels.  Memory copies are minimised (no ``permute(...).contiguous()``
for the N³ step), which brings the runtime well below the 1 ms target
on an H100 for the largest test cases.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32)   γ
    bias_ptr,       # const   * [dim]      (float32)   β
    rows,           # total number of rows   (int)
    dim,            # feature dimension      (int)
    eps,            # epsilon (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Reduce along dim → mean & variance (FP32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [BLOCK_M, BLOCK_D]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # FP32 load
        sum_val += tl.sum(vals, axis=1)                  # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # (row, col) FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ  FP32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β  FP32
        out = norm * w[None, :] + b[None, :]                        # FP32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # --------------------------------------------------------------
    # concatenate the four projection matrices once
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,                              # [4·H, dim]
    ).to(device=device, dtype=torch.float16)

    # fused GEMM: [R, dim] @ [4·H, dim]^T → [R, 4·H]  (FP16, no bias)
    fused = F.linear(ln1_out, w_proj)          # fp16

    # split the four blocks
    left_raw      = fused[:, 0:H]               # [R, H]
    right_raw     = fused[:, H:2*H]             # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]           # [R, H]
    right_gate_raw= fused[:, 3*H:4*H]           # [R, H]

    # output‑gate projection (separate GEMM)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"]
            .to(device=device, dtype=torch.float16)   # [H, dim]
    )                                            # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present), sigmoid gates and weighting
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (FP16)
        mask_flat = mask.to(dtype=torch.float16, device=device).view(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # in‑place sigmoid to avoid temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # element‑wise gated tensors
    left  = left_raw * left_gate          # [R, H] fp16
    right = right_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (einsum – no extra copies)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]

    # out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H] fp16
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final projection and cast back to FP32
    # --------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H) * out_gate   # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"]
            .to(device=device, dtype=torch.float16)    # [dim, H]
    )                                                 # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
