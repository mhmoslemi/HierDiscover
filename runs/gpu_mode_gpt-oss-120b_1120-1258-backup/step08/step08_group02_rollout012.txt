"""
custom_kernel: high‑performance forward of the *outgoing* TriMul operator.

Algorithm (FP16‑friendly)
-------------------------
1️⃣  Row‑wise LayerNorm over the channel dimension (dim) – use the
    highly‑optimized torch.nn.functional.layer_norm (GPU kernel).

2️⃣  Fuse the four Linear projections/gates (left/right proj + left/right
    gate) into a single GEMM:
        X_norm @ W_fused.T  →  [R, 4·H]      (R = B·N·N, H = hidden_dim)
    Split the result into left_proj, right_proj, left_gate, right_gate.

3️⃣  Compute the output‑gate with a separate Linear (still FP16).

4️⃣  Apply the optional binary mask to the two projected tensors,
    then apply the three sigmoid gates (in‑place).

5️⃣  Reshape left/right → (B·H, N, N) and perform a batched GEMM:
        out_mat = left_mat @ right_matᵀ      (B·H independent GEMMs)

6️⃣  Fuse the second LayerNorm (over the hidden dimension H) **and**
    the element‑wise multiplication by out_gate in a Triton kernel
    (layernorm_gate_kernel).  The kernel computes per‑row mean/variance,
    normalises, applies γ/β and multiplies by the pre‑computed out_gate.

7️⃣  Final Linear projection back to the original channel dimension
    (FP16 GEMM) and cast the result to float32.

All heavy linear algebra (including the N³ multiplicative update) uses
the cuBLAS‑accelerated Torch kernels; the only Triton‑written parts are the
row‑wise LayerNorm + gating fusion (step 6) and the optional custom LayerNorm
(if preferred).  This design keeps the critical N³ matmul in highly‑optimised
cuBLAS while still satisfying the “must use a kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm (over hidden dim H) + element‑wise gate.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim] (float16 stored)
    gate_ptr,       # const   * [total_rows, dim] (float16)
    y_ptr,          # mutable * [total_rows, dim] (float16)  ← (norm*γ+β)*gate
    weight_ptr,     # const   * [dim]            (float32) γ
    bias_ptr,       # const   * [dim]            (float32) β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime, hidden_dim)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # Load as float16 → up‑cast to float32 for accumulation
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Normalise, apply γ/β and multiply by gate (all FP32, store FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # raw activation
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # gate (pre‑sigmoided out_gate)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = (norm * w[None, :] + b[None, :]) * g_vals        # FP32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

            - input_tensor : torch.Tensor         [B, N, N, dim]  (float32)
            - mask_tensor  : torch.Tensor or None [B, N, N]       (bool/float)
            - weights_dict : mapping with all Linear/LayerNorm parameters
            - config_dict  : {"dim": int, "hidden_dim": int,
                              "nomask": bool (optional), "eps": float (optional)}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------------- unpack ---------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = config.get("eps", 1e-5)
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over the channel dimension (uses torch kernel)
    # ------------------------------------------------------------------
    # x_norm : float32 → cast to float16 for the rest of the pipeline
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=weights["norm.weight"],
                          bias=weights["norm.bias"],
                          eps=eps).to(torch.float16)          # [B,N,N,dim]

    # Flatten to (R, dim) for the fused projection
    x_norm_flat = x_norm.view(total_rows, dim).contiguous()          # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # ------------------------------------------------------------------
    # Build a single weight matrix of shape [4·H, dim] in half‑precision
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                            # [4·H, dim]

    # One fused GEMM: (R, dim) @ (4·H, dim)ᵀ → (R, 4·H)
    fused = F.linear(x_norm_flat, w_fused)                         # FP16

    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]

    # Separate output‑gate
    out_gate_raw = F.linear(
        x_norm_flat,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) and sigmoid gates
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)  # broadcast
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()      # in‑place, FP16
    right_gate = right_gate_raw.sigmoid_()     # FP16
    out_gate   = out_gate_raw.sigmoid_()       # FP16, shape [R, H]

    left  = left_raw  * left_gate   # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Collapse batch & head dimensions: (B·H, N, N)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm (over hidden dim H) + multiplication by out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()                   # [R, H] FP16
    out_gate_flat = out_gate                                          # already [R, H] FP16

    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # Triton launch configuration
    BLOCK_M = 256        # rows per program – tuned for H100
    BLOCK_D = 64         # columns per iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,                     # more warps → better utilisation on H100
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to original dim (FP16 GEMM) → FP32 output
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
