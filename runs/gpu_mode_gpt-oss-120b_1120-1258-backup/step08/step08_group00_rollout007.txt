def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B,N,_,_ = x.shape
    R = B*N*N
    H = hidden_dim
    # Convert weight to half for layernorm and linear
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    # 1) LayerNorm input (fp32) - we do it in fp32 and cast to half
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    x_norm = x_norm.to(torch.float16)

    # 2) fused linear: concat weights
    w_fused = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)  # shape (5*H, dim)
    proj = F.linear(x_norm, w_fused)  # (R,5*H) fp16

    # 3) split and apply mask/gates using Triton kernel
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy
    else:
        HAS_MASK = 1
        mask_fp16 = mask.to(dtype=torch.float16).reshape(R)  # (R,)
        mask_ptr = mask_fp16

    BLOCK_R = 256
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # release proj
    del proj

    # 4) batched gemm
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # 5) second layernorm on hidden
    out_norm = F.layer_norm(out, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out_norm = out_norm.to(torch.float16)  # ensure half

    # multiply by out_gate
    out_norm = out_norm * out_gate

    # 6) final linear
    final = F.linear(out_norm.view(R, H), weights["to_out.weight"].to(torch.float16))
    final = final.view(B, N, N, dim).to(torch.float32)
    return final

"""
Outgoing TriMul (Tri‑multiplicative update) – high‑performance forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – performed with
    torch.nn.functional.layer_norm (float32) and cast to fp16.
2️⃣  Fused linear projection: concatenate the five weight matrices
    (left_proj, right_proj, left_gate, right_gate, out_gate) → shape
    (5·hidden_dim, dim).  One GEMM (R×dim) @ (5·H×dim)ᵀ produces a tensor
    (R, 5·H) in fp16.
3️⃣  Triton kernel `split_fuse_gate_kernel`:
        * splits the fused result into the three tensors *left*, *right* and
          *out_gate*,
        * applies the optional pair‑wise mask,
        * evaluates sigmoid gates (computed in fp32 for stability) and writes
          the gated results (fp16).
4️⃣  Batched matrix multiplication over the hidden dimension:
        (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N).
5️⃣  Second LayerNorm on the hidden‑dim tensor (fp16) using torch.nn.functional.
6️⃣  Element‑wise multiplication with the output gate and a final linear
    projection back to `dim` (fp16 → fp32 output).

Only the split‑and‑gate step is written in Triton; the remaining heavy
linear algebra (large GEMMs and BMMs) is delegated to cuBLAS/Tensor Cores,
which gives the best performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left  = (proj_left      * mask) * sigmoid(proj_left_gate)
        right = (proj_right     * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    All ops are fp16; sigmoid is evaluated in fp32 for numerical stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load the (optional) per‑row mask and broadcast to the hidden dim.
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in BLOCK_H chunks.
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]       # [R,H] offset for seg 0

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ----- sigmoid in fp32 -------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -------------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- write results ----------------------------------------------------
        left_dst_ptr   = left_ptr   + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr  = right_ptr  + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,   left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,  right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    eps          = 1e-5
    device       = x.device
    B, N, _, _   = x.shape
    R            = B * N * N                # total rows (B·N·N)
    H            = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    # --------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(x,
                               normalized_shape=(dim,),
                               weight=weights["norm.weight"],
                               bias=weights["norm.bias"],
                               eps=eps)                     # [B,N,N,dim] float32
    x_norm = x_norm_fp32.to(torch.float16)                     # fp16 for the GEMM

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (one large GEMM)
    # --------------------------------------------------------------
    # w_concat : (5*H, dim)  (fp16)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # (R, dim) @ (5*H, dim)^T  → (R, 5*H)
    proj = F.linear(x_norm, w_concat)               # fp16
    del x_norm, w_concat                             # free intermediate memory

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates (single Triton kernel)
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy
    else:
        HAS_MASK = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)                  # [R]
        mask_ptr = mask_fp16

    BLOCK_R = 256
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr                               # free

    # --------------------------------------------------------------
    # 4️⃣ Batched matrix multiplication over the hidden dimension
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] → [B, H, N, N] → (B*H, N, N)
    left_mat  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_mat.view(B * H, N, N)      # (B·H, N, N)
    right_mat = right_mat.view(B * H, N, N)     # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (fp16) + output‑gate scaling
    # --------------------------------------------------------------
    # torch.layer_norm expects weight/bias of same dtype as the input.
    to_out_norm_weight_h = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias_h   = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(out,
                            normalized_shape=(H,),
                            weight=to_out_norm_weight_h,
                            bias=to_out_norm_bias_h,
                            eps=eps)                     # fp16
    out_norm = out_norm * out_gate                         # element‑wise gate
    del out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(out_norm.view(R, H),
                          weights["to_out.weight"].to(torch.float16))   # fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)   # fp32 output

    return final
