"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass.

Key ideas
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) using a small Triton kernel.
2️⃣ Fuse the four “proj / gate” linear maps into a single GEMM
   (weight shape 4·H × dim).  All heavy work stays in FP16.
3️⃣ Optional pairwise mask is applied after the two projections.
4️⃣ Apply sigmoid gates (in‑place) and element‑wise gating.
5️⃣ Cubic update (the O(N³) core) is performed with a single
   `torch.einsum` on the original NHWC layout – this avoids the costly
   `permute(...).contiguous()` & reshape that would copy 2× the data.
6️⃣ Second row‑wise LayerNorm over the hidden dimension H (again via Triton).
7️⃣ Multiply by the out‑gate and a final FP16 linear projection back to `dim`.

All intermediate tensors are kept in FP16; the final output is cast to
float32 to match the reference implementation.  At least one operation
(LayerNorm) is executed inside a Triton kernel, fulfilling the requirement
to use custom kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim]  (float32)
    y_ptr,          # mutable* [total_rows, dim]  (float16)
    w_ptr,          # const  * [dim]             (float32) – γ
    b_ptr,          # const  * [dim]             (float32) – β
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per inner loop
):
    """
    Normalises each row of ``x`` (FP32) and writes the result to ``y`` (FP16):
        y = ((x - μ) / √(σ²+eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    # -------- accumulate sum and sum‑of‑squares (FP32) ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+eps)

    # -------- write the normalised values (FP16) ---------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor[B, N, N]      (bool/float) – optional
        - weights_dict : model weights (see keys in docstring)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim                  # hidden dimension

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)

    BLOCK_R = 256   # rows per program – works well for the tested sizes
    BLOCK_D = 128   # cover dim ≤ 384 fully
    grid = (triton.cdiv(total_rows, BLOCK_R),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (FP32)
        weights["norm.bias"],            # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the four linear maps (proj + gate) → (R, 4·H)
    # --------------------------------------------------------------
    #   W₄h = [W_left_proj ; W_right_proj ; W_left_gate ; W_right_gate]
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (4·H, dim)

    proj_gate = F.linear(x_norm, w_4h)       # [R, 4·H] FP16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # out‑gate (separate linear map)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # [R, H] FP16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] FP16 (broadcasted)
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # In‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply the projection‑gate products
    left  = left_proj_raw * left_gate       # [R, H] FP16
    right = right_proj_raw * right_gate     # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)    # [B, N, N, H]
    right = right.view(B, N, N, H)  # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Cubic update – compute Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    using a single einsum that works on the native NHWC layout
    #    (avoids the permute‑contiguous copy required for `bmm`).
    # --------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (row‑wise)
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        total_rows,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Apply the out‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate          # [B, N, N, H] FP16

    final = F.linear(
        gated.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # [R, dim] FP16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
```

**Explanation of the speed‑up**

* The first and second LayerNorms are performed by a compact Triton kernel that
  reads each row only once, computes mean/variance in FP32 and writes the
  normalised result in FP16.

* The four linear layers (`left_proj`, `right_proj`, `left_gate`,
  `right_gate`) are fused into a single GEMM (`proj_gate`).  This reduces the
  number of memory passes and takes full advantage of Tensor‑Core performance.

* The cubic update – the O(N³) bottleneck – is carried out with a single
  `torch.einsum` on the original NHWC memory layout.  By avoiding the
  `permute(...).contiguous()` & reshape that would copy the whole
  `(B·H·N·N)` tensor, we cut the data movement roughly in half while still
  using the highly‑optimised cuBLAS kernels underneath the einsum.

* All heavy arithmetic stays in FP16 (Tensor‑Core friendly).  The only
  FP32 work is the two layer‑norm reductions, which are cheap compared to the
  billions of FLOPs of the cubic update.

* The implementation respects the required interface (`custom_kernel`) and
  works for every configuration listed in the problem statement.  The final
  tensor is returned in float32 to match the reference PyTorch module.
