@triton.jit
def layernorm_gate_kernel(
    inp_ptr,  # [rows, H] fp16
    gate_ptr, # [rows, H] fp16
    out_ptr,  # [rows, H] fp16 (normalized * gate)
    weight_ptr, # [H] fp32 gamma
    bias_ptr,   # [H] fp32 beta
    rows: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # accumulate sum and sumsq over H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        ptr = inp_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: normalized, apply gamma/beta, gating
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        ptr = inp_ptr + offs_m[:, None] * H + offs_h[None, :]
        inp_vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        norm = (inp_vals - mean[:, None]) * rstd[:, None]

        # gamma beta
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)  # gamma
        b = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)    # beta
        norm = norm * w[None, :] + b[None, :]

        # gate
        gate_ptr_loc = gate_ptr + offs_m[:, None] * H + offs_h[None, :]
        gate_vals = tl.load(gate_ptr_loc,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)  # maybe convert to fp32 for stability?

        # Compute sigmoid of gate? Wait out_gate already passed through sigmoid in earlier kernel. So it's already sigmoid output (0-1). Actually earlier split_fuse_gate computes out_gate = sigmoid(out_gate_raw). So out_gate is already sigmoid; we just multiply.

        # So we don't need to apply sigmoid again.

        out_vals = norm * gate_vals

        out_ptr_loc = out_ptr + offs_m[:, None] * H + offs_h[None, :]
        tl.store(out_ptr_loc,
                 out_vals.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

"""
Optimized Triton‑based forward pass for the **outgoing** TriMul operator.

Algorithm (all tensors stay on the GPU):
1️⃣ Row‑wise LayerNorm of the input tensor (fp32 → fp16).
2️⃣ One fused Linear   X_norm @ W_fusedᵀ   (W_fused = [left, right,
   left_gate, right_gate, out_gate]) → (R, 5·H)   (fp16, Tensor‑cores).
3️⃣ Single Triton kernel:
   • splits the fused projection,
   • applies the optional pair‑wise mask,
   • computes sigmoid‑gates,
   • writes *left*, *right* and *out_gate* (all fp16).
4️⃣ Batched matrix‑multiply over the hidden dimension:
   out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ   (torch.bmm, cuBLAS).
5️⃣ Fused LayerNorm + output‑gate (one Triton kernel) → (R, H) fp16.
6️⃣ Final Linear projection back to `dim` (fp16 → fp32).

All heavy element‑wise work (LayerNorms, masking, sigmoid) is done in
custom Triton kernels; the two large GEMM/BMM operations use cuBLAS /
torch‑ops which already exploit H100 Tensor‑cores.  This reduces memory
traffic and kernel launch overhead compared to the naïve reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- compute mean & variance (fp32 accumulation) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values (γ·(x-μ)·rstd + β) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops in fp16 (sigmoid computed in fp32 for stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # load per‑row mask if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0)                     # [BLOCK_R]
        mask_val = mask_val[:, None]                         # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ----- layout: [left, right, left_gate, right_gate, out_gate] -----
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]
        left_raw_ptr     = proj_ptr + base
        right_raw_ptr    = proj_ptr + base + H
        left_gate_ptr    = proj_ptr + base + 2 * H
        right_gate_ptr   = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        # ----- load raw tensors -----
        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ----- sigmoid (computed in fp32) -----
        left_gate  = (1.0 / (1.0 + tl.exp(-(left_g_raw).to(tl.float32))).to(tl.float16))
        right_gate = (1.0 / (1.0 + tl.exp(-(right_g_raw).to(tl.float32))).to(tl.float16))
        out_gate   = (1.0 / (1.0 + tl.exp(-(out_g_raw).to(tl.float32))).to(tl.float16))

        # ----- apply mask, gates and store -----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        left_dst_ptr  = left_ptr  + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr = right_ptr + offs_r[:, None] * H + offs_h[None, :]
        outg_dst_ptr  = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,  left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(outg_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ LayerNorm + output‑gate (second LN fused with out_gate)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    inp_ptr,          # const   *[rows, H] (float16)
    gate_ptr,         # const   *[rows, H] (float16)   – already sigmoid
    out_ptr,          # mutable *[rows, H] (float16)   – LN(..) * gate
    weight_ptr,       # const   *[H]      (float32)   γ (to_out_norm)
    bias_ptr,         # const   *[H]      (float32)   β (to_out_norm)
    rows: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = inp_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, apply LN weight/bias, multiply by gate ----------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # input value
        inp = tl.load(inp_ptr + offs_m[:, None] * H + offs_h[None, :],
                      mask=mask_m[:, None] & mask_h[None, :],
                      other=0.0).to(tl.float32)

        # normalize
        norm = (inp - mean[:, None]) * rstd[:, None]

        # γ and β from to_out_norm
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate (already sigmoid, keep fp32 for stability)
        gate = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate   # still fp32

        tl.store(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor       [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor or None  [B, N, N]      (bool/float)
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # rows for the row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (fp32 -> fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 1024
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right & gates (5 * H outputs)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5*H, dim]

    proj = F.linear(x_norm, w_fused)          # (R, 5*H)  fp16
    del x_norm, w_fused                        # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left   = torch.empty(R, H, dtype=torch.float16, device=device)
    right  = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        has_mask = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)   # [R]
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → permute to [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_M2 = 1024
    BLOCK_H2 = 32
    grid_ln2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out_flat, out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
