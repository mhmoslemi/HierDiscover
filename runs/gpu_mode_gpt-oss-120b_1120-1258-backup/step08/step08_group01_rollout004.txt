"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style) implemented with Triton + PyTorch.

The kernel fuses the two LayerNorms (fp32 → fp16) in Triton while all
large matrix multiplications are delegated to cuBLAS (via torch.nn.functional.linear
and torch.bmm). Mixed‑precision (fp16) is used everywhere except the final
cast to fp32, matching the reference implementation.

Key steps
---------
1️⃣ First row‑wise LayerNorm on the flattened pair representation.
2️⃣ One fused Linear producing left/right projections and their two gates
   (4·H outputs) + a separate Linear for the output‑gate (H outputs).
3️⃣ Optional mask application, sigmoid on the three gates and element‑wise
   weighting.
4️⃣ N³ multiplicative update: for each head h, compute
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   (batched bmm).
5️⃣ Second LayerNorm on the hidden dimension.
6️⃣ Apply the output‑gate, final Linear back to dim and cast to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (fp32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_tiles = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, num_tiles):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, num_tiles):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β fp32
        out = norm * w[None, :] + b[None, :]                    # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the following keys (all float32):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : dict with keys
            "dim" (int), "hidden_dim" (int), optional "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack & basic checks
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N                       # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over dim (fp32→fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()        # [R, dim]  float32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    # Block sizes – chosen to cover the typical dim (≤384) efficiently
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gates (mixed‑precision fused GEMM)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices for left/right & their gates
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
        ],
        dim=0,                              # [4·H, dim]
    ).to(device=device, dtype=torch.float16)

    # Fused GEMM: (R, dim) @ (4·H, dim).T -> (R, 4·H)
    fused = F.linear(ln1_out, w_proj)   # fp16

    left_raw      = fused[:, 0:H]                # [R, H]
    right_raw     = fused[:, H:2*H]              # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]           # [R, H]

    # Output‑gate (stand‑alone linear)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise weighting
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [R, 1] (float16) for broadcasting
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # in‑place sigmoid to save memory
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H] fp16
    right = right_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update via batched GEMM
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)       # [B, N, N, H]
    right = right.view(B, N, N, H)      # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)

    # Rearrange to [B, H, N, N] then collapse batch & head axes
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # [B·H, N, N]

    # Batched matrix multiplication (Tensor‑Core)
    # out_mat[b*h, i, j] = Σₖ left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Restore original layout -> [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H (fp32→fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final projection and cast back to fp32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)          # fp16
    out_norm = out_norm * out_gate               # element‑wise fp16 (gate)

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # float32 output
    return final
