def _apply_gate_and_mask_kernel(
    left_raw_ptr, right_raw_ptr, left_gate_ptr, right_gate_ptr,
    out_gate_ptr,
    mask_ptr,  # or zero if nomask
    out_left_ptr, out_right_ptr, out_gate_out_ptr,
    rows, H, nomask: tl.constexpr, BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # load
    left_raw = tl.load(left_raw_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :],
                       mask=mask_m[:, None], other=0.0)
    right_raw = tl.load(...)
    left_gate_raw = tl.load(...)
    right_gate_raw = tl.load(...)
    out_gate_raw = tl.load(...)
    # mask if needed
    if not nomask:
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0)
        left_raw = left_raw * mask_val[:, None]
        right_raw = right_raw * mask_val[:, None]

    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw))

    left = left_raw * left_gate
    right = right_raw * right_gate

    tl.store(out_left_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :],
             left, mask=mask_m[:, None])
    tl.store(out_right_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :],
             right, mask=mask_m[:, None])
    tl.store(out_gate_out_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :],
             out_gate, mask=mask_m[:, None])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # unpack
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    B, N, _, _ = input_tensor.shape
    R = B * N * N

    # flatten input
    x_flat = input_tensor.reshape(R, dim)   # view

    # layernorm 1 -> fp16
    x_norm = _layernorm_triton(x_flat,
                               weights["norm.weight"], weights["norm.bias"], eps)
    # fused proj+gate weight
    w_proj = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)

    proj = F.linear(x_norm, w_proj)  # [R, 5*H] fp16

    H = hidden_dim
    left_raw = proj[:, 0:H]
    right_raw = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    out_gate_raw = proj[:, 4*H:5*H]

    # allocate outputs
    left = torch.empty_like(left_raw)
    right = torch.empty_like(right_raw)
    out_gate = torch.empty_like(out_gate_raw)

    if not nomask:
        mask_flat = mask_tensor.reshape(R).to(torch.float16)
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)

    # apply gates and optional mask via Triton
    _apply_gate_mask_kernel[(triton.cdiv(R, 128),)](
        left_raw, right_raw,
        left_gate_raw, right_gate_raw,
        out_gate_raw,
        mask_flat,
        left, right, out_gate,
        R, H,
        BLOCK_M=128,
        BLOCK_H=32,
        NOMASK=int(nomask),
        num_warps=4,
    )

    # reshape for batched GEMM
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    left_bh = left.permute(0, 3, 1, 2).contiguous()
    right_bh = right.permute(0, 3, 1, 2).contiguous()

    # batched matmul
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))
    out = out_bh.permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # second layernorm
    out_flat = out.view(R, H).contiguous()
    out_norm = _layernorm_triton(out_flat,
                                 weights["to_out_norm.weight"],
                                 weights["to_out_norm.bias"], eps)

    # Apply out_gate and final linear
    # reshape out_gate to [B,N,N,H]
    out_gate = out_gate.view(B, N, N, H)

    out_norm = out_norm.view(B, N, N, H) * out_gate

    final = F.linear(out_norm.view(R, H), weights["to_out.weight"].to(torch.float16))
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the [B·N·N, C] view of the input (FP32 → FP16)
   – Triton kernel (`_layernorm_kernel`).

2️⃣ Fuse the four linear projections (left/right, left‑gate/right‑gate) **and**

   `proj = (x_norm @ W_fused.T)`  where `W_fused` has shape (5·H, C).

3️⃣ Convert the raw projection results into the gated tensors
   `left, right, out_gate` (FP16) while optionally applying the pairwise mask.

4️⃣ Reshape `left` / `right` to [B, H, N, N] and compute the expensive

5️⃣ Second LayerNorm on the hidden dimension (FP32 → FP16) – Triton.

6️⃣ Apply the output‑gate, project back to the original channel dimension

    y_ptr,          # mutable  * [rows, dim] (float16)
    weight_ptr,     # const    * [dim]      (float32) γ
    bias_ptr,       # const    * [dim]      (float32) β
    rows,           # int32    (runtime)
    dim,            # int32    (runtime)
    eps,            # float32  (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----------------------------------------------------------
    #   a) compute per‑row mean & variance
    # ----------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # stride = dim
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                               # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ----------------------------------------------------------
    #   b) write normalized output (fp16)
    # ----------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                         # fp32
        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm.
    Input  : [rows, dim]  (float32)
    Output : [rows, dim]  (float16)
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# --------------------------------------------------------------
# 3️⃣ Gate + optional mask (fused kernel)
# --------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_raw_ptr, right_raw_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_raw_ptr,
    mask_ptr,                     # [rows] (float16) – ignored if NOMASK==1
    out_left_ptr, out_right_ptr, out_gate_ptr,
    rows, H,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, NOMASK: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    num_h_tiles = tl.cdiv(H, BLOCK_H)

    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # raw tensors
        left_raw = tl.load(left_raw_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)                     # fp16
        right_raw = tl.load(right_raw_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(left_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)
        right_gate_raw = tl.load(right_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)
        out_gate_raw = tl.load(out_gate_raw_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)

        # optional mask (pairwise mask is broadcast along H)
        if NOMASK == 0:
            mask_val = tl.load(mask_ptr + offs_m,
                               mask=mask_m,
                               other=0.0)                # fp16, shape [BLOCK_M]
            left_raw  = left_raw  * mask_val[:, None]
            right_raw = right_raw * mask_val[:, None]

        # sigmoid (σ) = 1 / (1 + exp(-x))
        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # gated values
        left  = left_raw  * left_gate
        right = right_raw * right_gate

        # write results
        tl.store(out_left_ptr  + offs_m[:, None] * H + offs_h[None, :],
                 left,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_right_ptr + offs_m[:, None] * H + offs_h[None, :],
                 right,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_m[:, None] * H + offs_h[None, :],
                 out_gate,
                 mask=mask_m[:, None] & mask_h[None, :])


# --------------------------------------------------------------
#  Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping of all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    rows = B * N * N                       # total pairwise rows

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = inp.reshape(rows, dim)        # view, no copy
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        block_m=256,
        block_d=32,
    )                                      # [rows, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ fused projection (left, right, gate, out_gate) – one GEMM
    # ------------------------------------------------------------------
    # concatenate weight matrices: each is [hidden_dim, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, w_fused)        # [rows, 5*H] fp16

    H = hidden_dim
    left_raw       = proj[:, 0:H]
    right_raw      = proj[:, H:2*H]
    left_gate_raw  = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    out_gate_raw   = proj[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣ apply mask (if any) + σ‑gating  → left, right, out_gate
    # ------------------------------------------------------------------
    left  = torch.empty_like(left_raw)
    right = torch.empty_like(right_raw)
    out_gate = torch.empty_like(out_gate_raw)

    if not nomask:
        mask_flat = mask.reshape(rows).to(torch.float16)   # [rows]
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)

    # launch gate+mask kernel
    grid = (triton.cdiv(rows, 128),)           # BLOCK_M = 128 rows per program
    _gate_mask_kernel[grid](
        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        out_gate_raw,
        mask_flat,
        left,
        right,
        out_gate,
        rows,
        H,
        BLOCK_M=128,
        BLOCK_H=32,            # process 32 hidden channels per tile
        NOMASK=int(nomask),   # 1 → skip mask load
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 4️⃣ N³ update  (batched FP16 GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)           # [B, N, N, H]
    right = right.view(B, N, N, H)          # [B, N, N, H]

    # make them contiguous in (B, H, N, N) layout for cuBLAS
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))  # fp16
    out = out_bh.permute(0, 2, 3, 1).contiguous()               # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out_flat = out.view(rows, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        block_m=256,
        block_d=32,
    )                                               # [rows, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ gating + final projection back to `dim`
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)           # broadcastable shape
    out_norm = out_norm.view(B, N, N, H) * out_gate      # fp16 element‑wise

    final = F.linear(
        out_norm.view(rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                # [rows, dim] fp16

    final = final.view(B, N, N, dim).to(torch.float32)
    return final
