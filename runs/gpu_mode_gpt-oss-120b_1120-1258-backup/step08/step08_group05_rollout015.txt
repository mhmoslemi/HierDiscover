def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    B,N,_,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    eps = 1e-5
    nomask = config.get('nomask', True)
    total_rows = B * N * N
    # 1) layernorm on input x -> x_norm (fp16)
    # flatten to (total_rows, dim)
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = _layernorm_triton(x_flat, weights['norm.weight'], weights['norm.bias'], eps)
    # 2) compute fused projections
    # Concatenate weight matrices for left_proj, right_proj, left_gate, right_gate
    # all of shape [hidden_dim, dim] (torch.nn.Linear weight shape is [out_features, in_features])
    # weights stored as weight matrix (out_features, in_features)
    W_concat = torch.cat([weights['left_proj.weight'],
                         weights['right_proj.weight'],
                         weights['left_gate.weight'],
                         weights['right_gate.weight']], dim=0).to(torch.float16)
    proj_gate = F.linear(x_norm, W_concat)  # shape [R, 4*H]
    # split
    H = hidden_dim
    left_raw = proj_gate[:, :H]
    right_raw = proj_gate[:, H:2*H]
    left_gate_raw = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]
    # out_gate
    out_gate_raw = F.linear(x_norm, weights['out_gate.weight'].to(torch.float16))  # R, H
    
    # mask if needed
    if not nomask:
        mask = mask_tensor.to(torch.float16)
        mask = mask.reshape(total_rows, 1)
        left_raw = left_raw * mask
        right_raw = right_raw * mask
    
    # gate sigmoid
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()
    
    left = left_raw * left_gate   # shape [R, H] fp16
    right = right_raw * right_gate # [R, H] fp16
    
    # reshape left/right to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    
    # compute out = sum_k left[..., i, k, h] * right[..., j, k, h]
    # Use custom batched GEMM kernel
    out = batched_triton_gemm(left, right)  # shape [B, H, N, N] maybe
    # Actually we want output shape [B,N,N,H]. Let's compute out of shape [B,N,N,H] directly.
    
    # We'll compute out_perm = out (B,H,N,N) then permute back:
    out_BHN = out  # if we produce (B, H, N, N)
    out = out_BHN.permute(0,2,3,1).contiguous()  # shape [B,N,N,H]
    
    # 5) second layernorm on out
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = _layernorm_triton(out_flat, weights['to_out_norm.weight'], weights['to_out_norm.bias'], eps)
    out_norm = out_norm.view(B, N, N, H)
    # apply output gate
    out_norm = out_norm * out_gate.view(B, N, N, H)
    # final linear
    final = F.linear(out_norm.view(total_rows, H), weights['to_out.weight'].to(torch.float16))
    final = final.view(B, N, N, dim).to(torch.float32)
    
    return final

"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The algorithm follows the reference PyTorch implementation
but moves the two LayerNorms and the expensive N³ multiplicative update
into custom Triton kernels.  All heavy linear work is performed in FP16
with float‑32 accumulation, while the final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32 or fp16)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – γ
    bias_ptr,       # const   * [dim]      (float32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute per‑row mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # promote to fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized output in fp16
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]      # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim].
    Returns an fp16 tensor of the same shape."""
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ multiplicative update
# out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    # stride information for A (left)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    # stride information for B (right)
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    # stride information for C (output)
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    N,                # sequence length (rows / cols)
    H,                # hidden_dim (number of heads)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (batch*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    pid_bh = tl.program_id(0)            # combined batch * head index
    pid_m  = tl.program_id(1)            # block row
    pid_n  = tl.program_id(2)            # block col

    # de‑compose batch and head
    b = pid_bh // H
    h = pid_bh % H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the summed axis)
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: left[b,i,k,h]  → shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + b * stride_a_batch
            + h * stride_a_h
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile (transposed): right[b,j,k,h] accessed as B_T[k,j]
        #   -> rows = k (stride_b_k), cols = j (stride_b_j)
        b_ptrs = (
            B_ptr
            + b * stride_b_batch
            + h * stride_b_h
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_j
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # store result in C: out[b,i,j,h]
    c_ptrs = (
        C_ptr
        + b * stride_c_batch
        + h * stride_c_h
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : mapping of all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()     # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        eps=eps,
    )                                                       # fp16, [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear layers (half‑precision)
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection / gate weight matrices (each [H, dim])
    proj_gate_weight = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                    # [4·H, dim]

    # (R, dim) @ (dim, 4·H).T  → (R, 4·H)
    proj_gate = F.linear(x_norm, proj_gate_weight)          # fp16

    # split the fused result
    left_proj_raw  = proj_gate[:, :H]                       # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]                  # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]              # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]              # [R, H]

    # out‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),      # [H, dim]
    )                                                       # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gating + element‑wise product
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask
        right_proj_raw = right_proj_raw * mask

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate       # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update – custom Triton batched GEMM
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)                 # [B, N, N, H]
    right = right.view(B, N, N, H)               # [B, N, N, H]

    out = torch.empty_like(left)                 # fp16, same strides as left

    # fetch strides (in elements) for the three tensors
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # launch Triton GEMM: grid = (B*H, ceil(N/BM), ceil(N/BN))
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    _tri_mul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        N,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # out now has shape [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()  # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
    )                                                   # fp16, [R, H]
    out_norm = out_norm.view(B, N, N, H)               # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)    # fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),      # [dim, H]
    )                                                   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
