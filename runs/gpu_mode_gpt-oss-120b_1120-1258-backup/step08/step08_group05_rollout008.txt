"""
Optimized forward of the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the input tensor.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate) into a single
   half‑precision GEMM and split the result.
3️⃣ (Optional) apply the pair‑wise mask, compute gate sigmoids and element‑wise multiply.
4️⃣ Compute the cubic update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched half‑precision mat‑mul (cuBLAS Tensor‑Core kernel).
5️⃣ Row‑wise LayerNorm over the hidden dimension.
6️⃣ Apply the output gate, project back to the original channel dimension and return
   a FP32 tensor.

All heavy linear work is performed in FP16; only the two LayerNorm reductions run in
FP32.  The implementation works for any batch size ≤ 2, any sequence length in the
test suite and both masked / un‑masked variants.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (output in FP16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (any dtype, we cast to fp32)
    y_ptr,          # mutable * [rows, dim] (fp16)
    weight_ptr,     # const   * [dim]      (fp32) – γ
    bias_ptr,       # const   * [dim]      (fp32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute per‑row mean & variance (fp32 accumulators)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # Load as whatever dtype the pointer has, then cast to fp32
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized output in fp16
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]    # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β (fp32)
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim].
    Returns a FP16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if

    # ------------------------------------------------------------------

    hidden_dim = config["hidden_dim"]

        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                  # [4·H, dim]

    # (R, dim) @ (dim, 4·H).T → (R, 4·H)
    proj_gate = F.linear(x_norm, proj_gate_w)  # FP16

    # Split fused result
    left_proj_raw  = proj_gate[:, 0:H]                     # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]                 # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]             # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]             # [R, H]

    # Separate linear for the output‑gate (H, dim)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),  # [H, dim]
    )                                                    # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any) + sigmoid gating + element‑wise product
    # ------------------------------------------------------------------
    if not nomask:

        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate      # [R, H] FP16
    right = right_proj_raw * right_gate   # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]

    # Layout required by cuBLAS: (B, H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)
    )                                                   # FP16, [B, H, N, N]

    # Back to original layout
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → FP16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
    )                                             # FP16, [R, H]
    out_norm = out_norm.view(B, N, N, H)          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # FP16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                 # FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
