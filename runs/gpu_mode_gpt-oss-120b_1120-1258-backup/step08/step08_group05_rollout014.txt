@triton.jit
def _layernorm_gate_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr, gate_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute sum and sum of squares
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals_f16 = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals = vals_f16.to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized * gate
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals_f16 = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals = vals_f16.to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :], other=0.0)

        out_val = (norm * w[None, :] + b[None, :]) * gate_val

        tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_val.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

"""
Optimized Triton implementation of the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16) – Triton kernel.
2️⃣ Fuse the four projection Linear layers **and** the output‑gate Linear into a single
   half‑precision GEMM.  Split the result into left/right projections and the three
   sigmoid‑gates.
3️⃣ (Optional) apply the pair‑wise mask, then multiply the projections by their
   gates.
4️⃣ Batched half‑precision matrix multiplication:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented with a single `torch.matmul` on tensors of shape (B·H, N, N) – this
   maps to a cuBLAS Tensor‑Core GEMM and avoids costly data copies.
5️⃣ Second LayerNorm over the hidden dimension **and** element‑wise multiplication
   by the output‑gate – fused into one Triton kernel.
6️⃣ Final half‑precision Linear projection back to the original channel dimension,
   followed by a cast to float32.

All heavy linear work is performed in FP16; the only FP32 work is the two
row‑wise LayerNorm reductions.  The implementation works for the full test
suite (B∈{1,2}, N up to 1024, dim ∈{128,384}, hidden_dim = 128) and for both
masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – γ
    bias_ptr,       # const   * [dim]      (float32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute per‑row mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized output in fp16
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D fp32 tensor ``x`` of shape [rows, dim].
    Returns an fp16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm + element‑wise gate (fp16 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_gate_kernel(
    x_ptr,          # const   * [rows, dim] (float16)
    out_ptr,        # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32)
    bias_ptr,       # const   * [dim]      (float32)
    gate_ptr,       # const   * [rows, dim] (float16)
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean / variance ------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals_f16 = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        vals = vals_f16.to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized value, apply weight/bias and the gate -----------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # original value
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals_f16 = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        vals = vals_f16.to(tl.float32)

        # normalization
        norm = (vals - mean[:, None]) * rstd[:, None]      # (x‑μ)/σ

        # weight & bias (float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        # gate value (float16 → float32)
        gate_val_f16 = tl.load(
            gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        gate_val = gate_val_f16.to(tl.float32)

        out_val = (norm * w[None, :] + b[None, :]) * gate_val   # fp32

        tl.store(
            out_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_val.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def _layernorm_gate_triton(x: torch.Tensor,
                           gate: torch.Tensor,
                           weight: torch.Tensor,
                           bias: torch.Tensor,
                           eps: float = 1e-5,
                           block_m: int = 256,
                           block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm + element‑wise gate.
    ``x`` and ``gate`` must have identical shape [rows, dim] and dtype fp16.
    Returns an fp16 tensor of shape [rows, dim].
    """
    rows, dim = x.shape
    out = torch.empty_like(x)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_gate_kernel[grid](
        x,
        out,
        weight,
        bias,
        gate,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# TriMul forward (outgoing) – public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if
                         config["nomask"] == True
        - weights_dict : mapping of all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        eps=eps,
        block_m=256,
        block_d=32,
    )                                                               # fp16 [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + all gates (single GEMM)
    # ------------------------------------------------------------------
    # concatenate the four projection matrices and the output‑gate matrix
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                            # [5·H, dim]

    # (R, dim) @ (dim, 5·H).T  -> (R, 5·H)
    proj_all = F.linear(x_norm, fused_weight)                       # fp16

    # split
    H = hidden_dim
    left_proj_raw  = proj_all[:, 0 * H : 1 * H]                     # [R, H]
    right_proj_raw = proj_all[:, 1 * H : 2 * H]                     # [R, H]
    left_gate_raw  = proj_all[:, 2 * H : 3 * H]                     # [R, H]
    right_gate_raw = proj_all[:, 3 * H : 4 * H]                     # [R, H]
    out_gate_raw   = proj_all[:, 4 * H : 5 * H]                     # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any) + sigmoid gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcast over the hidden dimension
        mask_flat = mask_tensor.to(torch.float16).view(total_rows, 1)  # [R,1]
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()          # [R, H]

    left = left_proj_raw * left_gate             # [R, H] fp16
    right = right_proj_raw * right_gate           # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # reshape back to 4‑D tensors
    left = left.view(B, N, N, H)               # [B, N, N, H]
    right = right.view(B, N, N, H)             # [B, N, N, H]

    # view as (B, H, N, N) without materialising a copy
    left_bh = left.permute(0, 3, 1, 2)          # [B, H, N, N]  (non‑contiguous)
    right_bh = right.permute(0, 3, 1, 2)        # [B, H, N, N]

    # batched matrix multiplication over the “k” dimension
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)            # (B, H, N, N)ᵀ → (B, H, N, N)
    )                                          # fp16, shape (B, H, N, N)

    # back to (B, N, N, H) – we need a contiguous layout for the next LN step,
    # so we explicitly call .contiguous()
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension *plus* gating
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H)                # [R, H] (non‑contig)
    # make it contiguous because the kernel expects dense rows
    out_flat = out_flat.contiguous()
    out_gate_flat = out_gate.view(total_rows, H).contiguous()

    out_norm_gate = _layernorm_gate_triton(
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
        block_m=256,
        block_d=32,
    )                                           # fp16, shape [R, H]

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to the original channel dimension
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)   # float32 output
    return final
