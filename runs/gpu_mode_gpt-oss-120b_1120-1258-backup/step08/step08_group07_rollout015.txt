Implementation of kernel:

```
@triton.jit
def rowwise_mean_var_kernel(
    x_ptr, # const * [total_rows, dim] float32
    mean_ptr, # mutable * [total_rows] float32
    var_ptr,  # mutable * [total_rows] float32
    total_rows: tl.constexpr,
    dim: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean

    # Write results
    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(var_ptr + offs_m, var, mask=mask_m)
```

"""
Highly‑optimized forward pass for the “outgoing” TriMul operator.

Algorithm
---------
1️⃣  Row‑wise statistics (mean & variance) of the input tensor
    are computed by a tiny Triton kernel.  Only two fp32
    reductions per row are performed – no intermediate
    normalised tensor is materialised.

2️⃣  The four projection matrices (left/right) and the three
    gate matrices are fused into a single GEMM.  The
    pre‑normalisation scale (γ) of the LayerNorm is folded into
    the weight matrices (ŵ = w·γ).  After the GEMM we recover
    the true LayerNorm output with a cheap per‑row affine
    transformation:
        y = (z – μ·s)·r + b,
    where  z = x·ŵᵀ ,  s = Σₖ ŵ ,  b = Σₖ β·w .
    This eliminates the memory‑traffic of the original
    “normalize → linear” pattern.

3️⃣  Optional mask is applied, sigmoid gates are computed and the
    gated left/right tensors are formed.

4️⃣  The N³ multiplicative update is performed with a batched
    GEMM (torch.bmm).  The hidden dimension H (≤128) is treated
    as a batch dimension, yielding B·H independent N×N matrix
    multiplies.

5️⃣  A second Triton kernel computes row‑wise mean/variance of
    the hidden‑dim tensor produced by the update.  A fully
    fused LayerNorm + gate + final linear projection is then
    realised with plain tensor operations (no extra write‑back).

All heavy work (large GEMMs & batched GEMM) stays in the
high‑throughput cuBLAS kernels; the Triton kernels only handle
the cheap reductions, which makes the implementation well‑suited
to the H100 Tensor‑Core accelerator.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise mean & variance (no output tensor)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_mean_var_kernel(
    x_ptr,            # const   * [rows, dim] (float32)
    mean_ptr,         # mutable * [rows]      (float32)
    var_ptr,          # mutable * [rows]      (float32)
    rows: tl.constexpr,
    dim: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """Compute per‑row mean and variance of a 2‑D fp32 tensor.
    Writes mean and var (both fp32) to global memory.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # accumulators for sum and sum‑squared
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over the feature dimension in blocks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)          # fp32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(var_ptr  + offs_m, var,  mask=mask_m)


# ----------------------------------------------------------------------
# Helper: fold LayerNorm scale (γ) into a linear weight matrix
# ----------------------------------------------------------------------
def _prepare_weight(weight: torch.Tensor, norm_weight: torch.Tensor,
                    norm_bias: torch.Tensor):
    """
    weight:      [out_dim, dim]   (fp32)
    norm_weight: [dim]            (γ, fp32)
    norm_bias:   [dim]            (β, fp32)

    Returns
    -------
    w_scaled : fp16 tensor  = weight * γ
    w_sum    : fp32 tensor  = Σₖ (weight·γ)_out,k
    bias     : fp32 tensor  = Σₖ β_k·weight_out,k
    """
    w_scaled = (weight * norm_weight[None, :])                   # fp32
    w_sum    = w_scaled.sum(dim=1)                               # [out_dim] fp32
    bias     = (norm_bias[None, :] * weight).sum(dim=1)          # [out_dim] fp32
    return w_scaled.to(torch.float16), w_sum, bias


# ----------------------------------------------------------------------
# Main Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor       shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor       shape [B, N, N]     (bool/float)
        - weights_dict : dict of torch.Tensors containing all Linear/LayerNorm params
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = 1e-5
    device       = input_tensor.device
    nomask       = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R          = B * N * N                 # total pairwise positions
    H          = hidden_dim                # for brevity

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise statistics of the raw input (mean/var)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()   # [R, dim] fp32
    # allocate buffers for mean and variance
    mean = torch.empty(R, dtype=torch.float32, device=device)
    var  = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M = 256      # rows per program
    BLOCK_D = 128      # columns per tile (covers dim efficiently)
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_mean_var_kernel[grid](
        x_flat,
        mean,
        var,
        R,
        dim,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    rstd = torch.rsqrt(var + eps)                     # [R]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four projection + three gate linear layers
    # ------------------------------------------------------------------
    # LayerNorm parameters
    ln_weight = weights["norm.weight"]   # γ  [dim], fp32
    ln_bias   = weights["norm.bias"]     # β  [dim], fp32

    # Prepare scaled weight matrices and the accompanying constants
    left_proj_w,   left_wsum,   left_bias   = _prepare_weight(
        weights["left_proj.weight"], ln_weight, ln_bias)
    right_proj_w,  right_wsum,  right_bias  = _prepare_weight(
        weights["right_proj.weight"], ln_weight, ln_bias)
    left_gate_w,   leftg_wsum,  leftg_bias  = _prepare_weight(
        weights["left_gate.weight"], ln_weight, ln_bias)
    right_gate_w,  rightg_wsum, rightg_bias = _prepare_weight(
        weights["right_gate.weight"], ln_weight, ln_bias)
    out_gate_w,    outg_wsum,   outg_bias   = _prepare_weight(
        weights["out_gate.weight"], ln_weight, ln_bias)

    # Concatenate all five scaled weight matrices → single GEMM
    proj_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0
    )                                         # [5*H, dim] fp16
    # Concatenate the per‑output‑channel sums and biases
    proj_wsum = torch.cat(
        [left_wsum, right_wsum, leftg_wsum, rightg_wsum, outg_wsum],
        dim=0
    )                                         # [5*H] fp32
    proj_bias = torch.cat(
        [left_bias, right_bias, leftg_bias, rightg_bias, outg_bias],
        dim=0
    )                                         # [5*H] fp32

    # Cast the raw input to fp16 – the GEMM runs on Tensor‑Cores
    x_fp16 = x_flat.to(torch.float16)        # [R, dim] fp16
    proj_raw = F.linear(x_fp16, proj_weight)   # [R, 5*H] fp16 (bias = 0)

    # Apply the affine correction that accounts for the missing LayerNorm
    # (mean, rstd, wsum, bias are all fp32 → cast to fp16 for broadcasting)
    mean_b = mean.unsqueeze(-1).to(torch.float16)          # [R,1]
    rstd_b = rstd.unsqueeze(-1).to(torch.float16)          # [R,1]
    wsum_b = proj_wsum.unsqueeze(0).to(torch.float16)      # [1,5*H]
    bias_b = proj_bias.unsqueeze(0).to(torch.float16)      # [1,5*H]

    proj = (proj_raw - mean_b * wsum_b) * rstd_b + bias_b   # [R,5*H] fp16

    # Split into the five separate streams
    left_proj_raw = proj[:, 0:H]
    right_proj_raw = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    out_gate_raw = proj[:, 4*H:]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask & sigmoid gates
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(R, 1)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # [R, H] fp16

    # Gating on the projected values
    left  = left_proj_raw * left_gate             # [R, H] fp16
    right = right_proj_raw * right_gate           # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B,N,N,H] fp16
    right = right.view(B, N, N, H)         # [B,N,N,H] fp16

    # Move hidden dimension into the batch axis to obtain many independent GEMMs
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    left_mat  = left_perm.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)     # [B·H, N, N]

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to the original shape
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (row‑wise) – statistics only
    # ------------------------------------------------------------------
    R2 = B * N * N
    out_flat = out.view(R2, H).contiguous()          # [R2, H] fp16

    out_mean = torch.empty(R2, dtype=torch.float32, device=device)
    out_var  = torch.empty(R2, dtype=torch.float32, device=device)

    grid2 = (triton.cdiv(R2, BLOCK_M),)
    rowwise_mean_var_kernel[grid2](
        out_flat.to(torch.float32),   # temporary fp32 view for the kernel
        out_mean,
        out_var,
        R2,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_rstd = torch.rsqrt(out_var + eps)           # [R2]

    # ------------------------------------------------------------------
    # 6️⃣ Apply LayerNorm, output gate and final linear projection
    # ------------------------------------------------------------------
    # Cast statistics to fp16 for the arithmetic that follows
    out_mean_b = out_mean.unsqueeze(-1).to(out_flat.dtype)   # [R2,1]
    out_rstd_b = out_rstd.unsqueeze(-1).to(out_flat.dtype)   # [R2,1]

    # LN γ/β for the hidden dimension
    out_norm_weight = weights["to_out_norm.weight"]   # γ  [H] fp32
    out_norm_bias   = weights["to_out_norm.bias"]     # β  [H] fp32
    out_norm_weight_fp16 = out_norm_weight.to(out_flat.dtype)
    out_norm_bias_fp16   = out_norm_bias.to(out_flat.dtype)

    # Normalised hidden tensor (γ,β applied)
    out_norm_pre = (out_flat - out_mean_b) * out_rstd_b
    out_norm_pre = out_norm_pre * out_norm_weight_fp16 + out_norm_bias_fp16

    # Apply the output‑gate (element‑wise)
    out_norm = out_norm_pre * out_gate   # [R2, H] fp16

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm, to_out_weight)              # [R2, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
