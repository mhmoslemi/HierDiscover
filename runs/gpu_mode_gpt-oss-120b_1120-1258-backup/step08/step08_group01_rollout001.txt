"""
Optimized forward pass for the **outgoing** TriMul primitive (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) – Triton kernel.
2️⃣ Fuse the four linear maps
       left_proj, right_proj, left_gate, right_gate
   into one GEMM (`torch.nn.functional.linear`).  The result is split
   into the four tensors (raw projections + gate pre‑activations).
3️⃣ Separate GEMM for the output‑gate.
4️⃣ Optional mask (broadcast over the hidden dimension) and in‑place
   sigmoid on the three gates.
5️⃣ Multiplicative update:
   for each hidden channel we have a matrix‑multiply
   `out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is realized by reshaping to (B·H,N,N) and using a single
   batched `torch.bmm`.
6️⃣ Second LayerNorm over the hidden dimension (again Triton).
7️⃣ Apply the output‑gate, final linear projection and cast back to
   float32.

All heavy element‑wise work (both LayerNorms, mask, sigmoid) is
performed in Triton; the dense matrix multiplications use cuBLAS
(Tensor‑Cores) via `torch.nn.functional.linear` / `torch.bmm`.
The implementation works for any batch size ≤2, sequence length ≤1024,
and arbitrary `dim` / `hidden_dim` (including the 384‑dim case).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)  γ
    bias_ptr,       # const   * [dim]               (float32)  β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1️⃣ Compute per‑row mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                    # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------- 2️⃣ Write normalized output (fp16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β fp32
        out = norm * w[None, :] + b[None, :]                    # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with LayerNorm / Linear parameters (float32)
        - config_dict  : dict with keys ``dim``, ``hidden_dim``, optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack and constants
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N               # R = B·N·N
    H = hidden_dim                       # hidden dimension

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()   # [R, dim] float32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate four weight matrices of shape [H, dim] → [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # fp16

    # fused GEMM: [R, dim] @ [4·H, dim].T → [R, 4·H] (fp16, no bias)
    fused = F.linear(ln1_out, w_fused)   # fp16

    # split fused result
    left_raw      = fused[:, 0:H]                       # [R, H]
    right_raw     = fused[:, H:2 * H]                   # [R, H]
    left_gate_raw = fused[:, 2 * H:3 * H]               # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]              # [R, H]

    # Separate output‑gate GEMM
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16),  # [H, dim]
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask (if any), sigmoid gates, element‑wise weighting
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] → [R, 1] (float16)
        mask_f = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_f
        right_raw = right_raw * mask_f

    # In‑place sigmoid → avoid extra temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate    # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # move hidden dim to batch axis → [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # collapse (B·H) as batch for torch.bmm → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # restore [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Output‑gate, final linear projection, cast back to fp32
    # --------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)          # fp16
    out_norm = out_norm * out_gate.view(B, N, N, H)   # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # float32 output
    return final
