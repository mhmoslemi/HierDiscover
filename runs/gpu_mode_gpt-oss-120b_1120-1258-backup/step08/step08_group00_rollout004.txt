def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg['dim']; hidden_dim = cfg['hidden_dim']; eps=1e-5; device=x.device
    B,N,_,_ = x.shape
    R = B*N*N
    # reshape
    x_flat = x.view(R, dim)
    # layernorm using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x_flat, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)  # returns same dtype as input (float32). We need fp16 later. We'll use .to(torch.float16)
    x_norm = x_norm.to(torch.float16)

    # fused projection:
    w_concat = torch.cat([...], dim=0).to(torch.float16)  # shape (5*H, dim)
    proj = torch.nn.functional.linear(x_norm, w_concat) # results in [R,5*H] fp16

    # mask: convert
    if cfg.get('nomask', True):
        mask_flat = None
    else:
        mask_flat = mask.reshape(R).to(torch.float16)  # [R]
    
    # split proj
    H = hidden_dim
    left_raw = proj[:, :H]
    right_raw = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    out_gate_raw = proj[:, 4*H:5*H]

    if mask_flat is not None:
        left_raw = left_raw * mask_flat[:, None]
        right_raw = right_raw * mask_flat[:, None]

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    left = left_raw * left_gate
    right = right_raw * right_gate

    # reshape for batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_mat = left.reshape(B*H, N, N)
    right_mat = right.reshape(B*H, N, N)

    # batched matmul
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))

    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # [B,N,N,H]

    # second layernorm
    out_flat = out.view(R, H)
    out_norm = torch.nn.functional.layer_norm(out_flat, (H,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'], eps=eps)
    out_norm = out_norm.to(torch.float16)
    # multiply by out_gate
    out_norm = out_norm * out_gate

    final_half = torch.nn.functional.linear(out_norm, weights['to_out.weight'].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

"""
Tri‑multiplicative (outgoing) update – fast Triton / PyTorch implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – we reuse
    torch.nn.functional.layer_norm which is highly‑optimised on GPUs.
2️⃣ Fuse the five linear maps (left/right projections + three gates)
   into a single matrix multiplication:   X_norm @ W_fusedᵀ  → (R,5·H)
3️⃣ One Triton kernel splits the fused result, applies the optional
   pair‑wise mask, computes the σ‑gates and writes the three tensors
   `left`, `right` and `out_gate` (all fp16).
4️⃣ Batched GEMM over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
   (implemented with torch.bmm which maps to cuBLAS tensor‑cores).
5️⃣ Second LayerNorm on the hidden‑dim tensor, element‑wise multiplication
   with `out_gate`.
6️⃣ Final linear projection back to `dim` (fp16 → fp32 output).

Only the split‑+‑gate kernel is written in Triton; all heavy matrix
multiplications use cuBLAS through PyTorch and the two normalisations use
the highly‑optimised PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]     (fp16)  – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are fp16 (sigmoid evaluated in fp32 for stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load mask (if present) and broadcast to the hidden‑dim dimension.
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)    # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process the hidden‑dim axis in chunks of BLOCK_H.
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside the fused projection: [L,R,LG,RG,OG] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H] offset for segment 0

        left_raw_ptr     = proj_ptr + base
        right_raw_ptr    = proj_ptr + base + H
        left_gate_ptr    = proj_ptr + base + 2 * H
        right_gate_ptr   = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        left_raw      = tl.load(left_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_raw     = tl.load(right_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        left_gate_raw = tl.load(left_gate_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_gate_raw= tl.load(right_gate_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(out_gate_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)

        # sigmoid in fp32 → fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------- unpack arguments -----------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # --------------------- 1️⃣ first LayerNorm -------------------------
    # reshape to 2‑D and use the highly‑optimised PyTorch implementation
    x_flat = x.view(R, dim)
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)               # fp16 for the following GEMM

    # --------------------- 2️⃣ fused linear (5·H outputs) --------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # (5·H, dim)
    proj = F.linear(x_norm, w_fused)  # (R, 5·H), fp16

    # --------------------- 3️⃣ split, mask, gates (Triton) ----------
    left  = torch.empty((R, H), dtype=torch.float16, device=device)
    right = torch.empty((R, H), dtype=torch.float16, device=device)
    out_gate = torch.empty((R, H), dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        mask_ptr = mask.to(torch.float16).contiguous().view(R)

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj   # free intermediate memory

    # --------------------- 4️⃣ batched GEMM over hidden dim ----------
    # reshape to [B, H, N, N] → batch dimension B·H
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat  = left.reshape(B * H, N, N)   # (B·H, N, N)
    right_mat = right.reshape(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------- 5️⃣ second LayerNorm + out‑gate ----------
    out_flat = out.view(R, H)
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(torch.float16)

    out_norm = out_norm * out_gate           # element‑wise gate
    del out, out_gate                        # free memory

    # --------------------- 6️⃣ final linear back to `dim` -----------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # (R, dim), fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
