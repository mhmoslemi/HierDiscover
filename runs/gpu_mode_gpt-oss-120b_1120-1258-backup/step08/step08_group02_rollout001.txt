"""
custom_kernel  –  high‑performance forward pass of the *outgoing* TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (dim‑wise) → FP16.
2️⃣ Fuse the four linear projections (left/right + left/right‑gate) into a
   single GEMM:   X_norm @ W_fused.T  →  [R, 4·H]  (R = B·N·N, H = hidden_dim).
   Slice the result into left_raw, right_raw, left_gate_raw, right_gate_raw.
   Compute the output‑gate with a second (small) GEMM.
3️⃣ (Optional) Apply the pair‑mask to left_raw/right_raw.
4️⃣ Apply sigmoid to the three gate tensors (in‑place) and fuse the gates:
        left  = left_raw  * sigmoid(left_gate_raw)
        right = right_raw * sigmoid(right_gate_raw)
5️⃣ Reshape left/right to (B·H, N, N) and perform a batched GEMM:
        out = (B·H, N, N) @ (B·H, N, N)ᵀ      →  (B·H, N, N)
   This implements the ``einsum('... i k d, ... j k d -> ... i j d')``.
6️⃣ Fuse the second LayerNorm (over hidden dim H) and multiplication by the
   output‑gate into a single Triton kernel.
7️⃣ Final linear projection back to the original dimension (FP16 GEMM) and
   cast the result to FP32.

Only the two LayerNorm phases are written in Triton; all dense linear
operations use the highly‑optimized cuBLAS kernels (torch.nn.functional.linear,
torch.bmm).  The implementation works for any batch size ∈ {1,2},
sequence length up to 1024, and both masked / unmasked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)
    bias_ptr,       # const   * [dim]            (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                             # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β (FP16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # still FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    gate_ptr,       # const   * [total_rows, dim] (float16)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)
    bias_ptr,       # const   * [dim]            (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute mean & variance (ignoring gate)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalise, apply γ/β and multiply by gate (all FP32, store FP16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load raw activation (FP32)
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (FP16 → FP32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = (norm * w[None, :] + b[None, :]) * g_vals   # FP32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : {
                "norm.weight":           Tensor[dim] (float32)
                "norm.bias":             Tensor[dim] (float32)
                "left_proj.weight":      Tensor[hidden, dim] (float32)
                "right_proj.weight":     Tensor[hidden, dim] (float32)
                "left_gate.weight":      Tensor[hidden, dim] (float32)
                "right_gate.weight":     Tensor[hidden, dim] (float32)
                "out_gate.weight":       Tensor[hidden, dim] (float32)
                "to_out_norm.weight":    Tensor[hidden] (float32)
                "to_out_norm.bias":      Tensor[hidden] (float32)
                "to_out.weight":         Tensor[dim, hidden] (float32)
          }
        - config_dict : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for the row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # Tuned block sizes (work well for dim ∈ {128,384,768})
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate weight matrices (each [H, dim]) → [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # X_norm (FP16) @ W_fusedᵀ → [R, 4·H]  (single GEMM)
    fused = F.linear(x_norm_fp16, w_fused)          # FP16

    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Apply optional mask and sigmoid gates
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # broadcast mask to hidden dimension
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # [R, H] FP16

    left  = left_raw  * left_gate   # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # --------------------------------------------------------------
    # reshape to (B, N, N, H)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring hidden dimension to batch axis → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)   (FP16 matmul → Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Fuse LayerNorm over hidden dim H and multiplication by out_gate
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H] FP16
    out_gate_flat = out_gate                        # already [R, H] FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat,
                                          dtype=torch.float16,
                                          device=device)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    # Cast to FP32 as required by the API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
