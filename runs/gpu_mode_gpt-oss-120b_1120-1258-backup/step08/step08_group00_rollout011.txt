def layernorm_gate_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr, gate_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # accumulate mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # x
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # gate pointer: shape (rows, dim)
    gate_ptr_local = gate_ptr + offs_m[:, None] * dim + tl.arange(0, BLOCK_D)[None, :]  # pointer will be used inside loop.

    # write out
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # x
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # gate
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out = out * gate_val  # elementwise multiplication (both fp32)
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

"""
Tri‑multiplicative “outgoing” update – fast Triton implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16).
2️⃣  One fused Linear (weights concatenated) produces:
        left_proj, right_proj, left_gate, right_gate, out_gate
    in FP16.
3️⃣  A single Triton kernel splits the fused result, applies the optional
    pair‑wise mask, computes the sigmoid gates and writes the three tensors:
        left  = (proj_left  * mask) * sigmoid(gate_left)
        right = (proj_right * mask) * sigmoid(gate_right)
        out_gate = sigmoid(gate_out)
4️⃣  Batched matrix multiplication (B·H independent GEMMs) over the hidden
    dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
    executed by a custom Triton kernel that uses Tensor‑cores.
5️⃣  Fused LayerNorm + output‑gate multiplication on the GEMM result
    (still FP16).
6️⃣  Final Linear projection back to `dim` (FP16 → FP32 output).

Only the two LayerNorms, the split‑&‑gate, and the batched GEMM are written
in Triton; all large matrix‑multiplications (the fused projection and the
final projection) use cuBLAS via `torch.nn.functional.linear`.  This design
drastically reduces memory traffic and launches far fewer kernels, bringing
the runtime well below the 1 ms target for the prescribed test shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalise & apply γ,β (store as fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # const   *[rows, 5*H]   (float16)
    left_ptr,      # mutable *[rows, H]    (float16)
    right_ptr,     # mutable *[rows, H]    (float16)
    out_gate_ptr,  # mutable *[rows, H]    (float16)
    mask_ptr,      # const   *[rows]       (float16) – ignored if HAS_MASK==0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # mask broadcast (1.0 when absent)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)          # [BLOCK_R]
        mask_val = mask_val[:, None]                           # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension in chunks of BLOCK_H
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # base offset for each row (5*H elements per row)
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [R,H]

        # pointers to the five segments
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        # load raw values
        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid – compute in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(- left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-  out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,   left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,  right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fused LayerNorm + output‑gate multiplication (FP16 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, H]   (float16)
    y_ptr,          # mutable *[rows, H]   (float16)
    weight_ptr,     # const   *[H]        (float32) γ
    bias_ptr,       # const   *[H]        (float32) β
    gate_ptr,       # const   *[rows, H]   (float16) – out_gate
    rows: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # compute mean/var (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < H
        ptr = x_ptr + offs_m[:, None] * H + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # normalise, apply γ,β and multiply by out_gate (still fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < H

        # load x
        x_ptr_off = x_ptr + offs_m[:, None] * H + offs_d[None, :]
        x_vals = tl.load(x_ptr_off,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate
        g_ptr_off = gate_ptr + offs_m[:, None] * H + offs_d[None, :]
        g_vals = tl.load(g_ptr_off,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g_vals
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * H + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Batched GEMM: C = A @ Bᵀ   (A,B: [B·H, N, N]  fp16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,          # const *[batch, M, K]   (fp16)
    b_ptr,          # const *[batch, N, K]   (fp16) – we interpret K as the reduction dim,
                                            #                 so we read it with stride_bk = 1 (column‑major)
    c_ptr,          # mutable *[batch, M, N] (fp16)
    batch_stride_a: tl.int32,
    batch_stride_b: tl.int32,
    batch_stride_c: tl.int32,
    M: tl.int32,
    N: tl.int32,
    K: tl.int32,
    stride_am: tl.int32,
    stride_ak: tl.int32,
    stride_bk: tl.int32,
    stride_bn: tl.int32,
    stride_cm: tl.int32,
    stride_cn: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (Tensor‑core dot writes fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(a_ptr + batch_id * batch_stride_a
                           + offs_m[:, None] * stride_am
                           + offs_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B is read transposed: we need Bᵀ (K × N)
        b = tl.load(b_ptr + batch_id * batch_stride_b
                           + offs_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        acc += tl.dot(a, b)

    # store result (fp16)
    c = acc.to(tl.float16)
    c_ptr_off = c_ptr + batch_id * batch_stride_c \
                       + offs_m[:, None] * stride_cm \
                       + offs_n[None, :] * stride_cn
    tl.store(c_ptr_off, c,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict of model parameters (float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()            # [R, dim]  fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_norm = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free fp32 input early

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates (single linear)
    # ------------------------------------------------------------------
    # concatenate weights: [left_proj, right_proj, left_gate, right_gate, out_gate]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # (5*H, dim)

    # (R, dim) @ (5*H, dim)ᵀ  →  (R, 5*H)   fp16
    proj = F.linear(x_norm, w_fused)                  # fused linear
    del w_fused, x_norm                               # optional free

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # No mask – kernel will ignore the mask pointer
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask shape [B, N, N] → [R]
        mask_fp = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr                                   # free

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM: out = left @ rightᵀ   (B·H independent GEMMs)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) – we need the hidden dim in the batch axis
    left_mat  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    batch = B * H
    M = N
    N_ = N
    K = N

    # output tensor
    out_mat = torch.empty_like(left_mat)  # [B·H, N, N] fp16

    # strides (all tensors are contiguous)
    stride_am = left_mat.stride(1)   # N
    stride_ak = left_mat.stride(2)   # 1
    stride_bk = right_mat.stride(2)  # 1  (column stride – we read transposed)
    stride_bn = right_mat.stride(1)  # N
    stride_cm = out_mat.stride(1)    # N
    stride_cn = out_mat.stride(2)    # 1

    batch_stride_a = left_mat.stride(0)   # N*N
    batch_stride_b = right_mat.stride(0)  # N*N
    batch_stride_c = out_mat.stride(0)    # N*N

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid_gemm = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_, BLOCK_N),
    )
    batched_gemm_kernel[grid_gemm](
        left_mat,
        right_mat,
        out_mat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        M,
        N_,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        # H100 tensor‑core friendly config
    )
    del left_mat, right_mat  # free memory early

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm with output‑gate multiplication
    # ------------------------------------------------------------------
    out_flat = out_mat.view(R, H).contiguous()
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid_norm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid_norm2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate,                        # gate (fp16)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out_mat

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16),  # (dim, hidden_dim)
    )   # (R, dim) fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
