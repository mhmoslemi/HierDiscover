```python
def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps=1e-5
    device = x.device
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim
    
    # Reshape input to (R, dim)
    x_flat = x.view(R, dim)
    
    # kernel to compute per-row mean & inv_std
    mean = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    row_stats_kernel[grid](
        x_flat,
        mean,
        rstd,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # Convert input to half for linear
    x_half = x_flat.to(torch.float16)
    
    # Prepare weight scalings
    # First LN parameters
    gamma = weights["norm.weight"]   # shape [dim], fp32
    beta = weights["norm.bias"]     # shape [dim], fp32
    gamma_fp16 = gamma.to(torch.float16)
    beta_fp16 = beta.to(torch.float16)
    
    # ----- Left + left_gate -----
    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0) # shape [2H, dim]
    left_concat_w_scaled = left_concat_w * gamma_fp16[None, :]   # elementwise broadcast
    # Precompute A and B for this concat (size 2H)
    A_left = left_concat_w_scaled.sum(dim=1)   # fp16 but convert to fp32 later
    B_left = (left_concat_w * beta_fp16[None, :]).sum(dim=1)   # fp32
    
    # ----- Right + right_gate -----
    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)
    right_concat_w_scaled = right_concat_w * gamma_fp16[None, :]
    A_right = right_concat_w_scaled.sum(dim=1)
    B_right = (right_concat_w * beta_fp16[None, :]).sum(dim=1)
    
    # ----- Out gate -----
    out_gate_w = weights["out_gate.weight"]
    out_gate_w_scaled = out_gate_w * gamma_fp16[None, :]
    A_out_gate = out_gate_w_scaled.sum(dim=1)
    B_out_gate = (out_gate_w * beta_fp16[None, :]).sum(dim=1)
    
    # Convert scaling tensors to appropriate dtype:
    A_left = A_left.to(torch.float32)
    B_left = B_left.to(torch.float32)
    A_right = A_right.to(torch.float32)
    B_right = B_right.to(torch.float32)
    A_out_gate = A_out_gate.to(torch.float32)
    B_out_gate = B_out_gate.to(torch.float32)
    
    # Linear calls
    left_and_gate_raw = torch.nn.functional.linear(x_half, left_concat_w_scaled)   # (R,2H) fp16
    right_and_gate_raw = torch.nn.functional.linear(x_half, right_concat_w_scaled) # (R,2H) fp16
    out_gate_raw = torch.nn.functional.linear(x_half, out_gate_w_scaled)           # (R,H) fp16
    
    # Normalize raw outputs using per-row mean & rstd
    # Function to normalize: (raw_fp16.float() - mean[:,None] * A) * rstd[:,None] + B
    # We can write a helper:
    def normalize(raw_fp16, A, B):
        raw_fp32 = raw_fp16.to(torch.float32)
        # mean and rstd broadcast: shape (R,1)
        # subtract mean * A
        # raw_fp32 - mean[:,None] * A[None,:]  # need broadcasting
        return ((raw_fp32 - mean[:, None] * A[None, :]) * rstd[:, None] + B[None, :]).to(torch.float16)
    
    left_and_gate = normalize(left_and_gate_raw, A_left, B_left)   # (R,2H)
    right_and_gate = normalize(right_and_gate_raw, A_right, B_right)
    out_gate_norm = normalize(out_gate_raw, A_out_gate, B_out_gate)   # (R,H)

    left_proj_norm = left_and_gate[:, :H]
    left_gate_norm = left_and_gate[:, H:]
    right_proj_norm = right_and_gate[:, :H]
    right_gate_norm = right_and_gate[:, H:]
    
    # Release temporary tensors
    del left_and_gate_raw, right_and_gate_raw, out_gate_raw, left_and_gate, right_and_gate
    
    # Prepare mask
    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        mask_flat = None
    else:
        mask_fp16 = mask.to(torch.float16).reshape(R)   # [R]
        mask_ptr = mask_fp16
        has_mask = 1

    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

        left_proj_norm,
        left_gate_norm,
        right_proj_norm,
        right_gate_norm,
        out_gate_norm,
        mask_ptr,
        left,
        right,
        out_gate,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # free raw normalized tensors
    del left_proj_norm, left_gate_norm, right_proj_norm, right_gate_norm, out_gate_norm
    
    # Reshape left and right for BMM
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    
    left_mat = left.reshape(B*H, N, N)
    right_mat = right.reshape(B*H, N, N)
    
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))   # (B*H, N, N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(R, H).contiguous()
    
    # LN2 + out_gate scaling kernel (fuse)
    out_scaled = torch.empty_like(out_flat, dtype=torch.float16)
    
    # apply LN2 + gate in one kernel
    BLOCK_R2 = 256
    BLOCK_H2 = 32
    grid_ln2 = (triton.cdiv(R, BLOCK_R2),)
    apply_ln2_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        out_scaled,
        weights["to_out_norm.weight"],   # gamma2 float32
        weights["to_out_norm.bias"],     # beta2 float32
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out_flat, out_gate, out
    
    # Final linear
    final_half = torch.nn.functional.linear(out_scaled, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

```python
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def row_stats_kernel(x_ptr, mean_ptr, rstd_ptr, rows, dim, eps,
                     BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

@triton.jit
def gate_and_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_raw_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    rows, H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # mask broadcast
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load values
        lproj = tl.load(left_proj_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        lgate = tl.load(left_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        rproj = tl.load(right_proj_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        rgate = tl.load(right_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        outg_raw = tl.load(out_gate_raw_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)

        # sigmoid in fp32
        lg = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rg = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        og = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left = lproj * mask_val * lg
        right = rproj * mask_val * rg

        # store
        tl.store(left_out_ptr + offs_r[:, None] * H + offs_h[None, :],
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + offs_r[:, None] * H + offs_h[None, :],
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + offs_r[:, None] * H + offs_h[None, :],
                 og,
                 mask=mask_r[:, None] & mask_h[None, :])

@triton.jit
def apply_ln2_gate_kernel(
    out_ptr, out_gate_ptr, out_scaled_ptr,
    norm_weight_ptr, norm_bias_ptr,
    rows, H, eps,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # first pass: compute mean & variance
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: normalize, apply LN2 weight,bias and out_gate
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw out and gate
        out_val = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                          mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        gate_val = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # load LN2 weight & bias (fp32)
        w = tl.load(norm_weight_ptr + offs_h, mask=mask_h, other=0.0)
        b = tl.load(norm_bias_ptr + offs_h, mask=mask_h, other=0.0)

        # normalize
        norm = (out_val - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]

        # apply gate
        gated = norm * gate_val

        tl.store(out_scaled_ptr + offs_r[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Kernel: per‑row mean/variance for a 2‑D tensor
# --------------------------------------------------------------
@triton.jit
def row_stats_kernel(
    x_ptr,        # *[rows, dim] (float32)
    mean_ptr,     # *[rows] (float32)
    rstd_ptr,     # *[rows] (float32)
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# Kernel: apply mask and sigmoid gates (left/right/out_gate)
# --------------------------------------------------------------
@triton.jit
def gate_and_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_raw_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]           # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        lproj = tl.load(left_proj_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        lgate = tl.load(left_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        rproj = tl.load(right_proj_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        rgate = tl.load(right_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        outg_raw = tl.load(out_gate_raw_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)

        # sigmoid in fp32 -> back to fp16
        lgate_sig = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate_sig = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        outg_sig  = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = lproj * mask_val * lgate_sig
        right = rproj * mask_val * rgate_sig

        tl.store(left_out_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + offs_r[:, None] * H + offs_h[None, :],
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + offs_r[:, None] * H + offs_h[None, :],
                 outg_sig,
                 mask=mask_r[:, None] & mask_h[None, :])

# Kernel: second LayerNorm + multiplication by the already‑sigmoided out_gate
# --------------------------------------------------------------
@triton.jit
def apply_ln2_gate_kernel(
    out_ptr, out_gate_ptr, out_scaled_ptr,
    norm_weight_ptr, norm_bias_ptr,
    rows: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # ---- first pass: mean / variance ----
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: normalize, apply weight/bias, multiply by out_gate ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_val = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
        gate_val = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        w = tl.load(norm_weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ2
        b = tl.load(norm_bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β2

        norm = (out_val - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]
        gated = norm * gate_val

        tl.store(out_scaled_ptr + offs_r[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    The implementation fuses the two LayerNorms, the mask, the sigmoid gates
    and the final projection into a handful of Triton kernels while keeping
    the heavy matrix multiplications (the three projections and the final
    Linear) inside cuBLAS/Torch for maximal tensor‑core utilisation.
    """
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row wise mean/std (first LayerNorm statistics only)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim] float32
    mean = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_stats = (triton.cdiv(R, BLOCK_M),)
    row_stats_kernel[grid_stats](
        x_flat,
        mean,
        rstd,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Prepare scaled weights for the three linear projections
    # ------------------------------------------------------------------
    # First LayerNorm parameters (γ,β)
    gamma = weights["norm.weight"]   # [dim], fp32
    beta  = weights["norm.bias"]    # [dim], fp32
    gamma_fp16 = gamma.to(torch.float16)
    beta_fp16  = beta.to(torch.float16)

    # ----- left projection + left gate (2H) -----
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_concat_w_fp32 = torch.cat([left_proj_w, left_gate_w], dim=0)           # [2H, dim]
    left_concat_w_fp16 = left_concat_w_fp32.to(torch.float16)
    left_scaled_w = left_concat_w_fp16 * gamma_fp16[None, :]                    # *γ
    A_left = left_scaled_w.sum(dim=1).to(torch.float32)                         # Σ w·γ
    B_left = (left_concat_w_fp32 * beta_fp16[None, :]).sum(dim=1)              # Σ w·β

    # ----- right projection + right gate (2H) -----
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_concat_w_fp32 = torch.cat([right_proj_w, right_gate_w], dim=0)
    right_concat_w_fp16 = right_concat_w_fp32.to(torch.float16)
    right_scaled_w = right_concat_w_fp16 * gamma_fp16[None, :]
    A_right = right_scaled_w.sum(dim=1).to(torch.float32)
    B_right = (right_concat_w_fp32 * beta_fp16[None, :]).sum(dim=1)

    # ----- out gate (H) -----
    out_gate_w = weights["out_gate.weight"]
    out_gate_w_fp16 = out_gate_w.to(torch.float16)
    out_gate_scaled_w = out_gate_w_fp16 * gamma_fp16[None, :]
    A_outgate = out_gate_scaled_w.sum(dim=1).to(torch.float32)
    B_outgate = (out_gate_w * beta_fp16[None, :]).sum(dim=1)          # fp32

    # Cast input to half for the matmuls
    x_half = x_flat.to(torch.float16)

    # ----- three fused matmuls (cuBLAS) -----
    left_and_gate_raw = F.linear(x_half, left_scaled_w)               # [R, 2H] fp16
    right_and_gate_raw = F.linear(x_half, right_scaled_w)             # [R, 2H] fp16
    out_gate_raw = F.linear(x_half, out_gate_scaled_w)                # [R, H]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Normalise the raw matmul results using the previously computed mean/rstd
    # ------------------------------------------------------------------
    def normalize(raw_fp16, A, B):
        raw_fp32 = raw_fp16.to(torch.float32)
        # (raw - μ·A) * rstd + B
        out = ((raw_fp32 - mean[:, None] * A[None, :]) * rstd[:, None] + B[None, :])
        return out.to(torch.float16)

    left_and_gate = normalize(left_and_gate_raw, A_left, B_left)   # [R,2H]
    right_and_gate = normalize(right_and_gate_raw, A_right, B_right)  # [R,2H]
    out_gate_norm = normalize(out_gate_raw, A_outgate, B_outgate)      # [R,H]

    # split left/right from the concatenated tensors
    left_proj_norm  = left_and_gate[:, :H]
    left_gate_norm  = left_and_gate[:, H:]
    right_proj_norm = right_and_gate[:, :H]
    right_gate_norm = right_and_gate[:, H:]

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask and sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_ptr = mask.to(torch.float16).reshape(R)   # [R]

    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 32
    grid_gate = (triton.cdiv(R, BLOCK_R),)
    gate_and_mask_kernel[grid_gate](
        left_proj_norm,
        left_gate_norm,
        right_proj_norm,
        right_gate_norm,
        out_gate_norm,
        mask_ptr,
        left,
        right,
        out_gate,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # Clean up temporaries
    del left_proj_norm, left_gate_norm, right_proj_norm, right_gate_norm, out_gate_norm
    del left_and_gate_raw, right_and_gate_raw, out_gate_raw
    del left_and_gate, right_and_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication over the hidden dimension
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_mat = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(R, H).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + multiplication by the out‑gate (single kernel)
    # ------------------------------------------------------------------
    out_scaled = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_R2 = 256
    BLOCK_H2 = 32
    grid_ln2 = (triton.cdiv(R, BLOCK_R2),)
    apply_ln2_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        out_scaled,
        weights["to_out_norm.weight"],   # γ2, fp32
        weights["to_out_norm.bias"],     # β2, fp32
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 7️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(out_scaled, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

        mask_h = offs_h < H

        lproj = tl.load(left_proj_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        lgate = tl.load(left_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        rproj = tl.load(right_proj_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        rgate = tl.load(right_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
        outg_raw = tl.load(out_gate_raw_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        lgate_sig = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate_sig = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        outg_sig  = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = lproj * mask_val * lgate_sig
        right = rproj * mask_val * rgate_sig

        tl.store(left_out_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + offs_r[:, None] * H + offs_h[None, :],
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + offs_r[:, None] * H + offs_h[None, :],
                 outg_sig,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# Kernel 3: second LayerNorm + multiplication by already‑sigmoided out_gate
# --------------------------------------------------------------
@triton.jit
def apply_ln2_gate_kernel(
    out_ptr, out_gate_ptr, out_scaled_ptr,
    norm_weight_ptr, norm_bias_ptr,
    rows: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # ---- first pass: mean / variance ----
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: normalize, apply γ/β, multiply by out_gate ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_val = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
        gate_val = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        w = tl.load(norm_weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ₂
        b = tl.load(norm_bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β₂

        normed = (out_val - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]
        gated  = normed * gate_val

        tl.store(out_scaled_ptr + offs_r[:, None] * H + offs_h[None, :],
                 gated.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# Custom forward kernel – outgoing TriMul
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    The implementation fuses the two LayerNorms, mask handling, sigmoid
    gating and the final projection into a few Triton kernels while keeping
    the heavy matrix multiplications inside cuBLAS for peak performance.
    """
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm statistics (mean & inv‑std)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] float32
    mean = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_stats = (triton.cdiv(R, BLOCK_M),)
    row_stats_kernel[grid_stats](
        x_flat,
        mean,
        rstd,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Prepare scaled weights (γ from first LayerNorm) and bias terms (β)
    # ------------------------------------------------------------------
    gamma = weights["norm.weight"]           # [dim] fp32
    beta  = weights["norm.bias"]            # [dim] fp32
    gamma_fp16 = gamma.to(torch.float16)
    beta_fp16  = beta.to(torch.float16)

    # ----- left + left_gate (2H) -----
    left_concat_w_fp32 = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    )                                          # [2H, dim] fp32
    left_concat_w_fp16 = left_concat_w_fp32.to(torch.float16)
    left_scaled_w = left_concat_w_fp16 * gamma_fp16[None, :]  # *γ
    A_left = left_scaled_w.sum(dim=1).to(torch.float32)       # Σ w·γ
    B_left = (left_concat_w_fp32 * beta_fp16[None, :]).sum(dim=1)  # Σ w·β

    # ----- right + right_gate (2H) -----
    right_concat_w_fp32 = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    )
    right_concat_w_fp16 = right_concat_w_fp32.to(torch.float16)
    right_scaled_w = right_concat_w_fp16 * gamma_fp16[None, :]
    A_right = right_scaled_w.sum(dim=1).to(torch.float32)
    B_right = (right_concat_w_fp32 * beta_fp16[None, :]).sum(dim=1)

    # ----- out_gate (H) -----
    out_gate_w_fp32 = weights["out_gate.weight"]
    out_gate_w_fp16 = out_gate_w_fp32.to(torch.float16)
    out_gate_scaled_w = out_gate_w_fp16 * gamma_fp16[None, :]
    A_outgate = out_gate_scaled_w.sum(dim=1).to(torch.float32)
    B_outgate = (out_gate_w_fp32 * beta_fp16[None, :]).sum(dim=1)

    # --------------------------------------------------------------
    # 3️⃣ MatMuls (cuBLAS) – three GEMMs for the three weight groups
    # --------------------------------------------------------------
    x_half = x_flat.to(torch.float16)                    # [R, dim] fp16
    left_and_gate_raw = F.linear(x_half, left_scaled_w)   # [R, 2H] fp16
    right_and_gate_raw = F.linear(x_half, right_scaled_w) # [R, 2H] fp16
    out_gate_raw = F.linear(x_half, out_gate_scaled_w)   # [R, H]  fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply the first LayerNorm correction using the pre‑computed stats
    # --------------------------------------------------------------
    def normalize(raw_fp16, A, B):
        raw_fp32 = raw_fp16.to(torch.float32)
        # (raw - μ·A) * rstd + B   (all fp32)
        out = ((raw_fp32 - mean[:, None] * A[None, :]) *
               rstd[:, None] + B[None, :])
        return out.to(torch.float16)

    left_and_gate  = normalize(left_and_gate_raw,  A_left,  B_left)   # [R,2H]
    right_and_gate = normalize(right_and_gate_raw, A_right, B_right) # [R,2H]
    out_gate_norm  = normalize(out_gate_raw,        A_outgate, B_outgate)  # [R,H]

    # split left/right projections from the concatenated tensors
    left_proj_norm  = left_and_gate[:, :H]
    left_gate_norm  = left_and_gate[:, H:]
    right_proj_norm = right_and_gate[:, :H]
    right_gate_norm = right_and_gate[:, H:]

    # --------------------------------------------------------------
    # 5️⃣ Mask + sigmoid gates (single Triton kernel)
    # --------------------------------------------------------------
    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_ptr = mask.to(torch.float16).reshape(R)   # [R]

    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 32
    grid_gate = (triton.cdiv(R, BLOCK_R),)
    gate_and_mask_kernel[grid_gate](
        left_proj_norm,
        left_gate_norm,
        right_proj_norm,
        right_gate_norm,
        out_gate_norm,
        mask_ptr,
        left,
        right,
        out_gate,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # free temporaries
    del left_proj_norm, left_gate_norm, right_proj_norm, right_gate_norm, out_gate_norm
    del left_and_gate_raw, right_and_gate_raw, out_gate_raw, left_and_gate, right_and_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched matrix multiply over the hidden dimension
    # --------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)       # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_flat = out.view(R, H).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm + multiplication by out_gate (single kernel)
    # --------------------------------------------------------------
    out_scaled = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_R2 = 256
    BLOCK_H2 = 32
    grid_ln2 = (triton.cdiv(R, BLOCK_R2),)
    apply_ln2_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        out_scaled,
        weights["to_out_norm.weight"],   # γ₂  (float32)
        weights["to_out_norm.bias"],     # β₂  (float32)
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_scaled,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
