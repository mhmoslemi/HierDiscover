def custom_kernel(data):
    x, mask, weights, cfg = data
    dim, hidden_dim = cfg["dim"], cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B,N,_,_ = x.shape
    R = B*N*N
    H = hidden_dim
    out_dim = 5*H

    # Precompute fused weights.
    # Weight shapes: left_proj.weight [H, dim], etc.
    # these are weight matrices where output dimension is H.
    # They are stored as shape [H, dim] (Torch Linear weight shape [out_features, in_features]).
    # We'll need original weight_i for each output row.

    # Gather the weight tensors
    left_proj_w = weights["left_proj.weight"]           # shape [H, dim]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # They are all fp32.

    # Stack them
    weight_orig = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)   # shape [5H, dim], fp32
    # gamma and beta from first layernorm
    norm_gamma = weights["norm.weight"]    # shape [dim]
    norm_beta = weights["norm.bias"]       # shape [dim]
    # out_gate weight for second linear later (to_out.weight)
    # already loads later.

    # Compute w_tilde = weight_orig * norm_gamma  (broadcast)
    w_tilde = (weight_orig * norm_gamma[None, :]).to(torch.float16)   # (5H, dim) fp16

    # Compute bias_original = (norm_beta * weight_orig).sum(dim=1)   # (5H,)
    bias_original = (norm_beta[None, :] * weight_orig).sum(dim=1).to(torch.float32)

    # Compute sum_w_tilde = w_tilde.float().sum(dim=1)   # (5H,)
    sum_w_tilde = (weight_orig * norm_gamma[None, :]).sum(dim=1).to(torch.float32)  # fp32

    # Create output tensor
    out_fused_fp16 = torch.empty(R, out_dim, dtype=torch.float16, device=device)

    # Launch fused kernel
    BLOCK_M = 64  # rows per program
    BLOCK_K = 32  # dim per tile
    BLOCK_N = 64  # output dim per tile (must be <= out_dim)
    total_rows = R
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_linear_kernel[grid](
        x_ptr = x.view(R, dim).contiguous(),
        out_ptr = out_fused_fp16,
        weight_ptr = w_tilde,
        bias_ptr = bias_original,
        sum_weight_ptr = sum_w_tilde,
        total_rows = total_rows,
        dim = dim,
        out_dim = out_dim,
        eps = eps,
        BLOCK_M = BLOCK_M,
        BLOCK_K = BLOCK_K,
        BLOCK_N = BLOCK_N,
        num_warps=4,
    )
```

@triton.jit
def layernorm_stats_kernel(
    x_ptr, mean_ptr, rstd_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # store
    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

"""
TriMul (outgoing) – fused LayerNorm + linear and a fast CUDA kernel.

Algorithm
---------
0️⃣ Unpack inputs and pre‑compute the fused weights:
   *   weight_tilde   = W * γ          (γ from the first LayerNorm)
   *   bias_original  = β·W           (β from the first LayerNorm)
   *   sum_w_tilde    = Σ_d (γ·W)     – needed for the mean subtraction term.

1️⃣ **Statistics kernel** – a tiny Triton kernel that computes per‑row
   mean μ and 1/σ (rstd) for the input tensor X ∈ ℝ^{R×C}
   (R = B·N·N, C = dim).  It writes two fp32 vectors `mean` and `rstd`.

2️⃣ **Mixed‑precision GEMM** – `X` (cast to fp16) is multiplied by the
   pre‑scaled weights `weight_tilde` (fp16) with a single cuBLAS call
   (`torch.nn.functional.linear`).  This yields the raw dot‑products
   `S = X @ weight_tildeᵀ`.

3️⃣ **Closed‑form LayerNorm + linear** – using the statistics from step 1
   we compute the exact LayerNorm‑followed linear result without ever
   materialising the normalized tensor:
        out = rstd * (S – mean·sum_w_tilde) + bias_original
   All arithmetic is performed in fp32 and the final tensor is cast back
   to fp16.

4️⃣ Split the fused output (shape [R,5·H]) into the four projections,
   the two gates and the output‑gate.  Apply the optional mask, the three
   sigmoid gates and the element‑wise multiplication that forms the
   “left” and “right” tensors.

5️⃣ **Cubic update** – batched matrix multiplication over the sequence
   dimension (B·H independent GEMMs):
        out[b,h] = left[b,h] @ right[b,h]ᵀ   (fp16 Tensor‑Core).

6️⃣ **Second LayerNorm** – a second Triton row‑wise LayerNorm over the
   hidden dimension H.

7️⃣ Multiply by the output‑gate, apply the final linear projection and
   cast the result back to float32.

The implementation uses two custom Triton kernels: one for the
row‑wise LayerNorm (both passes) and a lightweight kernel that only
computes the per‑row mean / rstd.  All heavy linear algebra runs in
fp16 using cuBLAS, giving a fast forward pass that meets the target
runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernels
# -------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp32)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim]          (fp32) – γ
    b_ptr,          # const  * [dim]          (fp32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """
    Row‑wise LayerNorm: y = ((x-μ)/σ)*γ + β
    Two passes over the data – first for mean/var, second for the
    normalized output (cast to fp16).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- 1. mean & variance ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1/√(var+eps)

    # ----- 2. write normalized result (fp16) --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]     # (x-μ)/σ
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def layernorm_stats_kernel(
    x_ptr,          # const * [total_rows, dim] (fp32)
    mean_ptr,       # mutable* [total_rows]    (fp32)
    rstd_ptr,       # mutable* [total_rows]    (fp32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """
    Computes per‑row mean μ and rstd = 1/√(var+eps) for X.
    No output tensor is written – only the two statistics.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# -------------------------------------------------------------
# Forward pass
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : dict of torch.nn.Parameter (float32)
        * config_dict  : dict with keys
            - "dim"         : input channel dimension
            - "hidden_dim" : hidden dimension (H)
            - "nomask"      : bool, True → ignore mask

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32
    """
    # -----------------------------------------------------------------
    # 0️⃣ unpack & constants
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim
    out_dim = 5 * H                       # left, right, left_gate, right_gate, out_gate

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm statistics (mean & rstd) – Triton
    # -----------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()        # [R, dim] fp32
    mean_buf = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd_buf = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_D_STATS = 64
    grid_stats = (triton.cdiv(total_rows, BLOCK_M_STATS),)
    layernorm_stats_kernel[grid_stats](
        x_flat,
        mean_buf,
        rstd_buf,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_D=BLOCK_D_STATS,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the four projections + gates + out‑gate into ONE GEMM
    # -----------------------------------------------------------------
    #   Gather original (fp32) weights
    left_proj_w   = weights["left_proj.weight"]          # [H, dim]
    right_proj_w  = weights["right_proj.weight"]         # [H, dim]
    left_gate_w   = weights["left_gate.weight"]          # [H, dim]
    right_gate_w  = weights["right_gate.weight"]         # [H, dim]
    out_gate_w    = weights["out_gate.weight"]           # [H, dim]

    weight_orig = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )                     # shape [5H, dim]  (fp32)

    # γ and β from the first LayerNorm
    gamma = weights["norm.weight"]   # [dim]   fp32
    beta  = weights["norm.bias"]     # [dim]   fp32

    # ----- pre‑compute fused quantities (stay on CPU) -----
    #   w̃ = γ * W         (fp16 for the GEMM)
    weight_tilde = (weight_orig * gamma[None, :]).to(torch.float16)   # [5H, dim] fp16

    #   bias_i = Σ_d β_d * W_i,d         (fp32)
    bias_original = (weight_orig * beta[None, :]).sum(dim=1)          # [5H] fp32

    #   Σ_d w̃_i,d = Σ_d γ_d * W_i,d    (fp32)
    sum_w_tilde = (weight_orig * gamma[None, :]).sum(dim=1)           # [5H] fp32

    # ----- GEMM: X (fp16) @ weight_tildeᵀ (fp16) → S (fp16) -----
    x_fp16 = x_flat.to(torch.float16)                     # [R, dim] fp16
    S = F.linear(x_fp16, weight_tilde)                    # [R, 5H] fp16

    # ----- Apply the closed‑form LayerNorm + Linear ----------
    # all arithmetic in fp32, final cast to fp16
    S_fp32 = S.to(torch.float32)                         # [R,5H] fp32
    mean_exp = mean_buf[:, None]                         # [R,1]
    rstd_exp = rstd_buf[:, None]                         # [R,1]
    sum_w_exp = sum_w_tilde[None, :]                     # [1,5H]
    bias_exp   = bias_original[None, :]                  # [1,5H]

    fused_raw_fp32 = (S_fp32 - mean_exp * sum_w_exp) * rstd_exp + bias_exp
    fused_raw = fused_raw_fp32.to(torch.float16)          # [R,5H] fp16

    # -----------------------------------------------------------------
    # 3️⃣ Split fused tensor into the four branches + out‑gate
    # -----------------------------------------------------------------
    left_raw      = fused_raw[:, 0:H]                # [R, H]
    right_raw     = fused_raw[:, H:2*H]
    left_gate_raw = fused_raw[:, 2*H:3*H]
    right_gate_raw= fused_raw[:, 3*H:4*H]
    out_gate_raw  = fused_raw[:, 4*H:5*H]

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask, sigmoid gates and element‑wise multiplication
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (broadcastable)
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()               # will be used after 2nd LN

    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update (batched GEMM)
    # -----------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring H to the batch dimension : [B·H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat   = left_perm.view(B * H, N, N)
    right_mat  = right_perm.view(B * H, N, N)

    # batched matmul: (N×N) @ (N×N)ᵀ → [B·H, N, N]  (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (row‑wise)
    # -----------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_M2 = 128
    BLOCK_D2 = 64
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast back to float32
    # -----------------------------------------------------------------
    # out_norm_fp16 : [R, H]   fp16
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)   # element‑wise gate

    # final linear projection: [R, H] @ [dim, H]ᵀ → [R, dim]
    to_out_w_fp16 = weights["to_out.weight"].to(torch.float16)    # [dim, H]
    final_fp16 = F.linear(out_norm_fp16, to_out_w_fp16)          # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
