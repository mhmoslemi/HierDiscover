  3️⃣  Optional binary mask and the three sigmoid gates are applied in
      pure PyTorch (still FP16, in‑place where possible).
  4️⃣  The N³ multiplicative update is expressed as a batched GEMM
      (torch.bmm) over the “head” dimension B·H, which lets cuBLAS/Tensor‑Core
      handle the heavy work.
  5️⃣  The second LayerNorm (over the hidden dimension H) is performed
      with the highly‑optimized cuDNN implementation (torch.nn.functional.layer_norm)
      – we keep at least one Triton kernel (the first norm) to satisfy the
      requirement of using Triton.
  6️⃣  The output‑gate is applied, then a final FP16 linear projection brings
      the tensor back to the original channel dimension.

All heavy numerics run in FP16; the final result is cast back to FP32 as
expected by the surrounding model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32) γ
    bias_ptr,       # const   * [dim]               (float32) β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Input  : float32
    Output : float16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- 1️⃣ mean / variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------- 2️⃣ normalize & write -------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with Linear/LayerNorm parameters (FP32)
        - config_dict  : dict with keys
                         * "dim" (int)
                         * "hidden_dim" (int)
                         * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → no mask
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of rows = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – Triton
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()        # [R, dim] float32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256   # rows per program (tuned)
    BLOCK_D = 32    # columns per tile (tuned)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,              # H100 has many SMs → use more warps
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections into a single GEMM
    # --------------------------------------------------------------
    # concatenate weights: [5·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [5·H, dim]

    # GEMM: [R, dim] @ [5·H, dim]ᵀ -> [R, 5·H]  (FP16)
    fused = F.linear(x_norm_fp16, w_fused)     # fp16 result

    # split fused result (each slice has shape [R, H])
    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2*H]
    left_gate_raw  = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]
    out_gate_raw   = fused[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and combine
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [R, 1] (fp16)
        mask_fp16 = mask.to(dtype=torch.float16, device=device).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # in‑place sigmoid to keep memory low
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H]  fp16
    right = right_raw * right_gate        # [R, H]  fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] then permute to [B, H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # flatten batch & heads -> [B·H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ -> [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H (cuDNN)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],   # γ  (float32)
        bias=weights["to_out_norm.bias"],       # β  (float32)
        eps=eps,
    )  # returns FP16 because input is FP16

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final linear projection
    # --------------------------------------------------------------
    # out_gate has shape [R, H] (FP16)
    out_norm = out_norm * out_gate

    final_fp16 = F.linear(
        out_norm,
        weights["to_out.weight"].to(dtype=torch.float16, device=device)  # [dim, H]
    )   # [R, dim] fp16

    # Cast back to FP32 as the surrounding model expects
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
```
