@triton.jit
def final_norm_gate_proj_kernel(
    out_ptr,               # [R, H] fp16
    out_gate_ptr,          # [R, H] fp16
    weight_ptr,            # [C, H] fp16
    norm_weight_ptr,       # [H] fp32
    norm_bias_ptr,         # [H] fp32
    eps,
    R, H, C,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # compute mean and var per row over H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k0 in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k0 * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Loop over output dim tiles
    for d0 in range(0, tl.cdiv(C, BLOCK_N)):
        offs_d = d0 * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_d = offs_d < C
        # accumulator for this tile
        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        # inner loop over H tiles
        for k0 in range(0, tl.cdiv(H, BLOCK_K)):
            offs_k = k0 * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # load out and gate
            out_vals = tl.load(out_ptr + offs_m[:, None]*H + offs_k[None, :],
                               mask=mask_m[:, None] & mask_k[None, :],
                               other=0.0).to(tl.float32)
            gate_vals = tl.load(out_gate_ptr + offs_m[:, None]*H + offs_k[None, :],
                                mask=mask_m[:, None] & mask_k[None, :],
                                other=0.0).to(tl.float32)

            # normalize
            norm = (out_vals - mean[:, None]) * rstd[:, None]

            # load layernorm weight and bias
            w = tl.load(norm_weight_ptr + offs_k, mask=mask_k, other=0.0)
            b = tl.load(norm_bias_ptr   + offs_k, mask=mask_k, other=0.0)
            # broadcast across rows
            w_row = w[None, :]   # [1, BLOCK_K]
            b_row = b[None, :]   # broadcast
            norm = norm * w_row + b_row

            # apply gate
            gated = norm * gate_vals

            # load linear weight for this output tile

            # accumulate dot product (row wise)

            acc += tl.dot(gated, w_linear.T)   # w_linear.T shape (BLOCK_K, BLOCK_N)

"""
TriMul (outgoing) – fast Triton‑fused implementation
===================================================

The forward pass consists of:

1️⃣ Row‑wise LayerNorm on the input `x`      → custom Triton kernel.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate)
   into a single FP16 GEMM.
3️⃣ Compute the output‑gate with a second GEMM.
4️⃣ (Optional) apply the binary mask to the raw left/right projections.
5️⃣ Apply sigmoid to the three gate tensors and multiply‑gate the projections.
6️⃣ Batched matrix‑multiply `left @ rightᵀ`  → `torch.bmm` on (B·H, N, N).
7️⃣ Fuse the **second** LayerNorm, output‑gate multiplication and the final
   projection (`to_out`) into a single Triton kernel:
   for each row `r` (size = hidden_dim) we
   * compute mean / variance,
   * normalize, apply LN γ/β,
   * multiply by `out_gate`,
   * and finally compute the dot‑product with `to_out.weight`.
   This removes an extra kernel launch and a redundant memory copy.
8️⃣ Cast the result back to FP32.

Both kernels run on FP16 (except the reductions that use FP32) and are tuned
for the H100 with 8 warps per block.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32) γ
    bias_ptr,         # const   * [dim]               (float32) β
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- mean & variance -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- normalize & apply γ/β -------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2 – fused 2nd LayerNorm + out‑gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_final_kernel(
    # inputs
    out_ptr,               # fp16 * [R, H]      (output of the N³ mul)
    out_gate_ptr,          # fp16 * [R, H]
    weight_ptr,            # fp16 * [C, H]      (to_out.weight)
    ln_weight_ptr,         # fp32 * [H]         (to_out_norm.weight)
    ln_bias_ptr,           # fp32 * [H]         (to_out_norm.bias)
    # output
    out_proj_ptr,          # fp16 * [R, C]
    # dimensions
    R, H, C,               # runtime int values
    eps,                   # const float
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_N: tl.constexpr,   # output‑dim tile
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # ---------- row‑wise mean / variance ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        vals = tl.load(out_ptr + offs_m[:, None] * H + offs_k[None, :],
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- loop over output‑dim tiles ---------------------------------
    for d_off in range(0, tl.cdiv(C, BLOCK_N)):
        offs_d = d_off * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_d = offs_d < C

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for k_off in range(0, tl.cdiv(H, BLOCK_K)):
            offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # load raw out and gate values
            out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_k[None, :],
                               mask=mask_m[:, None] & mask_k[None, :],
                               other=0.0).to(tl.float32)
            gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_k[None, :],
                                mask=mask_m[:, None] & mask_k[None, :],
                                other=0.0).to(tl.float32)

            # ----- LayerNorm ---------------------------------------------------
            norm = (out_vals - mean[:, None]) * rstd[:, None]

            w_ln = tl.load(ln_weight_ptr + offs_k, mask=mask_k, other=0.0)   # γ
            b_ln = tl.load(ln_bias_ptr   + offs_k, mask=mask_k, other=0.0)   # β
            norm = norm * w_ln[None, :] + b_ln[None, :]

            # ----- apply output gate -------------------------------------------
            gated = norm * gate_vals

            # ----- linear projection (to_out) -----------------------------------
            w_lin = tl.load(weight_ptr + offs_d[:, None] * H + offs_k[None, :],
                            mask=mask_d[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [BLOCK_N, BLOCK_K]

            # accumulate dot product: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)
            acc += tl.dot(gated, w_lin.T)

        # store tile
        out_store_ptr = out_proj_ptr + offs_m[:, None] * C + offs_d[None, :]
        tl.store(out_store_ptr,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)   # True → no mask
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of (b,i,j) tuples
    H = hidden_dim
    C = dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm → fp16 (custom Triton kernel)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256   # rows per program
    BLOCK_D = 32    # cols per iteration
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],     # γ (float32)
        weights["norm.bias"],       # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates into one GEMM
    #    (torch.linear = FP16 TensorCore GEMM)
    # --------------------------------------------------------------
    # concatenate [left_proj, right_proj, left_gate, right_gate] → [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)        # [R, 4·H]  FP16

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2*H]
    left_gate_raw  = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]

    # output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(dtype=torch.float16, device=device)   # [H, dim]
    )   # [R, H] FP16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gates
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.view(R, 1).to(dtype=torch.float16, device=device)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()           # [R, H] FP16

    left  = left_raw * left_gate          # [R, H] FP16
    right = right_raw * right_gate        # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched matmul)
    #    left, right : [B, H, N, N] → (B·H, N, N)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # batched gemm: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [R, H]
    out_flat = out_mat.view(R, H).contiguous()   # [R, H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Fused 2nd LayerNorm + out‑gate + final projection
    # --------------------------------------------------------------
    final_fp16 = torch.empty(R, C, dtype=torch.float16, device=device)

    # block sizes tuned for H100
    BLOCK_M = 128      # rows per program
    BLOCK_N = 64       # output‑dim tile
    BLOCK_K = 32       # hidden‑dim tile

    grid_final = (triton.cdiv(R, BLOCK_M),)
    fused_final_kernel[grid_final](
        out_flat,                     # out_ptr
        out_gate,                     # out_gate_ptr
        weights["to_out.weight"].to(dtype=torch.float16, device=device),   # weight_ptr [C,H]
        weights["to_out_norm.weight"],   # ln_weight_ptr (fp32)
        weights["to_out_norm.bias"],     # ln_bias_ptr   (fp32)
        final_fp16,                    # out_proj_ptr
        R,
        H,
        C,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Cast back to FP32 and return
    # --------------------------------------------------------------
    out = final_fp16.view(B, N, N, C).to(torch.float32)
    return out
