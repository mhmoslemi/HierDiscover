    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    stride_x_row, stride_x_col,
    stride_y_row, stride_y_col,
    stride_w, stride_b,
    N: tl.constexpr,  # dimension size
    eps: tl.constexpr,
    BLOCK: tl.constexpr,
):
    row = tl.program_id(0)
    off = row * stride_x_row  # base offset for row start
    # Compute mean
    sum_val = tl.zeros([1], dtype=tl.float32)
    var_val = tl.zeros([1], dtype=tl.float32)

    # First pass for mean
    for i in range(0, N, BLOCK):
        col = tl.arange(0, BLOCK)
        mask = (i + col) < N
        # Compute offsets
        x_ptrs = x_ptr + off + (i + col) * stride_x_col
        x = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, mask=mask)

    mean = sum_val / N

    # Second pass for variance
    for i in range(0, N, BLOCK):
        col = tl.arange(0, BLOCK)
        mask = (i + col) < N
        x_ptrs = x_ptr + off + (i + col) * stride_x_col
        x = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32)
        diff = x - mean
        var_val += tl.sum(diff * diff, mask=mask)

    var = var_val / N

    # Compute rsqrt
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Third pass to write normalized values
    for i in range(0, N, BLOCK):
        col = tl.arange(0, BLOCK)
        mask = (i + col) < N
        x_ptrs = x_ptr + off + (i + col) * stride_x_col

        # store to y

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure weight dtype matches input dtype.
    # Perform first LayerNorm with Triton
    B, N, _, C = input_tensor.shape
    # flatten (B,N,N) into rows
    x = input_tensor
    x_flat = x.view(-1, dim).contiguous()
    # apply layernorm
    x_norm_flat = triton_layer_norm(x_flat, norm_weight, norm_bias, eps)
    x_norm = x_norm_flat.view(B, N, N, dim)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.to(x_norm.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Triangular multiplication (outgoing) as batch matmul
    # Prepare tensors for batched matmul
    left_perm = left.permute(0, 3, 1, 2)   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1) # (B, hidden_dim, N, N) where last two axes are (k,j)
    out = torch.matmul(left_perm, right_perm)  # (B, hidden_dim, N, N)
    out = out.permute(0, 2, 3, 1)               # (B, N, N, hidden_dim)

    # Apply to_out_norm (layernorm over hidden_dim)
    out_flat = out.view(-1, hidden_dim).contiguous()
    out_norm_flat = triton_layer_norm(out_flat, to_out_norm_weight, to_out_norm_bias, eps)
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # Apply out gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)

    return out_final

"""
TriMul (outgoing) – Triton‑fused implementation
================================================

This file defines a `custom_kernel` entry‑point that computes the forward
pass of the “triangle multiplicative update (outgoing)” module used in
AlphaFold‑3.  The implementation follows the reference PyTorch code but
off‑loads the two LayerNorms to a small Triton kernel, keeping the rest
of the heavy work (linear projections, the N‑by‑N‑by‑N batch multiplication
and the final linear) in regular PyTorch which already uses highly
optimised cuBLAS kernels.

Algorithm
---------
1. **LayerNorm over the channel dimension (`dim`)**  
   Implemented with a custom Triton kernel (`layernorm_kernel`).  
   The kernel processes each row of a flattened `[B*N*N, dim]` tensor,
   computes mean/variance and writes the normalised result with the
   learned weight/bias.

2. **Linear projections** (no bias) for *left* and *right* tensors.

3. **Optional mask** – applied elementwise before the gates.

4. **Gating** – sigmoid of linear projections of the normalised input
   (`left_gate`, `right_gate`, `out_gate`).  The left/right tensors are
   multiplied by their respective gates.

5. **Outgoing triangular multiplication**  
   For each hidden channel `d` compute  
   ``out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]``.  
   This is exactly a batched matrix multiplication:
   `left.permute(0,3,1,2) @ right.permute(0,3,2,1)`.

6. **LayerNorm over the hidden dimension (`hidden_dim`)** – again with the
   Triton kernel.

7. **Out‑gate** – elementwise multiplication with `out_gate`.

8. **Final linear projection** back to `dim`.

All tensors stay on the GPU, the kernels run on an H100 and the final
output is in `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm kernel (generic over any normalisation length)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *const float* input [M, N]
    y_ptr,            # *float* output [M, N]
    weight_ptr,       # *const float* weight [N]
    bias_ptr,         # *const float* bias   [N]
    stride_x_row,     # stride between rows in x
    stride_x_col,     # stride between columns in x (normally 1)
    stride_y_row,     # stride between rows in y
    stride_y_col,     # stride between columns in y
    stride_w,         # stride for weight (normally 1)
    stride_b,         # stride for bias   (normally 1)
    N,                # normalisation dimension (runtime)
    eps,              # epsilon for stability (runtime)
    BLOCK: tl.constexpr,   # tile size for the inner loop
):
    row = tl.program_id(0)                     # one program instance per row (M)

    # ---------- first pass: compute mean and variance ----------
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    for offset in range(0, N, BLOCK):
        col = tl.arange(0, BLOCK)
        mask = (offset + col) < N                # mask out‑of‑bounds
        x_off = x_ptr + row * stride_x_row + (offset + col) * stride_x_col
        x = tl.load(x_off, mask=mask, other=0.0).to(tl.float32)

        sum_val  += tl.sum(x)                    # Σ x
        sumsq_val += tl.sum(x * x)               # Σ x²

    mean = sum_val / N
    var  = sumsq_val / N - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalised values ----------
    for offset in range(0, N, BLOCK):
        col = tl.arange(0, BLOCK)
        mask = (offset + col) < N
        x_off = x_ptr + row * stride_x_row + (offset + col) * stride_x_col
        x = tl.load(x_off, mask=mask, other=0.0).to(tl.float32)

        # normalise
        x_norm = (x - mean) * inv_std

        # weight & bias
        w_off = weight_ptr + (offset + col) * stride_w
        b_off = bias_ptr   + (offset + col) * stride_b
        w = tl.load(w_off, mask=mask, other=0.0)
        b = tl.load(b_off, mask=mask, other=0.0)

        y = x_norm * w + b

        y_off = y_ptr + row * stride_y_row + (offset + col) * stride_y_col
        tl.store(y_off, y.to(tl.float32), mask=mask)


def triton_layer_norm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     block: int = 128) -> torch.Tensor:
    """
    Apply LayerNorm using the Triton kernel defined above.
    Expected shapes:
        x     : (M, N)
        weight: (N,)
        bias  : (N,)
    Returns a tensor of the same shape as `x`.
    """
    assert x.dim() == 2 and weight.dim() == 1 and bias.dim() == 1
    M, N = x.shape
    # Ensure contiguous layout for stride calculations
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = torch.empty_like(x)

    grid = (M,)
    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        stride_x_row=x.stride(0),
        stride_x_col=x.stride(1),
        stride_y_row=y.stride(0),
        stride_y_col=y.stride(1),
        stride_w=weight.stride(0),
        stride_b=bias.stride(0),
        N=N,
        eps=eps,
        BLOCK=block,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Input:
        data = (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]
        mask         : torch.Tensor [B, N, N]   (may be None or ignored)
        weights      : dict with all LayerNorm/Linear weights (no bias)
        config       : dict containing at least "dim", "hidden_dim",
                       optionally "nomask".
    Output:
        torch.Tensor of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # extract model weights
    # ------------------------------------------------------------------
    norm_weight      = weights["norm.weight"]
    norm_bias        = weights["norm.bias"]
    left_proj_w      = weights["left_proj.weight"]
    right_proj_w     = weights["right_proj.weight"]
    left_gate_w      = weights["left_gate.weight"]
    right_gate_w     = weights["right_gate.weight"]
    out_gate_w       = weights["out_gate.weight"]
    to_out_norm_w    = weights["to_out_norm.weight"]
    to_out_norm_b    = weights["to_out_norm.bias"]
    to_out_w         = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm (over `dim`) – Triton fused
    # --------------------------------------------------------------
    B, N, _, C = input_tensor.shape
    # flatten the first three axes so each row corresponds to a vector over `dim`
    x_flat = input_tensor.view(-1, dim).contiguous()
    x_norm_flat = triton_layer_norm(x_flat, norm_weight, norm_bias, eps=eps, block=128)
    x_norm = x_norm_flat.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x_norm, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_w)  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # --------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask is [B, N, N]; broadcast on the hidden dimension
        mask_exp = mask.to(x_norm.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4) Gating signals (sigmoid of linear projections of the *normalised* input)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))   # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Outgoing triangular multiplication (batched matmul)
    # --------------------------------------------------------------
    # reshape for batched matmul: (B, hidden_dim, N, N) x (B, hidden_dim, N, N).T
    left_perm  = left.permute(0, 3, 1, 2)   # [B, hidden_dim, N, i]
    right_perm = right.permute(0, 3, 2, 1)  # [B, hidden_dim, k, j] (k aligns with i)
    out = torch.matmul(left_perm, right_perm)   # [B, hidden_dim, N, N]
    out = out.permute(0, 2, 3, 1)                # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension (Triton)
    # --------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim).contiguous()
    out_norm_flat = triton_layer_norm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps, block=128)
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 7) Apply out gating
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate

    # --------------------------------------------------------------
    # 8) Final linear projection back to `dim`
    # --------------------------------------------------------------
    out_final = F.linear(out_norm, to_out_w)   # [B, N, N, dim]

    return out_final
