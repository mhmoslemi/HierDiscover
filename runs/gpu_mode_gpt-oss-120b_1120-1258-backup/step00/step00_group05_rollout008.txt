"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The implementation follows the reference PyTorch module:
  1) Layer‑norm over the input channel dimension.
  2) Linear projections (left, right) and three sigmoid gates.
  3) A custom Triton kernel fuses mask broadcasting, left/right masking,
     and multiplication with the corresponding gates.
  4) The N‑wise reduction 
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
     is performed as a batched matrix‑multiply (torch.bmm) after
     reshaping the tensors to (B·H, N, N).
  5) Layer‑norm over the hidden dimension, multiplication by the output gate,
     and a final linear projection back to the original channel dimension.

Only the mask‑gate fusion is executed on Triton; the heavy N³ reduction
uses cuBLAS via torch.bmm, which is optimal on H100.
"""

import torch
import triton
import triton.language as tl
import math

# ----------------------------------------------------------------------------- #
# Triton kernel: apply mask (broadcasted over hidden_dim) and element‑wise
# multiplication with the left/right gates.
# ----------------------------------------------------------------------------- #
@triton.jit
def mask_gate_kernel(
    left_ptr,            # const float*  (B,N,N,H)
    right_ptr,           # const float*  (B,N,N,H)
    mask_ptr,            # const float*  (B,N,N)
    left_gate_ptr,       # const float*  (B,N,N,H)
    right_gate_ptr,      # const float*  (B,N,N,H)
    left_out_ptr,        # float*        (B,N,N,H)
    right_out_ptr,       # float*        (B,N,N,H)
    B, N, H,                         # scalar dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    BLOCK_SIZE: tl.constexpr
):
    """Fuse mask & gate multiplication."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * H
    valid = offs < total

    # ---------- decode flat offset into (b,i,k,h) ----------
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    k = rem2 // H
    h = rem2 % H

    # ---------- compute linear indices ----------
    mask_index   = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    left_index   = b * stride_left_b + i * stride_left_i + k * stride_left_k + h * stride_left_h
    right_index  = b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    lg_index     = b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + h * stride_lg_h
    rg_index     = b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + h * stride_rg_h

    # ---------- loads (masked) ----------
    mask_val      = tl.load(mask_ptr + mask_index,  mask=valid).to(tl.float32)   # shape (BLOCK,)
    left_val      = tl.load(left_ptr + left_index,    mask=valid).to(tl.float32)
    right_val     = tl.load(right_ptr + right_index,  mask=valid).to(tl.float32)
    left_gate_val = tl.load(left_gate_ptr + lg_index, mask=valid).to(tl.float32)
    right_gate_val= tl.load(right_gate_ptr + rg_index,mask=valid).to(tl.float32)

    # ---------- fused multiply ----------
    gated_left  = left_val  * mask_val * left_gate_val
    gated_right = right_val * mask_val * right_gate_val

    # ---------- stores ----------
    tl.store(left_out_ptr  + left_index,  gated_left,  mask=valid)
    tl.store(right_out_ptr + right_index, gated_right, mask=valid)


# ----------------------------------------------------------------------------- #
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float, on CUDA)
            - weights     : dict of all linear and layer‑norm parameters
            - config      : dict containing at least:
                * "dim"         – input channel dimension
                * "hidden_dim"  – hidden dimension used in projections
                * "nomask"      – bool, True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------------- #
    # unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    # shortcuts to weight tensors
    # --------------------------------------------------------------------- #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    eps = 1e-5  # LayerNorm epsilon – matches the PyTorch default

    # --------------------------------------------------------------------- #
    # 1) Input LayerNorm
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2) Linear projections & gates (no bias)
    # --------------------------------------------------------------------- #
    left      = torch.nn.functional.linear(x, left_proj_weight)      # [B,N,N,hidden_dim]
    right     = torch.nn.functional.linear(x, right_proj_weight)     # [B,N,N,hidden_dim]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # --------------------------------------------------------------------- #
    # 3) Prepare mask (broadcast to hidden_dim)
    # --------------------------------------------------------------------- #
    nomask = config.get("nomask", True)
    if nomask or mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=torch.float32, device=input_tensor.device)
    else:
        mask = mask_tensor.to(dtype=torch.float32, device=input_tensor.device)

    # --------------------------------------------------------------------- #
    # 4) Fuse mask + gate multiplication with a Triton kernel
    # --------------------------------------------------------------------- #
    BLOCK_SIZE = 1024
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    # Allocate output buffers (same shape & dtype)
    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Ensure contiguity for the mask (required for stride calculations)
    mask_contig = mask.contiguous()

    mask_gate_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        mask_ptr=mask_contig.data_ptr(),
        left_gate_ptr=left_gate.data_ptr(),
        right_gate_ptr=right_gate.data_ptr(),
        left_out_ptr=left_out.data_ptr(),
        right_out_ptr=right_out.data_ptr(),
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask_contig.stride(0), mask_contig.stride(1), mask_contig.stride(2),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # --------------------------------------------------------------------- #
    # 5) N³ reduction via batched matrix‑multiply (torch.bmm)
    #    left_out:  [B, N, N, hidden_dim]
    #    right_out: [B, N, N, hidden_dim]
    #    Result   : [B, N, N, hidden_dim]
    # --------------------------------------------------------------------- #
    # Bring hidden dimension to the second axis so we can view (B*H, N, N)
    left_perm  = left_out.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * hidden_dim, N, N)          # (B·H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)         # (B·H, N, N)

    # Batched matmul: sum over k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)

    # Restore original layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------------- #
    # 6) Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, to_out_weight)   # back to [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # Return the final tensor (float32, CUDA)
    # --------------------------------------------------------------------- #
    return out
