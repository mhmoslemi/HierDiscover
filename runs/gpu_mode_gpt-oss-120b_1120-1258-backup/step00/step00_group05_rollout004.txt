def batched_matmul_triton(A, B, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32):
    # Assume A shape (B*H, N, N) and B shape (B*H, N, N) both contiguous.
    BH, M, K = A.shape
    _, N, _ = B.shape

    # Compute strides in elements
    stride_ah = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bh = B.stride(0)
    stride_bn = B.stride(1)
    stride_bk = B.stride(2)

    # Output shape BH, M, N
    C = torch.empty((BH, M, N), dtype=A.dtype, device=A.device)

    stride_ch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # Launch kernel
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (BH * num_blocks_m, num_blocks_n)

    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bn, stride_bk,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

"""
custom_kernel for the outgoing TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]).
2. Linear projections (no bias) to a hidden dimension `H`:
      left  = Linear(x, left_proj.weight)   # [B,N,N,H]
      right = Linear(x, right_proj.weight)  # [B,N,N,H]
3. Optional mask multiplication (broadcast on the hidden dim).
4. Gating:
      left_gate  = sigmoid(Linear(x, left_gate.weight))   # [B,N,N,H]
      right_gate = sigmoid(Linear(x, right_gate.weight))  # [B,N,N,H]
      out_gate   = sigmoid(Linear(x, out_gate.weight))    # [B,N,N,H]
   Apply the gates:  left  = left  * left_gate
                     right = right * right_gate
5. Compute the core “tri‑multiplicative” update:
      out_hidden[i,j] = Σₖ left[i,k] * right[j,k]   (per batch and per hidden channel)
   This is exactly a batched matrix multiplication
      out_hidden = left @ rightᵀ
   The heavy reduction is performed by a custom Triton kernel
   (`batched_matmul_kernel`) that processes many (batch·H) matrix pairs
   in parallel using tiling (BLOCK_M × BLOCK_N) and a K‑loop.
6. Layer‑norm the hidden result, multiply by `out_gate`,
   and linearly project back to the original channel size `C`.
7. Return the final tensor of shape [B,N,N,C].

The kernel works with float‑32 tensors (the whole forward pass is
performed in FP32 for numerical robustness).  The input and all
weights are cast to float‑32 at the beginning and the final result is
cast back to the original input dtype.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A,B shape: [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M=N=seq_len, K=seq_len)
    stride_ah, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bh, stride_bn, stride_bk,   # strides for B (batch, N, K)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)          # (batch·tile_m) axis
    pid1 = tl.program_id(1)          # tile_n axis

    # ------------------------------------------------------------------
    # Decode batch id and tile coordinates
    # ------------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id = pid0 // num_tiles_m
    tile_m   = pid0 % num_tiles_m
    tile_n   = pid1

    # Offsets for the output tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N
    offs_m = tl.where(mask_m, offs_m, 0)
    offs_n = tl.where(mask_n, offs_n, 0)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        offs_k = tl.where(mask_k, offs_k, 0)

        # Load A tile (M × K)
        a_ptrs = (A_ptr + batch_id * stride_ah
                  + offs_m[:, None] * stride_am
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (K × N) – note B is used transposed
        b_ptrs = (B_ptr + batch_id * stride_bh
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr + batch_id * stride_ch
              + offs_m[:, None] * stride_cm
              + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float32),
             mask=mask_m[:, None] & mask_n[None, :])


def batched_matmul_triton(A: torch.Tensor,
                          B: torch.Tensor,
                          BLOCK_M: int = 64,
                          BLOCK_N: int = 64,
                          BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B : [B*H, N, N]  (float32, contiguous)
    Returns C : [B*H, N, N] (float32)
    """
    assert A.is_contiguous() and B.is_contiguous()
    BH, M, K = A.shape
    _, N, _ = B.shape
    device = A.device

    C = torch.empty((BH, M, N), dtype=A.dtype, device=device)

    # Strides are expressed in number of elements (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    num_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (BH * num_tiles_m, num_tiles_n)

    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bn, stride_bk,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (same dtype as the input)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Cast everything to FP32 for numerical stability (the kernel works in FP32)
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float32)
    if mask is not None:
        mask = mask.to(torch.float32)

    # Cast all weight tensors to FP32
    wp = {k: v.to(torch.float32) for k, v in weights.items()}

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=wp["norm.weight"],
        bias=wp["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, wp["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, wp["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask if present and not disabled
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, wp["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, wp["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, wp["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched matmul kernel
    #   shape needed: (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # _ == hidden_dim
    # Permute to bring hidden dim to the second axis and flatten batch + hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # ------------------------------------------------------------------
    # Core TriMul: out_hidden = left @ rightᵀ   (batched matmul)
    # ------------------------------------------------------------------
    out_flat = batched_matmul_triton(left_flat, right_flat,
                                     BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)  # [B*H, N, N]

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out_hidden = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Output layer‑norm over hidden dimension
    # ------------------------------------------------------------------
    out_hidden_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=wp["to_out_norm.weight"],
        bias=wp["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Apply the final gating and output projection
    # ------------------------------------------------------------------
    out_gated = out_hidden_norm * out_gate
    out = torch.nn.functional.linear(out_gated, wp["to_out.weight"])   # [B, N, N, dim]

    # Cast back to the original dtype (e.g. fp16 / bf16) before returning
    return out.to(orig_dtype)
