"""
TriMul (outgoing) kernel for AlphaFold‑3 style pair‑wise updates.

Algorithm
---------
1. Layer‑normalize the 4‑D input (B, N, N, C) with learned weight/bias.
2. Project the normalized tensor to a hidden dimension H via three linear layers:
   * left  = x_norm @ left_proj_weight.T
   * right = x_norm @ right_proj_weight.T
3. (Optional) Apply a binary mask (broadcast on the hidden axis).
4. Compute element‑wise gates with sigmoids and multiply them:
   left  = left  * sigmoid(x_norm @ left_gate_weight.T)
   right = right * sigmoid(x_norm @ right_gate_weight.T)
5. **Core O(N³) operation** – for each hidden slice d perform:
        out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
   Implemented as a fused batched GEMM using a Triton kernel:
   `out_d = left_d @ right_d^T`  (fp16 × fp16 → fp32 accumulator).
6. Apply a second LayerNorm (learned weight/bias) on the hidden axis.
7. Multiply by an output gate `sigmoid(x_norm @ out_gate_weight.T)`.
8. Final linear projection back to the original channel dimension C.

Only the heavy O(N³) matrix multiplication is off‑loaded to Triton;
all element‑wise and LayerNorm work is performed with PyTorch for simplicity.
The kernel works for any batch size B∈{1,2}, sequence length N up to 1024,
and hidden dimension H up to 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ  (per batch‑hidden slice)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_out_kernel(
    left_ptr,            # fp16 pointer   [B,N,N,H]
    right_ptr,           # fp16 pointer   [B,N,N,H]
    out_ptr,             # fp32 pointer   [B,N,N,H]
    N: tl.constexpr,    # sequence length
    # strides for left
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr, BATCH: tl.constexpr
):
    pid_m = tl.program_id(0)      # block row  (i)
    pid_n = tl.program_id(1)      # block col  (j)
    pid_z = tl.program_id(2)      # flattened (batch, hidden)

    # --------------------------------------------------------------
    # Decode the (batch, hidden) pair from pid_z
    # --------------------------------------------------------------
    b = pid_z // HIDDEN_DIM          # batch index
    h = pid_z - b * HIDDEN_DIM        # hidden index (0 ≤ h < HIDDEN_DIM)

    # Base pointers for the current (b,h) slice
    left_base   = left_ptr  + b * stride_left_b   + h * stride_left_h
    right_base  = right_ptr + b * stride_right_b  + h * stride_right_h
    out_base    = out_ptr   + b * stride_out_b    + h * stride_out_h

    # --------------------------------------------------------------
    # Offsets for the current output tile (M×N)
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----------------------------------------------
        # Load left tile  (M × K)
        # ----------------------------------------------
        left_ptrs = left_base + \
            offs_m[:, None] * stride_left_i + \
            k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # ----------------------------------------------
        # Load right tile transposed (K × N)
        #   rightᵀ[k, j] == right[j, k]  -> use stride_right_i for j,
        #   stride_right_k for k.
        # ----------------------------------------------
        right_ptrs = right_base + \
            k_offset[:, None] * stride_right_k + \
            offs_n[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # ----------------------------------------------
        # Accumulate dot product
        # ----------------------------------------------
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Write out the result tile
    # --------------------------------------------------------------
    out_ptrs = out_base + \
        offs_m[:, None] * stride_out_i + \
        offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    use_mask = not config.get("nomask", True)

    # ------------------------------------------------------------------
    # Convenience aliases for the learned parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    # Input shape: [B, N, N, dim]
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias,
                          eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    # shape after proj: [B, N, N, hidden_dim]
    left = F.linear(x_norm, left_proj_weight)      # bias is None
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (binary 0/1)
    # ------------------------------------------------------------------
    if use_mask and mask_tensor is not None:
        # mask: [B, N, N] -> broadcast on hidden dim
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³) multiplication using Triton
    # ------------------------------------------------------------------
    # Convert to fp16 for the GEMM (half‑precision compute)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output buffer (FP32 accumulator)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    B, N, _, _ = left_fp16.shape

    # Strides (as python ints)
    lsb, lsi, lsk, lsh = (s.item() for s in left_fp16.stride())
    rsb, rsi, rsk, rsh = (s.item() for s in right_fp16.stride())
    osb, osi, osj, osh = (s.item() for s in out_fp32.stride())

    # Triton block sizes – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,        # blocks in M dimension (i)
        (N + BLOCK_N - 1) // BLOCK_N,        # blocks in N dimension (j)
        B * hidden_dim                       # one program per (batch, hidden)
    )

    tri_mul_out_kernel[grid](
        left_fp16, right_fp16, out_fp32,
        N,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim, BATCH=B,
        num_warps=4,                         # heuristic for H100
    )

    # ------------------------------------------------------------------
    # 6) Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out_fp32,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim] → [B, N, N, dim]
    out = F.linear(out, to_out_weight)   # bias is None

    return out
