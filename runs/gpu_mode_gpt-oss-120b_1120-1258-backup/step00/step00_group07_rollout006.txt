"""
TriMul (outgoing) custom kernel.

The heavy O(N³) part of the TriMul forward pass is the
einsum:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

For each channel `d` this is a matrix multiplication
    left[b,:, :,d]  (N×N)  @  right[b,:, :,d].T  (N×N)
We fuse this multiplication in a Triton kernel that processes all
(b × hidden_dim) matrices in parallel (batch‐hidden dimension).  The
kernel implements a tiled GEMM `C = A @ Bᵀ` with float16 inputs and
float32 accumulation.  The rest of the TriMul pipeline (LayerNorm,
gates, final linear) is performed in PyTorch.

Key steps
---------
1. Layer‑norm the input.
2. Linear projections → `left`, `right`.
3. (Optional) mask multiplication.
4. Gating (sigmoid on linear transforms) and element‑wise gating.
5. Fuse the `left @ rightᵀ` contraction with a Triton kernel.
6. Layer‑norm on the contracted tensor, multiply by output gate.
7. Final linear projection back to `dim`.

The kernel works for any batch size (1 or 2), any sequence length
up to 1024 and hidden dimensions 128/384 as required by the test
suite.  It returns a float32 tensor with the same shape as the
reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B are (B·H, N, N) tensors, fp16)
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (all = seq_len)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_n, stride_b_k,   # strides for B (original shape N×K)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  with tiling.
    A : (batch, M, K)  fp16
    B : (batch, N, K)  fp16 (will be read as K×N for the transpose)
    C : (batch, M, N)  fp32 accumulator → stored as fp32
    """
    pid_batch = tl.program_id(0)          # batch × hidden dimension
    pid_m = tl.program_id(1)              # tile row index
    pid_n = tl.program_id(2)              # tile col index

    # ---- 1‑D offsets for this tile -----------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---- Masks for out‑of‑bounds -------------------------------------
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- Base pointers for this batch --------------------------------
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # ---- Pointers to the beginning of the current tile ----------------
    #   A tile: (BLOCK_M, K)
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m
    #   B tile (will be accessed as K×BLOCK_N after transpose)
    b_ptrs = b_batch_ptr + offs_n[None, :] * stride_b_n
    #   C tile (BLOCK_M × BLOCK_N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n

    # ---- Accumulator (float32) ---------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- Loop over K dimension in BLOCK_K chunks --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        cur_offs_k = cur_k + offs_k
        mask_k = cur_offs_k < K

        # Load A block (BLOCK_M × BLOCK_K)
        a_mask = mask_m[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs + cur_offs_k[None, :] * stride_a_k,
                    mask=a_mask, other=0.0, dtype=tl.float16)

        # Load B block transposed (BLOCK_K × BLOCK_N)
        # Original B layout is (N, K); we read it as (K, N)
        b_mask = mask_k[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs + cur_offs_k[:, None] * stride_b_k,
                    mask=b_mask, other=0.0, dtype=tl.float16)

        # GEMM accumulation
        acc += tl.dot(a, b)

    # ---- Store the result --------------------------------------------
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


def trimul_gemm(A: torch.Tensor, B: torch.Tensor, C: torch.Tensor,
                BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    """
    Launches `trimul_matmul_kernel` for tensors:
        A : (B·H, N, N)  fp16
        B : (B·H, N, N)  fp16
        C : (B·H, N, N)  fp32 (output)
    """
    assert A.shape == B.shape == C.shape
    batch, M, K = A.shape
    N = B.shape[1]               # M == N == K == seq_len
    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Custom forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused Triton kernel.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Unpack configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weights (they are already on the correct device)
    # ------------------------------------------------------------------
    def W(name): return weights[name]

    # LayerNorm on input (dim)
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=W("norm.weight"),
        bias=W("norm.bias"),
        eps=eps,
    )

    # --------------------------------------------------------------
    # Linear projections
    # --------------------------------------------------------------
    # proj: (bs, N, N, hidden_dim)
    left = F.linear(x_norm, W("left_proj.weight"))   # bias=False in definition
    right = F.linear(x_norm, W("right_proj.weight"))

    # --------------------------------------------------------------
    # Optional mask
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [bs, N, N] → broadcast on channel dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, W("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x_norm, W("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x_norm, W("out_gate.weight")))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Prepare tensors for the Triton GEMM
    # --------------------------------------------------------------
    # Cast to fp16 for the kernel (all heavy work in fp16)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    bs, N, _, _ = left_h.shape  # shape = [B, N, N, hidden_dim]

    # Move hidden dimension to the leading position and make it contiguous
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & hidden_dim → treat each channel as an independent GEMM
    batch_hidden = bs * hidden_dim
    left_flat = left_perm.reshape(batch_hidden, N, N)   # (B·H, N, N)
    right_flat = right_perm.reshape(batch_hidden, N, N)

    # Output buffer (float32 as required by the reference implementation)
    out_flat = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Fused GEMM: out_flat = left_flat @ right_flatᵀ  (float32 result)
    # --------------------------------------------------------------
    # Block sizes are tuned for H100 – they work well for N up to 1024.
    trimul_gemm(left_flat, right_flat, out_flat,
                BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # --------------------------------------------------------------
    # Reshape back to [bs, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # Post‑contraction LayerNorm over hidden_dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=W("to_out_norm.weight"),
        bias=W("to_out_norm.bias"),
        eps=eps,
    )

    # --------------------------------------------------------------
    # Multiply by output gate and final linear layer back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, W("to_out.weight"))   # shape (bs, N, N, dim)

    return out
