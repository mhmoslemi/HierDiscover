    j = remain % N

    b_vec = tl.load(bias_ptr + d_offs, mask=mask, other=0.0)

    # store output
    tl.store(output_ptr + offset + d_offs * stride_d, y.to(tl.float16), mask=mask)

"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑Norm the input tensor x ∈ [B, N, N, D] (float32) → x̂ (float16).
   This is done with a custom Triton kernel that computes per‑position
   mean/variance over the last dimension D and applies weights/biases.

2. Project x̂ with four linear layers (no bias):
   * left_proj, right_proj  → shape [B, N, N, H]
   * left_gate , right_gate → shape [B, N, N, H] (sigmoid)
   * out_gate               → shape [B, N, N, H] (sigmoid)
   The projects are executed with torch.nn.functional.linear in fp16
   (the weight matrices are cast to fp16).

3. Apply the optional mask (broadcast on the hidden dimension) and the gates:
   left  = left_proj  * mask * left_gate
   right = right_proj * mask * right_gate

4. Core TriMul contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix multiplication for each hidden
   channel h.  By reshaping to (B·H, N, N) we can reuse cuBLAS via
   torch.bmm, which is highly optimized on H100.

5. Layer‑Norm the result over the hidden dimension H using the same
   Triton kernel (weights/bias = to_out_norm.{weight,bias}).

6. Gated output:  out ← out * out_gate

7. Final linear projection back to dim D (no bias) with torch.nn.functional.linear.

All heavy reductions (layer‑norm) are performed in Triton, the O(N³·H)
contraction uses cuBLAS (torch.bmm) which on an H100 with fp16
Tensor‑Cores comfortably meets the sub‑millisecond target.
The final tensor is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,      # pointers (float32 → float16 output)
    weight_ptr, bias_ptr,  # layer‑norm parameters (float16)
    stride_b, stride_i, stride_j, stride_d,  # strides of the 4‑D input
    B, N, D, eps,          # dimensions and epsilon
    BLOCK_D: tl.constexpr, # compile‑time block size for the channel dim
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # ----- decode (b,i,j) from linear pid -----
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base offset of the D‑slice for this (b,i,j)
    base = b * stride_b + i * stride_i + j * stride_j

    # ----- first pass: compute mean and variance -----
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for d in range(0, D, BLOCK_D):
        d_idx = d + tl.arange(0, BLOCK_D)
        mask = d_idx < D
        # load a BLOCK_D chunk of the last dimension
        x = tl.load(inp_ptr + base + d_idx * stride_d, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: apply affine transform and store -----
    for d in range(0, D, BLOCK_D):
        d_idx = d + tl.arange(0, BLOCK_D)
        mask = d_idx < D
        x = tl.load(inp_ptr + base + d_idx * stride_d, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + d_idx, mask=mask, other=0.0).to(tl.float32)
        b_ = tl.load(bias_ptr   + d_idx, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std * w + b_
        tl.store(out_ptr + base + d_idx * stride_d, y.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul update.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor shape [B, N, N, D], float32
            - mask: torch.Tensor shape [B, N, N] (bool/float) or None
            - weights: dict of the model parameters (see comment above)
            - config: dict with keys "dim" (D) and "hidden_dim" (H)

    Returns:
        torch.Tensor shape [B, N, N, D], float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Unpack configuration
    # ------------------------------------------------------------------
    D = config["dim"]
    H = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Cast weight tensors to fp16 (Tensor Core friendly)
    # ------------------------------------------------------------------
    to_fp16 = lambda t: t.to(device).to(torch.float16)
    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias   = to_fp16(weights["norm.bias"])

    left_proj_weight  = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])

    left_gate_weight  = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight   = to_fp16(weights["out_gate.weight"])

    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_fp16(weights["to_out_norm.bias"])

    to_out_weight = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1) Layer‑norm the input (over D)
    # ------------------------------------------------------------------
    x_norm = torch.empty_like(input_tensor, dtype=torch.float16)

    stride_b, stride_i, stride_j, stride_d = input_tensor.stride()
    BLOCK_D = 32  # works for D up to 384
    grid = (B * N * N,)
    layernorm_kernel[grid](
        input_tensor,                       # inp_ptr  (float32)
        x_norm,                             # out_ptr  (float16)
        norm_weight, norm_bias,             # LN parameters
        stride_b, stride_i, stride_j, stride_d,
        B, N, D, eps,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_ctas=1,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections + gates
    # ------------------------------------------------------------------
    # flatten the first three dims to use torch.nn.functional.linear
    flat = B * N * N
    x_norm_flat = x_norm.view(flat, D)          # (B·N·N, D)  fp16

    # Projections (no bias)
    left_proj = F.linear(x_norm_flat, left_proj_weight)   # (flat, H)
    right_proj = F.linear(x_norm_flat, right_proj_weight)

    # Reshape back to 4‑D
    left_proj = left_proj.view(B, N, N, H)
    right_proj = right_proj.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask_f = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_f = mask.to(device).to(torch.float16)
    mask_f = mask_f.unsqueeze(-1)                     # (B,N,N,1)

    left_proj = left_proj * mask_f
    right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm_flat, left_gate_weight).view(B, N, N, H).sigmoid()
    right_gate = F.linear(x_norm_flat, right_gate_weight).view(B, N, N, H).sigmoid()
    out_gate   = F.linear(x_norm_flat, out_gate_weight).view(B, N, N, H).sigmoid()

    # Apply gates to the projected tensors
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul contraction  (batched GEMM over the K dimension)
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the batch axis: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)          # (B·H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # out_flat[i] = left_flat[i] @ right_flat[i].T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B·H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 6) Layer‑norm on the hidden dimension H (to_out_norm)
    # ------------------------------------------------------------------
    out_norm = torch.empty_like(out, dtype=torch.float16)
    stride_bo, stride_oi, stride_oj, stride_od = out.stride()
    BLOCK_H = 32
    grid_out = (B * N * N,)
    layernorm_kernel[grid_out](
        out, out_norm,
        to_out_norm_weight, to_out_norm_bias,
        stride_bo, stride_oi, stride_oj, stride_od,
        B, N, H, eps,
        BLOCK_D=BLOCK_H,
        num_warps=4,
        num_ctas=1,
    )

    # ------------------------------------------------------------------
    # 7) Gate the normalized result and final linear projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                            # (B,N,N,H) fp16
    gated_flat = gated.view(flat, H)                       # (B·N·N, H)
    final_fp16 = F.linear(gated_flat, to_out_weight)      # (flat, D)

    # Cast to float32 as required by the API
    final = final_fp16.view(B, N, N, D).to(torch.float32)

    return final
