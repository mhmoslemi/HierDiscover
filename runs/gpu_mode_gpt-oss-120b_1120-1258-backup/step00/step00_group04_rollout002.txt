    mask_i = offs_i < N

    # Initialize accumulator

"""
TriMul forward pass accelerated with Triton.

- LayerNorm, linear projections and gating are performed in PyTorch.
- The heavy N³ interaction (einsum '... i k d, ... j k d -> ... i j d')
  is rewritten as a batched matrix multiplication for each hidden
  dimension: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d] =
  (left[b,:, :, d]) @ (right[b,:, :, d]).T.
- A custom Triton kernel computes this product block‑wise (64×64 i‑j
  tiles, 32‑wide k tiles) in FP16 while accumulating in FP32.
- The kernel is launched over (i‑tile, j‑tile, batch×hidden) grid.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,    # pointer to left tensor   [B,N,N,H] (FP16)
    right_ptr,   # pointer to right tensor  [B,N,N,H] (FP16)
    out_ptr,     # pointer to output tensor [B,N,N,H] (FP16)

    B, N, H,                     # runtime scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_I: tl.constexpr,       # i‑tile size
    BLOCK_J: tl.constexpr,       # j‑tile size
    BLOCK_K: tl.constexpr,       # k‑tile size
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]

    Grid dimensions:
        program_id(0) -> i‑tile
        program_id(1) -> j‑tile
        program_id(2) -> batch*hidden index (b*H + h)
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # i and j coordinates for this tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over the reduction dimension k
    for cur_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = cur_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left block: shape (BLOCK_I, BLOCK_K)
        left_offset = (offs_i[:, None] * stride_left_i) + (k_offset[None, :] * stride_left_k)
        left = tl.load(left_base + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block **transposed**: we want R_T[k, j] = right[b, j, k, h]
        # shape (BLOCK_K, BLOCK_J)
        right_offset = (offs_j[None, :] * stride_right_i) + (k_offset[:, None] * stride_right_k)
        right_T = tl.load(right_base + right_offset,
                          mask=mask_j[None, :] & mask_k[:, None],
                          other=0.0)

        # dot-product over k dimension
        # left: (BLOCK_I, BLOCK_K); right_T: (BLOCK_K, BLOCK_J)
        prod = tl.dot(left, right_T)            # FP16 result
        acc += prod.to(tl.float32)               # accumulate in FP32

    # write back the result (convert to FP16)
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    out_offset = (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_base + out_offset,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool or float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # unpack arguments
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", True)
    eps = 1e-5

    # ----------- 1. LayerNorm over the channel dimension -----------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----------- 2. Linear projections (no bias) ------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ----------- 3. Gating values --------------------------------------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # ----------- 4. Optional mask ---------------------------------------
    if use_mask and mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1] to broadcast
        mask_exp = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ----------- 5. Apply gates -----------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------- 6. Core O(N³) interaction via Triton -------------------
    # Convert to FP16 for the kernel (memory‑friendly & faster)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, _ = left_h.shape

    # Allocate output tensor (FP16) for the kernel
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=x.device)

    # Tiling parameters – chosen to fit an H100 SM comfortably
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        left_h.stride(0),
        left_h.stride(1),
        left_h.stride(2),
        left_h.stride(3),
        right_h.stride(0),
        right_h.stride(1),
        right_h.stride(2),
        right_h.stride(3),
        out_h.stride(0),
        out_h.stride(1),
        out_h.stride(2),
        out_h.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining float‑point heavy ops
    out = out_h.to(torch.float32)

    # ----------- 7. Output LayerNorm + gated scaling --------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # element‑wise gating

    # ----------- 8. Final linear projection back to original dim --------
    out = F.linear(out, weights["to_out.weight"])

    return out
