@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_A_batch, stride_A_i, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_i, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)  # row block
    pid_n = tl.program_id(axis=1)  # col block
    pid_b = tl.program_id(axis=2)  # matrix batch

    # compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A and B for current batch
    a_ptrs = A_ptr + pid_b * stride_A_batch + offs_m[:, None] * stride_A_i + offs_k[None, :] * stride_A_k
    b_ptrs = B_ptr + pid_b * stride_B_batch + offs_k[:, None] * stride_B_k + offs_n[None, :] * stride_B_n

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # use dot
        # shape a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_A_k
        b_ptrs += BLOCK_K * stride_B_k

    # write to C
    c_ptrs = C_ptr + pid_b * stride_C_batch + offs_m[:, None] * stride_C_i + offs_n[None, :] * stride_C_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    # decompress config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)   # defaults to False?

    # run layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # shape [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # permute and contiguous
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    # right transposed along inner axes
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    A_flat = left_perm.view(-1, N, N)
    B_flat = right_T.view(-1, N, N)

    # allocate output
    out_flat = torch.empty((A_flat.shape[0], N, N), dtype=torch.float32, device=input_tensor.device)

    # strides
    a_batch_stride = A_flat.stride(0)
    a_i_stride = A_flat.stride(1)
    a_k_stride = A_flat.stride(2)
    b_batch_stride = B_flat.stride(0)
    b_k_stride = B_flat.stride(1)  # row (k)
    b_n_stride = B_flat.stride(2)  # column (n)
    c_batch_stride = out_flat.stride(0)
    c_i_stride = out_flat.stride(1)
    c_n_stride = out_flat.stride(2)

    # define block sizes based on N
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), A_flat.shape[0])

    batched_matmul_kernel[grid](
        A_flat.data_ptr(),
        B_flat.data_ptr(),
        out_flat.data_ptr(),
        N, N, N,
        a_batch_stride, a_i_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_i_stride, c_n_stride,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back
    out_perm = out_flat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1)  # shape (B,N,N,H)

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul forward (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ [B, N, N, C].
2. Linear projections (no bias) to a hidden dimension H:
      left  = x @ W_left   (size [B,N,N,H])
      right = x @ W_right  (size [B,N,N,H])
3. Optional pair‑wise mask is applied (broadcasted on the hidden axis).
4. Gating:
      left  = left  * sigmoid(x @ W_left_gate)
      right = right * sigmoid(x @ W_right_gate)
5. Compute the core “tri‑multiplicative” update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication
      out = left @ rightᵀ   (per hidden slice).
   This step is performed by a fused Triton kernel that processes
   all (batch × hidden) matrices in parallel.
6. Layer‑norm on the hidden axis, multiply by the output gate
   (sigmoid(x @ W_out_gate)).
7. Final linear projection back to the original channel dimension C.

The heavy part (step 5) is implemented in Triton; the surrounding
linear layers, layer‑norms and gating are expressed with PyTorch
operations for simplicity and maintainability.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply A @ B  (both [M, N, K] in fp16,
# result accumulated in fp32 and stored as fp16).
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A, B, C,                # pointers
    M, N, K,                # matrix dimensions (rows×cols of A and B)
    stride_A_batch, stride_A_i, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_i, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)   # tile row index
    pid_n = tl.program_id(axis=1)   # tile column index
    pid_b = tl.program_id(axis=2)   # which (batch × hidden) matrix

    # ---- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- compute the base pointers for this matrix --------------------
    a_ptrs = A + pid_b * stride_A_batch + offs_m[:, None] * stride_A_i + offs_k[None, :] * stride_A_k
    b_ptrs = B + pid_b * stride_B_batch + offs_k[:, None] * stride_B_k + offs_n[None, :] * stride_B_n

    # ---- accumulate ----------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k_tiles):
        # mask for the current k‑tile
        cur_k = k_iter * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)                     # fp16·fp16 → fp32 accumulation

        # advance to the next k‑tile
        a_ptrs += BLOCK_K * stride_A_k
        b_ptrs += BLOCK_K * stride_B_k

    # ---- write result --------------------------------------------------
    c = acc.to(tl.float16)                     # store as fp16 (saves bandwidth)
    c_ptrs = C + pid_b * stride_C_batch + offs_m[:, None] * stride_C_i + offs_n[None, :] * stride_C_n
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask_c)


# ----------------------------------------------------------------------
# Public entry point used by the evaluator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------------
    # 1. Layer‑norm
    # ---------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]

    # ---------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------------
    # 3. Optional mask
    # ---------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N]; broadcast on hidden dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)            # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ---------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ---------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 5. Batched matrix multiplication with Triton
    #    left @ rightᵀ  per hidden slice.
    # ---------------------------------------------------------------
    # Cast to fp16 for speed; the kernel accumulates in fp32.
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Move hidden dimension to the batch-like axis and make tensors contiguous.
    # Shape after permute: [B, H, N, N]
    left_perm  = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    # Right‑hand side needs a transpose on the last two axes.
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape

    # Flatten the (batch, hidden) axes so that we have a batch of matrices.
    A = left_perm.view(-1, N, N)          # [B*H, N, N]  fp16
    Bmat = right_T.view(-1, N, N)        # [B*H, N, N]  fp16
    C = torch.empty((A.shape[0], N, N), dtype=torch.float16, device=device)

    # Strides for the flattened tensors.
    stride_A_batch = A.stride(0)   # = N*N
    stride_A_i     = A.stride(1)   # = N
    stride_A_k     = A.stride(2)   # = 1

    stride_B_batch = Bmat.stride(0)
    stride_B_k     = Bmat.stride(1)   # row stride (k)
    stride_B_n     = Bmat.stride(2)   # column stride (n)

    stride_C_batch = C.stride(0)
    stride_C_i     = C.stride(1)
    stride_C_n     = C.stride(2)

    # Choose tile sizes – 32 is a good trade‑off for the H100.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        A.shape[0],                       # number of (batch×hidden) matrices
    )

    _batched_matmul_kernel[grid](
        A, Bmat, C,
        N, N, N,
        stride_A_batch, stride_A_i, stride_A_k,
        stride_B_batch, stride_B_k, stride_B_n,
        stride_C_batch, stride_C_i, stride_C_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out_hidden = C.view(B, H, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # ---------------------------------------------------------------
    # 6. Output layer‑norm + output gate
    # ---------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate

    # ---------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ---------------------------------------------------------------
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out
