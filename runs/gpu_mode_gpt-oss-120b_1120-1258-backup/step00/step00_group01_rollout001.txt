import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# define kernel

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # x shape [B,N,N,dim]
    B,N,N2,dim_check = x.shape
    assert N==N2 and dim_check==dim

    # LayerNorm: using weight and bias
    norm_weight = weights['norm.weight'].view(1,1,1,dim)  # broadcast
    norm_bias = weights['norm.bias'].view(1,1,1,dim)
    # Compute mean,var across dim
    # Use torch.mean and torch.var for speed
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps) * norm_weight + norm_bias

    # Linear projections (no bias)
    # weight shapes: left_proj_weight: [hidden_dim, dim] (maybe shape hidden_dim x dim)
    left_proj_weight = weights['left_proj.weight'] # shape [hidden_dim, dim]
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']  # shape [dim, hidden_dim]

    # Apply linear using F.linear
    left = F.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden]
    right = F.linear(x_norm, right_proj_weight)
    # mask
    if mask is not None:
        # mask shape [B,N,N] broadcast over hidden
        mask_ = mask.unsqueeze(-1).to(x.dtype)  # broadcast
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for kernel
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    # shape [B,N,N,hidden]
    # Permute to [B, hidden, N, N] and reshape to [B*hidden, N, N]
    left_mat = left_f16.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_mat = right_f16.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    # Allocate output
    out_f16 = torch.empty_like(left_mat)  # same shape [B*hidden,N,N]

    # Compute strides
    # left_mat stride in elements:
    left_stride_batch, left_stride_m, left_stride_n = left_mat.stride()
    # For left_mat, shape (batch_h, M,N)= (B*hidden, N,N) where stride[0]=N*N, stride[1]=N, stride[2]=1 (contiguous)
    # For right_mat we need row stride and col stride separately:
    right_stride_batch, right_stride_m, right_stride_n = right_mat.stride()
    out_stride_batch, out_stride_m, out_stride_n = out_f16.stride()

    # define block sizes (tuned)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B*hidden_dim, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_T_kernel[grid](
        left_mat, right_mat, out_f16,
        B*hidden_dim, N, N, N,
        left_stride_batch, left_stride_m, left_stride_n,
        right_stride_batch, right_stride_m, right_stride_n,
        out_stride_batch, out_stride_m, out_stride_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out_hidden = out_f16.reshape(B, hidden_dim, N, N).permute(0,2,3,1).to(torch.float32)

    # to_out_norm: layernorm over hidden_dim
    # We have weight and bias shape [hidden_dim]
    out_norm = (out_hidden - out_hidden.mean(-1, keepdim=True)) / torch.sqrt(out_hidden.var(-1, unbiased=False, keepdim=True) + eps)
    out_norm = out_norm * to_out_norm_weight.view(1,1,1,hidden_dim) + to_out_norm_bias.view(1,1,1,hidden_dim)

    out_norm = out_norm * out_gate  # elementwise gating

    # final linear to dim
    out = F.linear(out_norm, to_out_weight)  # shape [B,N,N,dim]
    return out
```

"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch module but off‑loads the
computationally dominant pairwise multiplication (the einsum
'... i k d, ... j k d -> ... i j d') to a custom Triton kernel.
The kernel performs a batched matrix multiplication

    out[b, h] = left[b, h] @ right[b, h].T

for each batch b and hidden channel h (h = hidden_dim).  The rest of the
operations (LayerNorms, linear projections, sigmoids, gating) are carried
out with PyTorch to keep the code simple while still meeting the
requirement that “part of the operations” be implemented in a kernel.
All tensors are assumed to reside on the same CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_t_kernel(
    left_ptr,               # *half
    right_ptr,              # *half
    out_ptr,                # *half
    BATCH, M, N, K,         # int32
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul: out[b] = left[b] @ right[b].T

    left  : [batch, M, K]   (float16)
    right : [batch, N, K]   (float16) – accessed as (n, k) to realise transpose
    out   : [batch, M, N]   (float16)
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -------------------------------------------------------------
    # Block offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for a block of left  [M, K]
        left_ptrs = (
            left_ptr
            + pid_batch * stride_l_batch
            + (offs_m[:, None] * stride_l_m)
            + (offs_k[None, :] * stride_l_k)
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # pointers for a block of right transposed  [N, K]  (right[n, k])
        right_ptrs = (
            right_ptr
            + pid_batch * stride_r_batch
            + (offs_n[:, None] * stride_r_m)
            + (offs_k[None, :] * stride_r_n)
        )
        right = tl.load(
            right_ptrs,
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # dot accumulates in FP32
        acc += tl.dot(left, right)

    # -------------------------------------------------------------
    # Store result
    # -------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_batch * stride_o_batch
        + (offs_m[:, None] * stride_o_m)
        + (offs_n[None, :] * stride_o_n)
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    TriMul forward (outgoing) implementation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [batch, seq_len, seq_len, dim]
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # LayerNorm 1 (over dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]  # shape [dim, hidden_dim]

    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Optional masking (applied after projection)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask = mask.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating (sigmoid of linear transforms of the *pre‑masked* x)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to half for the kernel (half‑precision matmul)
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, _ = left_h.shape

    # reshape to [B * hidden_dim, N, N] to get a batch for the kernel
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer (half)
    out_mat = torch.empty_like(left_mat)

    # -------------------------------------------------------------
    # Prepare strides (in elements, not bytes) for the kernel
    # -------------------------------------------------------------
    stride_l_batch, stride_l_m, stride_l_n = left_mat.stride()
    stride_r_batch, stride_r_m, stride_r_n = right_mat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_mat.stride()

    # -------------------------------------------------------------
    # Triton launch configuration
    # -------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # batch dimension (including hidden)
        triton.cdiv(N, BLOCK_M),           # rows
        triton.cdiv(N, BLOCK_N),           # cols
    )

    batched_matmul_t_kernel[
        grid
    ](
        left_mat,
        right_mat,
        out_mat,
        B * hidden_dim,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_n,
        stride_r_batch,
        stride_r_m,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] (float32)
    # -------------------------------------------------------------
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # Second LayerNorm over hidden_dim
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # Apply output gate and final linear projection back to dim
    # -----------------------------------------------------------------
    out_hidden = out_hidden * out_gate  # gating
    out = F.linear(out_hidden, to_out_weight)  # shape [B, N, N, dim]

    return out
```
