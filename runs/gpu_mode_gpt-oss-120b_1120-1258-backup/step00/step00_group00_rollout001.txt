    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # index for batch*H
    # decode b, h
    total = B * H
    b = pid // H
    h = pid % H

    # i block index (axis=1) for i coordinate
    pid_i = tl.program_id(axis=1)
    # j block index (axis=2) for j coordinate
    pid_j = tl.program_id(axis=2)

    # compute start indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # K dimension loop offsets
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # Get weight tensors...
    # Ensure contiguous
    input_tensor = input_tensor.contiguous()
    if mask is not None:
        mask = mask.contiguous()
    else:
        mask = torch.ones((bs,N,N), dtype=input_tensor.dtype, device=input_tensor.device)

    # LayerNorm x
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if needed
    if not config.get("nomask", False):
        mask_uns = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_uns
        right = right * mask_uns
    else:
        # should already be all ones; but skip multiplication
        pass

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Compute einsum in Triton kernel
    out = trimul_einsum_fused(left, right, bs, N, hidden_dim)

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

"""
TriMul (outgoing) – fast Triton implementation of the heavy
einsum part.

The algorithm follows the reference PyTorch implementation:
1. Layer‑norm over the last channel (C).
2. Linear projections to a hidden dimension H (left/right) and three
   gating projections (left_gate, right_gate, out_gate).
3. Mask is applied after the projections (if a mask is supplied).
4. The two gated tensors are multiplied with a Triton kernel that
   computes, for each hidden channel h,
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]      (1)
   i.e. a batch of H independent N×N matrix multiplications.
5. Layer‑norm on the hidden channel dimension, gating with out_gate,
   and a final linear projection back to the original channel dimension.

Only the O(N³·H) contraction (1) is executed on the GPU with Triton;
all surrounding operations use regular PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the einsum contraction:
#   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,                # const float*  left  [B,N,N,H]
    right_ptr,               # const float*  right [B,N,N,H]
    out_ptr,                 #       float*  out   [B,N,N,H]
    B, N, H,                 # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,   # tile size in the i dimension
    BLOCK_N: tl.constexpr,   # tile size in the j dimension
    BLOCK_K: tl.constexpr,   # tile size in the reduction k dimension
):
    # program ids
    pid = tl.program_id(axis=0)                 # combines batch and hidden dim
    pid_i = tl.program_id(axis=1)               # i‑tile
    pid_j = tl.program_id(axis=2)               # j‑tile

    # decode batch index and hidden channel from pid
    b = pid // H
    h = pid % H

    # offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # [M]
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # [N]

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)       # [K]

        # Masks for out‑of‑bounds loads
        mask_i_k = (offs_i[:, None] < N) & (offs_k[None, :] < N)   # [M, K]
        mask_j_k = (offs_j[:, None] < N) & (offs_k[None, :] < N)   # [N, K]

        # ------------------------------------------------------------------
        # Load left  : shape (M, K)
        #   left[b, i, k, h]
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(left_ptrs, mask=mask_i_k, other=0.0)

        # ------------------------------------------------------------------
        # Load right transposed : shape (K, N)
        #   right_T[k, j, h] = right[b, j, k, h]
        # ------------------------------------------------------------------
        right_T_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k    # k as first axis
            + offs_j[None, :] * stride_right_i    # j as second axis
            + h * stride_right_h
        )
        right_T = tl.load(right_T_ptrs, mask=mask_j_k, other=0.0)

        # ------------------------------------------------------------------
        # Accumulate the dot product for this K‑tile
        # ------------------------------------------------------------------
        # tl.dot works on (M, K) @ (K, N) -> (M, N)
        acc += tl.dot(left, right_T)   # result is FP32 because acc is FP32

    # ------------------------------------------------------------------
    # Write the final block to global memory
    # ------------------------------------------------------------------
    mask_i_j = (offs_i[:, None] < N) & (offs_j[None, :] < N)   # [M, N]
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=mask_i_j)


def trimul_einsum_fused(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Calls the Triton kernel to compute the per‑channel matrix multiplication.
    `left` and `right` must have shape [B, N, N, H] and be contiguous.
    Returns a tensor of the same shape and dtype (float32).
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape

    # allocate output tensor
    out = torch.empty_like(left)

    # strides (int64 -> int32 for the kernel, safe because values < 2^31)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # grid: one slice per (batch, hidden) pair, plus 2‑D tiling over i/j
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    trimul_einsum_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
      - input_tensor: [B, N, N, C] (float32)
      - mask: [B, N, N] (bool/float, may be None)
      - weights: dict with all linear / layer‑norm parameters
      - config: dict containing at least `dim` and `hidden_dim`
    Returns:
      output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]          # input channel dimension C
    hidden_dim = config["hidden_dim"]   # H
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Initial LayerNorm over the channel dimension C
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, C]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) → hidden dimension
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (mask may be None – treat as all‑ones)
    # --------------------------------------------------------------
    if mask is not None:
        # ensure right dtype for multiplication
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Heavy contraction – Triton kernel
    # --------------------------------------------------------------
    # Ensure tensors are contiguous for simple stride handling
    left = left.contiguous()
    right = right.contiguous()
    out = trimul_einsum_fused(left, right)   # [B,N,N,H]

    # --------------------------------------------------------------
    # 6. Post‑processing: layer‑norm over hidden dim, out‑gate, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
