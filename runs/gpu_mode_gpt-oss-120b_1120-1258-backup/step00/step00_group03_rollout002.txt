"""
TriMul custom kernel (outgoing version) – AlphaFold‑3 core update.

The kernel computes the high‑cost part of the TriMul forward pass:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the projected, masked and gated tensors of
shape (B, N, N, H) (H = hidden_dim).  This is equivalent to the
einsum `'... i k d, ... j k d -> ... i j d'`.  The kernel fuses the
matrix‑multiply across the K‑dimension for many (i, j) tiles and many
(batch, hidden) slices in a single launch, using half‑precision for the
input tensors and float‑32 accumulation.

The surrounding Python code handles LayerNorms, linear projections,
gates, and the final linear output using standard PyTorch ops.
Only the O(N³·H) contraction is performed in Triton.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM over the K dimension (the "triangle" multiply)
# ----------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr,          # const * half
    right_ptr,         # const * half
    out_ptr,           # * float
    B, N, H,           # int32: batch, seq_len, hidden_dim
    # strides of left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides of right (original layout; we read it transposed in K dimension)
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides of out (same layout as left/right)
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Computes out = Σ_k left[..., i,k,d] * right[..., j,k,d]"""

    pid_m = tl.program_id(0)               # i‑tile
    pid_n = tl.program_id(1)               # j‑tile
    pid_bh = tl.program_id(2)              # batch * hidden slice

    # ----- decode batch and hidden indices --------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ----- compute tile offsets -------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds i/j (used for the final store)
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Base offsets for the current (b, h) slice
    base_l = b * stride_l_b + h * stride_l_h
    base_r = b * stride_r_b + h * stride_r_h
    base_o = b * stride_o_b + h * stride_o_h

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K ----------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- left tile: shape (BLOCK_M, BLOCK_K) -----------------------
        mask_l = mask_i[:, None] & mask_k[None, :]
        ptr_l = left_ptr + base_l \
                + offs_i[:, None] * stride_l_i \
                + offs_k[None, :] * stride_l_k
        left = tl.load(ptr_l, mask=mask_l, other=0.0).to(tl.float32)

        # ------- right tile (transposed view): shape (BLOCK_K, BLOCK_N) --
        # we need right_T[b, k, j, h] = right[b, j, k, h]
        mask_r = mask_k[:, None] & mask_j[None, :]
        ptr_r = right_ptr + base_r \
                + offs_j[None, :] * stride_r_i \
                + offs_k[:, None] * stride_r_k
        right = tl.load(ptr_r, mask=mask_r, other=0.0).to(tl.float32)

        # ------- accumulate -------------------------------------------------
        acc += tl.dot(left, right)

    # ----- write result ----------------------------------------------------
    ptr_o = out_ptr + base_o \
            + offs_i[:, None] * stride_o_i \
            + offs_j[None, :] * stride_o_k
    tl.store(ptr_o, acc, mask=mask_ij)


# ----------------------------------------------------------------------
# Python entry point – called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, D] (float32)
            - mask  : torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with the module's parameters
            - config : dict containing at least "dim" and "hidden_dim"
                      and optionally "nomask"

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]  # hidden dimension

    # ------------------------------------------------------------------
    # 1) first LayerNorm over the feature dimension D
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, D]

    # ------------------------------------------------------------------
    # 2) Linear projects (dim -> hidden_dim)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        # ensure mask is float for multiplication
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # apply gates
    left = (left * left_gate).to(torch.float16)   # cast to half for the kernel
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Core TriMul contraction – executed by Triton
    # ------------------------------------------------------------------
    # allocate output buffer (float32 for accumulation stability)
    out_intermediate = torch.empty((B, N, N, H), device=device, dtype=torch.float32)

    # Triton block sizes – tuned for H100, works for all tested N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid dimensions
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * H,                                 # (batch, hidden) slices
    )

    # strides (int64) – needed for pointer arithmetic inside the kernel
    stride_l = left.stride()   # (B_stride, i_stride, k_stride, h_stride)
    stride_r = right.stride()
    stride_o = out_intermediate.stride()

    trimul_out_kernel[grid](
        left,
        right,
        out_intermediate,
        B, N, H,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: to_out_norm, out_gate, final Linear
    # ------------------------------------------------------------------
    # LayerNorm over hidden dimension H
    out = torch.nn.functional.layer_norm(
        out_intermediate,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B,N,N,H]

    # Apply the output gate (broadcastable)
    out = out * out_gate

    # Final projection back to D (dim)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
