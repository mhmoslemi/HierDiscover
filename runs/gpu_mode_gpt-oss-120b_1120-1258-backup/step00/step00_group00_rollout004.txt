```python
import triton
import triton.language as tl

@triton.jit
def trimul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # group (batch*hidden)

    # compute block start indices
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute batch and hidden idx
    group = pid_z
    b = group // H
    h = group % H

    # compute base pointers per group
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # masks for valid indices
        i_mask = i_offset < N
        j_mask = j_offset < N
        k_mask = k_offset < N

        # compute pointers for A tile (left)
        a_ptrs = left_base + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        # compute pointers for B_T tile (right transposed)
        # B_T shape: (k, j)
        b_ptrs = right_base + k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_j

        # load
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # convert to fp32 for accumulation if needed
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

    # store result
    # apply final mask for i and j dimensions
    out_ptrs = out_base + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=i_mask[:, None] & j_mask[None, :])
```

import torch
import triton
import triton.language as tl
import math

@triton.jit
def trimul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    group = pid_z
    b = group // H
    h = group % H

    # base pointers for this (b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k_idx in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)

        i_mask = i < N
        j_mask = j < N
        k_mask = k < N

        # tile of left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # tile of right transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # store result
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    """
    Triton implementation of the “outgoing” TriMul operator.
    Performs LayerNorm, linear projections, gating, a batched matrix
    multiplication across the pairwise dimension using a Triton kernel,
    followed by a second LayerNorm, gating and a final linear projection.
    """
    import torch
    import triton

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm over channel dim of input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask:
        # mask shape [B,N,N] -> broadcast to last dim
        m = mask.to(left.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # H == hidden_dim
    # Allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # Strides (in elements)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsj, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # Choose block sizes. These values work well for N up to 1024 and hidden_dim <=384.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # tile size along the summed dimension

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch Triton kernel
    trimul_forward_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        lsb,
        lsi,
        lsk,
        lsh,
        rsb,
        rsj,
        rsk,
        rsh,
        osb,
        osi,
        osj,
        osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,  # empirically good for H100
    )

    # Second LayerNorm on the hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to original dim
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) – Triton‐accelerated implementation.

The operator receives a tensor x of shape [B, N, N, C] (C = dim) and a
pairwise mask.  It performs

1. LayerNorm over the last dimension (dim).
2. Two point‑wise linear projections (left_proj, right_proj) → hidden_dim.
3. Optional mask multiplication.
4. Gated projections using three additional linear layers + sigmoid.
5. **Core step** – a batched matrix multiplication across the summed
   sequence dimension k:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is executed by a custom Triton kernel that processes each
   (batch, hidden) pair as an independent GEMM.
6. LayerNorm over the hidden dimension, multiplication by the output gate,
   and a final linear projection back to dim.

Only the forward pass is required; gradients are not computed.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_forward_kernel(
    left_ptr,    # const * left  [B, N, N, H]
    right_ptr,   # const * right [B, N, N, H]
    out_ptr,     # * out          [B, N, N, H]
    B, N, H,     # scalar sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile over rows   (i)
    pid_j = tl.program_id(1)          # tile over cols   (j)
    pid_z = tl.program_id(2)          # tile over (batch, hidden)

    # ------------------------------------------------------------------
    # Compute coordinates for this tile
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Decode batch (b) and hidden (h) from the third program ID
    group = pid_z
    b = group // H
    h = group % H

    # Base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Accumulator (use fp32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the summed dimension k
    # ------------------------------------------------------------------
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_block * BLOCK_K + tl.arange(0, BLOCK_K)

        i_mask = i < N
        j_mask = j < N
        k_mask = k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # we load right as (k, j) so that the dot product yields (i, j)
        b_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0)

        # Cast to fp32 before dot‑product
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Accumulate block result
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write the computed tile to the output tensor
    # ------------------------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float32),
             mask=i_mask[:, None] & j_mask[None, :])

# ----------------------------------------------------------------------
# Entry‑point function used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : Tuple[torch.Tensor, torch.Tensor, dict, dict]
        (input, mask, weights, config) where
        - input  : [B, N, N, dim] tensor
        - mask   : [B, N, N] (may be all‑ones when `nomask` is True)
        - weights: model parameters (see keys in the code)
        - config : contains ``dim``, ``hidden_dim`` and ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the TriMul output.
    """
    import torch
    import triton

    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] with same dtype as left/right
        mask_broadcast = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of point‑wise linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul: batched GEMM over the pairwise dimension k
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # left.shape == (B, N, N, hidden_dim)
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=left.dtype,
                      device=left.device)

    # Strides (in elements, not bytes)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsj, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # Tile sizes – tuned for N up to 1024 and hidden_dim ≤ 384 on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j
        B * hidden_dim,                 # one program for each (batch, hidden) pair
    )

    trimul_forward_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        lsb, lsi, lsk, lsh,
        rsb, rsj, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # good default for H100
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden_dim), output gate and final linear
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out_norm = out_norm * out_gate

    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
