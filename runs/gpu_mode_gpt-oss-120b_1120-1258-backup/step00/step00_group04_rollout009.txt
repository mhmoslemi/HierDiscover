class TriMul(nn.Module):
    def __init__(self, dim, hidden_dim):
        super().__init__()
        self.norm = nn.LayerNorm(dim)
        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x, mask):
        batch_size, seq_len, _, dim = x.shape
        x = self.norm(x)  # LN over last dim
        left = self.left_proj(x)   # shape [B,N,N,hidden_dim]
        right = self.right_proj(x) # shape [B,N,N,hidden_dim]

        mask = mask.unsqueeze(-1) # shape [B,N,N,1]
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()
        left = left * left_gate
        right = right * right_gate

        # out shape [B,N,N,hidden_dim]
        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # i: first dimension seq (i index) from left,
        # j: second dimension seq (j index) from right,
        # k: third dimension from left/right (common).
        # That is sum over k broadcasting over d.
        out = self.to_out_norm(out) # LN hidden_dim
        out = out * out_gate
        return self.to_out(out)    # linear to dim
```

import torch
import triton
import triton.language as tl

@triton.jit
def batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bj,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Batch offsets
    a_batch_ptr = a_ptr + pid_batch * stride_ab
    b_batch_ptr = b_ptr + pid_batch * stride_bb
    c_batch_ptr = c_ptr + pid_batch * stride_cb

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + offs_k
        # Load A tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (cur_k[None, :] < K), other=0.0)
        # Load B tile
        b_ptrs = b_batch_ptr + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bj)
        b = tl.load(b_ptrs, mask=(cur_k[:, None] < K) & mask_n[None, :], other=0.0)
        # Compute dot
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    """
    Triton implementation of the “outgoing” TriMul operator.
    Performs LayerNorm, linear projections, gating, a batched
    matrix‑multiply (∑_k left[i,k,:]·right[j,k,:]) using a custom
    Triton GEMM kernel, second LayerNorm, output gating and final
    projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Resolve mask
    if mask is None or config.get("nomask", False):

    # ensure mask has same dtype

    # -------------------------------------------------
    #   1. LayerNorm over the trailing channel (dim)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # mean & variance over last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias   # broadcasting over last dim

    # -------------------------------------------------
    #   2. Linear projections (no bias)
    # -------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------------------------------
    #   3. Apply mask (if any)
    # -------------------------------------------------
    mask_exp = mask.unsqueeze(-1)   # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # -------------------------------------------------
    #   4. Gating (sigmoid)
    # -------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    #   5. Core TriMul:  out[b,i,j,d] = Σ_k left[b,i,k,d]·right[b,j,k,d]
    #      Implemented as a batched GEMM (B*hidden) using Triton.
    # -------------------------------------------------
    B, N, _, H = left.shape   # B,N,N,H
    # reshape to (B*H, N, N) where the second operand is transposed
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # k,j order

    # Triton batched matmul
    out_flat = triton_batch_matmul(left_flat, right_flat)
    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # -------------------------------------------------
    #   6. Output LayerNorm over hidden dimension
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # -------------------------------------------------
    #   7. Output gating & final linear projection
    # -------------------------------------------------
    out_norm = out_norm * out_gate   # out_gate shape [B,N,N,hidden]
    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_w)   # [B,N,N,dim]

    return output

"""
TriMul “outgoing” operator – Triton accelerated core.

The forward pass consists of:
1️⃣ LayerNorm over the input channel dimension (`dim`).
2️⃣ Linear projections → left/right tensors (no bias).
3️⃣ Optional pairwise mask & element‑wise sigmoid gates.
4️⃣ Core “triangle” multiplication:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is realised as a *batched* GEMM (batch = B·hidden_dim) using a
   custom Triton kernel.  The right tensor is first transposed so that the
   operation becomes a regular matrix product.
5️⃣ LayerNorm over the hidden dimension (`hidden_dim`) + output gate.
6️⃣ Final linear projection back to `dim`.

Only the heavy N³ reduction (step 4) is executed on Triton; the remaining
operations are performed with native PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B), where
#   A: (batch, M, K)   B: (batch, K, N)   C: (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bj,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch index = B * hidden_dim
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # --- coordinate offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # batch‑level pointers
    a_batch_ptr = a_ptr + pid_batch * stride_ab
    b_batch_ptr = b_ptr + pid_batch * stride_bb
    c_batch_ptr = c_ptr + pid_batch * stride_cb

    # out‑of‑bounds masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- loop over K -------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + offs_k

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (
            offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (cur_k[None, :] < K),
            other=0.0,
        )

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (
            cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(cur_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )

        # matrix multiplication for the tile
        acc += tl.dot(a, b)

    # --- write back ---------------------------------------------------------
    c_ptrs = c_batch_ptr + (
        offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm(a: torch.Tensor, b: torch.Tensor,
                  BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    Input tensors must be contiguous and have shapes
        a: (B, M, K)
        b: (B, K, N)
    Returns c of shape (B, M, N) (float32).
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    c = torch.empty((B, M, N), dtype=a.dtype, device=a.device)

    grid = (
        B,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        M=M,
        N=N,
        K=K,
        stride_ab=a.stride(0),
        stride_ai=a.stride(1),
        stride_ak=a.stride(2),
        stride_bb=b.stride(0),
        stride_bk=b.stride(1),
        stride_bj=b.stride(2),
        stride_cb=c.stride(0),
        stride_ci=c.stride(1),
        stride_cj=c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Entry point – the custom TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated implementation of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weight_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # mask handling
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones(
            input_tensor.shape[:3],
            dtype=input_tensor.dtype,
            device=input_tensor.device,
        )
    else:
        mask = mask_tensor.to(input_tensor.dtype)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (over the last dimension = dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b  # broadcasting over last dim

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present)
    # ------------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul – batched GEMM on (B * hidden_dim)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N) and transpose right for the gemm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Triton batched GEMM: out_flat = left_flat @ right_flat
    out_flat = _batched_gemm(left_flat, right_flat)

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7️⃣ Output gating & final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise gate
    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_w)   # [B,N,N,dim]

    return output
