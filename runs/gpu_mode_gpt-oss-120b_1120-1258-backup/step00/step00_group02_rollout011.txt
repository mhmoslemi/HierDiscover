def custom_kernel(data):
    """
    Custom Triton-accelerated implementation of the outgoing TriMul operator.
    Implements:
    - LayerNorm on input (dim)
    - Linear projections and gates
    - Masking
    - Core O(N^3) operation via a Triton batched matmul kernel that computes,
      for each hidden channel h, left[..., :, :] @ right[..., :, :].T.
    - Output norm, gating, and final linear projection.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs = input_tensor.shape[0]
    N = input_tensor.shape[1]
    # Step 1: LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Step 2: Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Step 3: Apply mask if provided
    if not config.get("nomask", True):
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Step 4: Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare for kernel
    # Permute and reshape to (M,N,K)
    M = bs * hidden_dim
    # left shape (B,N,N,H) -> (B,H,N,N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    # right_t shape (B,N,N,H) -> permute (B,H,N,N) with K swapped to J: (B,H,N,N)
    # Want right_t where we transpose last two dims.
    right_t = right.permute(0, 3, 2, 1).contiguous().view(M, N, N)  # shape (M,N,N) but with k and j swapped relative to original (makes terms right[j,k]).
    
    # Allocate output
    out_flat = torch.empty_like(left_reshape)  # shape (M,N,N), dtype float16?
    # We'll compute in float32, store float32; allocate float32.
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=left.device)

    # Compute strides
    N_int = N
    K_int = N
    stride_am = N_int * K_int
    stride_ai = K_int
    stride_ak = 1
    stride_bm = N_int * K_int
    stride_bi = K_int
    stride_bk = 1
    stride_cm = N_int * N_int
    stride_ci = N_int
    stride_cj = 1

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (M, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # Define kernel (triton function defined later)
    tri_mul_kernel[grid](
        left_reshape,
        right_t,
        out_flat,
        M,
        N,
        K_int,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bi,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape output back to (B,N,N,H)
    out_hidden = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # Step 5: to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate and final linear projection
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_start * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_m * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B tile transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + pid_m * stride_bm + offs_k[:, None] * stride_bi + offs_j[None, :] * stride_bk

    # store result
    c_ptrs = C_ptr + pid_m * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj

"""
custom_kernel: Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Linear projections (left / right) and three sigmoid gates (left, right, out).
3. Optional element‑wise mask on left/right.
4. Gate the projected tensors.
5. Core O(N³·H) operation:
   For each batch and hidden channel h compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batched matrix multiplication of
        left_reshaped  : (M, N, K)   where M = B·H, N = seq_len, K = seq_len
        right_transposed: (M, N, K)   (right with its last two axes swapped).
   The multiplication is performed by a tiled Triton kernel
   (`tri_mul_kernel`) that computes each (i,j) tile with a dot‑product
   reduction over k, accumulating in float32.
6. Reshape the kernel output back to [B, N, N, H].
7. Layer‑norm over the hidden dimension, apply the out‑gate and a final linear
   projection to recover the original channel dimension `dim`.

The heavy N³ reduction is fully fused in Triton, while the remaining
point‑wise ops use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    A_ptr,  # left tensor    [M, N, K]   (float16)
    B_ptr,  # right^T tensor [M, N, K]   (float16)
    C_ptr,  # output tensor  [M, N, N]   (float32)
    M, N, K,
    stride_am, stride_ai, stride_ak,   # strides for A (M,N,K)
    stride_bm, stride_bi, stride_bk,   # strides for B (M,N,K)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids: (m, i_tile, j_tile)
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulation buffer in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k_block in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k_block * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- load A tile (M,N,K) ---------------------------------------
        a_ptrs = (
            A_ptr
            + pid_m * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)  # (BLOCK_M, BLOCK_K)

        # ---- load B tile transposed (need B[k, j]) --------------------
        # B layout is (M, N=j, K=k); we load it as (k, j) to obtain B^T.
        b_ptrs = (
            B_ptr
            + pid_m * stride_bm
            + offs_k[:, None] * stride_bi   # rows: k -> use stride of N dimension
            + offs_j[None, :] * stride_bk   # cols: j
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)  # (BLOCK_K, BLOCK_N)

        # ---- dot product over k -----------------------------------------
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # ---- store result ----------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_m * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor [B, N, N, dim]
              - mask_tensor : torch.Tensor [B, N, N] (may be ignored)
              - weights_dict: contains all model parameters as tensors
              - config_dict : contains keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of projections
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) operation using Triton
    # -----------------------------------------------------------------
    # Cast to fp16 for memory bandwidth, accumulation in fp32 inside the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to (M, N, K) where M = B * hidden_dim, K = N
    M = B * hidden_dim
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)       # (M,N,K)
    # right needs its last two axes swapped before flattening: (B, hidden, K, J)
    right_transposed = right.permute(0, 3, 2, 1).contiguous().view(M, N, N)   # (M,N,K) where K is original k and N is original j

    # output buffer in float32
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=left.device)

    # strides for the contiguous (M,N,K) tensors
    stride_am = N * N          # stride between different m
    stride_ai = N              # stride between rows (i)
    stride_ak = 1              # stride between cols (k)

    stride_bm = N * N
    stride_bi = N
    stride_bk = 1

    stride_cm = N * N
    stride_ci = N
    stride_cj = 1

    # block sizes – tuned for H100
    BLOCK_M = 64   # handles one hidden‑channel (M dimension) per program
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        M,                                 # one program per (batch * hidden)
        (N + BLOCK_M - 1) // BLOCK_M,      # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,      # j‑tiles
    )

    # launch kernel
    tri_mul_kernel[grid](
        left_reshaped,
        right_transposed,
        out_flat,
        M,
        N,
        N,                 # K == N
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bi,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out_hidden = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, out‑gate, and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
