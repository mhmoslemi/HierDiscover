Now kernel:

```
@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    M, C,
    stride_xm, stride_xc,
    stride_outm, stride_outc,
    stride_w, stride_b,
    eps: tl.constexpr,
    BLOCK_C: tl.constexpr
):
    row = tl.program_id(0)
    # total number of blocks
    num_blocks = tl.cdiv(C, BLOCK_C)
    
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)
    
    # loop over blocks
    for block_idx in range(num_blocks):
        c_offset = block_idx * BLOCK_C
        offs_c = c_offset + tl.arange(0, BLOCK_C)
        mask = offs_c < C
        x = tl.load(x_ptr + row * stride_xm + offs_c * stride_xc,
                    mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=0)
        sumsq_val += tl.sum(x_f32 * x_f32, axis=0)
        
    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    
    # second pass to write normalized
    for block_idx in range(num_blocks):
        c_offset = block_idx * BLOCK_C
        offs_c = c_offset + tl.arange(0, BLOCK_C)
        mask = offs_c < C
        x_ptrs = x_ptr + row * stride_xm + offs_c * stride_xc
        x = tl.load(x_ptrs, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        # normalize
        y = (x_f32 - mean) * inv_std
        
        # load weight/bias
        w = tl.load(weight_ptr + offs_c * stride_w, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_c * stride_b, mask=mask, other=0.0).to(tl.float32)
        y = y * w + b
        
        y = y.to(x.dtype)  # back to original dtype
        # store
        tl.store(out_ptr + row * stride_outm + offs_c * stride_outc,
                 y, mask=mask)
```

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; 
    device = x.device
    
    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    
    # flatten first three dims
    B,N,_,D = x.shape
    assert D == dim
    # flatten to tensor of shape [M,R]
    x_reshape = x.reshape(-1, dim) # shape [M, dim]
    # call triton layernorm
    x_norm = triton_layernorm(x_reshape, norm_weight, norm_bias, eps=1e-5)
    # reshape back
    x_norm = x_norm.view(B,N,N,dim)
    
    # linear projections
    left_proj_weight = weights["left_proj.weight"] # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # compute projections using torch.nn.functional.linear (no bias)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    
    # apply mask if provided
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    
    # compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))
    
    left = left * left_gate
    right = right * right_gate
    
    # Batched matmul across hidden_dim
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    
    # Compute out = left_perm @ right_perm.transpose(-2,-1)
    # Use torch.matmul
    out = torch.matmul(left_perm, right_perm.transpose(-2,-1))
    # shape [B, hidden_dim, N, N]
    # permute back
    out = out.permute(0,2,3,1).contiguous()
    
    # to_out_norm
    # flatten first three dims for layernorm again
    out_reshape = out.reshape(-1, hidden_dim)
    out_norm = triton_layernorm(out_reshape, to_out_norm_weight, to_out_norm_bias, eps=1e-5)
    out_norm = out_norm.view(B,N,N,hidden_dim)
    
    # multiply by out_gate
    out = out_norm * out_gate
    
    # final linear to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]
    
    return out
```

"""
custom_kernel: fast forward pass of AlphaFold3's outgoing TriMul.
Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,D]) using a fused Triton kernel.
2. Linear projections to the hidden dimension (H) for the left/right tensors and
   three gating tensors (left_gate, right_gate, out_gate).  No bias terms are used.
3. Apply the pairwise mask (if provided) to left/right.
4. Multiply left/right by their respective sigmoid gates.
5. Compute the “tri‑mul”:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batch of H independent matrix‑multiplications
   (left_permuted @ right_permutedᵀ).  This is performed with torch.matmul
   (cuBLAS) which is highly optimized on H100.
6. Layer‑norm the intermediate tensor (shape [B,N,N,H]) with another fused
   Triton kernel.
7. Apply the output sigmoid gate and a final linear projection back to D.
8. Return the result in float32.

Only the LayerNorm steps are implemented in Triton, satisfying the
requirement that “part of the operations” be a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton fused LayerNorm (per‑row mean/variance, weight & bias)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const *float   (input)
    out_ptr,             # *float         (output)
    weight_ptr,          # const *float   (γ)
    bias_ptr,            # const *float   (β)
    M,                   # int            (number of rows)
    C,                   # int            (features per row)
    stride_xm, stride_xc,      # int stride of input
    stride_outm, stride_outc,  # int stride of output
    stride_w, stride_b,        # int stride of γ/β (normally 1)
    eps,                       # float32 scalar
    BLOCK_C: tl.constexpr,     # compile‑time block size
):
    row = tl.program_id(0)
    if row >= M:
        return

    # ------------------------------------------------------------------
    # First pass: compute sum and sum‑of‑squares for mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)
    offs_c = tl.arange(0, BLOCK_C)
    offset = 0
    while offset < C:
        cur_offs = offset + offs_c            # absolute column indices
        mask = cur_offs < C
        # load a block of the row (might be fp16 → cast to fp32)
        x = tl.load(
            x_ptr + row * stride_xm + cur_offs * stride_xc,
            mask=mask,
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=0)
        sumsq_val += tl.sum(x_f32 * x_f32, axis=0)
        offset += BLOCK_C

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize, apply γ/β and store
    # ------------------------------------------------------------------
    offset = 0
    while offset < C:
        cur_offs = offset + offs_c
        mask = cur_offs < C
        x = tl.load(
            x_ptr + row * stride_xm + cur_offs * stride_xc,
            mask=mask,
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        # (x - μ) / σ
        y = (x_f32 - mean) * inv_std

        # γ and β (assumed contiguous → stride 1)
        w = tl.load(weight_ptr + cur_offs * stride_w, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + cur_offs * stride_b, mask=mask, other=0.0).to(tl.float32)

        y = y * w + b
        # cast back to input dtype (we keep fp32 throughout)
        y = y.to(x.dtype)

        tl.store(
            out_ptr + row * stride_outm + cur_offs * stride_outc,
            y,
            mask=mask,
        )
        offset += BLOCK_C


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Fused LayerNorm using the Triton kernel above.
    x:    [..., C]  (any leading dimensions)
    weight, bias: [C] (γ and β)
    Returns a tensor of the same shape and dtype as x.
    """
    # flatten leading dimensions -> 2‑D tensor (M rows, C columns)
    orig_shape = x.shape
    C = x.shape[-1]
    M = x.numel() // C
    x_flat = x.reshape(M, C)
    out_flat = torch.empty_like(x_flat)

    # launch one program per row
    grid = (M,)
    BLOCK_C = 128  # works for C up to 384 (3 blocks max)

    layernorm_kernel[grid](
        x_flat,
        out_flat,
        weight,
        bias,
        M,
        C,
        x_flat.stride(0),
        x_flat.stride(1),
        out_flat.stride(0),
        out_flat.stride(1),
        weight.stride(0),
        bias.stride(0),
        eps,
        BLOCK_C=BLOCK_C,
    )
    return out_flat.view(orig_shape)


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask_tensor_or_None, weights_dict, config_dict)
    Returns: tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]          # input/output channel dimension D
    hidden_dim = config["hidden_dim"]  # hidden dimension H

    # ------------------------------------------------------------------
    # Unpack weights (all expected to be on the same device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)

    left_proj_weight   = weights["left_proj.weight"].to(device, dtype)   # [H, D]
    right_proj_weight  = weights["right_proj.weight"].to(device, dtype)  # [H, D]

    left_gate_weight   = weights["left_gate.weight"].to(device, dtype)   # [H, D]
    right_gate_weight  = weights["right_gate.weight"].to(device, dtype)  # [H, D]
    out_gate_weight    = weights["out_gate.weight"].to(device, dtype)    # [H, D]

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)  # [H]
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype)    # [H]

    to_out_weight = weights["to_out.weight"].to(device, dtype)  # [D, H]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (γ,β)
    # ------------------------------------------------------------------
    x_norm = triton_layernorm(input_tensor, norm_weight, norm_bias)  # [B,N,N,D]

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, H]
    left  = F.linear(x_norm, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x_norm, right_proj_weight)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Apply pairwise mask if present and not disabled
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask: [B, N, N] -> [B, N, N, 1] broadcastable to left/right
        mask_ = mask.to(dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Compute sigmoid gates from the same normalized input
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Tri‑multiplicative update:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    This is a batch of H independent matrix‑multiplies.
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the front to use torch.matmul (cuBLAS)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Batched GEMM: (B*H) matrix‑multiplications of size N×N
    out_hidden = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # [B, H, N, N]

    # Restore original ordering → [B, N, N, H]
    out_hidden = out_hidden.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm on hidden tensor
    # ------------------------------------------------------------------
    out_norm = triton_layernorm(out_hidden, to_out_norm_weight, to_out_norm_bias)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to D
    # ------------------------------------------------------------------
    out_gated = out_norm * out_gate
    out = F.linear(out_gated, to_out_weight)  # [B,N,N,D]

    # Ensure final dtype is float32 as required
    out = out.to(torch.float32)

    return out
