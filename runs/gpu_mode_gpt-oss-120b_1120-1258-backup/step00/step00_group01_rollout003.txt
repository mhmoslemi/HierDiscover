def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    stride_cm, stride_cn, stride_cb,

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    # compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_n = offs_n < N

"""
TriMul forward (outgoing) implementation with a fused Triton kernel.

The heavy part of the TriMul module is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
einsum, which is equivalent to a batched matrix multiplication
    L_d @ R_dᵀ  (for each hidden channel d).
We implement this step with a custom Triton kernel that performs
    C = A @ B   where A and B are (batch, N, N) tensors of
    dtype float16 and the accumulation is performed in float32.
All other operations (LayerNorm, Linear projections, Sigmoid gates,
masking) are expressed with native PyTorch kernels for simplicity.
The kernel is launched on a flattened batch dimension
    batch = B * hidden_dim, so each channel of each batch element
is processed independently.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : batched matrix multiplication C = A @ B
# A : (BATCH, M, K)    (float16)
# B : (BATCH, K, N)    (float16)
# C : (BATCH, M, N)    (float16, accumulation in float32)
# ----------------------------------------------------------------------
@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A strides
    stride_bk, stride_bn, stride_bb,   # B strides
    stride_cm, stride_cn, stride_cb,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block index in M dimension
    pid_n = tl.program_id(1)   # block index in N dimension
    pid_b = tl.program_id(2)   # batch index

    # Offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for the tail of the matrix
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current block (including the batch offset)
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # float16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # float16

        acc += tl.dot(a, b)                # dot → accumulates in float32

        # Advance pointers to the next k‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result (cast back to fp16)
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _triton_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the TriMul core operation:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a fused Triton batched GEMM.
    All tensors must be float16.
    """
    B, N, _, H = left.shape   # left.shape == (B, N, N, H)
    # Rearrange to (B, H, N, N) → (batch, M, K)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose the last two dimensions of right (K ↔ N)
    right_T = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N)

    batch = B * H
    A = left_perm.reshape(batch, N, N).contiguous()   # (batch, M=N, K=N)
    Bmat = right_T.reshape(batch, N, N).contiguous()  # (batch, K=N, N)

    C = torch.empty_like(A)  # (batch, N, N)

    # Strides (in elements) for the three tensors
    stride_am, stride_ak, stride_ab = A.stride(1), A.stride(2), A.stride(0)
    stride_bk, stride_bn, stride_bb = Bmat.stride(1), Bmat.stride(2), Bmat.stride(0)
    stride_cm, stride_cn, stride_cb = C.stride(1), C.stride(2), C.stride(0)

    # Tunable block sizes – chosen to fit H100 registers and shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _matmul_kernel[grid](
        A, Bmat, C,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = C.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Extract weights (all stored as float32 in the dict)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension (float32 for numerical safety)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], dtype float32

    # Cast to float16 for the inner heavy computation
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_weight.to(torch.float16))
    right = F.linear(x_h, right_proj_weight.to(torch.float16))

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight.to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight.to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul einsum – implemented in Triton
    # ------------------------------------------------------------------
    out = _triton_einsum(left, right)          # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden_dim) + gating + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    out = F.linear(out, to_out_weight.to(torch.float16))   # back to dim

    # Return float32 as per the reference implementation
    return out.to(torch.float32)
