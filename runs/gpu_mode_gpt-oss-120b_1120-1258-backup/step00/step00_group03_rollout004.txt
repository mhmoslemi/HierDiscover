def custom_kernel(data):
    x, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)
    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N, hidden
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # Gates and mask
    mask_exp = mask.unsqueeze(-1)  # shape B,N,N,1; ensure float
    left = left * mask_exp
    right = right * mask_exp
    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # Permute for matmul
    left_perm = left.permute(0,3,1,2).contiguous()   # (B, H, N, N) i,k
    right_perm = right.permute(0,3,2,1).contiguous()  # (B, H, N, N) k,j
    # Compute batched matmul using Triton
    out_perm = tri_mul_matmul(left_perm, right_perm)  # shape (B, H, N, N)
    # Permute back
    out = out_perm.permute(0,2,3,1).contiguous()   # (B, N, N, H)
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # Multiply by out_gate (already shape B,N,N,H)
    out = out * out_gate
    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_weight) # output dim dim
    return out
```

"""
TriMul “outgoing” kernel (AlphaFold3 style).

The operation works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} and performs:

1) Layer‑norm on the last channel (C = dim).
2) Two point‑wise linear projections (left/right) → hidden_dim channels.
3) Optional pairwise mask (broadcast on hidden_dim).
4) Three gating linear layers + sigmoid (left_gate, right_gate, out_gate).
5) Element‑wise gating of the projected tensors.
6) A batched matrix multiplication per hidden channel:
        out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
   which is equivalent to (left @ rightᵀ) for every hidden channel.
   This is the only heavy part – it is implemented with a custom
   Triton kernel that tiles over the i‑ and j‑axes and loops over k.
7) Layer‑norm on the hidden dimension, gate‑multiplication by out_gate,
   and a final linear projection back to dim.

All steps use float32 for numerical stability; the Triton mat‑mul works
in float32 and is fused across the batch×hidden dimension to exploit the
H100 tensor cores.
"""

import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,            # B*H*N*N  (i,k)
    right_ptr,           # B*H*N*N  (k,j)
    out_ptr,             # B*H*N*N  (i,j)

    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1)          # j‑tile
    pid_bh = tl.program_id(2)         # combined batch × hidden

    # decode batch and hidden indices
    bh = pid_bh
    b = bh // H
    h = bh % H

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_j = offs_j < N

    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # pointers to the start of the current tile
    a_ptrs = left_base  + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_idx in range(0, num_k):
        # global k index for this tile (used for masking)
        k_global = k_idx * BLOCK_K + offs_k
        mask_k = k_global < N

        # load tiles with proper masking
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate dot product (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to the next k‑tile
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # write the result
    c_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton mat‑mul for TriMul
# ----------------------------------------------------------------------
def tri_mul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched mat‑mul for TriMul:
        out_{b,h,i,j} = Σ_k left_{b,h,i,k} * right_{b,h,k,j}
    Arguments:
        left, right – tensors of shape (B, H, N, N), contiguous, dtype float32
    Returns:
        out – tensor of shape (B, H, N, N), dtype float32
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, N, _ = left.shape
    out = torch.empty_like(left)

    # tile sizes – chosen to fit H100 tensor‑core working set
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides in elements (not bytes)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left.stride()
    stride_r_b, stride_r_h, stride_r_k, stride_r_j = right.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_k,
        stride_r_j,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp32 on H100
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Parameters
    ----------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        - input tensor    : shape (B, N, N, dim)
        - mask tensor     : shape (B, N, N)   (may be all‑ones when nomask)
        - weights dict    : contains all model parameters
        - config dict     : {'dim': int, 'hidden_dim': int, 'nomask': bool, ...}
    Returns
    -------
    torch.Tensor
        Processed tensor of shape (B, N, N, dim) – dtype float32.
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Grab weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"] # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (hidden_dim,)

    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Point‑wise linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after linear: (B, N, N, hidden_dim)
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional pair‑wise mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)            # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp
    # if nomask, skip – left/right remain unchanged

    # ------------------------------------------------------------------
    # 4) Gating linear layers + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Ensure we are in float32 for the heavy reduction
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    # ------------------------------------------------------------------
    # 5) TriMul reduction via Triton (batched mat‑mul per hidden channel)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) where the second index is the hidden channel.
    # left  : (i, k)   -> we keep as is
    # right : (k, j)   -> we need to transpose the last two axes.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) i‑k
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) k‑j

    # Batched mat‑mul: out_perm[b, h, i, j] = Σ_k left_perm[b, h, i, k] * right_perm[b, h, k, j]
    out_perm = tri_mul_matmul(left_perm, right_perm)   # (B, H, N, N)

    # Permute back to the original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gate, final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
