def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # layernorm
    # Use torch.nn.functional.layer_norm
    # We need to treat last dimension dim
    weight_norm = weights["norm.weight"]
    bias_norm = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight_norm, bias_norm)
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # apply mask if present (mask is shape [bs,N,N])
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # broadcast
        left = left * mask
        right = right * mask
    # gate multiplication
    left = left * left_gate
    right = right * right_gate
    # compute trimul via triton
    out = trimul_outgoing_kernel(left, right, bs, N, hidden_dim, device)
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

hidden_dim. The kernel loads tiles of size BLOCK_M x BLOCK_K from left
and BLOCK_K x BLOCK_N from right (transposed on-the-fly) and accumulates
using fp32 precision. The rest of the TriMul module (layernorms,
linear projections, gates) is performed in PyTorch.
"""
import torch
import triton
import triton.language as tl

# Triton kernel
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_bh = tl.program_id(2) # combined batch * hidden

    # compute batch and hidden channel indices
    b = pid_bh // H
    h = pid_bh % H

    # offsets for i, j
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create masks for boundaries
    mask_i = i_offset < N
    mask_j = j_offset < N

    # base pointers for this (b, h) pair
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers for A (left) and B (right transposed)
        # left: shape (i, k)
        a_ptrs = left_base + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        # right: we need right[j, k] -> we load as (k, j)
        b_ptrs = right_base + k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_i

        # masks
        mask_a = mask_i[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_j[None, :]

        # load tiles
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_base + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    mask_c = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)

def trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Launches the Triton kernel for the Trimul outgoing operation.
    left, right: tensors of shape [B, N, N, H], dtype torch.float16.
    Returns out: tensor of shape [B, N, N, H], dtype torch.float32.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, N2, H = left.shape
    assert N == N2, "Input must be square"
    device = left.device

    # allocate output in float32
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # block sizes (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module using a fused Triton kernel
    for the core einsum. All other operations (LayerNorm, Linear, sigmoid)
    are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    # unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Device
    device = input_tensor.device

    # LayerNorm over last dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections (float32)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates (float32)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply mask if provided
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # Apply gates (still float32)
    left = left * left_gate
    right = right * right_gate

    # Cast to half to reduce memory traffic for the heavy matmul
    left_fp16 = left.half()
    right_fp16 = right.half()

    # Core trimul using Triton (output in float32)
    out = trimul_outgoing(left_fp16, right_fp16)  # shape [B,N,N,hidden_dim]

    # to_out_norm (LayerNorm) over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Multiply with out gate (float32)
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass optimized with Triton.

The heavy part of the TriMul module is the Einstein‑summation
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
which is a batched matrix multiplication of size N×N for every hidden
channel (d) and batch element (b).  This kernel treats the pair (b, d)
as a combined batch dimension and performs a GEMM per batch using
Triton's `tl.dot`.  Input matrices are provided in FP16 to reduce
memory traffic while the accumulation is performed in FP32 for numerical
stability.  All other operations (LayerNorms, Linear projections,
sigmoids, masking) are executed with PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,        # *FP16  [B, N, N, H]
    right_ptr,       # *FP16  [B, N, N, H]
    out_ptr,         # *FP32  [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 2‑D tile indices for the output matrix (i, j)
    pid_m = tl.program_id(0)        # i‑tile
    pid_n = tl.program_id(1)        # j‑tile
    # 1‑D index over the combined (batch, hidden) dimension
    pid_bh = tl.program_id(2)

    # Resolve batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # Global coordinates of the tile
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_i = i_off < N
    mask_j = j_off < N

    # Base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # Pointers to the current tiles
        # left tile: shape (M, K) -> left[i, k, h]
        a_ptrs = left_base \
                 + i_off[:, None] * stride_left_i \
                 + k_off[None, :] * stride_left_k
        # right tile needs right[j, k, h] i.e. we load (K, N) -> right[k, j, h]
        b_ptrs = right_base \
                 + k_off[:, None] * stride_right_k \
                 + j_off[None, :] * stride_right_i

        # Apply masks to avoid OOB loads
        mask_a = mask_i[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_j[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # FP16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # FP16

        # Accumulate (TL dot does FP16×FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # Store the computed tile (FP32)
    c_ptrs = out_base \
             + i_off[:, None] * stride_out_i \
             + j_off[None, :] * stride_out_j
    mask_c = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)


def trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM performed by the Triton kernel.
    left, right: [B, N, N, H] torch.float16 (contiguous)
    Returns:    out       [B, N, N, H] torch.float32
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, N2, H = left.shape
    assert N == N2, "Input matrix must be square."
    device = left.device

    # Allocate output (FP32)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides in *elements* (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tile sizes (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps for H100
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] torch.Tensor (float32)
            - mask:          [B, N, N] torch.Tensor (bool or float) or None
            - weights: dict of model parameters
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        output: [B, N, N, dim] torch.Tensor (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gates (float32, sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4) Optional masking  (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Heavy ternary multiplication via Triton (FP16 inputs, FP32 output)
    # ------------------------------------------------------------------
    left_fp16  = left.half().contiguous()
    right_fp16 = right.half().contiguous()
    out = trimul_outgoing(left_fp16, right_fp16)   # [B, N, N, hidden_dim] FP32

    # ------------------------------------------------------------------
    # 7) Output layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 8) Multiply by the output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
