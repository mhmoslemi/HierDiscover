                    stride_x, stride_out, dim, total_rows,
                    eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    row_mask = offs_m < total_rows
    # initialize sum and sumsq to zeros
    sum = tl.zeros([BLOCK_M], dtype=tl.float32)
    sumsq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        col_mask = offs_d < dim
        x_ptrs = x_ptr + (offs_m[:, None] * stride_x) + (offs_d[None, :] * 1)  # assume Dense
        x = tl.load(x_ptrs, mask=row_mask[:, None] & col_mask[None, :], other=0.0).to(tl.float32)
        sum += tl.sum(x, axis=1)
        sumsq += tl.sum(x * x, axis=1)
    mean = sum / dim
    var = sumsq / dim - mean * mean
    # Now compute normalized values and store.

    x_ptr, out_ptr,

    eps: tl.constexpr,

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # ensure contiguity
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N

    # flatten for layernorm
    x_flat = x.view(total_rows, dim)
    out_norm_flat = torch.empty_like(x_flat)

    # pointers
    x_ptr = x_flat.data_ptr()
    out_ptr = out_norm_flat.data_ptr()
    weight_ptr = weights["norm.weight"].data_ptr()
    bias_ptr = weights["norm.bias"].data_ptr()

    # launch layernorm kernel
    BLOCK_M = 128  # rows per program
    BLOCK_D = 32   # columns per chunk
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_ptr,
        out_ptr,
        weight_ptr,
        bias_ptr,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ensure kernel completion
    torch.cuda.synchronize()

    # reshape normalized tensor
    x_norm = out_norm_flat.view(B, N, N, dim)

    # linear projections and gates using torch.nn.functional.linear
    # flatten again for efficient matmul
    x_norm_flat = x_norm.view(total_rows, dim)

    # left and right projections
    left_proj = torch.nn.functional.linear(x_norm_flat, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm_flat, weights["right_proj.weight"])

    # mask handling
    if config.get("nomask", True):
        # No mask: skip
        pass
    else:
        # mask shape [B,N,N] -> flatten
        mask = mask_tensor.to(x.dtype).contiguous().view(total_rows, 1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # gates
    left_gate = torch.nn.functional.linear(x_norm_flat, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_flat, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_flat, weights["out_gate.weight"]).sigmoid()

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape to 4D [B,N,N,hidden_dim]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # TriMul: compute out = left @ right^T over the N dimension for each hidden_dim
    # Rearrange for batched gemm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k,j)
    B_H = B * hidden_dim
    left_flat = left_perm.view(B_H, N, N)
    right_flat = right_perm.view(B_H, N, N)
    out_flat = torch.bmm(left_flat, right_flat)   # [B*H, N, N]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # to_out_norm: LayerNorm over hidden_dim (size hidden_dim)
    out_flat2 = out.view(B * N * N, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat2)

    # pointers for to_out_norm
    out_ptr_in = out_flat2.data_ptr()
    out_ptr_out = out_norm_flat.data_ptr()
    weight_ptr2 = weights["to_out_norm.weight"].data_ptr()
    bias_ptr2 = weights["to_out_norm.bias"].data_ptr()
    total_rows2 = B * N * N
    dim2 = hidden_dim

    grid2 = (triton.cdiv(total_rows2, BLOCK_M),)
    layernorm_kernel[grid2](
        out_ptr_in,
        out_ptr_out,
        weight_ptr2,
        bias_ptr2,
        total_rows2,
        dim2,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # reshape normalized output
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # apply out_gate
    out_norm = out_norm * out_gate

    # final linear to dim
    out_norm_flat2 = out_norm.view(total_rows, hidden_dim)
    final = torch.nn.functional.linear(out_norm_flat2, weights["to_out.weight"])
    final = final.view(B, N, N, dim)

"""
custom_kernel for the “outgoing” TriMul operator (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the input tensor `x` over the channel dimension (`dim`).
   Implemented with a Triton kernel because it requires a reduction
   (mean/variance) that Torch‑compile cannot fuse efficiently.

2. Linear projections (no bias) and gating:
   * left_proj   = x_norm @ W_left_proj
   * right_proj  = x_norm @ W_right_proj
   * left_gate   = sigmoid(x_norm @ W_left_gate)
   * right_gate  = sigmoid(x_norm @ W_right_gate)
   * out_gate    = sigmoid(x_norm @ W_out_gate)

   If a mask is supplied, it is applied (broadcast) to the two
   projections before gating.

3. Apply the element‑wise gates:
   left  = left_proj  * left_gate
   right = right_proj * right_gate

4. Triangular multiplicative update (the expensive O(N³) part):
   For every hidden channel `h` we need
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   i.e. a matrix‑multiply `L_h @ R_hᵀ`.
   We reshape the tensors to [B·H, N, N] and call `torch.bmm`,
   which on H100 executes as a highly‑optimised GEMM.

5. Layer‑norm over the hidden dimension (`hidden_dim`).
   This second norm is also performed with a Triton kernel
   (identical to the first one, just with a different size).

6. Apply the final output gate and a linear projection back to `dim`.

All heavy reductions (the two LayerNorms) are done in Triton,
the remaining dense linear algebra uses PyTorch’s fused kernels.
The implementation works for all configurations described in the
prompt (different batch sizes, sequence lengths, channel sizes and
with/without a mask).  The final tensor is returned in `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,                # *const float*   [total_rows, dim]
    y_ptr,                # *mut   float*   [total_rows, dim]
    weight_ptr,           # *const float*   [dim]
    bias_ptr,             # *const float*   [dim]
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm (n = dim) for a 2‑D tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows                         # bool mask for rows

    # -----------------------------------------------------------------
    # First pass: compute sum and sum of squares (mean & variance)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer to element (row, col)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: write normalized values with γ (weight) and β (bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcasted over rows)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Prepare tensors (ensure contiguity)
    # -----------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                      # x: [B, N, N, dim]
    total_rows = B * N * N                     # rows for the flattened view

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim)
    # -----------------------------------------------------------------
    x_flat = x.view(total_rows, dim)           # [total_rows, dim]
    x_norm_flat = torch.empty_like(x_flat)     # output buffer

    # launch Triton kernel
    BLOCK_M = 128   # rows per program (tune if needed)
    BLOCK_D = 32    # columns per iteration (tune if needed)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # reshape back to 4‑D tensor
    x_norm = x_norm_flat.view(B, N, N, dim)    # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections + gates (pure PyTorch – highly tuned kernels)
    # -----------------------------------------------------------------
    x_norm_flat = x_norm.view(total_rows, dim)  # [total_rows, dim]

    # ----- Projections (no bias) -------------------------------------------------
    left_proj   = F.linear(x_norm_flat, weights["left_proj.weight"])
    right_proj  = F.linear(x_norm_flat, weights["right_proj.weight"])

    # ----- Optional mask ---------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] -> flatten & unsqueeze for broadcasting
        m = mask_tensor.to(x.dtype).contiguous().view(total_rows, 1)
        left_proj  = left_proj * m
        right_proj = right_proj * m

    # ----- Gates -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm_flat, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm_flat, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm_flat, weights["out_gate.weight"]))

    # ----- Apply element‑wise gates ---------------------------------------------
    left  = left_proj * left_gate          # [total_rows, hidden_dim]
    right = right_proj * right_gate

    # reshape to 4‑D tensors for the TriMul step
    left  = left.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 3) TriMul: L @ Rᵀ for each hidden channel (batched GEMM)
    # -----------------------------------------------------------------
    # Move hidden dimension to the second axis so we can collapse batch×hidden
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k, j)

    BH = B * hidden_dim
    left_mat   = left_perm.view(BH, N, N)    # [B*H, N, N]
    right_mat  = right_perm.view(BH, N, N)   # [B*H, N, N]

    # torch.bmm uses the GPU’s tensor‑core GEMM – very fast on H100
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N]

    # reshape back to 4‑D
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 4) LayerNorm over hidden_dim (second LayerNorm)
    # -----------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    out_norm_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 5) Apply the final output gate and the projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate                       # element‑wise

    # final linear projection (hidden_dim -> dim)
    out_norm_flat = out_norm.view(total_rows_h, hidden_dim)
    final = F.linear(out_norm_flat, weights["to_out.weight"])   # [total_rows_h, dim]
    final = final.view(B, N, N, dim)                           # [B, N, N, dim]

    return final
