TriMul (outgoing) – AlphaFold3 core operation.

The implementation follows the reference PyTorch module but fuses the
mask‑application and the element‑wise gating with a custom Triton kernel.
The heavy pairwise contraction  
    out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d]  
is expressed as a batched matrix multiplication (one GEMM per hidden
dimension) using Torch's highly‑optimised cuBLAS driver.  All LayerNorm
and linear layers are performed with `torch.nn.functional` for numerical
stability; only the mask + gate fusion is realised in Triton.

Input
-------
data = (input_tensor, mask_tensor, weight_dict, config_dict)

output
-------
Tensor of shape [bs, seq_len, seq_len, dim] (float32) – the forward pass
of the outgoing TriMul module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------
# Triton kernel: fuse mask multiplication and gating
# -----------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    # pointers to the contiguous input tensors (flattened view)
    left_proj_ptr,    # float32*
    right_proj_ptr,   # float32*
    left_gate_ptr,    # float32*
    right_gate_ptr,   # float32*
    mask_ptr,         # float32*  (shape [B, N, N] flattened)
    out_left_ptr,     # float32*
    out_right_ptr,    # float32*
    # scalar arguments
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,               # hidden dimension
    total_elems: tl.int32,     # B * N * N * H
    BLOCK_SIZE: tl.constexpr,  # number of elements processed per program
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bounds check
    in_range = offs < total_elems

    # load the four tensors (left/right proj & gates)
    left   = tl.load(left_proj_ptr + offs,  mask=in_range, other=0.0)
    right  = tl.load(right_proj_ptr + offs, mask=in_range, other=0.0)
    lgate  = tl.load(left_gate_ptr + offs,  mask=in_range, other=0.0)
    rgate  = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)

    # mask is stored without the hidden dimension -> replicate across H
    mask_idx = offs // H                 # integer division
    mask_val = tl.load(mask_ptr + mask_idx,
                       mask=in_range,
                       other=1.0)        # if mask missing treat as 1.0

    # fused gating + masking
    out_left  = left  * lgate * mask_val
    out_right = right * rgate * mask_val

    # store results
    tl.store(out_left_ptr + offs, out_left,  mask=in_range)
    tl.store(out_right_ptr + offs, out_right, mask=in_range)


# -----------------------------------------------------------
# Entry point
# -----------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Normalisation of the input (LayerNorm over the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim) – no bias
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_weight)   # [bs, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_weight)  # [bs, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Gate linear layers + sigmoid (also dim -> hidden_dim)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # used later

    # ------------------------------------------------------------------
    # Fuse mask + gate multiplication with Triton (if a mask is present)
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure contiguous layout for safe 1‑D pointer arithmetic
        left_proj  = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # Cast mask to same dtype as the projected tensors
        mask = mask_tensor.to(left_proj.dtype).contiguous()   # [bs, N, N]

        # Allocate output buffers
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        total_elems = left_proj.numel()                      # B * N * N * H
        BLOCK_SIZE = 1024                                    # 1‑K  elements per program

        # Launch Triton kernel
        grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
        mask_gate_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask,
            left,
            right,
            bs,
            seq_len,
            hidden_dim,
            total_elems,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,
        )
        # Release temporary tensors to save memory
        del left_proj, right_proj, left_gate, right_gate, mask
    else:
        # No mask → simple element‑wise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core pairwise contraction: out_{i,j,d} = Σₖ left_{i,k,d} * right_{j,k,d}
    # Implemented as a batch of GEMMs (one per hidden channel)
    # ------------------------------------------------------------------
    # reshape to (B * H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()     # [B, H, N, N]

    left_mat  = left_perm.view(bs * hidden_dim, seq_len, seq_len)
    right_mat = right_perm.view(bs * hidden_dim, seq_len, seq_len)

    # Batched GEMM (left @ rightᵀ) → [B*H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # Apply the output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate                                 # element‑wise gating
    out = F.linear(out, to_out_weight)                  # [B, N, N, dim]

    return out
