"""
TriMul (outgoing) forward pass implemented with a fused Triton GEMM.

Algorithm
---------
1. Layer‑norm over the last dimension (dim).
2. Linear projections → left/right (shape [B, N, N, H]).
3. Optional mask applied on the projected tensors.
4. Gating (sigmoid of linear projections) and element‑wise gating of left/right.
5. Compute the pair‑wise product:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix‑multiply (left @ rightᵀ) for each hidden
   slice.  The heavy N³ work is performed by a custom Triton kernel that
   tiles the GEMM (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates in FP32.
6. Layer‑norm on the hidden dimension, gating with out_gate and a final
   linear projection back to the original dim.

The kernel works on FP16 tensors (left/right) for memory‑bandwidth efficiency
and accumulates in FP32, then writes the result as FP32.  All other
operations are done with PyTorch (they are cheap compared to the GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul reduction.
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix shapes (all = seq_len)
    stride_am, stride_ah, stride_ai, stride_ak,   # A strides: batch, hidden, i, k
    stride_ba, stride_bh, stride_bk, stride_bn,   # B strides: batch, hidden, k, j  (B is read as (k, j))
    stride_cm, stride_ch, stride_ci, stride_cj,   # C strides: batch, hidden, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ  for each (batch, hidden) slice.
    A : (batch, hidden, M, K)  – fp16
    B : (batch, hidden, N, K)  – fp16  (we read it as (K, N))
    C : (batch, hidden, M, N)  – fp32
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # ---- decode batch & hidden indices ----
    batch_idx = pid_z // hidden_dim
    hidden_idx = pid_z - batch_idx * hidden_dim

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_ba + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_ch

    # ---- thread‑local offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- tiled K loop ----
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile  (M × K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile  (K × N)  – note B is accessed as (k, j)
        b_ptrs = b_base + (cur_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # ---- write back ----
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Entry point expected by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, S, S, C]  (C == dim)
            - mask_tensor:  [B, S, S]  (or None)
            - weights_dict: all linear / layer‑norm weights
            - config_dict: {'dim': C, 'hidden_dim': H, 'nomask': bool, ...}

    Returns:
        Tensor of shape [B, S, S, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Helper to fetch weight tensors (already on device)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape [B, S, S, dim]   (float32)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_weight)   # [B, S, S, H]
    right = F.linear(x, right_proj_weight)  # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 3. Optional mask
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, S, S, H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Compute the N³ reduction by a batched GEMM using Triton
    # ------------------------------------------------------------------- #
    # Cast to half for the GEMM kernel (memory‑bandwidth friendly)
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Rearrange to [B, H, S, S] so the hidden dimension is contiguous (stride=1)
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, S, S]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()  # [B, H, S, S]

    # Output buffer (FP32 accumulation)
    out_fp32 = torch.empty_like(left_h, dtype=torch.float32, device=device)  # [B, H, S, S]

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(S, BLOCK_M),          # rows
        triton.cdiv(S, BLOCK_N),          # cols
        B * hidden_dim                    # (batch * hidden)
    )

    _trimul_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left_h.stride(0),  # batch stride
        stride_ah=left_h.stride(1),  # hidden stride (==1)
        stride_ai=left_h.stride(2),  # i (row) stride
        stride_ak=left_h.stride(3),  # k (col) stride
        stride_ba=right_h.stride(0),
        stride_bh=right_h.stride(1),
        stride_bk=right_h.stride(3),  # note: we read B as (k, j)
        stride_bn=right_h.stride(2),  # j stride
        stride_cm=out_fp32.stride(0),
        stride_ch=out_fp32.stride(1),
        stride_ci=out_fp32.stride(2),
        stride_cj=out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
    )

    # ------------------------------------------------------------------- #
    # 6. Restore original layout: [B, S, S, H]
    # ------------------------------------------------------------------- #
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 7. Output LayerNorm (hidden dimension)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 8. Apply out‑gate and final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, S, S, dim]

    return out
