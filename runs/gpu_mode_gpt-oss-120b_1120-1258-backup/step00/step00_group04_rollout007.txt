def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if not nomask
    if not nomask:
        # mask shape [B,N,N]; unsqueeze to broadcast
        mask_expanded = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask_expanded
        right = right * mask_expanded

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast left/right to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # TriMul via Triton
    out_hidden = triton_tri_mul_forward(left_fp16, right_fp16, block_m=64, block_n=64, block_k=64)

    # out_hidden shape [B,N,N,hidden] float32
    # LayerNorm on hidden
    out_hidden = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out_hidden = out_hidden * out_gate

    # Final linear to dim
    out = torch.nn.functional.linear(out_hidden, to_out_weight)  # shape [B,N,N,dim]

    return out

    """

        BLOCK_N=block_n,

    )
    return out

"""
TriMul (outgoing) forward pass with a Triton kernel for the O(N³) reduction.

Algorithm
---------
1. Layer‑Norm over the last dimension (dim).
2. Linear projections → left/right (shape [B,N,N,H]) and three sigmoid‑gates.
3. Apply mask (if present) and element‑wise gates.
4. Compute 
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that treats each (batch, hidden) pair as an
   independent matrix‑multiply (M×K)·(K×N)ᵀ.  The kernel works on FP16 inputs
   and accumulates in FP32.
5. Layer‑Norm on the hidden dimension, multiply by the out‑gate and a final
   linear projection back to dim.

The kernel fuses the innermost “k” reduction and is the performance‑critical
part of the operator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    out_ptr,           # *fp32
    B, N, H,           # runtime sizes
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_jr, stride_kr, stride_hr,   # right strides
    stride_bo, stride_io, stride_jo, stride_ho,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)                     # tile row  (i)
    pid_n = tl.program_id(1)                     # tile col  (j)
    pid_bh = tl.program_id(2)                    # combined batch & head index

    b = pid_bh // H                              # batch index
    h = pid_bh % H                               # hidden‑dim index

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i coordinates
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j coordinates

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this (b,h) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_bo + h * stride_ho

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Reduction over k
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ---- left block (M x K) ------------------------------------------------
        left_ptrs = left_base \
                    + offs_m[:, None] * stride_il \
                    + cur_k[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16).to(tl.float32)

        # ---- right block (K x N) – we load it transposed so that we can use tl.dot
        # right original layout: (b, j, k, h)
        # transposed view we need: (k, j)
        right_ptrs = right_base \
                     + cur_k[:, None] * stride_kr \
                     + offs_n[None, :] * stride_jr
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_n[None, :],
                          other=0.0,
                          dtype=tl.float16).to(tl.float32)

        # ---- dot product over K ------------------------------------------------
        acc += tl.dot(left, right_T)      # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    out_ptrs = out_base \
               + offs_m[:, None] * stride_io \
               + offs_n[None, :] * stride_jo
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that invokes the Triton kernel
# ----------------------------------------------------------------------
def _tri_mul_triton(left_fp16: torch.Tensor, right_fp16: torch.Tensor,
                    block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    left/right: [B, N, N, H]  (fp16, contiguous)
    returns:    [B, N, N, H]  (fp32)
    """
    assert left_fp16.is_contiguous() and right_fp16.is_contiguous()
    B, N, _, H = left_fp16.shape

    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left_fp16.device)

    # strides in element units
    stride_bl, stride_il, stride_kl, stride_hl = left_fp16.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    grid = (
        (N + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
        B * H,
    )

    _tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_jr, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm (weight / bias from the supplied dict)
    # ------------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim] float32

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: [B,N,N]; expand to [B,N,N,1] and match dtype
        mask_exp = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy O(N^3) reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    out_hidden = _tri_mul_triton(left_fp16, right_fp16,
                                 block_m=64, block_n=64, block_k=64)   # [B,N,N,hidden] float32

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_hidden = out_hidden * out_gate

    out = F.linear(out_hidden, weights["to_out.weight"])      # [B,N,N,dim] float32
    return out
