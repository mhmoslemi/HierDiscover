@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_bl, stride_il, stride_kl,
    stride_br, stride_jr, stride_kr,
    stride_bo, stride_io, stride_jo,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for left: (i,k)
    left_ptrs = left_ptr + (pid_batch * stride_bl) + (offs_i[:, None] * stride_il) + (offs_k[None, :] * stride_kl)
    left = tl.load(left_ptrs, mask=(offs_i[:, None] < N) & (offs_k[None, :] < N), other=0.0)

    # Pointers for right: (j,k) - we need right block with shape (BLOCK_N, BLOCK_K)
    right_ptrs = right_ptr + (pid_batch * stride_br) + (offs_j[:, None] * stride_jr) + (offs_k[None, :] * stride_kr)
    right = tl.load(right_ptrs, mask=(offs_j[:, None] < N) & (offs_k[None, :] < N), other=0.0)

    # Compute dot with right transposed
    acc = tl.dot(left, right, trans_b=True)  # (BLOCK_M, BLOCK_N)

    # Store output
    out_ptrs = out_ptr + (pid_batch * stride_bo) + (offs_i[:, None] * stride_io) + (offs_j[None, :] * stride_jo)
    tl.store(out_ptrs, acc, mask=(offs_i[:, None] < N) & (offs_j[None, :] < N))

"""
TriMul (outgoing) – fast Triton implementation.

The core of the TriMul update is a three‑dimensional tensor contraction
```
out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
```
where `b` = batch, `i,j,k` = sequence positions and `d` = hidden dimension.
For each hidden dimension `d` this is exactly a matrix multiplication
`left_d @ right_d.T`.  The kernel fuses the `k`‑loop of this batched
matrix multiplication and runs with Triton 3.3.1 on H100.  All other
operations (LayerNorm, linear projections, gating, final linear) are
performed with regular PyTorch for simplicity.

The kernel works on a flattened batch dimension `B*D` (batch × hidden
size) and uses a 3‑D grid:
    program_id(0) → tile of output rows `i`
    program_id(1) → tile of output columns `j`
    program_id(2) → combined batch‑hidden index.
Tiling sizes `BLOCK_M/N/K` are set to 64, which gives good occupancy for
the target hardware and the supported sequence lengths (256‑1024).
The accumulation is performed in FP32 and the final tensor is returned
as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    left_ptr,               # *Pointer* to left tensor   [B*D, N, N]
    right_ptr,              # *Pointer* to right tensor  [B*D, N, N]
    out_ptr,                # *Pointer* to output tensor [B*D, N, N]
    N: tl.constexpr,        # sequence length
    # strides for left  = (batch, i, k)
    stride_bl, stride_il, stride_kl,
    # strides for right = (batch, j, k) – note that for the multiplication we need the
    #   right tensor **as is** (the kernel will use trans_b=True to treat it as (k, j))
    stride_br, stride_jr, stride_kr,
    # strides for out   = (batch, i, j)
    stride_bo, stride_io, stride_jo,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # combined batch x hidden index
    pid_i = tl.program_id(0)       # output row tile
    pid_j = tl.program_id(1)       # output column tile

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ----------- load left block (i, k) -------------
    left_ptrs = left_ptr + (pid_batch * stride_bl) \
                + (offs_i[:, None] * stride_il) \
                + (offs_k[None, :] * stride_kl)
    left_mask = (offs_i[:, None] < N) & (offs_k[None, :] < N)
    left = tl.load(left_ptrs, mask=left_mask, other=0.0)

    # ----------- load right block (j, k) ------------
    # we will later use trans_b=True, so we load it in (j, k) order.
    right_ptrs = right_ptr + (pid_batch * stride_br) \
                 + (offs_j[:, None] * stride_jr) \
                 + (offs_k[None, :] * stride_kr)
    right_mask = (offs_j[:, None] < N) & (offs_k[None, :] < N)
    right = tl.load(right_ptrs, mask=right_mask, other=0.0)

    # ----------- compute dot product (i, j) ----------
    # left is (M, K), right is (N, K) → dot with trans_b=True gives (M, N)
    acc = tl.dot(left, right, trans_b=True)   # (BLOCK_M, BLOCK_N)

    # ----------- store result ------------------------
    out_ptrs = out_ptr + (pid_batch * stride_bo) \
               + (offs_i[:, None] * stride_io) \
               + (offs_j[None, :] * stride_jo)
    out_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask_tensor  : torch.Tensor [B, N, N] (may be None if config["nomask"] is True)
            - weights_dict : mapping of weight name → torch.Tensor
            - config_dict  : {'dim': int, 'hidden_dim': int, 'nomask': bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device).
    """

    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Load model parameters (all are 1‑D or 2‑D tensors on the same device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1) Initial LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul contraction performed by a Triton kernel
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Re‑order to (B, H, N, N) → flatten batch×hidden for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(B * H, N, N) # [B*H, N, N]

    out_flat = torch.empty_like(left_flat)    # allocate output buffer

    # strides for the flattened tensors (they are contiguous)
    stride_bl = left_flat.stride(0)
    stride_il = left_flat.stride(1)
    stride_kl = left_flat.stride(2)

    stride_br = right_flat.stride(0)
    stride_jr = right_flat.stride(1)  # represents the “row” dimension j
    stride_kr = right_flat.stride(2)  # column k

    stride_bo = out_flat.stride(0)
    stride_io = out_flat.stride(1)
    stride_jo = out_flat.stride(2)

    # block sizes – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # launch grid: (tiles_i, tiles_j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_bl, stride_il, stride_kl,
        stride_br, stride_jr, stride_kr,
        stride_bo, stride_io, stride_jo,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # final projection back to `dim`

    return out
```
