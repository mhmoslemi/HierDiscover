"""
TriMul (outgoing) – fast Triton implementation.

The kernel computes, for each batch `b` and hidden channel `h`,
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

i.e. a batched matrix‑multiply over the sequence dimension `N` for every
hidden dimension.  The surrounding LayerNorms, Linear projections and
gating are done with PyTorch (they are cheap compared to the N³ work).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *float32   [B,N,N, H]
    right_ptr,           # *float32   [B,N,N, H]
    out_ptr,             # *float32   [B,N,N, H]

    B, N, H,             # scalar sizes
    # strides for left (B,N,N,H) layout
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (B,N,N,H) layout (j uses the same stride as i)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out (B,N,N,H) layout
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile along dimension i
    pid_j = tl.program_id(1)          # tile along dimension j
    pid_m = tl.program_id(2)          # combined batch * hidden index

    # ------------------------------------------------------------------
    # decode batch index `b` and hidden index `h` from pid_m
    # ------------------------------------------------------------------
    b = pid_m // H
    h = pid_m % H

    # ------------------------------------------------------------------
    # block offsets
    # ------------------------------------------------------------------
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # bounds checks for i and j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # pointer to the beginning of the current (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k dimension in blocks
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_offset = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ------------------- load left block (i,k) --------------------
        # shape (BLOCK_I, BLOCK_K)
        left_ptrs = (
            left_base
            + i_offset[:, None] * stride_left_i
            + k_offset[None, :] * stride_left_k
        )
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # ------------------- load right block (k,j) -------------------
        # note: we load it transposed so that it has shape (BLOCK_K, BLOCK_J)
        # shape (BLOCK_K, BLOCK_J)
        right_ptrs = (
            right_base
            + k_offset[:, None] * stride_right_k
            + j_offset[None, :] * stride_right_j
        )
        b_mat = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b_mat = b_mat.to(tl.float32)

        # ------------------- matrix multiply & accumulate ---------------
        acc += tl.dot(a, b_mat)      # (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_ptrs = out_base + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing) – uses a Triton kernel for the
    heavy N³ contraction.  All other linear / layer‑norm operations are
    performed with PyTorch.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, C]
            mask:        torch.Tensor [B, N, N]  (or None)
            weights:     dict of model parameters (torch.Tensors)
            config:      dict with at least ``dim`` and ``hidden_dim``,
                         optionally ``nomask`` (bool).

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # optional mask handling
    # ------------------------------------------------------------------
    if mask is None or nomask:
        # all‑ones mask – cheap broadcasting
        mask = torch.ones_like(input_tensor[..., 0], dtype=input_tensor.dtype, device=input_tensor.device)
    mask = mask.to(input_tensor.dtype)
    mask = mask.unsqueeze(-1)                 # [B, N, N, 1]

    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)

    # apply mask before gating (as in the reference implementation)
    left  = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the contraction over k
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty((B, N, N, hidden_dim),
                        dtype=input_tensor.dtype,
                        device=input_tensor.device)

    # strides in *elements* (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()    # (B, i, k, h)
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # tile sizes – tuned for H100
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm → gating → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                         # element‑wise gating
    out = F.linear(out, to_out_weight)           # final projection to C

    return out
