import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(A_ptr, B_ptr, C_ptr,
                   B, N, H,
                   stride_Ab, stride_Ai, stride_Ak, stride_Ah,
                   stride_Bb, stride_Bi, stride_Bk, stride_Bh,
                   stride_Cb, stride_Ci, stride_Ck, stride_Ch,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid0 = tl.program_id(0) # batch*hidden
    pid1 = tl.program_id(1) # i tile
    pid2 = tl.program_id(2) # j tile

    # Compute batch and hidden indices
    batch_idx = pid0 // H
    hidden_idx = pid0 % H

    # Coordinate offsets
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for A (left) – shape (BLOCK_M, BLOCK_K)
        a_ptrs = (A_ptr
                  + batch_idx * stride_Ab
                  + offs_i[:, None] * stride_Ai
                  + offs_k[None, :] * stride_Ak
                  + hidden_idx * stride_Ah)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Pointers for B (right transposed) – shape (BLOCK_K, BLOCK_N)
        b_ptrs = (B_ptr
                  + batch_idx * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_j[None, :] * stride_Bi
                  + hidden_idx * stride_Bh)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate dot product over k
        acc += tl.dot(a, b)

    # Write back to C
    c_ptrs = (C_ptr
              + batch_idx * stride_Cb
              + offs_i[:, None] * stride_Ci
              + offs_j[None, :] * stride_Ck
              + hidden_idx * stride_Ch)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_triton(left, right, BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    """Batched triangular multiplication: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    B, N, _, H = left.shape
    out = torch.empty_like(left)
    # Strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h = out.stride()
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_k, stride_c_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator using a Triton kernel
    for the heavy pairwise multiplication.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # LayerNorm over the last dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Linear projections (bias=False in the reference)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if provided (mask may be all‑ones when nomask=True)
    if config.get("nomask", True):
        mask = torch.ones_like(mask_tensor, dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)  # shape [B, N, N, 1]
    left = left * mask
    right = right * mask

    # Gating signals
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Heavy triangular multiplication via Triton
    out = tri_mul_triton(left, right,
                         BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # Output layer‑norm, gating and final projection
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass.

The heavy part of the TriMul module is the 4‑D tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

with shapes [B, N, N, H].  This is a batched matrix‑multiply where, for each
pair (b, h), we compute left_{b,:, :,h} @ right_{b,:, :,h}^T.  The kernel
below implements this contraction in Triton, tiled over the i and j axes
and looping over the k dimension.  All other operations (LayerNorm,
Linear projections, gating, and final projection) are performed with
standard PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers to left, right (transposed), output
    B, N, H,                           # batch size, sequence length, hidden dim
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,   # strides for A (left)
    stride_Bb, stride_Bi, stride_Bk, stride_Bh,   # strides for B (right)
    stride_Cb, stride_Ci, stride_Ck, stride_Ch,   # strides for C (output)
    BLOCK_M: tl.constexpr,                          # tile size for i‑axis
    BLOCK_N: tl.constexpr,                          # tile size for j‑axis
    BLOCK_K: tl.constexpr,                          # tile size for reduction axis
):
    # ----------------------------------------------------------------------
    # 1) Map program IDs to (batch, hidden) and tile indices (i, j)
    # ----------------------------------------------------------------------
    pid0 = tl.program_id(0)   # combined batch × hidden index
    pid_i = tl.program_id(1)  # tile index along i
    pid_j = tl.program_id(2)  # tile index along j

    batch_idx = pid0 // H           # batch index
    hidden_idx = pid0 % H           # hidden‑channel index

    # ----------------------------------------------------------------------
    # 2) Compute absolute offsets for the output tile (i, j)
    # ----------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑coordinates
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑coordinates

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (high precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # 3) Loop over k (the reduction dimension) in BLOCK_K sized chunks
    # ----------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load a tile of A (left) : shape (BLOCK_M, BLOCK_K) -----
        a_ptrs = (
            A_ptr
            + batch_idx * stride_Ab
            + offs_i[:, None] * stride_Ai
            + offs_k[None, :] * stride_Ak
            + hidden_idx * stride_Ah
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ----- Load a tile of B (right) transposed :
        #   we need right[b, j, k, h] → B_T[k, j, h]
        #   which corresponds to loading (BLOCK_K, BLOCK_N) -----
        b_ptrs = (
            B_ptr
            + batch_idx * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_j[None, :] * stride_Bi
            + hidden_idx * stride_Bh
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # ----- Accumulate the dot product over the k‑dimension -----
        # tl.dot performs a matrix‑multiply on the two tiles and returns
        # a (BLOCK_M, BLOCK_N) matrix.
        acc += tl.dot(a, b)

    # ----------------------------------------------------------------------
    # 4) Write the result back to C
    # ----------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch_idx * stride_Cb
        + offs_i[:, None] * stride_Ci
        + offs_j[None, :] * stride_Ck
        + hidden_idx * stride_Ch
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


def tri_mul_triton(left: torch.Tensor,
                   right: torch.Tensor,
                   BLOCK_M: int = 128,
                   BLOCK_N: int = 128,
                   BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched triangular multiplication using Triton.
    Input tensors must have shape [B, N, N, H] and be contiguous
    on the CUDA device.
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for A, B, C (they are identical for left/right/out)
    sAb, sAi, sAk, sAh = left.stride()
    sBb, sBi, sBk, sBh = right.stride()
    sCb, sCi, sCk, sCh = out.stride()

    grid = (
        B * H,                                   # one program per (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,           # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # j‑tiles
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        sAb, sAi, sAk, sAh,
        sBb, sBi, sBk, sBh,
        sCb, sCi, sCk, sCh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    The expensive pairwise contraction is delegated to a Triton kernel.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if config.get("nomask", True):
        mask = torch.ones_like(mask_tensor, dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype=device)
    mask = mask.unsqueeze(-1)       # shape [B, N, N, 1]

    left = left * mask
    right = right * mask

    # --------------------------------------------------------------
    # 4) Gating signals (before masking they are computed on the
    #    original normalized tensor)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Heavy triangular multiplication via Triton
    # --------------------------------------------------------------
    out = tri_mul_triton(
        left, right,
        BLOCK_M=128, BLOCK_N=128, BLOCK_K=32   # tuned for H100, can be adjusted
    )

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
