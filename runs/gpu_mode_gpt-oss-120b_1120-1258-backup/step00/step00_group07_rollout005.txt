    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; bs = input_tensor.shape[0]; N = input_tensor.shape[1]; # also N = shape[2]; confirm dims.

    # Retrieve norm weight/bias: shape [dim]
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x_normed = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

"""
TriMul (outgoing) forward implementation.

The operation works on a 4‑D tensor x ∈ ℝ^{B × N × N × C}.  The PyTorch reference
first applies a LayerNorm, linear projections (left/right) and three gating
vectors (left_gate, right_gate, out_gate).  After masking and gating the core
computation is an N³ contraction:

    out[b,i,j,d] = Σ_k left[b,i,k,d] · right[b,j,k,d]

which can be written as a batched matrix multiplication
    out_d = left_d @ right_dᵀ   for every hidden channel d.

The kernel below fuses this matmul for all (batch, hidden) pairs, using
FP16 arithmetic with FP32 accumulation.  It tiles the M (i) and N (j) axes
with BLOCK_M×BLOCK_N tiles and iterates over K (the summed dimension) with
BLOCK_K.  The result is reshaped back to the original layout and the remaining
LayerNorm/gating/linear steps are performed with PyTorch.

The implementation works for all configurations described in the problem
statement and targets H100 GPUs (Triton 3.3.1).
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                     # left  tensor   [B*H, M, K]   (FP16)
    b_ptr,                     # right tensor   [B*H, K, N]   (FP16)
    c_ptr,                     # out   tensor   [B*H, M, N]   (FP16)
    M, N, K,                   # dimensions (all equal to seq_len)
    stride_ah, stride_ai, stride_ak,       # strides for a
    stride_bh, stride_bk, stride_bj,       # strides for b
    stride_ch, stride_ci, stride_cj,       # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # combined batch * hidden
    pid_m    = tl.program_id(1)   # tile row   (i)
    pid_n    = tl.program_id(2)   # tile col   (j)

    # ----- block offsets -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ----- accumulator (FP32) -----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over the reduction axis K -----
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Pointers to the current tile of A and B
        a_ptrs = a_ptr + pid_batch * stride_ah \
                 + offs_m[:, None] * stride_ai \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_bh \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bj

        # Masks for out‑of‑bounds elements
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # FP16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # FP16

        # TL.dot accumulates into FP32 when the accumulator is FP32
        acc += tl.dot(a, b)

    # ----- write result -----
    c_ptrs = c_ptr + pid_batch * stride_ch \
             + offs_m[:, None] * stride_ci \
             + offs_n[None, :] * stride_cj
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]

    # Store as FP16; caller will cast to FP32
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Front‑end routine expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:          torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dictionary of model parameters (all float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32) – the TriMul forward output.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype  = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gate multiplicatively
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; broadcast over hidden dim
        mask = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the heavy N³ contraction
    # ------------------------------------------------------------------
    B, N1, N2, H = left.shape
    assert N1 == N2 == N, "Inconsistent sequence dimensions"
    # Permute to [B, H, N, N] so that the hidden dimension becomes the second axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()     # [B, H, N, N] (k,j order)

    # Flatten batch & hidden dimension → shape [B*H, N, N]
    left_flat  = left_perm.view(-1, N, N).to(torch.float16)   # A
    right_flat = right_perm.view(-1, N, N).to(torch.float16)  # Bᵀ

    # Output buffer (FP16)
    out_flat_fp16 = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    batch_hidden = left_flat.shape[0]          # = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat_fp16,
        N, N, N,                                 # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat_fp16.stride(0), out_flat_fp16.stride(1), out_flat_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = (
        out_flat_fp16
        .view(B, hidden_dim, N, N)                # [B, hidden, N, N]
        .permute(0, 2, 3, 1)                      # [B, N, N, hidden]
        .to(torch.float32)                        # back to FP32 for the rest
    )

    # ------------------------------------------------------------------
    # 8. Second LayerNorm, out gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                         # element‑wise gate

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
```
