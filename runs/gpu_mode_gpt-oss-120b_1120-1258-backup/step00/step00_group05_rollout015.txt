left = F.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
right = F.linear(x_norm, right_proj_weight)
# mask broadcast
mask_b = mask.unsqueeze(-1) # [B,N,N,1] of dtype same as input
left = left * mask_b
right = right * mask_b

# gates
left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

left = left * left_gate
right = right * right_gate

# compute out = einsum left * right via kernel
out = trmul_kernel(left, right)  # shape [B,N,N,hidden_dim]
# out_norm
out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
out_norm = out_norm * out_gate
output = F.linear(out_norm, to_out_weight)
```

                          B, N, D,
                          stride_left_b, stride_left_i, stride_left_k, stride_left_d,
                          stride_right_b, stride_right_j, stride_right_k, stride_right_d,
                          stride_out_b, stride_out_i, stride_out_j, stride_out_d,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr,
                          BLOCK_D: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range of i and j within block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask for i,j inside bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator for each d vector
    # we will compute for each BLOCK_D in a loop over d
    d = 0
    while d < D:
        cur_d = tl.minimum(D - d, BLOCK_D)
        offs_d = d + tl.arange(0, BLOCK_D)
        offs_d = offs_d  # shape (BLOCK_D)
        d_mask = offs_d < D

        # Initialize accumulator for each (i,j) element as vector of cur_d
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)
        # Loop over k dimension
        for k_start in range(0, N, BLOCK_K):
            cur_k = tl.minimum(N - k_start, BLOCK_K)
            offs_k = k_start + tl.arange(0, BLOCK_K)
            # Compute mask for k within bounds
            k_mask = offs_k < N

            # left tile pointer: shape (BLOCK_M, cur_k, cur_d)
            left_ptrs = left_ptr + pid_b * stride_left_b \
                + offs_i[:, None, None] * stride_left_i \
                + offs_k[None, :, None] * stride_left_k \
                + offs_d[None, None, :] * stride_left_d

            left = tl.load(left_ptrs,
                           mask=(mask_i[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]),
                           other=0.0)

            # right tile transposed shape (cur_k, BLOCK_N, cur_d)
            right_ptrs = right_ptr + pid_b * stride_right_b \
                + offs_k[:, None, None] * stride_right_k \
                + offs_j[None, :, None] * stride_right_j \
                + offs_d[None, None, :] * stride_right_d

            right = tl.load(right_ptrs,
                            mask=(k_mask[:, None, None] & mask_j[None, :, None] & d_mask[None, None, :]),
                            other=0.0)

            # Perform reduction over k dimension
            # left: (M,K,D), right: (K,N,D)
            # approach: for each k_idx, compute outer product across i and j
            # Use broadcasting: left_expand = left[:, :, None, :]  # (M,K,1,D)
            # right_expand = right[None, :, :, :]  # (1,K,N,D)
            # Multiply and sum over k:
            # acc += tl.sum(left_expand * right_expand, axis=1)
            # To avoid large temporaries, do manual loop over k
            # But we can use tl.sum after elementwise multiplication broadcasting across i and j.

            prod = left[:, :, None, :] * right[None, :, :, :]  # shape (M,K,N,D)

            + offs_i[:, None, None] * stride_out_i \
            + offs_j[None, :, None] * stride_out_j \
            + offs_d[None, None, :] * stride_out_d

                 acc,
                 mask=(mask_i[:, None, None] & mask_j[None, :, None] & d_mask[None, None, :]))
        d += BLOCK_D

Here's the kernel:

```
@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # offsets for i and j within the block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # offsets for hidden dimension (full)
    offs_d = tl.arange(0, D)

    # accumulator (M,N,D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, D), dtype=tl.float32)

    # loop over K
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile for current k block
        # shape (M, cur_k, D)
        left_ptrs = left_ptr + pid_b * stride_left_b \
            + offs_i[:, None, None] * stride_left_i \
            + offs_k[None, :, None] * stride_left_k \
            + offs_d[None, None, :] * stride_left_d

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & (offs_d[None, None, :] < D),
                       other=0.0)   # (M, K, D)

        # Load right tile transposed for current k block
        # shape (cur_k, N, D)
        right_ptrs = right_ptr + pid_b * stride_right_b \
            + offs_k[:, None, None] * stride_right_k \
            + offs_j[None, :, None] * stride_right_j \
            + offs_d[None, None, :] * stride_right_d

        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & (offs_d[None, None, :] < D),
                        other=0.0)   # (K, N, D)

        # loop over k within block
        # unroll small K? maybe manually:
        for kk in range(0, BLOCK_K):
            # check if kk < cur_k to avoid out-of-range
            if kk >= cur_k:
                continue  # compile-time constant? Actually tl.constexpr can't depend on runtime.

```

But we can't use runtime conditional like `if kk >= cur_k: continue` because tl.constexpr loops compile-time constant. However we can use mask inside loads and accumulations via selecting only valid entries. Better: we can loop over all BLOCK_K and rely on mask to zero out contributions for out-of-range elements. So we can just loop over `kk` from 0 to BLOCK_K constant and compute for each index; the mask ensures out-of-bounds loads produce zeros and contributions zero.

Thus:

```
for kk in range(0, BLOCK_K):
    left_vec = left[:, kk, :]   # (M, D)
    right_vec = right[kk, :, :] # (N, D)
    # compute outer product and accumulate
    # left_vec[:, None, :] shape (M,1,D)
    # right_vec[None, :, :] shape (1,N,D)
    acc += left_vec[:, None, :] * right_vec[None, :, :]   # (M,N,D)
```

The mask for out-of-bound k: we need to handle that left and right already have zeros for invalid k entries because of mask in loads. So the outer product will involve zeros, fine.

After finishing loop over all k blocks, we store acc to output.

Out pointer:

```
out_ptrs = out_ptr + pid_b * stride_out_b \
    + offs_i[:, None, None] * stride_out_i \
    + offs_j[None, :, None] * stride_out_j \
    + offs_d[None, None, :] * stride_out_d

tl.store(out_ptrs,
         acc,
         mask=mask_i[:, None, None] & mask_j[None, :, None] & (offs_d[None, None, :] < D))
```

Now we need to launch this kernel with grid dimensions:

- num_blocks_i = ceil(N / BLOCK_M)
- num_blocks_j = ceil(N / BLOCK_N)
- num_blocks_b = B

Thus grid = (num_blocks_i, num_blocks_j, B). We'll pass appropriate arguments.

Now the kernel implements full triple sum (i,j,k) for each d element.

def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i = pid_i
    j = pid_j

    # mask for out-of-bound i,j
    if i >= N or j >= N:
        return

    # offsets for D dimension
    offs_d = tl.arange(0, D)

    # accumulator
    acc = tl.zeros((D,), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k, BLOCK_K)
        offs_k = k + tl.arange(0, BLOCK_K)

        # load left vector for current k block: shape (cur_k, D)
        left_ptrs = left_ptr + pid_b * stride_left_b + i * stride_left_i + offs_k[:, None] * stride_left_k + offs_d[None, :] * stride_left_d
        left = tl.load(left_ptrs, mask=(offs_k[:, None] < N) & (offs_d[None,:] < D), other=0.0)  # shape (BLOCK_K, D)

        # load right vector for current block: shape (cur_k, D)
        right_ptrs = right_ptr + pid_b * stride_right_b + j * stride_right_j + offs_k[:, None] * stride_right_k + offs_d[None, :] * stride_right_d
        right = tl.load(right_ptrs, mask=(offs_k[:, None] < N) & (offs_d[None,:] < D), other=0.0)

        # accumulate product across k values
        # left[k,:] * right[k,:] summed over k dimension
        # we can compute elementwise multiplication and sum across k dimension:
        acc += tl.sum(left * right, axis=0)   # sum over k (first axis)
    # store acc in out
    out_ptrs = out_ptr + pid_b * stride_out_b + i * stride_out_i + j * stride_out_j + offs_d * stride_out_d

"""
TriMul (outgoing) implementation for AlphaFold3.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]).
2. Compute four linear projections of `x` (no bias):
       left_proj  = x @ W_left   (→ [B,N,N,H])
       right_proj = x @ W_right  (→ [B,N,N,H])
   and three gating projections (sigmoid):
       left_gate  = sigmoid(x @ W_left_gate)
       right_gate = sigmoid(x @ W_right_gate)
       out_gate   = sigmoid(x @ W_out_gate)
   Apply the optional mask (broadcasted on the hidden dimension) and the
   gates to the projected tensors.

3. Core “triangle‑multiplicative” update:
       out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   This is a batched matrix‑multiply for each hidden channel:
       out[b,:,:,h] = left[b,:,:,h] @ right[b,:,:,h].T
   To keep everything on‑device we flatten the hidden dimension into
   the batch dimension and run a custom Triton GEMM kernel that computes
   all (B·H) matrix products in parallel.

4. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate and a final linear projection back to the original dimension.

The Triton kernel (``batched_matmul_kernel``) is a classic tiled GEMM
kernel with three program‑ID axes:
    pid_m  – output‑row tile,
    pid_n  – output‑col tile,
    pid_b  – batch (combined B×H) tile.
It loads a (BLOCK_M×BLOCK_K) tile from A (left) and a
(BLOCK_K×BLOCK_N) tile from B (right transposed), performs the dot product
with ``tl.dot`` and accumulates in FP32.  The kernel works for arbitrary
batch size, sequence length and hidden dimension (the latter is folded into
the batch size).  All tensors are expected to be on the CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_am, stride_ar, stride_ac,   # A strides: batch, row (i), col (k)
    stride_bm, stride_br, stride_bc,   # B strides: batch, row (k), col (j) – note the transposed layout
    stride_cm, stride_cr, stride_cc,   # C strides: batch, row (i), col (j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Tiled batched GEMM: C = A @ B, where B is already transposed (K×N)."""
    pid_m = tl.program_id(0)   # row‑tile index
    pid_n = tl.program_id(1)   # col‑tile index
    pid_b = tl.program_id(2)   # batch (combined B×H) index

    # Compute the absolute coordinates of the tiles.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Allocate accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, K, BLOCK_K):
        cur_k = tl.minimum(K - k, BLOCK_K)
        offs_k = k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, cur_k)
        a_ptrs = a_ptr + pid_b * stride_am \
                 + offs_m[:, None] * stride_ar \
                 + offs_k[None, :] * stride_ac
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0).to(tl.float32)

        # Load B tile (already transposed): shape (cur_k, BLOCK_N)
        b_ptrs = b_ptr + pid_b * stride_bm \
                 + offs_k[:, None] * stride_br \
                 + offs_n[None, :] * stride_bc
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0).to(tl.float32)

        # Accumulate the matrix product for this K‑block.
        acc += tl.dot(a, b)

    # Write the result back to C.
    c_ptrs = c_ptr + pid_b * stride_cm \
             + offs_m[:, None] * stride_cr \
             + offs_n[None, :] * stride_cc
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused Triton GEMM.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA).
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape
    device = input_tensor.device

    # ----------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----------------------------------------------------------------------
    # Linear projections (no bias)
    # ----------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])       # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])     # [B,N,N,H]

    # ----------------------------------------------------------------------
    # Optional mask (broadcast to hidden dim)
    # ----------------------------------------------------------------------
    if config.get("nomask", False):
        # “no mask” test case – treat as all‑ones.
        mask = torch.ones(bs, seq_len, seq_len, 1, dtype=left_proj.dtype, device=device)
    else:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)   # [B,N,N,1]

    left_proj = left_proj * mask
    right_proj = right_proj * mask

    # ----------------------------------------------------------------------
    # Gates (sigmoid of linear projections, no bias)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ----------------------------------------------------------------------
    # Core TriMul: out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    # We fold the hidden dimension into the batch dimension and run a
    # batched GEMM kernel (left @ right^T) for each hidden channel.
    # ----------------------------------------------------------------------
    # Reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(-1, seq_len, seq_len)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(-1, seq_len, seq_len)

    # Allocate output tensor for the flattened computation.
    out_flat = torch.empty_like(left_flat)

    # Determine tiling sizes (tuned for H100 TensorCores).
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions: (row tiles, col tiles, batch_total)
    batch_total = bs * hidden_dim
    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        batch_total,
    )

    # Launch the Triton kernel.  B is already transposed by swapping the
    # K‑ and N‑strides of the right operand.
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,
        seq_len,
        seq_len,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(2),   # stride for K (original column)
        right_flat.stride(1),   # stride for N (original row)
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, H].
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # Second LayerNorm on the hidden dimension, gating and final projection.
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
