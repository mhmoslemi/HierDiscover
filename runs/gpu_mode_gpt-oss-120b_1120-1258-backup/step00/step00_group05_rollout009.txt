@triton.jit
def batch_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for left tile
    a_ptrs = left_ptr + batch * stride_left_batch + (offs_m[:, None] * stride_left_i) + (tl.arange(0, BLOCK_K)[None, :] * stride_left_k)

    # pointers for right tile (transposed)
    # we need rows (k) and cols (j)
    b_ptrs = right_ptr + batch * stride_right_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_right_i) + (offs_n[None, :] * stride_right_j)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K + tl.arange(0,BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_i  # moving to next k block rows

    # Store
    out_ptrs = out_ptr + batch * stride_out_batch + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start indices
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create masks to avoid out-of-bounds
    mask_m = off_m < M
    mask_n = off_n < N

    # Pointers to the start of the block
    a_block_ptr = a_ptr + batch * stride_a_batch + (off_m[:, None] * stride_a_m)
    b_block_ptr = b_ptr + batch * stride_b_batch + (off_n[None, :] * stride_b_n)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # create mask for K dimension
        mask_k = off_k < K

        # Load A tile: shape (M, K)
        a = tl.load(
            a_block_ptr + (off_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile, but B is transposed: we need rows = k, cols = n
        # B pointer currently points to start of row block for off_n columns: need to offset by k row offset
        b = tl.load(
            b_block_ptr + (off_k[:, None] * stride_b_k),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)

        # Increment pointers for next K block
        a_block_ptr += BLOCK_K * stride_a_k
        b_block_ptr += BLOCK_K * stride_b_k

    # Store result
    c = c_ptr + batch * stride_c_batch + (off_m[:, None] * stride_c_m) + (off_n[None, :] * stride_c_n)
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # gather weight parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # optional gate biases? Not present in spec; assume None.

    # Normalization
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gating logits
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply mask if provided
    if mask is not None:
        # mask shape [B,N,N]
        mask_ = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Apply elementwise gates
    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape
    # reshape for batched matmul
    left_flat = left.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute grid dims
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden_dim,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape output back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out_gate gating
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offset indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of this batch
    a_batch_ptr = a_ptr + batch * stride_a_batch
    b_batch_ptr = b_ptr + batch * stride_b_batch
    c_batch_ptr = c_ptr + batch * stride_c_batch

    # pointer base for m and n dims
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_i
    b_ptrs = b_batch_ptr + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        cur_mask = cur_k < K

"""
TriMul (outgoing) – Triton‑accelerated implementation.

The forward pass consists of:
1. Layer‑norm over the last dimension.
2. Linear projections → left, right (shape [B,N,N,H]).
3. Linear gates → left_gate, right_gate, out_gate (sigmoid).
4. Optional pair mask applied to left/right before gating.
5. Elementwise gating of left/right.
6. Batched matrix multiply:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is equivalent to a separate matmul for each (batch,head) pair:
      left_(b,h) @ right_(b,h).T.
   The heavy O(N³) reduction is performed in a fused Triton kernel.
7. Layer‑norm over the hidden dimension, gating with out_gate and a final
   linear projection back to the original channel dimension.

The kernel fuses the three‑dimensional batched GEMM (the cubic reduction)
and keeps the rest of the pipeline in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    a_ptr,                     # left  [B*H, N, N]
    b_ptr,                     # right [B*H, N, N] (used transposed)
    c_ptr,                     # out   [B*H, N, N]
    BATCH, M, N, K,            # dimensions (B*H, N, N, N)
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_i, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: for each batch `b` compute
        C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]
    where A and B have shape (BATCH, M, K) and (BATCH, K, N) respectively.
    The kernel is launched with a 3‑D grid (BATCH, ceil(M/BM), ceil(N/BN)).
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows i
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols j

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for the current batch
    # ------------------------------------------------------------------
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # pointers that already include the i‑ and j‑offsets
    a_ptrs = a_batch + offs_m[:, None] * stride_a_i
    b_ptrs = b_batch + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)           # (BLOCK_K,)
        cur_mask = cur_k < K

        # Load a tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptrs + cur_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & cur_mask[None, :],
            other=0.0,
        )
        # Load b tile (transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptrs + cur_k[:, None] * stride_b_k,
            mask=cur_mask[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c = c_batch + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_n
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be None)
        - weights      : dict of torch.Tensors containing all model weights
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # weight shortcuts (biases exist only for layer‑norms)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Pair representation layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Gating logits (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if (mask is not None) and (not nomask):
        mask_exp = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # H == hidden_dim

    # ------------------------------------------------------------------
    # 6) Fuse left/right into [B*H, N, N] and launch Triton GEMM
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------- launch parameters ---------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch*head dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
    )

    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        N,
        # strides (as pure Python ints)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8) Output layer‑norm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9) Apply final out‑gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
