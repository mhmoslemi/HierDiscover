@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_N)
    num_j = tl.cdiv(N, BLOCK_N)
    # total blocks = B * num_i * num_j
    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i0 = (rem // num_j) * BLOCK_N
    j0 = (rem % num_j) * BLOCK_N

    # offsets
    i = i0 + tl.arange(0, BLOCK_N)
    j = j0 + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    # Load mask tile
    mask = tl.load(mask_ptr + b * stride_mask_b + i[:,None] * stride_mask_i + j[None,:] * stride_mask_j,
                    mask=i_mask[:,None] & j_mask[None,:], other=0.0)

    # For loop over hidden dim
    for d0 in range(0, H, BLOCK_H):
        d = d0 + tl.arange(0, BLOCK_H)
        d_mask = d < H

        left = tl.load(left_ptr + b * stride_left_b + i[:,None,None]*stride_left_i + j[None,:,None]*stride_left_j + d[None,None,:]*stride_left_h,
                        mask=i_mask[:,None,None] & j_mask[None,:,None] & d_mask[None,None,:],
                        other=0.0)
        right = tl.load(right_ptr + b * stride_right_b + i[:,None,None]*stride_right_i + j[None,:,None]*stride_right_j + d[None,None,:]*stride_right_h,
                        mask=i_mask[:,None,None] & j_mask[None,:,None] & d_mask[None,None,:],
                        other=0.0)
        left_gate = tl.load(left_gate_ptr + b * stride_lgate_b + i[:,None,None]*stride_lgate_i + j[None,:,None]*stride_lgate_j + d[None,None,:]*stride_lgate_h,
                        mask=i_mask[:,None,None] & j_mask[None,:,None] & d_mask[None,None,:],
                        other=0.0)
        right_gate = tl.load(right_gate_ptr + b * stride_rgate_b + i[:,None,None]*stride_rgate_i + j[None,:,None]*stride_rgate_j + d[None,None,:]*stride_rgate_h,
                        mask=i_mask[:,None,None] & j_mask[None,:,None] & d_mask[None,None,:],
                        other=0.0)

        # Apply mask and gates
        left = left * mask[...,None] * left_gate
        right = right * mask[...,None] * right_gate

        # Store back
        tl.store(left_ptr + b * stride_left_b + i[:,None,None]*stride_left_i + j[None,:,None]*stride_left_j + d[None,None,:]*stride_left_h,
                  left, mask=i_mask[:,None,None] & j_mask[None,:,None] & d_mask[None,None,:])
        tl.store(right_ptr + b * stride_right_b + i[:,None,None]*stride_right_i + j[None,:,None]*stride_right_j + d[0,:,None]*stride_right_h,
                  right, mask=i_mask[:,None,None] & j_mask[None,:,None] & d_mask[None,None,:])
        # The pointer arithmetic for right store; note we used d[None,None,:] as offset.

"""
custom_kernel for the “outgoing” TriMul operator.

Algorithm:
1. Layer‐normalize the input tensor (last dimension) using the provided
   norm weight/bias.
2. Linear projections `left_proj` / `right_proj` (no bias) and the three
   gating linear layers, followed by a sigmoid.
3. If a mask is present (nomask=False) we fuse the mask, left‑gate and
   right‑gate into a single Triton kernel (`mask_gate_kernel`) that
   multiplies the projected tensors in‑place:
        left = left_proj * mask * left_gate
        right = right_proj * mask * right_gate
   For the nomask case we perform the same multiplication with plain
   PyTorch.
4. Compute the pairwise “gram” product:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to a batched matrix multiplication for each
   hidden channel.  By permuting the tensors we obtain a batch of GEMMs
   that are executed by cuBLAS:
        out = (left.permute(0,3,1,2) @ right.permute(0,3,1,2).transpose(-1,-2))
        out = out.permute(0,2,3,1)
5. Apply a second LayerNorm (`to_out_norm`), multiply by the output gate,
   and a final linear projection back to the original channel size.
6. Return the result tensor of shape [B, N, N, dim].

The kernel `mask_gate_kernel` demonstrates a genuine Triton implementation
by fusing the mask application and gating for all hidden channels.
The rest of the computation uses highly‑optimised PyTorch/CuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors
# --------------------------------------------------------------------------- #
@triton.jit
def mask_gate_kernel(
    left_ptr,               # *mut* left_proj   [B,N,N,H]
    right_ptr,              # *mut* right_proj  [B,N,N,H]
    mask_ptr,               # mask              [B,N,N]
    left_gate_ptr,          # left_gate         [B,N,N,H]
    right_gate_ptr,         # right_gate        [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Each program processes a TILE_N×TILE_N block of the (i,j) plane
    and iterates over the hidden dimension in blocks of size BLOCK_H.
    """
    pid = tl.program_id(0)

    # number of tiles per axis
    num_i = tl.cdiv(N, BLOCK_N)
    num_j = tl.cdiv(N, BLOCK_N)

    # map pid to (b, i0, j0)
    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i0 = (rem // num_j) * BLOCK_N
    j0 = (rem % num_j) * BLOCK_N

    # thread‑local coordinates
    i = i0 + tl.arange(0, BLOCK_N)
    j = j0 + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    # ---------- load the 2‑D mask tile ----------
    mask = tl.load(
        mask_ptr + b * stride_mask_b
                 + i[:, None] * stride_mask_i
                 + j[None, :] * stride_mask_j,
        mask=i_mask[:, None] & j_mask[None, :],
        other=0.0
    )  # shape: [BLOCK_N, BLOCK_N]

    # broadcast mask to hidden dim
    mask_ex = mask[:, :, None]   # [BLOCK_N, BLOCK_N, 1]

    # ---------- loop over hidden dimension ----------
    h_offset = 0
    while h_offset < H:
        d = h_offset + tl.arange(0, BLOCK_H)
        d_mask = d < H

        # load left, right, gates
        left = tl.load(
            left_ptr + b * stride_left_b
                     + i[:, None, None] * stride_left_i
                     + j[None, :, None] * stride_left_j
                     + d[None, None, :] * stride_left_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
            other=0.0
        )
        right = tl.load(
            right_ptr + b * stride_right_b
                      + i[:, None, None] * stride_right_i
                      + j[None, :, None] * stride_right_j
                      + d[None, None, :] * stride_right_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
            other=0.0
        )
        left_gate = tl.load(
            left_gate_ptr + b * stride_lgate_b
                           + i[:, None, None] * stride_lgate_i
                           + j[None, :, None] * stride_lgate_j
                           + d[None, None, :] * stride_lgate_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
            other=0.0
        )
        right_gate = tl.load(
            right_gate_ptr + b * stride_rgate_b
                            + i[:, None, None] * stride_rgate_i
                            + j[None, :, None] * stride_rgate_j
                            + d[None, None, :] * stride_rgate_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
            other=0.0
        )

        # fuse mask and gates
        left = left * mask_ex * left_gate
        right = right * mask_ex * right_gate

        # store back in‑place
        tl.store(
            left_ptr + b * stride_left_b
                     + i[:, None, None] * stride_left_i
                     + j[None, :, None] * stride_left_j
                     + d[None, None, :] * stride_left_h,
            left,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
        )
        tl.store(
            right_ptr + b * stride_right_b
                      + i[:, None, None] * stride_right_i
                      + j[None, :, None] * stride_right_j
                      + d[None, None, :] * stride_right_h,
            right,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
        )

        h_offset += BLOCK_H


# --------------------------------------------------------------------------- #
# Entry point used by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same device).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------- #
    # unpack configuration
    # ------------------------------------------------------------------- #
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # if True we skip mask

    # ------------------------------------------------------------------- #
    # unpack model weights (all are torch.Tensor, already on correct device)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5  # LayerNorm epsilon (same as PyTorch default)

    # ------------------------------------------------------------------- #
    # 1. LayerNorm on the input (over the last dimension)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------- #
    # 3. Gating tensors (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------- #
    # 4. Apply mask and gates (fused Triton kernel when mask is present)
    # ------------------------------------------------------------------- #
    if not nomask:
        # Ensure mask is float32 for arithmetic
        if mask.dtype != torch.float32:
            mask_f = mask.to(torch.float32)
        else:
            mask_f = mask

        # Make everything contiguous (required for correct stride handling)
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_f = mask_f.contiguous()

        # Strides for the tensors (in element units)
        stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
        stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
        stride_mask_b, stride_mask_i, stride_mask_j = mask_f.stride()
        stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h = left_gate.stride()
        stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h = right_gate.stride()

        B, N, _, _ = left.shape

        # Tuning parameters – chosen to fit comfortably within 1024 threads/program
        BLOCK_N = 64   # tile size in i/j dimensions
        BLOCK_H = 32   # tile size in hidden dimension

        # Determine grid size: one program per (b, i_tile, j_tile)
        num_i_tiles = (N + BLOCK_N - 1) // BLOCK_N
        num_j_tiles = (N + BLOCK_N - 1) // BLOCK_N
        total_progs = B * num_i_tiles * num_j_tiles

        mask_gate_kernel[(total_progs,)](
            left,
            right,
            mask_f,
            left_gate,
            right_gate,
            B, N, hidden_dim,
            stride_left_b, stride_left_i, stride_left_j, stride_left_h,
            stride_right_b, stride_right_i, stride_right_j, stride_right_h,
            stride_mask_b, stride_mask_i, stride_mask_j,
            stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
            stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
            BLOCK_N=BLOCK_N,
            BLOCK_H=BLOCK_H,
        )
        # After the kernel, `left` and `right` already contain the masked &
        # gated values.
    else:
        # No mask – simple elementwise multiplication with the gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Gram‑product via batched GEMM (equivalent to the Einstein sum)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------- #
    #   (B,N,N,D) -> (B,D,N,N)  for cuBLAS batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, K] with K=N
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, D, N, K]

    # Matrix multiply each (B*D) pair: (N x K) @ (N x K)^T  -> (N x N)
    out_perm = torch.matmul(
        left_perm,                # (B, D, N, K)
        right_perm.transpose(-1, -2)  # (B, D, K, N)
    )  # (B, D, N, N)

    # Bring back to original layout
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, D]

    # ------------------------------------------------------------------- #
    # 6. Final layer‑norm, output‑gate, and projection back to `dim`
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
