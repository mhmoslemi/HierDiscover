def custom_kernel(data):
   input_tensor, mask, weights, config = data
   # config contains dim (C), hidden_dim (H) maybe also other parameters.
   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   # weights are dict of tensors on same device and dtype (float16? default float32)
   # We'll use float32 for forward but convert to float16 for speed.

   # Compute LayerNorm: using torch.nn.functional.layer_norm
   # x = (input - mean) / sqrt(var+eps) * weight + bias
   eps = 1e-5
   x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

   # Compute left and right projections
   # Linear: out = x @ weight.T (weights shape (hidden_dim, dim))
   # Use torch.nn.functional.linear
   left = torch.nn.functional.linear(x, weights["left_proj.weight"])
   right = torch.nn.functional.linear(x, weights["right_proj.weight"])

   # mask if provided and not nomask:
   if mask is not None:
      left = left * mask.unsqueeze(-1)
      right = right * mask.unsqueeze(-1)

   # Compute gates
   left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

   left = left * left_gate
   right = right * right_gate

   # Cast to fp16 for kernel
   left = left.to(torch.float16)
   right = right.to(torch.float16)

   # Permute for kernel (B,H,N,N)
   B, N, _, _ = left.shape
   left_perm = left.permute(0, 3, 1, 2).contiguous()   # shape [B, hidden_dim, N, N]
   right_perm = right.permute(0, 3, 1, 2).contiguous()

   # Compute out using Triton kernel
   out_perm = tri_mul_matmul_kernel(left_perm, right_perm)  # Implementation that returns float32 output

   # Permute back to [B,N,N,hidden_dim] (float32)
   out = out_perm.permute(0, 2, 3, 1).contiguous()  # shape B,N,N,H

   # Apply to_out_norm: now need LayerNorm hidden_dim:
   out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

   # Multiply by out_gate (float32)
   out = out * out_gate

   # Final linear to dim:
   out = torch.nn.functional.linear(out, weights["to_out.weight"])

   return out

def tri_mul_matmul(left, right):
    B,N,N_,H = left.shape
    assert N == N_
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    # cast to fp16 if not already
    left_perm = left_perm.to(torch.float16)
    right_perm = right_perm.to(torch.float16)

    stride_b_batch, stride_b_hidden, stride_b_i, stride_b_k = right_perm.stride()

   dim = config["dim"]; hidden_dim = config["hidden_dim"]
   eps = 1e-5

"""
TriMul (outgoing) custom kernel.

The forward pass consists of
1. Layer‑norm over the last channel (dim).
2. Two linear projections (`left_proj`, `right_proj`) + optional mask.
3. Gated projections (`left_gate`, `right_gate`, `out_gate`).
4. Massive pair‑wise multiplication:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This reduces to a batched matrix multiplication
        out_d = left_d @ right_dᵀ   (for each hidden channel d).
   We implement this O(N³) step with a fused Triton kernel that
   processes the (B·hidden) * N×N matrices in FP16 and accumulates
   in FP32.
5. Layer‑norm over the hidden dimension, gating with `out_gate`,
   and a final linear projection back to the original `dim`.

Only the heavy N³ part is written in Triton; the remaining
operations use fast PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ   (left/right shape [B, H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    B, H, N,                            # sizes
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_hidden, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # Program IDs
    # axis 0 : combined (batch * hidden) * block_row
    # axis 1 : block_col
    # ------------------------------------------------------------------
    pid0 = tl.program_id(axis=0)
    pid1 = tl.program_id(axis=1)

    # number of blocks along the M dimension (rows)
    num_blocks_m = tl.cdiv(N, BLOCK_M)

    # decode batch, hidden and block‑row
    batch_hidden = pid0 // num_blocks_m          # integer division
    block_m      = pid0 %  num_blocks_m
    batch_idx    = batch_hidden // H
    hidden_idx   = batch_hidden %  H

    # decode block‑col
    block_n = pid1

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    row_offset = block_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑indices
    col_offset = block_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑indices

    row_mask = row_offset < N
    col_mask = col_offset < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)      # k‑indices
        k_mask = k_offset < N

        # ---------- load A tile (M × K) ----------
        a_ptrs = (a_ptr
                  + batch_idx   * stride_a_batch
                  + hidden_idx  * stride_a_hidden
                  + row_offset[:, None] * stride_a_i
                  + k_offset[None, :]   * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=row_mask[:, None] & k_mask[None, :],
                    other=0.0).to(tl.float32)   # fp16 → fp32

        # ---------- load Bᵀ tile (K × N) ----------
        # We want Bᵀ[k, j] = B[j, k]; therefore we treat k as the row
        # stride (stride_b_k) and j as the column stride (stride_b_i).
        b_ptrs = (b_ptr
                  + batch_idx   * stride_b_batch
                  + hidden_idx  * stride_b_hidden
                  + k_offset[:, None] * stride_b_k
                  + col_offset[None, :] * stride_b_i)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & col_mask[None, :],
                    other=0.0).to(tl.float32)   # fp16 → fp32

        # ---------- inner product ----------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write back the result tile (M × N) in fp32
    # ------------------------------------------------------------------
    c_ptrs = (c_ptr
              + batch_idx   * stride_c_batch
              + hidden_idx  * stride_c_hidden
              + row_offset[:, None] * stride_c_i
              + col_offset[None, :] * stride_c_j)

    tl.store(c_ptrs,
             acc,
             mask=row_mask[:, None] & col_mask[None, :])

# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _tri_mul_matmul_fp16(left_fp16: torch.Tensor, right_fp16: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H] (float16)
    returns:      [B, N, N, H] (float32)
    """
    B, N, _, H = left_fp16.shape
    # permute to [B, H, N, N] so that we can treat `H` as an extra batch dim
    a = left_fp16.permute(0, 3, 1, 2).contiguous()
    b = right_fp16.permute(0, 3, 1, 2).contiguous()
    out = torch.empty((B, H, N, N), dtype=torch.float32, device=a.device)

    # strides (in elements)
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k = a.stride()
    stride_b_batch, stride_b_hidden, stride_b_i, stride_b_k = b.stride()
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j = out.stride()

    # ------------------------------------------------------------------
    # Tunable block sizes – chosen experimentally for H100, fp16 inputs.
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    # combine (batch * hidden) with the row‑block dimension
    grid_x = B * H * grid_m
    grid_y = grid_n

    _tri_mul_gemm_kernel[(grid_x, grid_y)](
        a_ptr=a,
        b_ptr=b,
        c_ptr=out,
        B=B,
        H=H,
        N=N,
        stride_a_batch=stride_a_batch,
        stride_a_hidden=stride_a_hidden,
        stride_a_i=stride_a_i,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_hidden=stride_b_hidden,
        stride_b_i=stride_b_i,
        stride_b_k=stride_b_k,
        stride_c_batch=stride_c_batch,
        stride_c_hidden=stride_c_hidden,
        stride_c_i=stride_c_i,
        stride_c_j=stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # permute back to the original layout [B, N, N, H]
    return out.permute(0, 2, 3, 1).contiguous()

# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last channel = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (mask shape = [B, N, N])
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        # mask may be bool/int – cast to float for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate      # still float32
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) multiplication – perform in fp16 + Triton
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (reduces memory traffic)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Triton‑accelerated matmul: out = left @ rightᵀ  (per hidden channel)
    out = _tri_mul_matmul_fp16(left_fp16, right_fp16)          # [B, N, N, H] float32

    # ------------------------------------------------------------------
    # 6️⃣ Final hidden‑dim layer‑norm, gating, and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                                    # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])           # back to `dim`

    return out
