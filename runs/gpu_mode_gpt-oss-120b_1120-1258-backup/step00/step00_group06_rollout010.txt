@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute program id
    pid = tl.program_id(axis=0)
    total_blocks_i = tl.cdiv(N, BLOCK_M)
    total_blocks_j = tl.cdiv(N, BLOCK_N)
    total_blocks_hd = B * H
    pid_hd = pid // (total_blocks_i * total_blocks_j)
    pid = pid % (total_blocks_i * total_blocks_j)
    pid_i = pid // total_blocks_j
    pid_j = pid % total_blocks_j
    # get batch and hidden indices
    b = pid_hd // H
    h = pid_hd % H

    # Compute offsets for the block
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # Mask out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_k = k_offset < N

    a_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + i_offset[:,None] * stride_left_i + k_offset[None,:] * stride_left_k
    # left: shape (BLOCK_M, BLOCK_K)
    # right: need to load B[k,j]
    b_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + j_offset[None,:] * stride_right_j + k_offset[:,None] * stride_right_k
    # shape (BLOCK_K, BLOCK_N)

    # initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # load a and b
        a = tl.load(a_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0)
        # compute dot
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k
    # store result
    c_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + i_offset[:,None] * stride_out_i + j_offset[None,:] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16) if tin? else acc?? Wait we need float32 output. Keep as float32.

def custom_kernel(data):
    # parse input
    input_tensor, mask, weights, config = data
    # B, N, N, dim
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Linear projections
    left = F.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)
    # apply mask if provided
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f
    # Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))
    left = left * left_gate
    right = right * right_gate
    # Compute pairwise product across k dimension using Triton kernel
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]
    # Determine strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()
    # Block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # compute number of blocks
    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = num_blocks_i * num_blocks_j * B * hidden_dim
    # Launch kernel
    trimul_einsum_kernel[total_blocks](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],

        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # Compute block coordinates
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    # Total per (b,h) is num_blocks_i * num_blocks_j
    # Determine which (b,h) this pid belongs to
    pid_bd = pid // (num_blocks_i * num_blocks_j)  # index over B*H
    pid = pid % (num_blocks_i * num_blocks_j)
    pid_i = pid // num_blocks_j
    pid_j = pid % num_blocks_j
    # Determine batch and hidden indices
    b = pid_bd // H
    h = pid_bd % H

    # Compute offsets within block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers
        # left: shape (M,K)
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # right: we need values right[b, j, k, h] where k as row index, j as column; load as (K,N)
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_j[None, :] * stride_right_j + offs_k[:, None] * stride_right_k

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)

    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul "outgoing" forward pass implemented with a fused Triton kernel.

The kernel computes the core Einstein‑summation:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for all batches `b` and hidden channels `h`.  This is equivalent to an
N×N matrix multiplication L @ Rᵀ for each hidden channel.
All other operations (LayerNorms, linear projections, sigmoids, masking)
are performed with PyTorch for simplicity; the heavy N³ work is off‑loaded
to the Triton kernel.

The implementation supports variable batch size (1‑2), sequence length
up to 1024, and hidden dimensions 128‑384.  The final output is float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: fused batch‑hidden‑channel matrix multiply
# ------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,               # const void *
    right_ptr,              # const void *
    out_ptr,                # void *
    B, N, H,                # int32 scalars (batch, seq_len, hidden_dim)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # rows of output block (i dimension)
    BLOCK_N: tl.constexpr,  # cols of output block (j dimension)
    BLOCK_K: tl.constexpr,  # reduction block size (k dimension)
):
    """Compute out = Σ_k left[:, i, k, h] * right[:, j, k, h]"""

    pid = tl.program_id(0)                     # 1‑D grid of blocks

    #--- grid decomposition -------------------------------------------------
    num_blocks_i = tl.cdiv(N, BLOCK_M)         # blocks along i
    num_blocks_j = tl.cdiv(N, BLOCK_N)         # blocks along j
    # each (batch, hidden) pair gets a full 2‑D grid of (i,j) blocks
    pid_hd = pid // (num_blocks_i * num_blocks_j)   # index over B*H
    pid_rem = pid % (num_blocks_i * num_blocks_j)
    pid_i = pid_rem // num_blocks_j                 # block-id in i
    pid_j = pid_rem % num_blocks_j                  # block-id in j

    # decode batch and hidden-channel indices
    b = pid_hd // H
    h = pid_hd % H

    #--- block offsets ------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (no overflow for typical ranges)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #--- reduction over k ----------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = (left_ptr
                  + b * stride_left_b
                  + h * stride_left_h
                  + offs_i[:, None] * stride_left_i
                  + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[b, j, k, h]  -> we need (k, j) ordering => shape (BLOCK_K, BLOCK_N)
        b_ptrs = (right_ptr
                  + b * stride_right_b
                  + h * stride_right_h
                  + offs_j[None, :] * stride_right_j
                  + offs_k[:, None] * stride_right_k)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate dot product over the k‑axis
        acc += tl.dot(a, b)

    #--- write back ---------------------------------------------------------
    c_ptrs = (out_ptr
              + b * stride_out_b
              + h * stride_out_h
              + offs_i[:, None] * stride_out_i
              + offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ------------------------------------------------------------------
# Public entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (or None)
        - weights      : dict of torch.Tensor containing all linear / norm weights
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Extract shape / config
    # ------------------------------------------------------------------
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Unpack weight tensors (they are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (bias=False by definition)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections on the original normed input)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core Triple‑Multiplicative update via Triton (einsum reduction)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # same shape [B,N,N,hidden_dim]

    # Strides for the three tensors (all contiguous, but we keep them explicit)
    s_left  = left.stride()
    s_right = right.stride()
    s_out   = out.stride()

    # Block sizes (tuned for H100, can be adjusted)
    BLOCK_M = 64   # i‑dimension tile
    BLOCK_N = 64   # j‑dimension tile
    BLOCK_K = 64   # reduction tile over k

    # Total number of program instances (1‑D grid)
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = num_i * num_j * B * hidden_dim   # (batch * hidden) * tile_grid

    # Launch the kernel
    trimul_einsum_kernel[total_blocks](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=s_left[0],
        stride_left_i=s_left[1],
        stride_left_k=s_left[2],
        stride_left_h=s_left[3],
        stride_right_b=s_right[0],
        stride_right_j=s_right[1],
        stride_right_k=s_right[2],
        stride_right_h=s_right[3],
        stride_out_b=s_out[0],
        stride_out_i=s_out[1],
        stride_out_j=s_out[2],
        stride_out_h=s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Output LayerNorm, gating, and final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, to_out_weight)       # final linear (bias=False)

    return out
