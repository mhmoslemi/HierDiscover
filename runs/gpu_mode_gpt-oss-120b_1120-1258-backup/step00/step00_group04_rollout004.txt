import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    b = pid // H
    h = pid % H

    # base pointers for this (batch, hidden) slice
    left_base = left_ptr + b * stride_b + h * stride_d
    right_base = right_ptr + b * stride_b + h * stride_d
    out_base = out_ptr + b * stride_b + h * stride_d

    # block row/col indices
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_offset = kb * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)

        # Load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_m[:, None] * stride_i + k_idx[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (k_idx < N)[None, :]),
                    other=0.0).to(tl.float32)

        # Load B_T: shape (BLOCK_K, BLOCK_N) where B_T[k, j] = right[j, k]
        b_ptrs = right_base + k_idx[:, None] * stride_k + offs_n[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=((k_idx < N)[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result C[b,i,j,h] using same layout as left/right
    c_ptrs = out_base + offs_m[:, None] * stride_i + offs_n[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):

    Triton-accelerated implementation of the outgoing TriMul operation.
    Computes:
        out = ( (LN(x) @ W_left) * sigmoid(LN(x) @ W_left_gate) * mask )
                @ ( (LN(x) @ W_right) * sigmoid(LN(x) @ W_right_gate) * mask )^T
        out = LN_hidden(out) * sigmoid(LN(x) @ W_out_gate)
        out = out @ W_to_out
    where LN denotes LayerNorm.
    """
    import torch
    import triton

    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------------------
    # 1. LayerNorm on input (last dim = dim)
    # -------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.nn.functional.layer_norm expects normalized_shape tuple, weight and bias optional
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape: [B, N, N, dim]

    # -------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # shape: (hidden_dim, dim)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight) # [B,N,N,hidden_dim]

    # -------------------------------------------------------------------------
    # 3. Apply mask if present
    # -------------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] -> unsqueeze to broadcast
        mask_ = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -------------------------------------------------------------------------
    # 4. Gates
    # -------------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # compute gates (sigmoid after linear)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Core multiplicative update (the expensive N^3 part)
    #    Compute out_ijh = sum_k left_ik h * right_jk h   (for each h)
    #    Implemented via Triton kernel that performs a batch of
    #    B * hidden_dim independent matrix multiplications.
    # -------------------------------------------------------------------------
    B, N, _, _ = left.shape   # shape [B,N,N,hidden_dim]
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output tensor
    out_mul = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Strides (in elements, not bytes)
    stride_b = left.stride(0)   # B * N * N * H? Actually stride(0) = N*N*hidden_dim
    stride_i = left.stride(1)   # N*hidden_dim
    stride_k = left.stride(2)   # hidden_dim
    stride_d = left.stride(3)   # 1

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    # launch kernel
    trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_mul.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        stride_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------------------
    # 6. Hidden layernorm + out gate
    # -------------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    out_norm = torch.nn.functional.layer_norm(
        out_mul,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # apply out gate (broadcast)
    out = out_norm * out_gate

    # -------------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # -------------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    # Linear without bias
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return out

Triton‑accelerated outgoing TriMul (AlphaFold3 style)

The kernel computes the core N³ einsum:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where
    left  = (LN(x) @ W_left )   * sigmoid(LN(x) @ W_left_gate)   * mask
    right = (LN(x) @ W_right)   * sigmoid(LN(x) @ W_right_gate)  * mask

The heavy N³ part is expressed as a batch of independent matrix
multiplications (one per hidden dimension) and is executed in a
custom Triton kernel (BLOCK_M×BLOCK_N tiles, BLOCK_K inner reduction).
All other operations (LayerNorm, linear projections, sigmoids,
final LayerNorm + gating + linear) are performed with PyTorch tensors,

The implementation works for any batch size ≤2, sequence length up to
1024, and arbitrary dim / hidden_dim as required by the test suite.

import triton
import triton.language as tl

    left_ptr,                        # const *float32  left tensor
    right_ptr,                       # const *float32  right tensor

):

        C[i, j] = Σ_k A[i, k] * B_T[k, j]   (where B_T[k, j] = right[j, k])

        a = tl.load(
            a_ptrs,
            mask=(mask_m[:, None] & (k_idx < N)[None, :]),
            other=0.0,
        ).to(tl.float32)

        # ----------- load B_T = right[j, k] -------------
        # B_T has shape (K, N) where B_T[k, j] == right[j, k]
        b_ptrs = right_base + k_idx[:, None] * stride_k + offs_n[None, :] * stride_i
        b = tl.load(
            b_ptrs,
            mask=((k_idx < N)[:, None] & mask_n[None, :]),
            other=0.0,
        ).to(tl.float32)

        # ----------- accumulate partial product -------------
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ----------- write out C = out[i, j, h] -------------
    c_ptrs = out_base + offs_m[:, None] * stride_i + offs_n[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Input:
        data = (input_tensor, mask, weights, config)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    import torch.nn.functional as F

    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) LayerNorm on input over the last dim (dim)
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] -> unsqueeze to [B, N, N, 1]
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4) Gating – sigmoid(linear(x_norm))
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core multiplicative update (the O(N³) part) via Triton
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape                      # left.shape == (B, N, N, hidden_dim)
    left  = left.contiguous()
    right = right.contiguous()

    # Allocate output of the einsum
    out_mul = torch.empty((B, N, N, hidden_dim),
                          dtype=torch.float32,
                          device=left.device)

    # Strides in **elements** (not bytes)
    stride_b = left.stride(0)   # N * N * hidden_dim
    stride_i = left.stride(1)   # N * hidden_dim
    stride_k = left.stride(2)   # hidden_dim
    stride_d = left.stride(3)   # 1

    # Tile sizes – tuned for H100, fits easily into shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # one program per (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,       # rows
        (N + BLOCK_N - 1) // BLOCK_N,       # cols
    )

    _trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_mul.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        stride_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm + out_gate
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out_mul,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out_norm * out_gate

    # -----------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
