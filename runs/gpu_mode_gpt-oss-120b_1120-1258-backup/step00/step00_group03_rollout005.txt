"""
custom_kernel for the “outgoing” TriMul operation (AlphaFold3).

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor `x` (shape [B,N,N,D]) using the
   provided weight/bias (`norm.weight` / `norm.bias`).

2. Compute four linear projections of the normalized tensor (no bias):
   * left_proj   = Linear(D → H)   (weight: left_proj.weight)
   * right_proj  = Linear(D → H)   (weight: right_proj.weight)
   * left_gate   = Linear(D → H)   (weight: left_gate.weight)  → sigmoid
   * right_gate  = Linear(D → H)   (weight: right_gate.weight) → sigmoid
   * out_gate    = Linear(D → H)   (weight: out_gate.weight)   → sigmoid

3. If a mask is supplied, broadcast it to [...,1] and apply to the
   `left_proj` and `right_proj` tensors.

4. Fuse the projections with their gates:
        left  = left_proj  * left_gate
        right = right_proj * right_gate

   At this point `left` and `right` have shape [B,N,N,H].

5. **Heavy part – 3‑D tensor contraction**  
   Compute, for every batch `b` and hidden index `h`:

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   This is exactly a batched matrix‑multiply `L_h @ R_hᵀ`.
   A custom Triton kernel (`trimul_matmul_kernel`) performs the
   multiplication on‑the‑fly, tiled in `i`, `j` and reduced over `k`.
   The kernel processes one pair `(b,h)` per program‑ID‑2, using
   BLOCK_M × BLOCK_N tiles (default 64×64) and a BLOCK_K of 32.

6. Apply a second LayerNorm on the hidden dimension using
   `to_out_norm.weight` / `to_out_norm.bias`, multiply element‑wise
   with `out_gate`, and finally map back to the original dimension
   with `to_out.weight` (Linear H → D, no bias).

The function returns the final tensor of shape [B,N,N,D] and runs
the dominant O(N³·H) work on the GPU via the Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,           # pointers
    B, N, H,                               # sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_batch = tl.program_id(2)   # combined (b, h)

    # decode batch & hidden index
    b = pid_batch // H
    h = pid_batch % H

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over reduction dimension k
    # ------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to the current k‑block
        # left[b,i,k,h]   -> strides (b,i,k,h)
        left_offset = (
            b * stride_left_b
            + h * stride_left_d
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[b,j,k,h]  -> same layout, just j instead of i
        right_offset = (
            b * stride_right_b
            + h * stride_right_d
            + offs_j[:, None] * stride_right_i
            + offs_k[None, :] * stride_right_k
        )
        right = tl.load(right_ptr + right_offset,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # Dot product over the k‑dimension.
        # left : (BLOCK_M, BLOCK_K)
        # right: (BLOCK_N, BLOCK_K) -> need its transpose
        acc += tl.dot(left, right, trans_b=True)   # (M,N)

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    out_offset = (
        b * stride_out_b
        + h * stride_out_d
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    mask_store = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptr + out_offset, acc, mask=mask_store)


# ----------------------------------------------------------------------
# Helper: LayerNorm without bias for possible reuse (float32)
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    """
    Manual LayerNorm over the last dimension.
    x : [..., C]
    weight, bias : [C] (or None)
    Returns normalized tensor of same shape and dtype (float32).
    """
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    y = (x - mean) * inv_std
    if weight is not None:
        y = y * weight
    if bias is not None:
        y = y + bias
    return y


# ----------------------------------------------------------------------
# Main entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] on the same device
        as the inputs (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # LayerNorm epsilon used in AlphaFold3

    # ------------------------------------------------------------------
    # Extract model weights (all are 2‑D Tensors, no bias on Linear layers)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over dim)
    # ------------------------------------------------------------------
    x_norm = _layer_norm(input_tensor, norm_weight, norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)    # [B,N,N,hidden]
    right = F.linear(x_norm, right_proj_weight) # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (binary/float). If config["nomask"] is True the mask
    #    is ignored (treated as all ones).  Mask is broadcast over the
    #    hidden dimension.
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of Linear(x_norm))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # Fuse gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Heavy contraction via Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Triton expects explicit stride arguments (int64)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_d = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_d = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_d = out.stride(3)

    # Tunable block sizes – chosen to fit into shared memory on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out_norm = _layer_norm(out, to_out_norm_weight, to_out_norm_bias, eps=eps)

    # Multiply by the output gate (broadcast over B,N,N)
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(out_norm, to_out_weight)   # shape [B,N,N,dim]

    return final
