"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣ **LayerNorm (FP32 → FP16)** – a tiny Triton kernel computes per‑row μ,
   σ² and writes the normalized tensor in half precision, saving bandwidth.

2️⃣ **Fuse projection + gate** – for the left and right paths we concatenate the
   linear weight matrices (proj & gate) and perform a single FP16 GEMM each.
   This yields:
        left_proj, left_gate_pre, right_proj, right_gate_pre  (shape …×H)
   The output‑gate is computed once from the same normalized input.

3️⃣ **Optional mask** – if a mask is supplied it is flattened and broadcasted
   over the hidden dimension (H).  Mask is applied *after* the projections but
   *before* the sigmoid‑gates, exactly as in the reference implementation.

4️⃣ **Sigmoid gates** – apply sigmoid to the gate tensors and multiply with
   their corresponding projections.

5️⃣ **Tri‑multiplicative update** – the operation
        out[i,j,d] = Σₖ left[i,k,d]·right[j,k,d]
   becomes a batched matrix multiplication for each hidden channel:
        (B·H) matrices of shape N×N  ×  (N×N)ᵀ.
   We use `torch.bmm`, which on H100 maps to a highly‑optimised CUTLASS
   kernel (Tensor‑Core FP16).

6️⃣ **Second LayerNorm (FP16 output)** – same Triton kernel as step 1, but
   normalising over the hidden dimension.

7️⃣ **Output‑gate & final projection** – multiply by the sigmoid‑output‑gate
   and linearly project back to `dim`.  The final result is cast back to
   float32 to match the PyTorch reference.

Only the two LayerNorms are written in Triton; all dense linear algebra is
delegated to PyTorch’s native FP16 kernels which already achieve near‑peak
throughput on an H100.  The implementation works for arbitrary batch size,
sequence length, `dim ∈ {128,384}` and `hidden_dim = 128`, with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm (float32 → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Compute a per‑row LayerNorm on a 2‑D tensor stored in row‑major order.
    Internally works in FP32, writes the normalized result in FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------- Pass 1: mean & variance ------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ----------- Pass 2: write normalized values ----------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]               # FP32
        out_fp16 = out.to(tl.float16)                     # cast

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (optional, dtype float32)
        - weights_dict : mapping str → torch.Tensor containing all model weights
        - config_dict  : contains at least
              dim, hidden_dim, nomask (bool, optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32, on the same device.
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                               # rows for flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) using Triton
    # --------------------------------------------------------------
    # flatten to [total_rows, dim] (contiguous)
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    # output buffer (FP16)
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256            # rows per program (tuned)
    BLOCK_D = 64             # cols per inner loop (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # reshape back to 4‑D
    x_norm_h = x_norm_h.view(B, N, N, dim)               # [B, N, N, dim] (FP16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (4 GEMMs total)
    # --------------------------------------------------------------
    # Concatenate proj & gate weights for left / right paths
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)  # [2*H, dim]

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)  # [2*H, dim]

    # Linear pass on the flattened view
    flat_norm = x_norm_h.view(total_rows, dim)           # [rows, dim]

    left_concat = F.linear(flat_norm, left_concat_w)    # [rows, 2*H]
    right_concat = F.linear(flat_norm, right_concat_w)  # [rows, 2*H]

    # split into proj and pre‑gate tensors
    left_proj_h   = left_concat[:, :hidden_dim]          # [rows, H]
    left_gate_pre = left_concat[:, hidden_dim:]          # [rows, H]
    right_proj_h  = right_concat[:, :hidden_dim]         # [rows, H]
    right_gate_pre = right_concat[:, hidden_dim:]        # [rows, H]

    # output‑gate (single linear + sigmoid)
    out_gate_h = torch.sigmoid(
        F.linear(flat_norm, weights["out_gate.weight"].to(torch.float16))
    )   # [rows, H] (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Apply optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape [B, N, N] → flatten → [rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates & gated projections
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre)          # [rows, H]
    right_gate_h = torch.sigmoid(right_gate_pre)        # [rows, H]

    left_h = left_proj_h * left_gate_h                   # [rows, H]
    right_h = right_proj_h * right_gate_h                # [rows, H]

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # bring hidden dim to batch axis for bmm: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim               # effective batch for bmm
    left_mat = left_perm.view(BH, N, N)   # [BH, N, N]
    right_mat = right_perm.view(BH, N, N) # [BH, N, N]

    # out_mat[b*h, i, j] = Σ_k left[i, k] * right[j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16 output)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, hidden_dim)      # FP16

    # --------------------------------------------------------------
    # 7️⃣ Output‑gate multiplication + final projection
    # --------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, hidden_dim)      # FP16
    out_norm = out_norm * out_gate                        # gated output

    final_h = F.linear(
        out_norm.view(total_rows, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows, dim] (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)  # cast back to FP32
    return final
