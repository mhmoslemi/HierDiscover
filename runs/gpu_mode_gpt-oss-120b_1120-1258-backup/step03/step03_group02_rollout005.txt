"""
TriMul “outgoing” implementation (AlphaFold‑3 style) – Triton + PyTorch.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – a small Triton kernel that
   computes per‑row mean/variance in FP32 and writes the normalized
   result directly as FP16, reducing memory traffic.

2️⃣ Fuse projection + gate for the left and right paths:
   a single half‑precision GEMM per side produces both the projection
   and the pre‑sigmoid gate tensors (2 × hidden_dim columns).  This yields
   four GEMMs total (left/right × proj/gate).

3️⃣ Optional mask (broadcast over the hidden dimension) is applied
   in‑place on the projected tensors.

4️⃣ Apply sigmoid gates and multiply gated projections (still FP16).

5️⃣ Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
   Implemented as a batched GEMM: we reshape to
   (B·hidden_dim, N, N) and call `torch.bmm` with the right operand
   transposed.  This leverages cuBLAS/Tensor‑cores and is far faster than
   Python loops.

6️⃣ Second LayerNorm (FP16 output) – same Triton kernel as step 1.

7️⃣ Apply the output‑gate (sigmoid) and a final half‑precision linear
   projection back to *dim*.  The result is cast to FP32 for the final
   output.

Only the two LayerNorms are written in Triton; dense linear algebra is
performed with PyTorch’s highly‑optimised FP16 kernels.  The implementation
covers all configuration variants (mask/no‑mask, variable `dim` (≤384)
and `hidden_dim` (=128) and batch sizes 1‑2).  The returned tensor has
shape `[bs, seq_len, seq_len, dim]` and `torch.float32` dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm that reads FP32/FP16 input and writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const *       [total_rows, dim]  (input)
    y_ptr,          #       *       [total_rows, dim]  (output FP16)
    w_ptr,          # const *       [dim]              (γ)
    b_ptr,          # const *       [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static dimension size
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    """
    Compute LayerNorm for each row of a 2‑D tensor stored in row‑major
    order.  The kernel works in FP32 internally and writes the normalized
    result as FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows                         # mask for rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # number of column tiles needed (known at compile‑time because dim is constexpr)
    NUM_D = tl.cdiv(dim, BLOCK_D)

    for d in range(0, NUM_D):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # pointer arithmetic: row * dim + col
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [BLOCK_M, BLOCK_D]

        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        # cast to float32 for accumulation
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values
    # ------------------------------------------------------------------
    for d in range(0, NUM_D):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [BLOCK_M, BLOCK_D]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        vals_f32 = vals.to(tl.float32)

        # (x - μ) * rstd
        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        # load γ (weight) and β (bias)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]   # still FP32
        out_fp16 = out.to(tl.float16)         # cast to FP16 for storage

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point – TriMul outgoing
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape            # input shape: [B, N, N, dim]
    total_rows = B * N * N                      # number of rows after flattening
    # flatten to [total_rows, dim] (row‑major)
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    # allocate half‑precision output buffer
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # launch Triton LN kernel
    BLOCK_M = 256      # rows per program (tuned)
    BLOCK_D = 64       # columns per tile (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # reshape back to 4‑D tensor (still FP16)
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (4 GEMMs total) – all FP16
    # --------------------------------------------------------------
    # Concatenate projection and gate weight matrices for the left side
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # Same for the right side
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # Flatten normalized input for linear layers
    x_norm_flat = x_norm_h.view(total_rows, dim)

    # Linear passes (produces [rows, 2*hidden_dim])
    left_concat = F.linear(x_norm_flat, left_concat_w)   # no bias
    right_concat = F.linear(x_norm_flat, right_concat_w)

    # Slice projection and pre‑gate tensors
    left_proj_h      = left_concat[:, :hidden_dim]                # [rows, H]
    left_gate_pre_h  = left_concat[:, hidden_dim:]                # [rows, H]
    right_proj_h     = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # Output‑gate (single linear, same for whole tensor)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_flat, weights["out_gate.weight"].to(torch.float16))
    )   # [rows, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten → [rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates and multiply projections
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h     # [rows, H]
    right_h = right_proj_h * right_gate_h   # [rows, H]

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update – batched GEMM
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Move hidden dimension to batch axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim                # batch size for BMM
    left_mat = left_perm.view(BH, N, N)    # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)  # [BH, N, N]

    # Batched matrix multiplication, summing over k dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # Reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16 output)
    # --------------------------------------------------------------
    total_rows_h = total_rows                     # same number of rows
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    # Reshape out_gate to broadcast correctly
    out_gate = out_gate_h.view(B, N, N, hidden_dim)   # FP16
    out_norm = out_norm * out_gate                     # gated output (FP16)

    # Final linear projection: hidden_dim → dim (FP16)
    final_h = F.linear(
        out_norm.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows_h, dim] (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)   # cast to FP32

    return final
