2️⃣  Fuse the five linear layers (left/right projections, left/right gates, out_gate)
    into a single FP16 GEMM:  Y = X_norm @ W_fused.T.
    Split Y into the five branches and apply the optional mask (only to the two
    projections) and the sigmoid‑gates.
3️⃣  Triangular multiplicative update:
        out[i,j] = Σ_k left[i,k] * right[j,k]
    This is exactly a batch matrix‑multiply L @ Rᵀ for each hidden channel.
    We reshape the tensors to (B·H, N, N) and call torch.bmm with a transpose,
    avoiding any extra copies.
4️⃣  Second LayerNorm + output‑gate – fused in a single Triton kernel.
5️⃣  Final linear projection back to `dim` (FP16) and cast to float32.

Only the two LayerNorm‑style kernels are written in Triton; all dense linear
algebra uses cuBLAS/torch kernels which are already optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton fused LayerNorm + output gate (used for the second norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*  [total_rows, dim]   (input to LN)
    gate_ptr,       # const half*  [total_rows, dim]   (sigmoid‑gate)
    y_ptr,          #       half*  [total_rows, dim]   (output)
    w_ptr,          # const float* [dim]              (γ)
    b_ptr,          # const float* [dim]              (β)
    total_rows,     # int
    dim: tl.constexpr,          # static (here = hidden_dim)
    eps: tl.constexpr,          # static
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute mean / variance (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, affine transform, multiply gate and store (FP16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load normalized x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load already‑sigmoid‑ed gate
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out = out * g_vals                                 # apply output‑gate

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor of shape [B, N, N]   (float32 or bool)
        weights_dict : {
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        }
        config_dict  : {
            "dim"        : int,
            "hidden_dim" : int,
            "nomask"     : bool (True → ignore mask)
        }

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]          # input / output channel dimension
    hidden_dim = config["hidden_dim"]   # hidden dimension used inside TriMul
    eps        = 1e-5
    nomask     = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_pairs = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm  (use torch's highly‑optimised implementation)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                           # [B, N, N, dim]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear projections + gates into ONE GEMM (FP16)
    # ------------------------------------------------------------------
    # move to half precision for the heavy linear algebra
    x_norm_h = x_norm.view(-1, dim).to(torch.float16)               # [R, dim]

    # concatenate the five weight matrices once (shape: [5*H, dim])
    weight_concat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(dtype=torch.float16, device=device)               # [(5*H), dim]

    # Single GEMM – (R, dim) @ (dim, 5*H).  Result shape: [R, 5*H]
    proj = F.linear(x_norm_h, weight_concat)                        # FP16

    # Split into the five logical branches
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = \
        torch.split(proj, hidden_dim, dim=1)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) – only to the projections
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_pairs, 1).to(dtype=torch.float16, device=device)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates (still FP16)
    # ------------------------------------------------------------------
    left  = left_proj_raw  * torch.sigmoid(left_gate_raw)   # [R, H]
    right = right_proj_raw * torch.sigmoid(right_gate_raw) # [R, H]
    out_gate = torch.sigmoid(out_gate_raw)                 # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, hidden_dim)                  # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)                 # [B, N, N, H]

    # bring hidden channel into the batch dimension
    BH = B * hidden_dim
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(BH, N, N)   # [BH, i, k]
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(BH, N, N)  # [BH, j, k]

    # out = Σ_k left[i,k] * right[j,k]  →  left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))           # [BH, i, j]

    # restore original layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat      = out.view(total_pairs, hidden_dim)      # [R, H]   FP16
    out_gate_flat = out_gate.view(total_pairs, hidden_dim)# [R, H]   FP16
    out_norm_gated = torch.empty_like(out_flat)           # destination (FP16)

    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(out_flat.shape[0], BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_flat.shape[0],
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gated,
        weights["to_out.weight"].to(dtype=torch.float16, device=device)   # [dim, H]
    )   # [R, dim]  FP16

    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
