"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ First LayerNorm (C‑dim) – use PyTorch’s native kernel (fast, fused fp32
    reduction → fp16 output).  The result is cast to FP16 for the remaining
    GEMMs.
2️⃣ Fuse the five linear projections (left/right + three gates) into a single
    FP16 GEMM via `torch.nn.functional.linear`.
3️⃣ Optional mask (applied to left/right projections) – element‑wise FP16.
4️⃣ Apply sigmoid gates and element‑wise gating (still FP16).
5️⃣ Triangular multiplicative update:  out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k] .
   Realized as a batched FP16 GEMM `left @ rightᵀ` on tensors of shape
   (B·H, N, N) using `torch.bmm`.
6️⃣ Second LayerNorm (H‑dim) – custom Triton kernel that computes mean/var in
    FP32 and writes directly in FP16 (no extra cast).
7️⃣ Gated output × out_gate  (FP16)  → final linear projection back to C
   (FP16 GEMM) and cast to FP32.

Only the second LayerNorm is written in Triton (required “some Triton code”).
All heavy linear algebra is performed with cuBLAS/Tensor‑Core FP16 GEMMs,
keeping memory traffic low and achieving sub‑millisecond runtimes on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 reduction, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                 # *const   half*   [total_rows, dim]
    y_ptr,                 # *mut     half*   [total_rows, dim]
    weight_ptr,            # *const   float*  [dim]
    bias_ptr,              # *const   float*  [dim]
    total_rows,            # int (runtime)
    dim: tl.constexpr,     # int (compile‑time)
    eps: tl.constexpr,     # float (compile‑time)
    BLOCK_M: tl.constexpr, # rows per program
    BLOCK_D: tl.constexpr, # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows                         # mask for rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile of the input
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # FP16 → FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Normalise, scale, bias and write FP16 output
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # fp32

        out = norm * w[None, :] + b[None, :]            # fp32 result
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr,
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C], float32
        - mask_tensor : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights_dict : dict mapping parameter names to tensors
        - config_dict  : {"dim", "hidden_dim", "nomask" (bool, optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                         # rows for first LN

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm using PyTorch (fast, fp32 reduction)
    # ------------------------------------------------------------------
    # Torch’s native LayerNorm is highly optimized on H100.
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to FP16 for the remainder of the pipeline
    x_norm = x_norm_fp32.half()                    # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right + 3 gates) into one GEMM
    # ------------------------------------------------------------------
    # Shape: (total_rows, C) @ (C, 5*H)ᵀ  → (total_rows, 5*H)
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                         # (5*H, C)

    proj_all = F.linear(x_norm.view(total_rows, dim), proj_weight)  # FP16

    # Split into the five tensors
    splits = torch.split(proj_all, hidden_dim, dim=1)
    left_proj, right_proj, left_gate, right_gate, out_gate = splits

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied BEFORE gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [total_rows, 1] (float16)
        mask_fp16 = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate          # [total_rows, H]
    right = right_proj * right_gate       # [total_rows, H]

    # Reshape to 4‑D layout required for the triangular update
    left = left.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim) # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
    # ------------------------------------------------------------------
    # Bring the hidden dimension into the batch dimension.
    # This results in copies (contiguous) – required for torch.bmm.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # transpose on k‑axis

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)  # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat)   # FP16, shape [B*H, N, N]

    # Back‑to 4‑D tensor
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dimension) – Triton implementation
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)          # [total_rows_h, H]
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    # Tune block sizes: hidden_dim ≤ 384 ⇒ BLOCK_D = 64 works well.
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # Implicit dependency guarantees out_norm is ready.

    out_norm = out_norm.view(B, N, N, hidden_dim)          # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                           # FP16 element‑wise

    to_out_weight = weights["to_out.weight"].half()
    final_fp16 = F.linear(gated.view(total_rows_h, hidden_dim), to_out_weight)  # FP16

    # Cast back to FP32 for the final output
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
