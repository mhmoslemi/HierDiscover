   `[R, dim] @ [5*hidden, dim].T → [R, 5*hidden]`.  Slicing yields the two
   projected tensors and the three gate vectors.

3️⃣ Optional mask (applied only to the two projected tensors) and three
   sigmoid gates.

4️⃣ Triangular multiplicative update:
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This is a batched matrix multiplication per hidden dimension,
   computed with `torch.matmul` on tensors of shape `[B, H, N, N]`,
   fully exploiting Tensor‑Cores.

5️⃣ Second LayerNorm over the hidden dimension (again a Triton kernel).

6️⃣ Apply the output gate, project back to `dim` with a half‑precision
   linear layer, and finally cast to `float32`.

All heavy linear layers run in fp16, the two LayerNorms run in fp16
(their parameters are fp32), and the final result is returned as fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (works for FP32 or FP16 input, writes FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *  [total_rows, dim]   (float32 or float16)
    y_ptr,            # mutable *[total_rows, dim]   (float16 output)
    weight_ptr,       # const *  [dim]               (float32)
    bias_ptr,         # const *  [dim]               (float32)
    total_rows,       # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass: compute mean & variance per row
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2nd pass: write normalized value * gamma + beta (store as fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # beta
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom TriMul forward (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                       # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – result in fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()  # [R, dim], fp32
    x_norm_flat = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input ptr (float32)
        x_norm_flat,                # output ptr (float16)
        weights["norm.weight"],     # gamma (float32)
        weights["norm.bias"],       # beta  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse the five linear layers (proj + 3 gates) into one GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices: shape [5*hidden_dim, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # fp16

    # Single half‑precision linear transformation
    out_concat = F.linear(x_norm_flat, w_concat)   # [R, 5*hidden_dim] fp16

    hs = hidden_dim
    left_proj   = out_concat[:, 0:hs]                     # [R, H]
    right_proj  = out_concat[:, hs:2*hs]
    left_gate_p = out_concat[:, 2*hs:3*hs]
    right_gate_p= out_concat[:, 3*hs:4*hs]
    out_gate_p  = out_concat[:, 4*hs:5*hs]

    # Sigmoid gates (half precision)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # Optional mask (broadcast over hidden dim)
    if not nomask:
        # mask shape: [B, N, N] → flatten → [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Element‑wise gating
    left  = left_proj * left_gate          # [R, H] fp16
    right = right_proj * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 3) Reshape to 4‑D tensors [B, N, N, H]
    # --------------------------------------------------------------
    left  = left.view(B, N, N, hs)
    right = right.view(B, N, N, hs)
    out_gate = out_gate.view(B, N, N, hs)   # for later use

    # --------------------------------------------------------------
    # 4) Triangular multiplicative update using batched matmul
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Transpose right to align the k dimension:
    #   left : [B, H, i, k]   (after permute)
    #   right: [B, H, k, j]
    L = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    R = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    # Batched matmul over the leading dimensions (B*H)
    out_mat = torch.matmul(L, R)                # [B, H, N, N]

    # Rearrange back to [B, N, N, H]
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 5) Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hs).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        total_rows,
        hs,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_flat.view(B, N, N, hs)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6) Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # element‑wise, fp16

    # Final linear projection: hidden → dim
    final_half = F.linear(
        out_norm.view(total_rows, hs),            # [R, H] fp16
        weights["to_out.weight"].to(torch.float16)  # [dim, H]
    )                                            # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)  # cast back to fp32
    return final
```
