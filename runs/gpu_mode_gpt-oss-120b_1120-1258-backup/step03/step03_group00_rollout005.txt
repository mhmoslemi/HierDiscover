@triton.jit
def layernorm_fp16_gate_kernel(
    x_ptr, y_ptr, gate_ptr,
    weight_ptr, bias_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    # pass 1
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals_fp16 = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0) # half
        vals = vals_fp16.to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # pass 2
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals_fp16 = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals = vals_fp16.to(tl.float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32
        out = norm * w + b               # fp32
        # gating
        gate_vals_fp16 = tl.load(gate_ptr + (offs_m[:, None] * dim + offs_d[None, :]),
                                 mask=mask_m[:, None] & mask_d[None, :],
                                 other=0.0)
        gate_vals = gate_vals_fp16.to(tl.float32)
        out = out * gate_vals
        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])
```

"""
custom_kernel – Optimized “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm over the last channel *C* of the input tensor (fp32
    accumulation, fp16 output).  Implemented with a Triton kernel.

2️⃣  Fuse the five linear projections (left/right and three gates) into a
    single GEMM by concatenating the weight matrices.  All linear work
    is kept in fp16 (Tensor‑Core friendly).

3️⃣  Optional pair‑wise mask is applied to the left/right projections.

4️⃣  Apply sigmoid gates (fp16) and element‑wise gating.

5️⃣  Triangular multiplicative update:
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    realised as a batched matrix‑multiply (`torch.bmm`) on tensors of
    shape (B·H, N, N) in fp16.

6️⃣  Layer‑norm over the hidden dimension *H* and simultaneous multiplication
    by the output gate – fused in a second Triton kernel (no extra memory
    pass).

7️⃣  Final linear projection H → C (fp16) and cast back to fp32.

All heavy reductions are done by custom Triton kernels; dense linear algebra
leverages cuBLAS/Tensor‑Cores. The implementation supports optional masking
and works for the full range of test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernels
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                # *const float*   [total_rows, dim] (fp32 input)
    y_ptr,                # *mut   half*    [total_rows, dim] (fp16 output)
    weight_ptr,           # *const float*   [dim]            (γ)
    bias_ptr,             # *const float*   [dim]            (β)
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1: compute row‑wise mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32

        # (x - mean) * rstd  → fp32
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β are fp32; promote to fp16 right before store
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b            # fp32
        out_fp16 = out_fp32.to(tl.float16) # fp16

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def layernorm_fp16_gate_kernel(
    x_ptr,                # *const half*    [total_rows, dim] (fp16 input)
    y_ptr,                # *mut   half*    [total_rows, dim] (fp16 output)
    gate_ptr,             # *const half*    [total_rows, dim] (fp16 gate)
    weight_ptr,           # *const float*   [dim]            (γ)
    bias_ptr,             # *const float*   [dim]            (β)
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1: mean & variance
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals_fp16 = tl.load(ptrs,
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0)
        vals = vals_fp16.to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2: normalize, apply γ/β, multiply by gate, store
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # Load input
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals_fp16 = tl.load(ptrs,
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0)
        vals = vals_fp16.to(tl.float32)

        # Load γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        # Normalize
        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32
        out  = norm * w + b                                 # fp32 (γ·x̂ + β)

        # Load gate (fp16 → fp32) and apply
        gate_vals_fp16 = tl.load(gate_ptr + (offs_m[:, None] * dim + offs_d[None, :]),
                                 mask=mask_m[:, None] & mask_d[None, :],
                                 other=0.0)
        gate_vals = gate_vals_fp16.to(tl.float32)
        out = out * gate_vals                               # fp32

        # Store as fp16
        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor   [B, N, N, C] (float32)
        mask_tensor  : torch.Tensor   [B, N, N] (bool/float) – may be ignored
        weights_dict : dict of torch.Tensors
            Expected keys:
                norm.weight, norm.bias,
                left_proj.weight, right_proj.weight,
                left_gate.weight, right_gate.weight, out_gate.weight,
                to_out_norm.weight, to_out_norm.bias,
                to_out.weight
        config_dict  : dict with entries
            dim, hidden_dim, nomask (bool, optional)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape   # [B, N, N, C]
    total_rows = B * N * N             # rows for the flattened 2‑D view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # fp32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 128   # rows per program
    BLOCK_D = 64    # columns per tile (must divide dim, handled by mask)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # --------------------------------------------------------------
    # Concatenate weight matrices (shape: (5*H, C)) and cast to fp16
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Linear projection – bias is zero for all five Linear layers
    proj_all = F.linear(x_norm_fp16, concat_weight)   # [total_rows, 5*H] (fp16)

    # Split into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: [total_rows, H]

    # --------------------------------------------------------------
    # 2a️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 2b️⃣ Sigmoid gates (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating
    left = left_proj * left_gate      # [total_rows, H]
    right = right_proj * right_gate   # [total_rows, H]

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape back to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)      # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)    # [B, N, N, H]

    # Bring hidden dim to the second axis for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k‑axis transposed)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                # [B·H, N, N]
    right_mat = right_perm.view(BH, N, N)              # [B·H, N, N]

    # Batched matrix multiplication (fp16, Tensor‑Core friendly)
    out_mat = torch.bmm(left_mat, right_mat)          # [B·H, N, N] (fp16)

    # Reshape to original 4‑D layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output gate (fused)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(total_rows, hidden_dim).contiguous()
    out_norm_gated_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_gate_kernel[grid2](
        out_flat,
        out_norm_gated_fp16,
        out_gate_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final linear projection H → C and cast to fp32
    # --------------------------------------------------------------
    to_out_weight_fp16 = weights["to_out.weight"].half().contiguous()
    final_fp16 = F.linear(out_norm_gated_fp16, to_out_weight_fp16)   # [total_rows, C] (fp16)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
