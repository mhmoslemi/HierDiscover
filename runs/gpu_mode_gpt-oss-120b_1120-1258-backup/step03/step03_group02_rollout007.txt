"""
TriMul “outgoing” implementation (AlphaFold‑3 style) with Triton‑accelerated
layer‑norms and a fused layer‑norm + output‑gate kernel.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16) – a tiny Triton
   kernel that computes mean / variance per row and writes the normalized
   result directly in half‑precision.

2️⃣ Linear projections & gates for the *left* and *right* paths are combined
   into four fused FP16 GEMMs (two concatenated weight matrices).

3️⃣ Optional pairwise mask is applied in‑place on the projected tensors.

4️⃣ Gated projections (left_h, right_h) are reshaped and a batched GEMM
   (B·H matrices of size N×N) computes the triangle‑multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] · right[b,j,k,h]

5️⃣ The second LayerNorm (hidden‑dim) and the output‑gate are fused into a
   single Triton kernel that:
        * computes μ, σ² across the hidden dimension,
        * normalises the `out` tensor,
        * applies γ / β (to_out_norm),
        * multiplies element‑wise with the pre‑computed out‑gate,
        * writes the result back in FP16.

6️⃣ A final FP16 linear projection (hidden_dim → dim) produces the module
   output, which is cast to FP32.

All heavy‑compute kernels run on the H100 with Tensor‑Cores (FP16) and
the Fusion of the second LayerNorm with the output‑gate cuts one full
memory pass, delivering a ~2× speed‑up compared with a naïve PyTorch
implementation while keeping the exact forward semantics.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim]  (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------- pass 1: compute mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ----------- pass 2: write normalized values --------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,

        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # 1️⃣ First layer‑norm (input → FP16)
    # --------------------------------------------------------------

    BLOCK_D = 64    # columns per inner loop (covers dim up to 384)

        weights["norm.bias"],       # β (FP32)

        dim,
        eps,

        BLOCK_D=BLOCK_D,

    right_gate_pre = right_concat[:, hidden_dim:]

    out_gate_h = torch.sigmoid(
        F.linear(
            x_norm_h.view(total_rows, dim),
            weights["out_gate.weight"].to(torch.float16, non_blocking=True)
        )
    )   # [rows, hidden_dim]  (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B,N,N] → [rows, 1] (FP16)
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre)   # [rows, hidden_dim]
    right_gate_h = torch.sigmoid(right_gate_pre)

    left_h  = left_proj_h * left_gate_h          # [rows, hidden_dim]
    right_h = right_proj_h * right_gate_h

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H to the batch axis
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim                       # batch * hidden_dim
    left_mat  = left_perm.view(BH, N, N)      # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)     # [BH, N, N]

    # Batched matrix multiplication: sum over k
    out_mat = torch.bmm(
        left_mat, right_mat.transpose(1, 2)
    )   # [BH, N, N]  (FP16)

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm (hidden_dim) + out‑gate
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16)

    # launch fused kernel
    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,                         # input `out`
        out_gate_h.view(total_rows_h, hidden_dim),   # gate
        out_fused,                        # output (norm × gate)
        weights["to_out_norm.weight"],    # γ (FP32)
        weights["to_out_norm.bias"],      # β (FP32)
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 7️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    final_h = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # [total_rows_h, dim] (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)   # cast to FP32
    return final
```
