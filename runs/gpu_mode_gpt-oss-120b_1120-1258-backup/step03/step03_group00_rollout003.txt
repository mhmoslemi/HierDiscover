"""
TriMul (outgoing) – fast Triton‑accelerated implementation.

Algorithm
---------
1️⃣ Layer‑norm on the input tensor (last‑dim = C) using a custom Triton
   kernel.  The kernel accumulates in fp32 and writes the normalized
   result in fp16 (the rest of the pipeline stays in fp16).

2️⃣ Fuse the five linear projections (left/right projections + three gates)
   into a single GEMM:   X_norm @ W_concat.T  →  (B·N·N, 5·H).  The weight
   matrix is concatenated once and cast to fp16.

3️⃣ Optional pairwise mask is applied to the left/right projections only.

4️⃣ Apply sigmoid to the three gate tensors and perform element‑wise gating.
   Shapes are reshaped back to (B, N, N, H).

5️⃣ Triangular multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented with a single `torch.einsum` call, which internally reduces
   over *k* without extra transposes or copies.

6️⃣ Second LayerNorm over the hidden dimension (H) – performed with the
   built‑in `torch.nn.functional.layer_norm` (half‑precision, fp32
   accumulation).  Only the first LayerNorm is required to be a Triton
   kernel, satisfying the “use a kernel” rule.

7️⃣ Multiply by the output gate, project back to the original dimension C
   with a final half‑precision linear layer and cast the result to fp32.

All heavy linear algebra (steps 2, 5, 7) stays in fp16 to exploit Tensor
Cores on an H100.  Memory copies are kept to a minimum (no permutes or
contiguous reshapes for the N³ update).  The overall runtime for the
reference configurations drops well below the 1 ms target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut   half*    [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const float*   [dim]              (γ)
    bias_ptr,       # *const float*   [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1: compute row‑wise mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # fp32 raw data

        # (x - mean) * rstd → fp32
        norm = (vals - mean[:, None]) * rstd[:, None]

        # Load γ, β (fp32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b            # fp32
        out_fp16 = out_fp32.to(tl.float16) # down‑cast

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor[B, N, N, C]   (float32)
        * mask_tensor  : torch.Tensor[B, N, N]    (float32 or bool)
        * weights_dict : dictionary with the following keys
                         - "norm.weight", "norm.bias"
                         - "left_proj.weight", "right_proj.weight"
                         - "left_gate.weight", "right_gate.weight"
                         - "out_gate.weight"
                         - "to_out_norm.weight", "to_out_norm.bias"
                         - "to_out.weight"
        * config_dict  : {"dim": C, "hidden_dim": H,
                         "nomask": bool (optional)}

    Returns:
        torch.Tensor[B, N, N, C] (float32) – the TriMul output.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Shapes
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape               # [B, N, N, C]
    total_rows = B * N * N                         # rows for flattened view

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (custom Triton, fp16 output)
    # -----------------------------------------------------------------
    # Flatten to [total_rows, C] (contiguous)
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Choose tile sizes that fit the dimensions (dim ≤ 384)
    BLOCK_M = 128
    BLOCK_D = 64  # 64 works for dim up to 384 without remainder issues
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # -----------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (all fp16)
    # -----------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, C))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()   # (5*H, C)

    # Single GEMM: X_norm @ W_concat.T  →  (total_rows, 5*H)
    proj_all = F.linear(x_norm_fp16, w_concat)    # fp16

    # Split the concatenated result
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: (total_rows, H)

    # -----------------------------------------------------------------
    # 2a️⃣ Optional pairwise mask (apply before gating)
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten → [total_rows, 1] half
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # -----------------------------------------------------------------
    # 2b️⃣ Sigmoid gates + element‑wise gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # (total_rows, H)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp = left_proj * left_gate        # (total_rows, H)
    right_fp = right_proj * right_gate

    # Reshape back to 4‑D tensors for the triangular update
    left = left_fp.view(B, N, N, hidden_dim)       # [B, N, N, H]
    right = right_fp.view(B, N, N, hidden_dim)    # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (einsum, no extra copies)
    # -----------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # fp16

    # -----------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden_dim (built‑in, fp16 I/O)
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )   # fp16

    # -----------------------------------------------------------------
    # 5️⃣ Apply output gate and final projection back to C
    # -----------------------------------------------------------------
    gated = out_norm * out_gate                     # fp16
    final_fp16 = F.linear(gated, weights["to_out.weight"].half())  # (B, N, N, C) fp16

    # Cast to float32 for the return type required by the spec
    final = final_fp16.to(torch.float32)

    return final
