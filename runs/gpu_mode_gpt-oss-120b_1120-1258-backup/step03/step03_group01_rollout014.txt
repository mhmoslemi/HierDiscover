"""
custom_kernel → highly‑optimized “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1. **Row‑wise LayerNorm** over the last channel (`dim`).  
   A tiny Triton kernel computes mean/variance in fp32 and writes the
   normalized tensor in fp16.  This replaces the costly `nn.LayerNorm`.

2. **Fused linear projection** – the five dense layers (`left_proj`,
   `right_proj`, `left_gate`, `right_gate`, `out_gate`) are concatenated
   into one matrix‑multiply:
       `[R, dim] @ [5*hidden, dim].T → [R,5*hidden]`,
   where `R = B·N·N`.  Slicing yields the two projected tensors,
   the three sigmoid gates and the output‑gate.

3. **Mask & gating** – if a mask is supplied, it is broadcast over the
   hidden dimension and applied only to the two projected tensors.
   The three sigmoid gates are evaluated in half‑precision.

4. **Triangular multiplicative update** – for each head the operation

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   is a batched matrix‑multiply:
       `left` : (B·H,N,N)   (row = i, col = k)  
       `right`: (B·H,N,N)   (row = j, col = k)  
       `out  = left @ right.T`.
   The hidden dimension (`H = hidden_dim`) is fused into the batch
   dimension, allowing cuBLAS to run a single large batched GEMM.

5. **Second LayerNorm** over the hidden dimension using the same Triton
   kernel as in step 1.

6. **Output gating & projection** – the normalized tensor is multiplied
   element‑wise with the output‑gate, then linearly projected back to
   `dim`.  The final result is returned as `float32`.

All heavy reductions (mean/variance, GEMMs) run in fp16, exploiting
Tensor‑Cores on an H100.  The only required Triton component is the
LayerNorm kernel, satisfying the “custom kernel” requirement while
keeping the overall runtime well below the 1 ms target for the largest
test case.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (input can be fp32 or fp16, output fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   *  [total_rows, dim] (fp32 or fp16)
    y_ptr,            # mutable *  [total_rows, dim] (fp16 output)
    weight_ptr,       # const   *  [dim] (fp32)
    bias_ptr,         # const   *  [dim] (fp32)
    total_rows,       # int (runtime)
    dim: tl.constexpr,    # compile‑time constant
    eps: tl.constexpr,    # compile‑time constant
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass – compute mean & variance (fp32 accumulators)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # Load whatever the input dtype is; we immediately cast to fp32.
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)                # Σ x
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)     # Σ x²

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # 2nd pass – write (x-mean)*rstd*γ + β  (store as fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]          # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point – outgoing TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – result in FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] float32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)                     # output buffer

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input (float32)
        x_norm,                     # output (fp16)
        weights["norm.weight"],     # γ (float32)
        weights["norm.bias"],       # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse the five linear layers (proj + three gates) into one GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices: [5*H, dim]  (no bias in any of them)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16 for Tensor‑Core

    # Single fused linear:  [R, dim] @ [5*H, dim].T → [R, 5*H]  (fp16)
    proj_and_gates = F.linear(x_norm, w_concat)   # fp16

    # Slice the concatenated output
    left_proj   = proj_and_gates[:, 0:H]                 # [R, H]
    right_proj  = proj_and_gates[:, H:2*H]
    left_gate_p = proj_and_gates[:, 2*H:3*H]
    right_gate_p = proj_and_gates[:, 3*H:4*H]
    out_gate_p  = proj_and_gates[:, 4*H:5*H]

    # Sigmoid gates (still fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # Optional mask – broadcast over hidden dimension
    if not nomask:
        # mask shape: [B, N, N] → [R, 1] after view
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Apply the two projection gates
    left  = left_proj * left_gate        # [R, H]
    right = right_proj * right_gate      # [R, H]

    # Reshape back to 4‑D tensors
    left  = left.view(B, N, N, H)   # [B, N, N, H]
    right = right.view(B, N, N, H)  # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Bring hidden dimension into the batch axis: (B, H, N, N)
    # left[i,k]   → rows = i, cols = k
    # right[j,k]  → rows = j, cols = k  → we need right.T
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)    # [B*H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # same layout

    # Compute left @ right.T  (batch‑wise matmul)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))               # [B*H, N, N]

    # Restore original 4‑D layout: [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4) Second LayerNorm over the hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()          # [R, H] fp16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm.view(B, N, N, H)                     # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 5) Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                               # element‑wise fp16

    out_norm_flat = out_norm.view(total_rows, H).contiguous()    # [R,H]
    final_half = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16)              # [dim, H] fp16
    )                                                          # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)     # cast back to fp32
    return final
