@triton.jit
def final_proj_kernel(
    out_ptr, out_gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    weight_T_ptr,
    eps,
    total_rows, hidden_dim, dim,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid
    if row >= total_rows:
        return

    # Compute mean & var
    sum_val = tl.float32(0)
    sum_sq = tl.float32(0)
    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        ptr = out_ptr + row * hidden_dim + offs_h
        vals = tl.load(ptr, mask=mask_h, other=0.0).to(tl.float32)
        sum_val += tl.sum(vals)
        sum_sq += tl.sum(vals * vals)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Loop over output dimension blocks
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        acc = tl.zeros([BLOCK_D], dtype=tl.float32)

        # Loop over hidden dimension blocks
        for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            # Load raw values
            out_vals = tl.load(out_ptr + row * hidden_dim + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            gate_vals = tl.load(out_gate_ptr + row * hidden_dim + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            w_norm = tl.load(norm_weight_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            b_norm = tl.load(norm_bias_ptr   + offs_h, mask=mask_h, other=0.0).to(tl.float32)

            # LN + gate
            normed = (out_vals - mean) * rstd
            ln_out = normed * w_norm + b_norm
            gated = ln_out * gate_vals    # shape [BLOCK_H]

            # Load weight_T slice (hidden_dim x dim) part for this block
            # weight_T has shape (hidden_dim, dim)
            w_T = tl.load(weight_T_ptr + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :],
                          other=0.0).to(tl.float32)   # shape [BLOCK_H, BLOCK_D]

            # accumulate: gated (1, BLOCK_H) @ w_T (BLOCK_H, BLOCK_D) -> (1, BLOCK_D)
            # Equivalent to sum over H: gated[h] * w_T[h,d]
            # Use broadcasting: (gated[None, :] * w_T) sum over axis 0?
            # Let's compute: gated_vec = gated[None, :] shape (1, BLOCK_H)
            # mul = gated_vec * w_T          shape (BLOCK_H, BLOCK_D)? Actually broadcasting: (1, BLOCK_H) * (BLOCK_H, BLOCK_D) -> (BLOCK_H, BLOCK_D) where each row of w_T multiplied by gated element (broadcast across rows?). Need to broadcast across rows: we want each row h multiplied by gated[h]. So we should compute w_T * gated[:, None] if we want each row scaled. So:
            mul = w_T * gated[:, None]  # w_T shape (BLOCK_H, BLOCK_D), gated[:, None] shape (BLOCK_H,1). Multiply each row.
            acc += tl.sum(mul, axis=0)   # sum across H, resulting shape (BLOCK_D,)

        out_index = row * dim + offs_d
        tl.store(out_index + ????) ...

"""
Highly‑optimized Triton + PyTorch implementation of the *outgoing* TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ First LayerNorm over the channel dimension C using the native
   torch.nn.functional.layer_norm (fast fused kernel).

2️⃣ Fuse the five linear projections (left/right and three gates) into a
   single half‑precision GEMM by concatenating their weight matrices.
   The result is split back into the individual tensors.

3️⃣ Apply the optional pair‑wise mask and the three sigmoid gates.

4️⃣ Batched half‑precision matrix‑multiply implements the triangular
   multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
   (B * H independent GEMMs).

5️⃣ A custom Triton kernel fuses the second LayerNorm, the output‑gate
   multiplication and the final linear projection back to the original
   channel dimension:
        y = LN(out)                     # uses to_out_norm.{weight,bias}
        y = y * out_gate
        out = y @ to_out.weight.T
   The kernel computes the mean/variance of each row, applies LayerNorm,
   multiplies by the gate and reduces with the final weight matrix,
   writing fp16 results directly.

The kernel launches one program per output row (i.e. per (b,i,j) tuple),
which fully utilizes the H100 Tensor‑Cores and keeps the whole
tri‑mul forward pass well below the 1 ms target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def final_proj_kernel(
    out_ptr,                # *const half*  [total_rows_h, hidden_dim]    (tri‑mul output)
    out_gate_ptr,           # *const half*  [total_rows_h, hidden_dim]    (output gate)
    norm_weight_ptr,        # *const float* [hidden_dim]                  (to_out_norm.weight)
    norm_bias_ptr,          # *const float* [hidden_dim]                  (to_out_norm.bias)
    weight_T_ptr,           # *const half*  [hidden_dim, dim]            (to_out.weight.T)
    out_fp16_ptr,           # *mut   half*  [total_rows_h, dim]          (final result)
    eps,                    # float (scalar epsilon for LN)
    total_rows_h,           # int    (B * N * N)
    hidden_dim: tl.constexpr,
    dim: tl.constexpr,
    BLOCK_H: tl.constexpr,  # hidden‑dim tile (e.g. 32)
    BLOCK_D: tl.constexpr,  # output‑dim tile (e.g. 64)
):
    pid = tl.program_id(0)
    row = pid
    if row >= total_rows_h:
        return

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean / variance of the raw tri‑mul output (fp16 -> fp32)
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for h_start in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_start * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        ptr = out_ptr + row * hidden_dim + offs_h
        vals = tl.load(ptr, mask=mask_h, other=0.0).to(tl.float32)   # [BLOCK_H]
        sum_val += tl.sum(vals)
        sum_sq  += tl.sum(vals * vals)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Loop over output‑dimension blocks and accumulate the final
    #    projection (LN → gate → linear).  Each block produces BLOCK_D
    #    output elements.
    # ------------------------------------------------------------------
    for d_start in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        acc = tl.zeros([BLOCK_D], dtype=tl.float32)    # accumulator for this block

        # ----- scan hidden dimension ------------------------------------------------
        for h_start in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h_start * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            # raw tri‑mul output
            out_val = tl.load(out_ptr + row * hidden_dim + offs_h,
                              mask=mask_h,
                              other=0.0).to(tl.float32)          # [BLOCK_H]
            # output gate
            gate_val = tl.load(out_gate_ptr + row * hidden_dim + offs_h,
                               mask=mask_h,
                               other=0.0).to(tl.float32)         # [BLOCK_H]

            # LayerNorm parameters (to_out_norm.{weight,bias})
            w_norm = tl.load(norm_weight_ptr + offs_h,
                             mask=mask_h,
                             other=0.0).to(tl.float32)          # [BLOCK_H]
            b_norm = tl.load(norm_bias_ptr + offs_h,
                             mask=mask_h,
                             other=0.0).to(tl.float32)          # [BLOCK_H]

            # ----- LN + gate ---------------------------------------------------------
            normed   = (out_val - mean) * rstd                 # x̂
            ln_out   = normed * w_norm + b_norm                # γ·x̂ + β
            gated    = ln_out * gate_val                       # (γ·x̂+β)·gate

            # ----- weight_T block (hidden_dim × dim) --------------------------------
            # weight_T has shape (hidden_dim, dim) → stride0 = dim, stride1 = 1
            w_T = tl.load(weight_T_ptr + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :],
                          other=0.0).to(tl.float32)            # [BLOCK_H, BLOCK_D]

            # accumulate contribution: Σ_h gated[h] * weight_T[h, d]
            # broadcasting gated over the dim‑axis
            acc += tl.sum(w_T * gated[:, None], axis=0)        # [BLOCK_D]

        # ------------------------------------------------------------------
        # 3️⃣ Store the computed block (fp16)
        # ------------------------------------------------------------------
        out_ptr = out_fp16_ptr + row * dim + offs_d
        tl.store(out_ptr, acc.to(tl.float16), mask=mask_d)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32, on the same CUDA device)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dim C
    hidden_dim = config["hidden_dim"]   # projection dim H (always 128 in the tests)
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    B, N, _, C = input_tensor.shape
    total_rows = B * N * N               # rows for the first linear layer

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (native PyTorch – fast fused kernel)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                 # cast to fp16 for the unified GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + 3 gates)
    # ------------------------------------------------------------------
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).contiguous().half()               # shape (5*H, C)

    # Single GEMM: (B·N·N, C) @ (C, 5*H)^T → (B·N·N, 5*H)
    proj_all = F.linear(x_norm, concat_weight)   # fp16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = \
        torch.split(proj_all, hidden_dim, dim=1)

    # ------------------------------------------------------------------
    # 2a. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b. Apply sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)     # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)      # will be used later

    left_fp16  = left_proj_fp16 * left_gate       # [total_rows, H]
    right_fp16 = right_proj_fp16 * right_gate

    # Reshape to (B, N, N, H) for the triangular update
    left = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update via batched GEMM
    #    out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
    # ------------------------------------------------------------------
    # Permute to (B*H, N, N) where the multiplication axis is the last dim of left
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # right needs its k‑axis to be the first dimension for the GEMM
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat)               # fp16, shape (B*H,N,N)

    # Back to (B, N, N, H) and keep fp16
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Fuse second LayerNorm, output‑gate and final projection into a Triton kernel
    # ------------------------------------------------------------------
    total_rows_h = B * N * N                         # rows for the final kernel
    out_flat = out.view(total_rows_h, hidden_dim)    # [total_rows_h, H]
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)  # same shape

    # to_out_norm parameters (γ, β) – float32
    norm_weight = weights["to_out_norm.weight"]
    norm_bias   = weights["to_out_norm.bias"]

    # Final linear weight transposed: shape (H, C) → (hidden_dim, dim)
    to_out_weight_T = weights["to_out.weight"].t().contiguous().half()

    final_fp16 = torch.empty((total_rows_h, dim), dtype=torch.float16, device=device)

    # Tunable tile sizes – chosen to fit the H100 registers and shared memory
    BLOCK_H = 32            # hidden‑dim tile (covers hidden_dim ≤ 128)
    BLOCK_D = 64            # output‑dim tile (covers dim ≤ 384)

    grid = (total_rows_h,)
    final_proj_kernel[grid](
        out_flat,
        out_gate_flat,
        norm_weight,
        norm_bias,
        to_out_weight_T,
        final_fp16,
        eps,
        total_rows_h,
        hidden_dim,          # constexpr
        dim,                 # constexpr
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5️⃣ Cast back to float32 and restore the original 4‑D shape
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
