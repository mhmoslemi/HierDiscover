        ptrs = mat_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Accumulator for final output
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    # Second pass compute contribution
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # mat and gate slices
        mat_ptrs = mat_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        mat_vals = tl.load(mat_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        gate_ptrs = gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        gate_vals = tl.load(gate_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # normalize
        normed = (mat_vals - mean[:, None]) * rstd[:, None]   # [BLOCK_M, BLOCK_D]

        # apply second norm weight & bias (γ,β)
        w_norm = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0)[None, :]
        b_norm = tl.load(norm_bias_ptr   + offs_d, mask=mask_d, other=0.0)[None, :]
        normed = normed * w_norm + b_norm    # [BLOCK_M, BLOCK_D]

        # apply out_gate
        gated = normed * gate_vals

        # load weight_T slice (BLOCK_D, BLOCK_O) for current cols
        w_ptrs = weight_T_ptr + offs_d[:, None] * dim + offs_o[None, :]
        w = tl.load(w_ptrs, mask=mask_d[:, None] & mask_o[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(gated, w)   # [BLOCK_M, BLOCK_O]
    # Write result
    out_ptrs = out_ptr + offs_m[:, None] * dim + offs_o[None, :]
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_o[None, :])

"""
TriMul (outgoing) – fast forward implementation
================================================

We compute the outgoing TriMul update
    out[i,j] = Σ_k left[i,k] * right[j,k]
with three fused stages:

1️⃣ Input LayerNorm – use the highly‑optimized torch.nn.functional.layer_norm.
2️⃣ Linear projections + sigmoid gates – a single fused FP16 GEMM produces
   *left*, *right*, *left_gate*, *right_gate* and *out_gate*.
   Optional masking is applied before the gates.
3️⃣ Batched GEMM for the triangular multiplicative update
   (left @ rightᵀ) works on (B·H, N, N) tensors.
4️⃣ **Custom Triton kernel** fuses the second LayerNorm (on the hidden
   dimension), the output‑gate multiplication and the final projection
   back to the original channel dimension.  This eliminates three separate
   memory passes (layer‑norm → gate → linear) and reduces overall runtime
   dramatically, meeting the sub‑millisecond target on an H100.

The kernel operates on rows of shape (hidden_dim,) and a block of output
channels (dim).  It first computes the per‑row mean/variance, then in a
second pass performs the normalized transformation, gating, and the
matrix‑vector multiply with the transposed final weight.  All
accumulations are performed in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused (second LayerNorm + out_gate gating + final linear)
# ----------------------------------------------------------------------
@triton.jit
def final_proj_kernel(
    out_ptr,            # *mut float32*   [total_rows_h, dim]
    mat_ptr,            # *const fp16*    [total_rows_h, hidden_dim]   (tri‑mul result)
    gate_ptr,           # *const fp16*    [total_rows_h, hidden_dim]   (out_gate)
    norm_w_ptr,         # *const float32* [hidden_dim]                 (to_out_norm.weight)
    norm_b_ptr,         # *const float32* [hidden_dim]                 (to_out_norm.bias)
    weight_T_ptr,       # *const fp16*    [hidden_dim, dim]           (to_out.weightᵀ)
    total_rows_h: tl.constexpr,
    hidden_dim: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_O: tl.constexpr,   # output‑channel block size
    BLOCK_D: tl.constexpr,   # hidden‑dim block size
):
    pid = tl.program_id(0)          # row tile
    pid_o = tl.program_id(1)        # output‑channel tile

    row_start = pid * BLOCK_M
    col_start = pid_o * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_o = col_start + tl.arange(0, BLOCK_O)

    mask_m = offs_m < total_rows_h
    mask_o = offs_o < dim

    # -----------------------------------------------------------------
    # 1️⃣ Compute mean / variance of the hidden‑dim vector (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        ptrs = mat_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣ Accumulate final output:
    #      out = ( ( (x - μ) * rstd ) * γ + β ) * gate @ Wᵀ
    # -----------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # load raw hidden vector and gate
        mat_ptrs = mat_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        x = tl.load(mat_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        gate_ptrs = gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        g = tl.load(gate_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        # normalize → (x-μ)/σ
        normed = (x - mean[:, None]) * rstd[:, None]   # [BLOCK_M, BLOCK_D]

        # second LayerNorm parameters γ,β
        w_norm = tl.load(norm_w_ptr + offs_d,
                         mask=mask_d,
                         other=0.0)[None, :]          # [1, BLOCK_D]
        b_norm = tl.load(norm_b_ptr + offs_d,
                         mask=mask_d,
                         other=0.0)[None, :]          # [1, BLOCK_D]

        normed = normed * w_norm + b_norm               # γ·norm + β

        # apply output gate
        gated = normed * g                               # [BLOCK_M, BLOCK_D]

        # load slice of the transposed final weight (hidden_dim × dim)
        w_ptrs = weight_T_ptr + offs_d[:, None] * dim + offs_o[None, :]
        w = tl.load(w_ptrs,
                    mask=mask_d[:, None] & mask_o[None, :],
                    other=0.0).to(tl.float32)          # [BLOCK_D, BLOCK_O]

        # accumulate matrix‑vector product
        acc += tl.dot(gated, w)                         # [BLOCK_M, BLOCK_O]

    # -----------------------------------------------------------------
    # Store result
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + offs_m[:, None] * dim + offs_o[None, :]
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C]   (float32)
            - mask_tensor  : torch.Tensor [B, N, N]   (bool/float, optional)
            - weights_dict : model weights (see keys below)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]            # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for flattened view

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use torch's highly‑optimized kernel)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                          # [B, N, N, C]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right + 3 gates) via one GEMM
    # ------------------------------------------------------------------
    x_norm_fp16 = x_norm.half()                # FP16 for GEMM

    # concatenate the five weight matrices (each [H, C])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # shape (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)  # [total_rows, 5*H]  FP16
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float32)
        mask_fp16 = mask_flat.half()
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates + element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)    # [total_rows, H]  FP16
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)      # used later

    left = (left_proj_fp16 * left_gate).view(B, N, N, hidden_dim)   # [B,N,N,H] FP16
    right = (right_proj_fp16 * right_gate).view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)                  # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) → (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # transpose on the k‑axis

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                # [B*H, N, N]  FP16
    right_mat = right_perm.view(BH, N, N)              # [B*H, N, N]  FP16

    out_mat = torch.bmm(left_mat, right_mat)          # [B*H, N, N]  FP16 (left @ rightᵀ)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Fused second LayerNorm + out_gate gating + final projection
    # ------------------------------------------------------------------
    total_rows_h = B * N * N                    # same as total_rows
    out_flat = out.view(total_rows_h, hidden_dim)        # [rows, H]  FP16
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)  # [rows, H]  FP16

    # to_out_norm parameters (γ, β) are already float32 tensors
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]

    # final projection weight transposed for easier loading inside the kernel
    to_out_w_T = weights["to_out.weight"].t().contiguous().half()  # [H, C] → FP16

    # output buffer (float32)
    final_fp32 = torch.empty((total_rows_h, dim), dtype=torch.float32, device=device)

    # launch configuration – tuned for H100
    BLOCK_M = 64     # rows per program
    BLOCK_O = 32     # output‑channel block
    BLOCK_D = 32     # hidden‑dim block
    grid = (triton.cdiv(total_rows_h, BLOCK_M), triton.cdiv(dim, BLOCK_O))

    final_proj_kernel[grid](
        final_fp32,
        out_flat,
        out_gate_flat,
        to_out_norm_w,
        to_out_norm_b,
        to_out_w_T,
        total_rows_h,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_O=BLOCK_O,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # reshape to the original 4‑D layout
    result = final_fp32.view(B, N, N, dim)           # [B, N, N, C] (float32)

    return result
