def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # 1. First LayerNorm (use torch.nn.functional.layer_norm)
    # shape: [B,N,N,dim]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Convert to FP16 for linear
    total_rows = x_norm.shape[0] * x_norm.shape[1] * x_norm.shape[2]
    # flatten
    x_norm_flat = x_norm.view(-1, dim).to(torch.float16)

    # 2. Fuse linear projections
    weight_concat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape [5*hidden, dim]

    proj = F.linear(x_norm_flat, weight_concat)  # shape [R, 5*hidden]
    # split
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(proj, hidden_dim, dim=1)

    # mask
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # gates
    left = left_proj_raw * torch.sigmoid(left_gate_raw)
    right = right_proj_raw * torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)  # shape [R, hidden]

    # 3. Triangular multiplicative update
    B, N, _, _ = input_tensor.shape
    # reshape
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    # flatten batch and hidden dims
    BH = B * hidden_dim
    left_mat = left.view(BH, N, N)
    right_mat = right.view(BH, N, N)

    # batched matmul: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [BH,N,N]

    # reshape back to [B,N,N,H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # 4. Second LayerNorm + output gate (fused Triton kernel)
    out_flat = out.view(-1, hidden_dim)
    out_gate_flat = out_gate.view(-1, hidden_dim)
    out_norm_gated = torch.empty_like(out_flat)  # FP16

    # Determine block sizes
    BLOCK_M = 256  # number of rows per program
    BLOCK_D = 32 if hidden_dim <= 128 else 64  # columns per block

    grid = (triton.cdiv(out_flat.shape[0], BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_flat.shape[0],
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # 5. Final projection
    final_fp16 = F.linear(out_norm_gated, weights["to_out.weight"].to(torch.float16))
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final

"""
custom_kernel – highly‑optimized “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣ LayerNorm + linear projection are fused:
   * Compute per‑row mean/variance of the input (float32).
   * Apply γ/β (LayerNorm parameters) and fuse the β‑bias into the
     linear weight matrix → a single FP16 GEMM (weight_concat).
   * This removes the separate LayerNorm write‑back and the extra
     memory traffic of the original implementation.

2️⃣ Split the fused GEMM output into the five branches
   (left_proj, left_gate, right_proj, right_gate, out_gate),
   apply the optional mask, and fuse each projection with its sigmoid‑gate.

3️⃣ Triangular multiplicative update (the expensive O(N³) step):
   * Reshape left/right to (B, H, N, N) and use a single batched GEMM:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
     This is implemented as a batched matrix multiply
       left_mat @ right_matᵀ  (torch.bmm)  where
       left_mat  = left.reshape(B·H, N, N)
       right_mat = right.reshape(B·H, N, N).
   * Only one copy of the right tensor is performed (the necessary
     i↔j swap).

4️⃣ Second LayerNorm + output‑gate are fused in a custom Triton kernel
   (layernorm_gate_kernel).  The kernel computes mean/variance in FP32,
   applies the affine transform, multiplies by the pre‑computed sigmoid‑gate
   and writes the result back in FP16.

5️⃣ Final FP16 linear projection back to `dim` and cast to FP32.

All heavy linear algebra (steps 1, 3, 5) uses cuBLAS/Tensor‑Core kernels,
while the two LayerNorm kernels run on Triton.  The result is a
sub‑millisecond forward pass for the supported configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton fused LayerNorm + element‑wise gating (used for the second norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*  [total_rows, dim]   (input to LN)
    gate_ptr,       # const half*  [total_rows, dim]   (sigmoid‑gate)
    y_ptr,          #       half*  [total_rows, dim]   (output)
    w_ptr,          # const float* [dim]              (γ)
    b_ptr,          # const float* [dim]              (β)
    total_rows,     # int
    dim: tl.constexpr,          # static (here = hidden_dim)
    eps: tl.constexpr,          # static
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute mean / variance (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, affine transform, multiply gate and store (FP16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (already sigmoid‑ed)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out = out * g_vals                                 # apply output‑gate

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]          # input / output channel dimension
    hidden_dim  = config["hidden_dim"]   # hidden dimension used inside TriMul
    eps         = 1e-5
    nomask      = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape                # [B, N, N, dim]
    total_rows  = B * N * N                         # rows after flattening
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Fuse LayerNorm + all linear projections
    # ------------------------------------------------------------------
    # --- statistics for LayerNorm (float32) ---------------------------------
    # mean over last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)               # [B,N,N,1]
    # variance = E[x^2] - mean^2
    var = (input_tensor * input_tensor).mean(dim=-1, keepdim=True) - mean * mean
    rstd = torch.rsqrt(var + eps)                                # [B,N,N,1]

    # --- LayerNorm affine parameters ----------------------------------------
    gamma = weights["norm.weight"]               # [dim], float32
    beta  = weights["norm.bias"]                # [dim], float32

    # --- pre‑compute bias that compensates for the missing β in the linear ----
    # concatenate the five weight matrices (shape: [5*hidden_dim, dim])
    weight_concat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)                     # [(5*H), dim]

    # bias = β · Wᵀ  (shape: [5*hidden_dim])
    bias_out = torch.matmul(beta, weight_concat.t()).to(torch.float16)  # fp16

    # --- apply LayerNorm (without the β‑bias) and fuse into GEMM -------------
    # x_centered = (x - mean) * rstd           # float32
    x_centered = (input_tensor - mean) * rstd
    # scale by γ  (broadcast over the last dimension)
    x_scaled = x_centered * gamma                # float32
    # cast to FP16 for the GEMM
    x_scaled_h = x_scaled.to(torch.float16)

    # flatten to [R, dim] where R = B·N·N
    x_scaled_flat = x_scaled_h.view(-1, dim)     # [total_rows, dim]

    # fused GEMM:  (x_scaled_h) @ weight_concatᵀ   ->  [R, 5*hidden_dim]
    proj = F.linear(x_scaled_flat, weight_concat)   # FP16

    # add the pre‑computed β‑bias (broadcast over rows)
    proj = proj + bias_out

    # ------------------------------------------------------------------
    # Split into logical branches
    # ------------------------------------------------------------------
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )  # each [R, hidden]

    # ------------------------------------------------------------------
    # Apply mask (if present) – only to the projections
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # Apply sigmoid‑gates (still FP16)
    # ------------------------------------------------------------------
    left  = left_proj_raw  * torch.sigmoid(left_gate_raw)   # [R, hidden]
    right = right_proj_raw * torch.sigmoid(right_gate_raw) # [R, hidden]
    out_gate = torch.sigmoid(out_gate_raw)                 # [R, hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into the batch dimension
    left  = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i‑k)
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] (i‑k)

    BH = B * hidden_dim
    left_mat  = left.view(BH, N, N)                     # [BH, N, N]  (i, k)
    right_mat = right.view(BH, N, N)                    # [BH, N, N]  (i, k)

    # out = left @ rightᵀ  →  sum over k  (i, j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim)               # [R, hidden]
    out_gate_flat = out_gate.view(-1, hidden_dim)     # [R, hidden]
    out_norm_gated_flat = torch.empty_like(out_flat)  # FP16 destination

    # Choose block sizes that work well for hidden_dim up to 384
    BLOCK_M = 256
    BLOCK_D = 64 if hidden_dim > 128 else 32

    grid = (triton.cdiv(out_flat.shape[0], BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gated_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_flat.shape[0],
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gated_flat,
        weights["to_out.weight"].to(torch.float16)  # [dim, hidden]
    )   # [R, dim]   FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
