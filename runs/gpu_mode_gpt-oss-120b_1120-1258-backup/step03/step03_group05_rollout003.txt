"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  First LayerNorm – use the highly‑optimized cuDNN implementation via
    `torch.nn.functional.layer_norm`.  This removes the custom Triton launch
    for the biggest tensor and is already near‑peak performance.

2️⃣  Fuse the five linear projections (left/right projection, three gates) into
    a **single** half‑precision GEMM:
        Y = X_norm @ W_fused.T            #  (R, 5·H)   ,  R = B·N·N
    where `W_fused` stacks the five weight matrices.  The result is split
    into the logical branches and the optional pairwise mask is applied only
    to the two projection branches.

3️⃣  Apply the sigmoid gates (still in FP16) and reshape to
    `[B, N, N, H]`.

4️⃣  Triangular multiplicative update – a batched GEMM over the hidden
    dimension.  No extra copies are performed:
        left  = left.permute(0, 3, 1, 2)   # (B, H, N, N)
        right = right.permute(0, 3, 2, 1)  # (B, H, N, N)
        out   = torch.matmul(left, right) # (B, H, N, N)
        out   = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

5️⃣  Second LayerNorm + output‑gate – fused in a single Triton kernel
    (`layernorm_gate_kernel`).  The kernel works in FP32 for the reduction
    (mean / variance) and writes the gated, normalized result in FP16.

6️⃣  Final linear projection back to `dim` (FP16 GEMM) and cast to FP32.

Only the second LayerNorm (with the output‑gate) is implemented in Triton,
fulfilling the “must use Triton” requirement while keeping the overall
runtime well below the 1 ms target for the supported configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm (float32) + element‑wise gate (output‑gate)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*   [total_rows, dim]   (input to LN)
    gate_ptr,       # const half*   [total_rows, dim]   (sigmoid‑gate)
    y_ptr,          #       half*   [total_rows, dim]   (output)
    w_ptr,          # const float*  [dim]               (γ)
    b_ptr,          # const float*  [dim]               (β)
    total_rows: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute mean / variance (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  Normalize, apply affine (γ/β), multiply already‑sigmoid‑ed gate
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load normalized input
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (already sigmoid‑ed)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]          # LN + affine (FP32)
        out = out * g_vals                            # apply output‑gate

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel wrapper for the first (optional) LayerNorm – not needed
# ----------------------------------------------------------------------
# (We use torch.nn.functional.layer_norm instead, see `custom_kernel`.)

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data :
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [bs, seq_len, seq_len] (float32 or bool)
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict  : dict with keys
            * "dim"        – input / output channel dimension
            * "hidden_dim" – hidden dimension inside TriMul
            * "nomask"     – if True the mask is ignored (default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack and basic constants
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # e.g. 128 or 384
    hidden_dim = config["hidden_dim"]   # e.g. 128
    eps = 1e-5
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                 # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First layer‑norm – use cuDNN implementation (fast, no Triton needed)
    # ------------------------------------------------------------------
    # Input is [B, N, N, dim] -> reshape to [R, dim] for layer_norm.
    x_flat = input_tensor.view(total_rows, dim)
    x_norm_flat = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Back to 4‑D tensor (still FP32)
    x_norm = x_norm_flat.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all projections + gates into a single FP16 GEMM
    # ------------------------------------------------------------------
    # Cast to FP16 once (cuBLAS Tensor‑Core kernels love FP16)
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # Stack the five weight matrices (shape: (5·H, dim))
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)           # [(5·H), dim]

    # Single linear projection (no bias)
    proj = F.linear(x_norm_h, weight_concat)   # (R, 5·H)

    # Split into logical branches
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )   # each (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) – only to the two projection branches
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> flatten -> (R, 1) -> FP16 broadcast
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj_raw * torch.sigmoid(left_gate_raw)   # (R, H)
    right = right_proj_raw * torch.sigmoid(right_gate_raw)# (R, H)
    out_gate = torch.sigmoid(out_gate_raw)               # (R, H)

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)    # (B, N, N, H)
    right = right.view(B, N, N, hidden_dim)

    # Transpose to obtain batch‑wise matrix layout without extra copies
    left_perm = left.permute(0, 3, 1, 2)      # (B, H, N, N) – non‑contiguous
    right_perm = right.permute(0, 3, 2, 1)    # (B, H, N, N) – non‑contiguous (j↔k)

    # Batched GEMM: each (N,N) pair multiplies over the k‑dimension
    out = torch.matmul(left_perm, right_perm)  # (B, H, N, N)

    # Back to the original layout required by the next stage
    out = out.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # (R, H)
    out_gate_flat = out_gate.view(total_rows, hidden_dim)  # (R, H) – already FP16
    out_norm_gated = torch.empty_like(out_flat)          # FP16 destination

    # Triton launch configuration – tuned for H ≤ 128
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim` (FP16 GEMM) → FP32 output
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gated,
        weights["to_out.weight"].to(torch.float16),  # (dim, H)
    )   # (R, dim) – FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
