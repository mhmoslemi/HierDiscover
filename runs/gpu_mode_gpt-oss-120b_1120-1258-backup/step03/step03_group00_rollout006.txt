"""
Optimized Triton implementation of the “outgoing” TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm over the input channel dim (C).  A custom Triton kernel
    reads each row once (fp32 accumulation, fp16 output) and therefore
    halves the memory traffic compared to the two‑pass implementation.

2️⃣  Fuse the five linear projections (left/right & three gates) into a
    single half‑precision GEMM by concatenating their weight matrices.

3️⃣  Optional pairwise mask (broadcast over the hidden dimension).

4️⃣  Apply sigmoid gates and element‑wise gating (all in fp16).

5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    realized as a batched fp16 matrix‑multiply
    `torch.bmm` on tensors of shape (B·H, N, N).

6️⃣  Second Layer‑norm over the hidden dimension (H) – same Triton kernel.

7️⃣  Multiply by the output gate and project back to the original channel
    dimension `dim` with a final half‑precision linear layer.  The result
    is cast back to fp32 as required by the specification.
"""

import torch.nn.functional as F
import triton

    bias_ptr,       # *const float*   [dim]            (β)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns (= dim) processed per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    offs_d = tl.arange(0, BLOCK_D)                      # [BLOCK_D]
    mask_d = offs_d < dim

    # -----------------------------------------------------------------
    # Load one full row (or a masked sub‑row) and keep it in registers
    # -----------------------------------------------------------------
    x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
    vals = tl.load(
        x_ptrs,
        mask=mask_m[:, None] & mask_d[None, :],
        other=0.0,
    ).to(tl.float32)                                    # [BLOCK_M, BLOCK_D]

    # -----------------------------------------------------------------
    # Row‑wise mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.sum(vals, axis=1)                       # [BLOCK_M]
    sum_sq  = tl.sum(vals * vals, axis=1)                # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Normalization, scale & bias (fp32 → fp16)
    # -----------------------------------------------------------------
    norm = (vals - mean[:, None]) * rstd[:, None]       # [BLOCK_M, BLOCK_D]

    w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # [BLOCK_D]
    b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # [BLOCK_D]

    out_fp32 = norm * w + b                              # fp32
    out_fp16 = out_fp32.to(tl.float16)                  # fp16

    # -----------------------------------------------------------------
    # Store result
    # -----------------------------------------------------------------
    y_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
    tl.store(y_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/int/float)
            - weights_dict: dict of FP32 model weights
            - config_dict : {
                  "dim": C,
                  "hidden_dim": H,
                  "nomask": bool (optional, default True)
              }

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Shapes
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape               # [B, N, N, C]
    total_rows = B * N * N                         # rows for flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output, single‑pass Triton kernel)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 64          # rows per program (tunable)
    BLOCK_D = dim         # compile‑time column size = dim (max 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_onepass_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection + gates (all FP16)
    # --------------------------------------------------------------
    # concatenate the five projection weight matrices (shape: (5*H, C))
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()          # (5*H, C)

    # Linear: [total_rows, C] @ [C, 5*H]^T  → [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split back into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: [total_rows, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied before the gates)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten → [total_rows, 1]
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still FP16) and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)       # [total_rows, H]
    right_gate = torch.sigmoid(right_gate)     # [total_rows, H]
    out_gate = torch.sigmoid(out_gate)         # [total_rows, H]

    left = left_proj * left_gate                # [total_rows, H]
    right = right_proj * right_gate             # [total_rows, H]

    # Reshape to 4‑D tensors for the triangular update
    left = left.view(B, N, N, hidden_dim)       # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)    # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Transpose to bring the hidden dimension into the batch axis:
    # left  : (B, H, N, N)  where rows = i, cols = k
    # right : (B, H, N, N)  where rows = k, cols = j
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)    # [B*H, N, N]

    # Batched matrix multiply in FP16 (Tensor‑Core friendly)
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] fp16

    # Back to 4‑D tensor: [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                          # fp16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16, single‑pass)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    BLOCK_M2 = 64
    BLOCK_D2 = hidden_dim          # typically 128

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M2),)
    layernorm_fp16_onepass_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # fp16

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate          # [B, N, N, H] fp16

    # Final linear layer: H → C
    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # fp16 → shape [total_rows_h, C]

    # Cast back to FP32 as required by the spec
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
