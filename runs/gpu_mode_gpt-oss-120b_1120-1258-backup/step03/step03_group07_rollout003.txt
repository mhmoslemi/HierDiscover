"""
Highly‑optimized Triton‑augmented implementation of the **outgoing**
TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension (C).  The
    kernel reads the fp32 input once to compute mean/variance and a
    second time to write the normalized values directly as fp16.

2️⃣ Five linear projections (left, right and three gates) are fused
    into a single half‑precision GEMM (torch.nn.functional.linear).

3️⃣ Optional pair‑wise mask is applied to the *left* and *right*
    projections before gating.

4️⃣ Sigmoid is applied to the three gate tensors and element‑wise
    gating of the left/right projections is performed.

5️⃣ Triangular multiplicative update:
       out_h[i, j] = Σₖ left_h[i, k] * right_h[j, k]
   Implemented as a single batched matrix‑multiply
   (torch.bmm) on tensors of shape (B·H, N, N) where H = hidden_dim.

6️⃣ A second fused kernel performs LayerNorm over the hidden dimension,
    multiplies by the output‑gate and writes the result as fp16.

7️⃣ Final linear projection back to the original channel dimension
    (fp16 GEMM) and cast to float32.

Only the two LayerNorm steps are performed in custom Triton kernels;
all dense linear algebra uses PyTorch’s Tensor‑Core‑accelerated
operations.  The implementation works for any batch size (B≤2),
sequence length (N up to 1024) and supports both masked and unmasked
inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels (LayerNorm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                # const  *float   [total_rows, dim] (fp32 input)
    y_ptr,                # mut    *half     [total_rows, dim] (fp16 output)
    weight_ptr,           # const  *float   [dim]            (γ)
    bias_ptr,             # const  *float   [dim]            (β)
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm writing fp16 output."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- 1st pass: mean & variance (float32 accumulator) ----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                             # 1 / sqrt(var+eps)

    # ---- 2nd pass: write normalized values ----------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                      # fp32
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]

        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,                # const  *half   [total_rows, dim] (pre‑norm)
    gate_ptr,             # const  *half   [total_rows, dim] (output‑gate)
    y_ptr,                # mut    *half   [total_rows, dim] (LN·γ·β·gate)
    weight_ptr,           # const  *float  [dim]            (γ of 2nd LN)
    bias_ptr,             # const  *float  [dim]            (β of 2nd LN)
    total_rows,           # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """LayerNorm + output‑gate (γ·x̂+β)·gate → fp16."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- mean & variance -----------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized *gate ----------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)            # pre‑norm value
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)            # output‑gate

        norm = (x - mean[:, None]) * rstd[:, None]          # x̂

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g           # (γ·x̂+β)·gate
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom TriMul kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    B, N, _, C = input_tensor.shape
    assert C == dim, "config dim does not match input last dimension"

    total_rows = B * N * N                     # rows for the first LayerNorm

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (over C) → fp16
    # --------------------------------------------------------------
    # Flatten to [total_rows, C] (contiguous)
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    BLOCK_M = 256      # rows per program
    BLOCK_D = 32       # columns per iteration (must divide dim)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections (left/right + three gates)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices: (5*H, C)
    w_left_proj  = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate  = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate   = weights["out_gate.weight"]

    concat_W = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    ).half().contiguous()                     # shape (5*H, C)

    # Single half‑precision GEMM: [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]
    proj_all_fp16 = F.linear(x_norm_fp16, concat_W)   # fp16 output

    # Split projections back into their components
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(proj_all_fp16,
                                                    hidden_dim, dim=1)

    # --------------------------------------------------------------
    # 2a️⃣ Optional mask application (on left/right only)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [total_rows, 1]
        mask_flat = mask_tensor.reshape(total_rows, 1).to(dtype=torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b️⃣ Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)           # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)            # later used after 2nd LN

    left_fp16  = left_proj_fp16 * left_gate             # gated left
    right_fp16 = right_proj_fp16 * right_gate           # gated right

    # Reshape to [B, N, N, H] for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)       # saved for step 4

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σₖ left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Rearrange to (B*H, N, N) where the k‑dimension becomes the inner‑axis
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (B·H, N, N)

    # Batched matrix multiply (fp16, Tensor‑Core enabled)
    out_mat = torch.bmm(left_mat, right_mat)      # (B·H, N, N), fp16

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Fuse second LayerNorm, output‑gate, and final projection
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)          # [total_rows_h, H]
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)

    # Buffer for the result of LN·gate (fp16)
    fused_fp16 = torch.empty_like(out_flat)                # fp16

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_gate_flat,
        fused_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 5️⃣ Final linear projection back to the original channel dimension
    # --------------------------------------------------------------
    to_out_weight_fp16 = weights["to_out.weight"].half()   # (C, H)
    final_fp16 = F.linear(fused_fp16, to_out_weight_fp16)  # fp16 → [total_rows_h, C]

    # Cast to float32 and restore tensor shape [B, N, N, C]
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
