"""
Fast outgoing TriMul forward.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (`dim` channels) – custom Triton kernel.
   The kernel works in fp32 for the reduction, writes the normalized values
   directly in fp16 (the rest of the pipeline stays in fp16).

2️⃣ Fuse the five dense layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into one half‑precision GEMM:
   `[R, dim] @ [5*hidden, dim].T → [R, 5*hidden]`.  Simple slicing gives the
   three gate vectors and the two projected tensors.

3️⃣ Apply an optional mask (only on the two projected tensors) and the three
   sigmoid gates.

4️⃣ Triangular multiplicative update.
   For each hidden channel `d` we need
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched matrix‑multiply:
       out_d = left_d @ right_dᵀ.
   By reshaping to `[B*hidden, N, N]` we can use a single
   `torch.bmm` (cuBLAS/Tensor‑Core) which is far faster than the
   `einsum` in the reference implementation.

5️⃣ Second LayerNorm over the hidden dimension (`hidden_dim`), again with the
   Triton kernel.

6️⃣ Multiply by the output gate and a final half‑precision linear projection
   back to `dim`.  The result is cast to `float32` before returning.

All heavy linear work stays in fp16 to exploit H100 Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (input can be fp32 or fp16, output is fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *   [total_rows, dim]   (fp32 or fp16)
    y_ptr,            # mutable * [total_rows, dim]   (fp16 output)
    weight_ptr,       # const *   [dim]               (fp32)
    bias_ptr,         # const *   [dim]               (fp32)
    total_rows,       # int (runtime)
    dim: tl.constexpr,    # compile‑time constant
    eps: tl.constexpr,    # compile‑time constant
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass: sum and sum‑of‑squares (accumulate in fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2nd pass: write normalized (fp16) = (x-mean)*rstd*γ + β
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – output in fp16
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 1024   # larger block to reduce launch overhead
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse five linear layers (proj + 3 gates) into a single GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices: [5*hidden, dim]   (biases are all zero)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Linear transform in fp16
    out_concat = F.linear(x_norm_fp16, w_concat)                # [R, 5*H]

    H = hidden_dim
    left_proj   = out_concat[:, 0:H]                # [R, H]
    right_proj  = out_concat[:, H:2*H]
    left_gate_p = out_concat[:, 2*H:3*H]
    right_gate_p= out_concat[:, 3*H:4*H]
    out_gate_p  = out_concat[:, 4*H:5*H]

    # Sigmoid gates (fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # Optional mask (applied only to the projected tensors)
    if not nomask:
        # mask shape [B, N, N] → [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Element‑wise gating
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # Reshape back to 4‑D tensors [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 3) Triangular multiplicative update using batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Move hidden dimension to the second axis and make tensors contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Merge batch & hidden: (B*H, N, N)
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matmul: left_mat @ right_matᵀ  → [B*H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # fp16

    # --------------------------------------------------------------
    # 4) Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.contiguous().view(total_rows, H)      # [R, H]
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)            # fp16

    # --------------------------------------------------------------
    # 5) Apply output gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                         # element‑wise fp16

    out_norm_flat = out_norm.view(total_rows, H)           # [R, H]
    final_half = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16)        # [dim, H]
    )                                                     # [R, dim] fp16

    # Cast to float32 for the final result
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
