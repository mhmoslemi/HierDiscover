"""
Optimized forward kernel for the “outgoing” TriMul operation (AlphaFold‑3).

Algorithm
---------
1. **Input LayerNorm** – row‑wise (over the channel dimension `dim`) directly
   writes FP16 results with a custom Triton kernel (no extra casts).

2. **Fused linear projections & gates** – the five weight matrices
   (`left_proj`, `right_proj`, three gates) are concatenated and applied in a
   single half‑precision GEMM (Torch FP16 linear).  The result is split into the
   five tensors.

3. **Mask** (if present) and **sigmoid gates** are applied element‑wise in‑place.

4. **Triangular multiplicative update** – for each hidden channel `h` we need
   ``out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]``.
   This is exactly a batched matrix‑multiply `L_h @ R_hᵀ`.  We reshape the
   tensors to `(B·H, N, N)` and use a single `torch.bmm` (Tensor‑Core).

5. **Second LayerNorm + output gate** – fused into another Triton kernel that
   normalises, applies the learned scale/bias and finally multiplies by the
   pre‑computed `out_gate`.

6. **Final linear projection** – half‑precision GEMM back to the original
   dimension `dim`, followed by a cast to FP32 for the returned tensor.

All heavy arithmetic runs in FP16 (Tensor‑Core) and the two LayerNorm steps are
implemented in Triton to avoid extra memory traffic.  The implementation is
fully stateless, works for any `dim`/`hidden_dim` (including 128/384) and
supports optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm that directly outputs FP16
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float *  [total_rows, dim]   (FP32 input)
    y_ptr,          # mut   half  *  [total_rows, dim]   (FP16 output)
    w_ptr,          # const float *  [dim]               (γ)
    b_ptr,          # const float *  [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- first pass: mean & variance (float32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalize, apply γ/β, store as FP16 ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # (x-µ)/σ
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                  # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) Fused LayerNorm + output‑gate (both FP16 → FP16)
# ----------------------------------------------------------------------
@triton.jit

    y_ptr,          # mut   half *  [total_rows, dim]   (post‑norm⋅gate)

    b_ptr,          # const float * [dim]               (β)
    gate_ptr,       # const half *  [total_rows, dim]   (out_gate, already sigmoid)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / variance (float32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, γ/β, multiply gate ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                  # FP32

        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2) Fused projection + gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # concatenate five weight matrices (shape (5*hidden, dim))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                         # (5H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)    # (R, 5H) FP16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden, dim=1
        )                                         # each (R, H)

    # ------------------------------------------------------------------
    # 2a) optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B,N,N] → flatten to (R,1) and cast to half
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)            # (R, H) FP16

    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    #    out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden)          # (B,N,N,H)
    right = right_fp16.view(B, N, N, hidden)       # (B,N,N,H)

    # reshape to (B*H, N, N) for batched matrix multiplication
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H,N,N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H,N,N)

    out_mat = torch.bmm(left_mat, right_mat)       # (B*H,N,N) FP16

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4) Second LayerNorm + output gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat)        # (R, H) FP16

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                         # already sigmoid‑ed, FP16
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5) Final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()          # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16) # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # cast to FP32

    return final
```
