import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Highly‑optimized “outgoing” TriMul implementation (AlphaFold‑3).

Algorithm
---------
1. Row‑wise LayerNorm on the last channel (C).  Implemented with a custom
   Triton kernel that uses fp32 accumulation (fast & numerically stable).
2. Fuse the five linear projections (left/right and three gates) into a
   single fp16 GEMM by concatenating the weight matrices.
3. (Optional) mask the left/right projections before gating.
4. Apply sigmoid gates and element‑wise gating → tensors of shape
   [B, N, N, H] (H = hidden_dim).
5. Triangular multiplicative update:
      out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   realized as a batched GEMM `left @ rightᵀ` on tensors
   (B·H, N, N) in fp16.
6. Row‑wise LayerNorm on the hidden dimension (H) – same Triton kernel.
7. Multiply by the output gate, then a final fp16 linear projection back to
   the original channel dimension C.
8. Cast the result to fp32 and return.

Only the two LayerNorms are written in Triton; all heavy dense linear algebra
uses cuBLAS/Tensor‑Core fp16 GEMMs.  The implementation keeps memory traffic
low (in‑place where possible) and removes unnecessary host‑side
synchronisations to achieve sub‑millisecond runtimes on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16/fp32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const   float*   [total_rows, dim]
    y_ptr,          # *mut     float*   [total_rows, dim]
    weight_ptr,     # *const   float*   [dim]
    bias_ptr,       # *const   float*   [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣ compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the channel dimension in chunks of BLOCK_D
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # 2‑D pointer: [row, col]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣ write normalized output
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: mapping from layer names to weight tensors
            - config_dict : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        Tensor of shape [B, N, N, C] (float32, on the same CUDA device)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Prepare shapes
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N                     # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over the channel dimension C
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)           # [total_rows, C]
    x_norm_flat = torch.empty_like(x_flat)     # fp32 output buffer

    # Tuned block sizes: larger row block for better occupancy,
    # column block chosen to keep register pressure modest.
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No explicit sync – subsequent ops create an implicit dependency.

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + 3 gates) into a
    #     single fp16 GEMM.
    # ------------------------------------------------------------------
    # Convert LayerNorm output to half‑precision for the GEMM.
    x_norm_fp16 = x_norm_flat.half()           # [total_rows, C]

    # Concatenate weight matrices (each of shape [H, C]).
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]

    concat_w = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).half().contiguous()                      # shape (5*H, C)

    # Single fused linear: [total_rows, C] @ [C, 5*H]^T -> [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_w)  # fp16

    # Split the fused projection back into the five tensors.
    splits = torch.split(proj_all, hidden_dim, dim=1)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = splits

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [total_rows, 1] (float32 -> half)
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float32)
        mask_flat_fp16 = mask_flat.half()
        left_proj_fp16  = left_proj_fp16 * mask_flat_fp16
        right_proj_fp16 = right_proj_fp16 * mask_flat_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)   # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate   = torch.sigmoid(out_gate_fp16)

    left_fp16  = left_proj_fp16 * left_gate      # [total_rows, H]
    right_fp16 = right_proj_fp16 * right_gate    # [total_rows, H]

    # Reshape to the original 4‑D layout.
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #     out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) -> (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose on the k‑axis

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)  # [B*H, N, N]

    # Batched matrix‑multiply (fp16, Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat)  # [B*H, N, N] (fp16)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)       # [total_rows_h, H]
    out_norm_flat = torch.empty_like(out_flat)          # fp16 output buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # Implicit dependency ensures out_norm_flat is ready.

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate               # element‑wise, fp16

    # Linear: hidden_dim -> dim
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)    # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)    # [total_rows_h, C]

    # Cast back to fp32 for the final output.
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
