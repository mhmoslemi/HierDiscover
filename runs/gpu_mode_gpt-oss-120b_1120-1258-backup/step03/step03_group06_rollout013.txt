"""
Highly‑optimized forward kernel for the “outgoing” TriMul operation (AlphaFold‑3).

Algorithm
---------
1. Fast LayerNorm on the input tensor using torch.nn.functional.layer_norm
   (FP32 arithmetic → cast to FP16).  This is much faster than a hand‑written
   Triton kernel for the huge number of rows (B·N·N).

2. Fuse the five linear projections (left‑/right‑proj and three gates) into a
   single half‑precision GEMM by concatenating the weight matrices.
   The result is split into the individual tensors.

3. Optional binary mask is applied directly on the left/right projections.

4. Sigmoid gates (still FP16) are applied element‑wise.

5. Triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (B·H independent N×N matrix products) performed
   with torch.bmm on FP16 tensors (Tensor‑Core).

6. Second LayerNorm + output‑gate (γ,β) are fused into a single Triton kernel.
   The kernel computes mean/variance in FP32, normalises, applies the learned
   γ/β, multiplies by the pre‑computed out‑gate and writes the result as FP16.

7. Final linear projection (hidden → dim) is a standard FP16 GEMM with
   torch.nn.functional.linear.  The result is cast back to FP32 for the API.

Only the forward pass is required; no gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 2) Fused LayerNorm + output‑gate (all in FP16)
# --------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,                     # *const half* [total_rows, dim]
    y_ptr,                     # *mut   half* [total_rows, dim]
    weight_ptr,                # *const float* [dim]   (γ)
    bias_ptr,                  # *const float* [dim]   (β)
    gate_ptr,                  # *const half*  [total_rows, dim] (out_gate, already sigmoid‑ed)
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_D: tl.constexpr,     # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # First pass: mean & variance (FP32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)           # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass: normalise → γ,β → gate → store as FP16
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                # FP32

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32
        out = out * gate

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Entry‑point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (float32 or bool)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : dict containing:
            * dim (int)           – input channel dimension C
            * hidden_dim (int)    – projection dimension H
            * nomask (bool)       – True → no mask
            * eps (float)         – LayerNorm epsilon (optional)

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim   = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N           # rows for the flattened tensors (B·N·N)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 → cast to FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is heavily optimised on GPU.
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B,N,N,dim]   FP16

    # ------------------------------------------------------------------
    # 2) Fused linear projections + gates (single GEMM, FP16)
    #    weight matrix shape: (5*hidden, dim)
    # ------------------------------------------------------------------
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # flatten to (B·N·N, dim) for the linear
    x_flat = x_norm.view(total_rows, dim)
    proj_all = F.linear(x_flat, concat_w)     # (B·N·N, 5*H)  FP16

    # split into the five tensors of shape (B·N·N, H)
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden, dim=1
    )

    # ------------------------------------------------------------------
    # 3) Optional mask (apply before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (B·N·N, 1)
        mask_flat = mask.view(total_rows, 1).to(x_norm.dtype)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 4) Sigmoid gates & element‑wise gating (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)               # (B·N·N, H)

    left  = left_proj * left_gate                     # (B·N·N, H)
    right = right_proj * right_gate                   # (B·N·N, H)

    # ------------------------------------------------------------------
    # 5) Triangular multiplicative update (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    → for each head h we compute a matrix product left @ rightᵀ.
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_mat  = left.view(B, N, N, hidden).permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.view(B, N, N, hidden).permute(0, 3, 2, 1).reshape(B * hidden, N, N)

    # batched matmul: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)  FP16

    # restore the original 4‑D layout (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 6) Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()           # (R, H)  FP16
    out_gate_flat = out_gate.contiguous()                         # already (R, H) FP16
    out_norm_gate = torch.empty_like(out_flat)                    # (R, H) FP16 output

    BLOCK_M = 256          # rows per program (tuned for H100)
    BLOCK_D = 64           # cols per iteration (covers hidden dim efficiently)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],     # γ  (float32)
        weights["to_out_norm.bias"],       # β  (float32)
        out_gate_flat,
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()    # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate, to_out_w)  # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
