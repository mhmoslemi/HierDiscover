@triton.jit
def final_ln_linear_kernel(
    x_ptr,          # *const half*  [total_rows, hidden_dim]
    gate_ptr,       # *const half*  [total_rows, hidden_dim]
    weight_ptr,     # *const half*  [dim, hidden_dim]
    ln_weight_ptr,  # *const float* [hidden_dim]
    ln_bias_ptr,    # *const float* [hidden_dim]
    out_ptr,        # *mut float*   [total_rows, dim]
    total_rows,     # int
    hidden_dim: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim
        vals = tl.load(x_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)
    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Process output columns in blocks
    for o in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim

        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < hidden_dim

            # load x and gate
            x = tl.load(x_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)
            g = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

            # LayerNorm: (x - mean) * rstd
            norm = (x - mean[:, None]) * rstd[:, None]

            # Apply LN affine gamma & beta
            gamma = tl.load(ln_weight_ptr + offs_k, mask=mask_k, other=0.0)
            beta = tl.load(ln_bias_ptr + offs_k, mask=mask_k, other=0.0)
            norm = norm * gamma[None, :] + beta[None, :]

            # Apply gate
            gated = norm * g

            # Load weight block transposed (hidden_dim -> dim)
            w_T = tl.load(weight_ptr + offs_o[None, :] * hidden_dim + offs_k[:, None],
                          mask=mask_o[None, :] & mask_k[:, None],
                          other=0.0).to(tl.float32)
            # w_T shape [BLOCK_K, BLOCK_O]

            acc += tl.dot(gated, w_T)   # (BLOCK_M, BLOCK_K) * (BLOCK_K, BLOCK_O) -> (BLOCK_M, BLOCK_O)

        # Write out
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_o[None, :]
        tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_o[None, :])

"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The kernel sequence is:

1️⃣ Row‑wise LayerNorm over the input channel dimension (C).  This is a
   custom Triton kernel that writes the normalized tensor directly in fp16.

2️⃣ A single cuBLAS GEMM computes all five linear projections (left/right
   and three gates) from the normalized tensor.  The result is split into
   the individual tensors.

3️⃣ Apply the optional pair‑wise mask, sigmoid gates and element‑wise gating.

4️⃣ Triangular multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
   Realised as a batched half‑precision matrix‑multiply.

5️⃣ Fuse the second LayerNorm (over the hidden dimension H), the output‑gate
   multiplication and the final projection back to C into one Triton kernel.
   This eliminates the large intermediate tensor of the second LayerNorm and
   reduces memory traffic dramatically.

All heavy reductions run in custom Triton kernels, dense linear algebra uses
cuBLAS/Tensor‑Core GEMMs via ``torch.nn.functional.linear`` and ``torch.bmm``.
The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 stats → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # *const float*   [total_rows, dim]   (fp32 or fp16 input)
    y_ptr,            # *mut   half*    [total_rows, dim]  (fp16 output)
    weight_ptr,       # *const float*   [dim]               (γ)
    bias_ptr,         # *const float*   [dim]               (β)
    total_rows,       # int (runtime)
    dim: tl.constexpr,      # int (compile‑time)
    eps: tl.constexpr,      # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # First pass: compute mean and variance (float32)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # Second pass: write normalized values (γ·x̂+β) as fp16
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # x̂
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm + output‑gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def final_ln_linear_kernel(
    x_ptr,            # *const half*   [total_rows, hidden_dim]   (out of TriMul)
    gate_ptr,         # *const half*   [total_rows, hidden_dim]   (out_gate)
    weight_ptr,       # *const half*   [dim, hidden_dim]        (to_out weight)
    ln_weight_ptr,    # *const float*  [hidden_dim]            (to_out_norm.γ)
    ln_bias_ptr,      # *const float*  [hidden_dim]            (to_out_norm.β)
    out_ptr,          # *mut   float*  [total_rows, dim]       (final fp32)
    total_rows,       # int (runtime)
    hidden_dim: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # Compute per‑row mean & variance over the hidden dimension
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim
        ptr = x_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Produce final projection: LN(out) → gate → Linear → fp32 out
    # ---------------------------------------------------------------
    for o in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim

        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < hidden_dim

            # load out and gate
            x = tl.load(x_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_K]
            g = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_K]

            # LayerNorm affine transformation
            norm = (x - mean[:, None]) * rstd[:, None]          # x̂
            gamma = tl.load(ln_weight_ptr + offs_k, mask=mask_k, other=0.0)  # γ (hidden_dim)
            beta  = tl.load(ln_bias_ptr   + offs_k, mask=mask_k, other=0.0)  # β
            norm = norm * gamma[None, :] + beta[None, :]        # γ·x̂+β

            # Apply output‑gate
            gated = norm * g                                    # [BLOCK_M, BLOCK_K]

            # Load weight block transposed: shape (BLOCK_K, BLOCK_O)
            w_T = tl.load(
                weight_ptr + offs_o[None, :] * hidden_dim + offs_k[:, None],
                mask=mask_o[None, :] & mask_k[:, None],
                other=0.0,
            ).to(tl.float32)                                   # [BLOCK_K, BLOCK_O]

            # Accumulate matmul: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_O)
            acc += tl.dot(gated, w_T)

        # Store result (fp32)
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_o[None, :]
        tl.store(out_ptrs,
                 acc,
                 mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Custom Triton‑accelerated TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    B, N, _, C = input_tensor.shape   # C == dim
    total_rows = B * N * N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (over C) → fp16 output
    # --------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)          # fp32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections (left/right + three gates)
    # --------------------------------------------------------------
    # Concatenate the five projection weight matrices: shape (5*H, C)
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                   # (5*H, C)

    # One GEMM: [total_rows, C] @ [C, 5*H]^T -> [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split the concatenated result
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: [total_rows, H] (fp16)

    # --------------------------------------------------------------
    # 2a️⃣ Optional mask (applied only to left/right projections)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 2b️⃣ Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)   # shape: [total_rows, H]

    left = (left_proj * left_gate).view(B, N, N, hidden_dim)   # [B,N,N,H]
    right = (right_proj * right_gate).view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)             # later used after LN

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N) # (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)                           # fp16, (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Fuse second LayerNorm, output‑gate, and final projection
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()          # fp16
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim).contiguous()  # fp16

    final_fp32 = torch.empty((total_rows_h, dim), dtype=torch.float32,
                             device=device)

    # Kernel launch parameters (tuned for H100)
    BLOCK_O = 32   # output‑channel block
    BLOCK_K = 32   # hidden‑dim block
    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)

    final_ln_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        weights["to_out.weight"].half(),
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        final_fp32,
        total_rows_h,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_O=BLOCK_O,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 5️⃣ Reshape to the expected output layout
    # --------------------------------------------------------------
    out = final_fp32.view(B, N, N, dim)   # float32

    return out
