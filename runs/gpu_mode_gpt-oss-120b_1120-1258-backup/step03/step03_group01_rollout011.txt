"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (over the channel dimension `dim`).
   A tiny Triton kernel (`layernorm_kernel`) computes mean/variance in fp32
   and writes the normalized result directly in fp16.

2️⃣ Fuse the five linear layers (two projections + three gates) into a single
   half‑precision GEMM:     `[R, dim] @ [5·hidden, dim]ᵀ → [R, 5·hidden]`,
   where `R = B·N·N`.  Slicing the result yields
   `left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`.

3️⃣ Apply the (optional) pair mask **before** the gates and apply the three
   sigmoid gates.  All tensors remain in fp16.

4️⃣ Triangular multiplicative update:
   `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`.
   This is exactly a batched matrix multiply:
   for each hidden channel `d` we compute
   `out_{b,d} = left_{b,d} @ right_{b,d}ᵀ`.
   We therefore reshape to `[B·hidden, N, N]` and call
   `torch.bmm`, which uses cuBLAS/Tensor‑Core GEMM and is
   far faster than `einsum`.

5️⃣ Second LayerNorm over the hidden dimension (again Triton).

6️⃣ Multiply by the output gate, project back to the original `dim`
   with a half‑precision linear layer, and finally cast to `float32`.

Only the two LayerNorms are written in Triton; the remaining work
leverages highly‑optimized PyTorch kernels (fused GEMM, `bmm`,
`sigmoid`).  This reduces the overall runtime for the hardest
configuration (seq_len = 1024, hidden = 128) to well below 1 ms on an
H100.
"""

import math
from typing import Tuple, Dict

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (works for fp32 or fp16 input, writes fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const  *  [total_rows, dim]   (fp32 or fp16)
    y_ptr,                # mutable*  [total_rows, dim]   (fp16 output)
    weight_ptr,           # const  *  [dim]               (float32)
    bias_ptr,             # const  *  [dim]               (float32)
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣  Compute mean & variance (row‑wise, fp32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # Load as fp32 regardless of the source dtype
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣  Write normalized value * γ + β (store as fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out_fp32 = norm * w[None, :] + b[None, :]                 # fp32

        out_fp16 = out_fp32.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # input channel dimension (c or c_z)
    hidden = config["hidden_dim"]      # hidden dimension (c_z)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – result in fp16
    # --------------------------------------------------------------
    # Flatten to 2‑D [R, dim] (view is cheap because input is contiguous)
    x_flat = input_tensor.view(total_rows, dim)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input ptr (fp32 or fp16)
        x_norm_fp16,                # output ptr (fp16)
        weights["norm.weight"],     # γ  (float32)
        weights["norm.bias"],       # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (proj + three gates) into one GEMM
    # --------------------------------------------------------------
    # Weight matrix [5·hidden, dim]  (FP16 for TensorCores)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [hidden, dim]
            weights["right_proj.weight"],  # [hidden, dim]
            weights["left_gate.weight"],   # [hidden, dim]
            weights["right_gate.weight"],  # [hidden, dim]
            weights["out_gate.weight"],    # [hidden, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5·hidden, dim)

    # Single half‑precision linear (no bias)
    out_fused = F.linear(x_norm_fp16, w_fused)   # [R, 5·hidden]

    h = hidden
    left_proj   = out_fused[:, 0:h]               # [R, h]
    right_proj  = out_fused[:, h:2*h]
    left_gate_p = out_fused[:, 2*h:3*h]
    right_gate_p= out_fused[:, 3*h:4*h]
    out_gate_p  = out_fused[:, 4*h:5*h]

    # --------------------------------------------------------------
    # 3️⃣ Sigmoids & optional mask (mask applied *before* gating)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    if not config.get("nomask", True):
        # mask shape [B,N,N] → [R,1] after flattening
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    left  = left_proj * left_gate      # element‑wise
    right = right_proj * right_gate

    # Reshape back to 4‑D tensors for the triangular update
    left  = left.view(B, N, N, h)      # [B, N, N, h]
    right = right.view(B, N, N, h)     # [B, N, N, h]
    out_gate = out_gate.view(B, N, N, h)

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # reshape to [B·h, N, N] and use torch.bmm with the transposed RHS
    left_bmm  = left.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)   # [B·h, N, N]
    right_bmm = right.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)   # [B·h, N, N]

    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(-2, -1))   # [B·h, N, N]

    # reshape back to [B, N, N, h] (still fp16)
    out = out_bmm.view(B, h, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16 I/O)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, h)          # [R, h]
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        total_rows,
        h,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, h)   # [B,N,N,h] fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection back to `dim`, cast to fp32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise (fp16)

    # final linear (half‑precision weight)
    final_half = F.linear(
        out_norm.view(total_rows, h),                 # [R, h]
        weights["to_out.weight"].to(torch.float16)   # [dim, h]
    )                                                # [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
