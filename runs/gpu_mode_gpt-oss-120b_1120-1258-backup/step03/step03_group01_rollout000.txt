"""
Fast “outgoing” TriMul forward (AlphaFold‑3) – Triton + PyTorch

Algorithm
---------
1️⃣  Row‑wise LayerNorm of the input tensor (fp32 → fp16) – Triton kernel.
2️⃣  Fuse the three linear projections (left, right, three gates) into a
    single half‑precision GEMM:  X_norm @ W_catᵀ  →  [R,5·hidden].
3️⃣  Slice the concatenated result, apply the optional mask (only to the
    two projected tensors) and element‑wise sigmoid gates.
4️⃣  Reshape to `[B,N,N,hidden]` and compute the triangular multiplicative
    update using a batched matrix‑multiply (torch.bmm) per hidden channel:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    i.e. out = (leftᵀ @ right) for each hidden d.
5️⃣  Second LayerNorm over the hidden dimension (half‑precision) – Triton.
6️⃣  Multiply by the output gate, final linear projection back to `dim`,
    and cast the result to float32.

All heavy GEMMs run in fp16 so Tensor‑Core units on an H100 are fully
exploited.  The only explicit Triton kernels are the two LayerNorms, which
are heavily tiled (BLOCK_M × BLOCK_D) to achieve high occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise, works for fp32/fp16 input, writes fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *   [total_rows, dim]   (float32 or float16)
    y_ptr,            # mutable * [total_rows, dim]   (float16 output)
    weight_ptr,       # const *   [dim]               (float32)
    bias_ptr,         # const *   [dim]               (float32)
    total_rows,       # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass: compute mean & variance per row
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        # promote to fp32 regardless of input precision
        val_f32 = val.to(tl.float32)

        sum_val += tl.sum(val_f32, axis=1)
        sum_sq  += tl.sum(val_f32 * val_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2nd pass: write (x-mean)*rstd*γ+β as fp16
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        norm = (val_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out_fp32 = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out_fp32.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N  # R

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – fp32 → fp16
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_flat = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64   # 64 works for dim up to 1024 and hidden up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse the five linear layers (proj + 3 gates) into one GEMM
    # --------------------------------------------------------------
    # concat all five weight matrices (each [hidden_dim, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*hidden, dim]

    # fused linear: [R, dim] @ [5*hidden, dim].T → [R, 5*hidden]
    fused_out = F.linear(x_norm_flat, w_concat)   # fp16

    H = hidden_dim
    # slice the concatenated result
    left_proj   = fused_out[:, 0        :H]          # [R, H]
    right_proj  = fused_out[:, H        :2*H]       # [R, H]
    left_gate_p = fused_out[:, 2*H      :3*H]
    right_gate_p= fused_out[:, 3*H      :4*H]
    out_gate_p  = fused_out[:, 4*H      :5*H]

    # sigmoid gates (half‑precision)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # optional mask – broadcast over the hidden dimension
    if not nomask:
        # mask shape [B,N,N] → [R,1] (float16)
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # apply element‑wise gating
    left  = left_proj  * left_gate
    right = right_proj * right_gate

    # reshape back to 4‑D tensors
    left  = left.view(B, N, N, H)          # [B,N,N,H]
    right = right.view(B, N, N, H)         # [B,N,N,H]
    out_gate = out_gate.view(B, N, N, H)   # [B,N,N,H]

    # --------------------------------------------------------------
    # 3) Triangular multiplicative update via batched matmul
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # move hidden dim to the second axis for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2)  # [B, H, N, N]

    # flatten batch & hidden dim
    left_flat  = left_perm.reshape(B * H, N, N)          # [B*H, N, N]
    right_flat = right_perm.reshape(B * H, N, N)         # [B*H, N, N]

    # (left @ rightᵀ) for each hidden channel
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H,N,N]

    # reshape back to [B,H,N,N] → [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B,N,N,H] (fp16)

    # --------------------------------------------------------------
    # 4) Second LayerNorm over hidden dimension (fp16 → fp16)
    # --------------------------------------------------------------
    out_flat2 = out.contiguous().view(total_rows, H)   # [R, H]
    out_norm_flat = torch.empty_like(out_flat2, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat2,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_flat.view(B, N, N, H)      # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 5) Apply output gate, final projection, cast to float32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # element‑wise fp16

    # flatten again for the final linear projection
    out_norm_flat3 = out_norm.view(total_rows, H)   # [R, H]

    final_half = F.linear(
        out_norm_flat3,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
