"""
TriMul (outgoing) – a highly‑fused forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input channel C (custom Triton kernel).
2️⃣ Fuse the five linear projections (`left`, `right`, three gates) into a single
   matrix‑multiply by concatenating the weight matrices.  The input after LN is
   cast to FP16 so the GEMM runs on Tensor‑Cores.
3️⃣ (Optional) mask the `left`/`right` projections.
4️⃣ Apply sigmoid gates and element‑wise gating (still in FP16).
5️⃣ Triangular multiplicative update:
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   performed as a batched GEMM (`left @ rightᵀ`) on tensors of shape
   (B·H, N, N) in FP16.
6️⃣ Row‑wise LayerNorm on the hidden dimension H (same Triton kernel,
   output kept in FP16 to avoid extra casts).
7️⃣ Multiply by the output gate, then a final FP16 linear projection back to
   the original dimension C.
8️⃣ Cast the result back to FP32.

Both LayerNorms are written in Triton (the only required custom kernels);
all heavy linear algebra uses cuBLAS/Tensor‑Core fp16 GEMMs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16/fp32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const   float*   [total_rows, dim]
    y_ptr,          # *mut     float*   [total_rows, dim] (output dtype = y_ptr dtype)
    weight_ptr,     # *const   float*   [dim]
    bias_ptr,       # *const   float*   [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣ Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣ Write normalized output (γ·x̂ + β)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple containing
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be ignored.
            - weights_dict : dict with all model weights.
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool (opt.)}

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over the channel dimension C (custom Triton)
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N                         # flatten B×N×N

    x_flat = x.view(total_rows, dim)               # [total_rows, C]
    x_norm_flat = torch.empty_like(x_flat)         # fp32 output buffer

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # cast to FP16 for all subsequent GEMMs
    x_norm_fp16 = x_norm_flat.half()               # [total_rows, C]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + 3 gates)
    # ------------------------------------------------------------------
    # concatenate [left_proj, right_proj, left_gate, right_gate, out_gate]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                         # (5*H, C)

    # fused linear: [total_rows, C] @ [C, 5*H].T -> [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, w_concat)    # fp16, bias=False

    # split back into the five tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = \
        torch.split(proj_all, hidden_dim, dim=1)

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [total_rows, 1]
        mask_flat_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_flat_fp16
        right_proj_fp16 = right_proj_fp16 * mask_flat_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)     # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate   = torch.sigmoid(out_gate_fp16)

    left_fp16  = left_proj_fp16 * left_gate        # [total_rows, H]
    right_fp16 = right_proj_fp16 * right_gate     # [total_rows, H]

    # reshape back to the 4‑D layout
    left = left_fp16.view(B, N, N, hidden_dim)            # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #     out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    # left: (B, N, N, H) → (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: transpose on the k‑axis → (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)  # [B*H, N, N]

    # batched matrix‑multiply in fp16 (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)          # [B*H, N, N] (fp16)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # out shape: [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (custom Triton)
    # ------------------------------------------------------------------
    total_rows_h = total_rows                         # B·N·N
    out_flat = out.view(total_rows_h, hidden_dim)     # [total_rows_h, H]
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_D2 = 32  # hidden_dim is small; a tighter column block yields better regs.
    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D2,
    )

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)   # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # element‑wise (fp16)

    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)    # [total_rows_h, C] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to fp32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
