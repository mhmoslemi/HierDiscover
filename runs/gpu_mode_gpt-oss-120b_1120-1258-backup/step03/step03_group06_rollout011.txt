@triton.jit
def gated_proj_kernel(
    left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, out_left_ptr, out_right_ptr,
    total_rows, hidden,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_row = offs_m < total_rows

   offs_n = tl.arange(0, BLOCK_N)
   mask_n = offs_n < hidden

   # load left projection and gates
   left_proj = tl.load(left_proj_ptr + offs_m[:, None] * hidden + offs_n[None, :],
                       mask=mask_row[:, None] & mask_n[None, :],
                       other=0.0).to(tl.float32)   # cast to fp32 for better sigmoid?
   right_proj = tl.load(right_proj_ptr + offs_m[:, None] * hidden + offs_n[None, :],
                       mask=mask_row[:, None] & mask_n[None, :],
                       other=0.0).to(tl.float32)

   left_gate = tl.load(left_gate_ptr + offs_m[:, None] * hidden + offs_n[None, :],
                       mask=mask_row[:, None] & mask_n[None, :],
                       other=0.0).to(tl.float32)
   right_gate = tl.load(right_gate_ptr + offs_m[:, None] * hidden + offs_n[None, :],
                        mask=mask_row[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

   # compute sigmoid
   left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
   right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

   if mask_ptr is not None:
       mask_val = tl.load(mask_ptr + offs_m, mask=mask_row, other=0.0).to(tl.float32)  # shape (total_rows, 1)
       left_proj = left_proj * mask_val[:, None]
       right_proj = right_proj * mask_val[:, None]

   # apply gates
   out_left = left_proj * left_gate_sig
   out_right = right_proj * right_gate_sig

   # store as half
   tl.store(out_left_ptr + offs_m[:, None] * hidden + offs_n[None, :],
            out_left.to(tl.float16),
            mask=mask_row[:, None] & mask_n[None, :])
   tl.store(out_right_ptr + offs_m[:, None] * hidden + offs_n[None, :],
            out_right.to(tl.float16),
            mask=mask_row[:, None] & mask_n[None, :])

"""
TriMul “outgoing” forward pass (AlphaFold‑3) – highly‑optimized mixed‑precision version.

Algorithm
---------
1. **Row‑wise LayerNorm** over the last channel (`dim`) using the highly‑optimized
   `torch.nn.functional.layer_norm`.  Result is cast to fp16.
2. **Single fused linear projection**:
   – concatenate the five weight matrices (left/right proj + three gates) → 
     shape (5·hidden, dim)  (fp16).  
   – one GEMM (`F.linear`) produces five tensors in fp16:
        left_proj, right_proj, left_gate, right_gate, out_gate.
3. **Gate + optional mask** – a tiny Triton kernel fuses the three
   element‑wise ops per row:
        left  = left_proj  * sigmoid(left_gate)  * mask
        right = right_proj * sigmoid(right_gate) * mask
   (`mask` = 1 when `nomask=True`).  Only one kernel launch is required.
4. **Triangular multiplicative update** – batched matrix‑multiply:
        out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d] .
   Implemented with a single `torch.bmm` after reshaping
   to (B·hidden, N, N).  All tensors stay in fp16.
5. **Second LayerNorm + output gate** – use the standard CUDA‑accelerated
   `torch.nn.functional.layer_norm` (float32) followed by an fp16
   element‑wise multiplication with the sigmoid of `out_gate`.
6. **Final linear projection** back to `dim` (`to_out.weight`), again a
   half‑precision GEMM, and cast the result to float32.

Only the gate+mask fusion is written in Triton; the rest relies on cuBLAS
(and cuDNN) which delivers near‑peak Tensor‑Core throughput on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Triton kernel – fuse left/right gating with (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def gated_proj_kernel(
    left_proj_ptr,   # const half* [total_rows, hidden]
    right_proj_ptr,  # const half* [total_rows, hidden]
    left_gate_ptr,   # const half* [total_rows, hidden]   (sigmoid already applied)
    right_gate_ptr,  # const half* [total_rows, hidden]   (sigmoid already applied)
    mask_ptr,        # const half* [total_rows]           (1.0 if no mask)
    out_left_ptr,    # *mut half*  [total_rows, hidden]
    out_right_ptr,   # *mut half*  [total_rows, hidden]
    total_rows: tl.int32,
    hidden: tl.int32,
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 128)
    BLOCK_H: tl.constexpr,   # hidden‑dim tile size (e.g. 64)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # loop over hidden dimension tiles
    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # pointers for the current tile
        left_ptrs   = left_proj_ptr   + offs_m[:, None] * hidden + offs_h[None, :]
        right_ptrs  = right_proj_ptr  + offs_m[:, None] * hidden + offs_h[None, :]
        lgate_ptrs  = left_gate_ptr   + offs_m[:, None] * hidden + offs_h[None, :]
        rgate_ptrs  = right_gate_ptr  + offs_m[:, None] * hidden + offs_h[None, :]

        # load tensors (half‑precision)
        left   = tl.load(left_ptrs,
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=tl.float16(0.0))
        right  = tl.load(right_ptrs,
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=tl.float16(0.0))
        lgate  = tl.load(lgate_ptrs,
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=tl.float16(0.0))
        rgate  = tl.load(rgate_ptrs,
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=tl.float16(0.0))

        # scalar mask per row (broadcasted over hidden)
        row_mask = tl.load(mask_ptr + offs_m,
                           mask=mask_m,
                           other=tl.float16(1.0))
        row_mask = row_mask[:, None]   # shape (BLOCK_M, 1)

        # fused gating + mask
        out_l = left  * lgate * row_mask
        out_r = right * rgate * row_mask

        # store results
        tl.store(out_left_ptr  + offs_m[:, None] * hidden + offs_h[None, :],
                 out_l,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_right_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                 out_r,
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 2) Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C]    (float32)
            mask_tensor  : torch.Tensor [B, N, N]     (float32/bool) – may be None
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)          : input channel dimension C
                - hidden_dim (int)   : projection dimension H
                - nomask (bool)      : True → treat mask as all‑ones
                - eps (float)        : LayerNorm epsilon (optional)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows when we flatten the first three dims

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fast torch implementation)
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension only
    x_norm_fp16 = torch.nn.functional.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                        # [B, N, N, dim] → fp16

    # ------------------------------------------------------------------
    # 2) Fused linear projection (five heads) – all in FP16
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape (5*hidden, dim))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()          # (5*hidden, dim)

    # Single GEMM: (total_rows, dim) × (5*hidden, dim).T → (total_rows, 5*hidden)
    proj_all = F.linear(
        x_norm_fp16.view(total_rows, dim),   # flatten to 2‑D
        concat_w,
    )                                        # fp16 result

    # Split into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden, dim=1
    )   # each: (total_rows, hidden)

    # --------------------------------------------------------------
    # 3) Sigmoid on gates (still FP16)
    # --------------------------------------------------------------
    left_gate_sig  = torch.sigmoid(left_gate)
    right_gate_sig = torch.sigmoid(right_gate)
    out_gate_sig   = torch.sigmoid(out_gate)   # (total_rows, hidden)

    # --------------------------------------------------------------
    # 4) Fuse mask & gating via Triton (required kernel)
    # --------------------------------------------------------------
    if nomask:
        # mask of all ones – cheap tensor on the same device
        mask_fp16 = torch.ones(total_rows, dtype=torch.float16, device=device)
    else:
        # mask originally [B, N, N]; flatten & broadcast to hidden dim later
        mask_fp16 = mask.view(total_rows).to(torch.float16)

    # allocate output buffers for the gated projections
    left_fp16  = torch.empty_like(left_proj)   # (total_rows, hidden)
    right_fp16 = torch.empty_like(right_proj)

    BLOCK_M = 128    # rows per program
    BLOCK_H = 64     # hidden‑dim tile size (covers hidden up to 384 comfortably)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    gated_proj_kernel[grid](
        left_proj,
        right_proj,
        left_gate_sig,
        right_gate_sig,
        mask_fp16,
        left_fp16,
        right_fp16,
        total_rows,
        hidden,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5) Reshape for the triangular update
    # ------------------------------------------------------------------
    left  = left_fp16.view(B, N, N, hidden)    # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)   # (B, N, N, H)

    # Perform the “outgoing” TriMul:
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # Reshape to batched matrices: (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                         # (B·H, N, N), fp16
    out = (
        out_mat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                                                  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm + element‑wise output gate
    # ------------------------------------------------------------------
    out_norm_fp32 = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32
    out_norm_fp16 = out_norm_fp32.to(torch.float16)

    # Broadcast the sigmoid‑gated out_gate and multiply
    out_gate_fp16 = out_gate_sig.view(B, N, N, hidden)
    out_norm_gate = out_norm_fp16 * out_gate_fp16   # fp16, still (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    final_fp16 = F.linear(
        out_norm_gate.view(total_rows, hidden),  # flatten again
        to_out_w_fp16,
    )                                            # (total_rows, dim), fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast to float32

    return final
