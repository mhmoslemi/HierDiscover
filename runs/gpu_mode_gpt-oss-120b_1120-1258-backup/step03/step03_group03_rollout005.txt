"""
TriMul – outgoing version (AlphaFold‑3) optimized with Triton
----------------------------------------------------------------
Algorithm
1️⃣  Row‑wise LayerNorm (float32) on the input tensor [B,N,N,D].
2️⃣  Fuse the five linear projections (`left_proj`, `right_proj`,
    `left_gate`, `right_gate`, `out_gate`) into a single half‑precision
    GEMM:   X_norm @ W_concatᵀ  →  [R,5*H]   (R = B·N·N).
3️⃣  Optional pairwise mask is applied to the two projection parts.
    Gate tensors pass through a sigmoid and are fused with the projections.
4️⃣  **Tri‑multiplicative update** – a batched matrix multiplication
    L @ Rᵀ where L,R ∈ ℝ^{BH×N×N} (BH = B·H).  Implemented as a custom
    Triton kernel (`trimul_matmul_kernel`) that computes the sum over the
    intermediate dimension `k` tile‑wise, accumulating in FP32 and storing
    FP16.
5️⃣  Second LayerNorm (half‑precision) + output‑gate (`out_gate`) are
    fused in `layernorm_gate_kernel`.
6️⃣  Final linear projection back to the original channel dimension (FP16
    GEMM) and cast to FP32.

All heavy compute (both LayerNorms and the N³ reduction) runs on Triton,
while the cheap dense linear layers use cuBLAS via `torch.nn.functional.linear`.
The implementation supports both masked and un‑masked inputs and works for
any configuration from the test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32) – used on the input tensor
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       float*   [total_rows, dim] (output)
    weight_ptr,     # const float*   [dim]               (γ)
    bias_ptr,       # const float*   [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm in FP32."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)       # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- 1st pass: compute mean / variance (float32 accumulators) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)               # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)        # row‑wise sum‑of‑squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                        # 1 / sqrt(var+eps)

    # ---- 2nd pass: write normalized values (apply γ/β) ----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5️⃣ LayerNorm (half‑precision) + output‑gate fusion
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*    [total_rows, dim]   (input)
    y_ptr,          #       half*    [total_rows, dim]   (output)
    gate_ptr,       # const half*    [total_rows, dim]   (out_gate)
    weight_ptr,     # const half*    [dim]                (γ)
    bias_ptr,       # const half*    [dim]                (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """LayerNorm (FP16) followed by element‑wise multiplication with a gate."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- 1st pass: mean / variance (FP32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- 2nd pass: normalize, apply γ/β, multiply by gate ----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load input
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # normalize
        norm = (x_val - mean[:, None]) * rstd[:, None]

        # γ / β (cast to FP32 for the arithmetic)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # gate (half → FP32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        out = out * g

        # store back as half
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Tri‑multiplicative update (batched GEMM) – custom Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr,      # const half*   [BH, N, N]   (L)
    right_ptr,     # const half*   [BH, N, N]   (R)
    out_ptr,       #       half*   [BH, N, N]   (output)
    total_batches,  # i32   = BH
    N,               # i32   sequence length
    stride_batch,    # i32   stride between batches (N*N)
    stride_row,      # i32   stride between rows   (N)
    stride_col,      # i32   stride between columns (1)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul operation:

        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]

    The kernel treats the right operand as transposed on‑the‑fly
    (no explicit memory transpose).  Accumulation is performed in FP32
    for numerical stability.
    """
    batch_id = tl.program_id(0)          # which (b * hidden) slice
    pid_m    = tl.program_id(1)          # block row   (i)
    pid_n    = tl.program_id(2)          # block col   (j)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        k_offs = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # -------- left block --------  (M x K)
        left_ptrs = left_ptr \
                    + batch_id * stride_batch \
                    + offs_m[:, None] * stride_row \
                    + k_offs[None, :] * stride_col
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # -------- right block (transposed) --------  (K x N)
        # We need right[j,k] → in memory this is right[batch, j, k]
        right_ptrs = right_ptr \
                     + batch_id * stride_batch \
                     + offs_n[None, :] * stride_row \
                     + k_offs[:, None] * stride_col
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # accumulate the product
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # write back
    out_ptrs = out_ptr \
               + batch_id * stride_batch \
               + offs_m[:, None] * stride_row \
               + offs_n[None, :] * stride_col
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    hidden_dim   = config["hidden_dim"]    # H
    eps          = 1e-5
    nomask       = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                  # R = B·N·N

    # -------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # -------------------------------------------------
    x = input_tensor.contiguous()            # [B,N,N,D] FP32
    x_flat = x.view(total_rows, dim)         # [R,D]
    x_norm = torch.empty_like(x_flat)        # [R,D] FP32

    BLOCK_M = 256
    BLOCK_D = 32
    grid0 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid0](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  FP32 [D]
        weights["norm.bias"],     # β  FP32 [D]
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # -------------------------------------------------
    # 2️⃣ Fused projection & gating (half‑precision)
    # -------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)               # [R,D] FP16

    # Concatenate all five weight matrices → shape (5*H, D)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # (5H, D)

    # One big GEMM: [R,D] @ (5H,D)ᵀ → [R,5H] (FP16)
    proj_all = F.linear(x_norm_h, w_concat)           # [R, 5*H]

    H = hidden_dim
    left_proj = proj_all[:, 0:H]
    left_gate = proj_all[:, H:2*H]
    right_proj = proj_all[:, 2*H:3*H]
    right_gate = proj_all[:, 3*H:4*H]
    out_gate = proj_all[:, 4*H:5*H]

    # -------------------------------------------------
    # 3️⃣ Apply pair‑wise mask (if present) and sigmoid gates
    # -------------------------------------------------
    if not nomask:
        # mask has shape [B,N,N] → reshape to [R,1] for broadcasting
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # element‑wise gating
    left  = left_proj * left_gate          # [R, H] FP16
    right = right_proj * right_gate        # [R, H] FP16

    # -------------------------------------------------
    # 4️⃣ Tri‑multiplicative update – custom Triton kernel
    # -------------------------------------------------
    # reshape to 4‑D tensors and fuse batch & hidden dim
    left = left.view(B, N, N, H)   # [B,N,N,H]
    right = right.view(B, N, N, H) # [B,N,N,H]

    # bring hidden dim to batch axis
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_mat = right.permute(0, 3, 1, 2).contiguous() # [B,H,N,N]

    BH = B * H
    left_mat = left_mat.view(BH, N, N)   # [BH, N, N]
    right_mat = right_mat.view(BH, N, N) # [BH, N, N]

    # allocate output buffer
    out_mat = torch.empty_like(left_mat)   # FP16, shape [BH,N,N]

    # strides for a contiguous (batch, rows, cols) tensor
    stride_batch = left_mat.stride(0)   # N * N
    stride_row   = left_mat.stride(1)   # N
    stride_col   = left_mat.stride(2)   # 1

    # tile sizes – tuned for H100 (128‑bit registers)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid1 = (BH,
             triton.cdiv(N, BLOCK_M),
             triton.cdiv(N, BLOCK_N))
    trimul_matmul_kernel[grid1](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,
        stride_batch,
        stride_row,
        stride_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused)
    # -------------------------------------------------
    # Restore [B,N,N,H] layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out_flat = out.view(total_rows, H)                               # [R, H] FP16
    out_gate_flat = out_gate.view(total_rows, H)                     # [R, H] FP16

    out_norm_gate = torch.empty_like(out_flat)                       # FP16 buffer

    BLOCK_M2 = 256
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ  FP16 [H]
        weights["to_out_norm.bias"].to(torch.float16),     # β  FP16 [H]
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )

    # -------------------------------------------------
    # 6️⃣ Final projection back to original dim
    # -------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [D, H] FP16
    final_half = F.linear(out_norm_gate, w_to_out)         # [R, D] FP16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
