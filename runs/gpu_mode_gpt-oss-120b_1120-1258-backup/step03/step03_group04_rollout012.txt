def final_fused_kernel(...):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean & var first
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_ptrs = out_ptr + offs_m[:, None] * H + offs_h[None, :]
        out_vals = tl.load(out_ptrs, mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        out_f32 = out_vals.to(tl.float32)

        sum_val += tl.sum(out_f32, axis=1)
        sum_sq += tl.sum(out_f32 * out_f32, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Initialize per-d-block accumulators
    num_blocks = tl.cdiv(dim, BLOCK_D)
    # Create a list of accumulators
    acc = [tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32) for _ in range(num_blocks)]

    # Second pass: loop over H blocks and accumulate per d block
    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load out and out_gate
        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # Normalize out
        norm = (out_vals - mean[:, None]) * rstd[:, None]

        # Apply LN weight and bias
        ln_w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
        ln_b = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)

        norm = norm * ln_w[None, :] + ln_b[None, :]
        gated = norm * gate_vals   # shape (BLOCK_M, BLOCK_H)

        # Accumulate contributions to each dim block
        for d in range(0, num_blocks):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # Load weight transposed for (BLOCK_H, BLOCK_D)
            w_ptrs = to_out_weight_ptr + offs_h[:, None] + offs_d[None, :] * H
            w = tl.load(w_ptrs,
                mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            # Dot product: (BLOCK_M, BLOCK_H) @ (BLOCK_H, BLOCK_D) -> (BLOCK_M, BLOCK_D)
            acc[d] += tl.dot(gated, w)

    # Store results
    for d in range(0, num_blocks):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        out_ptrs = out_final_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs,
                 acc[d].to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ First LayerNorm over the channel dimension C.
   – Implemented in Triton, fp32 accumulation → fp16 output.
2️⃣ Fuse the five linear projections (left, right and three gates) into a
   single half‑precision GEMM (torch.nn.functional.linear on a concatenated
   weight matrix).
3️⃣ (Optional) mask the Left/Right projections before gating.
4️⃣ Apply sigmoid gates and element‑wise gating → tensors of shape
   [B, N, N, H] (H = hidden_dim, FP16).
5️⃣ Triangular multiplicative update:
      out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   realized as a batched GEMM `left @ rightᵀ` on tensors
   (B·H, N, N) in FP16.
6️⃣ Fuse the second LayerNorm, the output‑gate, and the final linear
   projection into a single Triton kernel.  This kernel:
   – computes per‑position mean/variance over H,
   – normalises, applies the output‑gate,
   – multiplies by the learned weight matrix `to_out.weight`
   (shape [dim, H]) and adds the LayerNorm scale/bias.
   All work is done in registers; only the final tensor
   [B, N, N, dim] (FP16) is written to memory.
7️⃣ Cast the result back to FP32 (the format expected by the caller).

The implementation removes two large intermediate buffers (the
second LayerNorm result and the final linear input) and keeps the
heaviest work on Tensor‑Core‑friendly FP16 GEMMs, giving a large
runtime reduction compared with the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton Row‑wise LayerNorm (float32 reduction → fp16 output)
# -------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # *const float32    [total_rows, dim]
    y_ptr,            # *mut   float16    [total_rows, dim]
    weight_ptr,       # *const float16    [dim] (γ)
    bias_ptr,         # *const float16    [dim] (β)
    total_rows,       # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣ compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over columns in chunks of BLOCK_D
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # ptrs: [row, col]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                      # fp32
        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣ write normalized + γ/β (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                      # fp32
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]   # fp32

        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp16
        beta  = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp16

        out_fp16 = (norm * gamma[None, :].to(tl.float32)
                         + beta[None, :].to(tl.float32)).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# -------------------------------------------------------------
# Triton kernel that fuses
#   • second LayerNorm over H
#   • output‑gate multiplication
#   • final linear projection (to_out)
# -------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,            # *const float16   [total_rows, H]   (tri‑mul result)
    out_gate_ptr,       # *const float16   [total_rows, H]   (output‑gate)
    ln_weight_ptr,      # *const float16   [H]              (to_out_norm.γ)
    ln_bias_ptr,        # *const float16   [H]              (to_out_norm.β)
    to_out_w_ptr,       # *const float16   [dim, H]         (to_out.weight)
    final_ptr,          # *mut   float16   [total_rows, dim]
    total_rows,         # int
    H: tl.constexpr,    # hidden dimension
    dim: tl.constexpr,  # output channel dimension
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim block size
    BLOCK_D: tl.constexpr,   # output‑dim block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------------------
    # 1️⃣ mean / variance over H (fp32)
    # -------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # out values (fp16 → fp32)
        ptrs = out_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0)                      # fp16
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------------------
    # 2️⃣ allocate per‑output‑dim block accumulators
    # -------------------------------------------------------------
    NUM_D_BLOCKS = tl.cdiv(dim, BLOCK_D)
    acc = [tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32) for _ in range(NUM_D_BLOCKS)]

    # -------------------------------------------------------------
    # 3️⃣ second pass: compute gated vector & accumulate GEMM
    # -------------------------------------------------------------
    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw out and gate (fp16 → fp32)
        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_H]
        gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)         # [BLOCK_M, BLOCK_H]

        # normalise
        norm = (out_vals - mean[:, None]) * rstd[:, None]        # fp32

        # layer‑norm scale & bias (γ / β)
        ln_w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)  # [BLOCK_H]
        ln_b = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0).to(tl.float32)  # [BLOCK_H]

        norm = norm * ln_w[None, :] + ln_b[None, :]                # fp32

        # apply output‑gate
        gated = norm * gate_vals                                   # fp32, shape (BLOCK_M, BLOCK_H)

        # ---------------------------------------------------------
        #   accumulate gated @ to_out_weightᵀ   (for every dim block)
        # ---------------------------------------------------------
        for d in range(0, NUM_D_BLOCKS):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # Load transposed weight slice: shape (BLOCK_H, BLOCK_D)
            # weight layout is [dim, H]; transposed address = h + d*H
            w_ptrs = to_out_w_ptr + offs_h[:, None] + offs_d[None, :] * H
            w = tl.load(w_ptrs,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                # fp32

            # (BLOCK_M, BLOCK_H) @ (BLOCK_H, BLOCK_D) -> (BLOCK_M, BLOCK_D)
            acc[d] += tl.dot(gated, w)

    # -------------------------------------------------------------
    # 4️⃣ write final output (fp16)
    # -------------------------------------------------------------
    for d in range(0, NUM_D_BLOCKS):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        out_ptrs = final_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs,
                 acc[d].to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# -----------------------------------------------------------------
# Main entry point used by the evaluation harness
# -----------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    Tensor
        Shape [B, N, N, dim], dtype float32, on the same CUDA device as the input.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5                         # LayerNorm epsilon
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 0️⃣ bookkeeping
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                       # rows for the first LayerNorm

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (C‑dim) → fp16
    # -----------------------------------------------------------------
    x = input_tensor.contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M1 = 256        # rows per program (good occupancy)
    BLOCK_D1 = 64         # columns per iteration (keeps regs modest)
    grid1 = (triton.cdiv(total_rows, BLOCK_M1),)

    layernorm_fp16_kernel[grid1](
        x.view(total_rows, dim),            # x_ptr  (fp32)
        x_norm_fp16,                        # y_ptr  (fp16)
        weights["norm.weight"].half(),      # γ (fp16)
        weights["norm.bias"].half(),        # β (fp16)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M1,
        BLOCK_D=BLOCK_D1,
    )

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into one GEMM
    # -----------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, C])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # shape (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)   # [total_rows, 5*H]  (fp16)

    # split back
    splits = torch.split(proj_all, hidden_dim, dim=1)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = splits

    # -----------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # -----------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # -----------------------------------------------------------------
    # 4️⃣ Sigmoid gates and element‑wise gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)       # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # reshape to [B, N, N, H]
    left = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # -----------------------------------------------------------------
    # left : [B, N, N, H] -> [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right : need transpose on the k‑axis, i.e. rightᵀ over the last two dims
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim                         # batch dimension for GEMM
    left_mat = left_perm.view(BH, N, N)          # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)        # [B*H, N, N]

    # batched matrix multiplication (fp16, Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat)    # [B*H, N, N]  (fp16)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # -----------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, output‑gate, and final projection
    # -----------------------------------------------------------------
    total_rows_h = B * N * N                     # rows for the second LN / final proj
    out_flat = out.view(total_rows_h, hidden_dim)        # [rows, H]  (fp16)
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)  # [rows, H]  (fp16)

    final_fp16 = torch.empty((total_rows_h, dim), dtype=torch.float16, device=device)

    # kernel launch parameters (tune if necessary)
    BLOCK_M2 = 64          # rows per program – keeps register pressure low
    BLOCK_H = 32           # hidden‑dim tile
    BLOCK_D = 32           # output‑dim tile
    grid2 = (triton.cdiv(total_rows_h, BLOCK_M2),)

    final_fused_kernel[grid2](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"].half(),
        weights["to_out_norm.bias"].half(),
        weights["to_out.weight"].half(),
        final_fp16,
        total_rows_h,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32 and return
    # -----------------------------------------------------------------
    out_final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return out_final
