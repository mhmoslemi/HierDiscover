@triton.jit
def tri_mul_batch_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_i_l, stride_k_l, stride_h_l,
    stride_br, stride_i_r, stride_k_r, stride_h_r,
    stride_bo, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    b = batch_id // H
    h = batch_id % H

    # compute base offsets for each pointer
    left_base = left_ptr + b * stride_bl + h * stride_h_l
    right_base = right_ptr + b * stride_br + h * stride_h_r
    out_base = out_ptr + b * stride_bo + h * stride_h_o

    # tile loops across i (M) and j (N)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # accumulate maybe fp32

    # loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_m[:, None] * stride_i_l + offs_k[None, :] * stride_k_l
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B tile: shape (BLOCK_K, BLOCK_N) because we need B[k,j]
        b_ptrs = right_base + offs_k[:, None] * stride_k_r + offs_n[None, :] * stride_i_r
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # results shape BLOCK_M x BLOCK_N

    # possibly scale if needed. Convert to float16
    out = acc.to(tl.float16)

    # store output
    out_ptrs = out_base + offs_m[:, None] * stride_i_o + offs_n[None, :] * stride_j_o
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

        BLOCK_D=BLOCK_D_LN,

    x_norm = x_norm_h.view(B, N, N, dim)   # fp16

    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)

"""Fast “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input   [B, N, N, C] → FP16.
2️⃣  Three fused linear projections (proj + gate) are performed in FP16:
        * left  :   (x·W_left_proj , x·W_left_gate)
        * right :   (x·W_right_proj, x·W_right_gate)
        * out‑gate : x·W_out_gate
3️⃣  A tiny Triton elementwise kernel fuses:
        * optional pairwise mask
        * sigmoid on the gate pre‑activations
        * multiplication “proj × gate”.
   The result are the gated tensors `left_h` and `right_h` (FP16).
4️⃣  Tri‑multiplicative update:
        out_h[b,i,j,h] = Σ_k left_h[b,i,k,h] * right_h[b,j,k,h]
   Implemented as a custom batched GEMM kernel that works directly on the
   original layout [B, N, N, H] – no costly permute+contig copies.
5️⃣  Second row‑wise LayerNorm on the hidden tensor.
6️⃣  Elementwise multiplication with the already‑sigmoided `out_gate`.
7️⃣  Final FP16 linear projection back to the model dimension and cast to
   FP32.

Only the reduction (LayerNorm), the “proj × gate” fusion and the
tri‑multiplicative update are written in Triton; all dense linear algebra
uses PyTorch’s highly‑optimised FP16 kernels.  This eliminates the large
memory copies of the reference implementation and brings the forward
pass well below the 1 ms target on an H100."""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # const float*   [rows, dim]
    y_ptr,            #       half*   [rows, dim]  (output)
    w_ptr,            # const float*  [dim]       (γ)
    b_ptr,            # const float*  [dim]       (β)
    rows,             # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- pass 1: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)            # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1/√(var+eps)

    # ---------- pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Elementwise kernel: (proj * mask) * sigmoid(gate)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_mask_kernel(
    proj_ptr,          # const half* [rows, hidden]
    gate_pre_ptr,      # const half* [rows, hidden]
    mask_ptr,          # const half* [rows]   (optional)
    out_ptr,           #       half* [rows, hidden]
    rows,
    hidden,
    MASKED: tl.constexpr,           # 1 → mask is present, 0 → no mask
    BLOCK_R: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid_r = tl.program_id(0)
    pid_c = tl.program_id(1)

    offs_r = pid_r * BLOCK_R + tl.arange(0, BLOCK_R)
    offs_c = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)

    mask_r = offs_r < rows
    mask_c = offs_c < hidden
    active = mask_r[:, None] & mask_c[None, :]

    # load projection and pre‑gate
    proj = tl.load(
        proj_ptr + offs_r[:, None] * hidden + offs_c[None, :],
        mask=active,
        other=0.0,
    ).to(tl.float32)

    gate_pre = tl.load(
        gate_pre_ptr + offs_r[:, None] * hidden + offs_c[None, :],
        mask=active,
        other=0.0,
    ).to(tl.float32)

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))

    # optional mask (binary 0/1)
    if MASKED:
        m = tl.load(
            mask_ptr + offs_r,
            mask=mask_r,
            other=0.0,
        ).to(tl.float32)                # [BLOCK_R]
        m = m[:, None]                   # broadcast over hidden
        out = proj * gate * m
    else:
        out = proj * gate

    out_fp16 = out.to(tl.float16)

    out_ptrs = out_ptr + offs_r[:, None] * hidden + offs_c[None, :]
    tl.store(out_ptrs, out_fp16, mask=active)


# ----------------------------------------------------------------------
# 3️⃣ Batched GEMM for the Tri‑multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batch_matmul_kernel(
    a_ptr,      # left  : const half* [B, N, N, H]
    b_ptr,      # right : const half* [B, N, N, H]
    c_ptr,      # out   :       half* [B, N, N, H]
    B, N, H,
    stride_batch,  # stride to next batch = N*N*H
    stride_i,      # stride for the first spatial axis (i or j) = N*H
    stride_k,      # stride for the reduction axis k = H
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)   # tile row   → i
    pid_n   = tl.program_id(2)   # tile col   → j

    # --------------------------------------------------------------
    # De‑compose batch_id → (batch, hidden)
    # --------------------------------------------------------------
    b_idx = batch_id // H
    h_idx = batch_id % H

    # base pointers for this (b, h)
    a_base = a_ptr + b_idx * stride_batch + h_idx
    b_base = b_ptr + b_idx * stride_batch + h_idx
    c_base = c_ptr + b_idx * stride_batch + h_idx

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 for numerical safety
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: shape (BLOCK_M, BLOCK_K)  ← left[i, k, h]
        a_ptrs = a_base + offs_m[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile: shape (BLOCK_K, BLOCK_N) ← right[k, j, h] (note transpose)
        b_ptrs = b_base + offs_k[:, None] * stride_k + offs_n[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Write result (cast back to FP16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_m[:, None] * stride_i + offs_n[None, :] * stride_k
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 output)
    # --------------------------------------------------------------
    total_rows = B * N * N                      # rows = B·N·N
    x_flat_fp32 = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat_fp32,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # reshape to [B, N, N, dim] (still FP16)
    x_norm = x_norm_fp16.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (proj + gate) – all in FP16
    # --------------------------------------------------------------
    # concatenate proj and gate weights for the left/right paths
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    out_gate_w = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    # flatten once for the three linear ops
    x_norm_flat = x_norm_fp16               # shape [rows, dim]

    left_concat = F.linear(x_norm_flat, left_concat_w)    # [rows, 2*H]
    right_concat = F.linear(x_norm_flat, right_concat_w)
    out_gate_pre = F.linear(x_norm_flat, out_gate_w)     # [rows, H]

    # split concatenated results
    left_proj_pre = left_concat[:, :hidden]
    left_gate_pre = left_concat[:, hidden:]

    right_proj_pre = right_concat[:, :hidden]
    right_gate_pre = right_concat[:, hidden:]

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask + sigmoid‑gate (Triton)
    # --------------------------------------------------------------
    # mask is broadcasted over the hidden dimension
    if not nomask:
        mask_flat = mask_tensor.view(total_rows, 1).to(torch.float16)
    else:
        # dummy tensor – never accessed because MASKED=0
        mask_flat = torch.empty((1,), dtype=torch.float16, device=device)

    # allocate output containers for the gated projections
    left_proj = torch.empty_like(left_proj_pre)
    right_proj = torch.empty_like(right_proj_pre)

    # kernel grid (rows × hidden)
    BLOCK_R = 256
    BLOCK_C = 64
    grid_gate = (
        triton.cdiv(total_rows, BLOCK_R),
        triton.cdiv(hidden, BLOCK_C),
    )

    # left side
    proj_gate_mask_kernel[grid_gate](
        left_proj_pre,
        left_gate_pre,
        mask_flat,
        left_proj,
        total_rows,
        hidden,
        MASKED=0 if nomask else 1,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
    )
    # right side
    proj_gate_mask_kernel[grid_gate](
        right_proj_pre,
        right_gate_pre,
        mask_flat,
        right_proj,
        total_rows,
        hidden,
        MASKED=0 if nomask else 1,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
    )
    torch.cuda.synchronize()

    # out‑gate (sigmoid only, no mask)
    out_gate = torch.sigmoid(out_gate_pre)   # FP16, shape [rows, H]

    # --------------------------------------------------------------
    # 4️⃣ Reshape to [B, N, N, H] (no copy – view only)
    # --------------------------------------------------------------
    left_h  = left_proj.view(B, N, N, hidden)   # FP16
    right_h = right_proj.view(B, N, N, hidden)  # FP16
    out_gate_h = out_gate.view(B, N, N, hidden)  # FP16 (same layout as left/right)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update: batched GEMM in Triton
    # --------------------------------------------------------------
    out_h = torch.empty_like(left_h)          # will hold the hidden result

    # Strides for the 4‑D tensors (contiguous layout)
    stride_batch = N * N * hidden          # distance between consecutive batches (B dimension)
    stride_i = N * hidden                  # stride for i (or j) dimension
    stride_k = hidden                      # stride for the reduction dimension

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    batch_dim = B * hidden

    grid_tri = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_batch_matmul_kernel[grid_tri](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out_h.view(total_rows_h, hidden).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final linear projection
    # --------------------------------------------------------------
    # flatten for elementwise multiplication
    out_norm_flat = out_norm
    out_gate_flat = out_gate_h.view(total_rows_h, hidden)

    gated = out_norm_flat * out_gate_flat          # FP16

    final_h = F.linear(
        gated,
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )                                               # [rows, dim] FP16

    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
