"""
Optimized outgoing TriMul (AlphaFold‑3) using Triton.

Algorithm
---------
1️⃣  Layer‑norm over the last channel (dim) – Triton kernel that
    accumulates in fp32 but stores the normalized result in fp16.
2️⃣  Fuse the 5 linear projections (left/right & three gates) into a
    single GEMM (`F.linear`) by concatenating the weight matrices.
    All work is kept in fp16 (Tensor‑Core friendly).
3️⃣  (Optional) apply the pair‑wise mask to the left/right projections.
4️⃣  Apply sigmoid gates and element‑wise gating (still in fp16).
5️⃣  Triangular multiplicative update:
      out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    realised as a batched matrix‑multiply (`torch.bmm`) on tensors of
    shape (B·H, N, N) in fp16.
6️⃣  Second layer‑norm over the hidden dimension (H) – same Triton kernel.
7️⃣  Multiply by the output gate and project back to `dim` with a final
    linear layer (fp16).  The result is cast to fp32 before returning.
All heavy reductions (the two LayerNorms) are performed by a custom
Triton kernel; dense linear algebra uses cuBLAS (Tensor‑Core) which
is already highly optimized on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton layer‑norm kernel – fp16 output, fp32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut   half*    [total_rows, dim]   (fp16 output)
    w_ptr,          # *const float*   [dim]              (γ)
    b_ptr,          # *const float*   [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: compute row‑wise mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        # (x - mean) * rstd  → fp32
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β are fp32; promote to fp16 right before store
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b            # fp32
        out_fp16 = out_fp32.to(tl.float16) # down‑cast

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)   # True → ignore mask

    # --------------------------------------------------------------
    # Flatten input for first layer‑norm
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                # [B, N, N, C]
    total_rows = B * N * N                         # rows for flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    # View as [total_rows, dim] – keep fp32 for mean/var accumulation
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Choose tiling that covers typical sizes (dim ≤ 384)
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (all in fp16)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, C))
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Single GEMM: [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16 output

    # Split concatenated result
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: [total_rows, H]

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N]  → flatten → [total_rows, 1]
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating
    left_fp16 = left_proj * left_gate
    right_fp16 = right_proj * right_gate

    # ------------------------------------------------------------------
    # Reshape for triangular update
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # left  : (B, N, N, H) → (B, H, N, N)
    # right : (B, N, N, H) → (B, H, N, N) and then transpose k ↔ i
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (k,i)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)       # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)     # [B*H, N, N]

    # Batched matrix‑multiply in fp16 (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)  # [B*H, N, N] (fp16)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden_dim (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)  # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply final output gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate          # fp16 (element‑wise)

    # Final linear projection H → C (fp16)
    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # fp16, shape [total_rows_h, C]

    # Cast back to fp32 as required by the spec
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
