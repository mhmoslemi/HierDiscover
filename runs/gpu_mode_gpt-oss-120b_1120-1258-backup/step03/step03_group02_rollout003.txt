"""
TriMul “outgoing” implementation (AlphaFold‑3 style) accelerated with Triton.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor [B, N, N, C] (FP32 → FP16).
2️⃣  Fuse the projection and gate linear layers for the left and right side:
    a single FP16 GEMM each yields `proj` and `gate_pre` (2·H columns).
3️⃣  Optional pair‑wise mask (broadcast over the hidden dimension) – applied
    in‑place to the two projected tensors.
4️⃣  Apply sigmoid to the gate pre‑activations and multiply with the projected
    tensors (still FP16).
5️⃣  Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
    Realised as a batched GEMM (B·H matrices of size N×N) with `torch.bmm`.
6️⃣  Second LayerNorm on the hidden representation (FP16 output).
7️⃣  Apply the output‑gate (sigmoid) and a final FP16 linear projection back
    to the original channel dimension.  The final result is cast to FP32.

Only the two LayerNorms are written in Triton; all dense linear algebra is
performed with PyTorch’s highly‑optimised FP16 kernels.  The function works
with or without a mask and supports the shapes listed in the problem
statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]   (input FP32)
    y_ptr,          #       half*   [total_rows, dim]   (output FP16)
    w_ptr,          # const float*  [dim]               (γ)
    b_ptr,          # const float*  [dim]               (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns (C or hidden_dim)
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Compute LayerNorm for each row of a 2‑D tensor stored in row‑major order.
    The kernel works in FP32 internally and writes the normalized result as
    FP16 to reduce memory traffic.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean and variance of each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_iters_d = tl.cdiv(dim, BLOCK_D)
    for d in range(0, num_iters_d):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer arithmetic: row * dim + col
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values (γ·x + β) as FP16
    # ------------------------------------------------------------------
    for d in range(0, num_iters_d):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # load γ and β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]      # FP32
        out_fp16 = out.to(tl.float16)            # cast to FP16

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

            - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights_dict : dict with all linear / LN weights & biases

    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    total_rows = B * N * N                    # rows when flattening (B·N²)

    # 1️⃣ First LayerNorm (FP32 → FP16) using Triton
    # --------------------------------------------------------------

    BLOCK_D = 64                      # columns per inner loop (covers dim up to 384)

        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    # Restore the 4‑D view (still FP16)

    # --------------------------------------------------------------

        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0

        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0

    x_norm_flat = x_norm_h.view(total_rows, dim)          # [rows, dim]

    left_gate_pre = left_concat[:, hidden_dim:]                # [rows, H]
    right_proj_h  = right_concat[:, :hidden_dim]
    right_gate_pre = right_concat[:, hidden_dim:]

    # Output‑gate (single linear, same for whole tensor)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_flat, weights["out_gate.weight"].to(torch.float16))
    )   # [rows, H]   (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten → [rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre)      # [rows, H]
    right_gate_h = torch.sigmoid(right_gate_pre)

    left_h  = left_proj_h * left_gate_h              # [rows, H]
    right_h = right_proj_h * right_gate_h

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to 4‑D tensors
    left  = left_h.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right_h.view(B, N, N, hidden_dim)

    # Move hidden dimension to batch axis for the batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)          # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)         # [BH, N, N]

    # out = left @ rightᵀ  (sum over k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # Reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16 → FP16) using Triton
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, hidden_dim)   # FP16
    out_norm = out_norm * out_gate                    # element‑wise gating

    final_h = F.linear(
        out_norm.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows_h, dim]  (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32
    return final
```
