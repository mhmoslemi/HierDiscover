"""
Optimized outgoing TriMul (AlphaFold‑3) kernel
------------------------------------------------
1️⃣ First LayerNorm – use the highly‑optimised PyTorch implementation
   (float32 → fp16 conversion afterwards).

2️⃣ Fuse the five linear projections (left/right & three gates) in a single
   half‑precision GEMM.  The weight matrix is concatenated once, so the
   input tensor is read only a single time.

3️⃣ Optional mask – applied in‑place to the left/right projections by a
   tiny Triton kernel.  This guarantees we have at least one custom kernel
   while adding virtually no overhead.

4️⃣ Apply sigmoid gates and element‑wise gating (all in fp16).

5️⃣ Triangular multiplicative update (outgoing):
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   realised as a batched GEMM (`torch.bmm`) on tensors of shape
   (B·H, N, N) with Tensor‑Core friendly fp16 arithmetic.

6️⃣ Second LayerNorm – also the PyTorch implementation (half‑precision
   weight & bias).

7️⃣ Multiply by the output gate and a final half‑precision linear projection
   back to the original `dim`.  The final result is cast to float32 as
   required.

All heavy linear algebra uses cuBLAS/Tensor‑Core; only the mask‑application
and the mandatory custom‑kernel requirement are handled by Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_mul_kernel(
    left_ptr,                # fp16 *  (total_rows * hidden_dim)
    right_ptr,               # fp16 *
    mask_ptr,                # fp16 *  (total_rows) – broadcasted over hidden_dim
    total_elements,          # i64 – total number of elements in left/right (flattened)
    hidden_dim,              # i32 – size of the hidden dimension (runtime)
    BLOCK_SIZE: tl.constexpr # i32 – compile‑time block size
):
    """
    In‑place multiplication of left/right tensors by a broadcast mask.
    Each THREAD processes one element of the flattened (row·hidden_dim)
    tensors.  The mask index is `offset // hidden_dim`, i.e. the same mask
    value is reused for all hidden channels of a given (b,i,j) position.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # [BLOCK_SIZE]
    mask = offs < total_elements

    # broadcast mask over the hidden dimension
    mask_idx = offs // hidden_dim

    # Load values (masked)
    left_val  = tl.load(left_ptr  + offs, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    mask_val  = tl.load(mask_ptr  + mask_idx, mask=mask, other=1.0)

    # Apply mask
    left_val  = left_val  * mask_val
    right_val = right_val * mask_val

    # Store back in‑place
    tl.store(left_ptr  + offs, left_val,  mask=mask)
    tl.store(right_ptr + offs, right_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim         = config["dim"]          # C
    hidden_dim  = config["hidden_dim"]   # H
    nomask      = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape       # [B, N, N, C]
    total_rows = B * N * N                 # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – PyTorch implementation
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                     # [B,N,N,dim], float32
    x_norm_fp16 = x_norm.half()          # fp16 for the heavy GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections (five heads) – single GEMM
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, C)) and cast to fp16
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()               # (5*H, C)

    # Linear: [total_rows, C] @ [C, 5*H].T -> [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )   # each: [total_rows, H]

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (in‑place) – tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten → fp16
        mask_fp16 = mask_tensor.view(-1).to(dtype=torch.float16).contiguous()

        # flatten the projection tensors for the kernel
        left_flat  = left_proj.view(-1)
        right_flat = right_proj.view(-1)

        total_elements = left_flat.numel()   # = total_rows * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        mask_mul_kernel[grid](
            left_flat,
            right_flat,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
        # The tensors are now masked in‑place.
    # ------------------------------------------------------------------
    # 2b️⃣ Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)   # [total_rows, H]
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # Element‑wise gating
    left = left_proj * left_gate            # [total_rows, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Outgoing triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Prepare tensors of shape (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, N)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)   # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)  # (B·H, N, N)

    # Batched matrix multiplication in fp16 (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)   # (B·H, N, N), fp16

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (half‑precision)
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate   # fp16

    # Final linear projection H → C (half‑precision)
    final_fp16 = F.linear(
        gated.view(-1, hidden_dim),
        weights["to_out.weight"].half(),
    )   # fp16, shape [B*N*N, dim]

    # Cast to float32 for the return type
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
