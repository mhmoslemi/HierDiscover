4️⃣ Apply sigmoid to the three gate tensors and gate the left/right projections.

5️⃣ Triangular multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]     (batched GEMM)
   This is performed as a batched fp16 matrix multiply (Tensor‑Core).

6️⃣ Fuse the second LayerNorm (over the hidden dimension H) with the final
   output‑gate multiplication in a single Triton kernel, producing a tensor that
   can be fed directly to the final linear projection.

7️⃣ Linear projection back to the original channel dimension C (fp16 GEMM) and
   cast the result to float32.

All heavy reductions (the two LayerNorms) run in custom Triton kernels while
dense linear algebra uses cuBLAS/Tensor‑Core via ``torch.nn.functional.linear`` and
``torch.bmm``.  The function ``custom_kernel`` follows the required signature:
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
and returns a float32 tensor of shape ``[B, N, N, C]``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # *const float*   [total_rows, dim] (input fp32)
    y_ptr,            # *mut   half*    [total_rows, dim] (output fp16)
    weight_ptr,       # *const float*   [dim]           (γ)
    bias_ptr,         # *const float*   [dim]           (β)
    total_rows,       # int (runtime)
    dim: tl.constexpr,      # int (compile‑time)
    eps: tl.constexpr,      # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per iteration
):
    """Row‑wise LayerNorm that writes fp16 output."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                     # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------------- first pass: mean & variance -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)               # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)    # 1 / sqrt(var+eps)

    # ----------------- second pass: write normalized value -------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # [BLOCK_M, BLOCK_D]

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0) # β

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]

        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,            # *const half*   [total_rows, dim]  (pre‑norm tensor)
    gate_ptr,         # *const half*   [total_rows, dim]  (output gate)
    y_ptr,            # *mut   half*   [total_rows, dim]  (LN·γ·β·gate)
    weight_ptr,       # *const float*  [dim]              (γ of 2nd LN)
    bias_ptr,         # *const float*  [dim]              (β of 2nd LN)
    total_rows,       # int (runtime)
    dim: tl.constexpr,      # int (compile‑time)
    eps: tl.constexpr,      # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """LayerNorm + output‑gate (γ·x̂+β)·gate, writes fp16."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                     # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------- mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------- write LN·gate ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)           # pre‑norm
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)           # out‑gate

        norm = (x - mean[:, None]) * rstd[:, None]            # x̂

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g              # (γ·x̂+β)·gate

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. First LayerNorm (over C) → fp16 output
    # --------------------------------------------------------------
    B, N, _, C = input_tensor.shape   # C == dim
    total_rows = B * N * N

    # reshape to 2‑D for the kernel
    x_flat = input_tensor.contiguous().view(total_rows, dim)   # fp32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    BLOCK_M = 256   # rows per program (tuned for H100)
    BLOCK_D = 32    # cols per iteration (covers dim up to 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 2. Fused linear projections + gates (all in fp16)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, C))
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Single GEMM: [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # --------------------------------------------------------------
    # 2a. Optional mask (skip if nomask=True)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(dtype=torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b. Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)      # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # reshape back to 4‑D tensors
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)       # used later

    # --------------------------------------------------------------
    # 3. Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Rearrange to (B*H, N, N) matrices: left[i,k] and right[k,j]
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, i, k)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N) # (B*H, k, j)

    out_mat = torch.bmm(left_mat, right_mat)   # fp16, shape (B*H, N, N)

    # back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4. Fuse second LayerNorm (over H) with the final output‑gate
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)                # [total_rows_h, H]
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)      # same shape

    fused_fp16 = torch.empty_like(out_flat)                      # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_gate_flat,
        fused_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 5. Final linear projection back to C and cast to float32
    # --------------------------------------------------------------
    to_out_weight_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(fused_fp16, to_out_weight_fp16)   # [total_rows_h, C] fp16

    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
