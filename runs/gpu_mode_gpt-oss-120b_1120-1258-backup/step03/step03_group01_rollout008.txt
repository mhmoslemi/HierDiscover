"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input pair‑wise tensor (`dim` channels).
   A custom Triton kernel computes mean / var in fp32 and writes the
   normalized result directly in fp16.

2️⃣ Fuse the five dense layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a *single* half‑precision GEMM:
   `[R, dim] @ [5*hidden, dim].T → [R, 5*hidden]` where `R = B·N·N`.
   Simple slicing yields the three projected tensors and the three
   gate vectors.

3️⃣ Optional binary mask (shape `[B,N,N]`) is applied only to the two
   projected tensors.  Sigmoid gates are computed in fp16.

4️⃣ Triangular multiplicative update:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is a batched matrix‑multiply per hidden channel:
   `left_perm  = left .permute(0,3,1,2)   # [B,H,N,N] (i×k)`
   `right_perm = right.permute(0,3,2,1)  # [B,H,k,j]`
   `out_perm   = torch.matmul(left_perm, right_perm)   # [B,H,i,j]`
   `out        = out_perm.permute(0,2,3,1)` (back to `[B,N,N,H]`).

5️⃣ Second LayerNorm over the hidden dimension (`hidden_dim`) – also a
   Triton kernel (fp16 → fp16).

6️⃣ Element‑wise multiplication by the output gate and a final
   half‑precision linear projection back to `dim`.  The result is cast
   to `float32` before returning.

All heavy matrix multiplies are performed in fp16 to exploit Tensor‑Core
throughput on an H100.  Only the two LayerNorms are implemented in
custom Triton code; the rest uses highly‑optimized PyTorch/CuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 input → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   *  [total_rows, dim]   (float32 or float16)
    y_ptr,            # mutable *  [total_rows, dim]   (fp16 output)
    weight_ptr,       # const   *  [dim]               (float32)
    bias_ptr,         # const   *  [dim]               (float32)
    total_rows,       # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass: accumulate sum and sum‑of‑squares
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: write normalized + affine (γ, β)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype=torch.float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    apply_mask = not config.get("nomask", True)   # False → no mask

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                       # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) → fp16 output
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)                    # [R, dim] fp16

    BLOCK_M = 256
    BLOCK_D = 128   # dim is at most 384 → 128 works well
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse the five linear layers into a single FP16 GEMM
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # Linear: [R, dim] @ [5*H, dim].T → [R, 5*H]
    proj_all = F.linear(x_norm, w_concat)   # fp16

    H = hidden_dim
    left_proj   = proj_all[:, 0:H]
    right_proj  = proj_all[:, H:2*H]
    left_gate_p = proj_all[:, 2*H:3*H]
    right_gate_p= proj_all[:, 3*H:4*H]
    out_gate_p  = proj_all[:, 4*H:5*H]

    # Sigmoid gates (fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 3) Optional mask (only on the two projected tensors)
    # --------------------------------------------------------------
    if apply_mask:
        # mask shape [B,N,N] → flatten → [R,1]
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Element‑wise gating
    left  = left_proj * left_gate   # [R, H] fp16
    right = right_proj * right_gate # [R, H] fp16

    # --------------------------------------------------------------
    # 4) Reshape back to 4‑D tensors
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5) Triangular multiplicative update (batched matmul)
    # --------------------------------------------------------------
    # left: [B,N,N,H] → [B,H,N,N]   (i × k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: [B,N,N,H] → [B,H,N,N] where axes are (k × j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # Batched matmul over the hidden dimension
    out_perm = torch.matmul(left_perm, right_perm)          # [B,H,N,N]
    # Back to [B,N,N,H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()        # fp16

    # --------------------------------------------------------------
    # 6) Second LayerNorm over hidden_dim (still fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()        # [R, H]
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, H)                   # fp16

    # --------------------------------------------------------------
    # 7) Apply output gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                         # fp16
    out_norm_flat = out_norm.view(total_rows, H).contiguous()

    # Final projection: weight shape [dim, H]
    final_half = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16),  # [dim, H]
    )                                                  # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```

The implementation follows the reference PyTorch `TriMul` module but replaces the costly Python‑level loops with:

* Two Triton‑based LayerNorm kernels (row‑wise, no auxiliary memory).
* A single fused half‑precision GEMM for the five linear layers.
* A batched matrix‑multiply (`torch.matmul`) that replaces the explicit `einsum`.
* All remaining operations are performed in fp16, with a final cast to fp32.

This satisfies the requirement of having custom Triton kernels while delivering a fast forward pass suitable for the test configurations.
