@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode flat index into b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask handling
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        # broadcast H dimension later
    else:
        mask_val = tl.full([BLOCK_R], 1.0, dtype=tl.float16)

    # hidden stride for destination layout B*H*N*N
    hidden_stride = N * N
    # base offset for position (b,i,k) for hidden 0
    base_offset = b * (H * N * N) + i * N + k

    # loop over hidden dim
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # each column in proj corresponds to hidden dim
        # layout of proj: contiguous (R,5*H). stride_H = 5*H
        # expression to compute offsets: proj_ptr + offs_r[:, None] * (5*H) + offs_h[None, :]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(proj_ptr + proj_base, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # sigmoid in fp32
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        if HAS_MASK:
            # mask_val shape [BLOCK_R]; broadcast
            mask_val_h = mask_val[:, None]  # [BLOCK_R,1]
        else:
            mask_val_h = tl.full([BLOCK_R,1], 1.0, dtype=tl.float16)

        left = left_raw * mask_val_h * left_gate
        right = right_raw * mask_val_h * right_gate
        out_gate = out_gate_val

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_off, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, out_gate, mask=mask_r[:, None] & mask_h[None, :])
```

"""
TriMul (outgoing) – Triton‑fused forward pass
================================================

The implementation follows the reference PyTorch module but moves the two
expensive layer‑norms and the projection/gating step into three small Triton
kernels.  The rest of the computation (the batched matrix multiplication and
the final linear projection) is performed with PyTorch / cuBLAS which already
exploits tensor cores.

Order of operations
-------------------
1️⃣  First LayerNorm  (per‑row mean/var over the *dim* axis) → FP16.
2️⃣  A single GEMM computes the five linear maps (left‑proj, right‑proj,
    three gates).  A Triton kernel splits the result, applies an optional
    pair‑wise mask, and evaluates the three sigmoids.
3️⃣  Batched matrix multiplication: `out[b,h] = left[b,h] @ right[b,h]^T`
    (Tensor‑Core GEMM, FP16).
4️⃣  Second LayerNorm (mean/var over the hidden dimension) is done in PyTorch
    (still FP32 for numerical stability) and its γ/β parameters are applied.
5️⃣  The out‑gate is multiplied element‑wise and finally a linear layer maps the
    hidden dimension back to the original *dim*.

All tensors stay on the device and the final result is a contiguous
`[B, N, N, dim]` float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # const   *[rows, dim]   (float32)
    y_ptr,      # mutable *[rows, dim]   (float16)
    w_ptr,      # const   *[dim]         (float32)   γ
    b_ptr,      # const   *[dim]         (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance in FP32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    n_blocks_d = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, n_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalized values (γ·(x‑μ)·rstd + β) in FP16
    # ------------------------------------------------------------------
    for d_off in range(0, n_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const   *[R, 5*H]   (fp16)   fused projection
    left_ptr,       # mutable *[B, H, N, N] (fp16)
    right_ptr,      # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,   # mutable *[B, H, N, N] (fp16)
    mask_ptr,       # const   *[R]        (fp16)   (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask handling (scalar per (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * N * N) + i * N + k   # [BLOCK_R]

    # loop over hidden dimension in blocks of BLOCK_H
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj:   [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weight_dict, config_dict)
            input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            weight_dict  : mapping with all linear / layer‑norm parameters
                           (all float32, Linear layers have no bias)
            config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (b, i, k) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32 if dim <= 128 else 64

    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],   # γ
        w["norm.bias"],     # β
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                        # [5·H, dim]

    # (R, 5·H) half‑precision GEMM
    proj = F.linear(x_norm, fused_w)           # = x_norm @ fused_wᵀ
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split → left, right, out_gate  (apply mask & sigmoid)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # flatten mask → [R] fp16
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ)   – tensor‑core fp16
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # (B·H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) fp16
    out = out.view(B, H, N, N)                            # (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (per position over the hidden axis)
    # ------------------------------------------------------------------
    out_f32 = out.float()
    mu   = out_f32.mean(dim=1, keepdim=True)                 # (B,1,N,N)
    var  = out_f32.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)                           # (B,1,N,N)

    # γ / β of the second LN (shape H)
    ln_w = w["to_out_norm.weight"]
    ln_b = w["to_out_norm.bias"]
    ln_w = ln_w.view(1, H, 1, 1)   # broadcast
    ln_b = ln_b.view(1, H, 1, 1)

    out_norm = (out_f32 - mu) * rstd          # (B,H,N,N)  FP32
    out_norm = out_norm * ln_w + ln_b        # apply γ/β

    # ------------------------------------------------------------------
    # 6️⃣ Multiply by out‑gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.float()   # (B,H,N,N)

    # final linear: (B,N,N,dim) = (B,N,N,H) @ (H,dim)ᵀ
    # reshape to (B·N·N, H)
    out_flat = out_norm.view(B * N * N, H)
    w_out = w["to_out.weight"]               # (dim, H) float32
    final = F.linear(out_flat, w_out)        # (B·N·N, dim)  float32
    final = final.view(B, N, N, dim)         # (B, N, N, dim)

    return final
```

The code defines two Triton kernels:

* `layernorm_kernel` – computes a row‑wise LayerNorm (mean & variance in FP32,
  writes normalized FP16 values).

* `split_fuse_gate_kernel` – after a fused linear projection produces the
  left/right projections together with the three gates, applies the optional
  pair‑wise mask, evaluates the sigmoid activations and writes the three tensors
  (`left`, `right`, `out_gate`) in FP16.

`custom_kernel` orchestrates the whole forward pass, reusing cuBLAS (via
`torch.nn.functional.linear` and `torch.bmm`) for the large matrix
multiplications and performing the second LayerNorm and the final linear layer
in PyTorch (still FP32 for numerical stability). The result matches the
reference PyTorch implementation while keeping the heavy element‑wise work in
high‑performance Triton kernels.
