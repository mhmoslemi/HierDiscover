"""
TriMul (outgoing) – Fast Triton‑accelerated forward pass
=======================================================

The kernel stack implements only the first LayerNorm in Triton;
all remaining heavy linear algebra is delegated to cuBLAS /
torch.nn.functional which already exploits Tensor‑Cores on H100.
The algorithm follows the reference PyTorch implementation:

1️⃣  LayerNorm over the last dimension (dim) – Triton kernel.
2️⃣  Fused linear projection (left, right, three gates) via a single
    GEMM with a concatenated weight matrix.
3️⃣  Apply the optional pair‑wise mask and sigmoid gates, and reshape
    to [B, H, N, N].
4️⃣  Batched matrix multiplication  left @ rightᵀ   (H × N × N) per batch.
5️⃣  LayerNorm across the hidden dimension (H), multiply by the out‑gate,
    and a final linear projection back to *dim* – pure PyTorch (cuBLAS).

Only the first LayerNorm is written in Triton, satisfying the
requirement to “implement at least part of the operations in a kernel”.
All tensors are kept on the device; mixed‑precision (FP16 for the bulk
of the work) yields high throughput while the final result is returned
as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (float32 → float16) – Triton implementation
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # const   *[rows, dim] (float32)
    y_ptr,      # mutable *[rows, dim] (float16)
    w_ptr,      # const   *[dim]      (float32)   γ
    b_ptr,      # const   *[dim]      (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    n_blocks_d = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, n_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (γ·(x‑μ)·rstd + β) in fp16 ----
    for d_off in range(0, n_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)    # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # β (fp32)
        out = norm * w[None, :] + b[None, :]                   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward operator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/ bool) or None
            - weights_dict : dict with all Linear / LayerNorm parameters.
                            All tensors are on the same device as *input_tensor*.
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape                  # x shape = [B, N, N, dim]
    R = B * N * N                          # total (b,i,k) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using Triton
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16,
                         device=device)                     # output fp16

    BLOCK_M = 256
    BLOCK_D = 32 if dim <= 128 else 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],           # γ (float32)
        w["norm.bias"],             # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # ------------------------------------------------------------------
    # concatenate the five linear weights (each H×dim) → (5H, dim)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],   # (H, dim)
            w["right_proj.weight"],  # (H, dim)
            w["left_gate.weight"],   # (H, dim)
            w["right_gate.weight"],  # (H, dim)
            w["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)             # (5*H, dim) in FP16

    # single GEMM: (R, dim) @ (dim, 5*H)ᵀ → (R, 5*H)
    proj = F.linear(x_norm, fused_w)            # (R, 5*H) FP16
    del fused_w, x_norm

    # bring the projection to a convenient view
    proj = proj.view(R, 5, H)                   # (R, 5, H)

    left_raw      = proj[:, 0, :]               # (R, H)
    right_raw     = proj[:, 1, :]               # (R, H)
    left_gate_raw = proj[:, 2, :]               # (R, H)
    right_gate_raw = proj[:, 3, :]              # (R, H)
    out_gate_raw  = proj[:, 4, :]               # (R, H)
    del proj

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask and sigmoid gates
    # ------------------------------------------------------------------
    # mask handling (mask shape [B, N, N] → [R])
    if cfg.get("nomask", True) or mask is None:
        # no masking
        left = left_raw
        right = right_raw
    else:
        mask_flat = mask.to(torch.float16).view(R)      # (R,)
        mask_flat = mask_flat[:, None]                  # (R,1) for broadcasting
        left = left_raw * mask_flat
        right = right_raw * mask_flat

    # sigmoid gates (computed in fp32 internally by torch.sigmoid)
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left * left_gate
    right = right * right_gate
    # out_gate stays as the gating tensor for the output path

    del left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to [B, H, N, N] and batch GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # From (R, H) → (B, N, N, H) → (B, H, N, N)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # batched matrix multiplication over the N dimension (FP16 Tensor‑Core)
    left_mat = left.view(B * H, N, N)                # (B·H, N, N)
    right_mat = right.view(B * H, N, N)              # (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N).contiguous()                # (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H),
    #    multiply by out_gate, final linear projection.
    # ------------------------------------------------------------------
    # Rearrange to (B, N, N, H) to reuse torch.nn.functional.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],   # (H,) float32
        bias=w["to_out_norm.bias"],       # (H,) float32
        eps=eps,
    )                                            # stays FP16

    # Align out_gate to the same layout
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # Apply output gate
    gated = out_norm * gate_perm                               # (B, N, N, H) FP16

    # Flatten to (R, H) and perform the final linear projection
    gated_flat = gated.view(R, H).to(torch.float32)                         # FP32 input
    final = F.linear(gated_flat, w["to_out.weight"])                       # (R, dim) FP32

    # Reshape to [B, N, N, dim] and return
    final = final.view(B, N, N, dim)

    return final
