@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows: tl.int32, H: tl.int32, N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)

"""
custom_tri_mul.py

TriMul (outgoing) – Triton‑accelerated forward pass
--------------------------------------------------

This implementation follows the “outgoing” version of the TriMul operator
described in the AlphaFold‑3 paper.  The forward pass is split into six
stages:

1️⃣  Layer‑norm over the last dimension (FP32 → FP16) using Torch.
2️⃣  Five linear projections (left, right, left‑gate, right‑gate, out‑gate)
    are performed with a half‑precision matmul; gates are passed through
    a sigmoid.
3️⃣  Optional mask is applied and the gated tensors are multiplied.
4️⃣  Batched half‑precision GEMM (`bmm`) computes the contraction

       out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]

    producing a tensor of shape ``[B, H, N, N]`` (``H = hidden_dim``).
5️⃣  A Triton kernel computes the per‑position mean and rsqrt‑variance
    of ``out`` across the hidden dimension (second LayerNorm statistics).
6️⃣  A second Triton kernel fuses the second LayerNorm, the out‑gate,
    and the final linear projection (``hidden_dim → dim``).  The result
    is written in FP32 as required by the model.

Only the statistics and the final projection are performed in Triton,
while the heavier linear layers and the batch GEMM use Torch’s highly
optimised kernels.  This hybrid approach yields a correct implementation
and a fast forward pass on modern GPUs (e.g. NVIDIA H100).

The entry point is ``custom_kernel(data)`` where ``data`` is a tuple
``(input, mask, weights, config)`` as described in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean / rsqrt‑variance of the contracted tensor (second LN)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # FP16   [B, H, N, N]
    mu_ptr,           # FP32   [R]       (R = B*N*N)
    rstd_ptr,         # FP32   [R]
    total_rows: tl.int32,   # R = B*N*N
    H: tl.int32,            # hidden dimension
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_K: tl.constexpr,  # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode (b,i,j) from a flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N                     # distance between two hidden planes
    base_offset = b * (H * N * N) + i * N + j  # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # accumulate over hidden dimension
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [M, K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Final LayerNorm + out_gate + projection (fused)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # FP16   [B, H, N, N]   (contracted tensor)
    gate_ptr,         # FP16   [B, H, N, N]   (out_gate)
    mu_ptr,           # FP32   [R]
    rstd_ptr,         # FP32   [R]
    ln_w_ptr,         # FP32   [H]   (to_out_norm.weight)
    ln_b_ptr,         # FP32   [H]   (to_out_norm.bias)
    w_out_T_ptr,      # FP16   [H, dim]  (to_out weight transposed)
    out_final_ptr,    # FP32   [B, N, N, dim]  (final output)
    total_rows: tl.int32,   # R = B*N*N
    H: tl.int32,
    N: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
    BLOCK_N: tl.constexpr,   # output‑dim tile
):
    pid_m = tl.program_id(0)   # position tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start of hidden‑plane 0

    # load per‑position statistics (μ, rstd) – FP32
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # iterate over hidden dimension in tiles
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor and out‑gate (FP16 → FP32)
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # second LayerNorm parameters (γ,β) – FP32 per hidden unit
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)

        # (out - μ) * rstd → normalized, then apply γ,β, then out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]
        term = norm * gate_vals                                    # [M, K]

        # cast to fp16 for the final matmul with w_out_T
        term_fp16 = term.to(tl.float16)

        # load a tile of the final projection matrix (K × dim)
        w_tile = tl.load(w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, dim_tile]

        # accumulate dot product
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final result (B, N, N, dim) – FP32
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over last dim (FP32 → FP16) – using torch for simplicity
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm expects weight/bias to be 1‑D tensors
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Five linear projections (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # cast weights to half‑precision for the matmuls
    left_proj_w   = w["left_proj.weight"].to(torch.float16)
    right_proj_w  = w["right_proj.weight"].to(torch.float16)
    left_gate_w   = w["left_gate.weight"].to(torch.float16)
    right_gate_w  = w["right_gate.weight"].to(torch.float16)
    out_gate_w    = w["out_gate.weight"].to(torch.float16)

    # projections (no bias)
    left_proj   = F.linear(x_norm, left_proj_w)          # [B,N,N,H] FP16
    right_proj  = F.linear(x_norm, right_proj_w)         # [B,N,N,H] FP16
    left_gate   = torch.sigmoid(F.linear(x_norm, left_gate_w))   # FP16
    right_gate  = torch.sigmoid(F.linear(x_norm, right_gate_w))  # FP16
    out_gate    = torch.sigmoid(F.linear(x_norm, out_gate_w))    # FP16

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask and gates
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask provided as bool / float; broadcast over hidden dim
        assert mask is not None, "mask must be supplied when nomask=False"
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
    else:
        mask_fp16 = torch.tensor(1.0, dtype=torch.float16,
                                 device=device).view(1, 1, 1, 1)

    left  = left_proj  * left_gate  * mask_fp16
    right = right_proj * right_gate * mask_fp16
    # out_gate already contains sigmoid; no mask applied
    out_gate = out_gate

    # free temporary tensors early
    del left_proj, right_proj, left_gate, right_gate, mask_fp16, x_norm

    # ------------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM (contract over the second sequence dim)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] and make memory layout contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    out_gate_perm = out_gate.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # flatten batch & hidden dim for torch.bmm
    left_flat  = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H,N,N) FP16
    out = out_flat.view(B, H, N, N).contiguous()                  # [B, H, N, N] FP16

    # clean up
    del left, right, left_perm, right_perm, left_flat, right_flat, out_flat

    # ------------------------------------------------------------------
    # 5️⃣ Compute per‑position mean / rsqrt‑variance across hidden dim
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 128
    BLOCK_K_STATS = 32
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)

    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    # to_out_norm parameters are in FP32
    ln_w = w["to_out_norm.weight"]   # shape [H] FP32
    ln_b = w["to_out_norm.bias"]     # shape [H] FP32

    # final linear weight: [dim, hidden_dim] → transpose to [H, dim] in FP16
    w_out_T = w["to_out.weight"].t().contiguous().to(torch.float16)   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64   # rows per program (positions)
    BLOCK_K_OUT = 32   # hidden‑dim tiles
    BLOCK_N_OUT = 64   # output‑dim tiles
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))

    final_output_kernel[grid_out](
        out,                  # contracted tensor
        out_gate_perm,        # out_gate
        mu,
        rstd,
        ln_w,
        ln_b,
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    # clean up intermediate buffers
    del out, out_gate_perm, mu, rstd, w_out_T

    return final
