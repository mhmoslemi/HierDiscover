@triton.jit
def rowwise_stats_contig_kernel(out_ptr, mu_ptr, rstd_ptr, rows, hidden_dim, eps, BLOCK_M, BLOCK_K):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], tl.float32)
    sum_sq = tl.zeros([BLOCK_M], tl.float32)

    for k in range(0, hidden_dim, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        ptr = out_ptr + offs_m[:, None] * hidden_dim + offs_k[None,:]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

@triton.jit
def compute_term_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    gamma_ptr, beta_ptr,
    term_ptr,
    rows, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # load mu and rstd per row
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)  # [M]

    for k in range(0, hidden_dim, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim
        ptr_out = out_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :]
        ptr_gate = gate_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :]

        out_vals = tl.load(ptr_out, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)  # [M,K]
        gate_vals = tl.load(ptr_gate, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # layernorm of out
        # diff = (out_vals - mu[:, None]) * rstd[:, None]   # [M, K]
        diff = (out_vals - mu[:, None]) * rstd[:, None]
        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta = tl.load(beta_ptr + offs_k, mask=mask_k, other=0.0)

        norm = diff * gamma[None, :] + beta[None, :]
        term = norm * gate_vals   # [M,K]
        # store term as fp16
        tl.store(term_ptr + offs_m[:, None] * hidden_dim + offs_k[None,:],
                 term.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])

## layernorm_kernel detailed:

```
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    w_ptr, b_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # first pass: compute sum and sumsq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d in range(num_d_blocks):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: normalize and apply gamma/beta
    for d in range(num_d_blocks):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :], out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])
```

Works.

"""
Outgoing TriMul (AlphaFold3) – fast forward pass (no backward).

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the flattened input (R = B·N·N rows,
   dim features) in FP32 → store FP16.

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single FP16 GEMM:
        proj = x_norm @ W_fusedᵀ   → (R, 5·H)

3️⃣ Split the fused projection, apply the optional pair‑wise mask,
   compute sigmoid gates and store three tensors `left`, `right`,
   `out_gate` in layout [B, H, N, N] (FP16).

4️⃣ Batched half‑precision GEMM (cuBLAS):
        out = left_mat × right_matᵀ   →   [B, H, N, N] (FP16)

5️⃣ Permute `out` & `out_gate` to [B, N, N, H] (hidden dim contiguous).
   Compute per‑row (i.e. per (b,i,j)) mean & rsqrt‑variance over H
   (row‑wise stats kernel).

6️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the
   final projection (to_out) into two kernels:
   • `compute_term_kernel` builds the normalized tensor
        term = LN(out) * out_gate    (FP16, shape R×H)
   • Final projection is a large dense matmul performed by cuBLAS:
        out_final = term @ to_out.weightᵀ   (FP32)

The whole pipeline avoids any Python‑side loops and keeps memory traffic
low by using FP16 wherever possible.  All kernels are written for
Triton 3.3.1 and target an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # [R, dim]  input FP32
    y_ptr,           # [R, dim]  output FP16
    w_ptr,           # [dim]    gamma (FP32)
    b_ptr,           # [dim]    beta  (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [M, D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [M, D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  fp16
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R] fp16 (ignored when HAS_MASK==0)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional pair‑wise mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + k   # start of hidden plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
#    (input layout = [R, H] – hidden dimension contiguous)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_contig_kernel(
    out_ptr,          # [R, H] fp16
    mu_ptr,           # [R]   fp32
    rstd_ptr,         # [R]   fp32
    rows, hidden_dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        ptr = out_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate multiplication → term tensor
#    (output layout = [R, H] fp16)
# ----------------------------------------------------------------------
@triton.jit
def compute_term_kernel(
    out_ptr,          # [R, H] fp16   (pre‑norm tensor)
    gate_ptr,         # [R, H] fp16   (out_gate)
    mu_ptr,           # [R]   fp32
    rstd_ptr,         # [R]   fp32
    gamma_ptr,        # [H]   fp32  (to_out_norm.weight)
    beta_ptr,         # [H]   fp32  (to_out_norm.bias)
    term_ptr,         # [R, H] fp16 (output)
    rows, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    for k_off in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        # load raw out and gate values (FP16 → FP32)
        out_vals  = tl.load(out_ptr  + offs_m[:, None] * hidden_dim + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # second LayerNorm (γ,β) in FP32
        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(beta_ptr  + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → LN → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * gate_vals                                        # [M, K]

        tl.store(term_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
                 term.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : contains
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise positions (rows)

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [5·H, dim]

    proj = F.linear(x_norm, fused_w)    # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * hidden_dim, N, N)                 # [B·H, N, N]
    right_mat = right.view(B * hidden_dim, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))        # [B·H, N, N] FP16
    out = out.view(B, hidden_dim, N, N)                         # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (contiguous H)
    # ------------------------------------------------------------------
    # permute to make hidden dimension contiguous: [B, N, N, H]
    out_cont = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    out_gate_cont = out_gate.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    del out, out_gate

    out_flat = out_cont.view(R, hidden_dim)          # [R, H] fp16
    gate_flat = out_gate_cont.view(R, hidden_dim)    # [R, H] fp16
    del out_cont, out_gate_cont

    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 32
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_contig_kernel[grid_stats](
        out_flat,
        mu,
        rstd,
        R,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate → term tensor (R, H)
    # ------------------------------------------------------------------
    term_flat = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)

    BLOCK_M_TERM = 256
    BLOCK_K_TERM = 32
    grid_term = (triton.cdiv(R, BLOCK_M_TERM),)
    compute_term_kernel[grid_term](
        out_flat,
        gate_flat,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        term_flat,
        R,
        hidden_dim,
        BLOCK_M=BLOCK_M_TERM,
        BLOCK_K=BLOCK_K_TERM,
        num_warps=8,
    )
    # free intermediate tensors
    del out_flat, gate_flat, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final projection (cuBLAS) and reshape
    # ------------------------------------------------------------------
    # to_out.weight shape: [dim, hidden] (float32).  Use FP16 weight for speed.
    to_out_w = w["to_out.weight"].to(torch.float16)    # [dim, hidden]
    # torch.nn.functional.linear: output = input @ weight.t()
    out_final_fp16 = F.linear(term_flat, to_out_w)    # [R, dim] fp16

    out_final = out_final_fp16.to(torch.float32).view(B, N, N, dim)   # [B,N,N,dim] FP32

    return out_final
