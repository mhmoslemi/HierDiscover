@triton.jit
def compute_term_kernel(
    out_ptr,
    gate_ptr,
    mu_ptr,
    rstd_ptr,
    ln_w_ptr,
    ln_b_ptr,
    term_ptr,
    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    r_start = pid * BLOCK_R
    offs_r = r_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # decode location
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # now loop over hidden chunk
    # We'll generate term for each hidden in BLOCK_H loops
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute offset for out and gate
        hidden_stride = N * N
        # offset for position (b,i,j) to start of hidden slice 0:
        base = b * (H * N * N) + i * N + j   # offset for hidden index = 0 (since each slice has stride hidden_stride)
        # For hidden index h, address = base + h * hidden_stride
        out_off = base[:, None] + offs_h[None, :] * hidden_stride
        gate_off = base[:, None] + offs_h[None, :] * hidden_stride

        out_vals = tl.load(out_ptr + out_off,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + gate_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)
        rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)
        # broadcast mu and rstd across hidden dim
        mu = mu[:, None]
        rstd = rstd[:, None]

        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)    # float32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

        diff = (out_vals - mu) * rstd
        norm = diff * gamma[None, :] + beta[None, :]
        term = norm * gate_vals

        # store term as fp16
        term_fp16 = term.to(tl.float16)
        # compute offset to term_ptr: shape (R, H) contiguous
        term_off = offs_r[:, None] * H + offs_h[None, :]
        tl.store(term_ptr + term_off,
                 term_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])
```

"""
TriMul (outgoing) forward – highly‑optimized Triton implementation.

Algorithm (FP16 everywhere, FP32 only for statistics):
1️⃣  Row‑wise LayerNorm of the flattened input (R = B·N·N rows,
    C = dim).  FP32 mean/variance → FP16 normalised tensor.
2️⃣  Fuse the five linear projections (left, right, left_gate,
    right_gate, out_gate) into a single matmul:
        X_norm @ W_fusedᵀ → (R, 5·H)   (H = hidden_dim)
3️⃣  Split the fused projection, apply the optional pair‑wise mask,
    evaluate the three sigmoids and write the three tensors
    (left, right, out_gate) directly in (B, H, N, N) layout.
4️⃣  Batched GEMM (cuBLAS) over the hidden dimension:
        left[b,h] @ right[b,h]ᵀ   →   out[b,h]  (shape B·H·N·N)
5️⃣  Row‑wise mean & rsqrt‑variance of `out` (across H) → μ, rsqrt(var).
6️⃣  Compute the second LayerNorm, multiply by `out_gate`,
    and fuse the final linear projection with a *cuBLAS* matrix
    multiplication:
        term = ((out - μ) * rsqrt(var) * γ_out + β_out) * out_gate
        out_final = term @ W_outᵀ
   (`γ_out`, `β_out` are from `to_out_norm`; `W_outᵀ` is the transposed
    weight of the final Linear).  The result is cast to FP32.

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
Only the forward pass is required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # const   *[rows, dim] (float32)
    y_ptr,    # mutable *[rows, dim] (float16)
    w_ptr,    # const   *[dim]      (float32)   γ
    b_ptr,    # const   *[dim]      (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- accumulate sums for mean / variance (fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalised values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid (writes left/right/out_gate)
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const   *[R, 5*H]   (fp16)
    left_ptr,       # mutable *[B, H, N, N] (fp16)
    right_ptr,      # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,   # mutable *[B, H, N, N] (fp16)
    mask_ptr,       # const   *[R]        (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)       # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                                   # distance between hidden slices
    base_offset = b * (H * N * N) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout (R,5*H): [left, right, left_gate, right_gate, out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in fp32 → back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate * mask_val

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr    + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,    # const *[B, H, N, N] (fp16)
    mu_ptr,     # mutable *[R]         (float32)
    rstd_ptr,   # mutable *[R]         (float32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # offset of h = 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# --------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and prepare term for cuBLAS
# --------------------------------------------------------------
@triton.jit
def compute_term_kernel(
    out_ptr,        # const *[B, H, N, N] (fp16)
    gate_ptr,       # const *[B, H, N, N] (fp16)
    mu_ptr,         # const *[R]          (float32)
    rstd_ptr,       # const *[R]          (float32)
    ln_w_ptr,       # const *[H]          (float32)  γ of to_out_norm
    ln_b_ptr,       # const *[H]          (float32)  β of to_out_norm
    term_ptr,       # mutable *[R, H]      (fp16)
    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    # decode (b,i,j) from flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * N * N) + i * N + j                # offset of hidden‑slice 0

    # load per‑row statistics once (broadcast later)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r,   other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r,   other=0.0)   # [BLOCK_R]
    mu   = mu[:, None]      # [BLOCK_R,1]
    rstd = rstd[:, None]    # [BLOCK_R,1]

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)       # [BLOCK_H]
        mask_h = offs_h < H

        # load out & gate (fp16 → fp32)
        out_off  = base[:, None] + offs_h[None, :] * hidden_stride
        gate_off = base[:, None] + offs_h[None, :] * hidden_stride

        out_vals  = tl.load(out_ptr  + out_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)   # [BLOCK_R, BLOCK_H]
        gate_vals = tl.load(gate_ptr + gate_off,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)   # [BLOCK_R, BLOCK_H]

        # LayerNorm parameters (γ,β) for the hidden channel
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H] fp32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H] fp32

        # (out - μ) * rsqrt(var) → normalized
        diff = (out_vals - mu) * rstd
        # second LayerNorm
        norm = diff * gamma[None, :] + beta[None, :]      # [BLOCK_R, BLOCK_H]
        # apply out‑gate
        term = norm * gate_vals                           # [BLOCK_R, BLOCK_H]

        # store term in row‑major (R, H) layout: offset = row*H + hidden
        term_off = offs_r[:, None] * H + offs_h[None, :]
        tl.store(term_ptr + term_off,
                 term.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with LayerNorm / Linear parameters
                             (all float32, Linear layers have no bias)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],   # γ (float32)
        w["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # shape (5·H, dim)

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (layout B·H·N·N)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) over the hidden dimension
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                             # (B, H, N, N)
    del left, right    # free memory we no longer need

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise statistics (mean & rsqrt‑var) of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 128
    BLOCK_K_STATS = 32
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out‑gate → term (R,H), then cuBLAS projection
    # ------------------------------------------------------------------
    term = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_TERM = 128
    BLOCK_H_TERM = 32
    grid_term = (triton.cdiv(R, BLOCK_R_TERM),)
    compute_term_kernel[grid_term](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        term,
        R,
        H,
        N,
        BLOCK_R=BLOCK_R_TERM,
        BLOCK_H=BLOCK_H_TERM,
        num_warps=4,
    )
    del out, out_gate, mu, rstd   # free intermediate tensors

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (cuBLAS) – the original “to_out”
    # ------------------------------------------------------------------
    # to_out.weight has shape (dim, hidden_dim); we need its transpose (H, dim)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    out_fp16 = torch.matmul(term, w_out_T)          # (R, dim) fp16, Tensor‑Core GEMM
    out_fp32 = out_fp16.to(torch.float32)           # cast back to FP32
    final = out_fp32.view(B, N, N, dim)             # reshape to original layout

    return final
