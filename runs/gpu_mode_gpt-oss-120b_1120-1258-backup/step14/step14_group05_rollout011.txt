```python
"""
Efficient outgoing TriMul forward pass.

Algorithm:
1. Row‑wise LayerNorm over input (FP32 mean/var → FP16 normalized).
2. Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM.
3. Triton kernel `split_fuse_gate_kernel` splits the fused projection,
   applies optional mask, computes sigmoid gates and stores the
   left/right/out_gate tensors directly in (B, H, N, N) layout.
4. Batched GEMM (torch.bmm) over the hidden dimension:
      out[b,h] = left[b,h] @ right[b,h]ᵀ   → (B·H, N, N) FP16.
5. Triton kernel `rowwise_stats_kernel` computes per‑position mean and
   rsqrt‑variance of `out` across the hidden dimension (required for
   second LayerNorm).
6. Triton kernel `final_output_kernel` fuses the second LayerNorm,
   element‑wise multiplication by the out‑gate and the final
   linear projection → (B, N, N, dim) FP32.

All kernels are written for Triton 3.3.1, targeting NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,  # const   *[rows, dim]  (float32)
    y_ptr,  # mutable *[rows, dim]  (float16)
    w_ptr,  # const   *[dim]       (float32)   γ
    b_ptr,  # const   *[dim]       (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    m_mask = offs_m < rows

    # accumulate sum and sum‑sq in fp32
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over dim in chunks of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=m_mask[:, None] & d_mask[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized values with γ,β
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=m_mask[:, None] & d_mask[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(w_ptr + offs_d, mask=d_mask, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=d_mask, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=m_mask[:, None] & d_mask[None, :])

# --------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates,
#    write directly in (B, H, N, N) layout
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # const   *[R, 5*H]   (fp16)
    left_ptr,      # mutable *[B, H, N, N] (fp16)
    right_ptr,     # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,  # mutable *[B, H, N, N] (fp16)
    mask_ptr,      # const   *[R]        (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    r_mask = offs_r < R

    # decode (b,i,k) from the flattened position (R = B*N*N)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcast over hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=r_mask, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between two hidden slices
    base_offset = b * (H * N * N) + i * N + k          # offset for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        h_mask = offs_h < H

        # offsets inside the fused projection
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)

        # sigmoid in fp32 then cast back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate * mask_val

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset,
                 left,
                 mask=r_mask[:, None] & h_mask[None, :])
        tl.store(right_ptr    + dst_offset,
                 right,
                 mask=r_mask[:, None] & h_mask[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=r_mask[:, None] & h_mask[None, :])

# --------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # const *[B, H, N, N] (fp16)
    mu_ptr,    # mutable *[R]         (float32)
    rstd_ptr,  # mutable *[R]         (float32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    m_mask = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # address of h=0 element

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        k_mask = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=m_mask[:, None] & k_mask[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=m_mask)
    tl.store(rstd_ptr + offs_m, rstd, mask=m_mask)

# --------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# --------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # const *[B, H, N, N] (fp16)
    gate_ptr,       # const *[B, H, N, N] (fp16)   (out_gate)
    mu_ptr,         # const *[R]            (float32)
    rstd_ptr,       # const *[R]            (float32)
    ln_w_ptr,       # const *[H]            (float32)   γ of to_out_norm
    ln_b_ptr,       # const *[H]            (float32)   β of to_out_norm
    w_out_T_ptr,    # const *[H, dim]       (fp16)      (H×dim)
    out_final_ptr,  # mutable *[B, N, N, dim] (float32)
    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    m_mask = offs_m < total_rows
    n_mask = offs_n < dim

    # decode (b,i,j) from flattened row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # address of h=0 slice

    mu   = tl.load(mu_ptr   + offs_m, mask=m_mask, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=m_mask, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        k_mask = offs_k < H

        # load out & gate (fp16 → fp32)

                            mask=m_mask[:, None] & k_mask[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

                            mask=m_mask[:, None] & k_mask[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

"""
Efficient outgoing TriMul forward pass.

Algorithm:
1️⃣ Row‑wise LayerNorm over the input (FP32 statistics → FP16 normalized tensor).
2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single GEMM.
3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused projection,
   applies the (optional) mask, computes sigmoid gates and stores the
   left/right/out_gate tensors directly in (B, H, N, N) layout.
4️⃣ Batched GEMM (torch.bmm) over the hidden dimension:
      out[b,h] = left[b,h] @ right[b,h]ᵀ   → (B·H, N, N) FP16.
5️⃣ Triton kernel `rowwise_stats_kernel` computes per‑position mean and
   rsqrt‑variance of `out` across the hidden dimension (required for the
   second LayerNorm).
6️⃣ Triton kernel `final_output_kernel` fuses the second LayerNorm,
   element‑wise multiplication by the out‑gate and the final linear
   projection → (B, N, N, dim) FP32.

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed‑precision
(FP16 for the bulk of the work, FP32 for statistics).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,  # const   *[rows, dim]  (float32)
    y_ptr,  # mutable *[rows, dim]  (float16)
    w_ptr,  # const   *[dim]       (float32)   γ
    b_ptr,  # const   *[dim]       (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    m_mask = offs_m < rows

    # ----------------- compute mean & variance (fp32) -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=m_mask[:, None] & d_mask[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------------- write normalized values -----------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=m_mask[:, None] & d_mask[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(w_ptr + offs_d, mask=d_mask, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=d_mask, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=m_mask[:, None] & d_mask[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates,
#    write directly in (B, H, N, N) layout
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # const   *[R, 5*H]   (fp16)
    left_ptr,      # mutable *[B, H, N, N] (fp16)
    right_ptr,     # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,  # mutable *[B, H, N, N] (fp16)
    mask_ptr,      # const   *[R]        (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    r_mask = offs_r < R

    # decode (b,i,k) from the flattened index (R = B·N·N)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcast over hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=r_mask, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between two hidden slices
    base_offset = b * (H * N * N) + i * N + k          # offset for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        h_mask = offs_h < H

        # offsets inside the fused projection (layout: [R, 5·H])
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=r_mask[:, None] & h_mask[None, :],
                                other=0.0)

        # sigmoid in fp32 then cast back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate * mask_val

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset,
                 left,
                 mask=r_mask[:, None] & h_mask[None, :])
        tl.store(right_ptr    + dst_offset,
                 right,
                 mask=r_mask[:, None] & h_mask[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=r_mask[:, None] & h_mask[None, :])


# --------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # const *[B, H, N, N] (fp16)
    mu_ptr,    # mutable *[R]         (float32)
    rstd_ptr,  # mutable *[R]         (float32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    m_mask = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # address of h=0 element

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        k_mask = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=m_mask[:, None] & k_mask[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=m_mask)
    tl.store(rstd_ptr + offs_m, rstd, mask=m_mask)


# --------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# --------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # const *[B, H, N, N] (fp16)
    gate_ptr,       # const *[B, H, N, N] (fp16)   (out_gate)
    mu_ptr,         # const *[R]            (float32)
    rstd_ptr,       # const *[R]            (float32)
    ln_w_ptr,       # const *[H]            (float32)   γ of to_out_norm
    ln_b_ptr,       # const *[H]            (float32)   β of to_out_norm
    w_out_T_ptr,    # const *[H, dim]       (fp16)      (H×dim)
    out_final_ptr,  # mutable *[B, N, N, dim] (float32)
    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    m_mask = offs_m < total_rows
    n_mask = offs_n < dim

    # decode (b,i,j) from flattened row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # address of h=0 slice

    mu   = tl.load(mu_ptr   + offs_m, mask=m_mask, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=m_mask, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        k_mask = offs_k < H

        # load out & gate (fp16 → fp32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=m_mask[:, None] & k_mask[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=m_mask[:, None] & k_mask[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # second LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=k_mask, other=0.0)   # [K] fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=k_mask, other=0.0)   # [K] fp32

        # normalize over hidden dimension
        diff = (out_vals - mu[:, None]) * rstd[:, None]          # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]              # [M, K]

        # apply out‑gate
        term = norm * gate_vals                                   # [M, K]
        term_fp16 = term.to(tl.float16)                           # fp16 for dot

        # weight tile (H×dim) – contiguous over hidden dim
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=k_mask[:, None] & n_mask[None, :],
                         other=0.0)                                 # [K, N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=m_mask[:, None] & n_mask[None, :])

    # 5️⃣ Row‑wise statistics (mean & rsqrt‑var) of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 128
    BLOCK_K_STATS = 32
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    # to_out.weight is (dim, hidden_dim) → transpose to (hidden_dim, dim)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64          # rows per program
    BLOCK_K_OUT = 32          # hidden‑dim tile
    BLOCK_N_OUT = 64          # output‑dim tile
    grid_out = (triton.cdiv(R, BLOCK_M_OUT),
                triton.cdiv(dim, BLOCK_N_OUT))

    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
```
