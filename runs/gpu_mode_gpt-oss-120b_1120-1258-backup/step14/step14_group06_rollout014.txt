2️⃣  A single fused linear projection that produces the three
    matrices (left, right) and three sigmoid gates in one GEMM.
3️⃣  A kernel that splits the fused projection, applies the optional pair‑wise
    mask, evaluates the sigmoid gates and stores the results in a

4️⃣  Batched half‑precision GEMM (cuBLAS) computes
    `out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]`.
5️⃣  Row‑wise mean & rsqrt‑variance of `out` (still [B, H, N, N] fp16).
6️⃣  A final Triton kernel fuses the second LayerNorm, the element‑wise
    multiplication with `out_gate`, and the final linear projection to
    the output embedding dimension.
All kernels operate on contiguous tensors and use fp16 for compute

import triton
import triton.language as tl

# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # const   *[rows, dim] (float32)
    y_ptr,      # mutable *[rows, dim] (float16)
    w_ptr,      # const   *[dim]      (float32) γ
    b_ptr,      # const   *[dim]      (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values -----------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)    # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # β (fp32)
        out = norm * w[None, :] + b[None, :]                   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H] (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]        (fp16) – ignored if HAS_MASK=0
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask handling

                           other=1.0)                # [BLOCK_R]

    else:

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + k          # start of h=0 slice

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # column base for this row & hidden block

                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,

                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        tl.store(right_ptr    + dst_offset,

        tl.store(out_gate_ptr + dst_offset,

                 mask=mask_r[:, None] & mask_h[None, :])

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # offset of h=0

    # per‑position mean / rsqrt‑var
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor and gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # second LayerNorm parameters for the k‑th hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32

        # (out - μ) * rstd → normalize, then γ/β, then multiply by gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]
        term = norm * gate_vals                                    # [M, K] fp32
        term_fp16 = term.to(tl.float16)                           # cast for dot

        # load slice of the final weight (H×dim) – contiguous over hidden dim
        w_ptr = w_out_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                               # [K, N]

        # accumulate using Tensor‑Core dot (fp16×fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile)

    # write final result: (B, N, N, dim) → flattened as (R, dim)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],   # γ (float32)
        w["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + 3 gates)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(device).to(torch.float16)               # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)              # fp16, shape [R, 5·H]
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (layout B·H·N·N)
    # --------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM over the hidden dimension
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for cuBLAS batched matmul
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                             # (B, H, N, N)

    # free the projection tensors we no longer need
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise statistics (mean & rsqrt‑var) of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 128
    BLOCK_K_STATS = 32
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate & final linear projection
    # --------------------------------------------------------------
    # to_out.weight has shape [dim, H]; we need (H, dim) for the dot
    w_out_T = w["to_out.weight"].to(device).to(torch.float16).t().contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64          # rows per program (positions)
    BLOCK_K_OUT = 32          # hidden‑dim chunk
    BLOCK_N_OUT = 64          # output‑dim chunk
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
