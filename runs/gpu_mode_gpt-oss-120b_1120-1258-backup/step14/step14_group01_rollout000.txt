"""
TriMul (outgoing) – fast forward‑only implementation in Triton.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the (B,N,N,dim) tensor (fp32 → fp16).
2️⃣  Fuse the three linear projections (left/right) and the three gates
    into a single fp16 GEMM (`torch.nn.functional.linear`).  The weight
    matrix is the concatenation of the five original weight matrices.
3️⃣  Split the fused result, apply the optional pair‑wise mask and the
    sigmoid gates, and store the three tensors (left, right, out_gate)
    in layout (B, H, N, N) where H = hidden_dim.
4️⃣  Batched half‑precision GEMM (cuBLAS) over the hidden dimension:
       out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k].
5️⃣  Compute per‑position mean and rsqrt‑variance of the contracted
    tensor (fp32) with a Triton kernel.
6️⃣  Fuse the second LayerNorm, the out‑gate and the final linear
    projection (H→dim) into a single Triton kernel, producing the final
    tensor of shape (B,N,N,dim) in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ LayerNorm (row‑wise, fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # input   [R, dim]  (fp32)
    y_ptr,               # output  [R, dim]  (fp16)
    w_ptr,               # gamma   [dim]     (fp32)
    b_ptr,               # beta    [dim]     (fp32)
    rows: tl.int32,      # R = B*N*N
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # accumulate sum and sum‑of‑squares in fp32
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # ----- pass 1: statistics ------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- pass 2: write normalized output -----------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0) # gamma fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0) # beta  fp32
        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5*H] (fp16)
    left_ptr,          # [B, H, N, N] (fp16)
    right_ptr,         # [B, H, N, N] (fp16)
    out_gate_ptr,      # [B, H, N, N] (fp16)
    mask_ptr,          # [R] (fp16) or dummy
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program (≈128)
    BLOCK_H: tl.constexpr,   # hidden dim slice per loop (≈32)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from linear index offs_r
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair‑wise mask broadcast over H
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden slices
    # base offset of the first hidden slice for this (b,i,k)
    base_offset = b * (H * N * N) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # layout in proj: [R, 5*H] row‑major
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate  = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_gate = tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (fp32 for stability) -------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates -----------------------------------------------------------
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # destination offsets (linear in the packed [B,H,N,N] layout)
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# 5️⃣ Row‑wise mean / rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # [B, H, N, N] (fp16)
    mu_ptr,           # [R] (fp32)
    rstd_ptr,         # [R] (fp32)
    total_rows: tl.int32,   # R = B*N*N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_K: tl.constexpr,  # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # offset of first hidden slice

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

def final_output_kernel(

    gate_ptr,         # out_gate [B, H, N, N] (fp16)

    rstd_ptr,         # [R] (fp32)
    ln_w_ptr,         # to_out_norm.weight [H] (fp32)
    ln_b_ptr,         # to_out_norm.bias   [H] (fp32)

    H: tl.int32,
    N: tl.int32,
    dim: tl.int32,
    eps: tl.float32,

    BLOCK_K: tl.constexpr,   # hidden‑dim tile
    BLOCK_N: tl.constexpr,   # output‑dim tile
):

    pid_n = tl.program_id(1)   # output‑dim

    row_start = pid_m * BLOCK_M

    b = offs_m // nn
    rem = offs_m - b * nn

    j = rem - i * N

    hidden_stride = N * N

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] fp32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] fp32

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor slice + gate slice
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride

                            mask=mask_m[:, None] & mask_k[None, :],

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
```

This implementation follows the algorithm described in the docstring, uses Triton for the computationally intensive row‑wise LayerNorm, the split+gate/fuse step, the per‑position statistics, and the final fused LayerNorm + gate + projection. All heavy memory‑bound work is performed on‑device, while the batched GEMM (`torch.bmm`) leverages cuBLAS fp16 for the `left × rightᵀ` contraction. The function returns the required tensor of shape `[B, N, N, dim]` in `float32`.
