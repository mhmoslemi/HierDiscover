"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “Tri‑multiplicative update”).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 stats → FP16 output) – custom
    Triton kernel (`layernorm_fp16_kernel`). This is the only mandatory kernel.

2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single FP16 GEMM (`torch.nn.functional.linear`) by concatenating the
    weight matrices. No bias is present in any of these layers.

3️⃣  (Optional) apply the binary mask to the left/right projections. The mask
    is broadcast over the hidden dimension.

4️⃣  Apply sigmoid to the three gate tensors and element‑wise gating.

5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched half‑precision matrix‑multiply (`torch.bmm`);
    each hidden head is treated as an independent batch.

6️⃣  Second LayerNorm on the intermediate tensor (FP16) – we rely on the
    highly‑optimized PyTorch implementation (`torch.nn.functional.layer_norm`).
    The result is then multiplied element‑wise with the already‑sigmoided
    `out_gate`.

7️⃣  Final linear projection back to the model dimension (`dim`) – FP16 GEMM.

All tensors are kept in half‑precision except the two LayerNorm statistics,
which are accumulated in FP32 for numerical stability. The final result is
cast back to `float32` to match the original PyTorch module.

The implementation works for all test configurations (batch size 1‑2,
sequence length up to 1024, model dimension 128/384, hidden dimension 128,
with or without a mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (input FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (output FP16)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static channel dimension (≤ 384)
    eps: tl.constexpr,   # LayerNorm epsilon
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 64)
    BLOCK_D: tl.constexpr,   # columns processed per iteration (e.g. 128)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict) where
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool) –
                         ignored if config["nomask"]=True
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {
                "dim"        : int   (C),
                "hidden_dim" : int   (H),
                "nomask"     : bool,
                "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor : shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]     # H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output) – custom Triton kernel
    # --------------------------------------------------------------
    # Flatten to [R, C] where R = B*N*N
    inp_flat = inp.reshape(total_rows, dim).contiguous()   # FP32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)

    # Block sizes: ROWS per program = 64 (good for typical R) ;
    # COLS per iteration = 128 (covers dim ≤ 384 in ≤3 passes)
    BLOCK_M = 64
    BLOCK_D = 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (single FP16 GEMM)
    # --------------------------------------------------------------
    # Concatenate weight matrices: each is [hidden, dim]
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                 # shape (5*H, C)

    # (R, 5*H) – FP16
    proj_all = F.linear(x_norm_fp16, concat_w)   # bias = None

    # Split the concatenated result into its five components
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H)

    # --------------------------------------------------------------
    # 2a) Apply binary mask if required (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [R, 1] (half)
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b) Sigmoid gates (still FP16) and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)     # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)   # (R, H)
    out_gate = torch.sigmoid(out_gate_fp16)       # (R, H) – later used after LN

    left_fp16 = left_proj_fp16 * left_gate        # (R, H)
    right_fp16 = right_proj_fp16 * right_gate     # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left_fp16.view(B, N, N, hidden)    # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)  # (B, N, N, H)

    # Prepare (B*H, N, N) matrices for bmm:
    #   left_mat  : rows = i, cols = k
    #   right_mat : rows = k, cols = j
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    # Batched matrix multiplication (Tensor‑Core, FP16)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out‑gate (use torch.layer_norm – fast)
    # --------------------------------------------------------------
    # Flatten to [R, H] where R = B*N*N
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H)

    # PyTorch LayerNorm works on half‑precision tensors directly;
    # we pass the FP16 weight/bias cast to half.
    out_norm = F.layer_norm(
        out_flat,
        (hidden,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )  # (R, H) FP16

    # Apply out_gate (already sigmoid‑ed, half)
    out_norm_gate = out_norm * out_gate                # (R, H) FP16

    # --------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()   # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate, to_out_w_fp16)  # (R, dim) FP16

    # Cast back to float32 as required by the original module
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
