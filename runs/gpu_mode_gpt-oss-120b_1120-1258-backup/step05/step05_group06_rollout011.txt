    proj_ptr,    # const half [R, 5*H]
    mask_ptr,   # const half [R,1] (if mask, else not used)
    left_ptr, right_ptr, out_gate_ptr,
    total_rows: tl.int32,
    hidden: tl.constexpr,
    nomask: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0,BLOCK_R)
    mask_r = offs_r < total_rows

    # Process hidden dimension in loop
    for h_begin in range(0, hidden, BLOCK_H):
        offs_h = h_begin + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # compute offsets
        # indices: left_proj at offset h, right_proj at offset hidden + h,
        # left_gate offset 2*hidden + h, right_gate offset 3*hidden + h,
        # out_gate offset 4*hidden + h

        # load each component
        ptr = proj_ptr + offs_r[:, None] * (5*hidden) + offs_h[None, :]  # left_proj part
        left_val = tl.load(ptr, mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # right_gate
        ptr = proj_ptr + offs_r[:, None] * (5*hidden) + (3*hidden + offs_h)[None, :]

@triton.jit
def gate_mask_fuse_kernel(
    proj_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows: tl.int32,
    hidden: tl.int32,
    NOMASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # pre-load mask for rows if needed
    if NOMASK == 0:
        mask_val = tl.load(mask_ptr + offs_r[:, None], mask=mask_r[:, None], other=0.0)
    else:
        mask_val = tl.zeros((BLOCK_R, 1), dtype=tl.float16)

    # compute number of hidden blocks
    num_h_blocks = tl.cdiv(hidden, BLOCK_H)

    for hb in range(0, num_h_blocks):
        h_start = hb * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # compute stride between rows
        row_stride = 5 * hidden

        # load each component
        left = tl.load(proj_ptr + offs_r[:, None] * row_stride + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right = tl.load(proj_ptr + offs_r[:, None] * row_stride + (hidden + offs_h)[None, :],
                        mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate = tl.load(proj_ptr + offs_r[:, None] * row_stride + (2*hidden + offs_h)[None, :],
                            mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate = tl.load(proj_ptr + offs_r[:, None] * row_stride + (3*hidden + offs_h)[None, :],
                             mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate = tl.load(proj_ptr + offs_r[:, None] * row_stride + (4*hidden + offs_h)[None, :],
                           mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        if NOMASK == 0:
            # mask_val shape (BLOCK_R,1) broadcast
            left = left * mask_val
            right = right * mask_val

        # sigmoid (1/(1+exp(-x))) in half precision
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate))

        # gated outputs
        left_out = left * left_gate
        right_out = right * right_gate

        # store
        tl.store(left_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                 left_out, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                 right_out, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])
```

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “Tri‑multiplicative update”).

Algorithm
---------
1️⃣ Input LayerNorm – use PyTorch’s native `layer_norm` (FP32 → FP16) which
    performs the reduction in a single kernel and is much faster than a
    handcrafted Triton version.

2️⃣ Single fused linear projection for the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) using a
   half‑precision GEMM (`torch.nn.functional.linear`).

3️⃣ Triton kernel `gate_mask_fuse_kernel`:
   * optional binary mask application,
   * sigmoid for the three gates,
   * element‑wise gating to obtain `left`, `right` and `out_gate`.
   All of the above are fused into one pass over the data.

4️⃣ Triangular multiplicative update:
   `out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]`
   Implemented as a batched half‑precision matrix multiplication
   (`torch.bmm`).  Each head (hidden dimension) is treated as an
   independent batch (`B*H` matrices of size `N×N`).

5️⃣ Output LayerNorm + out‑gate:
   a) LayerNorm on the `hidden` dimension using PyTorch (FP32 stats → FP16).
   b) Element‑wise multiplication with the already‑sigmoided `out_gate`.

6️⃣ Final linear projection back to the model dimension (`dim`) using a
   half‑precision GEMM, then cast to `float32` as required by the original
   module.

Only the gating/mask fusion is written in Triton; all dense linear algebra
uses cuBLAS/Tensor‑Cores which are optimal on an H100.  The implementation
covers all admissible shapes (B∈{1,2}, N∈{128,…,1024}, dim∈{128,384},
hidden_dim=128) and works with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel that fuses mask application, sigmoid gates and gating.
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    proj_ptr,            # const half*  [total_rows, 5*hidden]
    mask_ptr,            # const half*  [total_rows, 1] (ignored if NOMASK==1)
    left_ptr,            # *mut half*   [total_rows, hidden]
    right_ptr,           # *mut half*   [total_rows, hidden]
    out_gate_ptr,        # *mut half*   [total_rows, hidden]
    total_rows: tl.int32,
    hidden: tl.int32,
    NOMASK: tl.constexpr,        # 0 → mask is used, 1 → mask ignored
    BLOCK_R: tl.constexpr,       # rows processed per program (e.g. 128)
    BLOCK_H: tl.constexpr,       # hidden‑dim block size (e.g. 64)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < total_rows

    # Load mask once per row block if required
    if NOMASK == 0:
        # mask is stored as (total_rows, 1) → stride = 1
        mask_row = tl.load(mask_ptr + offs_r[:, None],
                           mask=mask_r[:, None],
                           other=0.0)                         # [BLOCK_R, 1]
    else:
        mask_row = tl.zeros((BLOCK_R, 1), dtype=tl.float16)

    # number of hidden blocks
    n_h_blocks = tl.cdiv(hidden, BLOCK_H)

    # stride between rows in the concatenated projection tensor
    row_stride = 5 * hidden

    for hb in range(0, n_h_blocks):
        h_start = hb * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)               # [BLOCK_H]
        mask_h = offs_h < hidden

        # ----------- load the five slices -----------------------------
        left = tl.load(
            proj_ptr + offs_r[:, None] * row_stride + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right = tl.load(
            proj_ptr + offs_r[:, None] * row_stride + (hidden + offs_h)[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate = tl.load(
            proj_ptr + offs_r[:, None] * row_stride + (2 * hidden + offs_h)[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate = tl.load(
            proj_ptr + offs_r[:, None] * row_stride + (3 * hidden + offs_h)[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate = tl.load(
            proj_ptr + offs_r[:, None] * row_stride + (4 * hidden + offs_h)[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # ----------- optional binary mask ------------------------------
        if NOMASK == 0:
            # mask_row shape (BLOCK_R,1) broadcasts over hidden dim
            left  = left  * mask_row
            right = right * mask_row

        # ----------- sigmoid gates (half precision) --------------------
        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_gate))

        # ----------- gating -------------------------------------------
        left_out  = left  * left_gate
        right_out = right * right_gate

        # ----------- store results ------------------------------------
        tl.store(
            left_ptr + offs_r[:, None] * hidden + offs_h[None, :],
            left_out,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + offs_r[:, None] * hidden + offs_h[None, :],
            right_out,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
            out_gate,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict) where
        - input_tensor : torch.Tensor, shape [B, N, N, C] (`float32`)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (`bool`/`float32`);
                         ignored if config["nomask"] is True
        - weights_dict : mapping of the model's parameters (all `float32`)
        - config_dict  : {
                "dim"        : int   (C),
                "hidden_dim" : int   (H),
                "nomask"     : bool,
                "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (`float32`) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack input
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]     # H (always 128 in the provided tests)
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # PyTorch's layer_norm computes mean/var in FP32 and is heavily
    # optimized; we keep the result in half‑precision for the subsequent GEMM.
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.half)                # shape [B,N,N,dim] (half)

    # ------------------------------------------------------------------
    # 2️⃣ Single fused linear projection for all five weight matrices
    # ------------------------------------------------------------------
    # Concatenate weight matrices (shape: 5*H × dim) and run one GEMM.
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, dim)

    # Flatten input to (R, dim) for the GEMM.
    x_flat = x_norm.reshape(total_rows, dim)
    proj_all = F.linear(x_flat, concat_w)     # (R, 5*H), half‑precision

    # ------------------------------------------------------------------
    # 3️⃣ Triton kernel: mask, sigmoid, and gating (fused)
    # ------------------------------------------------------------------
    left_fp16   = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    right_fp16  = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    out_gate_fp16 = torch.empty(total_rows, hidden, dtype=torch.half, device=device)

    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(torch.half)
    else:
        # dummy tensor – will never be accessed because NOMASK==1
        mask_flat = torch.empty(0, dtype=torch.half, device=device)

    # Launch parameters – tuned for the target hardware
    BLOCK_R = 128   # rows per program
    BLOCK_H = 64    # hidden‑dim block
    grid = (triton.cdiv(total_rows, BLOCK_R),)

    gate_mask_fuse_kernel[grid](
        proj_all,
        mask_flat,
        left_fp16,
        right_fp16,
        out_gate_fp16,
        total_rows,
        hidden,
        NOMASK=1 if nomask else 0,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H) and prepare batched matrices:
    #   left_mat  : (B*H, N, N)   where rows = i, cols = k
    #   right_mat : (B*H, N, N)   where rows = k, cols = j
    left = left_fp16.view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)

    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)

    # Half‑precision batched matmul (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat)        # (B*H, N, N), half

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm + out‑gate (both in FP32 for maximum accuracy)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)                # (B,N,N,H)

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                # (B,N,N,H)

    # Apply the already‑sigmoided out‑gate
    out_gate_fp32 = out_gate_fp16.to(torch.float32)
    out_norm_gate = out_norm * out_gate_fp32          # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()       # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate.view(total_rows, hidden), to_out_w)  # (R, dim)
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
