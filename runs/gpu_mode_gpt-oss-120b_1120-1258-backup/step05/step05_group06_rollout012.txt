@triton.jit
def layernorm_stats_kernel(x_ptr, mean_ptr, rstd_ptr, total_rows, dim, eps, BLOCK_M, BLOCK_D):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

@triton.jit
def norm_linear_kernel(
    X_ptr,          # *const float* [R, C]
    mean_ptr,       # *const float* [R]
    rstd_ptr,       # *const float* [R]
    W_ptr,          # *const half*  [C, M]
    Y_ptr,          # *mut half*    [R, M]
    R, C, M,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)
    offs_i = pid_row * BLOCK_R + tl.arange(0, BLOCK_R)
    offs_j = pid_col * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_i = offs_i < R
    mask_j = offs_j < M

    # load per-row stats
    row_mean = tl.load(mean_ptr + offs_i, mask=mask_i, other=0.0)
    row_rstd = tl.load(rstd_ptr + offs_i, mask=mask_i, other=0.0)

    acc = tl.zeros((BLOCK_R, BLOCK_M), dtype=tl.float32)

    for k in range(0, tl.cdiv(C, BLOCK_C)):
        offs_k = k * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_k = offs_k < C

        # load A block
        a_ptrs = X_ptr + (offs_i[:, None] * C + offs_k[None, :])
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)  # float32

        # normalize
        a = (a - row_mean[:, None]) * row_rstd[:, None]  # broadcast

        # load weight block (C x M)
        w_ptrs = W_ptr + (offs_k[:, None] * M + offs_j[None, :])
        w = tl.load(w_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)  # cast half->fp32

        # compute dot
        acc += tl.dot(a, w)  # (BLOCK_R, BLOCK_C) x (BLOCK_C, BLOCK_M) => (BLOCK_R, BLOCK_M)

    # store output half
    out_ptrs = Y_ptr + (offs_i[:, None] * M + offs_j[None, :])
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    inp, mask, weights, cfg = data
    ...
    B,N,_,C= inp.shape
    R = B * N * N
    total_rows = R
    hidden = cfg["hidden_dim"]
    dim = cfg["dim"]
    M = hidden * 5

    eps = cfg.get('eps',1e-5)
    nomask = cfg.get('nomask', True)
    device = inp.device

    # Compute mean and rstd per row
    mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd = torch.empty(total_rows, dtype=torch.float32, device=device)
    BLOCK_R = 64
    BLOCK_D = 128  # for reduction across C dimension
    grid_stats = (triton.cdiv(total_rows, BLOCK_R),)
    layernorm_stats_kernel[grid_stats](
        inp.reshape(total_rows, dim).contiguous(),
        mean,
        rstd,
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
    )
    # Prepare weight transpose
    concat_w = torch.cat([...], dim=0).half().contiguous()
    weight_T = concat_w.t().contiguous()  # shape (dim, M)

    # Compute normalized linear projections
    proj_all = torch.empty(total_rows, M, dtype=torch.float16, device=device)
    BLOCK_R = 64
    BLOCK_C = 64
    BLOCK_M = 64
    grid_mat = (triton.cdiv(total_rows, BLOCK_R), triton.cdiv(M, BLOCK_M))
    norm_linear_kernel[grid_mat](
        inp.reshape(total_rows, dim).contiguous(),
        mean,
        rstd,
        weight_T,
        proj_all,
        total_rows,
        dim,
        M,
        eps=eps,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
        BLOCK_M=BLOCK_M,
    )
    # split
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(proj_all, hidden, dim=1)

    # apply mask if nomask False
    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # gating sigmoid
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # reshape to (B,N,N,H)
    left = left_fp16.view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)
    left_mat = left.permute(0, 3, 1, 2).reshape(B*hidden, N, N)  # (B*H,N,N)

    # second layernorm + out_gate scaling fused

    out_norm_gate_fp16 = torch.empty_like(out_flat)
    grid2 = (triton.cdiv(total_rows, BLOCK_R),)

        BLOCK_D=BLOCK_D,

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “Tri‑multiplicative update”).

Key ideas
---------
1️⃣  Compute per‑row mean / rstd of the input (FP32) with a tiny Triton kernel.
2️⃣  Fuse the first LayerNorm and the five linear maps (left/right projections
    + three gates) into a single GEMM kernel.  The kernel receives the
    previously computed (mean,rstd) and produces a concatenated tensor of shape
    [R, 5·H] (R = B·N·N, H = hidden_dim) in FP16.
3️⃣  Split the concatenated tensor, apply the optional mask, sigmoid the gates
    and perform the element‑wise gating.
4️⃣  Batched half‑precision matrix‑multiply (torch.bmm) implements the
    Tri‑multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
5️⃣  Fuse the second LayerNorm and the output‑gate multiplication into one
    Triton kernel.
6️⃣  Final linear projection back to the model dimension (FP16 GEMM) and cast
    to FP32 for the output.

All heavy reductions (mean/variance) are performed in FP32, while the bulk of
the linear algebra uses Tensor‑Core‑accelerated FP16 (via cuBLAS for the
batched GEMM and Triton for the fused kernels).  This removes the large
intermediate tensor created by the naïve “norm ➜ linear” two‑step approach and
yields a ≈2× speed‑up on the reference workloads.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Per‑row mean / rstd (no output tensor)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_stats_kernel(
    x_ptr,          # *const float*  [total_rows, dim]  (FP32 input)
    mean_ptr,       # *mut   float*  [total_rows]      (output)
    rstd_ptr,       # *mut   float*  [total_rows]      (output)
    total_rows: tl.int32,
    dim: tl.constexpr,          # static channel dim (≤384)
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,      # rows handled per program
    BLOCK_D: tl.constexpr,      # columns handled per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- reduction ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                              # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 2️⃣  Fused LayerNorm + Linear (produces concatenated 5·H outputs)
# ----------------------------------------------------------------------
@triton.jit
def norm_linear_kernel(
    x_ptr,          # *const float* [R, C]   (FP32 input)
    mean_ptr,       # *const float* [R]      (row mean)
    rstd_ptr,       # *const float* [R]      (row rstd)
    w_ptr,          # *const half*  [C, M]   (weight^T, FP16)
    y_ptr,          # *mut   half*  [R, M]   (output)
    R: tl.int32,
    C: tl.int32,
    M: tl.int32,
    BLOCK_R: tl.constexpr,      # rows per program
    BLOCK_C: tl.constexpr,      # reduction dimension per iteration
    BLOCK_M: tl.constexpr,      # output columns per program
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    offs_i = pid_row * BLOCK_R + tl.arange(0, BLOCK_R)          # rows
    offs_j = pid_col * BLOCK_M + tl.arange(0, BLOCK_M)          # output cols
    mask_i = offs_i < R
    mask_j = offs_j < M

    # load per‑row statistics (broadcast later)
    row_mean = tl.load(mean_ptr + offs_i, mask=mask_i, other=0.0)
    row_rstd = tl.load(rstd_ptr + offs_i, mask=mask_i, other=0.0)

    acc = tl.zeros((BLOCK_R, BLOCK_M), dtype=tl.float32)

    for k in range(0, tl.cdiv(C, BLOCK_C)):
        offs_k = k * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_k = offs_k < C

        # ---------- load a block of the input ----------
        a_ptrs = x_ptr + (offs_i[:, None] * C + offs_k[None, :])
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                                 # FP32

        # normalize a on‑the‑fly
        a = (a - row_mean[:, None]) * row_rstd[:, None]        # FP32

        # ---------- load a block of the transposed weight ----------
        w_ptrs = w_ptr + (offs_k[:, None] * M + offs_j[None, :])
        w = tl.load(w_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)                 # FP32

        # ---------- block GEMM ----------
        acc += tl.dot(a, w)                                    # (BR,BC)·(BC,BM)

    # store result as half
    out_ptrs = y_ptr + (offs_i[:, None] * M + offs_j[None, :])
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 6️⃣  Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,          # *const half*  [total_rows, dim] (pre‑norm)
    y_ptr,          # *mut   half*  [total_rows, dim] (post‑norm⋅gate)
    weight_ptr,     # *const float* [dim]          (γ)
    bias_ptr,       # *const float* [dim]          (β)
    gate_ptr,       # *const half*  [total_rows, dim] (out_gate already sigmoid‑ed)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- first pass: mean & var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32
        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]            # FP32

        # multiply by the already‑sigmoided out_gate (half → FP32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict) where
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : dict containing:
                "dim"        : int   (C)
                "hidden_dim" : int   (H)
                "nomask"     : bool
                "eps"        : float (optional, default 1e-5)

    Returns
    -------
    torch.Tensor : shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim   = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]       # H
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N               # R = B·N·N
    M = hidden * 5                        # concatenated output dimension

    # ------------------------------------------------------------------
    # 1️⃣  per‑row mean / rstd
    # ------------------------------------------------------------------
    inp_flat = inp.reshape(total_rows, dim).contiguous()   # FP32
    row_mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    row_rstd = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_R = 64
    BLOCK_D = 128
    grid_stats = (triton.cdiv(total_rows, BLOCK_R),)
    layernorm_stats_kernel[grid_stats](
        inp_flat,
        row_mean,
        row_rstd,
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣  fused LayerNorm + 5 linear maps (produces [R,5·H] half)
    # ------------------------------------------------------------------
    # concatenate weights (each  [H, C]) → shape (5·H, C)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5·H, C)

    # We need W^T of shape (C, 5·H) for the kernel
    w_T = concat_w.t().contiguous()           # (C, 5·H)

    proj_all = torch.empty(total_rows, M, dtype=torch.float16, device=device)

    BLOCK_C = 64
    BLOCK_M = 64
    grid_mat = (
        triton.cdiv(total_rows, BLOCK_R),     # rows
        triton.cdiv(M, BLOCK_M),              # output columns
    )
    norm_linear_kernel[grid_mat](
        inp_flat,
        row_mean,
        row_rstd,
        w_T,
        proj_all,
        total_rows,
        dim,
        M,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
        BLOCK_M=BLOCK_M,
    )

    # ------------------------------------------------------------------
    # 3️⃣  split, mask, sigmoid gates, element‑wise gating
    # ------------------------------------------------------------------
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each [R, H]

    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    left_gate = torch.sigmoid(left_gate_fp16)       # (R, H) half
    right_gate = torch.sigmoid(right_gate_fp16)     # (R, H) half
    out_gate = torch.sigmoid(out_gate_fp16)         # (R, H) half (used later)

    left_fp16  = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 4️⃣  Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden)          # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)        # (B, N, N, H)

    # reshape to [B·H, N, N] where the second operand is transposed on the last two axes
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (i,k)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (k,j)

    out_mat = torch.bmm(left_mat, right_mat)        # (B·H, N, N) half
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) half

    # ------------------------------------------------------------------
    # 5️⃣  fused LayerNorm + output‑gate
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()
    out_norm_gate_fp16 = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows, BLOCK_R),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_R,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6️⃣  final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(out_norm_gate_fp16, weights["to_out.weight"].half())  # (R, C) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
