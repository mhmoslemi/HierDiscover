"""
Highly‑optimized forward pass for the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32‑reduction → FP16 output) – Triton.
2️⃣ Fused Linear+Gate for the “left” and “right” branches (FP16 GEMM + sigmoid).
3️⃣ Optional binary mask – simple element‑wise broadcast.
4️⃣ Triangular multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that processes a tile
   (BLOCK_I × BLOCK_J) of the output matrix while iterating over the
   sequence dimension *k* and accumulating a full hidden vector (BLOCK_H)
   for each (i,j) pair.  This fuses the *H* independent GEMMs used in the
   reference implementation, dramatically reducing memory traffic.
5️⃣ Fused LayerNorm + output‑gate – Triton.
6️⃣ Final Linear projection back to the model dimension – FP16 GEMM.

All heavy arithmetic is in FP16 (Tensor‑Core) with FP32 reductions,
final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (FP16)
    weight_ptr,          # *const float*  [dim] (γ)
    bias_ptr,            # *const float*  [dim] (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- mean & variance (FP32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ----- normalise, scale, bias (store FP16) -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, hidden] (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, hidden] (post‑norm⋅gate)
    weight_ptr,          # *const float* [hidden] (γ)
    bias_ptr,            # *const float* [hidden] (β)
    gate_ptr,            # *const half*  [total_rows, hidden] (out_gate already sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- mean & variance (FP32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- normalise, scale, bias, multiply by out_gate -----
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 4) Triangular multiplicative update (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,   # *const half*  [B, N, N, H]
    right_ptr,  # *const half*  [B, N, N, H]
    out_ptr,    # *mut   half*  [B, N, N, H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    H_START: tl.constexpr,          # start index of the hidden block handled by this launch
):
    # program ids: 0 → batch, 1 → i‑tile, 2 → j‑tile
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    b = pid_b

    # ----- accumulator (FP32) -----
    acc = tl.zeros([BLOCK_I, BLOCK_J, BLOCK_H], dtype=tl.float32)

    # ----- coordinate vectors -----
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    offs_h = H_START + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # ----- loop over the reduction dimension k -----
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K

        # iterate over the inner k‑elements (unrolled)
        for kk in range(0, BLOCK_K):
            k = k_start + kk
            # out‑of‑range k are silently masked out
            mask_k = k < N

            # ----- load left[i,k,:]  (shape [BLOCK_I, BLOCK_H]) -----
            ptrs_left = (left_ptr
                         + b * stride_left_b
                         + offs_i[:, None] * stride_left_i
                         + k * stride_left_k
                         + offs_h[None, :] * stride_left_h)
            left_vals = tl.load(ptrs_left,
                                mask=mask_i[:, None] & mask_k & mask_h[None, :],
                                other=0.0).to(tl.float32)      # [BLOCK_I, BLOCK_H]

            # ----- load right[j,k,:] (shape [BLOCK_J, BLOCK_H]) -----
            ptrs_right = (right_ptr
                          + b * stride_right_b
                          + offs_j[:, None] * stride_right_j
                          + k * stride_right_k
                          + offs_h[None, :] * stride_right_h)
            right_vals = tl.load(ptrs_right,
                                 mask=mask_j[:, None] & mask_k & mask_h[None, :],
                                 other=0.0).to(tl.float32)     # [BLOCK_J, BLOCK_H]

            # ----- outer product + accumulate across h -----
            # left_vals[:, None, :] → (BLOCK_I, 1, BLOCK_H)
            # right_vals[None, :, :] → (1, BLOCK_J, BLOCK_H)
            acc += left_vals[:, None, :] * right_vals[None, :, :]   # (BLOCK_I, BLOCK_J, BLOCK_H)

    # ----- write results -------------------------------------------------
    ptrs_out = (out_ptr
                + b * stride_out_b
                + offs_i[:, None, None] * stride_out_i
                + offs_j[None, :, None] * stride_out_j
                + offs_h[None, None, :] * stride_out_h)

    tl.store(ptrs_out,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (AlphaFold‑3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                     # input / output channel dimension C
    hidden = cfg["hidden_dim"]              # hidden dimension H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)                # [R, C] half

    BLOCK_M = 128
    BLOCK_D = 64 if dim <= 256 else 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Two fused linear‑+‑gate passes (left & right) + out‑gate
    # ------------------------------------------------------------------
    # cast weights to half once (they are small)
    left_proj_w  = weights["left_proj.weight"].half().contiguous()
    left_gate_w  = weights["left_gate.weight"].half().contiguous()
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()

    # left side (proj + gate)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    left_comb   = F.linear(x_norm_fp16, left_comb_w)            # (R, 2*H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw).half()
    left = left_proj_raw * left_gate                             # (R, H)

    # right side (proj + gate)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)
    right_comb   = F.linear(x_norm_fp16, right_comb_w)
    right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw).half()
    right = right_proj_raw * right_gate                         # (R, H)

    # out‑gate (used after the triangular update)
    out_gate_raw = F.linear(x_norm_fp16, out_gate_w)            # (R, H)
    out_gate = torch.sigmoid(out_gate_raw).half()               # (R, H)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Reshape tensors for the kernel
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden).contiguous()
    right = right.view(B, N, N, hidden).contiguous()
    out = torch.empty_like(left)                               # (B, N, N, H) half

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update – Triton kernel
    # ------------------------------------------------------------------
    # tiling parameters – tuned for H100
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32
    BLOCK_H = 32                         # must divide hidden (or be masked)

    grid_tri = (B,
                triton.cdiv(N, BLOCK_I),
                triton.cdiv(N, BLOCK_J))

    # stride helpers (element‑wise strides, not bytes)
    stride_left_b  = left.stride(0)
    stride_left_i  = left.stride(1)
    stride_left_k  = left.stride(2)
    stride_left_h  = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b   = out.stride(0)
    stride_out_i   = out.stride(1)
    stride_out_j   = out.stride(2)
    stride_out_h   = out.stride(3)

    for h_start in range(0, hidden, BLOCK_H):
        tri_mul_outgoing_kernel[grid_tri](
            left,
            right,
            out,
            B,
            N,
            hidden,
            stride_left_b, stride_left_i, stride_left_k, stride_left_h,
            stride_right_b, stride_right_j, stride_right_k, stride_right_h,
            stride_out_b, stride_out_i, stride_out_j, stride_out_h,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
            H_START=h_start,
        )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()
    out_norm_gate = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                   # already sigmoid’ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()            # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)        # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
