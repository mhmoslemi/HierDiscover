"""
Highly‑optimized Triton+PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3, Chai, Protenix).

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP16 output) – reduced in FP32 for numerical
    stability, written back as half‑precision.

2️⃣  Two fused linear + gating passes (still FP16):
    * left  = Linear(x_norm) → split → sigmoid(gate) → proj * gate
    * right = Linear(x_norm) → split → sigmoid(gate) → proj * gate
   The two linear layers are fused by concatenating the projection and
   gate weight matrices, cutting the number of memory reads in half.

3️⃣  Optional binary mask – broadcasted over the hidden dimension and
   applied after gating (cheap element‑wise multiply).

4️⃣  Triangular multiplicative update:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched half‑precision GEMM
    `out = torch.bmm(left_mat, right_mat)` where
    `right_mat` is already transposed to the needed orientation.

5️⃣  Fused LayerNorm + output‑gate (both FP16):
    The hidden‑dim tensor is normalised, scaled/bias added, then multiplied
    element‑wise by the sigmoid‑gated output‑gate, all inside one Triton
    kernel to avoid extra passes.

6️⃣  Final linear projection back to the original dimension (`dim`) – an
    FP16 GEMM (`torch.nn.functional.linear`).  The result is cast back to
    FP32, matching the reference implementation.

All heavy reductions (means/variances) are done in FP32, while the
matrix‑multiplications exploit Tensor‑Cores on H100 (FP16).  The kernel
launch configuration adapts to the `dim`/`hidden_dim` size (≤384) and
covers any batch‑size (1 or 2) and sequence length (up to 1024) supported
by the tests.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP16 output)
# ---------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (FP32 input)
    y_ptr,               # *mut   half*   [total_rows, dim] (FP16 output)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static dim (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------------------------------------------------------------
    # Reduce to mean / variance (FP32)
    # ----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ----------------------------------------------------------------
    # Normalise, apply γ/β and store as half
    # ----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    BLOCK_M = 128                                 # rows per program
    BLOCK_D = 128 if dim > 256 else 64          # columns per iteration

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Two fused linear‑+‑gate passes (left & right)
    # ------------------------------------------------------------------
    # Cast model weights to half once
    left_proj_w  = weights["left_proj.weight"].half().contiguous()
    left_gate_w  = weights["left_gate.weight"].half().contiguous()
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()

    # ------- left side (proj + gate fused) -------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    left_comb   = F.linear(x_norm_fp16, left_comb_w)            # (R, 2*H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)                    # (R, H)
    left = left_proj_raw * left_gate                            # (R, H)

    # ------- right side (proj + gate fused) -------
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2*H, C)
    right_comb   = F.linear(x_norm_fp16, right_comb_w)            # (R, 2*H)
    right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)                     # (R, H)
    right = right_proj_raw * right_gate                           # (R, H)

    # ------- out‑gate (single linear + sigmoid) -------
    out_gate_raw = F.linear(x_norm_fp16, out_gate_w)               # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)                         # (R, H), already half

    # ------------------------------------------------------------------
    # 2a️⃣ Apply binary mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        # mask [B,N,N] → [R,1] (half).  Broadcast over hidden dim.
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched matmul)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, hidden)    # (B, N, N, H)
    right = right.view(B, N, N, hidden)   # (B, N, N, H)

    # Fuse batch & head dimension for GEMM
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, i, k)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, k, j)

    # Half‑precision batched matmul → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)

    # Back‑to [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)            # (R, H) half

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid’ed, half‑precision
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()            # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)        # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
