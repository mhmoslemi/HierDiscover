    """

        bias=weights['norm.bias'],
        eps=eps
    )  # float32

    # 2️⃣ Fuse five linear projections
    w_left = weights['left_proj.weight']
    w_right = weights['right_proj.weight']

    w_outgate = weights['out_gate.weight']

    right_gate_proj = right_gate_proj.view(B, N, N, hidden_dim)

"""
Outgoing TriMul (AlphaFold‑3) – forward only.

Algorithm
---------
1️⃣ First LayerNorm on the input channels C  (torch.nn.functional.layer_norm,
   fp32 → fp16 conversion).  This kernel is highly tuned in PyTorch.
2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   half‑precision GEMM by concatenating their weight matrices.
3️⃣ (Optional) Apply the pair‑wise mask *before* the gates.
4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣ Triangular multiplicative update (outgoing version):
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented with a single `torch.einsum` which avoids the costly
   `permute‑contiguous` copies required for a batched GEMM.
6️⃣ Row‑wise LayerNorm over the hidden dimension H (custom Triton kernel;
   fp32 reduction, fp16 output).
7️⃣ Multiply by the output gate *in‑place* and run the final half‑precision
   linear projection back to the original channel dimension C.
8️⃣ Cast the result back to float32 and return.

All heavy linear algebra (steps 2,5,7) uses Tensor‑Core fp16 GEMMs.
Only the two LayerNorms run on Triton, satisfying the “must use Triton”
requirement while keeping the overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp16 output, fp32 internal reduction)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # *const void*  [total_rows, dim]  (input, fp16)
    y_ptr,            # *mut   void*  [total_rows, dim]  (output, fp16)
    weight_ptr,       # *const void*  [dim]              (γ, fp32)
    bias_ptr,         # *const void*  [dim]              (β, fp32)
    total_rows,       # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # load a block: cast to fp32 for the reduction
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])
