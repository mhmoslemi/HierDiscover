def gating_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_ptr,
    mask_ptr,
    left_out_ptr,
    right_out_ptr,
    out_gate_out_ptr,
    total_rows,
    hidden,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    DO_MASK: tl.constexpr
):
    pid = tl.program_id(0)  # row block
    # offsets
    row_start = pid * BLOCK_M
    rows = row_start + tl.arange(0, BLOCK_M)
    row_mask = rows < total_rows

    hid_start = tl.program_id(1) * BLOCK_H
    hids = hid_start + tl.arange(0, BLOCK_H)
    hid_mask = hids < hidden

    # load: shape (BLOCK_M, BLOCK_H)
    left_proj = tl.load(left_proj_ptr + rows[:, None] * hidden + hids[None,:],
                        mask=row_mask[:, None] & hid_mask[None,:], other=0.0)
    right_proj = tl.load(... similarly)
    left_gate = tl.load(...).   same
    right_gate = tl.load(...).   same
    out_gate = tl.load(...).    same

    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate))

    right_out = right_proj * right_gate

"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Input LayerNorm (FP32 reduction → FP16 output) – custom Triton kernel.
2️⃣ Fuse the five linear projections (left/right + three gates) into a
   single FP16 GEMM (torch.nn.functional.linear) – the heaviest compute step.
3️⃣ Fuse mask, sigmoid‑gating and element‑wise multiplication for the
   left/right projections in one Triton kernel (gate_kernel).  This removes
   three separate PyTorch kernels (mask, sigmoid, mul).
4️⃣ Triangular multiplicative update – a batched matrix‑multiply
   ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`` performed with
   ``torch.bmm`` on half‑precision tensors (Tensor‑Core).
5️⃣ Fuse the final LayerNorm, out‑gate and projection to the output dimension
   in a single Triton kernel (layernorm_gate_linear_kernel).

The implementation stays entirely on‑device and returns a float‑32 tensor
with shape ``[B, N, N, dim]``.  It supports all test configurations and
uses mixed‑precision (FP16 for the bulk of the work, FP32 for the numerically
sensitive reductions).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                     # *const float*  [total_rows, dim] (input FP32)
    y_ptr,                     # *mut   half*   [total_rows, dim] (output FP16)
    weight_ptr,                # *const float*  [dim]          (γ)
    bias_ptr,                  # *const float*  [dim]          (β)
    total_rows,                # int32
    dim,                       # tl.constexpr   (≤ 384)
    eps,                       # tl.constexpr
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_D: tl.constexpr,     # dim chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3) Fuse mask, sigmoid‑gating and element‑wise multiplication
# ----------------------------------------------------------------------
@triton.jit
def gate_kernel(
    left_proj_ptr,   # *const half* [total_rows, hidden]  (proj)
    right_proj_ptr,  # *const half* [total_rows, hidden]
    left_gate_ptr,   # *const half* [total_rows, hidden] (pre‑sigmoid)
    right_gate_ptr,  # *const half* [total_rows, hidden]
    out_gate_ptr,    # *const half* [total_rows, hidden] (pre‑sigmoid)
    mask_ptr,        # *const half* [total_rows, 1] (0/1) – ignored if DO_MASK=0
    left_out_ptr,    # *mut   half* [total_rows, hidden]
    right_out_ptr,   # *mut   half* [total_rows, hidden]
    out_gate_out_ptr,# *mut   half* [total_rows, hidden] (sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    DO_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    row_start = pid_m * BLOCK_M
    col_start = pid_h * BLOCK_H

    rows = row_start + tl.arange(0, BLOCK_M)
    cols = col_start + tl.arange(0, BLOCK_H)

    mask_rows = rows < total_rows
    mask_cols = cols < hidden

    # -----------------------------------------------------------------
    # Load the five tensors (FP32 for math, cast back to half later)
    # -----------------------------------------------------------------
    off = rows[:, None] * hidden + cols[None, :]   # stride = hidden

    lp = tl.load(left_proj_ptr + off,
                 mask=mask_rows[:, None] & mask_cols[None, :],
                 other=0.0).to(tl.float32)

    rp = tl.load(right_proj_ptr + off,
                 mask=mask_rows[:, None] & mask_cols[None, :],
                 other=0.0).to(tl.float32)

    lg = tl.load(left_gate_ptr + off,
                 mask=mask_rows[:, None] & mask_cols[None, :],
                 other=0.0).to(tl.float32)

    rg = tl.load(right_gate_ptr + off,
                 mask=mask_rows[:, None] & mask_cols[None, :],
                 other=0.0).to(tl.float32)

    og = tl.load(out_gate_ptr + off,
                 mask=mask_rows[:, None] & mask_cols[None, :],
                 other=0.0).to(tl.float32)

    # -----------------------------------------------------------------
    # Optional mask (binary 0/1) – broadcasted over the hidden dim.
    # -----------------------------------------------------------------
    if DO_MASK:
        m = tl.load(mask_ptr + rows[:, None],
                    mask=mask_rows[:, None],
                    other=1.0).to(tl.float32)      # shape (BLOCK_M, 1)
        lp = lp * m
        rp = rp * m

    # -----------------------------------------------------------------
    # Sigmoid (1 / (1+exp(-x))) – performed in FP32 for stability.
    # -----------------------------------------------------------------
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # -----------------------------------------------------------------
    # Apply the gates
    # -----------------------------------------------------------------
    left_out  = lp * lg
    right_out = rp * rg

    # -----------------------------------------------------------------
    # Store results back as FP16
    # -----------------------------------------------------------------
    tl.store(left_out_ptr + off,
             left_out.to(tl.float16),
             mask=mask_rows[:, None] & mask_cols[None, :])

    tl.store(right_out_ptr + off,
             right_out.to(tl.float16),
             mask=mask_rows[:, None] & mask_cols[None, :])

    tl.store(out_gate_out_ptr + off,
             og.to(tl.float16),
             mask=mask_rows[:, None] & mask_cols[None, :])


# ----------------------------------------------------------------------
# 5) Fused LayerNorm + out‑gate + final linear projection
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden]   (pre‑norm)
    gate_ptr,            # *const half*  [total_rows, hidden]   (out_gate, sigmoid’ed)
    norm_w_ptr,          # *const float* [hidden]               (γ of to_out_norm)
    norm_b_ptr,          # *const float* [hidden]               (β of to_out_norm)
    out_w_ptr,           # *const half*  [dim, hidden]          (to_out weight)
    out_ptr,             # *mut   float* [total_rows, dim]      (final FP32 output)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # output‑channel chunk
):
    pid = tl.program_id(0)          # row block
    pid_d = tl.program_id(1)        # output‑channel block
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Load the whole hidden vector for each selected row (BLOCK_M × hidden)
    # -----------------------------------------------------------------
    offs_h = tl.arange(0, hidden)                               # [hidden]
    mask_h = offs_h < hidden

    x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                       # (BLOCK_M, hidden)

    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                       # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Compute mean / rstd for each row (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)                                 # (BLOCK_M)
    sum_sq  = tl.sum(x * x, axis=1)                             # (BLOCK_M)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Apply LayerNorm weight/bias and the out‑gate
    # -----------------------------------------------------------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)

    norm = (x - mean[:, None]) * rstd[:, None]                     # (BLOCK_M, hidden)
    norm = norm * w_norm[None, :] + b_norm[None, :]                # apply γ,β
    gated = norm * g                                                # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Output‑channel block handling
    # -----------------------------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)                       # [BLOCK_D]
    mask_d = offs_d < dim

    # Load the slice of the final weight matrix (hidden × BLOCK_D)
    # out_w has shape [dim, hidden]; we need its transpose (hidden, BLOCK_D)
    out_w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                      mask=mask_h[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)                  # (hidden, BLOCK_D)

    # Compute the mat‑mul: (BLOCK_M, hidden) × (hidden, BLOCK_D) → (BLOCK_M, BLOCK_D)
    out_chunk = tl.dot(gated, out_w_T)                            # (BLOCK_M, BLOCK_D)

    # Store the result (FP32)
    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]            (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – ignored if config["nomask"]=True
        - weights_dict : mapping of parameter names → torch.Tensor (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim    = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)

    BLOCK_M = 64
    BLOCK_D = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5 linear layers → one GEMM)
    # --------------------------------------------------------------
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                         # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)   # (R, 5*H)  FP16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
    right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                            # each (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask, sigmoid‑gates and element‑wise multiplication
    # --------------------------------------------------------------
    # Buffers for the gated tensors
    left_fp16 = torch.empty(total_rows, hidden,
                            dtype=torch.half, device=device)
    right_fp16 = torch.empty(total_rows, hidden,
                             dtype=torch.half, device=device)
    out_gate_sig_fp16 = torch.empty(total_rows, hidden,
                                    dtype=torch.half, device=device)

    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
    else:
        # a dummy mask of all‑ones (will be ignored in the kernel)
        mask_flat = torch.ones(total_rows, 1, dtype=torch.half, device=device)

    BLOCK_H = 64
    grid_gate = (triton.cdiv(total_rows, BLOCK_M),
                 triton.cdiv(hidden, BLOCK_H))

    gate_kernel[grid_gate](
        left_proj_fp16,
        right_proj_fp16,
        left_gate_fp16,
        right_gate_fp16,
        out_gate_fp16,
        mask_flat,
        left_fp16,
        right_fp16,
        out_gate_sig_fp16,
        total_rows,
        hidden=hidden,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        DO_MASK=0 if nomask else 1,
    )

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched matmul)
    # --------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left = left_fp16.view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)

    # (B, H, N, N) where the last two axes are (i,k) and (k,j) respectively
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)          # (i,k)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)        # (k,j)

    out_mat = torch.bmm(left_mat, right_mat)                               # (i,j)
    # back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + out‑gate + final projection
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()                    # (R, H)

    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    grid_out = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    layernorm_gate_linear_kernel[grid_out](
        out_flat,                                   # x_ptr
        out_gate_sig_fp16,                          # gate_ptr (already sigmoid)
        weights["to_out_norm.weight"],              # γ
        weights["to_out_norm.bias"],                # β
        weights["to_out.weight"].half().contiguous(),# to_out weight (dim, hidden)
        final_fp32,                                 # output buffer
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape to final tensor shape
    # --------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)           # [B, N, N, dim] float32
    return final
