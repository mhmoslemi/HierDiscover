@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    h_start = pid_h * BLOCK_H

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_h = h_start + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # output accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left values shape (BLOCK_M, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + \
            offs_i[:, None, None] * stride_left_i + \
            cur_k[None, :, None] * stride_left_k + \
            offs_h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0).to(tl.float32)   # (BM, BK, BH)

        # right values shape (BLOCK_N, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + \
            offs_j[:, None, None] * stride_right_j + \
            cur_k[None, :, None] * stride_right_k + \
            offs_h[None, None, :] * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                        other=0.0).to(tl.float32)   # (BN, BK, BH)

        # accumulate outer products across k dimension
        # left: (BM, BK, BH), right: (BN, BK, BH)
        # iterate over k within BK
        for kk in range(0, BLOCK_K):
            # left slice (BM, BH)
            left_slice = left[:, kk, :]   # (BM, BH)
            # right slice (BN, BH)
            right_slice = right[:, kk, :]  # (BN, BH)
            # broadcast and multiply

        offs_j[None, :, None] * stride_out_j + \
        offs_h[None, None, :] * stride_out_h

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AFA3 “Tri‑multiplicative update”).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input (FP32 stats → FP16 output).  Triton.
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single half‑precision GEMM (torch.nn.functional.linear) by concatenating
    the weight matrices.
3️⃣  Optional binary mask applied to the left/right projections (broadcasted).
4️⃣  Apply sigmoid to the three gate tensors and element‑wise gating.
5️⃣  Tri‑multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a custom Triton kernel that directly computes the outer‑product
    reduction over the sequence dimension.  The kernel works on tiles
    (i‑tile, j‑tile, h‑slice) and accumulates in FP32, storing the result in FP16.
6️⃣  Fused LayerNorm + output‑gate (both FP16).  Triton.
7️⃣  Final linear projection back to the model dimension (half‑precision GEMM).
8️⃣  Cast the result back to float32 (as required by the reference module).

All kernels are written for H100 GPUs (triton‑3.3.1).  The implementation
handles the presence/absence of a mask and works for all test configurations
(B≤2, N∈{128,…,1024}, dim∈{128,384}, hidden_dim=128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (input FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (output FP16)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static channel dimension (≤ 384)
    eps: tl.constexpr,   # LayerNorm epsilon
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, dim] (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, dim] (post‑norm⋅gate)
    weight_ptr,          # *const float* [dim]          (γ)
    bias_ptr,            # *const float* [dim]          (β)
    gate_ptr,            # *const half*  [total_rows, dim] (out_gate, already sigmoid‑ed)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        # multiply by the already‑sigmoided out_gate (half → FP32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5) Tri‑multiplicative update (batched outer‑product reduction)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,       # *const half*  [N, N, H]   (left projection, batch‑slice)
    right_ptr,      # *const half*  [N, N, H]   (right projection, batch‑slice)
    out_ptr,        # *mut   half*  [N, N, H]   (output, batch‑slice)
    N: tl.int32,    # sequence length
    H: tl.int32,    # hidden dimension
    stride_i: tl.int32,   # stride for the first axis (i or j) = N*H
    stride_k: tl.int32,   # stride for the reduction axis = H
    stride_h: tl.int32,   # stride for the hidden axis = 1
    stride_out_i: tl.int32,
    stride_out_j: tl.int32,
    stride_out_h: tl.int32,
    BLOCK_M: tl.constexpr,   # tile size over i
    BLOCK_N: tl.constexpr,   # tile size over j
    BLOCK_K: tl.constexpr,   # tile size over k (set to 1)
    BLOCK_H: tl.constexpr,   # tile size over hidden dimension
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    h_start = pid_h * BLOCK_H

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_h = h_start + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over the reduction dimension k
    # ---------------------------------------------------------------
    for k in range(0, N, BLOCK_K):          # BLOCK_K is 1 → pure scalar loop
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left slice  : (BLOCK_M, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + \
            offs_i[:, None, None] * stride_i + \
            cur_k[None, :, None] * stride_k + \
            offs_h[None, None, :] * stride_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] &
                            mask_h[None, None, :],
                       other=0.0).to(tl.float32)          # FP32

        # right slice : (BLOCK_N, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + \
            offs_j[:, None, None] * stride_i + \
            cur_k[None, :, None] * stride_k + \
            offs_h[None, None, :] * stride_h
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] &
                             mask_h[None, None, :],
                        other=0.0).to(tl.float32)          # FP32

        # Outer‑product accumulation across the hidden slice
        # left  : (BM, BK, BH)   → slice left[:, kk, :] : (BM, BH)
        # right : (BN, BK, BH)   → slice right[:, kk, :] : (BN, BH)
        for kk in range(0, BLOCK_K):
            l = left[:, kk, :]               # (BM, BH)
            r = right[:, kk, :]              # (BN, BH)
            # broadcast to (BM, BN, BH) and accumulate
            acc += l[:, None, :] * r[None, :, :]

    # ---------------------------------------------------------------
    # Store the result (cast back to half)
    # ---------------------------------------------------------------
    out_ptrs = out_ptr + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_h[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None] &
                  mask_h[None, None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool)
                         – ignored if config["nomask"] is True
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt)}

    Returns
    -------
    torch.Tensor : shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]                 # input channel dimension C
    hidden = cfg["hidden_dim"]       # hidden dimension H (always 128 in the tests)
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.reshape(total_rows, dim).contiguous()   # FP32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)

    BLOCK_M_LN = 64
    BLOCK_D_LN = 128
    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)
    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                 # (5*H, C)

    # Linear projection (R, 5*H) – half precision, bias‑free
    proj_all = F.linear(x_norm_fp16, concat_w)   # (R, 5*H)

    # Split the concatenated result
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask if needed (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates (still FP16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)          # (R, H)

    left_fp16 = (left_proj_fp16 * left_gate).contiguous()
    right_fp16 = (right_proj_fp16 * right_gate).contiguous()

    # ------------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update (custom Triton kernel)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] for the kernel
    left = left_fp16.view(B, N, N, hidden).contiguous()
    right = right_fp16.view(B, N, N, hidden).contiguous()
    out_fp16 = torch.empty_like(left)                # (B, N, N, H)

    # Tile sizes – tuned for H100 registers & shared memory
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 1                 # scalar reduction step
    BLOCK_H = 8

    stride_i = N * hidden       # stride for the first axis (i or j)
    stride_k = hidden
    stride_h = 1
    # Output strides (i, j, h)
    stride_out_i = N * hidden
    stride_out_j = hidden
    stride_out_h = 1

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(hidden, BLOCK_H),
    )

    for b in range(B):
        trimul_outgoing_kernel[grid](
            left[b],
            right[b],
            out_fp16[b],
            N,
            hidden,
            stride_i,
            stride_k,
            stride_h,
            stride_out_i,
            stride_out_j,
            stride_out_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out_fp16.view(total_rows, hidden).contiguous()   # (R, H)
    out_norm_gate_fp16 = torch.empty_like(out_flat)             # (R, H)

    BLOCK_M_LN2 = 64
    BLOCK_D_LN2 = 128
    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M_LN2),)

    fused_layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M_LN2,
        BLOCK_D=BLOCK_D_LN2,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()          # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)  # (R, dim)

    # Cast back to float32 as required by the original module
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
