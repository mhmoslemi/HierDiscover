@triton.jit
def fused_ln_gate_kernel(
    src_ptr, gate_ptr, dst_ptr,
    ln_weight_ptr, ln_bias_ptr,
    total_rows, hidden_dim, seq_len,
    stride_b, stride_h, stride_i, stride_j,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        val = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        raw = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (raw - mean[:, None]) * rstd[:, None]

        w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(ln_bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]

        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        out = out * gate

        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Fast native LayerNorm (torch.nn.functional.layer_norm) on the
    input tensor – FP32 → FP16.
2️⃣  Fuse the five linear projections (left, right, three gates) into
    a single FP16 GEMM by concatenating their weight matrices.
3️⃣  Optional pair‑wise mask applied only to the left/right projections.
4️⃣  Apply sigmoid gates and reshape the gated tensors to [B, N, N, H].
5️⃣  Compute the pair‑wise interaction with a single torch.einsum,
    yielding a tensor of shape [B, N, N, H] (no extra permutes).
6️⃣  Fuse the second LayerNorm, the output‑gate and the write‑out to a
    contiguous (B·N·N, H) buffer with a custom Triton kernel

        • normalises, applies γ/β,
        • multiplies by the pre‑computed output‑gate (FP16),
        • stores the result as FP16.
7️⃣  Final projection back to the model dimension with a FP16 GEMM

operations use highly‑optimised Torch kernels, giving a large

        + i_idx * stride_i
        + j_idx * stride_j
    )  # [BLOCK_M]

    # --------------------------------------------------------------
    # Pass 1 – compute mean & variance over hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        # Load a tile of the hidden vector (strided by stride_h)
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        val = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – normalize, apply γ/β, multiply with out_gate and store
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # Load raw hidden values
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        raw = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        # LayerNorm
        norm = (raw - mean[:, None]) * rstd[:, None]      # fp32

        # γ/β (fp32)
        w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                # fp32

        # Multiply with out_gate (fp16 → promote)
        gate = tl.load(
            gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = out * gate

        # Store as fp16
        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # 0) Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # model dimension (e.g. 128 or 384)
    hidden_dim = config["hidden_dim"]   # hidden dimension (always 128 in tests)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                        # total rows = B·N·N

    # ------------------------------------------------------------------
    # 1) First LayerNorm (native CuDNN) – FP32 → FP16
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim) for each (b,i,j) entry.
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)          # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Fused linear projection producing five hidden tensors
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape (hidden_dim, dim) each)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # (5*H, dim)

    # Linear: [B·N·N, dim] @ [dim, 5*H]ᵀ → [B·N·N, 5*H]
    proj = F.linear(x_norm.view(R, dim), w_concat)   # fp16

    # Split projected tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=1
    )                                               # each (R, H)

    # --------------------------------------------------------------
    # 3) Sigmoid gates (remain in fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)               # (R, H)

    # --------------------------------------------------------------
    # 4) Optional pair‑wise mask (applied only to left/right projections)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(R, 1)  # (R,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 5) Apply gates → gated tensors `left` and `right`
    # --------------------------------------------------------------
    left = left_proj * left_gate                      # (R, H)
    right = right_proj * right_gate                  # (R, H)

    # Reshape back to (B, N, N, H) for the pair‑wise interaction
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6) Pair‑wise interaction using a single einsum
    #    out[b, i, j, h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    out = torch.einsum(
        "b i k h, b j k h -> b i j h", left, right
    )  # shape (B, N, N, H), contiguous, fp16

    # ------------------------------------------------------------------
    # 7) Second LayerNorm + out‑gate (fused Triton kernel)
    #    Produces contiguous (R, H) tensor `hidden_out`
    # ------------------------------------------------------------------
    # Stride information for the source tensor `out`
    stride_b = out.stride(0)   # B‑stride = N*N*H
    stride_h = out.stride(3)   # H‑stride = 1 (innermost)
    stride_i = out.stride(1)   # i‑stride = N*H
    stride_j = out.stride(2)   # j‑stride = H

    hidden_out = torch.empty(R, hidden_dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)

    fused_ln_gate_kernel[grid](
        out,                                 # src (non‑contig) - actually contig but ok
        out_gate,                            # gate (R,H)
        hidden_out,                          # dst (contig)
        weights["to_out_norm.weight"],       # γ (fp32)
        weights["to_out_norm.bias"],         # β (fp32)
        R,
        hidden_dim,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 8) Final linear projection back to model dimension
    # ------------------------------------------------------------------
    w_out = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    final_half = F.linear(hidden_out, w_out)             # (R, dim), fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
