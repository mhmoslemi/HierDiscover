def custom_kernel(data):
   inp, mask, weights, cfg = data
   device = inp.device
   dim = cfg["dim"]
   hidden = cfg["hidden_dim"]
   eps = cfg.get("eps", 1e-5)
   nomask = cfg.get("nomask", True)

   B, N, _, _ = inp.shape
   total_rows = B * N * N

   # layernorm
   inp_flat = inp.view(total_rows, dim).contiguous()
   x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

   BLOCK_M = 128
   BLOCK_D = 128 if dim > 256 else 64
   grid = (triton.cdiv(total_rows, BLOCK_M), )
   layernorm_fp16_kernel[grid](
      inp_flat,
      x_norm_fp16,
      weights["norm.weight"],
      weights["norm.bias"],
      total_rows,
      dim,
      eps,
      BLOCK_M=BLOCK_M,
      BLOCK_D=BLOCK_D,
   )

   # Build combined weight for left/right projections and gates and out_gate
   left_proj_w = weights["left_proj.weight"].half().contiguous()
   left_gate_w = weights["left_gate.weight"].half().contiguous()
   right_proj_w = weights["right_proj.weight"].half().contiguous()
   right_gate_w = weights["right_gate.weight"].half().contiguous()
   out_gate_w = weights["out_gate.weight"].half().contiguous()

   # Concatenate all 5 weight matrices
   combined_w = torch.cat([
       left_proj_w,
       left_gate_w,
       right_proj_w,
       right_gate_w,
       out_gate_w
   ], dim=0)   # shape (5*hidden, dim)

   # Linear pass
   comb = F.linear(x_norm_fp16, combined_w)  # (R,5*hidden)

   # split
   splits = torch.split(comb, hidden, dim=1)
   left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = splits

   left_gate = torch.sigmoid(left_gate_raw)
   right_gate = torch.sigmoid(right_gate_raw)
   out_gate = torch.sigmoid(out_gate_raw)

   left = left_proj_raw * left_gate
   right = right_proj_raw * right_gate

   # Apply mask if needed (mask shape B,N,N)
   if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

"""
Highly‑optimized Triton‑accelerated forward pass for the **outgoing**
TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Input LayerNorm (FP16) – compute per‑row mean/var in FP32, write a
    half‑precision normalised tensor.

2️⃣  One fused linear projection:
       * concatenate the four projection weights (left/right) and the
         two gate weights together with the output‑gate weight → a single
         weight matrix of shape (5×H, C).
       * a single GEMM (CUBLAS) produces all five tensors in one pass.
       * split, apply sigmoid to the three gate tensors and form the gated
         left/right tensors.  The output‑gate is kept for the final step.

3️⃣  Optional binary mask – applied to the gated left/right tensors.

4️⃣  Tri‑multiplicative update:
       out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
       Implemented as a batched FP16 matrix‑multiply
       (B·H independent GEMMs of size N×N).

5️⃣  Second LayerNorm + output‑gate – a fused Triton kernel that computes the
    mean/var, normalises the hidden tensor, applies the learned scale/bias
    and multiplies element‑wise with the pre‑computed out‑gate.

6️⃣  Final linear projection (hidden → C) – a regular FP16 GEMM (cuBLAS);
    the result is cast back to FP32.

The implementation removes one full‑precision read of the normalised
input (step 2) and fuses the second LayerNorm with its gate, reducing
memory traffic and kernel launch overhead while keeping all heavy
computations in FP16/Tensor‑Core.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, C] (FP32)
    y_ptr,               # *mut   half*   [total_rows, C] (FP16)
    weight_ptr,          # *const float*  [C]          (γ)
    bias_ptr,            # *const float*  [C]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- first pass: mean & variance (FP32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: normalise, apply γ, β and store as FP16 -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5) Second LayerNorm + out‑gate (both FP16) – fused
# ----------------------------------------------------------------------
@triton.jit

    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,

    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- first pass: mean & variance (FP32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_sq  += tl.sum(vals * vals, axis=1)

    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):

        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                 # C
    hidden = cfg["hidden_dim"]          # H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                      # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)                # [R, C] half

    BLOCK_M = 128
    BLOCK_D = 128 if dim > 256 else 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ One fused linear pass that yields left/right projections, gates,
    #    and the out‑gate in a single matrix‑multiply.
    # ------------------------------------------------------------------
    # Convert all projection / gate weights to FP16 once
    left_proj_w   = weights["left_proj.weight"].half().contiguous()
    left_gate_w   = weights["left_gate.weight"].half().contiguous()
    right_proj_w  = weights["right_proj.weight"].half().contiguous()
    right_gate_w  = weights["right_gate.weight"].half().contiguous()
    out_gate_w    = weights["out_gate.weight"].half().contiguous()

    # Stack all five weight matrices → (5*H, C)
    all_w = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w
    ], dim=0)                                            # shape (5*H, C)

    # Single GEMM
    combined = F.linear(x_norm_fp16, all_w)               # (R, 5*H)

    # Split into the five tensors
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = \
        torch.split(combined, hidden, dim=1)

    # Apply gates (sigmoid) and form gated left/right tensors
    left_gate = torch.sigmoid(left_gate_raw)               # (R, H)
    right_gate = torch.sigmoid(right_gate_raw)             # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)                 # (R, H) – used later

    left = left_proj_raw * left_gate                       # (R, H)
    right = right_proj_raw * right_gate                    # (R, H)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update (batched matrix‑multiply)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden)                      # (B, N, N, H)
    right = right.view(B, N, N, hidden)                    # (B, N, N, H)

    # reshape to (B*H, N, N) for a single batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    out_hidden = torch.bmm(left_mat, right_mat)                     # (B*H, N, N)

    # back to (B, N, N, H) and flatten for the next kernel
    out_hidden = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden_flat = out_hidden.view(total_rows, hidden).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_norm_gate = torch.empty_like(out_hidden_flat)               # (R, H) half

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_hidden_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                  # already sigmoid’ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,                           # reuse same BLOCK_D as above
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half().contiguous()   # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)            # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```

The code defines two Triton kernels – one for the initial LayerNorm and one for the second LayerNorm + output‑gate – and a single PyTorch‑based fused linear pass that produces all four projected tensors and the output‑gate in **one** matrix‑multiply.  
The heavy triangular update remains a batched FP16 matrix‑multiply (`torch.bmm`).  
All operations finally return a float‑32 tensor of shape `[B, N, N, C]`.
