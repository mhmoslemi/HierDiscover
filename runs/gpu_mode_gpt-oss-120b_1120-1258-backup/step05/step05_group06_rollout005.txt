import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,     # const half* [total_rows, dim]
    y_ptr,     # mutable half* [total_rows, dim]
    weight_ptr,# const float* [dim]
    bias_ptr,  # const float* [dim]
    gate_ptr,  # const half* [total_rows, dim]
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # 1) compute mean & variance (FP32)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # 2) normalize, apply gamma/beta, multiply gate, store half
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # float
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # float

        out = norm * w[None, :] + b[None, :]   # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)   # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Algorithm
    ----------
    1.  Row‑wise LayerNorm on the input (FP32 → FP16) using torch.nn.functional.layer_norm.
    2.  Concatenate the five linear weight matrices (left/right projections,
        three gates) and perform a single half‑precision GEMM.
    3.  Optional binary mask on the left/right projections.
    4.  Sigmoid gates and element‑wise gating.
    5.  Batched matrix multiplication (B*hidden, N, N) implementing the
        triangular multiplicative update.
    6.  Fused LayerNorm + out‑gate (custom Triton kernel).
    7.  Final linear projection back to the model dimension.
    8.  Cast result back to float32.
    """
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 -> FP16) using torch (cuDNN optimized)
    # ------------------------------------------------------------------
    x_flat = inp.reshape(total_rows, dim).contiguous()  # FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # FP32
    x_norm_fp16 = x_norm.half()  # (total_rows, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate weight matrices [5*hidden, dim]
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()  # (5*hidden, dim)

    proj_all = F.linear(x_norm_fp16, concat_w)  # (total_rows, 5*hidden), no bias

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (total_rows, hidden)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask if present
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates and gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)  # (total_rows, hidden), half

    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden)   # (B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden) # (B, N, N, hidden)

    # reshape to (B*hidden, N, N) for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N)

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate (custom Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()  # (R, hidden), half
    out_norm_gate_fp16 = torch.empty_like(out_flat)       # (R, hidden), half

    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()   # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)  # (R, dim), half

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```

# -*- coding: utf-8 -*-
"""
Highly‑optimised forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “tri‑multiplicative update”).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input (FP32 → FP16) via
    `torch.nn.functional.layer_norm` (cuDNN‑optimised).

2️⃣  Concatenate the five linear weight matrices (left/right projections +
    three gates) and apply a single half‑precision GEMM (`torch.nn.functional.linear`);
    this fuses five matrix multiplies into one cuBLAS call.

3️⃣  (Optional) binary mask is applied to the left/right projections.

4️⃣  Sigmoid gates are evaluated and used to gate the projected tensors.

5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched half‑precision matrix‑multiply (`torch.bmm`);
    each hidden head is treated as an independent batch.

6️⃣  Fuse the second LayerNorm (γ/β) and the output gate in a custom
    Triton kernel (`fused_layernorm_gate_kernel`).  Statistics are
    accumulated in FP32, the result is written back in FP16.

7️⃣  Final linear projection back to the model dimension (`dim`) – another
    half‑precision GEMM.

8️⃣  Cast the result back to FP32 (as required by the original PyTorch module).

Only the final LayerNorm + gate step is written in Triton; the rest relies on
highly‑optimised cuDNN/cuBLAS kernels, guaranteeing sub‑millisecond runtimes
for the prescribed test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6️⃣ Fused LayerNorm + out‑gate (half‑precision, FP32 stats)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,            # const half*  [total_rows, dim]   (pre‑norm)
    y_ptr,            # mut   half*  [total_rows, dim]   (post‑norm·gate)
    weight_ptr,       # const float* [dim]              (γ)
    bias_ptr,         # const float* [dim]              (β)
    gate_ptr,         # const half*  [total_rows, dim]   (out_gate, sigmoid‑ed)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean & variance (FP32) – two‑pass algorithm
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1/σ

    # ------------------------------------------------------------------
    # 2️⃣  Normalise, apply γ/β, multiply by out‑gate and store as FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ  (FP32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β  (FP32)

        out = norm * w[None, :] + b[None, :]            # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)           # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑backed forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool)
        - weight_dict  : dict of model parameters (float32)
        - config_dict  : {
              "dim"        : int   (C),
              "hidden_dim" : int   (H),
              "nomask"     : bool,
              "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim      = cfg["dim"]                     # model dimension C
    hidden   = cfg["hidden_dim"]              # hidden dimension H
    eps      = cfg.get("eps", 1e-5)
    nomask   = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                    # rows when flattening the [B,N,N] axis

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – use cuDNN for speed
    # ------------------------------------------------------------------
    x_flat = inp.view(total_rows, dim).contiguous()           # (R, C) FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                                          # FP32
    x_norm_fp16 = x_norm.half()                               # (R, C) FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left/right proj + three gates)
    # ------------------------------------------------------------------
    # concatenate weights: each has shape (hidden, dim)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                     # (5*H, C)

    # single GEMM: (R, 5*H) = (R, C) @ (5*H, C)^T
    proj_all = F.linear(x_norm_fp16, concat_w)                # (R, 5*H)

    # split back to the five tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden, dim=1
        )                                                    # each (R, H)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask if it exists
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)  # (R, 1)
        left_proj_fp16  = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates + element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)                # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)               # (R, H)
    out_gate   = torch.sigmoid(out_gate_fp16)                 # (R, H)   ← half

    left_fp16  = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_fp16.view(B, N, N, hidden)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)  # (B, N, N, H)

    # view as (B*H, N, N) where each head is a separate batch
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                       # (B*H, N, N)

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate (custom Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()           # (R, H) half
    out_norm_gate_fp16 = torch.empty_like(out_flat)                # (R, H) half

    BLOCK_M = 128          # rows per program – chosen to keep occupancy high
    BLOCK_D = 128          # columns processed per inner loop (≤ hidden)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],    # γ, FP32
        weights["to_out_norm.bias"],      # β, FP32
        out_gate,                         # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()               # (dim, H)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)      # (R, dim) half

    # reshape to [B, N, N, dim] and cast to float32 (as in the reference)
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
