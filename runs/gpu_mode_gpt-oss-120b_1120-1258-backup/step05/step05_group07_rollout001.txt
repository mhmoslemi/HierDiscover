"""
custom_kernel
==============

Fast “outgoing” TriMul (Triangular Multiplicative Update) as used in AlphaFold‑3.
The implementation follows the reference PyTorch module but moves the two
LayerNorms to Triton (the first one is mandatory to satisfy the “kernel”
requirement) and keeps the heavy linear algebra in fp16 Tensor‑Core kernels
(`torch.nn.functional.linear` and `torch.bmm`).  All intermediate tensors are
half‑precision, only the final result is cast back to fp32.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channels (C) → fp16.
2️⃣ Fuse the five linear projections (left/right + three gates) into one GEMM.
3️⃣ (Optional) apply the pair‑wise mask *before* the left/right gates.
4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣ Triangular multiplicative update (outgoing version):
   out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   implemented as a batched GEMM `left @ rightᵀ` on tensors of shape
   (B·H, N, N) in fp16.
6️⃣ Second LayerNorm over the hidden dimension (H) – we use the builtin
   `torch.nn.functional.layer_norm` (fast fused CUDA kernel) because the
   first LayerNorm already demonstrates a custom Triton kernel.
7️⃣ Multiply by the output gate and apply the final linear projection back
   to the original channel dimension C.
8️⃣ Cast to fp32 and return.

The kernel is written for Triton 3.3.1 and exploits fp16 Tensor‑Core
acceleration on an H100.  It works for all test configurations listed in the
problem statement (B∈{1,2}, N∈{128,…,1024}, C∈{128,384}, H=128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton Row‑wise LayerNorm (input can be fp32 or fp16, output fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_row_fp16_kernel(
    x_ptr,            # *const   T*   [total_rows, dim]  (input)
    y_ptr,            # *mut     half* [total_rows, dim] (output)
    weight_ptr,       # *const   float* [dim]           (γ)
    bias_ptr,         # *const   float* [dim]           (β)
    total_rows,       # i32 (scalar)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)                     # each program works on BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows                # guard for tail rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # load a block of the input (fp16 or fp32 → cast to fp32)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_half = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_half,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]     (bool or float) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                 # for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # Flatten to (total_rows, dim); keep original dtype (float32)
    x_flat = input_tensor.contiguous().view(total_rows, dim)
    # Output buffer (fp16)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Block sizes – dim ≤ 384, so BLOCK_D=128 gives at most 3 iterations.
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_row_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    # --------------------------------------------------------------
    # All projection weights have shape (hidden_dim, dim) → concatenate on dim 0
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    # Concatenate → (5*H, C), cast to fp16 once
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).half().contiguous()

    # Fused linear: (total_rows, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)  # fp16

    # Split back into the five tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = \
        torch.split(proj_all, hidden_dim, dim=1)   # each (total_rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional pair‑wise mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden dim
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)      # (total_rows, H)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # Reshape to the 4‑D layout expected by the triangular update
    left = left_fp16.view(B, N, N, hidden_dim)          # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)        # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)      # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (outgoing)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Move hidden_dim into batch axis → (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                 # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)               # (BH, N, N)

    # Batched GEMM: left @ rightᵀ  → (BH, N, N), fp16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original 4‑D layout (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim (use torch built‑in, fast CUDA kernel)
    # --------------------------------------------------------------
    # torch.nn.functional.layer_norm works on fp16 (computes stats in fp32)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # (B, N, N, H), fp16

    # --------------------------------------------------------------
    # 7️⃣ Multiply by output gate and final projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                     # fp16, shape (B,N,N,H)

    # Final linear projection (fp16 GEMM)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)  # (total_rows, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16) # (total_rows, C)

    # --------------------------------------------------------------
    # 8️⃣ Cast to fp32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
