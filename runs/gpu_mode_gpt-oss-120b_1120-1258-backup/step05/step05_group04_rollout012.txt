Highly‑optimized Triton + PyTorch forward of the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Input LayerNorm – a custom Triton kernel that normalises each
    row of the flattened [B·N·N, C] tensor and writes the result as FP16.

2️⃣  Five linear projections are fused into a single GEMM (half precision).
    The result is split into:
        left_proj, right_proj, left_gate, right_gate, out_gate.

3️⃣  The three element‑wise steps
        * apply the pairwise mask   (broadcast over the hidden dim)
        * sigmoid‑gate the left/right projections
        * multiply the gates with the projected tensors

    This reduces the memory traffic from three passes to a single pass.

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    channel dimension.  This eliminates two extra memory‑bound passes.

The whole pipeline stays on‑device and returns a float‑32 tensor

def layernorm_fp16_kernel(
    x_ptr,          # *const float*  [total_rows, dim]   (FP32 input)
    y_ptr,          # *mut   half*   [total_rows, dim]   (FP16 output)
    weight_ptr,     # *const float*  [dim]               (γ)
    bias_ptr,       # *const float*  [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass: compute mean & variance (single global‑memory read)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1/√(var+eps)

    # -----------------------------------------------------------------
    # 2nd pass: normalise, apply γ/β and store as FP16
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]         # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2) Fuse mask‑application, sigmoid‑gating and element‑wise product
#    (left & right branches) – one global‑memory load per tensor.
# ----------------------------------------------------------------------
@triton.jit
def gate_and_mask_kernel(
    proj_ptr,    # *const half*  [total_rows, hidden]  (left or right proj)
    gate_ptr,    # *const half*  [total_rows, hidden]  (left or right gate)
    mask_ptr,    # *const half*  [total_rows]         (1.0 or 0.0 mask)
    out_ptr,     # *mut   half*  [total_rows, hidden]  (gated output)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # Load the broadcast mask (scalar per row)
    mask_val = tl.load(mask_ptr + offs_m,
                       mask=mask_m,
                       other=0.0)                       # [BLOCK_M]

    for d in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = d * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < hidden

        # Load proj and gate
        proj = tl.load(proj_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0)                        # [BLOCK_M, BLOCK_H]
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0)

        # sigmoid(gate)  = 1/(1+exp(-x))
        gate_sig = 1.0 / (1.0 + tl.exp(-gate))

        # Apply mask (broadcast) and gating
        out = proj * gate_sig * mask_val[:, None]

        tl.store(out_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                 out,
                 mask=mask_m[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 6) Fused output LayerNorm + out‑gate + final linear projection
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden]   (pre‑norm)
    gate_ptr,            # *const half*  [total_rows, hidden]   (out_gate, sigmoid’ed)
    norm_w_ptr,          # *const float* [hidden]               (γ of to_out_norm)
    norm_b_ptr,          # *const float* [hidden]               (β of to_out_norm)
    out_w_ptr,           # *const half*  [dim, hidden]          (to_out weight)
    out_ptr,             # *mut   float* [total_rows, dim]      (final FP32 output)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # output‑channel chunk
):
    pid_m = tl.program_id(0)          # row block
    pid_d = tl.program_id(1)          # output‑channel block
    row_start = pid_m * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Load the whole hidden vector for the selected rows
    # -----------------------------------------------------------------
    offs_h = tl.arange(0, hidden)                               # [hidden]
    mask_h = offs_h < hidden

    x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                       # (BLOCK_M, hidden)
    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                       # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Compute mean / rstd per row (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)                                 # (BLOCK_M)
    sum_sq  = tl.sum(x * x, axis=1)                             # (BLOCK_M)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Apply LayerNorm weight/bias and the out‑gate
    # -----------------------------------------------------------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)

    norm = (x - mean[:, None]) * rstd[:, None]                     # (BLOCK_M, hidden)
    norm = norm * w_norm[None, :] + b_norm[None, :]                # γ,β
    gated = norm * g                                                # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Output‑channel block
    # -----------------------------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)                       # [BLOCK_D]
    mask_d = offs_d < dim

    # Load slice of the final weight matrix (shape [dim, hidden])
    # We need its transpose (hidden, BLOCK_D)
    out_w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                      mask=mask_h[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)                  # (hidden, BLOCK_D)

    # Mat‑mul: (BLOCK_M, hidden) × (hidden, BLOCK_D) → (BLOCK_M, BLOCK_D)
    out_chunk = tl.dot(gated, out_w_T)                            # (BLOCK_M, BLOCK_D)

    # Store result (FP32)
    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool or float) – ignored when config["nomask"]=True
        - weights_dict  : model parameters (float32)
        - config_dict   : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (optional)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim    = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                       # rows for flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)

    # Tuned block sizes (dim ≤ 384 → BLOCK_D = 32 works well)
    BLOCK_M = 128
    BLOCK_D = 32
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·hidden) – single large GEMM
    # --------------------------------------------------------------
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                   # (5·H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)               # (R, 5·H)   FP16

    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden, dim=1
    )                                                        # each (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Gate + mask fusion (one Triton kernel)
    # --------------------------------------------------------------
    # Prepare mask (broadcast over hidden dim).  If no mask, use all‑ones.
    if nomask:
        mask_fp16 = torch.ones(total_rows, dtype=torch.half, device=device)
    else:
        mask_fp16 = mask.view(total_rows).to(torch.float16)

    # Allocate output tensors for the gated left / right branches
    left  = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Kernel dimensions
    BLOCK_M_G = 128          # rows per program
    BLOCK_H_G = 64           # hidden dimension chunk
    grid_gate = (triton.cdiv(total_rows, BLOCK_M_G),)

    # Left branch
    gate_and_mask_kernel[grid_gate](
        left_proj,
        left_gate,
        mask_fp16,
        left,
        total_rows,
        hidden,
        BLOCK_M=BLOCK_M_G,
        BLOCK_H=BLOCK_H_G,
    )
    # Right branch
    gate_and_mask_kernel[grid_gate](
        right_proj,
        right_gate,
        mask_fp16,
        right,
        total_rows,
        hidden,
        BLOCK_M=BLOCK_M_G,
        BLOCK_H=BLOCK_H_G,
    )

    # out_gate still needs a sigmoid (cheap, separate pass)
    out_gate = torch.sigmoid(out_gate)                     # (R, H) half

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # reshape for bmm: (B·H, N, N) where right is transposed on the last two axes
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                # (B·H, N, N)  FP16

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Fused output LayerNorm + out‑gate + final projection
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()    # (R, H) half
    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    # Grid: rows × output‑channel blocks
    BLOCK_M_O = 128
    BLOCK_D_O = 64
    grid_out = (
        triton.cdiv(total_rows, BLOCK_M_O),
        triton.cdiv(dim, BLOCK_D_O),
    )
    layernorm_gate_linear_kernel[grid_out](
        out_flat,                                         # x_ptr
        out_gate,                                         # gate_ptr (sigmoid'ed)
        weights["to_out_norm.weight"],                    # norm γ
        weights["to_out_norm.bias"],                      # norm β
        weights["to_out.weight"].half().contiguous(),    # final linear weight (dim, hidden)
        final_fp32,                                       # output buffer
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M_O,
        BLOCK_D=BLOCK_D_O,
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape to final tensor shape (float32)
    # --------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)                 # [B, N, N, dim] float32
    return final
