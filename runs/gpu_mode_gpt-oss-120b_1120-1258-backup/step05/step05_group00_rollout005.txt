"""
Custom Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (shape [B·N·N, dim]).
   A tiny Triton kernel computes mean / variance per row in FP32
   and writes the normalized result in FP16.

2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single half‑precision GEMM:
       [R, dim] @ [4·H, dim].T → [R, 4·H]   (R = B·N·N, H = hidden_dim)
   The result is split into the four tensors.

3️⃣ Compute the output‑gate linear layer separately (also FP16).

4️⃣ (Optional) Apply the mask (broadcast over the head dimension) to the
   two projected tensors. Apply the three sigmoid gates in‑place and
   obtain the gated tensors `left` and `right`.

5️⃣ Triangular multiplicative update (the N³ part):
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   Implemented as a batched half‑precision GEMM:
       (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)

6️⃣ Second LayerNorm over the hidden dimension (again Triton, FP16).

7️⃣ Multiply by the previously computed output‑gate and a final
   FP16 linear projection back to `dim`.  Cast the result to float32.

All heavy normalisations are done with Triton; the large dense
matrix multiplications (projections, N³ update, final projection) use
cuBLAS Tensor‑Cores via `torch.nn.functional.linear` / `torch.bmm`
for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)  γ
    bias_ptr,       # const   * [dim]               (float32)  β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # cols per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Accumulation is performed in FP32, output is FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the feature dimension in blocks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)         # fp32 load
        sum_val += tl.sum(vals, axis=1)                     # sum over d
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------

    """

    Args:

            "left_proj.weight", "right_proj.weight",

            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"

    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------

                              dtype=torch.float16,
                              device=device)

    BLOCK_D = 32    # feature block
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],      # γ (fp32)
        weights["norm.bias"],        # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gating into a single GEMM
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices needed for left/right + gates
    # Each weight is [hidden_dim, dim] (out_features, in_features)
    w_concat_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [4·H, dim]

    # One half‑precision GEMM: [R, dim] @ [4·H, dim].T → [R, 4·H]
    proj_gate = F.linear(x_norm_half, w_concat_4h)        # fp16

    # Split the concatenated result
    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]              # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]            # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]            # [R, H]

    # Separate linear for the output gate
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B,N,N] → flatten to [R,1] (fp16) and broadcast
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid to save memory
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate        # [R, H] fp16
    right = right_proj_raw * right_gate      # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ triangular multiplicative update
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # Move the head dimension into the batch axis for a batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)  # [B·H, N, N]

    # Batched half‑precision matrix multiplication:
    #   (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (row‑wise) → FP16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast to fp32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise gating (fp16)

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                   # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
