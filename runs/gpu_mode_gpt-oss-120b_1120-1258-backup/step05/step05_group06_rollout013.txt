def fused_norm_linear_kernel(
    x_ptr, gate_ptr, out_ptr,
    gamma_ptr, beta_ptr, w_out_ptr,
    total_rows, hidden, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # First pass: compute mean and var across hidden dimension.
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    k = 0
    while k < hidden:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        # load x
        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # shape [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

        k += BLOCK_K

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: compute output accumulation.
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # compute column offsets once
    offs_n = tl.arange(0, BLOCK_N)
    mask_n = offs_n < dim

    # Second pass for K
    k = 0
    while k < hidden:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        # load x values
        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_K]

        # Normalize
        norm = (x - mean[:, None]) * rstd[:, None]  # [M,K]

        # Apply gamma (weight_norm) and beta
        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)
        beta = tl.load(beta_ptr + offs_k, mask=mask_k, other=0.0)

        # gamma,beta may be float16 or float32; ensure float32
        gamma_f = gamma.to(tl.float32)
        beta_f = beta.to(tl.float32)

        norm = norm * gamma_f[None, :] + beta_f[None, :]  # [M,K]

        # Apply out_gate (sigmoid)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M,K]

        norm = norm * gate

        # Load weights for output projection: shape (hidden, dim)
        # Our weight_out_ptr has shape (dim, hidden); we want [hidden, dim]
        # Compute tile (K,N):
        w = tl.load(w_out_ptr + offs_k[:, None] * dim + offs_n[None, :],
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)   # [K,N]

        # Accumulate
        acc += tl.dot(norm, w)  # (M,K) dot (K,N) -> (M,N)

        k += BLOCK_K

    # Store results
    out_ptrs = out_ptr + offs_m[:, None] * dim + offs_n[None, :]

def custom_kernel(data):
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B,N,_,_ = inp.shape
    total_rows = B * N * N

    # Step1: Input layernorm using torch.nn.functional.layer_norm
    inp_flat = inp.view(-1, dim)
    x_norm = torch.nn.functional.layer_norm(inp_flat, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm_fp16 = x_norm.half() # shape (total_rows, dim)

    # Step2: concat weights and linear projection
    concat_w = torch.cat(
            [weights["left_proj.weight"],
             weights["right_proj.weight"],
             weights["left_gate.weight"],
             weights["right_gate.weight"],
             weights["out_gate.weight"]], dim=0).half()
    proj_all = torch.nn.functional.linear(x_norm_fp16, concat_w)   # shape (total_rows, 5*hidden)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(proj_all, hidden, dim=1)

    # Optional mask
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)   # shape (total_rows, hidden)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # TriMul
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    left_mat = left.permute(0, 3, 1, 2).reshape(B*hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B*hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*hidden, N, N)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # flatten for fused kernel
    out_flat = out.view(total_rows, hidden).contiguous()
    # ready out_gate flatten already (total_rows, hidden) half

    # allocate output
    final_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    # kernel launch
    BLOCK_M = 64
    BLOCK_K = 32   # hidden <= 128
    BLOCK_N = 128 if dim <= 128 else 256 if dim <= 256 else 384
    # Determine BLOCK_N = 128 maybe fine for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_norm_linear_kernel[grid](
        out_flat,               # x_ptr
        out_gate,               # gate_ptr
        final_fp16,             # out_ptr
        weights["to_out_norm.weight"].half(),
        weights["to_out_norm.bias"].half(),
        weights["to_out.weight"].half(),
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
    )

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

def fused_norm_linear_kernel(
    x_ptr, gate_ptr, out_ptr,
    gamma_ptr, beta_ptr, w_out_ptr,
    total_rows, hidden, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean and variance over hidden
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    k = 0
    while k < hidden:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

        k += BLOCK_K

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Loop over column tiles
    n_start = 0
    while n_start < dim:
        offs_n = n_start + tl.arange(0, BLOCK_N)
        mask_n = offs_n < dim

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        # loop over K to accumulate
        k = 0
        while k < hidden:
            offs_k = k + tl.arange(0, BLOCK_K)
            mask_k = offs_k < hidden

            # load x
            x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

            # normalize
            norm = (x - mean[:, None]) * rstd[:, None]

            # gamma & beta
            gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)
            beta = tl.load(beta_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)
            norm = norm * gamma[None, :] + beta[None, :]

            # gate
            gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)
            norm = norm * gate

            # weight_out tile (K x N)
            w = tl.load(w_out_ptr + offs_k[:, None] * dim + offs_n[None, :],
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

            # accumulate
            acc += tl.dot(norm, w)  # (M,K)*(K,N) -> (M,N)
            k += BLOCK_K

        # store results
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_n[None, :]
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_n[None, :])

        n_start += BLOCK_N

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 “Tri‑multiplicative update”).

Algorithm
---------
1️⃣  Input LayerNorm (FP32 → FP16) – we reuse PyTorch’s native
    `torch.nn.functional.layer_norm`, which is a heavily‑optimised
    fused kernel on H100.
2️⃣  Fuse the five linear maps (two projections + three gates) into a
    single FP16 GEMM by concatenating their weight matrices.
3️⃣  (Optional) apply the binary mask to the left/right projections.
4️⃣  Apply sigmoid to the three gate tensors and perform element‑wise gating.
5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched FP16 matrix‑multiply (`torch.bmm`);
    each head (`hidden_dim`) is treated as an independent batch.
6️⃣  **Fused** LayerNorm + out‑gate + final linear projection.
    This Triton kernel:
      • computes mean/variance per row (FP32 reduction),
      • normalises and applies `to_out_norm.{weight,bias}`,
      • multiplies by the sigmoid‑ed out‑gate,
      • projects to the model dimension (`dim`) with `to_out.weight`,
      • writes the result in FP16.
    The fusion removes an intermediate tensor write/read and
    reduces memory traffic.
7️⃣  Cast the result back to float32 (original module output type).

All heavy linear algebra still uses Tensor‑Core‑accelerated cuBLAS.
Only the first and the fused‑norm‑linear stages are written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# 6) Fused LayerNorm + out‑gate + final projection (all in FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden] (input after TriMul)
    gate_ptr,            # *const half*  [total_rows, hidden] (sigmoid(out_gate))
    out_ptr,             # *mut   half*  [total_rows, dim]    (final output)
    gamma_ptr,           # *const half*  [hidden]            (to_out_norm.weight)
    beta_ptr,            # *const half*  [hidden]            (to_out_norm.bias)
    w_out_ptr,           # *const half*  [hidden, dim]       (to_out.weight transposed)
    total_rows: tl.int32,
    hidden: tl.int32,
    dim: tl.int32,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 64)
    BLOCK_K: tl.constexpr,   # reduction tile over hidden (e.g. 32)
    BLOCK_N: tl.constexpr,   # columns per program (e.g. 128)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # 1️⃣ First pass: compute mean & variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    k = 0
    while k < hidden:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

        k += BLOCK_K

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # 2️⃣ Outer loop over output‑feature tiles (dim)
    # ---------------------------------------------------------------
    n_start = 0
    while n_start < dim:
        offs_n = n_start + tl.arange(0, BLOCK_N)
        mask_n = offs_n < dim

        # accumulator for the current (M,N) tile
        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        # -----------------------------------------------------------
        #   Inner reduction over hidden dimension (K)
        # -----------------------------------------------------------
        k = 0
        while k < hidden:
            offs_k = k + tl.arange(0, BLOCK_K)
            mask_k = offs_k < hidden

            # load raw input
            x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)           # [M,K]

            # normalise
            norm = (x - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

            # γ (weight) & β (bias) for the second LayerNorm
            gamma = tl.load(gamma_ptr + offs_k,
                            mask=mask_k, other=0.0).to(tl.float32)
            beta  = tl.load(beta_ptr  + offs_k,
                            mask=mask_k, other=0.0).to(tl.float32)
            norm = norm * gamma[None, :] + beta[None, :] # [M,K]

            # multiply by the *already‑sigmoided* out‑gate
            gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)       # [M,K]
            norm = norm * gate

            # load weight slice: shape (K, N) from the transposed weight matrix
            w = tl.load(w_out_ptr + offs_k[:, None] * dim + offs_n[None, :],
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)           # [K,N]

            # accumulate GEMM contribution
            acc += tl.dot(norm, w)                         # (M,K)*(K,N) → (M,N)

            k += BLOCK_K

        # -----------------------------------------------------------
        #   Store the result tile (cast back to FP16)
        # -----------------------------------------------------------
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_n[None, :]
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_n[None, :])

        n_start += BLOCK_N


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict) where
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {
                "dim"        : int   (C),
                "hidden_dim" : int   (H),
                "nomask"     : bool,
                "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor : shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim     = cfg["dim"]               # model dimension C
    hidden  = cfg["hidden_dim"]       # hidden dimension H
    eps     = cfg.get("eps", 1e-5)
    nomask  = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – use PyTorch’s fast kernel
    # ------------------------------------------------------------------
    inp_flat = inp.view(-1, dim)                     # [R, C]  (FP32)
    x_norm = torch.nn.functional.layer_norm(
        inp_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm_fp16 = x_norm.half()                      # [R, C]  (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Concatenated linear projection (covers proj & gates)
    # ------------------------------------------------------------------
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                            # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)        # (R, 5*H)

    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        proj_all, hidden, dim=1
    )                                                # each (R, H)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask if required (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)          # (R, H)
    right_gate = torch.sigmoid(right_gate_pre)         # (R, H)
    out_gate   = torch.sigmoid(out_gate_pre)           # (R, H) – for the final Stage‑6

    left  = left_proj  * left_gate                     # (R, H)
    right = right_proj * right_gate                    # (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched FP16 GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden)                # (B, N, N, H)
    right = right.view(B, N, N, hidden)               # (B, N, N, H)

    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)   # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)           # (B*H, N, N)   FP16
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate + final projection (Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()      # (R, H) half
    # out_gate is already sigmoid‑ed and half‑precision
    out_gate_flat = out_gate

    # Prepare the transposed final weight: (hidden, dim)
    w_out_T = weights["to_out.weight"].t().contiguous().half()   # (H, C)

    # Allocate output buffer (FP16)
    final_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    # Block sizes – tuned for H100 Tensor‑Core occupancy
    BLOCK_M = 64               # rows per program
    BLOCK_K = 32               # tile over hidden (H ≤ 128)
    BLOCK_N = 128              # columns per program (covers up to 384 dims with 3 tiles)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_norm_gate_linear_kernel[grid](
        out_flat,                               # x_ptr
        out_gate_flat,                          # gate_ptr
        final_fp16,                             # out_ptr
        weights["to_out_norm.weight"].half(),   # gamma
        weights["to_out_norm.bias"].half(),     # beta
        w_out_T,                                # w_out (transposed)
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Cast back to FP32 (original module output type)
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
