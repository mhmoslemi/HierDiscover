```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton layernorm kernel (row-wise)
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Accumulate mean/variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Triton batched GEMM (A @ Bᵀ) where A,B are [B, N, N] half
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute for each batch:
        C = A @ Bᵀ
    A : [batch, M, K]   (half)
    B : [batch, N, K]   (half)   # stored in row‑major order, will be read transposed
    C : [batch, M, N]   (half)
    """
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # col block
    pid_b = tl.program_id(2)   # batch index

    # start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * batch_stride_a
    b_batch = b_ptr + pid_b * batch_stride_b
    c_batch = c_ptr + pid_b * batch_stride_c

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + k_off[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_batch + k_off[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # a,b are half; dot produces fp32 accumulation
        acc += tl.dot(a, b)

    # write result
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Algorithm (mixed‑precision):
    1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16).
    2️⃣ Fuse left/right projections and their two gates into a single GEMM.
    3️⃣ Compute the output gate with another GEMM.
    4️⃣ Apply optional mask, sigmoid gates and element‑wise gating.
    5️⃣ Batched half‑precision GEMM implementing the N³ “tri‑multiply”:
       C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
    6️⃣ Second LayerNorm over the hidden dimension.
    7️⃣ Multiply by the output gate and a final linear projection.
    All heavy per‑row normalisations are performed by a custom Triton kernel.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N      # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm  (fp32 -> fp16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ, float32
        weights["norm.bias"],     # β, float32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # ------------------------------------------------------------------
    H = hidden_dim
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # shape [4*H, dim]

    # [R, dim] @ [4*H, dim].T → [R, 4*H]
    proj_gate = F.linear(x_norm, w_proj_gate)   # fp16

    left_proj_raw   = proj_gate[:, 0:H]                # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]              # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]            # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]            # [R, H]

    # output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)
    )                                         # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B,N,N] → flat [R,1]
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate    # [R, H] fp16
    right = right_proj_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for batched N³ multiplication
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring hidden dimension to batch axis
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch_dim = B * H
    left_mat = left_mat.view(batch_dim, N, N)   # [BH, N, N]  fp16
    right_mat = right_mat.view(batch_dim, N, N) # [BH, N, N]  fp16

    out_mat = torch.empty_like(left_mat)        # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM C = A @ Bᵀ  (TriMul core)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_dim,
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,            # M
        N,            # N
        N,            # K
        left_mat.stride(1),   # stride_am (row stride)
        left_mat.stride(2),   # stride_ak (col stride)
        right_mat.stride(2),  # stride_bk (K stride after transposition)
        right_mat.stride(1),  # stride_bn (N stride after transposition)
        out_mat.stride(1),    # stride_cm
        out_mat.stride(2),    # stride_cn
        left_mat.stride(0),   # batch stride a
        right_mat.stride(0),  # batch stride b
        out_mat.stride(0),    # batch stride c
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate          # [B,N,N,H] fp16

    final = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)
    )                                        # fp16

    final = final.view(B, N, N, dim).to(torch.float32)
    return final

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass – mixed‑precision Triton‑accelerated.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the [B,N,N,dim] input (fp32 → fp16) – custom Triton kernel.
2️⃣ Fuse the four linear layers (left/right projection + their two gates) into one
   GEMM:   X_norm @ W_concat.T → [R,4*H]  where  R = B·N·N.
   Split the result into left_proj, right_proj, left_gate, right_gate.
3️⃣ Compute the output gate with a second GEMM.
4️⃣ (Optional) apply the pair‑wise mask, then sigmoid the three gates and
   multiply left/right by their respective gates.
5️⃣ Core N³ “tri‑multiply’’:
   For each head h we need   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched half‑precision GEMM C = A @ Bᵀ with batch size B·H.
   It is implemented with a custom Triton kernel that processes all batches
   in a single launch, avoiding the many small cuBLAS calls.
6️⃣ Second LayerNorm over the hidden dimension H – same Triton kernel.
7️⃣ Multiply by the output gate and linearly project back to `dim`
   (half‑precision GEMM).  Cast the final result to float32.

All heavy per‑row normalisations are done by a Triton kernel,
while the large matrix multiplications use either cuBLAS (via torch.nn.functional)
or a fused Triton GEMM for the N³ update.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const   * [total_rows, dim]   (float32 or float16)
    y_ptr,                # mutable * [total_rows, dim]   (float16)
    weight_ptr,           # const   * [dim]               (float32)
    bias_ptr,             # const   * [dim]               (float32)
    total_rows,           # int (runtime)
    dim,                  # int (runtime)
    eps,                  # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Accumulation is fp32, output stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean and variance per row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (half → half)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes for every batch:
        C = A @ Bᵀ
    A : [batch, M, K]   (fp16)
    B : [batch, N, K]   (fp16) – interpreted transposed inside the kernel.
    C : [batch, M, N]   (fp16)
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    a_batch = a_ptr + pid_b * batch_stride_a
    b_batch = b_ptr + pid_b * batch_stride_b
    c_batch = c_ptr + pid_b * batch_stride_c

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension (tiles of size BLOCK_K)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + k_off[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_batch + k_off[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # a,b are fp16; tl.dot accumulates in fp32
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters
        - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],            # γ  (float32)
        weights["norm.bias"],              # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their two gates (4·H outputs)
    # ------------------------------------------------------------------
    w_concat_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # [4*H, dim]

    proj_gate = F.linear(x_norm, w_concat_4h)         # [R, 4*H] fp16

    left_proj_raw  = proj_gate[:, 0:H]                # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]              # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]            # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]            # [R, H]

    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)
    )                                                # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates, element‑wise gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)           # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # move hidden dimension into the batch axis
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch_dim = B * H
    left_mat = left_mat.view(batch_dim, N, N)          # [BH, N, N] fp16
    right_mat = right_mat.view(batch_dim, N, N)        # [BH, N, N] fp16

    out_mat = torch.empty_like(left_mat)               # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: C = A @ Bᵀ   (TriMul core)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_dim,
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M
        N,               # N (output cols)
        N,               # K
        left_mat.stride(1),              # stride_am (row stride)
        left_mat.stride(2),              # stride_ak (col stride)
        right_mat.stride(2),             # stride_bk (K stride after transposition)
        right_mat.stride(1),             # stride_bn (N stride after transposition)
        out_mat.stride(1),               # stride_cm
        out_mat.stride(2),               # stride_cn
        left_mat.stride(0),              # batch stride a
        right_mat.stride(0),             # batch stride b
        out_mat.stride(0),               # batch stride c
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate          # [B,N,N,H] fp16

    final = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)
    )                                        # [R, dim] fp16

    final = final.view(B, N, N, dim).to(torch.float32)
    return final
