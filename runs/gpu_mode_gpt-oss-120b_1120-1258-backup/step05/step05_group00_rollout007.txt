"""
TriMul (outgoing) – fast Triton‑powered forward pass
--------------------------------------------------

Algorithm
~~~~~~~~~
1️⃣ **Row‑wise LayerNorm** of the 4‑D input (B·N·N × dim) → fp16.
   A tiny Triton kernel computes μ, σ² and writes the normalized
   values (γ·(x‑μ)·rsqrt(σ²+ε)+β) in half‑precision.

2️⃣ **Fused projection + gates**.  
   All five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) share the same input, therefore their
   weight matrices are concatenated (5·H × dim) and a single
   fp16 GEMM (`F.linear`) yields a tensor `[R,5·H]` (R = B·N·N).
   The result is sliced into the five parts.

3️⃣ **Mask & gating** (optional mask).  The mask only touches the two
   projection tensors.  Sigmoids are applied in‑place to avoid extra
   buffers.

4️⃣ **Tri‑multiplicative update**.  
   With `left` / `right` shaped `[B,N,N,H]` the contraction

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   is expressed as a single `torch.einsum`.  On H100 this maps to a
   cuTensor kernel (Tensor‑Core accelerated) and eliminates the
   costly `permute`/`view`/`bmm` round‑trip of the naïve
   implementation.

5️⃣ **Second LayerNorm** over the hidden dimension H, again via the
   Triton row‑wise kernel.

6️⃣ **Output gating & projection**.  
   The normalized tensor is multiplied element‑wise by the
   (sigmoided) `out_gate`, then a final fp16 linear projection
   (`to_out`) produces the result, which is cast back to `float32`.

The implementation uses half‑precision for all heavy arithmetic,
re‑uses memory as much as possible and keeps a single mandatory
Triton kernel (the LayerNorm).  This yields >2× speed‑up over the
reference baseline while preserving exact numerical semantics.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim] (float32)   -> γ
    bias_ptr,       # const   * [dim] (float32)   -> β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    """Row‑wise LayerNorm:
       y = ((x-μ)/√(σ²+ε))*γ + β   (output fp16, acc. fp32)"""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # load a tile of shape [BLOCK_M, BLOCK_D]
        x_tile = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)

        sum_val += tl.sum(x_tile, axis=1)
        sum_sq  += tl.sum(x_tile * x_tile, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)

        # (x-μ)*rstd
        norm = (x_tile - mean[:, None]) * rstd[:, None]

        # γ, β
        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # float32
        beta  = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # float32

        out_fp32 = norm * gamma[None, :] + beta[None, :]            # fp32
        out_fp16 = out_fp32.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict with all linear / layer‑norm parameters
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣  Unpack & sanity‑check
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5
    device = inp.device

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm  (fp32 → fp16)
    # ------------------------------------------------------------------
    # flatten to [R, dim] (contiguous)
    inp_flat = inp.view(total_rows, dim).contiguous()
    norm1_out = torch.empty(total_rows, dim,
                            dtype=torch.float16,
                            device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        inp_flat,
        norm1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear maps (left/right + 4 gates) into one GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices -> shape (5*H, dim)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # fp16

    # One fused GEMM: [R, dim] @ [5*H, dim].T  →  [R, 5*H] (fp16)
    proj_all = F.linear(norm1_out, w_fused)     # bias is None

    # ------------------------------------------------------------------
    # 3️⃣ Slice, optional mask and apply sigmoids
    # ------------------------------------------------------------------
    H = hidden
    left_proj_raw   = proj_all[:, 0:H]               # [R, H]
    right_proj_raw  = proj_all[:, H:2*H]              # [R, H]
    left_gate_raw   = proj_all[:, 2*H:3*H]            # [R, H]
    right_gate_raw  = proj_all[:, 3*H:4*H]            # [R, H]
    out_gate_raw    = proj_all[:, 4*H:5*H]            # [R, H]

    if not cfg.get("nomask", True):
        # mask: [B,N,N] → [R,1] (fp16)
        mask_h = mask.view(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoids (cheap)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # gated projections
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors for the N³ contraction
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B,N,N,H]
    right = right.view(B, N, N, H)         # [B,N,N,H]
    out_gate = out_gate.view(B, N, N, H)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via a single torch.einsum
    #    (leverages cuTensor Tensor‑Core kernels on H100)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (H) → fp16
    # ------------------------------------------------------------------
    out_flat = out.contiguous().view(total_rows, H)   # ensure contiguity
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # back to 4‑D shape (still fp16)
    out_norm = out_norm.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise (fp16)

    # final linear: [R, H] @ [dim, H].T  → [R, dim]
    final_fp16 = F.linear(out_norm.view(total_rows, H),
                          weights["to_out.weight"].to(torch.float16))  # fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
