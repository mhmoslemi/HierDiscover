@triton.jit
def apply_mask_kernel(x_ptr, mask_ptr, y_ptr, total_rows, hidden_dim,
                      BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    rows = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_valid = rows < total_rows
    
    for d in range(tl.cdiv(hidden_dim, BLOCK_D)):
        cols = d * BLOCK_D + tl.arange(0, BLOCK_D)
        col_valid = cols < hidden_dim
        
        mask_val = tl.load(mask_ptr + rows, mask=mask_valid, other=0.0)  # shape [BLOCK_M]
        # broadcast to [BLOCK_M, BLOCK_D]
        mask_val = mask_val[:, None]   # shape [BLOCK_M,1]
        x = tl.load(x_ptr + rows[:, None] * hidden_dim + cols[None, :],
                    mask=mask_valid[:, None] & col_valid[None, :],
                    other=0.0)
        y = x * mask_val
        tl.store(y_ptr + rows[:, None] * hidden_dim + cols[None, :],
                 y, mask=mask_valid[:, None] & col_valid[None, :])
```

@triton.jit
def mask_mul_kernel(tensor_ptr, mask_ptr, out_ptr, total, hidden_dim, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    rows = row_start + tl.arange(0, BLOCK_M)
    row_mask = rows < total

    for d in range(tl.cdiv(hidden_dim, BLOCK_D)):
        cols = d * BLOCK_D + tl.arange(0, BLOCK_D)
        col_mask = cols < hidden_dim
        mask_val = tl.load(mask_ptr + rows, mask=row_mask, other=0.0)
        mask_val = mask_val[:, None]
        x = tl.load(tensor_ptr + rows[:, None] * hidden_dim + cols[None,:], mask=row_mask[:, None] & col_mask[None, :], other=0.0)
        y = x * mask_val
        tl.store(out_ptr + rows[:, None] * hidden_dim + cols[None,:], y, mask=row_mask[:, None] & col_mask[None, :])

    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps=1e-5; nomask=config.get("nomask", True)

    # First LayerNorm using torch.nn.functional.layer_norm (fast)
    x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm_fp16 = x_norm.half()   # [B,N,N,C]

    # Linear projection fused
    concat_w = torch.cat([...], dim=0).half()
    proj_all = F.linear(x_norm_fp16, concat_w) # yields shape [B,N,N,5*H] as half

    # split

    # mask

        mask_half = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_half

    # gates sigmoid

    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # permute to [B, H, N, N] and contiguous
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

    # shape as [BH, N, N]
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)
    # BMM
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_apply_kernel(tensor_ptr, mask_ptr, out_ptr, total_rows, hidden_dim,
                      BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    rows = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    row_mask = rows < total_rows

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        cols = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        col_mask = cols < hidden_dim

        # Load mask values: one per row
        mask_val = tl.load(mask_ptr + rows, mask=row_mask, other=0.0)   # [BLOCK_M]
        mask_val = mask_val[:, None]                                   # broadcast

        # Load tensor values
        x = tl.load(tensor_ptr + rows[:, None] * hidden_dim + cols[None, :],
                    mask=row_mask[:, None] & col_mask[None, :],
                    other=0.0)

        y = x * mask_val
        tl.store(out_ptr + rows[:, None] * hidden_dim + cols[None, :],
                 y,
                 mask=row_mask[:, None] & col_mask[None, :])

def custom_kernel(data):
    """
    Fast forward pass of the outgoing TriMul operator (AlphaFold‑3) using
    half‑precision linear layers, cuBLAS batched GEMM and a tiny Triton kernel
    for mask‑application. The heavy LayerNorms are done with the native
    PyTorch implementation (CUDA‑optimized). The final output is float32.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32, then cast to fp16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm_fp16 = x_norm.half()  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # ------------------------------------------------------------------
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    # concatenate along output dimension (rows)
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0).half().contiguous()
    # fused projection
    proj_all = F.linear(x_norm_fp16, concat_w)   # [B, N, N, 5*H] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Split projections and gates
    # ------------------------------------------------------------------
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=3
    )   # each [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (in‑place via tiny Triton kernel)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # flatten mask to [total_rows, 1] fp16
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        # view tensors as 2‑D for the kernel
        left_view = left_proj_fp16.view(total_rows, hidden_dim)
        right_view = right_proj_fp16.view(total_rows, hidden_dim)

        BLOCK_M = 1024
        BLOCK_D = 64
        grid = (triton.cdiv(total_rows, BLOCK_M),)

        mask_apply_kernel[grid](
            left_view,
            mask_fp16,
            left_view,          # in‑place write back
            total_rows,
            hidden_dim,
            BLOCK_M=BLOCK_M,
            BLOCK_D=BLOCK_D,
        )
        mask_apply_kernel[grid](
            right_view,
            mask_fp16,
            right_view,
            total_rows,
            hidden_dim,
            BLOCK_M=BLOCK_M,
            BLOCK_D=BLOCK_D,
        )

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)          # [B, N, N, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left = left_proj_fp16 * left_gate                  # [B, N, N, H]
    right = right_proj_fp16 * right_gate               # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM: out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    #    (equivalent to left @ rightᵀ for each hidden head)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                 # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)               # [BH, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (half‑precision)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )   # still fp16

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                        # [B, N, N, H] fp16
    final_fp16 = F.linear(gated.view(total_rows, hidden_dim),
                          weights["to_out.weight"].half())  # [total_rows, C] fp16
    final = final_fp16.view(B, N, N, dim).float()       # cast back to fp32
    return final

# =============================================================================
# Custom forward kernel for the *outgoing* TriMul operator (AlphaFold‑3).
# The heavy linear algebra (GEMM / BMM) is handled by cuBLAS/Tensor‑Core.
# A tiny Triton kernel is used to apply the optional pair‑wise mask in‑place.
# LayerNorms are performed with the native PyTorch implementation (highly
# optimized, fp32 for the first LN and fp16 for the second).  All
# computations after the first LN use FP16, and the final result is cast back
# to FP32.
# =============================================================================
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: multiply a 2‑D tensor (rows×cols) by a per‑row scalar mask.
# The mask is a contiguous tensor of shape [rows, 1] (fp16).  The operation
# is in‑place: out_ptr may be the same as tensor_ptr.
# -------------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    tensor_ptr,        # fp16  [rows, cols]
    mask_ptr,          # fp16  [rows, 1]
    out_ptr,           # fp16  [rows, cols] (may be the same as tensor_ptr)
    rows: tl.constexpr,
    cols: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # cols processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # Iterate over the column dimension in BLOCK_D sized chunks
    for d in range(0, tl.cdiv(cols, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)         # [BLOCK_D]
        mask_d = offs_d < cols

        # Load one mask value per row and broadcast across the current block
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
        mask_val = mask_val[:, None]                                 # [BLOCK_M,1]

        # Load the corresponding block of the tensor
        x = tl.load(
            tensor_ptr + offs_m[:, None] * cols + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )  # [BLOCK_M, BLOCK_D]

        # Apply mask
        y = x * mask_val

        # Store result (in‑place if out_ptr == tensor_ptr)
        tl.store(
            out_ptr + offs_m[:, None] * cols + offs_d[None, :],
            y,
            mask=mask_m[:, None] & mask_d[None, :],
        )


def custom_kernel(data):
    """
    Fast forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights       : dict mapping layer‑name → torch.Tensor (float32)
            - config        : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32, then cast to fp16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B, N, N, C]  fp32
    x_norm_fp16 = x_norm.half()         # [B, N, N, C]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right + three gates)
    # ------------------------------------------------------------------
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    # concat on the output‑feature dimension (rows of the weight matrix)
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0).half().contiguous()
    # fused projection:  [B,N,N,C] @ [C, 5*H]^T → [B,N,N,5*H]
    proj_all = F.linear(x_norm_fp16, concat_w)

    # ------------------------------------------------------------------
    # 3️⃣ Split into the five tensors
    # ------------------------------------------------------------------
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=3
    )  # each: [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Apply optional mask (in‑place) using a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # shape [total_rows, 1] fp16
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)

        # view as 2‑D tensors for the kernel
        left_view = left_proj_fp16.view(total_rows, hidden_dim)
        right_view = right_proj_fp16.view(total_rows, hidden_dim)

        BLOCK_M = 1024
        BLOCK_D = 64
        grid = (triton.cdiv(total_rows, BLOCK_M),)

        mask_apply_kernel[grid](
            left_view,
            mask_fp16,
            left_view,                     # in‑place write‑back
            total_rows,
            hidden_dim,
            BLOCK_M=BLOCK_M,
            BLOCK_D=BLOCK_D,
        )
        mask_apply_kernel[grid](
            right_view,
            mask_fp16,
            right_view,
            total_rows,
            hidden_dim,
            BLOCK_M=BLOCK_M,
            BLOCK_D=BLOCK_D,
        )

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) + element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)    # [B, N, N, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left = left_proj_fp16 * left_gate            # [B, N, N, H]
    right = right_proj_fp16 * right_gate         # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Triangular multiplicative update (batched GEMM)
    #        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    #        = left @ rightᵀ for each hidden head
    # ------------------------------------------------------------------
    # rearrange to (B, H, N, N) → view as (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                 # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)               # [BH, N, N]

    # batched matrix multiplication: left @ rightᵀ  →  [BH, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )   # remains fp16

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # [B, N, N, H] fp16
    final_fp16 = F.linear(
        gated.view(total_rows, hidden_dim),
        weights["to_out.weight"].half(),
    )  # [total_rows, C] fp16

    # Cast back to fp32 for the model output
    final = final_fp16.view(B, N, N, dim).float()
    return final
