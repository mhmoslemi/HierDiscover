@triton.jit
def final_proj_ln_gate_kernel(
    out_ptr, out_gate_ptr, gamma_ptr, beta_ptr, weight_ptr, out_ptr_out,
    total_rows, hidden_dim, out_dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_O: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # allocate per row accumulators for mean and sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # First pass: compute mean & var over hidden dim
    for ho in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = ho * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        # load out val half
        out_vals = tl.load(out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)  # [BLOCK_M, BLOCK_H]
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)
    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute final output
    # We'll allocate output accumulators for each output block
    for oo in range(0, tl.cdiv(out_dim, BLOCK_O)):
        # init accumulator for each row and each output feature
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for ho in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = ho * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            out_vals = tl.load(out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                               mask=mask_m[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)  # FP32
            gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                                mask=mask_m[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)  # FP32

            # normalize
            norm = (out_vals - mean[:, None]) * rstd[:, None]  # [BLOCK_M, BLOCK_H]
            # apply to_out_norm scale and bias
            gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)  # [BLOCK_H]

    out_ptr,      # half* [total_rows, total_out_features] (output)
    gamma_ptr,    # float* [dim] (LN scale)
    beta_ptr,     # float* [dim] (LN bias)
    weight_ptr,   # half* [total_out_features, dim] (linear weight)
    total_rows,
    dim: tl.constexpr,
    total_out: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute mean and rstd per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :], mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_sq += tl.sum(x * x, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute projection
    # iterate over output feature blocks
    for o in range(0, tl.cdiv(total_out, BLOCK_O)):
        # accumulator init
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            # load raw x

            # apply gamma and beta
            gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
            beta = tl.load(beta_ptr + offs_d, mask=mask_d, other=0.0)
            norm = norm * gamma[None, :] + beta[None, :]
            # load weight block
            offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
            mask_o = offs_o < total_out

                        mask=mask_o[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)   # shape [BLOCK_O, BLOCK_D]
            # dot product: accumulate norm (M,D) * w.T (D,O) => (M,O)

def ln_linear_kernel(
    x_ptr,          # float* [total_rows, dim]
    out_ptr,        # half* [total_rows, total_out]
    gamma_ptr,      # float* [dim]
    beta_ptr,       # float* [dim]
    weight_ptr,     # half* [total_out, dim]
    total_rows: tl.constexpr,
    dim: tl.constexpr,
    total_out: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # first pass: compute mean and variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load x
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute projection output
    for o in range(0, tl.cdiv(total_out, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < total_out
        # accumulator for each row and each output in block
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # Load x again
            x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

            # Normalize
            norm = (x - mean[:, None]) * rstd[:, None]

            # Apply gamma and beta
            gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
            beta = tl.load(beta_ptr + offs_d, mask=mask_d, other=0.0)

                        mask=mask_o[None, :] & mask_d[:, None],
                        other=0.0).to(tl.float32)

            # Accumulate dot product: (M,D) @ (D,O) => (M,O)
            acc += tl.dot(norm, w)

        # Store accumulator to output
        out_ptrs = out_ptr + offs_m[:, None] * total_out + offs_o[None, :]

"""
custom_kernel – TriMul “outgoing” forward pass (AlphaFold‑3) in Triton+PyTorch.

Algorithm
---------
1️⃣  Fuse the first LayerNorm and all linear projections (left/right
    proj‑gate + out‑gate) into a single Triton kernel:
        * Row‑wise mean / variance → rstd (FP32)
        * Normalise the row, apply LN scale/shift (γ,β)
        * Multiply the normalised row by the *concatenated* weight matrix
          (5·hidden_dim × dim) – this yields
          `[left_proj, left_gate, right_proj, right_gate, out_gate]`
          in one memory pass.

2️⃣  Optional pair‑wise mask is applied (only to left/right
    projection tensors).

3️⃣  Apply sigmoid gates, i.e. `left = left_proj * σ(left_gate)`,
    `right = right_proj * σ(right_gate)`, and build `out_gate = σ(out_gate)`.

4️⃣  Tri‑multiplicative update:
        `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    realised as a batched GEMM (`torch.bmm`) over the hidden dimension.

5️⃣  Second LayerNorm (over the hidden dimension) using the fast
    PyTorch `F.layer_norm` implementation.

6️⃣  Element‑wise gating by `out_gate` and final linear projection
    (hidden → dim).  The final result is cast back to FP32.

Only the first LayerNorm + projection step is written in Triton
(`ln_linear_kernel`).  All other heavy primitives (batched GEMM,
layer‑norm, sigmoid, linear) are delegated to highly‑optimised
torch kernels.
"""

import torch.nn.functional as F

# Triton kernel : row‑wise LayerNorm + fused linear projection.
#   Input  : x               – float32 [total_rows, dim]
#   Output : y               – float16 [total_rows, out_feat]  (out_feat = 5·hidden_dim)
#   We compute:
#         mean, var  → rstd
#         norm = (x-mean)*rstd * gamma + beta            (LN)
#         y_i   = Σ_d norm_d * W_i,d                     (linear)
#   The kernel performs two passes over the input: one for the statistics,
#   a second for the matrix‑multiply.

@triton.jit
def ln_linear_kernel(
    x_ptr,          # float*   [total_rows, dim]
    y_ptr,          # half*    [total_rows, out_feat]
    gamma_ptr,      # float*   [dim]        (LN scale)
    beta_ptr,       # float*   [dim]        (LN bias)
    weight_ptr,     # half*    [out_feat, dim]   (concatenated projection weights)
    total_rows,     # int
    dim,            # int
    out_feat,       # int   (=5*hidden_dim)
    eps,            # float (constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim block size
    BLOCK_O: tl.constexpr,   # output‑feature block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  First pass – compute mean & variance for each row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        # x is float32 already
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                      # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Second pass – fused LN + linear projection
    # --------------------------------------------------------------
    for o in range(0, tl.cdiv(out_feat, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_feat

        # accumulator for a block of rows × a block of output features
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # reload the input row (needed for the dot‑product)
            x = tl.load(
                x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            # ---- Layer‑norm of the current block ----
            norm = (x - mean[:, None]) * rstd[:, None]          # (x‑μ)/σ
            gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
            beta  = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)
            norm = norm * gamma[None, :] + beta[None, :]        # γ·Ŵ+β

            # ---- Weight block (transposed) ----
            # weight layout: [out_feat, dim]; we need a [BLOCK_D, BLOCK_O] slice.
            w = tl.load(
                weight_ptr + offs_o[None, :] * dim + offs_d[:, None],
                mask=mask_o[None, :] & mask_d[:, None],
                other=0.0,
            ).to(tl.float32)                                   # [BLOCK_D, BLOCK_O]

            # ---- Accumulate dot‑product (M×D)·(D×O) → (M×O) ----
            acc += tl.dot(norm, w)

        # ---- Write the output block ----
        out_ptrs = y_ptr + offs_m[:, None] * out_feat + offs_o[None, :]
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_m[:, None] & mask_o[None, :],

            "right_gate.weight":      Tensor[hidden_dim, dim]
            "out_gate.weight":        Tensor[hidden_dim, dim]
            "to_out_norm.weight":     Tensor[hidden_dim]
            "to_out_norm.bias":       Tensor[hidden_dim]
            "to_out.weight":          Tensor[dim, hidden_dim]
        }
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True),
        }

    Returns:
        torch.Tensor[float32]  shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    total_rows = B * N * N
    out_feat = hidden_dim * 5            # 2*proj + 2*gate + out_gate
    assert hidden_dim * 5 == out_feat

    # ------------------------------------------------------------------
    # 1️⃣  Fuse first LayerNorm + all five linear projections (Triton)
    # ------------------------------------------------------------------
    # Concatenate all projection / gate weights (dim → 5·hidden_dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)   # [5·H, dim]

    # Allocate output buffer for the fused projection
    combined_out = torch.empty(
        (total_rows, out_feat), dtype=torch.float16, device=device
    )

    # Triton launch configuration
    BLOCK_M = 64            # rows per program (tunable)
    BLOCK_D = 64            # dim block size (covers dim ≤ 384)
    BLOCK_O = 64            # output‑feature block size (covers 5·H ≤ 640)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    # launch kernel
    ln_linear_kernel[grid](
        input_tensor.view(total_rows, dim).contiguous(),
        combined_out,
        weights["norm.weight"],
        weights["norm.bias"],
        combined_weight,
        total_rows,
        dim,
        out_feat,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Split the fused tensor into the individual components
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = torch.split(
        combined_out, [h, h, h, h, h], dim=1
    )   # each -> [total_rows, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 3️⃣  Optional pair‑wise mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16, non_blocking=True)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣  Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)      # FP16 → FP16 (still half)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left_proj * left_gate                  # [rows, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Tri‑multiplicative update (batched GEMM over the hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # bring H into the batch dimension → [B·H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)               # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)             # [BH, N, N]

    # GEMM: (N×N) @ (N×N)^T  → sum over k dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm (over hidden_dim) – use the highly‑optimised
    #     torch.nn.functional.layer_norm implementation.
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # FP16 output

    # ------------------------------------------------------------------
    # 7️⃣  Gating by out_gate and final linear projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # element‑wise FP16
    final = F.linear(
        gated.view(B * N * N, hidden_dim),
        weights["to_out.weight"].to(torch.float16, non_blocking=True),
    )   # [rows, dim] in FP16

    # Cast back to FP32 to match the reference module's output dtype
    final = final.view(B, N, N, dim).to(torch.float32)
    return final
```
