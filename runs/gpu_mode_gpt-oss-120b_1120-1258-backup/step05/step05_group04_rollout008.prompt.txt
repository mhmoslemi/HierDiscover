import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Input LayerNorm (FP32 reduction → FP16 output) – custom Triton kernel.
2️⃣  Fuse the 5 linear projections (left/right + 3 gates) into a single
    cheap FP16 GEMM (torch.nn.functional.linear).  This is the most
    compute‑intensive step (R × C × 5H FLOPs).
3️⃣  (Optional) apply the binary mask to the left/right projections.
4️⃣  Apply sigmoid on the three gate tensors and fuse the element‑wise
    gating.
5️⃣  Triangular multiplicative update – a batched matrix‑multiply
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]` performed with
    `torch.bmm` (Tensor‑Core, FP16).
6️⃣  **Fused** LayerNorm + out‑gate + final linear projection:
    a custom Triton kernel reads the `out` tensor, computes the per‑row
    mean/variance, applies the LayerNorm (`to_out_norm`), multiplies by
    the sigmoid‑gated `out_gate` and immediately projects to `dim`
    channels with the `to_out` weight.  This eliminates two
    memory‑bound passes (the separate LayerNorm kernel and the final
    `torch.nn.functional.linear` call) and reduces total memory traffic.

The implementation stays entirely on‑device, returns a float32 tensor
with shape `[B, N, N, dim]` and satisfies all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (input FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (output FP16)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,               # static dimension (≤ 384)
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,           # rows per program
    BLOCK_D: tl.constexpr,           # columns processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate + final linear projection (all FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden]   (pre‑norm)
    gate_ptr,            # *const half*  [total_rows, hidden]   (out_gate, sigmoid’ed)
    norm_w_ptr,          # *const float* [hidden]               (γ of to_out_norm)
    norm_b_ptr,          # *const float* [hidden]               (β of to_out_norm)
    out_w_ptr,           # *const half*  [dim, hidden]          (to_out weight)
    out_ptr,             # *mut   float* [total_rows, dim]      (final FP32 output)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program   (e.g. 64)
    BLOCK_D: tl.constexpr,   # output‑channel chunk (e.g. 64)
):
    pid = tl.program_id(0)          # row block index
    pid_d = tl.program_id(1)        # output‑channel block index
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Load the whole hidden vector for each selected row (BLOCK_M × hidden)
    # -----------------------------------------------------------------
    offs_h = tl.arange(0, hidden)                               # [hidden]
    mask_h = offs_h < hidden

    # x and gate are stored as half → cast to FP32 for math
    x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                       # (BLOCK_M, hidden)
    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                       # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Compute mean / rstd for each row (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)                                 # (BLOCK_M)
    sum_sq  = tl.sum(x * x, axis=1)                             # (BLOCK_M)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Apply LayerNorm weight/bias and the out‑gate
    # -----------------------------------------------------------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)

    norm = (x - mean[:, None]) * rstd[:, None]                     # (BLOCK_M, hidden)
    norm = norm * w_norm[None, :] + b_norm[None, :]                # apply γ,β
    gated = norm * g                                                # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Output‑channel block handling
    # -----------------------------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)                       # [BLOCK_D]
    mask_d = offs_d < dim

    # Load the slice of the final weight matrix (hidden × BLOCK_D)
    # out_w has shape [dim, hidden]; we need its transpose (hidden, BLOCK_D)
    out_w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                      mask=mask_h[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)                  # (hidden, BLOCK_D)

    # Compute the mat‑mul: (BLOCK_M, hidden) × (hidden, BLOCK_D) → (BLOCK_M, BLOCK_D)
    out_chunk = tl.dot(gated, out_w_T)                            # (BLOCK_M, BLOCK_D)

    # Store the result (FP32)
    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim    = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                 # rows for flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)

    BLOCK_M = 64
    BLOCK_D = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate (single FP16 GEMM)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                   # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)               # (R, 5*H) FP16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
    right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                                        # each (R, H)

    # --------------------------------------------------------------
    # 2a) Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16  * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)               # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)              # (R, H)
    out_gate   = torch.sigmoid(out_gate_fp16)                # (R, H) – used later

    left = left_proj_fp16 * left_gate                        # (R, H)
    right = right_proj_fp16 * right_gate                     # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched matmul)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, hidden)                       # (B, N, N, H)
    right = right.view(B, N, N, hidden)                      # (B, N, N, H)

    # reshape for bmm: (B*H, N, N) where left has rows = i, cols = k,
    # right needs rows = k, cols = j → we transpose its last two axes
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                # (B*H, N, N) FP16

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate + final projection
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()    # (R, H) FP16
    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    grid_out = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    layernorm_gate_linear_kernel[grid_out](
        out_flat,                                      # x_ptr
        out_gate,                                      # gate_ptr (already sigmoid)
        weights["to_out_norm.weight"],                 # γ
        weights["to_out_norm.bias"],                   # β
        weights["to_out.weight"].half().contiguous(), # to_out weight (dim, hidden)
        final_fp32,                                    # output buffer
        total_rows,
        dim=dim,
        hidden=hidden,                                 # compile‑time constant
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5️⃣ Reshape to final tensor shape and cast back to FP32
    # --------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)                 # [B, N, N, dim] float32
    return final
