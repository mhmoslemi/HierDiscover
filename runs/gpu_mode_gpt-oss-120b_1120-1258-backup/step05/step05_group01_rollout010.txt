@triton.jit
def fused_layernorm_proj_kernel(
    x_ptr,                 # const float32 input [R, dim]
    y_ptr,                 # mutable fp16 output [R, out_dim] (5*hidden_dim)
    weight_ptr,            # const fp16 weight [dim, out_dim]
    total_rows, dim, out_dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D_IN: tl.constexpr,
    BLOCK_D_OUT: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------- mean/var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D_IN)):
        offs_d = d * BLOCK_D_IN + tl.arange(0, BLOCK_D_IN)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- linear projection ----------
    # loop over output tiles
    for o in range(0, tl.cdiv(out_dim, BLOCK_D_OUT)):
        offs_o = o * BLOCK_D_OUT + tl.arange(0, BLOCK_D_OUT)
        mask_o = offs_o < out_dim
        acc = tl.zeros([BLOCK_M, BLOCK_D_OUT], dtype=tl.float32)
        # accumulate over input dim tiles
        for d in range(0, tl.cdiv(dim, BLOCK_D_IN)):
            offs_d = d * BLOCK_D_IN + tl.arange(0, BLOCK_D_IN)
            mask_d = offs_d < dim
            # load input chunk
            ptr_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            x_chunk = tl.load(ptr_x,
                              mask=mask_m[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)

            # normalize
            norm = (x_chunk - mean[:, None]) * rstd[:, None]   # (BLOCK_M, BLOCK_D_IN)

            # load weight chunk
            # weight layout [dim, out_dim]
            ptr_w = weight_ptr + offs_d[:, None] * out_dim + offs_o[None, :]
            w_chunk = tl.load(ptr_w,
                              mask=mask_d[:, None] & mask_o[None, :],
                              other=0.0).to(tl.float32)   # (BLOCK_D_IN, BLOCK_D_OUT)
            # Dot
            acc += tl.dot(norm, w_chunk)   # (BLOCK_M, BLOCK_D_OUT)
        # store result
        out_ptr = y_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
        tl.store(out_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])

@triton.jit
def fused_ln_gate_out_kernel(
    src_ptr,         # const fp16 [B, H, N, N]
    gate_ptr,        # const fp16 [R, H]
    ln_weight_ptr,   # const fp32 [H]
    ln_bias_ptr,     # const fp32 [H]
    w_out_ptr,       # const fp16 [H, dim] (transposed)
    out_ptr,         # mutable fp32 [R, dim]
    total_rows, hidden_dim, out_dim,
    seq_len,
    stride_b, stride_h, stride_i, stride_j,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j    # offset base for hidden dim

    # ---------- compute mean/var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        ptr = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_h[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- compute final output ----------
    for o in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
        for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim
            # load raw hidden
            src_ptrs = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
            raw = tl.load(src_ptrs,
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
            # normalize
            norm = (raw - mean[:, None]) * rstd[:, None]
            # apply LN affine
            w_g = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)
            b_b = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0)
            norm = norm * w_g[None, :] + b_b[None, :]
            # multiply gate
            gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            gated = norm * gate          # (BLOCK_M, BLOCK_H)

            # load w_out slice (H, out_dim). we store w_out_T as [hidden_dim, out_dim]
            w_ptr = w_out_ptr + offs_h[:, None] * out_dim + offs_o[None, :]
            w = tl.load(w_ptr,

    dim = config["dim"]

            weights["right_proj.weight"],
            weights["left_gate.weight"],

        dim,

        eps,

        BLOCK_D_IN=BLOCK_D_IN,
        BLOCK_D_OUT=BLOCK_D_OUT,
    )

    H = hidden_dim
    left_proj = proj_out[:, 0:H]
    right_proj = proj_out[:, H:2*H]
    left_gate = proj_out[:, 2*H:3*H]
    right_gate = proj_out[:, 3*H:4*H]
    out_gate = proj_out[:, 4*H:5*H]

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # optional mask
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(R, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Gated left/right
    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape for matmul
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    left_perm = left.permute(0,3,1,2)   # B H N N
    right_perm = right.permute(0,3,1,2)

    # matmul
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1,-2))   # B H N N (fp16)

    # prepare final kernel
    # w_out transposed
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)   # shape [hidden_dim, dim]

    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    # stride info
    stride_b = out_perm.stride(0)
    stride_h = out_perm.stride(1)
    stride_i = out_perm.stride(2)
    stride_j = out_perm.stride(3)

    BLOCK_M_OUT = 128
    BLOCK_H = 64
    BLOCK_O = 64
    grid_out = (triton.cdiv(R, BLOCK_M_OUT),)
    fused_ln_gate_out_kernel[grid_out](
        out_perm,
        out_gate,

        weights["to_out_norm.bias"],

        R,
        H,

        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_H=BLOCK_H,
        BLOCK_O=BLOCK_O,
    )

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass (no gradients)

Algorithm (all heavy work stays on the GPU):
1️⃣  Fuse the first LayerNorm (row‑wise over the model dimension) with the
    5‑way linear projection (left_proj, right_proj, left_gate, right_gate,
    out_gate).  One Triton kernel loads a row of the input, computes its
    mean/variance in FP32, normalises it, then performs a small GEMM
    (dim ≤ 384, out_dim = 5·hidden_dim ≤ 640) to produce the five
    projected tensors in FP16.

2️⃣  Apply sigmoid to the gating tensors, optionally mask the two
    projections, and multiply the gates → gated tensors *left* and *right*.

3️⃣  Reshape and permute to (B, H, N, N) and compute the batched matrix
    multiplication
        out_perm[b, h, i, j] = Σₖ left[b, i, k, h] · right[b, j, k, h]
    with a single torch.matmul (FP16 Tensor‑Core).

4️⃣  Fuse the second LayerNorm, the output‑gate, and the final linear
    projection back to the model dimension in another Triton kernel.
    The kernel:
        – reads the non‑contiguous result `out_perm`,
        – computes per‑row mean/variance over the hidden dimension,
        – applies γ/β,
        – multiplies by the pre‑computed *out_gate*,
        – performs the final weight multiplication (H × dim) and writes
          the result in FP32.

All kernels use FP32 accumulation for numerical stability and cast to
FP16 only for storage, matching the reference implementation’s precision.

The implementation supports the full test matrix (various batch sizes,
sequence lengths, model/hidden dimensions, masked or un‑masked inputs,
and normal/Cauchy data distributions).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Fused LayerNorm + 5‑way projection (first part of TriMul)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_proj_kernel(
    x_ptr,               # const   *  [total_rows, dim]   (fp32)
    y_ptr,               # mutable *  [total_rows, out_dim] (fp16)
    w_ptr,               # const   *  [dim, out_dim]      (fp16)
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    out_dim,             # int (runtime) = 5 * hidden_dim
    eps,                 # float (compile‑time)
    BLOCK_M: tl.constexpr,       # rows per program
    BLOCK_D_IN: tl.constexpr,    # input‑dim tile
    BLOCK_D_OUT: tl.constexpr,   # output‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – compute per‑row mean & variance over `dim`
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D_IN)):
        offs_d = d * BLOCK_D_IN + tl.arange(0, BLOCK_D_IN)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – normalise and multiply with weight matrix
    # --------------------------------------------------------------
    for o in range(0, tl.cdiv(out_dim, BLOCK_D_OUT)):
        offs_o = o * BLOCK_D_OUT + tl.arange(0, BLOCK_D_OUT)
        mask_o = offs_o < out_dim
        acc = tl.zeros([BLOCK_M, BLOCK_D_OUT], dtype=tl.float32)

        for d in range(0, tl.cdiv(dim, BLOCK_D_IN)):
            offs_d = d * BLOCK_D_IN + tl.arange(0, BLOCK_D_IN)
            mask_d = offs_d < dim

            # Load a tile of the input row
            ptr_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            x_tile = tl.load(ptr_x,
                             mask=mask_m[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)

            # Normalise (FP32)
            norm = (x_tile - mean[:, None]) * rstd[:, None]   # [M, D_in]

            # Load the matching weight sub‑matrix (dim × out_dim)
            # weight layout : [dim, out_dim] (row‑major)
            ptr_w = w_ptr + offs_d[:, None] * out_dim + offs_o[None, :]
            w_tile = tl.load(ptr_w,
                             mask=mask_d[:, None] & mask_o[None, :],
                             other=0.0).to(tl.float32)   # [D_in, D_out]

            # Accumulate GEMM contribution
            acc += tl.dot(norm, w_tile)    # [M, D_out]
        # Store the completed outputs (FP16)
        out_ptr = y_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
        tl.store(out_ptr,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# 6️⃣  Fuse second LayerNorm + out‑gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_out_kernel(
    src_ptr,                 # const   *  [B, H, N, N] (fp16, non‑contig)
    gate_ptr,                # const   *  [R, H]      (fp16)
    ln_w_ptr,                # const   *  [H]         (fp32)
    ln_b_ptr,                # const   *  [H]         (fp32)
    w_out_ptr,               # const   *  [H, D]      (fp16)  (H = hidden_dim, D = model dim)
    out_ptr,                 # mutable *  [R, D]      (fp32)
    total_rows,              # int (runtime) = B·N·N
    hidden_dim,              # int (compile‑time)
    out_dim,                 # int (compile‑time) = model dim
    seq_len,                 # int (runtime) = N
    stride_b,                # int (runtime) stride on dim 0 (batch) of src
    stride_h,                # int (runtime) stride on dim 1 (hidden) of src
    stride_i,                # int (runtime) stride on dim 2 (seq i) of src
    stride_j,                # int (runtime) stride on dim 3 (seq j) of src
    eps,                     # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim tile size
    BLOCK_O: tl.constexpr,   # output‑dim tile size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Decode (b, i, j) from the linear row index
    # ------------------------------------------------------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j   # [BLOCK_M]

    # ------------------------------------------------------------------
    # Pass 1 – mean & variance over hidden dimension
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        ptr = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_h[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass 2 – normalise, apply LN affine, multiply gate, and final proj.
    # ------------------------------------------------------------------
    for o in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            # Load raw hidden values
            ptr = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
            raw = tl.load(ptr,
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

            # Normalise (FP32)
            norm = (raw - mean[:, None]) * rstd[:, None]

            # LayerNorm affine parameters (γ/β)
            w = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # γ (FP32)
            b = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # β (FP32)
            norm = norm * w[None, :] + b[None, :]

            # Multiply with out‑gate (gate is FP16 → promote)
            gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            gated = norm * gate            # [M, H_tile]

            # Load slice of final weight matrix (hidden_dim × out_dim)
            # w_out_ptr has shape [H, D] (row‑major)
            w_ptr = w_out_ptr + offs_h[:, None] * out_dim + offs_o[None, :]
            w_chunk = tl.load(w_ptr,
                              mask=mask_h[:, None] & mask_o[None, :],
                              other=0.0).to(tl.float32)   # [H_tile, O_tile]

            # Accumulate dot product
            acc += tl.dot(gated, w_chunk)   # [M, O_tile]
        # Store the final result (FP32)
        out_ptrs = out_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
        tl.store(out_ptrs,
                 acc,
                 mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator (no grads)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]           # model dimension (e.g., 128 or 384)
    hidden_dim   = config["hidden_dim"]   # hidden dimension (always 128 in tests)
    eps          = 1e-5
    device       = input_tensor.device

    B, N, _, _   = input_tensor.shape
    total_rows   = B * N * N                     # R = B·N·N
    out_dim_5h   = hidden_dim * 5                # size of the concatenated projection

    # --------------------------------------------------------------
    # 1) First LayerNorm + 5‑way linear projection (fused)
    # --------------------------------------------------------------
    # Flatten input to [R, dim] (FP32)
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    # Output buffer for the concatenated projection [R, 5*H] (FP16)
    proj_out = torch.empty(total_rows, out_dim_5h,
                           dtype=torch.float16, device=device)

    # Build the concatenated weight matrix and transpose it to [dim, 5*H]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # each slice: (hidden_dim, dim)
    )                     # shape (5*H, dim)
    w_concat_T = w_concat.t().contiguous()   # shape (dim, 5*H), FP16

    # Triton launch configuration
    BLOCK_M      = 128
    BLOCK_D_IN   = 64               # tile over `dim` (≤384)
    BLOCK_D_OUT  = 64               # tile over `5*hidden_dim` (≤640)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_proj_kernel[grid](
        x_flat,
        proj_out,
        w_concat_T,
        total_rows,
        dim,
        out_dim_5h,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D_IN=BLOCK_D_IN,
        BLOCK_D_OUT=BLOCK_D_OUT,
    )

    # --------------------------------------------------------------
    # 2) Split projections and apply sigmoid gates
    # --------------------------------------------------------------
    H = hidden_dim
    left_proj   = proj_out[:, 0:H]
    right_proj  = proj_out[:, H:2*H]
    left_gate   = proj_out[:, 2*H:3*H]
    right_gate  = proj_out[:, 3*H:4*H]
    out_gate    = proj_out[:, 4*H:5*H]

    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 3) Optional mask (only on the two projections)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_flat = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4) Apply gates → gated tensors `left` and `right`
    # --------------------------------------------------------------
    left  = left_proj * left_gate      # [R, H]
    right = right_proj * right_gate    # [R, H]

    # Reshape back to (B, N, N, H) – still contiguous because H is last dim
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5) Batched matrix multiplication: out_perm = left @ rightᵀ
    #    (shape (B, H, N, N), FP16)
    # --------------------------------------------------------------
    # Permute to (B, H, N, N) – view, no copy.
    left_perm  = left.permute(0, 3, 1, 2)          # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)         # (B, H, N, N)
    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)  – cuBLAS Tensor‑Core GEMM
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))

    # --------------------------------------------------------------
    # 6) Second LayerNorm + out‑gate + final projection (fused)
    # --------------------------------------------------------------
    # Prepare strides for the non‑contiguous `out_perm` tensor
    stride_b = out_perm.stride(0)   # stride over batch dimension B
    stride_h = out_perm.stride(1)   # stride over hidden dimension H (== N*N)
    stride_i = out_perm.stride(2)   # stride over sequence index i
    stride_j = out_perm.stride(3)   # stride over sequence index j

    final = torch.empty(total_rows, dim,
                        dtype=torch.float32, device=device)

    # Transpose the final weight once: (hidden_dim, dim)
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)

    BLOCK_M_OUT = 128
    BLOCK_H = 64            # tile over hidden_dim (≤128)
    BLOCK_O = 64            # tile over output dim (≤384)
    grid_out = (triton.cdiv(total_rows, BLOCK_M_OUT),)
    fused_ln_gate_out_kernel[grid_out](
        out_perm,                     # src (B, H, N, N) – FP16, non‑contig
        out_gate,                     # gate (R, H) – FP16, contig
        weights["to_out_norm.weight"],   # LN γ (FP32)
        weights["to_out_norm.bias"],     # LN β (FP32)
        w_out_T,                         # final weight (H, D) – FP16
        final,                           # output buffer (R, D) – FP32
        total_rows,
        hidden_dim,
        dim,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_H=BLOCK_H,
        BLOCK_O=BLOCK_O,
    )

    # --------------------------------------------------------------
    # 7) Reshape to the expected output layout
    # --------------------------------------------------------------
    return final.view(B, N, N, dim)
```
