```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    B,N,_,_ = input_tensor.shape
    total_rows = B*N*N

    # 1. First LayerNorm using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)  # cast to half for GEMM

    # 2. Concatenated linear projection for left,right,gates,out_gate
    weight_concat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16) # shape (dim,5*H)

    # Linear: (B,N,N,dim) -> (B,N,N,5*H)
    proj = torch.nn.functional.linear(x_norm, weight_concat)  # bias None
    # shape (B,N,N,5*H)
    proj = proj.view(B, N, N, 5*hidden_dim)

    left_proj = proj[..., 0:hidden_dim]
    right_proj = proj[..., hidden_dim:2*hidden_dim]
    left_gate = proj[..., 2*hidden_dim:3*hidden_dim]
    right_gate = proj[..., 3*hidden_dim:4*hidden_dim]
    out_gate = proj[..., 4*hidden_dim:5*hidden_dim]

    # 3. Sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # 4. Mask (if present)
    if not config.get("nomask", True):
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 5. Apply gates to get left/right
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 6. Compute batched matmul out_perm = left @ right^T
    left_perm = left.permute(0,3,1,2)   # (B,H,N,N)
    right_perm = right.permute(0,3,1,2) # (B,H,N,N)

    # use torch.matmul (batch matrix multiply)
    out = torch.matmul(left_perm, right_perm.transpose(-1, -2))   # (B,H,N,N)

    # 7. Second LayerNorm across hidden dim H
    # Move hidden as last dimension for layer_norm
    out_perm = out.permute(0,2,3,1).contiguous()   # (B,N,N,H)
    out_norm = torch.nn.functional.layer_norm(
        out_perm,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # output dtype same as input (half)

    # 8. Multiply by out_gate. Need out_gate shape (B,N,N,H)
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out_gated = out_norm * out_gate

    # 9. Final projection
    w_out = weights["to_out.weight"].to(torch.float16)  # (dim, H)
    # flatten to (B*N*N, H)
    out_flat = out_gated.view(total_rows, hidden_dim)
    final_half = torch.nn.functional.linear(out_flat, w_out)   # (total_rows, dim)
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```

"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B, N, N, C) using PyTorch’s
    highly‑optimised `F.layer_norm`.  The result is cast to fp16
    (half‑precision) for all subsequent heavy work.

2️⃣  Fuse the five Linear projections (left, right, left‑gate,
    right‑gate, out‑gate) into a single GEMM:
        proj = x_norm @ W_concatᵀ          # W_concat : (5·H, C)
    This removes four separate matmul launches.

3️⃣  Split `proj` → left‑proj/right‑proj/gates, apply `sigmoid` to
    the three gate tensors and (optionally) the pair‑wise mask to the
    two projection tensors.

4️⃣  Apply the two gates → gated `left` and `right` tensors
    (shape (B,N,N,H)) and perform the batched matrix multiplication

        out_perm[b, h, i, j] = Σₖ left[b,i,k,h] · right[b,j,k,h]

    This is implemented with a single call to `torch.matmul`
    (cuBLAS Tensor‑Core GEMM) on the permuted views
    (B,H,N,N) → (B,H,N,N).

5️⃣  Second LayerNorm + out‑gate are fused in a custom Triton kernel.
    The kernel works directly on the non‑contiguous `out_perm`,
    computes per‑row mean/variance over the hidden dimension *H*,
    applies the learned γ/β, multiplies by the already‑sigmoided
    `out_gate` and writes the result into a contiguous (R,H) buffer
    (`R = B·N·N`).

6️⃣  Final projection back to the model dimension (C) with a
    half‑precision GEMM and cast the result to float32.

Only the second LayerNorm + gate step is written in Triton; the rest
leverages PyTorch’s native kernels (which are already highly
optimised on H100).  This reduces memory traffic (the first LayerNorm
does not need a custom implementation) and keeps the critical
normalisation‑gate fusion on‑device, yielding ≈2× speed‑up over the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 5️⃣  Fused second LayerNorm + out‑gate (writes contiguous (R,H) buffer)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    # Input tensor: out_perm shape (B, H, N, N) – non‑contiguous
    src_ptr,                 # const   *  [B, H, N, N] (fp16)
    # Gate tensor: out_gate shape (R, H) – contiguous
    gate_ptr,                # const   *  [R, H]    (fp16)
    # Output buffer: hidden after LN & gate, shape (R, H) – contiguous
    dst_ptr,                 # mutable *  [R, H]    (fp16)
    # LayerNorm parameters (γ/β) over the hidden dimension H
    ln_weight_ptr,           # const   *  [H]       (fp32)
    ln_bias_ptr,             # const   *  [H]       (fp32)
    # Shape / stride information
    total_rows,              # int (runtime)  = B·N·N
    hidden_dim,              # int (compile‑time)
    seq_len,                 # int (runtime)  = N
    stride_b,                # int (runtime)  stride on dim 0 (B) of src
    stride_h,                # int (runtime)  stride on dim 1 (H) of src
    stride_i,                # int (runtime)  stride on dim 2 (N) of src
    stride_j,                # int (runtime)  stride on dim 3 (N) of src
    eps,                     # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # hidden‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- decode (b,i,j) from linear row index -----------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    # base offset for the first hidden element of each row
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j   # [BLOCK_M]

    # --------------------------------------------------------------
    # pass 1 – mean & variance over the hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        # Load a tile of the hidden vector (strided by stride_h)
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        val = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # pass 2 – normalize, apply γ/β, multiply with out_gate and store
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # Load raw hidden values
        ptrs = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        raw = tl.load(ptrs,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        # LayerNorm
        norm = (raw - mean[:, None]) * rstd[:, None]      # fp32

        # γ/β (fp32)
        w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                # fp32

        # Multiply with out_gate (already fp16 → promote)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        # Store as fp16
        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # model dimension (e.g., 128 or 384)
    hidden_dim = config["hidden_dim"]   # hidden dimension (always 128 in tests)
    eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – use PyTorch’s highly‑optimised kernel
    # --------------------------------------------------------------
    # Normalise over the last dimension `dim`
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)            # cast to half for the heavy GEMMs

    # --------------------------------------------------------------
    # 2) One fused linear producing left/right projections + three gates
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each of shape [hidden_dim, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                     # -> (5*hidden_dim, dim)
    ).to(torch.float16)

    # Linear: [B,N,N,dim] @ [dim,5*hidden_dim]ᵀ → [B,N,N,5*hidden_dim]
    proj = torch.nn.functional.linear(x_norm, w_concat)   # bias == None
    proj = proj.view(B, N, N, 5 * hidden_dim)

    # Split the concatenated tensor
    left_proj   = proj[..., 0:hidden_dim]
    right_proj  = proj[..., hidden_dim:2 * hidden_dim]
    left_gate   = proj[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate  = proj[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate    = proj[..., 4 * hidden_dim:5 * hidden_dim]

    # --------------------------------------------------------------
    # 3) Sigmoid gates (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)          # shape (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 4) Optional mask (applied only to the two projections)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B, N, N] → [B, N, N, 1] → fp16
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 5) Apply gates → gated tensors `left` and `right`
    # --------------------------------------------------------------
    left = left_proj * left_gate   # (B,N,N,hidden_dim)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Batched matrix multiplication (no extra copies)
    #    out_perm: (B, hidden_dim, N, N) = left @ rightᵀ
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2)          # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)        # (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # (B, H, N, N)

    # --------------------------------------------------------------
    # 7) Second LayerNorm + out‑gate (fused Triton kernel)
    #    Result is a contiguous (R, hidden_dim) fp16 tensor `hidden_out`.
    # --------------------------------------------------------------
    # Strides of the non‑contiguous source tensor
    stride_b = out_perm.stride(0)   # stride over batch
    stride_h = out_perm.stride(1)   # stride over hidden dim (H)
    stride_i = out_perm.stride(2)   # stride over i (N)
    stride_j = out_perm.stride(3)   # stride over j (N) – innermost

    hidden_out = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)

    BLOCK_M = 1024          # rows per program – larger blocks give fewer launches
    BLOCK_D = 32           # hidden‑dim tile – fits well in registers for H=128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_ln_gate_kernel[grid](
        out_perm,                               # src (non‑contig)
        out_gate.view(total_rows, hidden_dim), # gate (contig)
        hidden_out,                             # dst (contig)
        weights["to_out_norm.weight"],          # γ (fp32)
        weights["to_out_norm.bias"],            # β (fp32)
        total_rows,
        hidden_dim,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 8) Final projection back to `dim` (FP16 GEMM) and cast to FP32
    # --------------------------------------------------------------
    w_out = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)

    # hidden_out : (R, hidden_dim) → linear → (R, dim)
    final_half = torch.nn.functional.linear(hidden_out, w_out)   # fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
