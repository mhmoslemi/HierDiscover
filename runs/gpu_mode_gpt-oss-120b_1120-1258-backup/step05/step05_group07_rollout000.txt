"""
Outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last channel (C) – Triton kernel
   * fp32 accumulation, fp16 output.
2️⃣ Fuse the five linear projections (left/right + three gates) into one
   half‑precision GEMM by concatenating their weight matrices.
3️⃣ (Optional) apply the pairwise mask to the left/right projected tensors.
4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣ Triangular multiplicative update (outgoing version):
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   realised as a batched GEMM `left @ rightᵀ` (fp16 Tensor‑Core).
6️⃣ Second LayerNorm over the hidden dimension (H) – same Triton kernel.
7️⃣ Multiply by the output gate and a final linear projection back to C.
8️⃣ Cast result to float32.

All heavy linear algebra uses PyTorch / cuBLAS fp16 kernels;
the only custom Triton kernels are the two LayerNorms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,           # *const T*   [total_rows, dim]   (input, fp32 or fp16)
    y_ptr,           # *mut   f16* [total_rows, dim]   (output)
    weight_ptr,      # *const f32* [dim]               (scale γ)
    bias_ptr,        # *const f32* [dim]               (bias β)
    total_rows,      # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # cols processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a block of the input (any dtype, cast to fp32)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (bool/float, optional)
        - weight_dict  : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # flatten B×N×N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)      # (R, C) fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Tuned block sizes – dim ≤ 384 → BLOCK_D = 64 works well
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left, right, three gates)
    # ------------------------------------------------------------------
    # Each weight shape: (hidden_dim, dim)
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    # Concatenate along output dimension → (5*H, C)
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).half().contiguous()
    # Single fused GEMM, bias‑free → (R, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)

    # Slice back to individual tensors (R, H)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden_dim, dim=1
        )

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied BEFORE gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: (B, N, N) → (R, 1) → fp16 broadcast
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)   # (R, H)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape to (B, N, N, H) for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (outgoing version)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    #    → batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Move hidden dim into the batch axis → (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # Batched matmul: left @ rightᵀ  → (BH, N, N) fp16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H (fp16 output)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)   # (R, H)
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,                     # input (fp16) – cast to fp32 inside kernel
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Final gate + projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # (B, N, N, H) fp16

    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)     # (R, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # (R, C)

    # ------------------------------------------------------------------
    # 8️⃣ Cast to float32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
