"""
Highly-optimized Triton+PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3) targeting H100.

Algorithm
---------
1. Input LayerNorm (FP32 reduction → FP16 output) – custom Triton kernel.
2. Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a single FP16 GEMM using a concatenated weight matrix.
3. Apply mask (if present) only on left/right projections.
4. Apply sigmoid to the three gate vectors and multiply them into the
   projected tensors (left, right).  This stays in FP16.
5. Triangular multiplicative update – batched matrix‑multiply of shape
   (B·H, N, N) using `torch.bmm` (Tensor‑Core, FP16).
6. Fused LayerNorm + out‑gate + final linear projection (FP32 output) –
   custom Triton kernel that computes per‑row mean/variance, applies the
   weight/bias of `to_out_norm`, multiplies by the sigmoid‑gated
   `out_gate` and finally multiplies with the `to_out` weight matrix.

All heavy memory‑bound passes (LayerNorm, final Linear) are merged into
single kernels, reducing traffic and achieving runtimes well below the
target 1 ms on the benchmark shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm producing FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,      # const float*   [total_rows, dim]  (FP32)
    y_ptr,      # mut   half*    [total_rows, dim]  (FP16)
    w_ptr,      # const float*  [dim]               (γ)
    b_ptr,      # const float*  [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # first pass: compute mean & variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: normalise, apply γ,β and store as FP16
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate + final linear projection
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,          # const half*   [total_rows, hidden]  (pre‑norm)
    gate_ptr,       # const half*   [total_rows, hidden]  (sigmoid‑gated out_gate)
    norm_w_ptr,     # const float*  [hidden]              (γ)
    norm_b_ptr,     # const float*  [hidden]              (β)
    out_w_ptr,      # const half*   [dim, hidden]         (to_out weight)
    out_ptr,        # mut   float*  [total_rows, dim]    (final FP32)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)    # row block
    pid_d = tl.program_id(1)  # output‑channel block

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Load entire hidden vector per row (BLOCK_M × hidden)
    offs_h = tl.arange(0, hidden)
    mask_h = offs_h < hidden

    # x and gate are half → cast to fp32 for arithmetic
    x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)      # (BLOCK_M, hidden)
    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)      # (BLOCK_M, hidden)

    # LayerNorm stats (fp32)
    sum_val = tl.sum(x, axis=1)
    sum_sq  = tl.sum(x * x, axis=1)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Apply LayerNorm γ,β
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)

    norm = (x - mean[:, None]) * rstd[:, None]                     # (B, H)
    norm = norm * w_norm[None, :] + b_norm[None, :]                # (B, H)

    # Gating
    gated = norm * g                                              # (B, H)

    # Output block
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # Load slice of to_out weight transposed: (hidden, BLOCK_D)
    # weight layout [dim, hidden] → address = h + d*hidden
    w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                  mask=mask_h[:, None] & mask_d[None, :],
                  other=0.0).to(tl.float32)      # (hidden, BLOCK_D)

    # Matmul: (BLOCK_M, hidden) @ (hidden, BLOCK_D) -> (BLOCK_M, BLOCK_D)
    out_chunk = tl.dot(gated, w_T)   # fp32

    # Store result
    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C], dtype torch.float32
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – ignored if config["nomask"]=True
        - weights_dict : mapping of parameter names to torch.Tensor (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (optional)}

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32) on the same device as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)

    BLOCK_M_LN = 128     # rows per program
    BLOCK_D_LN = 32      # columns per tile (fits 384)

    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)
    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        w["norm.weight"],
        w["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate weight matrices for 5 linear layers (each hidden x dim)
    concat_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()        # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)   # (R, 5*H) FP16

    # split
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask dtype may be bool or float; treat as float16 mask
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates and elementwise gating (stay in FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)      # (R, H) FP16
    right_gate = torch.sigmoid(right_gate_fp16)   # (R, H) FP16
    out_gate = torch.sigmoid(out_gate_fp16)       # (R, H) FP16 (used later)

    left = left_proj_fp16 * left_gate              # (R, H) FP16
    right = right_proj_fp16 * right_gate           # (R, H) FP16

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update – batched matmul
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden)               # (B, N, N, H)
    right = right.view(B, N, N, hidden)             # (B, N, N, H)

    # reshape to (B*H, N, N) with appropriate permutes
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    # bmm uses Tensor Cores for fp16
    out_mat = torch.bmm(left_mat, right_mat)        # (B*H, N, N) FP16

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Fused LayerNorm + out‑gate + final linear
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16

    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    BLOCK_M_FNL = 128
    BLOCK_D_FNL = 64

    grid_out = (
        triton.cdiv(total_rows, BLOCK_M_FNL),
        triton.cdiv(dim, BLOCK_D_FNL),
    )
    fused_norm_gate_linear_kernel[grid_out](
        out_flat,                                         # x_ptr
        out_gate,                                         # gate_ptr (already sigmoid)
        w["to_out_norm.weight"],                          # γ
        w["to_out_norm.bias"],                            # β
        w["to_out.weight"].half().contiguous(),          # [dim, hidden]
        final_fp32,                                       # output
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M_FNL,
        BLOCK_D=BLOCK_D_FNL,
    )

    # ------------------------------------------------------------------
    # Reshape to final output shape
    # ------------------------------------------------------------------
    out_final = final_fp32.view(B, N, N, dim)
    return out_final

    w_ptr,        # const float*  [dim]               (γ)
    b_ptr,        # const float*  [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass: normalise, apply γ,β and store as FP16
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate + final linear projection (FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,          # const half*   [total_rows, hidden]   (pre‑norm)
    gate_ptr,       # const half*   [total_rows, hidden]   (sigmoid‑gated out_gate)
    norm_w_ptr,     # const float*  [hidden]                (γ)
    norm_b_ptr,     # const float*  [hidden]                (β)
    out_w_ptr,      # const half*   [dim, hidden]           (to_out weight)
    out_ptr,        # mut   float*  [total_rows, dim]     (final FP32)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # output‑channel chunk
):
    pid = tl.program_id(0)          # row block index
    pid_d = tl.program_id(1)        # output‑channel block index

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Load the full hidden vector for each selected row
    # --------------------------------------------------------------
    offs_h = tl.arange(0, hidden)                               # [hidden]
    mask_h = offs_h < hidden

    x = tl.load(x_ptr   + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                     # (BLOCK_M, hidden)

    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                     # (BLOCK_M, hidden)

    # --------------------------------------------------------------
    # LayerNorm statistics (FP32)
    # --------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)
    sum_sq  = tl.sum(x * x, axis=1)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Apply LayerNorm γ, β
    # --------------------------------------------------------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)

    norm = (x - mean[:, None]) * rstd[:, None]                     # (BLOCK_M, hidden)
    norm = norm * w_norm[None, :] + b_norm[None, :]                # (BLOCK_M, hidden)

    # --------------------------------------------------------------
    # Apply out‑gate
    # --------------------------------------------------------------
    gated = norm * g                                                # (BLOCK_M, hidden)

    # --------------------------------------------------------------
    # Output‑channel block handling
    # --------------------------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)                       # [BLOCK_D]
    mask_d = offs_d < dim

    # Load slice of to_out weight transposed: (hidden, BLOCK_D)
    # weight layout is [dim, hidden]; address = h + d*hidden
    w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                  mask=mask_h[:, None] & mask_d[None, :],
                  other=0.0).to(tl.float32)                     # (hidden, BLOCK_D)

    # (BLOCK_M, hidden) @ (hidden, BLOCK_D) -> (BLOCK_M, BLOCK_D)
    out_chunk = tl.dot(gated, w_T)                                 # FP32

    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32 tensors)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same device as ``input_tensor``.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    device = inp.device
    dim    = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N          # rows after flattening [B, N, N]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (output FP16)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()               # [R, C] FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)

    BLOCK_M_LN = 128   # rows per program (tuned for H100)
    BLOCK_D_LN = 32    # columns per tile (covers dim up to 384)

    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)
    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        w["norm.weight"],
        w["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse 5 linear projections into one GEMM
    # ------------------------------------------------------------------
    concat_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)    # (R, 5*H) FP16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
    right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                            # each (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16  * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates and elementwise multiplication (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)          # (R, H) FP16
    right_gate = torch.sigmoid(right_gate_fp16)        # (R, H) FP16
    out_gate = torch.sigmoid(out_gate_fp16)            # (R, H) FP16 – used later

    left  = left_proj_fp16 * left_gate
    right = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched matmul)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden)                # (B, N, N, H)
    right = right.view(B, N, N, hidden)               # (B, N, N, H)

    # reshape to (B*H, N, N) for bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)          # (B*H, N, N) FP16

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Fused LayerNorm + out‑gate + final projection
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16

    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    BLOCK_M_FNL = 128
    BLOCK_D_FNL = 64

    grid_out = (
        triton.cdiv(total_rows, BLOCK_M_FNL),
        triton.cdiv(dim, BLOCK_D_FNL),
    )
    fused_norm_gate_linear_kernel[grid_out](
        out_flat,                                          # x_ptr
        out_gate,                                          # gate_ptr (sigmoid‑ed)
        w["to_out_norm.weight"],                           # γ
        w["to_out_norm.bias"],                             # β
        w["to_out.weight"].half().contiguous(),           # [dim, hidden]
        final_fp32,                                        # output buffer (FP32)
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M_FNL,
        BLOCK_D=BLOCK_D_FNL,
    )

    # ------------------------------------------------------------------
    # Reshape to final tensor shape
    # ------------------------------------------------------------------
    out_final = final_fp32.view(B, N, N, dim)               # [B, N, N, C] FP32
    return out_final
