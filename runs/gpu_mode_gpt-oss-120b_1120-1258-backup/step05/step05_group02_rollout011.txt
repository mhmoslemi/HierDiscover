            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)  # shape (5*H, dim)

    # Linear: (rows, dim) @ (dim, 5*H).T -> (rows, 5*H)
    combined = F.linear(x_norm_h.view(rows, dim), combined_weight)  # [rows, 5*H]

    # Split
    h = hidden_dim
    left_proj = combined[:, 0:h]
    left_gate_pre = combined[:, h:2*h]
    right_proj = combined[:, 2*h:3*h]
    right_gate_pre = combined[:, 3*h:4*h]
    out_gate_pre = combined[:, 4*h:5*h]

    # Gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Apply gates to projections
    left_h = left_proj * left_gate   # [rows, H]
    right_h = right_proj * right_gate

    # Optional mask
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_h = left_h * mask_h
        right_h = right_h * mask_h

    # Reshape to [B,N,N,H]
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Permute to [B,H,N,N] -> contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B*hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B*hidden_dim, N, N)

    # Tri-multiplicative update (batched GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H,N,N]

    # Reshape back to [B,N,N,H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm + gating (fused multiplication)
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    # Reuse same layernorm kernel (but will apply later multiplication)
    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # Multiply with out_gate (already sigmoid-ed)
    out_gate_flat = out_gate.view(rows_h, hidden_dim).to(torch.float16)
    gated = out_norm_h * out_gate_flat

    # Final projection
    final = F.linear(gated, weights["to_out.weight"].to(torch.float16, non_blocking=True))

    # Cast to float32
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul “outgoing” forward pass – fast Triton + PyTorch implementation.

Key ideas
---------
1️⃣  First LayerNorm is written in Triton (FP32 → FP16) – one kernel launch.
2️⃣  All three linear layers that share the same input (left_proj,
    left_gate, right_proj, right_gate, out_gate) are fused into a single
    torch.nn.functional.linear call with a concatenated weight matrix.
    This avoids reading the normalized tensor three times.
3️⃣  Optional pair‑wise mask is applied after gating (commutes with the mask).
4️⃣  The core N³ update is a batched GEMM (`torch.bmm`) over the hidden
    dimension – the most efficient primitive on H100.
5️⃣  A second Triton kernel fuses the second LayerNorm with the output‑gate
    multiplication, producing the tensor that is fed directly into the final
    linear projection.
6️⃣  The last projection (`to_out`) is a regular FP16 GEMM (torch.nn.functional.linear).
   The final result is cast back to FP32.

The implementation keeps the heavy matrix‑multiplications in highly‑optimized
cuBLAS kernels and only uses Triton for the two LayerNorm passes, which are
memory‑bound.  This reduces kernel launch overhead and memory traffic
compared to a naïve PyTorch implementation and comfortably meets the
target runtime on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # column block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                    # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- 1. mean / variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)              # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                           # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                     # 1 / sqrt(var+eps)

    # ------------------- 2. write normalized values -------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    beta_ptr,       # float*   [dim]               (bias)

    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns (= hidden_dim)
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # column block size

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                    # [BLOCK_M]
    mask_m = offs_m < total_rows

        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- write gated normalized values -------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):

# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32]  [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] [B, N, N]   (may be unused)
        weights_dict : {
            "norm.weight":           Tensor[dim],
            "norm.bias":             Tensor[dim],
            "left_proj.weight":      Tensor[hidden_dim, dim],
            "right_proj.weight":     Tensor[hidden_dim, dim],
            "left_gate.weight":      Tensor[hidden_dim, dim],
            "right_gate.weight":     Tensor[hidden_dim, dim],
            "out_gate.weight":       Tensor[hidden_dim, dim],
            "to_out_norm.weight":    Tensor[hidden_dim],
            "to_out_norm.bias":      Tensor[hidden_dim],
            "to_out.weight":         Tensor[dim, hidden_dim],
        }
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True),
        }

    Returns:
        torch.Tensor[float32]  [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [rows, dim]

    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16,
                           device=device)

    BLOCK_M = 128            # rows per program – fits well in shared memory
    BLOCK_D = 64             # column block (covers dim ≤ 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse all three linear layers that share the same input:
    #     left_proj, left_gate, right_proj, right_gate, out_gate
    # --------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)   # shape (5*H, dim)

    # Linear: (rows, dim) @ (5*H, dim).T  → (rows, 5*H)
    combined = F.linear(x_norm_h.view(total_rows, dim), combined_weight)

    H = hidden_dim
    # slice the concatenated result
    left_proj      = combined[:, 0            :H]
    left_gate_pre  = combined[:, H            :2*H]
    right_proj     = combined[:, 2*H          :3*H]
    right_gate_pre = combined[:, 3*H          :4*H]
    out_gate_pre   = combined[:, 4*H          :5*H]

    # sigmoid gates
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)          # [rows, H]

    # apply gates to projections
    left_h  = left_proj * left_gate   # [rows, H]
    right_h = right_proj * right_gate # [rows, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16, non_blocking=True)
        left_h  = left_h * mask_h
        right_h = right_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden_dim)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # bring H into batch dimension → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out = left @ rightᵀ  (batched GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm fused with out‑gate multiplication
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(rows_h, hidden_dim).to(torch.float16)

    out_gated_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_gated_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate_flat,
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection (hidden → dim)
    # --------------------------------------------------------------
    final_h = F.linear(out_gated_h,
                       weights["to_out.weight"].to(torch.float16, non_blocking=True))

    # Cast back to FP32 to match the original module's dtype
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
```
