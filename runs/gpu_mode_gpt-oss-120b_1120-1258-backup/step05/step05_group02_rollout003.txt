"""
custom_kernel
==============

Highly‑optimized forward implementation of the “outgoing” TriMul operation
used in AlphaFold‑3 (and similar protein‑structure models).

Algorithm
---------
1. **Layer‑norm (FP32 → FP16)** on the 4‑D input `[B, N, N, C]` using a
   custom Triton kernel (`layernorm_fp16_kernel`).  Statistics are computed
   in FP32, the normalized tensor is stored in FP16.

2. **Fused projection + gate** for the left and right paths:
   * A single linear layer is performed with a weight matrix that concatenates
     the projection and gate weights (`[2·H, C]`).  The result is split into
     the raw projection and pre‑gate tensors.
   * An additional linear layer provides the *output* gate.

3. **Optional pair‑wise mask** (broadcast over the hidden dimension) is applied
   to the raw projections before the sigmoidal gates.

4. **Apply sigmoidal gates** and obtain the gated left/right tensors (`[B,N,N,H]`).

5. **Tri‑multiplicative update** – a batched matrix‑multiply of the form

        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]

   This is implemented as a custom Triton GEMM (`tri_mul_gemm_kernel`) that
   computes `left @ rightᵀ` for every head (`H`) in a single launch.  All
   arithmetic inside the kernel is FP16 with FP32 accumulation.

6. **Second Layer‑norm** (over the hidden dimension) – the same Triton kernel
   as step 1.

7. **Output gating & final projection** to the original feature dimension.
   The final tensor is cast back to FP32 to match the reference module.

All heavy‑weight work (both LayerNorms and the O(N³·H) TriMul) is performed
in Triton, while cheap element‑wise ops and linear layers use native
torch kernels (FP16).  This yields a fast, memory‑efficient forward pass
compatible with the test configurations (B ≤ 2, N ≤ 1024, C ∈ {128,384},
H = 128).

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (reads any dtype, computes in FP32,
# writes FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *      [total_rows, dim]  (input)
    y_ptr,          # *[half] [total_rows, dim] (output)
    gamma_ptr,      # *      [dim]             (scale)
    beta_ptr,       # *      [dim]             (bias)
    total_rows,     # i32    runtime number of rows
    dim: tl.constexpr,          # i32 static number of columns
    eps: tl.constexpr,          # fp32 constant epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # column block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1) Compute mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) Write normalized values (γ·x̂+β) as FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM for the Tri‑multiplicative update.
# Computes for each head h:
#   out_h = left_h @ right_h.T   where left_h/right_h ∈ ℝ^{N×N}
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    BH, N,                                   # runtime: batch*head, seq‑len
    stride_lbh, stride_li, stride_lk,        # left strides (int)
    stride_rbh, stride_rj, stride_rk,        # right strides
    stride_obh, stride_oi, stride_oj,        # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    eps: tl.constexpr = 0.0,
):
    pid_head = tl.program_id(0)   # which (batch, head)
    pid_m    = tl.program_id(1)   # output‑row block
    pid_n    = tl.program_id(2)   # output‑col block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile  [M, K] -------------------------------------------------
        left_ptrs = (left_ptr
                     + pid_head * stride_lbh
                     + offs_m[:, None] * stride_li
                     + offs_k[None, :] * stride_lk)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # (M,K)

        # ---- right tile transposed  [K, N] ------------------------------------
        # we need right_T[k, j] = right[j, k]
        right_ptrs = (right_ptr
                      + pid_head * stride_rbh
                      + offs_k[:, None] * stride_rk
                      + offs_n[None, :] * stride_rj)
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_n[None, :],
                          other=0.0).to(tl.float32)          # (K,N)

        # ---- accumulate -------------------------------------------------------
        acc += tl.dot(left, right_T)      # (M,N)

    # ---- write result ---------------------------------------------------------
    out_ptrs = (out_ptr
                + pid_head * stride_obh
                + offs_m[:, None] * stride_oi
                + offs_n[None, :] * stride_oj)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32]  shape [B, N, N, C]
        mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be ignored)
        weights_dict : mapping from weight names to tensors (all in float32)
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True)
        }

    Returns:
        torch.Tensor[float32] shape [B, N, N, C]
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, C).contiguous()          # [R, C]
    x_norm_h = torch.empty((total_rows, C), dtype=torch.float16,
                           device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_norm = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_norm](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        C,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape back to [B, N, N, C] (still FP16)
    x_norm_h = x_norm_h.view(B, N, N, C)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (two linear layers in one)
    # ------------------------------------------------------------------
    # concatenate projection & gate weights for left and right paths
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)          # [2·H, C]
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)          # [2·H, C]

    rows = total_rows
    # Linear projections (FP16)
    left_concat = F.linear(x_norm_h.view(rows, C), left_concat_w)   # [R, 2·H]
    right_concat = F.linear(x_norm_h.view(rows, C), right_concat_w)

    # Split into projection & pre‑gate tensors
    left_proj_h      = left_concat[:, :H]                # [R, H]
    left_gate_pre_h  = left_concat[:, H:]                # [R, H]
    right_proj_h     = right_concat[:, :H]
    right_gate_pre_h = right_concat[:, H:]

    # Out‑gate (shared across positions)
    out_gate_h = torch.sigmoid(
        F.linear(
            x_norm_h.view(rows, C),
            weights["out_gate.weight"].to(torch.float16, non_blocking=True)
        )
    )   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h          # [R, H]
    right_h = right_proj_h * right_gate_h        # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] and permute so that (B·H) becomes the batch.
    left = left_h.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right_h.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * H
    left_mat = left.view(BH, N, N)     # [BH, N, N]
    right_mat = right.view(BH, N, N)   # [BH, N, N]
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)  # [BH, N, N]

    # Strides for the three tensors (assume contiguous layout)
    stride_lb, stride_li, stride_lk = left_mat.stride()
    stride_rb, stride_rj, stride_rk = right_mat.stride()
    stride_ob, stride_oi, stride_oj = out_mat.stride()

    # Launch GEMM kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid_gemm = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_gemm_kernel[grid_gemm](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,
        stride_lb, stride_li, stride_lk,
        stride_rb, stride_rj, stride_rk,
        stride_ob, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        eps=eps,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid_norm2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid_norm2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_h.view(B, N, N, H)               # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate_h.view(B, N, N, H)       # element‑wise FP16

    final_h = F.linear(
        gated.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # [rows_h, C]  (FP16)

    final = final_h.view(B, N, N, C).to(torch.float32)   # cast to FP32
    return final
