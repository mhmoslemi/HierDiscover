"""
TriMul outgoing kernel (AlphaFold‑3)

Algorithm (all tensors stay on the GPU):
1️⃣ Input LayerNorm  – row‑wise FP32 reduction → FP16 output (Triton).
2️⃣ Fuse the five linear maps (left/right projections + 3 gates) into a single
   half‑precision GEMM by concatenating the weight matrices.
3️⃣ (Optional) apply the binary mask to the left/right projection results.
4️⃣ Apply sigmoid to the three gate tensors and perform element‑wise gating.
5️⃣ Tri‑multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched FP16 GEMM (`torch.bmm`) where the hidden dimension
   is merged into the batch axis (size B·H).
6️⃣ Fused LayerNorm + out‑gate (both half) – Triton.
7️⃣ Final linear projection back to the model dimension (`dim`) – FP16 GEMM.
8️⃣ Cast the result back to FP32 (as the original PyTorch module does).

Only the two LayerNorm passes are written in Triton; all dense linear algebra
uses cuBLAS/Tensor‑Core which is already optimal on an H100.  The kernels are
parameterised for `dim ≤ 384` and `hidden_dim = 128` (the only hidden‑dim used
in the tests).  Block sizes are chosen to keep kernel launch overhead low while
still fitting comfortably into registers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 stats → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # const float*   [total_rows, dim]  (input FP32)
    y_ptr,               # mut   half*    [total_rows, dim]  (output FP16)
    weight_ptr,          # const float*  [dim]              (γ)
    bias_ptr,            # const float*  [dim]              (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static channel dimension (≤384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------
    # First pass: compute mean and variance (FP32 accumulation)
    # ---------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-µ)/σ
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------

def fused_layernorm_gate_kernel(

    bias_ptr,            # const float*  [dim]              (β)
    gate_ptr,            # const half*   [total_rows, dim] (sigmoid‑ed out_gate)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------
    # First pass: mean / variance (FP32)
    # ---------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by gate, store FP16
    # ---------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],

                 mask=mask_m[:, None] & mask_d[None, :])

    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------

                "hidden_dim" : int (H),
                "nomask"     : bool,
                "eps"        : float, optional (default 1e-5)
          }

    -------
    torch.Tensor

    """

    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]     # H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # 1️⃣ Input LayerNorm (produces FP16)
    # ------------------------------------------------------------------
    inp_flat = inp.reshape(total_rows, dim).contiguous()   # FP32

    BLOCK_D = 128      # covers dim ≤ 384 with ≤ 3 passes

    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each: [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # shape (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)   # bias is zero in the reference

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H)

    # 2a️⃣ Apply mask (if required) – broadcast on hidden dimension
    # ------------------------------------------------------------------
    if not nomask:

        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates and element‑wise gating (still half)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)       # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)     # (R, H)
    out_gate = torch.sigmoid(out_gate_fp16)         # (R, H) for later use

    left_fp16 = left_proj_fp16 * left_gate          # (R, H)
    right_fp16 = right_proj_fp16 * right_gate       # (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left = left_fp16.view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)

    # Rearrange to (B*H, N, N) where the hidden dim is merged into the batch axis.
    # The `.contiguous()` ensures a compact layout for cuBLAS.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N) # (B*H, N, N)

    # Batched matrix multiplication (Tensor‑Core FP16)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate_fp16 = torch.empty_like(out_flat)        # (R, H) half

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()          # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16) # (R, dim) half

    # Cast back to FP32 as expected by the reference module
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
