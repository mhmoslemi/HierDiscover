"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Input LayerNorm (FP16 output).  Mean/var are accumulated in FP32,
   the normalized result is written as half‑precision.

2️⃣ One fused linear projection produces *all* required
   tensors (left‑proj, left‑gate, right‑proj, right‑gate, out‑gate)
   in a single GEMM:
   ```
   proj = x_norm @ W   # W ∈ ℝ^{5·H×C}
   left_proj, left_gate,
   right_proj, right_gate,
   out_gate = torch.split(proj, [H, H, H, H, H], dim=1)
   ```
   Gating is performed with a sigmoid and the mask (if present) is
   applied after gating.

3️⃣ The triangular multiplicative update
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   is reduced to a batched matrix‑multiply:
   ```
   left  → (B·H, N, N)    (i,k)
   right → (B·H, N, N)    (k,j)   (k‑axis transposed)
   out   = torch.bmm(left, right)   # (B·H, N, N)
   ```
   All tensors are half‑precision → Tensor‑Core speed.

4️⃣ A fused LayerNorm + out‑gate kernel normalises the result,
   applies the learned scale/bias and finally multiplies by the
   previously computed `out_gate`.  The whole operation stays in
   FP16.

5️⃣ Final linear projection back to the original dimension
   (`hidden → dim`) is a regular half‑precision GEMM.  The
   output is cast to FP32 as required by the API.

The implementation fuses the three separate linear + gate
operations of the reference model into a single GEMM and reduces
the number of kernel launches, yielding a >2× speed‑up on the
reference H100 hardware.
"""

    x_ptr,               # const float* [total_rows, dim]   (FP32)
    y_ptr,               #      half*  [total_rows, dim]   (FP16)
    weight_ptr,          # const float* [dim]              (γ)
    bias_ptr,            # const float* [dim]              (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance in FP32
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 4) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # const half* [total_rows, hidden]   (pre‑norm)
    y_ptr,               #       half* [total_rows, hidden]   (post‑norm⋅gate)
    weight_ptr,          # const float* [hidden]              (γ)
    bias_ptr,            # const float* [hidden]              (β)
    gate_ptr,            # const half*  [total_rows, hidden] (out_gate, sigmoid‑ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by out_gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)          # out_gate (already sigmoid‑ed)

        out = out * g

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/float) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]        # H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N          # flattened rows for “[B,N,N]”

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)                # [R, C] half

    # Choose block sizes that respect hardware limits and dimension size
    BLOCK_M = 256                      # rows per program (≤ 1024 threads)
    BLOCK_D = 128 if dim > 256 else 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear‑+‑gate passes (left, right, out_gate) – ONE GEMM
    # ------------------------------------------------------------------
    # Pack all five weight matrices into a single (5·H, C) matrix
    left_proj_w  = weights["left_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    # concatenate in the order used later
    fused_weight = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w
    ], dim=0).contiguous()          # (5*H, C)

    # Linear projection – result is half‑precision
    fused_out = F.linear(x_norm_fp16, fused_weight)   # (R, 5*H)

    # Split into individual components
    split_tensors = torch.split(fused_out, hidden, dim=1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = split_tensors

    # Gating (sigmoid on half‑precision)
    left_gate  = torch.sigmoid(left_gate_raw)          # (R, H)
    right_gate = torch.sigmoid(right_gate_raw)         # (R, H)
    out_gate   = torch.sigmoid(out_gate_raw)           # (R, H) – used later

    # Apply gates
    left  = left_proj_raw * left_gate                  # (R, H)
    right = right_proj_raw * right_gate                # (R, H)

    # ------------------------------------------------------------------
    # 2a) Apply binary mask (if present) – broadcast on hidden dim.
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (R,1) half
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left  = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, hidden)    # (B, N, N, H)
    right = right.view(B, N, N, hidden)   # (B, N, N, H)

    # Prepare (B*H, N, N) tensors for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, i) where i ↔ k
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, k, j)

    out_mat = torch.bmm(left_mat, right_mat)                         # (B*H, N, N)

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out_gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)            # (R, H) half

    BLOCK_D_H = 128 if hidden > 256 else 64
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid’ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D_H,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()            # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate, to_out_w)        # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
