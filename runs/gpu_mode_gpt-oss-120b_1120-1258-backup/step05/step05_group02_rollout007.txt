"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton + PyTorch.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the 4‑D tensor.
2️⃣  Fuse each projection with its gating weight (dim → 2·hidden_dim) via a single
    `torch.nn.functional.linear`.  The result is split into projection and pre‑gate
    tensors and the sigmoid is applied.
3️⃣  Optional pair‑wise mask (broadcast over the hidden dim) is applied.
4️⃣  Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
    is a batch of `BH = B·hidden_dim` square matrix multiplications.
    This is realised by a custom Triton kernel that computes all `BH` GEMMs
    in a single launch using Tensor‑Core‑accelerated `tl.dot`.  The kernel works
    on FP16 data and accumulates in FP32, then writes the result back as FP16.
5️⃣  Second LayerNorm (over the hidden dimension) – same Triton kernel as step 1.
6️⃣  Output gate (sigmoid) and final linear projection (hidden → dim).  The
    result is cast back to float32.

All heavy compute (two LayerNorms and the batched GEMM) runs on Triton,
while projections/gates use high‑throughput cuBLAS via `F.linear`.  The
implementation works for the full test matrix (B≤2, N≤1024, dim∈{128,384},
hidden_dim=128) on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr,      # input [rows, dim] (FP32), output [rows, dim] (FP16)
    gamma_ptr, beta_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- pass 1: compute mean / var ----------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                     # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- pass 2: write normalized values ------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                     # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the Tri‑multiplicative update.
# Computes, for each (batch, head) pair:
#   C = A @ B^T   where A,B ∈ ℝ^{N×N} (FP16), result C ∈ ℝ^{N×N} (FP16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # [B*H, N, N] FP16 pointers
    batch: tl.constexpr,               # total number of BH matrices
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)            # which (B*H) matrix
    pid_m = tl.program_id(0)            # tile row
    pid_n = tl.program_id(1)            # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension (the shared “seq_len” dimension)
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_row
                            + offs_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # FP16

        # B tile (transposed): we need B[j,k] -> load as [BLOCK_K, BLOCK_N]
        b_ptrs = b_batch + (offs_n[None, :] * stride_b_row
                            + offs_k[:, None] * stride_b_col)
        b = tl.load(b_ptrs,
                    mask=mask_n[None, :] & mask_k[:, None],
                    other=0.0)                     # FP16

        # matmul accumulation (FP32)
        acc += tl.dot(a, b)                # (M,K)·(K,N) → (M,N)

    # write result as FP16
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_row
                        + offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32]  shape [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] shape [B, N, N]  (may be ignored)
        weights_dict : {
            "norm.weight":           Tensor[dim],
            "norm.bias":             Tensor[dim],
            "left_proj.weight":      Tensor[hidden_dim, dim],
            "right_proj.weight":     Tensor[hidden_dim, dim],
            "left_gate.weight":      Tensor[hidden_dim, dim],
            "right_gate.weight":     Tensor[hidden_dim, dim],
            "out_gate.weight":       Tensor[hidden_dim, dim],
            "to_out_norm.weight":    Tensor[hidden_dim],
            "to_out_norm.bias":      Tensor[hidden_dim],
            "to_out.weight":         Tensor[dim, hidden_dim],
        }
        config_dict : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True),
        }

    Returns
    -------
    torch.Tensor[float32]  shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [rows, dim] FP32

    x_norm_fp16 = torch.empty((total_rows, dim),
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # column block (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape back to [B, N, N, dim] (still FP16)
    x_norm_fp16 = x_norm_fp16.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Project + gate (fused linear for proj & gate)
    # --------------------------------------------------------------
    # concatenate proj and gate weights (dim → 2·hidden_dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]],
        dim=0,
    ).to(dtype=torch.float16, device=device, non_blocking=True)   # [2·H, dim]

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]],
        dim=0,
    ).to(dtype=torch.float16, device=device, non_blocking=True)  # [2·H, dim]

    rows = total_rows
    # Linear: (rows, dim) @ (2·H, dim).T → (rows, 2·H)
    left_concat = F.linear(x_norm_fp16.view(rows, dim), left_concat_w)   # FP16
    right_concat = F.linear(x_norm_fp16.view(rows, dim), right_concat_w)

    # Split projection / pre‑gate
    left_proj_fp16 = left_concat[:, :hidden_dim]          # (rows, H)
    left_gate_pre_fp16 = left_concat[:, hidden_dim:]      # (rows, H)
    right_proj_fp16 = right_concat[:, :hidden_dim]
    right_gate_pre_fp16 = right_concat[:, hidden_dim:]

    # out‑gate (shared across i,j)
    out_gate_fp16 = torch.sigmoid(
        F.linear(
            x_norm_fp16.view(rows, dim),
            weights["out_gate.weight"].to(dtype=torch.float16,
                                          device=device,
                                          non_blocking=True),
        )
    )   # (rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional pair‑wise mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.view(rows, 1).to(dtype=torch.float16,
                                                device=device,
                                                non_blocking=True)
        left_proj_fp16 = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate_fp16 = torch.sigmoid(left_gate_pre_fp16)
    right_gate_fp16 = torch.sigmoid(right_gate_pre_fp16)

    left_fp16 = left_proj_fp16 * left_gate_fp16       # (rows, H)
    right_fp16 = right_proj_fp16 * right_gate_fp16    # (rows, H)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via custom Triton GEMM
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)

    # bring H into batch dimension → [B·H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)   # (BH, N, N) FP16
    right_mat = right_perm.view(BH, N, N) # (BH, N, N) FP16

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)  # (BH, N, N)

    # strides (as Python ints)
    stride_a_batch = left_mat.stride(0)
    stride_a_row   = left_mat.stride(1)
    stride_a_col   = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row   = right_mat.stride(1)
    stride_b_col   = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row   = out_mat.stride(1)
    stride_c_col   = out_mat.stride(2)

    # launch parameters (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )
    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,          # M
        N,          # N (square)
        N,          # K
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Output gate & final projection
    # --------------------------------------------------------------
    out_gate = out_gate_fp16.view(B, N, N, hidden_dim)   # FP16
    gated = out_norm * out_gate                           # element‑wise FP16

    final_fp16 = F.linear(
        gated.view(rows_h, hidden_dim),
        weights["to_out.weight"].to(dtype=torch.float16,
                                    device=device,
                                    non_blocking=True),
    )   # (rows_h, dim) FP16

    # Cast back to FP32 for the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
