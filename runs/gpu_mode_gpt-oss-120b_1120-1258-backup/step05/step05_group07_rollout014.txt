@triton.jit
def layernorm_kernel(
    x_ptr,   # pointer to input (fp16)
    y_ptr,   # pointer to output (fp16)
    weight_ptr,  # gamma (fp32)
    bias_ptr,    # beta (fp32)
    rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # allocate shared memory for a tile
    tile = tl.alloc_shared((BLOCK_M, BLOCK_D), dtype=tl.float32)

    sum = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # first pass load into shared mem + accumulate
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # compute indices
        x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals_fp16 = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        # cast to fp32 for accumulation
        vals = vals_fp16.to(tl.float32)

        # store to shared memory
        # compute shared memory ptrs: tile + (offs_m offset for rows?) Actually tile is (BLOCK_M, BLOCK_D) per block, we can store using offset in rows dimension.
        tl.store(tile + (tl.arange(0, BLOCK_M)[:, None] * BLOCK_D + tl.arange(0, BLOCK_D)[None, :]), vals, mask=mask_m[:, None] & mask_d[None, :])
        # But we cannot store across whole BLOCK_M including rows that are out-of-range; maybe need masked store.

        sum += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute normalized output reading from shared memory.
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load from shared memory
        vals = tl.load(tile + (tl.arange(0, BLOCK_M)[:, None] * BLOCK_D + tl.arange(0, BLOCK_D)[None, :]), mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32) # but tile is already fp32.
        # Actually we need to load each row's tile offset relative.

        # compute normalized
        norm = (vals - mean[:, None]) * rstd[:, None]

        # load weight & bias
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        # store to output
        y_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])

Pseudo-code:

```
@triton.jit
def layernorm_fp16_onepass_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # allocate shared mem
    tile = tl.alloc_shared((BLOCK_M, BLOCK_D), dtype=tl.float32)

    # 1. load tile, compute sum and sumsq
    sum = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load fp16 values
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals_fp16 = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        )
        vals = vals_fp16.to(tl.float32)

        # store in shared memory for later use
        # compute shared memory offsets
        tile_ptrs = tile + (tl.arange(0, BLOCK_M)[:, None] * BLOCK_D + tl.arange(0, BLOCK_D)[None, :])
        tl.store(tile_ptrs, vals, mask=mask_m[:, None] & mask_d[None, :])

        sum += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # 2. Normalize and write output
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load from shared mem
        tile_ptrs = tile + (tl.arange(0, BLOCK_M)[:, None] * BLOCK_D + tl.arange(0, BLOCK_D)[None, :])
        vals = tl.load(tile_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        )  # already fp32

        # normalize
        norm = (vals - mean[:, None]) * rstd[:, None]

        # weight & bias
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        # store result
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])
```

def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    # row indices handled by this block
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # allocate shared memory for row values
    # shape (BLOCK_M, dim)
    row_vals = tl.alloc_shared((BLOCK_M, dim), dtype=tl.float32)

    # accumulators
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # first pass: load and store to shared memory, compute sum & sumsq
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # compute pointers for each row and column
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals_fp16 = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        vals = vals_fp16.to(tl.float32)

        # store to shared memory
        row_vals_ptr = row_vals + (tl.arange(0, BLOCK_M)[:, None] * dim + offs_d[None, :])
        tl.store(row_vals_ptr,
                 vals,
                 mask=mask_m[:, None] & mask_d[None, :])

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: normalize and write output
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load from shared memory

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the input channels (C) – Triton kernel.
    * One global load per element (FP16) → the values are kept in shared memory
      and reused for the second (normalisation) pass, halving the traffic.
2️⃣  Fuse the five linear projections (left/right + three gates) into a single
    half‑precision GEMM by concatenating their weight matrices.
3️⃣  Apply the optional pair‑wise mask *before* the element‑wise gates.
4️⃣  Apply sigmoid gates and element‑wise gating (still FP16).
5️⃣  Triangular multiplicative update (outgoing version):
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    – realised as a batched GEMM `left @ rightᵀ` on tensors of shape
      (B·H, N, N) with Tensor‑Core kernels.
6️⃣  Second LayerNorm over the hidden dimension (H) – same Triton kernel.
7️⃣  Multiply by the output gate and project back to the original dimension C.
8️⃣  Cast the result to float32.

All heavy linear algebra uses fp16 Tensor‑Core kernels; the only Triton
kernels are the two LayerNorms.  The implementation works for the full
range of test configurations (B∈{1,2}, N∈{128…1024}, C∈{128,384},
H=128) and returns a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp16 input, fp32 stats, fp16 output)
#   * reads each element once from global memory,
#   * stores a copy in shared memory,
#   * computes mean/var,
#   * normalises from the shared copy.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                     # *const fp16*  [total_rows, dim]
    y_ptr,                     # *mut   fp16*  [total_rows, dim]
    weight_ptr,                # *const fp32*  [dim]    (γ)
    bias_ptr,                  # *const fp32*  [dim]    (β)
    total_rows,                # i32
    dim: tl.constexpr,         # channel dimension (C or H)
    eps: tl.constexpr,         # small constant for numerical stability
    BLOCK_M: tl.constexpr,     # rows handled per program (≈32)
    BLOCK_D: tl.constexpr,     # columns handled per inner loop (≈64)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    #  Allocate shared memory for ONE BLOCK_M × dim tile.
    #  Total shared memory per block = BLOCK_M * dim * 4 bytes.
    #  For dim≤384 and BLOCK_M=32 this fits easily into the 64 KB limit.
    # ------------------------------------------------------------------
    row_vals = tl.alloc_shared((BLOCK_M, dim), dtype=tl.float32)

    # accumulators for mean / variance (one scalar per row)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # ------------------------------------------------------------------
    # 1️⃣ Load the row, store a copy to shared memory and accumulate stats.
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        # global pointer for this tile
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals_fp16 = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )                                 # fp16
        vals = vals_fp16.to(tl.float32)                     # fp32 for reduction

        # store a copy into shared memory (will be reused later)
        store_ptr = row_vals + (tl.arange(0, BLOCK_M)[:, None] * dim + offs_d[None, :])
        tl.store(store_ptr, vals, mask=mask_m[:, None] & mask_d[None, :])

        # update per‑row sums
        sum_val += tl.sum(vals, axis=1)      # Σ_k x
        sum_sq  += tl.sum(vals * vals, axis=1)  # Σ_k x²

    # ------------------------------------------------------------------
    # Compute mean, variance and rstd (1/√(var+eps))  – fp32.
    # ------------------------------------------------------------------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Normalise from the shared copy, apply γ & β and write fp16 out.
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        load_ptr = row_vals + (tl.arange(0, BLOCK_M)[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            load_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )                                 # fp32 (original value)

        # normalisation
        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        # γ (weight) and β (bias) – fp32 scalars
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        # store result
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be unused.
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Shape preparation
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output) – Triton
    # --------------------------------------------------------------
    # Convert input to fp16 once – the kernel works entirely in fp16.
    x_fp16 = input_tensor.to(torch.float16).contiguous()
    x_flat = x_fp16.view(total_rows, dim)      # (total_rows, C)

    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # Block size choice: BLOCK_M=32 rows → ~50 KB shared memory for dim=384.
    BLOCK_M = 32
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],               # γ  (float32)
        weights["norm.bias"],                 # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # --------------------------------------------------------------
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    # Concatenate and cast once to fp16
    concat_w = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).half().contiguous()                         # (5*H, C)

    # Fused GEMM
    proj_all = F.linear(x_norm_fp16, concat_w)     # (total_rows, 5*H)  fp16

    # Split the fused projection
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden_dim, dim=1
        )                                          # each (total_rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied BEFORE gating)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(total_rows, 1)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)          # (total_rows, H)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape to 4‑D tensors for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)          # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Move H into the batch dimension – this yields BH independent GEMMs
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)                # (BH, N, N)

    # Batched matrix multiply: left @ rightᵀ   (fp16 Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N)

    # Restore 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (fp16 output)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)            # (total_rows, H)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)    # (B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                            # (B, N, N, H)  fp16

    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(gated.view(total_rows, hidden_dim), to_out_w_fp16)  # (total_rows, C)

    # --------------------------------------------------------------
    # 8️⃣ Cast to fp32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
