"""
Trinion (outgoing) “TriMul’’ forward pass – AlphaFold‑3 style

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the input channel dimension `C`.
    *Implemented in Triton.*  Mean/variance are accumulated in fp32,
    the normalized output is emitted in fp16.

2️⃣  Fuse the five linear projections (left/right + three gates) into a
    single half‑precision GEMM (`torch.nn.functional.linear`).  This
    reduces memory traffic from 5 separate GEMMs to one.

3️⃣  (Optional) element‑wise mask is applied to the *left* and *right*
    projections **before** the gates.

4️⃣  Apply sigmoid to the three gate tensors and use them to gate the
    left and right projections.

5️⃣  Outgoing triangular multiplication:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    The heavy O(N³·H) work is performed by a batched half‑precision
    GEMM (`torch.bmm`).  We reshape the tensors to (B·H, N, N) – a copy
    is required because the hidden dimension must become the batch
    axis for cuBLAS.

6️⃣  Second LayerNorm over the hidden dimension `H` (again Triton).

7️⃣  Element‑wise gate `out_gate` followed by the final linear projection
    back to the original channel dimension `C`.

All heavy arithmetic stays in float16 (Tensor‑Core friendly) and the
final result is cast back to float32 as required by the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const  float*  [rows, dim]  (fp32 or fp16)
    y_ptr,               # *mut    float16* [rows, dim] (output)
    weight_ptr,          # *const  float*  [dim]       (γ)
    bias_ptr,            # *const  float*  [dim]       (β)
    total_rows,          # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a block (fp32 → fp32, fp16 is also OK because we cast)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                 # rows when the tensor is flattened

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (fp16 output) – Triton
    # ------------------------------------------------------------------
    # flatten to 2‑D; keep the original fp32 dtype for accurate stats
    x_flat = input_tensor.contiguous().view(total_rows, dim)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Choose block sizes – dim ≤ 384 ⇒ BLOCK_D = 128 covers it in ≤3 loops
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Fuse five linear projections (left/right + three gates)
    # ------------------------------------------------------------------
    # All projection weights share the same shape   [hidden_dim, dim]
    # Concatenate them on the output (row) axis and perform a single GEMM
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()   # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)   # (total_rows, 5*H), fp16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )                                            # each (total_rows, H)

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (applied BEFORE gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [total_rows, 1] → fp16 broadcast
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣  Apply sigmoid gates and element‑wise gating (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)      # (total_rows, H)
    right_gate = torch.sigmoid(right_gate_fp16)    # (total_rows, H)
    out_gate   = torch.sigmoid(out_gate_fp16)      # (total_rows, H)

    left_fp16  = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Outgoing triangular multiplication (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to 4‑D layout
    left  = left_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Move hidden dimension into the batch axis – copy required for cuBLAS
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)                # (BH, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N), fp16

    # Restore original 4‑D layout (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm over the hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # (total_rows, H)
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,                     # input (fp16) – cast to fp32 inside kernel
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣  Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # (B, N, N, H), fp16

    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)     # (total_rows, H)

    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # (total_rows, C)

    # ------------------------------------------------------------------
    # 8️⃣  Cast to fp32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
