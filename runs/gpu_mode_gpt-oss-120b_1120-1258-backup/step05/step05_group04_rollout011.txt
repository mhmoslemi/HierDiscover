    pid_j = tl.program_id(1)

"""
Highly‑optimized forward implementation of the **outgoing** TriMul operator
(AlphaFold‑3).  The kernel fuses the expensive N³ matrix‑multiply across the
hidden dimension using a custom Triton GEMM, eliminating the intermediate
permutes / copies that the naïve torch.bmm implementation needs.  The
pipeline is:

1️⃣  FP32 → FP16 LayerNorm (custom Triton, two‑pass).
2️⃣  Fuse the five linear projections (left/right + 3 gates) into one cheap
    half‑precision GEMM.
3️⃣  Apply the optional binary mask and the three sigmoid gates.
4️⃣  **Custom batched GEMM** that computes
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    directly into a `[B, N, N, H]` half‑tensor (no view/permute).
5️⃣  Fused LayerNorm + out‑gate + final linear projection (custom Triton).

All heavy work stays on‑device, the final result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# 1) LayerNorm (FP32 → FP16) – two‑pass reduction
# ------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,  # *const float*  [rows, dim]
    y_ptr,  # *mut half*      [rows, dim]
    w_ptr,  # *const float*  [dim]      (γ)
    b_ptr,  # *const float*  [dim]      (β)
    rows: tl.int32,
    DIM: tl.constexpr,          # static hidden dimension (≤ 384)
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per iteration
):
    pid = tl.program_id(0)
    row_base = pid * BLOCK_M
    offs_m = row_base + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: mean & variance (FP32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM
        ptr = x_ptr + offs_m[:, None] * DIM + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [M, D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / DIM
    var  = sum_sq / DIM - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # ---- second pass: normalise, γ, β and store as FP16 ----
    for d in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM
        ptr = x_ptr + offs_m[:, None] * DIM + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]

        tl.store(y_ptr + offs_m[:, None] * DIM + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ------------------------------------------------------------
# 4) Batched GEMM for the triangular multiplication
#    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#    left / right / out have shape [B, N, N, H] (contiguous, H last)
# ------------------------------------------------------------
@triton.jit
def trimul_batch_matmul_kernel(
    left_ptr,   # *const half*  [B, N, N, H]
    right_ptr,  # *const half*  [B, N, N, H]
    out_ptr,    # *mut   half*  [B, N, N, H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    # strides for the 4‑D tensors
    stride_lb, stride_li, stride_lk, stride_lh,   # left
    stride_rb, stride_rj, stride_rk, stride_rh,   # right
    stride_ob, stride_oi, stride_oj, stride_oh,   # out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)               # tile over rows (i)
    pid_j = tl.program_id(1)               # tile over cols (j)
    pid_bh = tl.program_id(2)              # combined batch id = b*H + h

    # decode batch and hidden channel
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # start indices for the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # loop over k dimension
    for k0 in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k0 * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]  (M x K)
        a_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right_T[k, j]  (K x N)  – we load the transpose directly
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Matrix multiply for this K‑tile
        acc += tl.dot(a, b)   # (M, K) @ (K, N) → (M, N)

    # write back (store as FP16)
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate + final linear projection
# ------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # *const half*  [rows, hidden]
    gate_ptr,            # *const half*  [rows, hidden]  (sigmoid‑ed out_gate)
    norm_w_ptr,          # *const float* [hidden]       (γ)
    norm_b_ptr,          # *const float* [hidden]       (β)
    out_w_ptr,           # *const half*  [dim, hidden] (to_out weight, transposed)
    out_ptr,             # *mut   float* [rows, dim]
    rows: tl.int32,
    DIM: tl.constexpr,          # output dim (≤ 384)
    HIDDEN: tl.constexpr,       # hidden dim (≤ 384)
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # output‑channel chunk
):
    pid = tl.program_id(0)          # row block
    pid_d = tl.program_id(1)        # output‑channel block
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- load hidden vector ----------
    offs_h = tl.arange(0, HIDDEN)
    mask_h = offs_h < HIDDEN

    x = tl.load(x_ptr + offs_m[:, None] * HIDDEN + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)               # (M, H)

    g = tl.load(gate_ptr + offs_m[:, None] * HIDDEN + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)               # (M, H)

    # ---------- per‑row LayerNorm ----------
    sum_val = tl.sum(x, axis=1)
    sum_sq  = tl.sum(x * x, axis=1)
    mean = sum_val / HIDDEN
    var  = sum_sq / HIDDEN - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # ---------- apply γ, β ----------
    w = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # γ (H)
    b = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # β (H)

    norm = (x - mean[:, None]) * rstd[:, None]
    norm = norm * w[None, :] + b[None, :]

    # ---------- out‑gate ----------
    gated = norm * g

    # ---------- final linear (dim × hidden) ----------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < DIM

    # weight is stored as [dim, hidden]; we need its transpose
    w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * HIDDEN,
                  mask=mask_h[:, None] & mask_d[None, :],
                  other=0.0).to(tl.float32)                  # (H, D)

    out_chunk = tl.dot(gated, w_T)   # (M, D)  FP32

    tl.store(out_ptr + offs_m[:, None] * DIM + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])

    hidden = cfg["hidden_dim"]       # H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                # flattened rows for LayerNorm & final LN

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()      # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)

    BLOCK_M = 64
    BLOCK_D = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection + 5 gate projections
    # --------------------------------------------------------------
    # concatenate the five weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    proj = F.linear(x_norm_fp16, concat_w)    # (total_rows, 5*hidden)  FP16

    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj, hidden, dim=1
    )                                          # each (R, H)

    # --------------------------------------------------------------
    # 2a) optional mask (broadcasted across hidden)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 2b) sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)      # (R, H) half
    right_gate = torch.sigmoid(right_gate)     # (R, H) half
    out_gate   = torch.sigmoid(out_gate)       # (R, H) half  – later used in final LN

    left  = left_proj * left_gate              # (R, H)
    right = right_proj * right_gate            # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Reshape to 4‑D tensors for the triangular update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, hidden)   # (B, N, N, H)
    right = right.view(B, N, N, hidden)  # (B, N, N, H)

    # --------------------------------------------------------------
    # 4️⃣ Custom batched GEMM for the N³ multiplicative update
    # --------------------------------------------------------------
    out_fp16 = torch.empty_like(left)   # (B, N, N, H), half‑precision

    # strides of the 4‑D tensors (in element counts)
    stride_lb = left.stride(0)          # B‑stride = N*N*H
    stride_li = left.stride(1)          # i‑stride = N*H
    stride_lk = left.stride(2)          # k‑stride = H
    stride_lh = left.stride(3)          # h‑stride = 1

    stride_rb = right.stride(0)
    stride_rj = right.stride(1)         # j‑stride = N*H
    stride_rk = right.stride(2)         # k‑stride = H
    stride_rh = right.stride(3)

    stride_ob = out_fp16.stride(0)
    stride_oi = out_fp16.stride(1)
    stride_oj = out_fp16.stride(2)
    stride_oh = out_fp16.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_matmul = (
        triton.cdiv(N, BLOCK_M),               # i‑tiles
        triton.cdiv(N, BLOCK_N),               # j‑tiles
        B * hidden,                            # batch*hidden
    )
    trimul_batch_matmul_kernel[grid_matmul](
        left,
        right,
        out_fp16,
        B,
        N,
        hidden,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final fused LayerNorm + out‑gate + linear projection
    # --------------------------------------------------------------
    out_flat = out_fp16.view(total_rows, hidden).contiguous()   # (R, H) half
    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    grid_out = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    layernorm_gate_linear_kernel[grid_out](
        out_flat,
        out_gate,                       # already sigmoid‑ed (half)
        weights["to_out_norm.weight"],  # γ (float32)
        weights["to_out_norm.bias"],    # β (float32)
        weights["to_out.weight"].half().contiguous(),  # (dim, hidden)
        final_fp32,
        total_rows,
        dim=dim,
        HIDDEN=hidden,
        EPS=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape to final tensor layout
    # --------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)   # [B, N, N, C] float32
    return final
```
