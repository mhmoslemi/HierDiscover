"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass

Main ideas
----------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) is delegated to
    ``torch.nn.functional.layer_norm`` – an already‑highly tuned kernel.

2️⃣  Five linear projections (left, right, three gates) are fused into a
    single FP16 GEMM by concatenating the five weight matrices.

3️⃣  Optional mask is applied only to the two value projections.

4️⃣  The expensive N³ “pairwise multiplication” is performed with a
    batched ``torch.matmul`` on the permuted tensors (B·H independent
    GEMMs of size N×N).  This uses Tensor‑Core kernels on H100.

5️⃣  The second LayerNorm, the output‑gate, and the final linear projection
    back to the model dimension are **fused** into ONE Triton kernel.
    The kernel:
        – loads the non‑contiguous result of the pair‑wise multiplication,
        – computes per‑row mean/variance across the hidden dimension,
        – normalises, applies γ/β,
        – multiplies by the per‑row output‑gate,
        – performs a GEMM with ``to_out.weight`` (H → dim),
        – writes the final FP16 result directly into a contiguous buffer.
    This eliminates the intermediate hidden tensor and a second GEMM,
    drastically cutting memory traffic.

The implementation below follows the reference PyTorch code but
replaces the two custom kernels with the faster strategy described
above.  Only the fused LN + gate + output‑linear kernel is written in
Triton; all other heavy operations use PyTorch’s highly‑optimised
CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 5️⃣  Fused second LayerNorm + out‑gate + final linear (single kernel)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_out_kernel(
    src_ptr,               # const   *  [B, H, N, N] (fp16) – non‑contiguous
    gate_ptr,              # const   *  [R, H]      (fp16) – contiguous
    dst_ptr,               # mutable *  [R, D]      (fp16) – contiguous output
    ln_weight_ptr,         # const   *  [H]         (fp32)
    ln_bias_ptr,           # const   *  [H]         (fp32)
    w_out_ptr,             # const   *  [D, H]      (fp16) – row‑major (out_dim × hidden_dim)
    total_rows,            # int                 = B·N·N
    hidden_dim,            # int                 = H  (compile‑time)
    out_dim,               # int                 = D  (compile‑time)
    seq_len,               # int                 = N
    stride_b,              # int                 stride on batch   of src
    stride_h,              # int                 stride on hidden  of src
    stride_i,              # int                 stride on dim i   of src
    stride_j,              # int                 stride on dim j   of src
    eps,                   # float               LayerNorm epsilon
    BLOCK_M: tl.constexpr, # rows per program
    BLOCK_H: tl.constexpr, # hidden‑dim tile
    BLOCK_D: tl.constexpr, # output‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Decode (b,i,j) from the linear row index
    # ------------------------------------------------------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    # base offset for the first hidden element of each row
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j   # [BLOCK_M]

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance over the hidden dimension
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < hidden_dim

        # Load a tile of the hidden vector (stride_h)
        ptr = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
        val = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass 2 – LayerNorm, out‑gate, and final linear projection
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(out_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < out_dim

        # accumulator for the dot‑product with w_out
        acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

        for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
            mask_h = offs_h < hidden_dim

            # ----- load raw hidden value -------------------------------------------------
            src = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
            raw = tl.load(src,
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

            # ----- LayerNorm ------------------------------------------------------------
            norm = (raw - mean[:, None]) * rstd[:, None]       # fp32

            # ----- γ/β (hidden‑dim) ----------------------------------------------------
            w_ln = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
            b_ln = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)   # fp32
            hidden = norm * w_ln[None, :] + b_ln[None, :]      # fp32

            # ----- out‑gate -------------------------------------------------------------
            gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            hidden = hidden * gate                            # fp32

            # layout: w_out[d, h]; stride0 = hidden_dim, stride1 = 1
            w = w_out_ptr + offs_d[:, None] * hidden_dim + offs_h[None, :]   # [BLOCK_D, BLOCK_H]
            w_tile = tl.load(w,
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)                     # fp32

            # ----- matrix‑multiply hidden (M×H_tile) × w_tileᵀ (H_tile×D_tile) ----------
            acc += tl.dot(hidden, tl.trans(w_tile))            # (M, D)

        # store the final result (fp16) -------------------------------------------------
        dst = dst_ptr + offs_m[:, None] * out_dim + offs_d[None, :]
        tl.store(dst, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
#  Custom forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # 0) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]           # model dimension (e.g. 128 or 384)
    hidden_dim   = config["hidden_dim"]   # hidden dimension (always 128 in the tests)
    nomask       = config.get("nomask", True)
    eps          = 1e-5
    device       = input_tensor.device
    B, N, _, _   = input_tensor.shape
    total_rows   = B * N * N                # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (FP32 → FP16) – use PyTorch's highly tuned kernel
    # --------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)   # FP16 for the following GEMM

    # --------------------------------------------------------------
    # 2) One fused linear producing the five projections (5×hidden_dim outputs)
    # --------------------------------------------------------------
    # concatenate weight matrices once and cast to FP16
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5*H, dim)

    proj = F.linear(x_norm, w_concat)       # (R, 5*H), fp16, no bias

    H = hidden_dim
    left_proj  = proj[:, 0:H]
    right_proj = proj[:, H:2*H]
    left_gate  = proj[:, 2*H:3*H]
    right_gate = proj[:, 3*H:4*H]
    out_gate   = proj[:, 4*H:5*H]          # (R, H)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4) Optional mask (applied only to the two value projections)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)   # (R,1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5) Apply gates → gated tensors `left` and `right`
    # ------------------------------------------------------------------
    left  = left_proj * left_gate        # (R, H)
    right = right_proj * right_gate      # (R, H)

    # reshape back to (B, N, N, H) – still contiguous because H is the last dim
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Batched matrix multiplication (N³ operation)
    #    out_perm[b, h, i, j] = Σₖ left[b, i, k, h] · right[b, j, k, h]
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2)   # (B, H, N, N) – view only
    right_perm = right.permute(0, 3, 1, 2)  # (B, H, N, N) – view only
    # (B·H, N, N) @ (B·H, N, N)ᵀ   → (B·H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # fp16

    # ------------------------------------------------------------------
    # 7) Fused LN + out‑gate + final linear (single Triton kernel)
    # ------------------------------------------------------------------
    # stride information for the non‑contiguous source tensor
    stride_b = out_perm.stride(0)   # batch stride
    stride_h = out_perm.stride(1)   # hidden‑dim stride (N·N)
    stride_i = out_perm.stride(2)   # i‑dim stride
    stride_j = out_perm.stride(3)   # j‑dim stride (1)

    # buffer for the final FP16 result (R, dim)
    out_fp16 = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # kernel launch configuration (tuned for H100)
    # ------------------------------------------------------------------
    BLOCK_M    = 256                     # rows per program
    BLOCK_H    = 32                      # hidden‑dim tile
    BLOCK_D    = 64                      # output‑dim tile (≤ dim)

    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_ln_gate_out_kernel[grid](
        out_perm,                                             # src (B,H,N,N)
        out_gate,                                             # gate (R,H)
        out_fp16,                                             # dst (R, dim)
        weights["to_out_norm.weight"],                       # LN γ (fp32)
        weights["to_out_norm.bias"],                         # LN β (fp32)
        weights["to_out.weight"].to(torch.float16),          # final linear weight (dim, H)
        total_rows,
        hidden_dim,
        dim,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 8) Reshape to the requested output layout and cast to FP32
    # ------------------------------------------------------------------
    final = out_fp16.view(B, N, N, dim).to(torch.float32)
    return final
