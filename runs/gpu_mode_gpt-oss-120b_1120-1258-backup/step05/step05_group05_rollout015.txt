Pseudo code for kernel:

```
@triton.jit
def fused_trimul_kernel(
    x_ptr, mask_ptr,
    wLP_ptr, wLG_ptr, wRP_ptr, wRG_ptr, wOG_ptr,
    sum_wLP_ptr, sum_wLG_ptr, sum_wRP_ptr, sum_wRG_ptr, sum_wOG_ptr,
    biasLP_ptr, biasLG_ptr, biasRP_ptr, biasRG_ptr, biasOG_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows, dim, hidden_dim,
    eps: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_O: tl.constexpr,
):
    pid0 = tl.program_id(0)  # row tile
    pid1 = tl.program_id(1)  # output dim tile
    row_start = pid0 * BLOCK_M
    col_start = pid1 * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_O)  # [BLOCK_O]
    mask_m = offs_m < total_rows
    mask_n = offs_n < hidden_dim

    # accumulators
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    raw_LP = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
    raw_LG = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
    raw_RP = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
    raw_RG = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
    raw_OG = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_k = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_k = offs_k < dim

        # load X chunk [BLOCK_M, BLOCK_D] half
        x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_k[None, :])
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # accumulate stats
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

        # compute raw dot for each weight
        # each weight pointer -> slice
        # left_proj
        wLP_ptrs = wLP_ptr + (offs_n[:, None] * dim + offs_k[None, :])
        wLP = tl.load(wLP_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        raw_LP += tl.dot(x, wLP)   # (BLOCK_M, BLOCK_O)

        # left_gate
        wLG_ptrs = wLG_ptr + (offs_n[:, None] * dim + offs_k[None, :])
        wLG = tl.load(wLG_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        raw_LG += tl.dot(x, wLG)

        # right_proj
        wRP_ptrs = wRP_ptr + (offs_n[:, None] * dim + offs_k[None, :])
        wRP = tl.load(wRP_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        raw_RP += tl.dot(x, wRP)

        # right_gate
        wRG_ptrs = wRG_ptr + (offs_n[:, None] * dim + offs_k[None, :])
        wRG = tl.load(wRG_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        raw_RG += tl.dot(x, wRG)

        # out_gate
        wOG_ptrs = wOG_ptr + (offs_n[:, None] * dim + offs_k[None, :])
        wOG = tl.load(wOG_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        raw_OG += tl.dot(x, wOG)

    # compute mean, var, rstd
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # load sum_w and bias for each category
    sum_w_LP = tl.load(sum_wLP_ptr + offs_n, mask=mask_n, other=0.0)
    sum_w_LG = tl.load(sum_wLG_ptr + offs_n, mask=mask_n, other=0.0)
    sum_w_RP = tl.load(sum_wRP_ptr + offs_n, mask=mask_n, other=0.0)
    sum_w_RG = tl.load(sum_wRG_ptr + offs_n, mask=mask_n, other=0.0)
    sum_w_OG = tl.load(sum_wOG_ptr + offs_n, mask=mask_n, other=0.0)

    bias_LP = tl.load(biasLP_ptr + offs_n, mask=mask_n, other=0.0)
    bias_LG = tl.load(biasLG_ptr + offs_n, mask=mask_n, other=0.0)
    bias_RP = tl.load(biasRP_ptr + offs_n, mask=mask_n, other=0.0)
    bias_RG = tl.load(biasRG_ptr + offs_n, mask=mask_n, other=0.0)
    bias_OG = tl.load(biasOG_ptr + offs_n, mask=mask_n, other=0.0)

    # compute normalized outputs
    # shape broadcast: mean [BLOCK_M, 1], rstd [BLOCK_M, 1]
    norm_LP = (raw_LP - mean[:, None] * sum_w_LP[None, :]) * rstd[:, None] + bias_LP[None, :]
    norm_LG = (raw_LG - mean[:, None] * sum_w_LG[None, :]) * rstd[:, None] + bias_LG[None, :]
    norm_RP = (raw_RP - mean[:, None] * sum_w_RP[None, :]) * rstd[:, None] + bias_RP[None, :]
    norm_RG = (raw_RG - mean[:, None] * sum_w_RG[None, :]) * rstd[:, None] + bias_RG[None, :]
    norm_OG = (raw_OG - mean[:, None] * sum_w_OG[None, :]) * rstd[:, None] + bias_OG[None, :]

    # compute sigmoid for gates
    gate_L = 1.0 / (1.0 + tl.exp(-norm_LG))
    gate_R = 1.0 / (1.0 + tl.exp(-norm_RG))
    gate_O = 1.0 / (1.0 + tl.exp(-norm_OG))

    # compute final left,right,out_gate
    left = norm_LP * gate_L
    right = norm_RP * gate_R
    out_gate = gate_O

    # apply mask if provided
    if mask_ptr is not None:
        mask = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)
        left = left * mask[:, None]
        right = right * mask[:, None]

    # Store to output pointers
    left_ptrs = left_ptr + (offs_m[:, None] * hidden_dim + offs_n[None, :])
    right_ptrs = right_ptr + (offs_m[:, None] * hidden_dim + offs_n[None, :])
    out_gate_ptrs = out_gate_ptr + (offs_m[:, None] * hidden_dim + offs_n[None, :])

    tl.store(left_ptrs, left.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
    tl.store(right_ptrs, right.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
    tl.store(out_gate_ptrs, out_gate.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(...) {
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None,:], mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)
    }

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass, compute normalized and gating
    for d in range(...) {
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        norm = (x - mean[:, None]) * rstd[:, None] * w[None,:] + b[None,:]   # fp32

        out_fp32 = norm * g

        tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None,:], out_fp32.to(tl.float16), mask=mask_m[:, None] & mask_d[None,:])
    }

def layernorm_gate_kernel(
    x_ptr, gate_ptr, out_ptr,
    weight_ptr, bias_ptr,
    total_rows, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # first pass: compute mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: normalize, apply gamma & beta, multiply gate, store
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        norm = ((x - mean[:, None]) * rstd[:, None]) * w[None, :] + b[None, :]

                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        out = (norm * g).to(tl.float16)

        tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out,

"""
custom_kernel
==============

Fast forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm (mixed fp16/fp32, Triton‑fused kernels)
------------------------------------------------
1️⃣  Layer‑norm + five linear projections (left/right + three gates) are
    fused into a single Triton kernel.  The kernel:
      – computes per‑row mean / variance (fp32)
      – accumulates raw GEMM results for the five weight matrices
      – finishes the LayerNorm analytically using the pre‑computed
        weight‑wise sums (γ·W) and biases (β·W) and applies the sigmoid
        gates, yielding the gated tensors `left`, `right` and the
        output‑gate `out_gate`.
    The mask (if present) is applied in‑kernel, avoiding extra passes.

2️⃣  The triangular multiplicative update is a batched GEMM
    `out = left @ rightᵀ` (torch.bmm, half‑precision, Tensor‑Cores).

3️⃣  A second Triton kernel fuses the hidden‑dim LayerNorm,
    the output‑gate multiplication and the write‑back of the gated
    tensor `Z = LN(out) * out_gate`.

4️⃣  Final linear projection `to_out` is performed with a single
    Torch linear (fp16 weight) and the result is cast back to fp32.

All heavy reductions (mean / variance) are performed in fp32;
the bulk of the arithmetic stays in fp16 to exploit Tensor‑Cores on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 : fused LayerNorm + five linear projections + gating + mask
# ----------------------------------------------------------------------
@triton.jit
def fused_trimul_kernel(
    # inputs
    x_ptr,                 # half   [total_rows, dim]
    mask_ptr,              # half   [total_rows]  (1.0 if no mask)
    # weight matrices (γ·W)    -> half
    wLP_ptr, wLG_ptr, wRP_ptr, wRG_ptr, wOG_ptr,
    # sums Σ_i γ·W_i for each output row (float32)
    sum_wLP_ptr, sum_wLG_ptr, sum_wRP_ptr, sum_wRG_ptr, sum_wOG_ptr,
    # biases Σ_i β·W_i (float32)
    biasLP_ptr, biasLG_ptr, biasRP_ptr, biasRG_ptr, biasOG_ptr,
    # outputs
    left_ptr,              # half   [total_rows, hidden_dim]
    right_ptr,             # half   [total_rows, hidden_dim]
    out_gate_ptr,          # half   [total_rows, hidden_dim]
    # scalars
    total_rows: tl.int32,
    dim:        tl.int32,
    hidden_dim: tl.int32,
    eps:        tl.constexpr,
    BLOCK_M:    tl.constexpr,   # rows per program (set to 1)
    BLOCK_D:    tl.constexpr,   # dim block size
    BLOCK_O:    tl.constexpr,   # hidden_dim block size
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_O)            # [BLOCK_O]
    mask_m = offs_m < total_rows
    mask_n = offs_n < hidden_dim

    # ------------------------------------------------------------------
    # 1. Reduce: mean & variance of the input row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2. Accumulate raw GEMM results for the five projections
    # ------------------------------------------------------------------
    raw_LP = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
    raw_LG = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
    raw_RP = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
    raw_RG = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)
    raw_OG = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_k = d * BLOCK_D + tl.arange(0, BLOCK_D)            # [BLOCK_D]
        mask_k = offs_k < dim

        # load X block [BLOCK_M, BLOCK_D] as fp32
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # statistics
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

        # each weight matrix slice is [BLOCK_O, BLOCK_D] half → cast to fp32
        wLP = tl.load(wLP_ptr + offs_n[:, None] * dim + offs_k[None, :],
                      mask=mask_n[:, None] & mask_k[None, :],
                      other=0.0).to(tl.float32)
        wLG = tl.load(wLG_ptr + offs_n[:, None] * dim + offs_k[None, :],
                      mask=mask_n[:, None] & mask_k[None, :],
                      other=0.0).to(tl.float32)
        wRP = tl.load(wRP_ptr + offs_n[:, None] * dim + offs_k[None, :],
                      mask=mask_n[:, None] & mask_k[None, :],
                      other=0.0).to(tl.float32)
        wRG = tl.load(wRG_ptr + offs_n[:, None] * dim + offs_k[None, :],
                      mask=mask_n[:, None] & mask_k[None, :],
                      other=0.0).to(tl.float32)
        wOG = tl.load(wOG_ptr + offs_n[:, None] * dim + offs_k[None, :],
                      mask=mask_n[:, None] & mask_k[None, :],
                      other=0.0).to(tl.float32)

        # matmul: (M,D)·(N,D)ᵀ → (M,N)
        raw_LP += tl.dot(x, wLP)
        raw_LG += tl.dot(x, wLG)
        raw_RP += tl.dot(x, wRP)
        raw_RG += tl.dot(x, wRG)
        raw_OG += tl.dot(x, wOG)

    # ------------------------------------------------------------------
    # 3. Complete LayerNorm analytically (γ·W already baked in)
    # ------------------------------------------------------------------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # load Σ_i γ·W_i  and   Σ_i β·W_i for the current output block
    sum_wLP = tl.load(sum_wLP_ptr + offs_n, mask=mask_n, other=0.0)
    sum_wLG = tl.load(sum_wLG_ptr + offs_n, mask=mask_n, other=0.0)
    sum_wRP = tl.load(sum_wRP_ptr + offs_n, mask=mask_n, other=0.0)
    sum_wRG = tl.load(sum_wRG_ptr + offs_n, mask=mask_n, other=0.0)
    sum_wOG = tl.load(sum_wOG_ptr + offs_n, mask=mask_n, other=0.0)

    biasLP = tl.load(biasLP_ptr + offs_n, mask=mask_n, other=0.0)
    biasLG = tl.load(biasLG_ptr + offs_n, mask=mask_n, other=0.0)
    biasRP = tl.load(biasRP_ptr + offs_n, mask=mask_n, other=0.0)
    biasRG = tl.load(biasRG_ptr + offs_n, mask=mask_n, other=0.0)
    biasOG = tl.load(biasOG_ptr + offs_n, mask=mask_n, other=0.0)

    # normalised outputs (fp32)
    norm_LP = (raw_LP - mean[:, None] * sum_wLP[None, :]) * rstd[:, None] + biasLP[None, :]
    norm_LG = (raw_LG - mean[:, None] * sum_wLG[None, :]) * rstd[:, None] + biasLG[None, :]
    norm_RP = (raw_RP - mean[:, None] * sum_wRP[None, :]) * rstd[:, None] + biasRP[None, :]
    norm_RG = (raw_RG - mean[:, None] * sum_wRG[None, :]) * rstd[:, None] + biasRG[None, :]
    norm_OG = (raw_OG - mean[:, None] * sum_wOG[None, :]) * rstd[:, None] + biasOG[None, :]

    # sigmoid gates (≈ 1/(1+exp(-x)))
    gate_L = 1.0 / (1.0 + tl.exp(-norm_LG))
    gate_R = 1.0 / (1.0 + tl.exp(-norm_RG))
    gate_O = 1.0 / (1.0 + tl.exp(-norm_OG))

    # gated left/right tensors
    left  = norm_LP * gate_L
    right = norm_RP * gate_R
    out_gate = gate_O                     # already sigmoid‑ed

    # apply mask (mask is 1.0 for “nomask” case)
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0).to(tl.float32)
    left  = left  * mask_val[:, None]
    right = right * mask_val[:, None]

    # store results as half
    left_ptrs  = left_ptr  + offs_m[:, None] * hidden_dim + offs_n[None, :]
    right_ptrs = right_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :]
    out_gate_ptrs = out_gate_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :]

    tl.store(left_ptrs,  left.to(tl.float16),  mask=mask_m[:, None] & mask_n[None, :])
    tl.store(right_ptrs, right.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
    tl.store(out_gate_ptrs, out_gate.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Kernel 2 : hidden‑dim LayerNorm + out‑gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    # inputs
    x_ptr,                     # half  [total_rows, hidden_dim]
    gate_ptr,                  # half  [total_rows, hidden_dim]
    # outputs
    y_ptr,                     # half  [total_rows, hidden_dim] = LN(x) * gate
    # LayerNorm parameters (to_out_norm)
    ln_weight_ptr,             # float [hidden_dim]
    ln_bias_ptr,               # float [hidden_dim]
    total_rows: tl.int32,
    hidden_dim: tl.int32,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1) mean / variance of the row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        xv = tl.load(x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                     mask=mask_m[:, None] & mask_d[None, :],
                     other=0.0).to(tl.float32)

        sum_val += tl.sum(xv, axis=1)
        sum_sq  += tl.sum(xv * xv, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2) normalize, apply (γ,β), multiply by gate, write out
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        xv = tl.load(x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                     mask=mask_m[:, None] & mask_d[None, :],
                     other=0.0).to(tl.float32)

        w = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)

        # (x – μ) / σ  * γ + β
        norm = ((xv - mean[:, None]) * rstd[:, None]) * w[None, :] + b[None, :]

        g = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        out = (norm * g).to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                 out,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, C] (float32)
            - mask  : torch.Tensor [B, N, N] (bool/float) or None
            - weights : dict of model parameters (float32)
            - config  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H (always 128 in the tests)
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 0) Prepare the input as half‑precision, flatten to [total_rows, dim]
    # ------------------------------------------------------------------
    x_fp16 = input_tensor.view(total_rows, dim).contiguous().half()

    # ------------------------------------------------------------------
    # 1) Pre‑scale the five projection weight matrices with LayerNorm γ
    #    and pre‑compute Σ_i β·W_i and Σ_i γ·W_i for each output row.
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]  # [dim] float32
    norm_bias   = weights["norm.bias"]    # [dim] float32

    def prep(weight):
        # weight: [hidden_dim, dim] float32
        w_scaled = (weight * norm_weight).half()                     # γ·W   (fp16)
        sum_w = w_scaled.to(torch.float32).sum(dim=1)                # Σ_i γ·W_i  (fp32)
        bias   = (weight * norm_bias).sum(dim=1)                     # Σ_i β·W_i  (fp32)
        return w_scaled, sum_w, bias

    wLP, sum_wLP, biasLP = prep(weights["left_proj.weight"])
    wLG, sum_wLG, biasLG = prep(weights["left_gate.weight"])
    wRP, sum_wRP, biasRP = prep(weights["right_proj.weight"])
    wRG, sum_wRG, biasRG = prep(weights["right_gate.weight"])
    wOG, sum_wOG, biasOG = prep(weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 2) Allocate output buffers for left, right and out_gate
    # ------------------------------------------------------------------
    left_fp16      = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    right_fp16     = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    out_gate_fp16 = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 3) Mask handling – a tensor of ones if the model is in “nomask” mode
    # ------------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones(total_rows, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.reshape(total_rows).to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Launch the fused Triton kernel (row‑wise + hidden‑dim blocks)
    # ------------------------------------------------------------------
    BLOCK_M = 1          # one row per program (good occupancy on H100)
    BLOCK_D = 64         # dim‑chunk
    BLOCK_O = 32         # hidden‑dim chunk (384/32 = 12 max, fine)

    grid = (triton.cdiv(total_rows, BLOCK_M), triton.cdiv(hidden_dim, BLOCK_O))
    fused_trimul_kernel[grid](
        x_fp16,
        mask_fp16,
        wLP, wLG, wRP, wRG, wOG,
        sum_wLP, sum_wLG, sum_wRP, sum_wRG, sum_wOG,
        biasLP, biasLG, biasRP, biasRG, biasOG,
        left_fp16,
        right_fp16,
        out_gate_fp16,
        total_rows,
        dim,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )

    # ------------------------------------------------------------------
    # 5) Reshape to [B, N, N, H] and perform the triangular update
    # ------------------------------------------------------------------
    left  = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate_fp16.view(B, N, N, hidden_dim)

    # batched GEMM: left @ rightᵀ   (half‑precision, Tensor‑Core)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)                # [BH, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Second LayerNorm + out_gate multiplication (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # [total_rows, H]
    gated_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_fp16,
        gated_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7) Final linear projection back to C dimensions
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half().contiguous()   # [C, H]
    final_fp16 = F.linear(gated_fp16, to_out_w_fp16)               # (half)

    # ------------------------------------------------------------------
    # 8) Cast back to FP32 and reshape to the original 4‑D layout
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
