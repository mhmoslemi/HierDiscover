import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Input LN using torch.nn.functional.layer_norm (fast)
# Output fused Kernel
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,            # FP16 pointer: hidden tensor [R, H]
    gate_ptr,         # FP16 pointer: out_gate sigmoid [R, H]
    norm_w_ptr,       # FP32 weight gamma [H]
    norm_b_ptr,       # FP32 bias beta [H]
    out_w_ptr,        # FP16 weight matrix [dim, H]  (to_out weight)
    out_ptr,          # FP32 output [R, dim]
    R: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_r = tl.program_id(0)   # row block
    pid_d = tl.program_id(1)   # output channel block

    # row range
    row_start = pid_r * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # load hidden rows
    offs_h = tl.arange(0, hidden)
    mask_h = offs_h < hidden

    # Load input hidden vectors and gates
    x = tl.load(x_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)   # (BLOCK_R, hidden)
    g = tl.load(gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)   # (BLOCK_R, hidden)

    # Compute per-row mean and variance (FP32)
    sum_val = tl.sum(x, axis=1)
    sum_sq = tl.sum(x * x, axis=1)
    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Apply LayerNorm scale and bias
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)  # FP32
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)

    normed = (x - mean[:, None]) * rstd[:, None]    # (BLOCK_R, hidden)
    normed = normed * w_norm[None, :] + b_norm[None, :]  # (BLOCK_R, hidden)

    # Gating
    gated = normed * g

    # Output channel block
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # Load weight slice (dim, hidden) -> transposed
    # weight_T shape (hidden, BLOCK_D)
    w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                  mask=mask_h[:, None] & mask_d[None, :],
                  other=0.0).to(tl.float32)   # (hidden, BLOCK_D)

    # Compute GEMM
    out_block = tl.dot(gated, w_T)    # (BLOCK_R, BLOCK_D)

    # Store
    tl.store(out_ptr + offs_r[:, None] * dim + offs_d[None, :],
             out_block,
             mask=mask_r[:, None] & mask_d[None, :])

def custom_kernel(data):
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    # 1. Input LayerNorm (torch.nn.functional.layer_norm) to float32
    # weight and bias are float32
    x_norm = F.layer_norm(inp,
                          normalized_shape=(dim,),
                          weight=weights["norm.weight"],
                          bias=weights["norm.bias"],
                          eps=eps)   # shape [B,N,N,dim] float32

    # Cast to half for subsequent linear
    x_norm_h = x_norm.to(torch.half)

    # 2. Fused linear projection for left/right & gates (5 * hidden)
    concat_w = torch.cat([weights["left_proj.weight"],
                          weights["right_proj.weight"],
                          weights["left_gate.weight"],
                          weights["right_gate.weight"],
                          weights["out_gate.weight"]], dim=0).half().contiguous()

    # Linear does not have bias, shape (R, 5*hidden)
    R = B * N * N
    proj_all = F.linear(x_norm_h, concat_w)    # (R, 5*hidden) half

    # split
    left_proj_fp, right_proj_fp, left_gate_fp, right_gate_fp, out_gate_fp = torch.split(proj_all, hidden, dim=1)

    # 3. Optional mask
    if not nomask:
        # mask shape [B,N,N] -> flatten to [R,1] broadcast to hidden
        mask_flat = mask.reshape(R, 1).to(torch.float16)
        left_proj_fp = left_proj_fp * mask_flat
        right_proj_fp = right_proj_fp * mask_flat

    # 4. Sigmoid gates
    left_gate = torch.sigmoid(left_gate_fp)
    right_gate = torch.sigmoid(right_gate_fp)
    out_gate = torch.sigmoid(out_gate_fp)  # (R, hidden)

    left = left_proj_fp * left_gate        # (R, hidden)
    right = right_proj_fp * right_gate     # (R, hidden)

    # 5. Triangular multiplicative update
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # Transform to (B*hidden, N, N) for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # FP16 result (B*H, N, N)

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden) FP16

    # 6. Fused output LayerNorm + out_gate gating + final linear projection
    out_flat = out.view(R, hidden).contiguous()   # (R, hidden) half
    final_fp32 = torch.empty(R, dim, dtype=torch.float32, device=device)

    BLOCK_R = 128   # rows per block
    BLOCK_D = 64    # output channel block size (adjust if dim > 64)
    grid = (triton.cdiv(R, BLOCK_R), triton.cdiv(dim, BLOCK_D))
    fused_norm_gate_linear_kernel[grid](
        out_flat,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        weights["to_out.weight"].half().contiguous(),
        final_fp32,
        R,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
    )

    # 7. Reshape final output
    final = final_fp32.view(B, N, N, dim)
    return final

"""
Highly‑optimized forward pass for the **outgoing** TriMul operation (AlphaFold‑3).

Algorithm
---------
1️⃣  Input LayerNorm – done with `torch.nn.functional.layer_norm` (fast cuDNN
    implementation).  The normalized tensor is cast to FP16 for the
    subsequent cheap GEMM.

2️⃣  Fuse the five linear projections (left/right + 3 gates) into a single
    FP16 GEMM (`torch.nn.functional.linear`).  This eliminates four extra
    matrix‑multiplications.

3️⃣  (Optional) binary mask applied only to the left/right projections.

4️⃣  Sigmoid on the three gate tensors and element‑wise gating.

5️⃣  Triangular multiplicative update:
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
    Realised as a batched FP16 matrix‑multiply (`torch.bmm`) over the
    flattened batch dimension `B*H`.

6️⃣  **Fused** LayerNorm + out‑gate + final linear projection.
    A custom Triton kernel reads the hidden tensor (`H = hidden_dim`),
    computes per‑row mean/variance (FP32), applies the LayerNorm parameters,
    multiplies by the sigmoid‑gated `out_gate`, and immediately projects to
    `dim` channels with the `to_out` weight (half‑precision weight,
    FP32 accumulation).  The kernel writes the result directly in FP32,
    eliminating two separate memory‑bound passes.

All heavy‑compute steps stay on‑device; the final result is a float‑32
tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate + final linear projection.
#   Input   : x   – half,   [total_rows, hidden]
#   Input   : gate – half,   [total_rows, hidden] (already sigmoid‑ed)
#   Output  : y   – float,  [total_rows, dim]
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,          # const   half*  [R, H]
    gate_ptr,       # const   half*  [R, H]
    norm_w_ptr,     # const   float* [H]   (γ of to_out_norm)
    norm_b_ptr,     # const   float* [H]   (β of to_out_norm)
    out_w_ptr,      # const   half*  [dim, H] (to_out weight)
    out_ptr,        #         float* [R, dim] (final output)
    R: tl.int32,    # total rows = B * N * N
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program (e.g. 128)
    BLOCK_D: tl.constexpr,   # output‑channel chunk (e.g. 64)
):
    pid_r = tl.program_id(0)   # row block
    pid_d = tl.program_id(1)   # output‑channel block

    # -----------------------------------------------------------------
    # 1. Load a block of hidden vectors and gates (FP32 for arithmetic)
    # -----------------------------------------------------------------
    row_start = pid_r * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    offs_h = tl.arange(0, hidden)
    mask_h = offs_h < hidden

    x = tl.load(x_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)          # (BLOCK_R, hidden)
    g = tl.load(gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)          # (BLOCK_R, hidden)

    # -----------------------------------------------------------------
    # 2. Row‑wise LayerNorm (FP32 reduction → FP32 output)
    # -----------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)                     # (BLOCK_R)
    sum_sq  = tl.sum(x * x, axis=1)                 # (BLOCK_R)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Load γ/β for to_out_norm
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)

    # Normalise, apply γ/β and the out‑gate
    normed = (x - mean[:, None]) * rstd[:, None]                # (BLOCK_R, hidden)
    normed = normed * w_norm[None, :] + b_norm[None, :]         # (BLOCK_R, hidden)
    gated  = normed * g                                          # (BLOCK_R, hidden)

    # -----------------------------------------------------------------
    # 3. Linear projection to `dim` (FP16 weight, FP32 accumulation)
    # -----------------------------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # Load weight slice (dim, hidden) → transpose to (hidden, BLOCK_D)
    w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                  mask=mask_h[:, None] & mask_d[None, :],
                  other=0.0).to(tl.float32)                     # (hidden, BLOCK_D)

    out_block = tl.dot(gated, w_T)                               # (BLOCK_R, BLOCK_D)

    # Store result (FP32)
    tl.store(out_ptr + offs_r[:, None] * dim + offs_d[None, :],
             out_block,
             mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim": C,
                          "hidden_dim": H,
                          "nomask": bool,
                          "eps": float (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim    = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N   # flattened batch over (i,j) pairs

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (fast cuDNN) → FP16
    # --------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x_norm = F.layer_norm(inp,
                          normalized_shape=(dim,),
                          weight=weights["norm.weight"],
                          bias=weights["norm.bias"],
                          eps=eps)                      # [B,N,N,dim] float32
    x_norm_h = x_norm.to(torch.half)                       # cast to FP16

    # --------------------------------------------------------------
    # 2️⃣ One‑shot linear projection for left/right & three gates
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                 # (5*H, dim)

    # Perform the fused GEMM: (R, dim) × (5*H, dim)^T → (R, 5*H)
    proj_all = F.linear(x_norm_h, concat_w)                # (R, 5*hidden) half

    # Split into individual tensors
    left_proj_fp, right_proj_fp, left_gate_fp, right_gate_fp, out_gate_fp = torch.split(
        proj_all, hidden, dim=1
    )                                                       # each (R, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Optional masking (broadcasted on hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp = left_proj_fp * mask_flat
        right_proj_fp = right_proj_fp * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates & element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp)                # (R, hidden)
    right_gate = torch.sigmoid(right_gate_fp)              # (R, hidden)
    out_gate = torch.sigmoid(out_gate_fp)                  # (R, hidden) – used later

    left = left_proj_fp * left_gate                         # (R, hidden)
    right = right_proj_fp * right_gate                      # (R, hidden)

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched matmul)
    # --------------------------------------------------------------
    left = left.view(B, N, N, hidden)                       # (B,N,N,H)
    right = right.view(B, N, N, hidden)                     # (B,N,N,H)

    # Reshape for batched GEMM: (B*H, N, N) where each slice multiplies
    # left[i,k] with right[j,k]^T
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)               # (B*H, N, N) fp16

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 6️⃣ Fused LayerNorm + out‑gate + final linear projection
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, hidden) fp16
    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    BLOCK_R = 128                       # rows per program (tuned for H100)
    BLOCK_D = 64                        # output‑channel chunk
    grid = (triton.cdiv(total_rows, BLOCK_R),
            triton.cdiv(dim, BLOCK_D))

    fused_norm_gate_linear_kernel[grid](
        out_flat,                              # x_ptr (hidden)
        out_gate,                              # gate_ptr (already sigmoid)
        weights["to_out_norm.weight"],         # γ
        weights["to_out_norm.bias"],           # β
        weights["to_out.weight"].half().contiguous(),  # to_out weight (dim, hidden)
        final_fp32,                            # output buffer
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 7️⃣ Reshape to final tensor shape
    # --------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)               # [B, N, N, dim] float32
    return final
