@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          BATCH, M, N, K,
                          stride_Ab, stride_Am, stride_Ak,
                          stride_Bb, stride_Bk, stride_Bn,
                          stride_Cb, stride_Cm, stride_Cn,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_rem = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid_rem // tl.cdiv(N, BLOCK_N)
    pid_n = pid_rem % tl.cdiv(N, BLOCK_N)

    # coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # offsets for each block in k loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for A and B_T
        a_ptrs = A_ptr + batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B_ptr + batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store
    c_ptrs = C_ptr + batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul outgoing kernel (AlphaFold3) implementation.

The operator performs:
1. LayerNorm over channel dimension (C) using a fast Triton kernel.
2. Five linear projections (left, right, left_gate, right_gate, out_gate) are fused
   into a single GEMM (half precision).
3. Optional mask multiplication.
4. Sigmoid gating and element‑wise multiplication.
5. Batched matrix multiplication `out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]`
   performed by a custom Triton batched GEMM kernel.
6. LayerNorm over the hidden dimension (H) using the same Triton kernel.
7. Gating by `out_gate` and final projection back to the original channel
   dimension.
All heavy reduction operations are done in Triton with FP32 accumulation,
while dense linear algebra uses half‑precision Tensor‑Core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton row‑wise LayerNorm (FP16 output, FP32 reduction)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const  void*   [total_rows, dim]
    y_ptr,          # *mut    half*   [total_rows, dim]
    weight_ptr,     # *const  float*  [dim]   (γ)
    bias_ptr,       # *const  float*  [dim]   (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # accumulate sum and sum of squares
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over dim
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized output
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = (norm * w[None, :] + b[None, :]).to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out,
                 mask=mask_m[:, None] & mask_d[None, :])

# --------------------------------------------------------------
# Triton batched GEMM kernel for left @ rightᵀ
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,   # A: [batch, M, K]
    stride_Bb, stride_Bk, stride_Bn,   # B: [batch, K, N] (original B is [batch, N, K])
    stride_Cb, stride_Cm, stride_Cn,   # C: [batch, M, N]
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles
        a_ptrs = A_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B_ptr + batch_id * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_K]
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)      # [BLOCK_K, BLOCK_N]

        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping from weight name to torch.Tensor (float32)
            - config_dict: {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # ----------------------------------------------------------
    # 1️⃣ First LayerNorm (custom Triton kernel, fp16 output)
    # ----------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # select block sizes (tuned for H100)
    BLOCK_M = 1024          # rows per program (max 1024 threads)
    BLOCK_D = 64            # column chunk
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ----------------------------------------------------------
    # 2️⃣ Fuse five linear projections into a single GEMM
    # ----------------------------------------------------------
    # concatenate weight matrices (each [hidden_dim, dim])
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).half().contiguous()  # shape [(5*H), C]

    proj_all = F.linear(x_norm_fp16, concat_w)  # (total_rows, 5*H) half

    # split
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each [total_rows, H]

    # ----------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ----------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ----------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # ----------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # reshape to [B, N, N, H]
    left = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ----------------------------------------------------------

    # ----------------------------------------------------------

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)   # [BH, N, N] half
    right_mat = right_perm.view(BH, N, N) # [BH, N, N] half

    # allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides (in elements)
    stride_Ab = left_mat.stride(0)
    stride_Am = left_mat.stride(1)
    stride_Ak = left_mat.stride(2)
    stride_Bb = right_mat.stride(0)
    stride_Bk = right_mat.stride(1)
    stride_Bn = right_mat.stride(2)
    stride_Cb = out_mat.stride(0)
    stride_Cm = out_mat.stride(1)
    stride_Cn = out_mat.stride(2)

    # launch Triton batched GEMM
    GRID_B = BH
    GRID_M = triton.cdiv(N, 128)   # BLOCK_M = 128
    GRID_N = triton.cdiv(N, 128)   # BLOCK_N = 128
    batched_gemm_kernel[(GRID_B, GRID_M, GRID_N)](
        left_mat,
        right_mat,
        out_mat,
        N,   # M = N (rows)
        N,   # N = N (cols)
        N,   # K = N (summation dim)
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (custom kernel)
    # ----------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)

    # ----------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # ----------------------------------------------------------
    gated = out_norm * out_gate   # [B,N,N,H] half
    gated_flat = gated.view(total_rows, hidden_dim)
    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # (total_rows, C) half

    # ----------------------------------------------------------
    # 8️⃣ Cast back to FP32
    # ----------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

"""
TriMul outgoing kernel (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension C (FP32 accumulation,
   FP16 output). Implemented in Triton.
2️⃣ Fuse the five linear projections (left, right, three gates) into one GEMM
   by concatenating their weight matrices.
3️⃣ Optional pair‑wise mask applied before gating.
4️⃣ Sigmoid gates + element‑wise gating (still FP16).
5️⃣ Batched matrix multiplication   out_h[i, j] = Σ_k left_h[i, k]·right_h[j, k]
   implemented as a custom Triton GEMM (A @ Bᵀ) on the hidden dimension H.
6️⃣ Row‑wise LayerNorm over H (same Triton kernel as in step 1).
7️⃣ Multiply by the output gate and a final linear projection back to C.
8️⃣ Cast the result to FP32 and return.

All heavy reductions are done with a Triton kernel, the dense linear algebra
uses half‑precision Tensor‑Core kernels for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton row‑wise LayerNorm (FP16 output, FP32 reduction)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const void*   [total_rows, dim]   (float32 or float16)
    y_ptr,          # *mut   half*   [total_rows, dim]   (output)
    weight_ptr,     # *const float*  [dim]               (γ)
    bias_ptr,       # *const float*  [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program (≤1024)
    BLOCK_D: tl.constexpr,   # columns processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- mean / variance (FP32) -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [row, col]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- write normalized output (FP16) ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]          # FP32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = (norm * w[None, :] + b[None, :]).to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out,
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) for the triangular update
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,       # A: [batch, M, K]
    stride_Bb, stride_Bk, stride_Bn,       # Bᵀ: [batch, K, N] (row‑major view of Bᵀ)
    stride_Cb, stride_Cm, stride_Cn,       # C: [batch, M, N]
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile:   [BLOCK_M, BLOCK_K]
        a_ptrs = A_ptr + batch_id * stride_Ab \
                 + offs_m[:, None] * stride_Am \
                 + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Bᵀ tile:  [BLOCK_K, BLOCK_N]   (Bᵀ = Bᵀ)
        b_ptrs = B_ptr + batch_id * stride_Bb \
                 + offs_k[:, None] * stride_Bk \
                 + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)       # sum over K

    c_ptrs = C_ptr + batch_id * stride_Cb \
             + offs_m[:, None] * stride_Cm \
             + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : mapping from name to torch.Tensor (float32)
            - config_dict   : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ----------------------------------------------------------
    # unpack arguments
    # ----------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # ----------------------------------------------------------
    # 1️⃣ First LayerNorm (custom Triton kernel, FP16 output)
    # ----------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # block size tuned for H100 (max 1024 threads per program)
    BLOCK_M = 1024
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ----------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right + three gates)
    # ----------------------------------------------------------
    # each weight has shape (hidden_dim, dim)
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).half().contiguous()            # [(5*H), C]

    # fused GEMM: (total_rows, C) @ (C, 5*H).T → (total_rows, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)               # half

    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden_dim, dim=1
        )   # each [total_rows, H]

    # ----------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ----------------------------------------------------------
    if (not nomask) and mask_tensor is not None:
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ----------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # ----------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # reshape back to 4‑D layout
    left = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ----------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ----------------------------------------------------------
    # permute to [B, H, N, N] and merge batch/head dimensions
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)   # [BH, N, N]  half
    right_mat = right_perm.view(BH, N, N)  # [BH, N, N]  half

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides in element counts
    stride_Ab = left_mat.stride(0)   # batch stride
    stride_Am = left_mat.stride(1)   # row stride (i)
    stride_Ak = left_mat.stride(2)   # col stride (k)

    stride_Bb = right_mat.stride(0)   # batch stride
    # transpose B on‑the‑fly: row‑stride = original col‑stride, col‑stride = original row‑stride
    stride_Bk = right_mat.stride(2)   # after transpose, stride over K (original k)
    stride_Bn = right_mat.stride(1)   # after transpose, stride over N (original i)

    stride_Cb = out_mat.stride(0)
    stride_Cm = out_mat.stride(1)
    stride_Cn = out_mat.stride(2)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,      # M
        N,      # N
        N,      # K
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to the original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ----------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H
    # ----------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)

    # ----------------------------------------------------------
    # 7️⃣ Final gating + projection back to C
    # ----------------------------------------------------------
    gated = out_norm * out_gate                     # [B,N,N,H] half
    gated_flat = gated.view(total_rows, hidden_dim)
    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)  # (total_rows, C) half

    # ----------------------------------------------------------
    # 8️⃣ Cast back to FP32
    # ----------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
