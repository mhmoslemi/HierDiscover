"""
Custom Triton‑accelerated implementation of the **outgoing** TriMul
operator (AlphaFold‑3). The forward pass consists of:

1️⃣ Input LayerNorm (FP32”‑>FP16) using PyTorch’s highly‑optimised
   `torch.nn.functional.layer_norm`.
2️⃣ A *single* fused linear projection that simultaneously computes the
   five linear transforms required (left/right projections, their gates
   and the output gate).  All four matrices are concatenated and a
   single `torch.nn.functional.linear` (FP16) is executed, which reduces
   memory traffic to the normalised tensor.
3️⃣ Optional binary mask applied to the two projections before gating.
4️⃣ Batched matrix‑multiply `out[b,i,j,d] = Σ_k left[b,i,k,d]·right[b,j,k,d]`
   implemented with `torch.bmm` on half‑precision tensors – this
   leverages Tensor‑Cores on H100.
5️⃣ Fused second LayerNorm + output‑gate implemented in Triton
   (`fused_layernorm_gate_kernel`).  The kernel computes per‑row mean/
   variance in FP32, applies the learned scale/bias and finally multiplies
   with the pre‑computed `out_gate`.
6️⃣ Final linear projection back to the original `dim` (FP16 GEMM) and a
   cast to `float32`.

The implementation respects the “no‑mask’’ configuration,
uses mixed‑precision (FP16 for the heavy compute), and keeps all large
tensor allocations contiguous to maximise memory‑bandwidth utilisation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # const half*    [total_rows, hidden]
    y_ptr,               #   mut half*    [total_rows, hidden]
    weight_ptr,          # const float*  [hidden]   (γ)
    bias_ptr,            # const float*  [hidden]   (β)
    gate_ptr,            # const half*   [total_rows, hidden]  (out_gate, already sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean / variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalise, apply γ/β and multiply by gate ----------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                     # FP32

        out = out * g

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool) – ignored
                         if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C,
                          "hidden_dim":H,
                          "nomask":bool,
                          "eps":float (optional)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ---------------------------- unpack ----------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]                # C
    hidden = cfg["hidden_dim"]      # H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # R = B·N·N

    # --------------------------- 1. LayerNorm ---------------------------
    # Flatten to (R, C) and run the highly‑optimised PyTorch LayerNorm.
    inp_flat = inp.view(total_rows, dim).contiguous()
    #   (FP32) → (FP16)  – we keep the half‑precision tensor for the
    #   subsequent GEMMs.
    x_norm = F.layer_norm(
        inp_flat.float(),
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()                                   # (R, C) half

    # ------------------- 2. Fused linear + gates (single matmul) -------
    # Concatenate the five weight matrices (left_proj, left_gate,
    # right_proj, right_gate, out_gate).  All are stored as FP16.
    left_proj_w = weights["left_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    #   Shape after concat: (5*H, C)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).contiguous()

    # Single linear projection (R, 5*H)
    raw = F.linear(x_norm, combined_w)                 # (R, 5*H) half
    # Split into the logical parts
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        raw, hidden, dim=1
    )                                                   # each (R, H)

    # ---------------------------- mask --------------------------------
    if not nomask:
        # mask shape (B, N, N) → (R, 1) half
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # --------------------- 3. Apply gates ---------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # (R, H) half
    right_gate = torch.sigmoid(right_gate_raw) # (R, H) half
    out_gate = torch.sigmoid(out_gate_raw)     # (R, H) half, used later

    left = left_proj_raw * left_gate           # (R, H) half
    right = right_proj_raw * right_gate        # (R, H) half

    # ------------------- 4. Triangular multiplicative update ---------
    # Reshape to (B, N, N, H)
    left = left.view(B, N, N, hidden)          # (B, N, N, H) half
    right = right.view(B, N, N, hidden)        # (B, N, N, H) half

    # Prepare tensors for batched GEMM:
    #   (B*H, N, N)  where left_mat[b*H + h] = left[b, :, :, h]
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N) half
    # right needs the k‑dimension on the second axis to achieve the
    # “i‑k * j‑k” contraction.
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N) half

    # Batched matrix multiply – Tensor‑Core accelerated.
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N) half

    # Return to the `[B, N, N, H]` layout.
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) half

    # ------------------- 5. Second LayerNorm + out‑gate -------------
    # Flatten for the Triton kernel.
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half

    # Allocate output buffer.
    out_norm_gate = torch.empty_like(out_flat)            # (R, H) half

    # Triton launch configuration – we use a fairly large BLOCK_M to
    # reduce the number of program IDs.
    BLOCK_M = 256
    BLOCK_D = 64 if hidden <= 256 else 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid’ed, half‑precision
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------- 6. Final linear projection -----------------
    to_out_w = weights["to_out.weight"].half()          # (C, H) half
    final_fp16 = F.linear(out_norm_gate, to_out_w)      # (R, C) half

    # Cast back to FP32 and restore original 4‑D shape.
    final = final_fp16.view(B, N, N, dim).float()
    return final

):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣ Compute row‑wise mean / variance in FP32
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2️⃣ Normalise, apply γ/β and multiply by the pre‑computed gate
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                     # FP32

        out = out * g

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32 or bool);
                         ignored if config["nomask"] is True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim": C,
                          "hidden_dim": H,
                          "nomask": bool,
                          "eps": float (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – the TriMul output.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                     # C
    hidden = cfg["hidden_dim"]              # H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                  # R = B·N·N

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()  # (R, C) float32
    # PyTorch LayerNorm is heavily optimised (uses cuDNN/flash‑LN)
    x_norm = F.layer_norm(
        inp_flat.float(),
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()  # (R, C) half

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear + gating (single FP16 GEMM)
    # -----------------------------------------------------------------
    # Cast all five weight matrices to FP16 once.
    left_proj_w = weights["left_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    # Concatenate: order matches the splits below.
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).contiguous()                                 # (5*H, C)

    # Single GEMM: (R, C) @ (C, 5*H)ᵀ → (R, 5*H)
    raw = F.linear(x_norm, combined_w)             # (R, 5*H) half

    # Split into logical parts.
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        raw, hidden, dim=1
    )                                              # each (R, H) half

    # -----------------------------------------------------------------
    # 3️⃣ Optional binary mask (applied before gating)
    # -----------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)  # (R,1)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # -----------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)    # (R, H) half
    right_gate = torch.sigmoid(right_gate_raw)   # (R, H) half
    out_gate   = torch.sigmoid(out_gate_raw)     # (R, H) half – used later

    left  = left_proj_raw * left_gate            # (R, H) half
    right = right_proj_raw * right_gate          # (R, H) half

    # -----------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # -----------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left  = left.view(B, N, N, hidden)           # (B, N, N, H) half
    right = right.view(B, N, N, hidden)          # (B, N, N, H) half

    # Prepare for batched matrix multiplication:
    #   left_mat[b*H + h]  = left[b, :, :, h]
    #   right_mat[b*H + h] = right[b, :, :, h]ᵀ   (note the transposition)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    # Half‑precision batched GEMM → Tensor‑Core accelerated.
    out_mat = torch.bmm(left_mat, right_mat)    # (B*H, N, N) half

    # Back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) half

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # -----------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()  # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)           # (R, H) half

    # Triton launch configuration – a relatively large BLOCK_M reduces
    # the number of program IDs and thus launch overhead.
    BLOCK_M = 256
    BLOCK_D = 64 if hidden <= 256 else 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid‑ed, half‑precision
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()   # (C, H) half
    final_fp16 = F.linear(out_norm_gate, to_out_w)   # (R, C) half

    # Cast back to FP32 and restore the original 4‑D shape.
    final = final_fp16.view(B, N, N, dim).float()
    return final
