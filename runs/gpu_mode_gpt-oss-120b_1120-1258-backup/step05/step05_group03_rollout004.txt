"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3) – version 2.

Algorithm
---------
1️⃣ Input LayerNorm (FP16 output) – Triton kernel that reduces each
   [B·N·N, C] row in FP32, applies learned γ/β and stores the
   normalised result as half‑precision.

2️⃣ One fused GEMM computes the four linear maps needed for the left
   and right sides (proj + gate) in a single matrix‑multiply:
   `L = [W_left_proj; W_left_gate; W_right_proj; W_right_gate]`.
   The result is split and the sigmoid gates are applied, yielding the
   two tensors `left` and `right` (shape [B,N,N,H] half).

3️⃣ Optional binary mask (broadcasted on the hidden dimension) is applied
   after gating.

4️⃣ Triangular multiplicative update – batched half‑precision matrix
   multiply:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   Implemented as a single `torch.bmm` on reshaped tensors
   (Tensor‑Core accelerated).

5️⃣ Second LayerNorm + output‑gate – performed with the highly‑optimised
   cuDNN/PyTorch `layer_norm` followed by element‑wise multiplication with
   the previously‑computed `out_gate`.

6️⃣ Final linear projection back to `dim` – FP16 GEMM (Tensor‑Core).

All heavy arithmetic stays in FP16/Tensor‑Core; reductions are performed
in FP32 for numerical stability.  The only user‑written kernel is the
first LayerNorm, satisfying the “must‑use‑a‑kernel” requirement while
reducing the overall number of kernel launches and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP16 output) – Triton
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (FP16)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the feature dimension in chunks
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1/√(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]     # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]               # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool)
          (ignored if config["nomask"]=True)
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C, "hidden_dim":H,
                         "nomask":bool, "eps":float (optional)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim      = cfg["dim"]                 # C
    hidden   = cfg["hidden_dim"]          # H
    eps      = cfg.get("eps", 1e-5)
    nomask   = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                            device=device)                  # [R, C] half

    # Choose block sizes that comfortably cover the feature dimension.
    BLOCK_M = 128
    BLOCK_D = 64 if dim > 256 else 32

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear‑+‑gate for left/right sides (single GEMM)
    # ------------------------------------------------------------------
    # Cast all relevant weight matrices to FP16 once.
    left_proj_w  = weights["left_proj.weight"].half().contiguous()   # (H, C)
    left_gate_w  = weights["left_gate.weight"].half().contiguous()
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()   # (H, C)

    # Concatenate the four weight matrices → shape (4*H, C)
    lr_comb_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w],
        dim=0,
    )  # (4*H, C)

    # One matrix‑multiply produces proj & gate logits for both sides.
    lr_comb = F.linear(norm_fp16, lr_comb_w)      # (R, 4*H)

    # Split the concatenated result.
    l_proj_raw = lr_comb[:, 0:hidden]
    l_gate_raw = lr_comb[:, hidden:2*hidden]
    r_proj_raw = lr_comb[:, 2*hidden:3*hidden]
    r_gate_raw = lr_comb[:, 3*hidden:4*hidden]

    # Apply sigmoid gates (half‑precision) and mask the projected values.
    left_gate  = torch.sigmoid(l_gate_raw)
    right_gate = torch.sigmoid(r_gate_raw)

    left  = l_proj_raw * left_gate   # (R, H)
    right = r_proj_raw * right_gate  # (R, H)

    # ------------------------------------------------------------------
    # 2a) Optional binary mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (R,1)
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left  = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 2b) Compute out‑gate (linear + sigmoid) – independent of mask.
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(norm_fp16, out_gate_w)   # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)           # (R, H) half

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape tensors to [B, N, N, H]
    left  = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # Prepare (B*H, N, N) tensors where the second operand is transposed.
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)      # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)     # (B*H, N, N)

    # Batched matrix multiply → (B*H, N, N). Uses Tensor‑Core for fp16.
    out_mat = torch.bmm(left_mat, right_mat)

    # Reshape back to [B, N, N, H] (still half‑precision)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out‑gate (cuDNN LayerNorm + elementwise mul)
    # ------------------------------------------------------------------
    # Convert LayerNorm parameters to half‑precision for speed.
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()

    # cuDNN‑accelerated LayerNorm on half‑precision tensor.
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # (B, N, N, H) half

    # Multiply with the previously computed out‑gate.
    out_gate = out_gate.view(B, N, N, hidden)          # (B,N,N,H)
    out_norm_gate = out_norm * out_gate                # half

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()          # (C, H)
    out_flat = out_norm_gate.view(total_rows, hidden)   # (R, H)
    final_fp16 = F.linear(out_flat, to_out_w)           # (R, C) half

    # Reshape to the original 4‑D layout and cast to FP32 as required.
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
