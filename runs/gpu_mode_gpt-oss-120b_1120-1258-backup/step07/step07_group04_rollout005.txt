"""
TriMul (outgoing) – high‑performance forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (shape [B·N·N, dim]).
   Implemented with a custom Triton kernel that:
   * computes per‑row mean & variance in FP32,
   * writes the normalized result in FP16.

2️⃣ Fuse the four linear layers (left/right projections + left/right gates)
   into a single FP16 GEMM:
        X_norm (R×dim)  ×  W_fusedᵀ (dim×4·H)  →  (R×4·H)
   The result is split into *left_raw*, *right_raw*, *left_gate_raw*,
   *right_gate_raw*.

3️⃣ Compute the output‑gate with a second FP16 GEMM.

4️⃣ (Optional) Apply the binary mask to the left/right projections.
   Apply the sigmoid non‑linearity in‑place to all gate tensors.

5️⃣ Triangular multiplicative update (the O(N³) core):
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   Implemented as a batched FP16 matrix multiplication
   (B·H, N, N) @ (B·H, N, N)ᵀ using `torch.bmm`.

6️⃣ Second LayerNorm over the hidden dimension (H) → FP16
   (same Triton kernel as step 1).

7️⃣ Multiply by the output‑gate, apply the final FP16 linear projection,
   and cast the result back to FP32.

All heavy linear algebra runs in FP16 to exploit Tensor‑Core throughput,
while the two LayerNorms are executed by a dedicated Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)  gamma
    bias_ptr,         # const   * [dim]               (float32)  beta
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """Row‑wise LayerNorm:
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the feature dimension
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # broadcast
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (LayerNorm parameters)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # beta

        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim        = config["dim"]                 # input channel dimension
    hidden_dim = config["hidden_dim"]          # hidden dimension (H)
    nomask     = config.get("nomask", True)   # True → no mask
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    # Flatten to [R, dim] – keep contiguous layout
    x_flat = x.view(total_rows, dim).contiguous()          # fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Triton launch parameters – tuned for H100
    BLOCK_M = 256          # rows per program (adjustable)
    BLOCK_D = 64           # columns per tile (must divide dim reasonably)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # gamma (float32)
        weights["norm.bias"],        # beta  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates (4·H outputs)
    # --------------------------------------------------------------
    # Stack all four weight matrices (shape [4·H, dim]) and cast to FP16.
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                      # [4·H, dim]

    # Linear: X_norm (R×dim) @ W_fusedᵀ → (R×4·H)   (FP16 GEMM)
    fused = F.linear(x_norm_fp16, w_fused)   # fp16

    # Split into the four required tensors
    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H]  fp16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and combine
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # Broadcast mask to [R, 1] and cast to FP16
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid (FP16) for lower memory pressure
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_raw  * left_gate   # [R, H]   fp16
    right = right_raw * right_gate  # [R, H]   fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # From [R, H] → [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Bring hidden dimension into the batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & head dimensions → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched GEMM (FP16 Tensor‑Core): (B·H, N, N) @ (B·H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → FP16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                       # element‑wise fp16

    # Final projection: hidden → dim
    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)      # [dim, H]
    )   # [R, dim]  fp16

    # Cast back to FP32 (the model expects float32)
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
