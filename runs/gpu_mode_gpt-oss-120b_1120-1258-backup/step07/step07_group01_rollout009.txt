"""
custom_kernel
==============

Fast Triton‑accelerated implementation of the **outgoing TriMul** operator
used in AlphaFold‑3.  The overall algorithm follows the reference
PyTorch module but moves the two LayerNorms to custom row‑wise Triton
kernels and fuses the many linear projections/gates into a single
matrix‑multiply.  All heavy tensor work is done in FP16 (or TF32 on
H100) while the final result is returned in FP32.

Key ideas
---------
* **Row‑wise LayerNorm** – a small Triton kernel that computes the mean,
  variance and writes the normalized tensor (FP16) in two passes.
* **Fused projection/gate** – the four linear projections (`left_proj`,
  `right_proj`, `left_gate`, `right_gate`) and the `out_gate` linear are
  concatenated into one weight matrix, turning five `nn.Linear`s into a
  single `torch.nn.functional.linear`.
* **Masked gating** – the optional mask is applied only to the left/right
  tensors (as in the reference implementation) after the projection.
* **Batched N³ update** – the costly contraction
  `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
  is expressed as a batched GEMM:
  `(B·H, N, N) @ (B·H, N, N)ᵀ`.  cuBLAS/Tensor‑Core kernels on H100
  execute this extremely fast.
* **Second LayerNorm + output gate + final projection** – the hidden‑
  dimension tensor is normalised, gated and projected back to `dim`.
  Only the two LayerNorms are Triton kernels; the remaining ops stay in
  PyTorch (highly‑optimised BLAS).

The implementation keeps the memory footprint low (all intermediate
tensors stay in FP16) and reduces the number of PyTorch calls to the
minimum possible while still satisfying the required functionality.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]        (float32) – gamma
    bias_ptr,       # const   * [dim]        (float32) – beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (single pass over dim)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Normalise, apply weight & bias, write FP16 output
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]           # still fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 1024,
                      block_d: int = 64) -> torch.Tensor:
    """
    Row‑wise LayerNorm for a 2‑D tensor ``x`` of shape [rows, dim] (float32)
    → FP16 tensor of the same shape.

    Parameters
    ----------
    x : torch.Tensor
        Input tensor (float32) of shape [rows, dim].
    weight, bias : torch.Tensor
        Layer‑norm weight (γ) and bias (β), shape [dim] (float32).
    eps : float
        Numerical epsilon.
    block_m, block_d : int
        Tiling sizes for the Triton kernel.

    Returns
    -------
    torch.Tensor
        Normalised tensor (float16) of shape [rows, dim].
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y

# ----------------------------------------------------------------------

    """

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict of all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain ``"dim"``, ``"hidden_dim"``,

    torch.Tensor

    # --------------------------------------------------------------
    # 0️⃣ Unpack ----------------------------------------------------
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    B, N, _, _ = input_tensor.shape
    R = B * N * N   # total number of (b,i,k) rows for the first LayerNorm

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] FP32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)

        block_m=1024,
        block_d=64,
    )                                                        # FP16, [R, dim]

    # 2️⃣ Fused projection + gates (5·H outputs)
    # ------------------------------------------------------------------

    # Order: left_proj, right_proj, left_gate, right_gate, out_gate

            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                      # [5·H, dim]

    # Single GEMM: X_norm (R,dim) @ W_fused.T (dim,5·H) → (R,5·H)
    fused_out = F.linear(x_norm, w_fused)                    # FP16

    # Split the fused result
    left_proj_raw  = fused_out[:, 0 * H : 1 * H]             # [R, H]
    right_proj_raw = fused_out[:, 1 * H : 2 * H]             # [R, H]
    left_gate_raw  = fused_out[:, 2 * H : 3 * H]             # [R, H]
    right_gate_raw = fused_out[:, 3 * H : 4 * H]             # [R, H]
    out_gate_raw   = fused_out[:, 4 * H : 5 * H]             # [R, H]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid for gates (still FP16)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply element‑wise gates
    left  = left_proj_raw * left_gate      # [R, H] FP16
    right = right_proj_raw * right_gate    # [R, H] FP16

    # 4️⃣ N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------

    out_gate = out_gate.view(B, N, N, H) # [B, N, N, H]

    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
        block_m=1024,
        block_d=64,
    )                                                     # FP16, [R, H]

    out_norm = out_norm.view(B, N, N, H)                  # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final projection back to `dim`
    # ------------------------------------------------------------------
    # Element‑wise gating (still FP16)
    out_norm = out_norm * out_gate                        # [B, N, N, H]

    # Final Linear: (R, H) @ (dim, H).T → (R, dim)  (FP16 → FP32)
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
