    norm_weight_ptr, norm_bias_ptr,
    total_rows: tl.int32, H: tl.int32, dim: tl.int32, eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean and variance across H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):

        mask_h = offs_h < H

                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Prepare accumulator for output
    acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

    # second pass: compute normalized, apply layernorm weight/bias, gate, matmul

def custom_kernel(data):
    """..."""
    x, mask, weights, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; eps = 1e-5
    device = x.device; B,N,_,_ = x.shape
    R = B*N*N

    # First rowwise layernorm
    x_flat = x.view(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 256; BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # gamma
        weights["norm.bias"],     # beta
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # fused projection + gates + out_gate (5 parts)
    weight_proj = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)

    proj_all = F.linear(x_norm, weight_proj)   # shape (R, 5*H) fp16

    H = hidden_dim
    left_proj_raw = proj_all[:, 0:H]
    right_proj_raw = proj_all[:, H:2*H]
    left_gate_raw = proj_all[:, 2*H:3*H]
    right_gate_raw = proj_all[:, 3*H:4*H]
    out_gate_raw = proj_all[:, 4*H:5*H]

    # Optional mask
    if not cfg.get("nomask", True):
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # Tri-multiplicative update as before
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B * H, N, N)       # (B*H, N, N)
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H) fp16

    # Fuse second LayerNorm + out_gate scaling + final linear via Triton kernel
    out_flat = out.view(R, H).contiguous()
    final_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Use fused kernel
    BLOCK_M2 = 128
    BLOCK_H = 32
    BLOCK_D = 128   # adjust if dim not multiple

    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_linear_gate_kernel[grid2](
        out_flat,
        out_gate.view(R, H).contiguous(),
        weights["to_out.weight"].to(torch.float16).contiguous(),
        final_half,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

    gate_ptr,            # const   *[total_rows, H]  (float16)   out_gate
    w_ptr,               # const   *[dim, H]        (float16)   weight

    norm_weight_ptr,     # const   *[H]            (float32)   gamma
    norm_bias_ptr,       # const   *[H]            (float32)   beta
    total_rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- compute mean & variance across H for each row ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = x_ptr + offs_m[:, None] * H + offs_h[None, :]

                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- Loop over output dim blocks and accumulate product ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # accumulator for this dim block
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

                             other=0.0).to(tl.float32)

            # apply layernorm weight and bias (γ,β)

                             mask=mask_h,
                             other=0.0).to(tl.float32)   # [BLOCK_H]
            b_norm = tl.load(norm_bias_ptr + offs_h,
                             mask=mask_h,
                             other=0.0).to(tl.float32)   # [BLOCK_H]
            norm = norm * w_norm[None, :] + b_norm[None, :]

            # load gate (already sigmoid-ed)
            gate_ptrs = gate_ptr + offs_m[:, None] * H + offs_h[None, :]

                                mask=mask_m[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # load weight slice for current dim block: shape [BLOCK_H, BLOCK_D]

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 -> FP16)
# ---------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const *[total_rows, dim] (float32)
    y_ptr,          # mutable *[total_rows, dim] (float16)
    weight_ptr,     # const *[dim] (float32)   γ
    bias_ptr,       # const *[dim] (float32)   β
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------- compute mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------- write normalized output (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]           # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    H: tl.int32,

    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

def custom_kernel(data):

    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    The implementation follows the reference PyTorch code but fuses several
    heavyweight operations to reduce kernel launches and memory traffic:
        * Row‑wise LayerNorm on the input (Triton)
        * A single fused Linear that produces left/right projections, their
          gates, and the output‑gate.
        * Tri‑multiplicative update computed with a batched GEMM (cuBLAS).
        * A fused Triton kernel that combines the second LayerNorm,
          the element‑wise out‑gate multiplication and the final Linear
          projection back to the original model dimension.

    All heavy matrix‑multiplications are performed in FP16; the two
    LayerNorms keep FP32 accumulators for numerical stability.
    The final tensor is returned in FP32 as required by the model.
    """
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                              # total rows for row‑wise ops

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()      # [R, dim]  float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 2️⃣ Fuse left/right projections, their gates and the output‑gate
    # -----------------------------------------------------------------
    # Concatenate the five weight matrices (4*H for proj/gates + 1*H for out_gate)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # shape [5*H, dim]

    proj_all = F.linear(x_norm, w_fused)       # [R, 5*H]  FP16

    H = hidden_dim
    left_proj_raw  = proj_all[:, 0:H]
    right_proj_raw = proj_all[:, H:2*H]
    left_gate_raw  = proj_all[:, 2*H:3*H]
    right_gate_raw = proj_all[:, 3*H:4*H]
    out_gate_raw   = proj_all[:, 4*H:5*H]

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask & sigmoid gates
    # -----------------------------------------------------------------
    if not cfg.get("nomask", True):
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate         # [R, H] FP16
    right = right_proj_raw * right_gate       # [R, H] FP16

    # -----------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # -----------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] FP16

    # -----------------------------------------------------------------
    # 5️⃣ Fused LayerNorm‑gate‑Linear for the hidden‑dim tensor
    # -----------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    final_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M2 = 128
    BLOCK_H = 32
    BLOCK_D = 128   # works for dim up to 384 (3 blocks)
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_linear_gate_kernel[grid2](
        out_flat,
        out_gate.view(R, H).contiguous(),
        weights["to_out.weight"].to(torch.float16).contiguous(),
        final_half,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Cast back to float32 and reshape
    # -----------------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
