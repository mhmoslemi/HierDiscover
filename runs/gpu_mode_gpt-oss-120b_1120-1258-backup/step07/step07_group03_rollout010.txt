"""
Highly‑optimized forward pass of the **outgoing TriMul** operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the input tensor [B,N,N,dim]  
    – custom Triton kernel (two‑pass: mean/var then normalization).

2️⃣  Fuse the four projection/gate Linear layers **and** the output‑gate Linear
    into a **single** half‑precision GEMM:
        [R, dim]  @  [5·H, dim]ᵀ  →  [R, 5·H]  (R = B·N·N,  H = hidden_dim)

3️⃣  Apply the (optional) pairwise mask, compute in‑place sigmoid gates and
    element‑wise gating:
        left  = left_proj  * sigmoid(left_gate)
        right = right_proj * sigmoid(right_gate)
        out_gate = sigmoid(out_gate)

4️⃣  Reshape left/right to [B, H, N, N] and perform the N³ multiplicative update
    as a **batched half‑precision GEMM**:
        out = (left_mat) @ (right_mat)ᵀ      # shape (B·H, N, N)

5️⃣  LayerNorm over the hidden dimension (FP32 → FP16) using the same Triton
    kernel.

6️⃣  Scale by the output‑gate and project back to the original dimension
    with a final half‑precision Linear layer.  The result is cast to FP32.

All heavy linear algebra (GEMMs + BMM) uses Tensor‑Core‑accelerated cuBLAS;
only the two LayerNorms are hand‑written in Triton.  This reduces kernel
launch overhead (the out‑gate Linear is now fused) and minimizes memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,        # const   * [total_rows, dim] (float32)
    y_ptr,        # mutable * [total_rows, dim] (float16)
    w_ptr,        # const   * [dim]            (float32) γ
    b_ptr,        # const   * [dim]            (float32) β
    total_rows,   # runtime int
    dim,          # runtime int
    eps,          # compile‑time float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # vector width (must divide dim nicely)
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ² + eps)) * γ + β
       – reduction is done in fp32, output in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1️⃣ Compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # Load tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32 for accumulation

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # ---------- 2️⃣ Write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out_fp32 = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor  shape [B, N, N]   (bool/float) – may be ignored
        - weights_dict : dict of torch.Tensors containing all Linear/LayerNorm params
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor   shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack & prepare
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]           # H
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                               # total rows for the first LN

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp32 → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] fp32
    x_norm_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Triton launch (tune block size for typical dims)
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections, gates and out‑gate into a single GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    # concat weights: [left_proj; right_proj; left_gate; right_gate; out_gate]    shape (5·H, dim)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)       # half‑precision for Tensor‑Core GEMM

    # GEMM: [R, dim] @ [5·H, dim]ᵀ  →  [R, 5·H]   (no bias)
    proj_gate = F.linear(x_norm_half, w_fused)          # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Slice the fused result
    # ------------------------------------------------------------------
    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]             # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]           # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]           # [R, H]
    out_gate_raw    = proj_gate[:, 4*H:5*H]           # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (applied before the sigmoid gates)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask Tensor: [B, N, N] → [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Sigmoid gates (in‑place) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Reshape for the N³ multiplicative update
    #      left/right: [B, N, N, H] → [B·H, N, N]
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]

    # bring hidden dimension to batch axis (contiguous layout)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # [B·H, N, N]

    # ------------------------------------------------------------------
    # 7️⃣ Batched matrix multiplication (half‑precision)
    #      out_mat[b·H] = left_mat[b·H] @ right_mat[b·H]ᵀ
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (fp32 → fp16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )   # fp16, same shape as out

    # ------------------------------------------------------------------
    # 9️⃣ Apply output‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # gating, fp16

    final_half = F.linear(
        out_norm.view(R, H),                     # [R, H]
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)   # cast back to fp32
    return final
