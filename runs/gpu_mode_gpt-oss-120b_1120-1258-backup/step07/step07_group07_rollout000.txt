"""
TriMul – outgoing version (AlphaFold‑3) – fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) on the input tensor.
2️⃣ Fuse the four linear projections (left/right + left/right gates) into a
   single mat‑mul by concatenating the weight matrices.
3️⃣ Optional mask is applied to the left/right projections.
4️⃣ Apply sigmoid gates (in‑place) and element‑wise multiplication.
5️⃣ Reshape to (B, H, N, N) and compute the cubic update with a single
   batched GEMM:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
   (equivalently torch.bmm on (B·H, N, N) matrices).
6️⃣ Fuse the second LayerNorm with the output‑gate scaling in a Triton kernel.
7️⃣ Final linear projection back to the original `dim` (fp16 → fp32).

Only two custom Triton kernels are launched (first LayerNorm, second
LayerNorm+gate).  All heavy arithmetic stays in FP16, giving a fast H100
implementation while producing the exact FP32 result required by the reference
PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [R, dim]   (float32)
    y_ptr,          # mutable * [R, dim]   (float16)
    w_ptr,          # const   * [dim]      (float32)   – γ
    b_ptr,          # const   * [dim]      (float32)   – β
    R,              # total number of rows (int)
    dim,            # feature dimension (int)
    eps,            # epsilon (float)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """Row‑wise LayerNorm storing the result in FP16."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # ------------------------------------------------------------------
    # ① Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                       # 1/√(var+eps)

    # ------------------------------------------------------------------
    # ② Normalise, apply γ/β and store as FP16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)       # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)       # β
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6️⃣ Row‑wise LayerNorm + output‑gate (both fp16 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [R, dim]   (float32)
    y_ptr,          # mutable * [R, dim]   (float16)
    w_ptr,          # const   * [dim]      (float32)   – γ
    b_ptr,          # const   * [dim]      (float32)   – β
    gate_ptr,       # const   * [R, dim]   (float16)   – element‑wise gate
    R,              # total rows (int)
    dim,            # feature dim (int)
    eps,            # epsilon (float)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """LayerNorm + gate multiplication, result stored in fp16."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # ---------- mean / var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalisation + gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalise
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # scale & bias
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        norm = norm * w[None, :] + b[None, :]                 # fp32

        # load gate (fp16 → fp32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                     # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣‑7️⃣ Full forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float, optional)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape                     # input shape = [B, N, N, dim]
    R = B * N * N                             # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (row‑wise)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()   # [R, dim]  fp32
    x_norm = torch.empty(R, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 2️⃣ fused projection + gate linear layers
    #    (left_proj, right_proj, left_gate, right_gate) → 4·H × dim weight
    # --------------------------------------------------------------
    # Concatenate weight matrices once (float16 for speed)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4H, dim]

    # One linear pass produces all four tensors
    proj_gate = F.linear(x_norm, w_fused)   # [R, 4H]  fp16

    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2*H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]

    # out_gate is separate
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ optional mask, sigmoid gates
    # --------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask: [B,N,N] → [R,1] broadcast over H
        mask_flat = mask.reshape(R, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # In‑place sigmoid for the gate tensors
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()    # [R, H] fp16

    # Apply gates
    left  = left_proj_raw * left_gate       # [R, H] fp16
    right = right_proj_raw * right_gate     # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ reshape to 4‑D tensors
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)           # [B, N, N, H] fp16
    right = right.view(B, N, N, H)          # [B, N, N, H] fp16
    out_gate = out_gate.view(B, N, N, H)    # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ cubic multiplicative update via batched GEMM
    #    out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Merge batch (B) and head (H) dimensions → (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B·H, N, N] fp16
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [B·H, N, N] fp16

    # torch.bmm uses Tensor‑Cores on H100, far faster than a custom Triton loop
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N] fp16

    # Back to the original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)           # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm + output‑gate (fused)
    # --------------------------------------------------------------
    out_flat = out.reshape(R, H).contiguous()                    # [R, H] fp16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.reshape(R, H),          # gate tensor (fp16)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )
    out_norm = out_norm.view(B, N, N, H)        # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 7️⃣ final linear projection back to `dim`
    # --------------------------------------------------------------
    final = F.linear(
        out_norm.reshape(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                 # [R, dim] fp16

    return final.view(B, N, N, dim).to(torch.float32)
