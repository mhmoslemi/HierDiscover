"""
custom_kernel
=============
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3).  The implementation

1️⃣  uses Torch’s native fused LayerNorm for the first normalisation
    (FP32 → FP16) – it is much faster than a hand‑written Triton kernel.

2️⃣  concatenates the four linear projections (left, right, left‑gate,
    right‑gate) **and** the output‑gate projection into a single
    half‑precision GEMM, cutting the number of matmuls from two to one.

3️⃣  applies the optional mask and the three sigmoid gates.

4️⃣  performs the batched matrix multiplication
    `out[i,j,k] = Σ_l left[i,l,k] * right[j,l,k]` via
    `torch.bmm` on half‑precision tensors (full Tensor‑Core utilisation).

5️⃣  fuses the final hidden‑dim LayerNorm and the multiplication by the
    output‑gate into a custom Triton kernel (layernorm_gate_kernel).

6️⃣  projects the hidden dimension back to the original `dim` with a final
    half‑precision linear layer and returns FP32 output.

Only the final LayerNorm + gate multiplication is written in Triton,
fulfilling the “must use a kernel” requirement while keeping the
overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel: fused hidden‑dim LayerNorm + element‑wise gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,                # input / output  [R, D] (FP16)
    weight_ptr, bias_ptr,        # LN parameters γ, β (FP32)
    gate_ptr,                    # gate tensor, same shape as x (FP16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Row range for this program
    # --------------------------------------------------------------
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute mean & variance in FP32
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # Normalise, apply γ, β and multiply by the gate
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load data
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # load gate (FP16 → FP32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # γ, β (FP32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        # (x‑μ)/σ * γ + β  then * gate
        out = ((vals - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]) * gate
        out_fp16 = out.to(tl.float16)

        # store result
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(
            y_ptrs,
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (bool/float)
        - weights_dict : mapping from parameter name to torch.Tensor
        - config_dict  : must contain ``dim`` (int) and ``hidden_dim`` (int);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use highly‑optimised torch implementation)
    # ------------------------------------------------------------------
    # torch.layer_norm works on FP32, we then cast to FP16 for the rest.
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)          # [R, dim] in half

    # ------------------------------------------------------------------
    # 2️⃣ Single fused linear projection for left/right + gates + out_gate
    # ------------------------------------------------------------------
    # Convert all projection weights to half once.
    w_left   = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16)  # [H, dim]
    w_lgate  = weights["left_gate.weight"].to(torch.float16)  # [H, dim]
    w_rgate  = weights["right_gate.weight"].to(torch.float16)  # [H, dim]
    w_outgate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]

    # Concatenate along the output dimension → [5*H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)

    # Linear (R, dim) @ (5H, dim).T → (R, 5H).  FP16 GEMM (Tensor cores)
    proj = F.linear(x_norm, w_concat)               # [R, 5H] half

    # Slice the five blocks
    left_proj_raw  = proj[:, 0:H]                     # [R, H]
    right_proj_raw = proj[:, H:2 * H]                # [R, H]
    left_gate_raw  = proj[:, 2 * H:3 * H]            # [R, H]
    right_gate_raw = proj[:, 3 * H:4 * H]            # [R, H]
    out_gate_raw   = proj[:, 4 * H:5 * H]            # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask application
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_half = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_half
        right_proj_raw = right_proj_raw * mask_half

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates (in‑place for speed)
    # ------------------------------------------------------------------
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Element‑wise gating
    left  = left_proj_raw * left_gate               # [R, H] half
    right = right_proj_raw * right_gate             # [R, H] half

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM implementing the einsum: ...i k d, ...j k d → ...i j d
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] then move H into the batch dimension.
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N] half
    right_mat = right.view(B * H, N, N)     # [B·H, N, N] half

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(total_rows, H)                                 # [R, H] half

    # ------------------------------------------------------------------
    # 6️⃣ Fused hidden‑dim LayerNorm + multiplication by the output‑gate
    # ------------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 64           # columns per inner loop (tuned for H ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        out_gate.view(total_rows, H),    # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] half

    # Cast to FP32 to match the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
