"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

Key ideas
---------
*   Row‑wise LayerNorm is performed by a custom Triton kernel (FP32 → FP16).
*   The four linear maps (left_proj, right_proj, left_gate, right_gate) are
    fused into a single GEMM, eliminating three extra matmul calls.
*   The cubic i‑j‑k update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is expressed as a batched matrix multiplication
        (B·H, N, N) @ (B·H, N, N)ᵀ
    which is dispatched to cuBLAS (torch.bmm).  This is the most
    compute‑intensive part and cuBLAS on H100 is already optimal.
*   A second row‑wise LayerNorm (over the hidden dimension H) is again
    executed by the same Triton kernel.
*   All heavy linear work is carried out in FP16 so Tensor‑Cores are
    fully utilised; the final result is cast back to FP32 to match the
    reference implementation.

The function ``custom_kernel`` adheres to the required signature:
    (input, mask, weights, config) → output.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 input → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [R, D] (float32)
    y_ptr,          # mutable * [R, D] (float16)
    w_ptr,          # const   * [D]   (float32)  – γ
    b_ptr,          # const   * [D]   (float32)  – β
    R,              # total rows (int32)
    D,              # dim per row (int32)
    eps,            # eps for stability (float32)
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """Row‑wise LayerNorm.
    Computes per‑row mean/variance in FP32 and writes the normalised result
    in FP16: y = ((x-μ)/√(σ²+eps))*γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # 1️⃣ compute sum and sum‑of‑squares over the dim axis
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < D
        # linearised address: row * D + col
        ptr = x_ptr + offs_r[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ write the normalised values (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        ptr = x_ptr + offs_r[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x-μ) * rstd  -->  normalized value
        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * D + offs_d[None, :]
        tl.store(out_ptr,
                 out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
        - weights_dict : model parameters (see keys in docstring)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (row‑wise) – FP32 → FP16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()            # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 256          # rows per program
    BLOCK_D = 128          # tile size for dim (covers dim up to 384)
    grid = (triton.cdiv(R, BLOCK_R),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (FP32)
        weights["norm.bias"],            # β (FP32)
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ fused projection + gate linear maps (4·H × dim)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices once (FP16)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    proj_gate = F.linear(x_norm, w_4h)       # [R, 4·H] FP16
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # out‑gate (separate linear map) – shape (H, dim)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # [R, H] FP16

    # ------------------------------------------------------------------
    # 3️⃣ optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] (float16)
        mask_fp16 = mask_tensor.reshape(R, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # in‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate          # [R, H] FP16
    right = right_proj_raw * right_gate        # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ reshape to (B, N, N, H) for the cubic update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H).contiguous()
    right = right.view(B, N, N, H).contiguous()
    out_gate = out_gate.view(B, N, N, H).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ cubic update via batched GEMM:
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # bring H into the batch dimension => (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)     # [B·H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)    # [B·H, N, N]

    # batched matmul (left @ rightᵀ) – cuBLAS uses Tensor‑Cores for FP16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))      # [B·H, N, N]

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ second LayerNorm over the hidden dimension H (row‑wise)
    # ------------------------------------------------------------------
    out_flat = out.reshape(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_R),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)          # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 7️⃣ apply out‑gate and final linear projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # FP16

    final = F.linear(
        gated.view(R, H),                         # [R, H]
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # [R, dim] FP16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
