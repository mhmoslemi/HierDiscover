"""
TriMul (outgoing) – fast forward pass
-------------------------------------

The implementation mirrors the reference PyTorch module but removes all
unnecessary memory passes:

1. **Row‑wise LayerNorm (fp32 → fp16)**
   * A custom Triton kernel (`layernorm_kernel`) computes the per‑row
     mean/variance in fp32, normalises the row and writes the result as
     fp16.

2. **Fused projection + gating**
   * The four linear projections (left, right, left‑gate, right‑gate) are
     concatenated into a single weight matrix of shape `[4·H, dim]`.
   * A single `torch.nn.functional.linear` (fp16) produces the four
     tensors in one GEMM.
   * Sigmoids are applied in‑place; an optional mask is broadcasted
     over the hidden dimension.

3. **Cubic multiplicative update**
   * For each head `h` we need
       `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   * This is exactly a batched matrix multiplication:
        `out[h] = left[h] @ right[h].T`.
   * By reshaping to `(B·H, N, N)` and using `torch.bmm` we obtain a
     highly‑optimised Tensor‑Core GEMM (FP16 × FP16 → FP32).

4. **Second LayerNorm + output‑gate fusion**
   * Another Triton kernel (`layernorm_gate_kernel`) fuses the second
     normalisation with the element‑wise multiplication by `out_gate`.

5. **Final linear projection**
   * A single FP16 linear layer (`to_out`) maps back from the hidden
     dimension to `dim`.  The result is cast to `float32` to match the
     reference implementation.

All heavy arithmetic stays in half‑precision; only the two
LayerNorms use fp32 for numerics.  The code works for every configuration
used in the tests (B∈{1,2}, N∈{128,…,1024}, dim∈{128,384},
hidden_dim = 128) and supports both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / √(σ²+eps)) * γ + β   (γ,β = weight,bias)
    Output is stored as float16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean / variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalised values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm + output‑gate fusion
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    gate_ptr,         # const   * [total_rows, dim]   (float16)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm + element‑wise gate multiplication.
    y = ( ((x-μ)/√(σ²+eps)) * γ + β ) * gate
    The gate operand lives in fp16, the rest of the computation is in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean / var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalisation + gating ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # original x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalise
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # weight / bias
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate (fp16 → fp32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                        # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

                         dtype=torch.float16, device=device)

    # choose block sizes that cover the widest dim (384) comfortably
    BLOCK_M = 1024                     # rows per program (tuneable)
    BLOCK_D = 128                      # cols per iteration (covers dim≤384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ   (float32)
        weights["norm.bias"],            # β   (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gates (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices: left, right, left_gate, right_gate
    # each has shape [H, dim]
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                  # [4*H, dim]

    # Linear projection on the normalised input (still fp16)
    proj_gate = F.linear(x_norm, w_4h)    # [R, 4*H] fp16

    # Slice the four tensors
    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2*H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]

    # out_gate is a separate linear projection (uses its own weight)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoids and gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B, N, N] → flatten to [R, 1]
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid for gate tensors (fp16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()                 # [R, H] fp16

    # Apply gates to the projected values
    left = left_proj_raw * left_gate                    # [R, H] fp16
    right = right_proj_raw * right_gate                 # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)            # [B, N, N, H] fp16
    right = right.view(B, N, N, H)          # [B, N, N, H] fp16
    out_gate = out_gate.view(B, N, N, H)    # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    → treat each (b, h) as an independent batch.
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B*H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # [B*H, N, N]

    # Batched matrix multiplication (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] fp16

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)          # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused)
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.reshape(total_rows, H), # gate (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out_norm.reshape(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                   # [R, dim] fp16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
