    BLOCK_K: tl.constexpr,

"""
Fast “outgoing” TriMul implementation (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 ×→ FP16) on the input tensor (B·N·N, C).
2️⃣ Fuse the four linear projections/gates into a single FP16 GEMM.
3️⃣ Apply the mask (if present) and the three sigmoid gates.
4️⃣ Compute the N³ multiplicative update
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched half‑precision GEMM (B·H independent matmuls) realised with a
   custom Triton kernel (much lower launch overhead than cuBLAS for many small
   matrices).
5️⃣ Row‑wise LayerNorm on the hidden dimension.
6️⃣ Element‑wise out‑gate, final FP16 linear projection and cast to FP32.

Only the two LayerNorms and the N³ update are written in Triton; all other
operations use fused PyTorch kernels (FP16 linear, sigmoid, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32 or float16)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – gamma
    bias_ptr,       # const   * [dim]      (float32) – beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]      # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # beta
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm (float32 → float16) using Triton.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel: batched half‑precision GEMM for the N³ update
# Computes C = A @ Bᵀ   where A,B have shape (batch, N, N) and dtype fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr,          # const * [batch, M, K]   (fp16)
    B_ptr,          # const * [batch, K, N]   (fp16) – already transposed
    C_ptr,          # mutable * [batch, M, N] (fp16)
    batch,          # int32   (runtime)
    M, N, K,        # int32   (runtime)
    stride_Ab, stride_Am, stride_Ak,   # strides for A (batch, M, K)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, K, N)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids: (batch, block_i, block_j)
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptr = A_ptr + pid_batch * stride_Ab \
                + offs_m[:, None] * stride_Am \
                + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile (already transposed): (BLOCK_K, BLOCK_N)
        b_ptr = B_ptr + pid_batch * stride_Bb \
                + offs_k[:, None] * stride_Bk \
                + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back result in fp16
    c_ptr = C_ptr + pid_batch * stride_Cb \
            + offs_m[:, None] * stride_Cm \
            + offs_n[None, :] * stride_Cn
    tl.store(c_ptr,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def _batched_matmul(A: torch.Tensor,
                    B_T: torch.Tensor,
                    block_m: int = 128,
                    block_n: int = 128,
                    block_k: int = 32) -> torch.Tensor:
    """
    Batched GEMM: C = A @ B_T   (all tensors fp16, shape [batch, N, N]).
    Returns C in fp16.
    """
    batch, M, K = A.shape
    _, K2, N = B_T.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty_like(A)   # same shape & dtype (fp16)

    grid = (
        batch,
        triton.cdiv(M, block_m),
        triton.cdiv(N, block_n),
    )
    _batched_matmul_kernel[grid](
        A,
        B_T,
        C,
        batch,
        M,
        N,
        K,
        A.stride(0), A.stride(1), A.stride(2),   # A strides
        B_T.stride(0), B_T.stride(1), B_T.stride(2),   # B_T strides
        C.stride(0), C.stride(1), C.stride(2),   # C strides
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=8,      # more warps give higher occupancy for N≥256
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if config["nomask"] is True
        - weights_dict : dict with Linear / LayerNorm parameters (float32)
        - config_dict  : must contain keys "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()       # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # gamma (float32)
        weights["norm.bias"],     # beta  (float32)
        eps=eps,
    )                                                            # fp16, [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (produces 4·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four weight matrices (each [H, dim]) and cast once
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [4·H, dim]

    # (R, dim) @ (dim, 4·H)ᵀ → (R, 4·H)   (FP16 GEMM)
    proj_gate = F.linear(x_norm, w_proj_gate)                   # fp16

    left_proj_raw  = proj_gate[:,          0:H]                 # [R, H]
    right_proj_raw = proj_gate[:,          H:2*H]               # [R, H]
    left_gate_raw  = proj_gate[:,    2*H:3*H]                   # [R, H]
    right_gate_raw = proj_gate[:,    3*H:4*H]                   # [R, H]

    # out‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),           # [H, dim]
    )                                                            # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoids + element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)  # [R,1]
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM via Triton)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]

    # bring hidden dim to the batch dimension: (B·H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    batch = B * H
    left_mat = left_bh.view(batch, N, N)                # [batch, N, N] fp16
    right_mat = right_bh.view(batch, N, N)              # [batch, N, N] fp16
    right_mat_T = right_mat.transpose(-2, -1).contiguous()  # transpose once

    out_mat = _batched_matmul(left_mat, right_mat_T)    # fp16, shape [batch, N, N]

    # back to the original layout
    out_bh = out_mat.view(B, H, N, N)                   # [B, H, N, N]
    out = out_bh.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (row‑wise over hidden dim) → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()     # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        eps=eps,
    )                                                   # fp16, [R, H]

    out_norm = out_norm.view(B, N, N, H)                # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Apply out‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                       # element‑wise, fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),      # [dim, H]
    )                                                    # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

This implementation keeps the memory footprint low, performs the heavy
N³ update with a custom batched GEMM kernel (significantly reducing launch
overhead compared to cuBLAS’s batched gemm), and uses Triton for the two
LayerNorms to avoid extra synchronization. The final tensor is returned in
`float32` as required.
