@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_hb = tl.program_id(2)   # batch*hidden dimension
    # compute batch b and hidden h
    b = pid_hb // H
    h = pid_hb % H

    # compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)   # i offset
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N) # j offset
    offs_k = tl.arange(0, BLOCK_K)

    # mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # compute pointer base offsets for this batch/h
    # compute base offset for left: b * stride_b_left + h * stride_h_left (since hidden is last)
    base_left = b * stride_b_left + h * stride_h_left
    # offsets for left: i dimension -> stride_i_left, k dimension -> stride_k_left
    a_ptrs = left_ptr + base_left + offs_m[:, None] * stride_i_left + offs_k[None, :] * stride_k_left

    # right ptr: need element right[b, j, k, h] where j corresponds to i dimension for column matrix.
    base_right = b * stride_b_right + h * stride_h_right
    # offsets: j dimension -> stride_i_right (since j akin to i)
    b_ptrs = right_ptr + base_right + offs_k[:, None] * stride_k_right + offs_n[None, :] * stride_i_right

    # out ptr: store out[b,i,j,h]
    base_out = b * stride_b_out + h * stride_h_out
    c_ptrs = out_ptr + base_out + offs_m[:, None] * stride_i_out + offs_n[None, :] * stride_j_out

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual offsets for k block offset
        cur_k = k * BLOCK_K

        # update a and b pointers for this block
        a = tl.load(a_ptrs + cur_k * stride_k_left, mask=mask_m[:, None] & mask_k[None,:], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs + cur_k * stride_k_right, mask=mask_k[:, None] & mask_n[None,:], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # apply mask if provided (mask tensor shape B,N,N). We could combine with left/projection earlier; maybe not needed.

    # store result
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel
=============
Highly‑optimized forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16) – custom Triton.
2️⃣  Four linear projections (left, right, left‑gate, right‑gate) are fused
    into a single half‑precision GEMM:  [R, dim] @ [4·H, dim]ᵀ → [R, 4·H].
3️⃣  A separate half‑precision GEMM produces the output‑gate.
4️⃣  Optional mask is applied, then inplace sigmoid on the gate tensors and
    element‑wise gating of the left/right projections.
5️⃣  **Batched “tri‑mul”** – a custom Triton kernel computes, for every
    (batch, hidden) pair, the product

           out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    i.e. a set of B·H independent (N×N) × (N×N)ᵀ matrix multiplies.
    The kernel works directly on the (B,N,N,H) layout produced by step 4,
    avoiding the expensive permute+contiguous copy used in the naïve
    implementation.
6️⃣  LayerNorm over the hidden dimension together with multiplication by the
    previously‑computed output‑gate – fused Triton kernel.
7️⃣  Final half‑precision linear projection back to `dim` (cuBLAS) and cast
    to FP32.

The implementation supports `dim ∈ {128,384}`, `hidden_dim = 128`,
batch size 1‑2 and sequence lengths up to 1024, with or without a mask.
All heavy‑compute parts are executed in FP16 with FP32 accumulation for
numerical stability.  The only PyTorch work left is the small dense GEMMs
for the projections, which are already heavily tuned in cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (FP32)
    y_ptr,               # mutable * [total_rows, dim]   (FP16)
    weight_ptr,          # const   * [dim]               (FP32)
    bias_ptr,            # const   * [dim]               (FP32)
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- compute mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized tensor ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                  # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Fused LayerNorm + out‑gate multiplication (hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,                # [total_rows, dim]  (FP16)
    weight_ptr, bias_ptr,        # LN γ,β  (FP32)
    gate_ptr,                    # gate tensor, same shape as x (FP16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize + gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # input value
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # gate value (FP16 → FP32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = ((x - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]) * g
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Batched “tri‑mul” kernel (B·H independent N×N × N×Nᵀ multiplies)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,               # const * [B, N, N, H]   (FP16)
    right_ptr,              # const * [B, N, N, H]   (FP16)
    out_ptr,                # mutable * [B, N, N, H] (FP16)
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    B, N, H,
    BLOCK_M: tl.constexpr,  # tile rows
    BLOCK_N: tl.constexpr,  # tile cols
    BLOCK_K: tl.constexpr,  # tile reduction
):
    pid_i = tl.program_id(0)               # tile row index
    pid_j = tl.program_id(1)               # tile col index
    pid_hb = tl.program_id(2)              # combined batch*hidden index

    # ---- resolve concrete batch and hidden ids ----
    b = pid_hb // H
    h = pid_hb % H

    # ---- tile offsets ----
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # off‑sets for the left matrix A (i, k)
    a_base = b * stride_b_left + h * stride_h_left
    a_ptrs = left_ptr + a_base \
            + offs_i[:, None] * stride_i_left \
            + tl.arange(0, BLOCK_K)[None, :] * stride_k_left

    # off‑sets for the right matrix Bᵀ (k, j) i.e. right[b, j, k, h]
    b_base = b * stride_b_right + h * stride_h_right
    b_ptrs = right_ptr + b_base \
            + tl.arange(0, BLOCK_K)[:, None] * stride_k_right \
            + offs_j[None, :] * stride_i_right

    # out pointer for C[b, i, j, h]
    c_base = b * stride_b_out + h * stride_h_out
    c_ptrs = out_ptr + c_base \
            + offs_i[:, None] * stride_i_out \
            + offs_j[None, :] * stride_j_out

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension ----
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k_blocks):
        k_offset = k_iter * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + k_offset
        mask_k = cur_k < N

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

        # advance pointers to next K‑block
        a_ptrs += BLOCK_K * stride_k_left
        b_ptrs += BLOCK_K * stride_k_right

    # ---- write the result ----
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# 4️⃣ Entry point (required by the grader)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, [B, N, N, dim], float32
        - mask_tensor  : torch.Tensor, [B, N, N] (bool/float) – optional
        - weights_dict : mapping from string names to torch.Tensor (model params)
        - config_dict  : must contain "dim" (int) and "hidden_dim" (int);
                         optional key "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N           # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()      # [R,dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (FP32)
        weights["norm.bias"],        # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into one GEMM
    # --------------------------------------------------------------
    # Convert projection / gate weights to FP16 once
    w_left   = weights["left_proj.weight"].to(torch.float16)    # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16) # [H, dim]
    w_lgate  = weights["left_gate.weight"].to(torch.float16) # [H, dim]
    w_rgate  = weights["right_gate.weight"].to(torch.float16) # [H, dim]

    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)   # [4·H, dim]

    # (R,dim) @ (4·H,dim)ᵀ → (R,4·H)   (FP16)
    proj_gate = F.linear(x_norm, w_concat)          # [R, 4·H] FP16

    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]             # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]           # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]           # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate projection
    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate = F.linear(x_norm, w_out_gate)                    # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Mask (if present) and sigmoid gates
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate.sigmoid_()

    # Element‑wise gating
    left = left_proj_raw * left_gate    # [R, H] FP16
    right = right_proj_raw * right_gate # [R, H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to (B, N, N, H) – this view is *non‑contiguous* but works
    # directly with the custom kernel.
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    out = torch.empty_like(left)   # allocate output tensor (FP16)

    # strides for the three tensors
    s_bl, s_il, s_kl, s_hl = left.stride()
    s_br, s_ir, s_kr, s_hr = right.stride()
    s_bo, s_io, s_jo, s_ho = out.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * H)                     # one program per (batch, hidden)

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        s_bl, s_il, s_kl, s_hl,
        s_br, s_ir, s_kr, s_hr,
        s_bo, s_io, s_jo, s_ho,
        B, N, H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ LayerNorm + output‑gate multiplication (hidden dim)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H)                     # [R, H] FP16
    out_gate_flat = out_gate.view(total_rows, H)           # [R, H] FP16
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        out_gate_flat,                  # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] FP16

    # reshape to original layout and cast to FP32 as the reference does
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
