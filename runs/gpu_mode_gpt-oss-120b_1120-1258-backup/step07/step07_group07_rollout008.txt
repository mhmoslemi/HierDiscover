@triton.jit
def apply_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, out_ptr, mask_ptr,
    total_rows, hidden_dim,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < hidden_dim

    # Load proj and gate
    proj = tl.load(proj_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0)
    gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0)
    if HAS_MASK:
        mask = tl.load(mask_ptr + offs_m[:, None],
                        mask=mask_m[:, None],
                        other=1.0)
        out = proj * gate * mask
    else:
        out = proj * gate
    tl.store(out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
             out,
             mask=mask_m[:, None] & mask_h[None, :])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    ...
    # Input LayerNorm using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    x_norm = x_norm.to(torch.float16)

    # Combined linear for projections and gates
    proj_weights = torch.cat([...], dim=0).to(torch.float16) # shape (4*hidden_dim, dim)
    proj = torch.nn.functional.linear(x_norm, proj_weights) # [B,N,N,4*hidden_dim]

    # Split
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw = torch.split(proj, hidden_dim, dim=-1)
    # out_gate as separate linear
    out_gate_raw = torch.nn.functional.linear(x_norm, weights["out_gate.weight"].to(torch.float16))

    # Sigmoid gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply mask and gate for left/right using triton kernel
    total_rows = B * N * N
    BLOCK_M = 256
    BLOCK_H = hidden_dim  # up to 128
    if not config.get("nomask", True):
        mask_flat = mask_tensor.reshape(total_rows).to(torch.float16)
        HAS_MASK = 1
    else:
        mask_flat = torch.empty(0, device=device, dtype=torch.float16)
        HAS_MASK = 0

    left = torch.empty_like(left_proj_raw)  # shape [R, H]
    right = torch.empty_like(right_proj_raw)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    apply_proj_gate_mask_kernel[grid](
      left_proj_raw,
      left_gate,
      left,
      mask_flat,
      total_rows,
      hidden_dim,
      BLOCK_M=BLOCK_M,
      BLOCK_H=hidden_dim,
      HAS_MASK=HAS_MASK,
      num_warps=4,
    )
    apply_proj_gate_mask_kernel[grid](
      right_proj_raw,
      right_gate,
      right,
      mask_flat,
      total_rows,
      hidden_dim,
      BLOCK_M=BLOCK_M,
      BLOCK_H=hidden_dim,
      HAS_MASK=HAS_MASK,
      num_warps=4,
    )

    # reshape to [B,N,N,H]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # shape (B*H,N,N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

"""
TriMul (outgoing) – highly‑fused forward pass

Algorithm
---------
1️⃣  Input LayerNorm – use the highly‑optimized torch.nn.functional.layer_norm
    (row‑wise over the last dimension).  Result is cast to float16 to
    feed the subsequent half‑precision linear layers.

2️⃣  Four linear projections (left/right + two gates) are merged into a single
    matrix‑multiply by concatenating the four weight matrices.  A second
    linear layer provides the output‑gate.  All linear ops run in FP16.

3️⃣  A tiny Triton kernel (`apply_proj_gate_mask_kernel`) fuses the element‑wise
    mask (when present) and the sigmoid‑gate multiplication for the left
    and right branches, reducing two memory passes to one.

4️⃣  The cubic update
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched matrix multiplication:
        (B·H, N, N) × (B·H, N, N)ᵀ  →  (B·H, N, N)
    which exploits Tensor‑Cores on H100.

5️⃣  A second Triton kernel (`layernorm_gate_kernel`) fuses the final
    LayerNorm (over the hidden dimension) with the element‑wise multiplication
    by the already‑computed output‑gate.

6️⃣  A final half‑precision linear projection brings the tensor back to the
    original `dim` and it is cast to float32 for the return value.

Only the two small Triton kernels are required; the heavy linear algebra is
handled by cuBLAS (via torch.nn.functional.linear / torch.bmm), which gives
the best possible throughput on modern GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: apply projection, sigmoid‑gate and optional mask in one pass
# ----------------------------------------------------------------------
@triton.jit
def apply_proj_gate_mask_kernel(
    proj_ptr,          # const  * [total_rows, hidden] (fp16)
    gate_ptr,          # const  * [total_rows, hidden] (fp16)
    out_ptr,           # mutable* [total_rows, hidden] (fp16)
    mask_ptr,          # const  * [total_rows]       (fp16) – only used when HAS_MASK=1
    total_rows: tl.int32,
    hidden: tl.int32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim block (≤128)
    HAS_MASK: tl.constexpr,  # compile‑time 0/1
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    offs_h = tl.arange(0, BLOCK_H)                           # [BLOCK_H]
    mask_h = offs_h < hidden

    # Load projection and gate
    proj = tl.load(
        proj_ptr + offs_m[:, None] * hidden + offs_h[None, :],
        mask=mask_m[:, None] & mask_h[None, :],
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
        mask=mask_m[:, None] & mask_h[None, :],
        other=0.0,
    )
    if HAS_MASK:
        # mask is broadcasted over the hidden dimension
        m = tl.load(
            mask_ptr + offs_m[:, None],
            mask=mask_m[:, None],
            other=1.0,
        )
        out = proj * gate * m
    else:
        out = proj * gate

    tl.store(
        out_ptr + offs_m[:, None] * hidden + offs_h[None, :],
        out,
        mask=mask_m[:, None] & mask_h[None, :],
    )


# ----------------------------------------------------------------------
# Kernel 2: fused LayerNorm (fp32 statistics) + output‑gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,            # const   * [total_rows, hidden] (fp16)
    y_ptr,            # mutable * [total_rows, hidden] (fp16)
    weight_ptr,       # const   * [hidden]           (fp32)
    bias_ptr,         # const   * [hidden]           (fp32)
    gate_ptr,         # const   * [total_rows, hidden] (fp16)
    total_rows: tl.int32,
    hidden: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # feature‑dim block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute per‑row mean & variance in fp32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over hidden dimension in BLOCK_D chunks
    for d_off in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                    # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Normalisation + weight/bias + gate
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden

        x_ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        x_vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate (fp16 → fp32)
        g_ptrs = gate_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        gate = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = norm * gate                                         # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        tl.store(
            y_ptrs,
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]    (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain
            * "dim"         – input channel dimension
            * "hidden_dim" – projection hidden size (always 128 in tests)
            * "nomask"      – True → ignore mask, False → use mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape            # [B, N, N, dim]
    total_rows = B * N * N                      # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm – use torch's highly tuned implementation
    # ------------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)      # half‑precision for the rest

    # ------------------------------------------------------------------
    # 2️⃣ One‑shot linear for four projections + separate out‑gate
    # ------------------------------------------------------------------
    # concatenate the four projection matrices (left, right, left_gate, right_gate)
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [hidden, dim]
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [4*hidden, dim]

    proj = F.linear(x_norm, w_proj)            # [R, 4*hidden]  fp16
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw = torch.split(
        proj, hidden, dim=-1
    )

    # out‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),  # [hidden, dim]
    )                                            # [R, hidden] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    left_gate = left_gate_raw.sigmoid_()       # fp16
    right_gate = right_gate_raw.sigmoid_()     # fp16
    out_gate = out_gate_raw.sigmoid_()         # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask (if any) + gate multiplication via Triton
    # ------------------------------------------------------------------
    BLOCK_M = 256                               # rows per program
    BLOCK_H = hidden                             # ≤128, fits in a single block

    if not config.get("nomask", True):
        # mask shape: [B, N, N] → flatten to [R] and broadcast over hidden
        mask_flat = mask_tensor.reshape(total_rows).to(torch.float16)
        HAS_MASK = 1
    else:
        mask_flat = torch.empty(0, device=device, dtype=torch.float16)  # dummy
        HAS_MASK = 0

    # Allocate output buffers for the gated projections
    left = torch.empty_like(left_proj_raw)
    right = torch.empty_like(right_proj_raw)

    grid = (triton.cdiv(total_rows, BLOCK_M),)

    apply_proj_gate_mask_kernel[grid](
        left_proj_raw,
        left_gate,
        left,
        mask_flat,
        total_rows,
        hidden,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )
    apply_proj_gate_mask_kernel[grid](
        right_proj_raw,
        right_gate,
        right,
        mask_flat,
        total_rows,
        hidden,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Reshape to 4‑D tensors and perform the cubic update via batched GEMM
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden)          # [B, N, N, H]
    right = right.view(B, N, N, hidden)        # [B, N, N, H]

    # (B, H, N, N) → (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # Batched matrix multiplication (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] fp16

    # Restore (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + element‑wise output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, hidden).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    # Use the same block size as the first kernel (covers hidden ≤128)
    layernorm_gate_kernel[grid](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.reshape(total_rows, hidden),  # gate (fp16)
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=128,                     # hidden_dim never exceeds 128
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, hidden)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out_norm.reshape(total_rows, hidden),
        weights["to_out.weight"].to(torch.float16),   # [dim, hidden]
    )                                                # [R, dim] fp16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
```
