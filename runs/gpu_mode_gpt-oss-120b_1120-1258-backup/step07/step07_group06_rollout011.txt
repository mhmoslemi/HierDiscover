```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # compute batch index from program_id(0)
    batch_idx = tl.program_id(0)
    # compute tile i, j indices
    tile_i = tl.program_id(1)
    tile_j = tl.program_id(2)
    i_start = tile_i * BLOCK_M
    j_start = tile_j * BLOCK_N
    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        # compute offsets
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_j = j_start + tl.arange(0, BLOCK_N)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N
        # left: shape [M, K]
        left_ptrs = left_ptr + batch_idx * stride_left_batch + \
                    offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        # right (transposed): shape [K, N] actually we need [K, N] where N corresponds to j dimension.
        # Offsets for right: row = offs_k (k dimension), col = offs_j (j dimension) ...
        right_ptrs = right_ptr + batch_idx * stride_right_batch + \
                     offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)
        # Multiply: left (M,K) @ right (K,N) -> (M,N)
        acc += tl.dot(left, right)
    # after loop, store
    out_ptrs = out_ptr + batch_idx * stride_out_batch + \
                offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])
```

"""
Tri‑Multiplicative (outgoing) update – AlphaFold‑3 style.

Algorithm (all tensors stay on the GPU):
1️⃣  Row‑wise LayerNorm on the [B·N·N, dim] input (fp32 → fp16).
2️⃣  Fuse the four linear maps (left/right & their gates) into one fp16 GEMM.
    Split the result into left_proj, right_proj, left_gate, right_gate.
3️⃣  Compute the output‑gate with a second fp16 GEMM.
4️⃣  (Optional) apply the pair‑wise mask, then sigmoid the gates and
    obtain gated tensors `left` and `right`.
5️⃣  **Tri‑multiplicative update** – compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a custom Triton kernel that performs a batched GEMM over the
    combined (B·H) dimension (H = hidden_dim).  The kernel works on fp16
    data with fp32 accumulation and writes fp16 output.
6️⃣  Second LayerNorm over the hidden dimension (fp16) while *fusing* the
    multiplication by the output‑gate → a single Triton kernel.
7️⃣  Linear projection back to `dim` (fp16) and cast to float32.

Only the forward pass is needed; gradients are not computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   *[total_rows, dim]   (float32)
    y_ptr,               # mutable *[total_rows, dim]   (float16)
    weight_ptr,          # const   *[dim]               (float32)  γ
    bias_ptr,            # const   *[dim]               (float32)  β
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # write normalized values  γ·(x‑μ)·rstd + β   (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) LayerNorm + element‑wise multiplication by the output‑gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,               # const   *[total_rows, dim]   (float16)
    y_ptr,               # mutable *[total_rows, dim]   (float16)
    weight_ptr,          # const   *[dim]               (float32) γ
    bias_ptr,            # const   *[dim]               (float32) β
    mult_ptr,            # const   *[total_rows, dim]   (float16)  (out_gate)
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # mean & variance (fp32) – note that x_ptr is fp16, we promote
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # write normalized * mult   (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load raw value
        x_ptr_tile = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(
            x_ptr_tile,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        # load multiplier (output‑gate)
        mult_tile = mult_ptr + offs_m[:, None] * dim + offs_d[None, :]
        mult = tl.load(
            mult_tile,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = (norm * w[None, :] + b[None, :]) * mult   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3) Batched GEMM for the Tri‑multiplicative update
#    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#    (equivalently: out_flat[b·h] = left_flat @ right_flatᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,                # const   *[B·H, N, N]   (fp16)
    right_ptr,               # const   *[B·H, N, N]   (fp16)   (will be read transposed)
    out_ptr,                 # mutable *[B·H, N, N]   (fp16)
    total_batch: tl.int32,   # B·H
    N: tl.int32,
    stride_left_batch: tl.int32,
    stride_left_i: tl.int32,
    stride_left_k: tl.int32,
    stride_right_batch: tl.int32,
    stride_right_i: tl.int32,   # row stride (k dimension in the transposed view)
    stride_right_k: tl.int32,   # col stride (j dimension)
    stride_out_batch: tl.int32,
    stride_out_i: tl.int32,
    stride_out_j: tl.int32,
    BLOCK_M: tl.constexpr,      # tile size along i
    BLOCK_N: tl.constexpr,      # tile size along j
    BLOCK_K: tl.constexpr,      # tile size along k (reduction)
):
    batch = tl.program_id(0)                     # which (b·h) we are processing
    tile_i = tl.program_id(1)
    tile_j = tl.program_id(2)

    i_start = tile_i * BLOCK_M
    j_start = tile_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile: shape [M, K] -----
        left_ptrs = left_ptr + \
            batch * stride_left_batch + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16 → promotes to fp32 in dot

        # ----- right tile (transposed): shape [K, N] -----
        # we read right as (k, j) → row stride = stride_right_i, col stride = stride_right_k
        right_ptrs = right_ptr + \
            batch * stride_right_batch + \
            offs_k[:, None] * stride_right_i + \
            offs_j[None, :] * stride_right_k
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # matrix multiply‑accumulate
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # write result (fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        batch * stride_out_batch + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for the row‑wise ops

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim]  float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_norm1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm1](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate projection and gate matrices -> shape [4*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # fp16

    # fused GEMM:  [R, dim] @ [4*H, dim]ᵀ → [R, 4*H]  (fp16)
    proj_gate = F.linear(x_norm, w_concat)                 # fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )   # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates, apply gating
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask shape [B,N,N] → broadcast to [R,1]
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate       # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (custom Triton GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring hidden dimension into batch axis → [B·H, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # allocate output buffer
    out_flat = torch.empty_like(left_flat)   # fp16, shape [B·H, N, N]

    # strides (all tensors are contiguous)
    sL_b, sL_i, sL_k = left_flat.stride()
    sR_b, sR_i, sR_k = right_flat.stride()
    sO_b, sO_i, sO_j = out_flat.stride()

    # kernel launch parameters
    BLOCK_M_T = 64
    BLOCK_N_T = 64
    BLOCK_K_T = 64
    grid_tri = (
        B * H,                           # batch * hidden
        triton.cdiv(N, BLOCK_M_T),      # tiles over i
        triton.cdiv(N, BLOCK_N_T),      # tiles over j
    )
    tri_mul_outgoing_kernel[grid_tri](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        sL_b,
        sL_i,
        sL_k,
        sR_b,
        sR_i,          # row stride for k dimension (transposed view)
        sR_k,          # col stride for j dimension
        sO_b,
        sO_i,
        sO_j,
        BLOCK_M=BLOCK_M_T,
        BLOCK_N=BLOCK_N_T,
        BLOCK_K=BLOCK_K_T,
        num_warps=8,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiply by output‑gate (fused)
    # ------------------------------------------------------------------
    R2 = B * N * N
    out_view = out.view(R2, H).contiguous()
    out_gate_view = out_gate.view(R2, H).contiguous()

    out_norm = torch.empty_like(out_view, dtype=torch.float16)

    grid_norm2 = (triton.cdiv(R2, BLOCK_M),)
    layernorm_mul_kernel[grid_norm2](
        out_view,
        out_norm,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        out_gate_view,                   # multiplier (out_gate)
        R2,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
