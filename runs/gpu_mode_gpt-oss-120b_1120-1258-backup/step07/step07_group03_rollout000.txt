"""
Outgoing TriMul (AlphaFold‑3) forward pass – Triton‑accelerated.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim).  Custom Triton
   kernel that does fp32 reduction but writes fp16 results.

2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate,
   right_gate) into a single GEMM:
        [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (R = B·N·N, H = hidden_dim)

3️⃣ Separate linear for the output‑gate (R × H).

4️⃣ Optional mask (broadcast over the hidden dimension) is applied
   *before* the sigmoids.  Sigmoid gates and gating are performed in‑place
   on fp16 tensors.

5️⃣ Reshape to [B, N, N, H] and view the hidden dimension as part of the
   batch axis:  (B·H, N, N).  The costly N³ multiplicative update becomes
   a batched half‑precision GEMM:
        out = left @ rightᵀ   →   (B·H, N, N)

6️⃣ Second row‑wise LayerNorm over the hidden dimension (Triton).

7️⃣ Apply the output‑gate, final linear projection back to `dim`,
   and cast the result to float32.

Only the two LayerNorm steps are written in Triton; all dense matrix
multiplications use PyTorch’s highly‑optimized FP16 GEMM (Tensor Core)
which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,                 # const   * [total_rows, dim] (float32)
    y_ptr,                 # mutable * [total_rows, dim] (float16)
    w_ptr,                 # const   * [dim]          (float32)   γ
    b_ptr,                 # const   * [dim]          (float32)   β
    total_rows,            # int runtime
    dim,                   # int runtime
    eps,                   # compile‑time float
    BLOCK_M: tl.constexpr, # rows per program
    BLOCK_D: tl.constexpr, # cols per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Reductions are performed in fp32; the final result is stored as fp16.
    """
    pid = tl.program_id(0)                     # one program per tile of rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean and variance for each row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile [BLOCK_M, BLOCK_D] from x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals,  axis=1)          # sum over columns
        sum_sq  += tl.sum(vals * vals, axis=1)    # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                    # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized output
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑μ)/σ

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * gamma[None, :] + beta[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : dict with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          All weights are float32 and biases are present only for the
          LayerNorms (as in the reference code).
        - config_dict  : must contain "dim", "hidden_dim" and optionally
          "nomask" (True means mask is ignored).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for the first LayerNorm

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # Flatten to [R, dim]
    x_flat = input_tensor.view(R, dim).contiguous()   # fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16,
                         device=device)               # fp16 output

    BLOCK_M = 256    # rows per program
    BLOCK_D = 32     # columns per tile
    grid = (triton.cdiv(R, BLOCK_M),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections (single GEMM)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate weight matrices of the four linear layers:
    #   left_proj, right_proj, left_gate, right_gate
    proj_gate_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,                          # shape [4*H, dim]
    ).to(torch.float16)                  # fp16 for Tensor‑Core GEMM

    # Fused linear: [R, dim] @ [4*H, dim]ᵀ → [R, 4*H] (fp16 output)
    proj_gate = F.linear(x_norm, proj_gate_w)   # bias = None

    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]            # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]          # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]          # [R, H]

    # Output‑gate linear (separate weight)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim] → fp16
    )   # → [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating (all fp16)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask ∈ {0,1} broadcast over hidden dimension
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (half‑precision)
    left_gate  = left_gate_raw.sigmoid_()   # [R, H]
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Gating
    left  = left_proj_raw * left_gate       # [R, H]
    right = right_proj_raw * right_gate    # [R, H]

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)       # [B, N, N, H]
    right = right.view(B, N, N, H)      # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)

    # Bring hidden dimension into the batch axis → shape [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                 # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H] → fp16
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)   # fp32 output
    return final
