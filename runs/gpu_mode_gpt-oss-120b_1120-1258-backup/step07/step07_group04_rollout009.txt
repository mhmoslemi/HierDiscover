"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (over the last channel) on the input tensor.
   Implemented by a custom Triton kernel that reads float32, computes
   mean/variance in fp32 and writes the normalized result in fp16.

2️⃣ Fuse the four linear layers (left/right projections + left/right gates)
   into a single FP16 GEMM:
        [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]    (R = B·N·N)
   The result is split into *left_raw*, *right_raw*, *left_gate_raw*,
   *right_gate_raw*.

3️⃣ Compute the output‑gate with a second FP16 GEMM.

4️⃣ (Optional) Apply the binary mask to the left/right projections.
   Apply sigmoid in‑place to the three gate tensors and fuse them:
        left  = left_raw  * sigmoid(left_gate_raw)
        right = right_raw * sigmoid(right_gate_raw)

5️⃣ Triangular multiplicative update (batched “einsum”):
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented with a single call to ``torch.einsum`` – this avoids the
   costly ``permute → contiguous`` copies that a naïve ``bmm`` pipeline
   would incur.

6️⃣ Second LayerNorm over the hidden dimension (H) → fp16
   (same Triton kernel as step 1, just a different dim).

7️⃣ Multiply by the output‑gate, apply the final linear projection
   (FP16 GEMM) and cast the result back to float32.

All heavy linear algebra is performed in FP16 to exploit Tensor‑Core
throughput, while the two LayerNorms are handled by a lightweight
Triton kernel.  The implementation requires at least one Triton kernel
(as mandated) and is fully compatible with the test configurations.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 input → float16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                    # const   * [total_rows, dim]   (float32)
    y_ptr,                    # mutable * [total_rows, dim]   (float16)
    weight_ptr,               # const   * [dim]               (float32)
    bias_ptr,                 # const   * [dim]               (float32)
    total_rows,               # int (runtime)
    dim,                      # int (runtime)
    eps,                      # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols processed per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute per‑row mean and variance in float32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values: γ·(x−μ)·rstd + β  (store as fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)       # True → ignore mask
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim                  # short alias

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # Flatten input to [R, dim] (contiguous)
    x_flat = x.view(total_rows, dim).contiguous()          # float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Triton launch parameters (tuned for H100)
    BLOCK_M = 256            # rows per program
    BLOCK_D = 64             # columns per iteration (larger for speed)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],     # γ  (float32)
        weights["norm.bias"],       # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM
    # --------------------------------------------------------------
    # Concatenate the four weight matrices (no bias present)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]  (FP16 GEMM)
    fused = F.linear(x_norm_fp16, w_fused)          # fp16

    # Split fused result (all tensors stay FP16)
    left_raw        = fused[:, 0:H]                 # [R, H]
    right_raw       = fused[:, H:2*H]                # [R, H]
    left_gate_raw   = fused[:, 2*H:3*H]              # [R, H]
    right_gate_raw  = fused[:, 3*H:4*H]              # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # --------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and sigmoid gates
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1]  (fp16)
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid (fp16) – cheap and saves a temporary
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate      # [R, H]  fp16
    right = right_raw * right_gate     # [R, H]  fp16

    # --------------------------------------------------------------
    # 5️⃣ Reshape to [B, N, N, H] and compute the triangular update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)      # [B, N, N, H]
    right = right.view(B, N, N, H)     # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)

    # Batched einsum: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # This avoids the costly permute‑→‑contiguous copies required for
    # a raw torch.bmm implementation.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (H) → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)    # fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection and cast to float32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),               # [R, H]
        weights["to_out.weight"].to(torch.float16)  # [dim, H]
    )                                                # [R, dim], fp16

    # Reshape back to the original 4‑D layout and convert to fp32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
