def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps=1e-5
    device = input_tensor.device
    B,N,_,_ = input_tensor.shape
    total_rows = B * N * N
    H = hidden_dim
    # 1. Initial LN
    x_norm_fp32 = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)  # FP32
    x_norm = x_norm_fp32.to(torch.float16)
    # Flatten
    x_flat = x_norm.view(total_rows, dim)
    # 2. Concatenated projection
    w_concat = torch.cat([
        weights["left_proj.weight"].to(torch.float16),
        weights["right_proj.weight"].to(torch.float16),
        weights["left_gate.weight"].to(torch.float16),
        weights["right_gate.weight"].to(torch.float16)
    ], dim=0)
    proj = F.linear(x_flat, w_concat)  # (R,4H) fp16
    left_proj_raw = proj[:, 0:H]
    right_proj_raw = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    # 3. Out gate linear
    out_gate_raw = F.linear(x_flat, weights["out_gate.weight"].to(torch.float16))  # (R,H)
    # 4. Mask and gates
    if not config.get("nomask", True):
        mask_fp16 = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()
    left = left_proj_raw * left_gate  # (R,H)
    right = right_proj_raw * right_gate
    # 5. Triangular multiplication
    left = left.view(B, N, N, H).permute(0,3,1,2).contiguous()  # (B,H,N,N)
    right = right.view(B, N, N, H).permute(0,3,1,2).contiguous()
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N)
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,H)
    out_flat = out.view(total_rows, H)
    # 6. fused LN + out gate
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],  # FP32
        weights["to_out_norm.bias"],    # FP32
        out_gate.view(total_rows, H),   # FP16
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # 7. final linear
    w_to_out = weights["to_out.weight"].to(torch.float16)
    final_half = F.linear(out_norm_gated, w_to_out)  # (R,dim) fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

        (B·N·N, dim) @ (4·H, dim).T → (B·N·N, 4·H)

3️⃣  Compute the output‑gate projection (another half‑precision GEMM).

4️⃣  Optional mask (broadcast over the hidden dimension) and in‑place sigmoid
    on the three gate tensors.  Element‑wise gating of the left/right projections.

5️⃣  Triangular multiplicative update (`einsum('... i k d, ... j k d -> ... i j d')`):
    reshape to (B·H, N, N) and perform a batched half‑precision matrix multiply
    with its transpose (`torch.bmm`).  This is the N³ work.

6️⃣  Fused hidden‑dim LayerNorm + multiplication by the output‑gate in a single
    Triton kernel (`layernorm_gate_kernel`).  This reduces memory traffic
    compared with a separate `layer_norm + mul`.

7️⃣  Final linear projection back to `dim` (FP16 GEMM) and cast to FP32.

The implementation runs entirely on the GPU, mixes FP16 for the heavy
computations and keeps the final result in FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6️⃣  Fused hidden‑dim LayerNorm + gate multiplication (Triton)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,                # const  * [total_rows, dim]  (FP16)
    y_ptr,                # mutable* [total_rows, dim]  (FP16)  ← output
    weight_ptr,           # const  * [dim]               (FP32) γ
    bias_ptr,             # const  * [dim]               (FP32) β
    gate_ptr,             # const  * [total_rows, dim]  (FP16)  ← out_gate after sigmoid
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute per‑row mean & variance (FP32 accumulator)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # Normalize, apply γ/β and multiply by gate
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x (FP16 → FP32)
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # load gate (FP16 → FP32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # layer‑norm
        norm = (x - mean[:, None]) * rstd[:, None]      # FP32

        # load γ/β (FP32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = (norm * w[None, :] + b[None, :]) * gate   # FP32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 0️⃣  Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : mapping from string names to torch.Tensor (model params)
        - config_dict  : contains "dim" (int) and "hidden_dim" (int);
                         optional key "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N            # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Initial LayerNorm (FP32 → FP16) using the highly‑optimized torch kernel
    # ------------------------------------------------------------------
    # Note: we keep the result in FP16 for the rest of the pipeline.
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32
    x_norm = x_norm_fp32.to(torch.float16)                     # FP16
    x_flat = x_norm.view(total_rows, dim).contiguous()        # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Fuse left/right projections + their gates into a single GEMM
    # ------------------------------------------------------------------
    # Convert weights to half‑precision once (they are small enough to fit in cache)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"].to(torch.float16),   # [H, dim]
            weights["right_proj.weight"].to(torch.float16),  # [H, dim]
            weights["left_gate.weight"].to(torch.float16),   # [H, dim]
            weights["right_gate.weight"].to(torch.float16),  # [H, dim]
        ],
        dim=0,
    )  # [4·H, dim]

    proj = F.linear(x_flat, w_concat)           # [R, 4·H]  FP16
    left_proj_raw  = proj[:, 0:H]               # [R, H]
    right_proj_raw = proj[:, H:2 * H]           # [R, H]
    left_gate_raw  = proj[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = proj[:, 3 * H:4 * H]       # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣  Output‑gate linear projection
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_flat,
        weights["out_gate.weight"].to(torch.float16),  # [H, dim]
    )  # [R, H]  FP16

    # ------------------------------------------------------------------
    # 4️⃣  Apply mask (if present) and sigmoid to gate tensors
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → [R, 1]
        mask_fp16 = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Element‑wise gating
    left = left_proj_raw * left_gate   # [R, H]  FP16
    right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H) → (B, H, N, N)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_flat = out.view(total_rows, H).contiguous()                 # [R, H]

    # ------------------------------------------------------------------
    # 6️⃣  Fused hidden‑dim LayerNorm + multiplication by out_gate (Triton)
    # ------------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_M = 256          # rows per program – fits well in shared memory
    BLOCK_D = 64           # columns per inner loop (covers hidden_dim ≤ 128 efficiently)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        out_gate.view(total_rows, H),    # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣  Final linear projection back to `dim` and cast to FP32
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim]  FP16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
