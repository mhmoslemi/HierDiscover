"""
TriMul (outgoing) – fast forward pass using Triton + Torch.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last channel (dim) of the input.
   Implemented with a custom Triton kernel that reads fp32, computes the
   mean/variance per row and writes the normalized result in fp16.

2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single fp16 GEMM:
        [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   ,   R = B·N·N
   Split the result into left_raw, right_raw, left_gate_raw, right_gate_raw.

3️⃣ Compute the output‑gate with a second fp16 GEMM.

4️⃣ (Optional) Apply the binary mask to left_raw / right_raw.
   Apply sigmoid in‑place to the three gate tensors and fuse the gates:
        left  = left_raw  * sigmoid(left_gate_raw)
        right = right_raw * sigmoid(right_gate_raw)

5️⃣ Reshape for the N³ multiplicative update and perform a batched GEMM:
        out[b, i, j, h] = Σₖ left[b, i, k, h] · right[b, j, k, h]
   This is realized as a batched matmul
        (B·H, N, N) @ (B·H, N, N)ᵀ   using torch.bmm (Tensor‑cores).

6️⃣ Second LayerNorm over the hidden dimension H (again via Triton,
   output fp16).

7️⃣ Apply the output‑gate, a final linear projection back to dim, and
   cast the result to float32.

All heavy linear algebra runs in FP16 to exploit H100 Tensor Cores.
Only the two LayerNorms are written in Triton; everything else uses
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)  – γ
    bias_ptr,         # const   * [dim]               (float32)  – β
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel reads `x` as fp32, computes the statistics in fp32,
    writes the normalized result as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)         # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)      # True → no mask applied
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total number of rows for layernorms
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp32 → fp16
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] fp32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # launch Triton layernorm kernel (tuned block sizes)
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # ------------------------------------------------------------------
    # concat the four weight matrices (each: [H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)  # [4·H, dim]

    # fused GEMM: [R, dim] @ [4·H, dim]ᵀ -> [R, 4·H]   (FP16)
    fused = F.linear(x_norm_fp16, w_fused)  # fp16

    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]

    # output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] → flatten to [R, 1] and cast to fp16
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # in‑place sigmoid (fp16) – no extra temporaries
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate    # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring hidden dim into batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch & heads → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched GEMM (FP16) – each head multiplies N×N @ (N×N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H) – fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final projection & cast to fp32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # fp16
    # final linear: [R, H] @ [dim, H]ᵀ → [R, dim]
    final_fp16 = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
