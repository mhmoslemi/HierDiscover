   ``out = left_mat @ right_mat.transpose(1,2)`` (half‑precision
   Tensor‑Core GEMM).  This computes the Einstein‐summation
   ``Σ_k left[...,k,:] * right[...,k,:]``.

5. **Second LayerNorm + output gate** – a custom Triton kernel
   (`layernorm_gate_kernel`) fuses Row‑wise LayerNorm, the affine
   transform, and the element‑wise multiplication by `out_gate`.  The
   kernel works on the flattened tensor `[R, H]` (`R = B·N·N`).

6. **Final linear projection** – a half‑precision GEMM projects back to
   the original `dim`.  The result is cast to `float32` to match the
   reference implementation.

All heavy arithmetic stays in **fp16** (except the layer‑norm statistics
which are accumulated in fp32 for numerical stability).  The kernel
launches are minimized (only one Triton kernel) and the dominant
matrix‑multiplication work is handled by cuBLAS/Tensor‑Cores, giving
sub‑millisecond runtimes on the H100 for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton Row‑wise LayerNorm + output‑gate fusion (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    gate_ptr,         # const   * [total_rows, dim]   (float16)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm + element‑wise gate.
    y = ( ((x-μ)/√(σ²+eps)) * γ + β ) * gate
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- compute mean / variance (fp32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- normalisation + gating (fp16 output) -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load original x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalise
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # affine γ,β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate (fp16 → fp32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out_fp16 = (norm * gate).to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict mapping weight name → torch.Tensor (float32)
        - config_dict  : contains
            * "dim"          – input feature dimension
            * "hidden_dim"  – hidden dimension (always 128)
            * "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape           # [B, N, N, dim]
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use fused torch implementation – fast)
    # ------------------------------------------------------------------
    # Reshape to [R, dim] and normalise across dim.
    x_flat = input_tensor.view(total_rows, dim)
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)    # cast once for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all four linear projections (left/right + left/right gates)
    # ------------------------------------------------------------------
    # concat weights → shape [4*H, dim] (fp16)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    proj_gate = F.linear(x_norm, w_4h)          # [R, 4*H]  fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # out_gate is a separate linear layer (no bias)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoids and gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True) and mask_tensor is not None:
        # broadcast mask over hidden dimension
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()               # [R, H] fp16

    left = left_proj_raw * left_gate                  # [R, H] fp16
    right = right_proj_raw * right_gate               # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors for the cubic update
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update via batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Merge batch and head dimensions for a single bmm call.
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B*H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [B*H, N, N]

    # BMM uses Tensor‑Core half‑precision kernels.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B*H, N, N] fp16

    # Restore the original layout.
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)          # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    # Choose a relatively large BLOCK_M to lower kernel launch overhead.
    BLOCK_M = 1024        # rows per program
    BLOCK_D = 128         # works for H up to 384

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        out_gate.reshape(total_rows, H), # gate (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out_norm.reshape(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] fp16

    # Cast to float32 as required by the reference implementation.
    return final.view(B, N, N, dim).to(torch.float32)
```
