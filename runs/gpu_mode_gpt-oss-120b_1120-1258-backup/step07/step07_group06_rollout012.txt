def custom_kernel(data):
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape  # note shape: bs, seq_len, seq_len, dim
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    # flatten rows
    R = B * N * N

    # Layernorm 1 using Triton
    x_flat = x.view(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 1024   # adjust
    BLOCK_D = 64     # can handle dim up to 384
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # Fuse linear projections & gates + out_gate
    # concat
    w_concat = torch.cat([
        weights["left_proj.weight"],   # [hidden_dim, dim]
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape [5*hidden_dim, dim]

    # Linear via F.linear (M=R)
    proj = torch.nn.functional.linear(x_norm, w_concat)  # shape [R, 5*H]
    # split
    H = hidden_dim
    left_proj_raw = proj[:, 0:H]
    right_proj_raw = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw = proj[:, 3*H:4*H]
    out_gate_raw = proj[:, 4*H:5*H]

    # mask
    if not cfg.get("nomask", True):
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # gated tensors
    left = left_proj_raw * left_gate   # shape [R,H]
    right = right_proj_raw * right_gate

    # reshape to [B,N,N,H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Tri-multiplicative update: batched matmul
    # Bring hidden dim to batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,H] fp16

    # Second LayerNorm (over hidden dim)
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final projection (to dim)
    final = torch.nn.functional.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),
    )

    final = final.view(B, N, N, dim).to(torch.float32)

    return final

"""
Tri‑multiplicative (outgoing) update – highly‑optimised forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) via a small Triton kernel.
2️⃣ Fuse the four linear projections (left/right & their gates) **and** the
   output‑gate linear map into a single FP16 GEMM:
        X_norm @ W_concatᵀ → [R, 5·H]   (R = B·N·N)
3️⃣ Split the concatenated result → left_proj, right_proj,
   left_gate, right_gate, out_gate.
4️⃣ Optional pair‑wise mask, sigmoid‑gates and element‑wise gating.
5️⃣ Tri‑multiplicative update:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched GEMM on the view (B·H, N, N).
6️⃣ Second LayerNorm (fp16) – same Triton kernel as step 1.
7️⃣ Scale by out_gate and a final FP16 linear projection back to *dim*.
   The result is cast back to float32.

Only the two row‑wise LayerNorms are written in Triton; all large
matrix‑multiplications use cuBLAS (torch.nn.functional.linear / torch.bmm),
which already gives near‑optimal performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]   (float32)
    y_ptr,          # mutable *[total_rows, dim]   (float16)
    weight_ptr,     # const   *[dim]               (float32)   γ
    bias_ptr,       # const   *[dim]               (float32)   β
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / sqrt(σ²+eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized values (γ·(x‑μ)·rstd + β) as fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim]  float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    # tune block sizes – dim ≤ 384 → BLOCK_D = 64 works well
    BLOCK_M = 1024
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear maps (left/right & gates + out_gate)
    # ------------------------------------------------------------------
    # concat order: left_proj, right_proj, left_gate, right_gate, out_gate
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                         # shape [5*H, dim]

    # single GEMM: (R, dim) @ (dim, 5*H).T → (R, 5*H)
    proj = F.linear(x_norm, w_concat)          # fp16, shape [R, 5*H]

    # split the concatenated result
    left_proj_raw  = proj[:, 0:H]
    right_proj_raw = proj[:, H:2 * H]
    left_gate_raw  = proj[:, 2 * H:3 * H]
    right_gate_raw = proj[:, 3 * H:4 * H]
    out_gate_raw   = proj[:, 4 * H:5 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking, sigmoid‑gates, and gating
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask shape [B,N,N] → [R,1] broadcastable over hidden dim
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H]  fp16
    right = right_proj_raw * right_gate    # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring hidden dim into batch axis → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H] (still fp16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Scale by output‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # fp16

    final = F.linear(
        out_norm.view(R, H),                            # fp16
        weights["to_out.weight"].to(torch.float16),    # [dim, H]
    )   # fp16

    final = final.view(B, N, N, dim).to(torch.float32)
    return final
```
