@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr, mask_ptr,
    weight_ptr, bias_ptr, weight_sum_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows, dim, H, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean and variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Pre-load bias and weight_sum per group
    # bias per group
    bias0 = tl.load(bias_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)    # shape H
    bias1 = tl.load(bias_ptr + H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    bias2 = tl.load(bias_ptr + 2*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    bias3 = tl.load(bias_ptr + 3*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    bias4 = tl.load(bias_ptr + 4*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)

    # weight_sum per group
    ws0 = tl.load(weight_sum_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    ws1 = tl.load(weight_sum_ptr + H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    ws2 = tl.load(weight_sum_ptr + 2*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    ws3 = tl.load(weight_sum_ptr + 3*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    ws4 = tl.load(weight_sum_ptr + 4*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)

   # Initialize accumulators for dot products:
    left_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    right_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    left_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    right_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    out_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
         offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
         mask_d = offs_d < dim

         x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
         x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)

         # weight tile for each group
         # define rows for tile: rows = offs_d
         rows = offs_d

         # compute for each group
         for group_idx in range(5):
              group_offset = group_idx * H
              w_ptrs = weight_ptr + rows[:, None] + (group_offset + tl.arange(0, H))[None, :] * dim   # shape (BLOCK_D, H)
              w = tl.load(w_ptrs, mask=mask_d[:, None] & (tl.arange(0, H)[None,:] < H), other=0.0).to(tl.float32)
              # dot
              if group_idx == 0:
                  left_acc += tl.dot(x, w)
              elif group_idx == 1:
                  right_acc += tl.dot(x, w)
              elif group_idx == 2:
                  left_gate_acc += tl.dot(x, w)
              elif group_idx == 3:
                  right_gate_acc += tl.dot(x, w)
              else:
                  out_gate_acc += tl.dot(x, w)

    # Normalize and bias add
    # Expand mean and rstd to (BLOCK_M,1)
    mean_exp = mean[:, None]
    rstd_exp = rstd[:, None]

    # compute outputs:
    left = (left_acc - mean_exp * ws0[None, :]) * rstd_exp + bias0[None, :]
    right = (right_acc - mean_exp * ws1[None, :]) * rstd_exp + bias1[None, :]
    left_gate = (left_gate_acc - mean_exp * ws2[None, :]) * rstd_exp + bias2[None, :]
    right_gate = (right_gate_acc - mean_exp * ws3[None, :]) * rstd_exp + bias3[None, :]
    out_gate = (out_gate_acc - mean_exp * ws4[None, :]) * rstd_exp + bias4[None, :]

    # apply sigmoid for gates
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate_sig = 1.0 / (1.0 + tl.exp(-out_gate))

    # mask handling
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0).to(tl.float32)  # shape (BLOCK_M)
    else:
        mask_val = tl.full([BLOCK_M], 1.0, dtype=tl.float32)

    mask_val = mask_val[:, None]  # broadcast

    left = left * left_gate_sig * mask_val
    right = right * right_gate_sig * mask_val

    # store
    left_ptrs = left_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :]
    right_ptrs = right_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :]
    out_gate_ptrs = out_gate_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :]

    tl.store(left_ptrs, left.to(tl.float16), mask=mask_m[:, None] & (tl.arange(0, H)[None,:] < H))
    tl.store(right_ptrs, right.to(tl.float16), mask=mask_m[:, None] & (tl.arange(0, H)[None,:] < H))
    tl.store(out_gate_ptrs, out_gate_sig.to(tl.float16), mask=mask_m[:, None] & (tl.arange(0, H)[None,:] < H))

@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr, mask_ptr,
    weight_ptr, bias_ptr, weight_sum_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows, dim, H, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, HAS_MASK: tl.constexpr,
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < total_rows

   # mean & var
   sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim
       x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
       x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
       sum_val += tl.sum(x, axis=1)
       sum_sq += tl.sum(x * x, axis=1)
   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   rstd = tl.rsqrt(var + eps)

   # initialize accumulators
   left_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
   right_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
   left_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
   right_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
   out_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)

   # per d tile dot
   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim
       x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None,:]
       x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)

       # weight tiles for each group
       # base for each group offset
       for g in range(5):
           g_off = g * H
           # weights shape (BLOCK_D, H)
           w_ptrs = weight_ptr + offs_d[:, None] + (g_off + tl.arange(0, H))[None, :] * dim
           w = tl.load(w_ptrs, mask=mask_d[:, None] & (tl.arange(0, H)[None, :] < H), other=0.0).to(tl.float32)
           if g == 0:
               left_acc += tl.dot(x, w)
           elif g == 1:
               right_acc += tl.dot(x, w)
           elif g == 2:
               left_gate_acc += tl.dot(x, w)
           elif g == 3:
               right_gate_acc += tl.dot(x, w)
           else:
               out_gate_acc += tl.dot(x, w)

   # load bias and weight_sum per group
   # bias groups
   bias0 = tl.load(bias_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
   bias1 = tl.load(bias_ptr + H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
   bias2 = tl.load(bias_ptr + 2*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
   bias3 = tl.load(bias_ptr + 3*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
   bias4 = tl.load(bias_ptr + 4*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)

   # weight sum
   ws0 = tl.load(weight_sum_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
   ws1 = tl.load(weight_sum_ptr + H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
   ws2 = tl.load(weight_sum_ptr + 2*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
   ws3 = tl.load(weight_sum_ptr + 3*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
   ws4 = tl.load(weight_sum_ptr + 4*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)

   # apply normalisation + bias
   m = mean[:, None]
   s = rstd[:, None]

   left = (left_acc - m * ws0[None, :]) * s + bias0[None, :]
   right = (right_acc - m * ws1[None, :]) * s + bias1[None, :]
   left_gate = (left_gate_acc - m * ws2[None, :]) * s + bias2[None, :]
   right_gate = (right_gate_acc - m * ws3[None, :]) * s + bias3[None, :]
   out_gate = (out_gate_acc - m * ws4[None, :]) * s + bias4[None, :]

   # sigmoid gates
   left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
   right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
   out_gate_sig = 1.0 / (1.0 + tl.exp(-out_gate))

   # mask
   if HAS_MASK:
       mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0).to(tl.float32)
   else:
       mask_val = tl.full([BLOCK_M], 1.0, dtype=tl.float32)
   mask_val = mask_val[:, None]

   left = left * left_gate_sig * mask_val
   right = right * right_gate_sig * mask_val

   # store
   left_ptrs = left_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :]
   right_ptrs = right_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :]
   out_gate_ptrs = out_gate_ptr + offs_m[:, None] * H + tl.arange(0, H)[None, :]

   tl.store(left_ptrs, left.to(tl.float16), mask=mask_m[:, None] & (tl.arange(0, H)[None, :] < H))
   tl.store(right_ptrs, right.to(tl.float16), mask=mask_m[:, None] & (tl.arange(0, H)[None, :] < H))
   tl.store(out_gate_ptrs, out_gate_sig.to(tl.float16), mask=mask_m[:, None] & (tl.arange(0, H)[None, :] < H))

@triton.jit
def fused_layernorm_gate_linear_kernel(
    out_ptr, out_gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    weight_to_out_ptr,
    final_ptr,
    total_rows, H, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # First pass: mean and var over H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptrs = out_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_h[None,:], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Load layernorm weight & bias vectors
    gamma = tl.load(norm_weight_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)  # (H,)
    beta = tl.load(norm_bias_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)   # (H,)

    # Second pass over H tiles and output dim tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None,:],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None,:],
                           other=0.0).to(tl.float32)

        # normalize
        norm = (out_vals - mean[:, None]) * rstd[:, None]   # (BLOCK_M, BLOCK_H)

        # apply to_out_norm weight & bias
        gamma_tile = gamma[offs_h]   # (BLOCK_H)
        beta_tile = beta[offs_h]
        norm = norm * gamma_tile[None, :] + beta_tile[None, :]

        # apply out_gate
        gated = norm * gate_vals   # (BLOCK_M, BLOCK_H)

        # linear projection to dim
        for d_off in range(0, tl.cdiv(dim, BLOCK_O)):
            offs_o = d_off * BLOCK_O + tl.arange(0, BLOCK_O)
            mask_o = offs_o < dim

            # weight slice shape (BLOCK_H, BLOCK_O)
            w_ptrs = weight_to_out_ptr + offs_h[:, None] + (offs_o[None, :] * H)
            w = tl.load(w_ptrs,
                        mask=mask_h[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)   # (BLOCK_H, BLOCK_O)

            out_tile = tl.dot(gated, w)   # (BLOCK_M, BLOCK_O)

            out_ptrs = final_ptr + offs_m[:, None] * dim + offs_o[None, :]
            tl.store(out_ptrs, out_tile,
                     mask=mask_m[:, None] & mask_o[None, :])
```

TriMul – outgoing version (AlphaFold‑3) forward pass

This implementation fuses the two LayerNorms, the four linear projections
(left/right + the two gates) and the out‑gate computation into a single
Triton kernel, thus avoiding the costly intermediate writes of the
reference implementation.  The cubic multiplicative update
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed with a single batched GEMM (torch.bmm) which is already
heavily Tensor‑Core optimised.  The second LayerNorm, the out‑gate
multiplication and the final linear projection are also fused into a
single Triton kernel.  All arithmetic (except the final cast back to
float32) stays in float16/float32 to maximise throughput on an H100.

Supported shapes:
    B ∈ {1,2}, N ∈ {128,256,512,1024},
    dim ∈ {128,384}, hidden_dim ∈ {128}
Mask may be absent (nomask=True) or a float16‑cast binary mask.
"""

        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  GEMM for the five groups (left, right, left_gate,
    #          right_gate, out_gate) – result in fp32 accumulators
    # ------------------------------------------------------------------
    left_acc      = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    right_acc     = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    left_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    right_gate_acc= tl.zeros((BLOCK_M, H), dtype=tl.float32)
    out_gate_acc  = tl.zeros((BLOCK_M, H), dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                     # (B_M,D_tile)

        # ----- loop over the 5 groups -----
        for g in range(5):
            g_off = g * H
            # weight tile shape (D_tile, H) → rows are dim, columns are out‑channels
            w = tl.load(weight_ptr
                         + offs_d[:, None]                # dim index
                         + (g_off + tl.arange(0, H))[None, :] * dim,   # channel stride
                        mask=mask_d[:, None] & (tl.arange(0, H)[None, :] < H),
                        other=0.0).to(tl.float32)                     # (D_tile, H)

            if g == 0:
                left_acc      += tl.dot(x, w)
            elif g == 1:
                right_acc     += tl.dot(x, w)
            elif g == 2:
                left_gate_acc += tl.dot(x, w)
            elif g == 3:
                right_gate_acc+= tl.dot(x, w)
            else:
                out_gate_acc  += tl.dot(x, w)

    # ------------------------------------------------------------------
    # 3️⃣  Normalise the GEMM results, add bias, apply sigmoid gates,
    #      and (optionally) mask left/right tensors.
    # ------------------------------------------------------------------
    # load per‑group bias and weight‑sum (both fp32)
    bias0 = tl.load(bias_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    bias1 = tl.load(bias_ptr + H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    bias2 = tl.load(bias_ptr + 2*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    bias3 = tl.load(bias_ptr + 3*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    bias4 = tl.load(bias_ptr + 4*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)

    ws0 = tl.load(weight_sum_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    ws1 = tl.load(weight_sum_ptr + H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    ws2 = tl.load(weight_sum_ptr + 2*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    ws3 = tl.load(weight_sum_ptr + 3*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)
    ws4 = tl.load(weight_sum_ptr + 4*H + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0)

    m = mean[:, None]                     # (B_M,1)
    s = rstd[:, None]                     # (B_M,1)

    left_raw      = (left_acc      - m * ws0[None, :]) * s + bias0[None, :]
    right_raw     = (right_acc     - m * ws1[None, :]) * s + bias1[None, :]
    left_gate_raw = (left_gate_acc - m * ws2[None, :]) * s + bias2[None, :]
    right_gate_raw= (right_gate_acc- m * ws3[None, :]) * s + bias3[None, :]
    out_gate_raw  = (out_gate_acc  - m * ws4[None, :]) * s + bias4[None, :]

    # sigmoid gates
    left_gate_sig  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate_raw))
    out_gate_sig   = 1.0 / (1.0 + tl.exp(-out_gate_raw))

    # mask (if present)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_m,
                           mask=mask_m,
                           other=1.0).to(tl.float32)[:, None]
    else:
        mask_val = tl.full([BLOCK_M, 1], 1.0, dtype=tl.float32)

    left  = left_raw  * left_gate_sig  * mask_val
    right = right_raw * right_gate_sig * mask_val

    # ------------------------------------------------------------------
    # 4️⃣  Store results (fp16)
    # ------------------------------------------------------------------
    offs_h = tl.arange(0, H)

    left_ptrs      = left_ptr      + offs_m[:, None] * H + offs_h[None, :]
    right_ptrs     = right_ptr     + offs_m[:, None] * H + offs_h[None, :]
    out_gate_ptrs  = out_gate_ptr  + offs_m[:, None] * H + offs_h[None, :]

    tl.store(left_ptrs,
             left.to(tl.float16),
             mask=mask_m[:, None] & (offs_h[None, :] < H))
    tl.store(right_ptrs,
             right.to(tl.float16),
             mask=mask_m[:, None] & (offs_h[None, :] < H))
    tl.store(out_gate_ptrs,
             out_gate_sig.to(tl.float16),
             mask=mask_m[:, None] & (offs_h[None, :] < H))


# ----------------------------------------------------------------------
# Kernel 2 – fused second LayerNorm + out‑gate multiplication + final linear
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_linear_kernel(
    # pointers
    out_ptr,          # const   *[R, H]   (float16) – result of the cubic update
    out_gate_ptr,     # const   *[R, H]   (float16)
    ln_weight_ptr,    # const   *[H]      (float32) – to_out_norm.weight
    ln_bias_ptr,      # const   *[H]      (float32) – to_out_norm.bias
    linear_weight_ptr,# const   *[dim, H] (float16) – to_out.weight
    final_ptr,        # mutable *[R, dim] (float32)
    # runtime args
    total_rows, H, dim,
    eps,
    # compile‑time constants
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance on the raw tensor (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        vals = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  Load layer‑norm γ / β (fp32) – broadcasted per tile
    # --------------------------------------------------------------
    gamma = tl.load(ln_weight_ptr + tl.arange(0, H),
                    mask=tl.arange(0, H) < H, other=0.0)   # (H,)
    beta  = tl.load(ln_bias_ptr   + tl.arange(0, H),
                    mask=tl.arange(0, H) < H, other=0.0)   # (H,)

    # --------------------------------------------------------------
    # 3️⃣  Second pass: normalise, apply out‑gate, and finish with a
    #          tiny GEMM to the final dim.
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw out and its gate
        out_raw = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)            # (B_M, B_H)
        gate_raw= tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)            # (B_M, B_H)

        # normalisation
        normed = (out_raw - mean[:, None]) * rstd[:, None]          # (B_M, B_H)
        # layer‑norm weight & bias
        gamma_tile = gamma[offs_h]                                   # (B_H,)
        beta_tile  = beta[offs_h]                                    # (B_H,)
        normed = normed * gamma_tile[None, :] + beta_tile[None, :]   # (B_M, B_H)

        # multiply by the pre‑computed out‑gate
        gated = normed * gate_raw                                    # (B_M, B_H)

        # tiny linear projection: gated @ to_out.weight^T (dim × H)
        for o_off in range(0, tl.cdiv(dim, BLOCK_O)):
            offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)
            mask_o = offs_o < dim

            # weight slice shape (B_H, B_O) → rows = hidden dim, cols = output dim tile
            w = tl.load(linear_weight_ptr
                         + offs_h[:, None]                # h index
                         + (offs_o[None, :] * H),         # d*H offset
                        mask=mask_h[:, None] & mask_o[None, :],
                        other=0.0).to(tl.float32)         # (B_H, B_O)

            # (B_M, B_H) @ (B_H, B_O) → (B_M, B_O)
            out_tile = tl.dot(gated, w)

            out_ptrs = final_ptr + offs_m[:, None] * dim + offs_o[None, :]
            tl.store(out_ptrs,
                     out_tile,
                     mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Python entry point – orchestrates the three stages
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]     (float32)
        - mask_tensor   : torch.Tensor [B, N, N]       (bool/float, optional)
        - weights_dict  : name → torch.Tensor (float32)
        - config_dict   : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                          # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 0️⃣  Prepare mask (flattened) – always pass a pointer, rely on HAS_MASK
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(torch.float16).reshape(R).contiguous()
    else:
        mask = torch.ones(R, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 1️⃣  Fuse first LayerNorm + 4 linear projections + out‑gate
    # ------------------------------------------------------------------
    # concatenate the 5 weight matrices (left/right + 2 gates + out_gate)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).contiguous()                                   # [5*H, dim]  float32
    # γ‑scale the weights
    gamma = weights["norm.weight"]                    # [dim]
    w_gamma = (w_fused * gamma[None, :]).to(torch.float16)   # [5*H, dim]  fp16
    # bias term = w_fused @ β (β = norm.bias)
    beta = weights["norm.bias"]                       # [dim]
    bias_fused = torch.matmul(w_fused.float(), beta.float())   # [5*H] float32
    # weight‑sum needed for mean subtraction ( Σ_d w_γ )
    weight_sum_fused = w_gamma.float().sum(dim=1)    # [5*H] float32

    # allocate outputs of the first kernel
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    # launch kernel
    BLOCK_M = 64          # rows per program
    BLOCK_D = 32          # dim tile (covers up to dim=384)
    grid = (triton.cdiv(R, BLOCK_M),)
    fused_norm_proj_gate_kernel[grid](
        input_tensor.reshape(R, dim).contiguous(),
        mask,
        w_gamma,
        bias_fused,
        weight_sum_fused,
        left,
        right,
        out_gate,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        H=H,
        HAS_MASK=int(not nomask),
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Cubic multiplicative update via a single batched GEMM
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) and permute to (B*H, N, N)
    left_mat  = left .view(B, N, N, H).permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.view(B, N, N, H).permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out = left @ rightᵀ   -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out_flat = out_mat.view(R, H)          # [R, H]  fp16

    # ------------------------------------------------------------------
    # 3️⃣  Fuse second LayerNorm, out‑gate multiplication and final linear
    # ------------------------------------------------------------------
    # final linear weight (dim, H) – already fp16
    w_to_out = weights["to_out.weight"].to(torch.float16).contiguous()
    # LayerNorm parameters for the output branch
    ln_w = weights["to_out_norm.weight"]   # (H,)   fp32
    ln_b = weights["to_out_norm.bias"]     # (H,)   fp32

    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    BLOCK_O = 64            # dim tile
    grid2 = (triton.cdiv(R, BLOCK_M),)
    fused_layernorm_gate_linear_kernel[grid2](
        out_flat,
        out_gate,
        ln_w,
        ln_b,
        w_to_out,
        final,
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=32,          # hidden dim tile (hides inside the kernel)
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )

    # reshape back to 4‑D tensor
    out = final.view(B, N, N, dim)
    return out
