"""
Outgoing TriMul (AlphaFold‑3) – fast Triton implementation
========================================================

This module implements the forward pass of the outgoing triangular
multiplicative update (TriMul) used in AlphaFold‑3 and other protein
structure models.  The implementation follows the reference PyTorch
module but

* fuses the four linear projections/gates into a single GEMM,
* optionally masks the projections,
* executes the costly N³ update with a custom batched GEMM kernel
  written in Triton (avoiding the per‑batch launch overhead of cuBLAS
  for many small N×N matrices),
* runs the two LayerNorms with a row‑wise Triton kernel,
* performs the majority of the work in FP16 while keeping the final
  output in FP32.

The function `custom_kernel` receives a tuple
`(input, mask, weights, config)` and returns the processed tensor
`[B, N, N, dim]` (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm (FP32 → FP16) – Triton
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – γ
    bias_ptr,       # const   * [dim]      (float32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------- 1️⃣ row‑wise mean & variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- 2️⃣ write normalized output (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32 → fp16

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Apply row‑wise LayerNorm to a 2‑D float32 tensor ``x`` and return a
    float16 tensor.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM kernel used for the N³ TriMul update
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr,          # const * [batch, M, K] (fp16)
    B_ptr,          # const * [batch, K, N] (fp16) – *original* right tensor
    C_ptr,          # mutable * [batch, M, N] (fp16) – output
    batch, M, N, K,
    stride_ab, stride_am, stride_ak,   # A strides (batch, rows, cols)
    stride_bb, stride_bk, stride_bn,   # B strides (batch, rows=K, cols=N)
    stride_cb, stride_cm, stride_cn,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]
    The kernel is a direct analogue of the dense GEMM example, but the
    batch dimension is fused into the program grid (axis‑0).
    """
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of this batch
    a_batch_ptr = A_ptr + batch_idx * stride_ab
    b_batch_ptr = B_ptr + batch_idx * stride_bb
    c_batch_ptr = C_ptr + batch_idx * stride_cb

    # ------------------------------------------------------------------
    # Accumulator (use fp32 for better numeric stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch_ptr + \
                 offs_m[:, None] * stride_am + \
                 offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (note that B is *original* right tensor,
        # i.e. we are loading B[k, n] which is Bᵀ[i, j] in the
        # mathematical formulation).
        b_ptrs = b_batch_ptr + \
                 offs_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back the result (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# TriMul forward kernel (Python wrapper) – combines everything
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dictionary with all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, cfg = data
    device = input_tensor.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                               # total rows for LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()              # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )                                                            # fp16, [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates
    # --------------------------------------------------------------
    H = hidden_dim

    # Concatenate the 5 weight matrices (left, right, left_gate, right_gate, out_gate)
    # Each matrix has shape [H, dim]; result shape [5*H, dim]
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                          # [5H, dim]

    # Single GEMM: (R, dim) @ (dim, 5H).T → (R, 5H)
    proj = F.linear(x_norm, fused_w)                               # fp16, [R, 5H]

    left_proj_raw   = proj[:, 0 * H:1 * H]                         # [R, H]
    right_proj_raw  = proj[:, 1 * H:2 * H]                         # [R, H]
    left_gate_raw   = proj[:, 2 * H:3 * H]                         # [R, H]
    right_gate_raw  = proj[:, 3 * H:4 * H]                         # [R, H]
    out_gate_raw    = proj[:, 4 * H:5 * H]                         # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (stay in fp16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply element‑wise gating
    left = left_proj_raw * left_gate        # [R, H] fp16
    right = right_proj_raw * right_gate     # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ update
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)            # [B, N, N, H]
    right = right.view(B, N, N, H)          # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)    # [B, N, N, H]

    # Bring hidden dimension to batch dimension: (B, H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch = B * H
    A = left_bh.view(batch, N, N)          # [batch, M=N, K=N]
    Bmat = right_bh.view(batch, N, N)      # [batch, K=N, N]

    out_bh = torch.empty_like(A)           # [batch, N, N] fp16

    # ------------------------------------------------------------------
    # Launch custom batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        A,
        Bmat,
        out_bh,
        batch, N, N, N,
        A.stride(0), A.stride(1), A.stride(2),     # stride_ab, stride_am, stride_ak
        Bmat.stride(0), Bmat.stride(1), Bmat.stride(2),  # stride_bb, stride_bk, stride_bn
        out_bh.stride(0), out_bh.stride(1), out_bh.stride(2),  # stride_cb, stride_cm, stride_cn
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dim) → FP16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )                                                # fp16, [R, H]
    out_norm = out_norm.view(B, N, N, H)             # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Output gate + final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                    # [B, N, N, H] fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
