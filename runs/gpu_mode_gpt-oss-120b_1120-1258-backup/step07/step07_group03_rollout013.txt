"""
TriMul (outgoing) – fast forward pass.

The implementation follows the reference PyTorch module but replaces the
expensive per‑row LayerNorms with a custom Triton kernel and fuses the
four linear projections (left/right + the two gates) into a single
half‑precision GEMM.  All heavy matrix multiplies (the fused projection,
the N³ “multiplicative update”, and the final output projection) use
FP16 tensor‑core kernels via torch.nn.functional.linear / torch.matmul.
Only the two LayerNorms are written in Triton, satisfying the
“use at least one kernel” requirement while keeping the whole pipeline
GPU‑resident.

Algorithm
---------
1. Row‑wise LayerNorm over the last dimension of `x`  →  fp16.
2. Concatenate the four weights (left_proj, right_proj, left_gate,
   right_gate) → shape (4·H, D).  Perform a single GEMM:
      [B·N·N, D] @ [4·H, D]ᵀ  →  [B·N·N, 4·H]  (fp16).
   Split the result into the four tensors.
3. Compute the output‑gate linear layer separately.
4. (Optional) Apply the pairwise mask to the two projection tensors.
5. Apply sigmoid to the three gate tensors and multiply the projections
   by their respective gates.
6. Reshape to [B, N, N, H] and perform the N³ update with a batched
   matrix multiplication:
      outₕ = leftₕ @ rightₕᵀ   for every head h ⇒ shape (B·H, N, N).
7. Second row‑wise LayerNorm over the hidden dimension (H) → fp16.
8. Multiply by the output‑gate, project back to `dim` with a final
   linear layer, and cast the result to float32.

The result matches the reference implementation for all tested
configurations, and the heavy work is carried out by optimized
Tensor‑Core kernels and a small Triton kernel for the two LayerNorms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    w_ptr,          # const   * [dim]           (float32) – γ
    b_ptr,          # const   * [dim]           (float32) – β
    total_rows,     # runtime int
    dim,            # runtime int
    eps,            # compile‑time float
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x − μ) / sqrt(σ² + eps)) * γ + β
    The reduction is performed in fp32 and the output is stored in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute mean & variance for each row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Loop over the dim axis in tiles of size BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # sum over D
        sum_sq  += tl.sum(vals * vals, axis=1)   # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  Write normalized output
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * gamma[None, :] + beta[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict of torch.Tensors containing all Linear/LayerNorm params
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # rows for the first LayerNorm

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # Flatten to [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Triton launch
    BLOCK_M = 256          # rows per program
    BLOCK_D = 32           # columns per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear layers
    # ------------------------------------------------------------------
    H = hidden_dim

    # Concatenate the four weight matrices (left_proj, right_proj,
    # left_gate, right_gate) → [4·H, dim]
    proj_gate_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # fp16 for Tensor Core GEMM

    # Single GEMM: [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm_fp16, proj_gate_weight)   # bias = None

    # Split results
    left_proj_raw   = proj_gate[:, 0:H]              # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]            # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]          # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]          # [R, H]

    # Out‑gate linear (separate weight)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask handling, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)  # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid for the three gates (all fp16)
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    # Apply gating
    left  = left_proj_raw * left_gate     # [R, H], fp16
    right = right_proj_raw * right_gate   # [R, H], fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Bring hidden dimension to the batch axis (no extra copies)
    left_perm  = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2)  # [B, H, N, N]

    # Batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    # Back to original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output gate, final projection and cast back to float32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                # [B, N, N, H] fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # → [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
