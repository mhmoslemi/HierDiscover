1. **Row‑wise LayerNorm** on the input (FP32 → FP16) using a custom Triton
   kernel – this eliminates the PyTorch `nn.LayerNorm` call.
2. **Fused projection & gating** – a single cuBLAS GEMM produces
   `left_proj, right_proj, left_gate, right_gate` from the normalized input.
3. **Optional pairwise mask** applied to the two projected tensors.
4. **Sigmoid gates** are applied in‑place, yielding gated tensors
   `left` and `right`.
5. **Cubic update** – for each hidden slice we need a matrix multiplication
   `out = left @ rightᵀ`.  A custom Triton batched‑GEMM kernel processes the
   whole `(B·H, N, N)` batch in one launch, avoiding the overhead of many
   tiny cuBLAS kernels.
6. **LayerNorm + output‑gate** – a second Triton kernel computes the per‑row
   mean/variance, normalises the result, and multiplies by `out_gate` in a
   single pass.
7. **Final projection** – a cuBLAS linear layer maps the hidden dimension back
   to `dim`.  The result is cast back to FP32 to match the reference
   implementation.

All heavy linear work stays in FP16, so Tensor‑cores are fully utilised.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # const  * [total_rows, dim] (float32)
    y_ptr,      # mutable* [total_rows, dim] (float16)
    w_ptr,      # const  * [dim]            (float32) – γ
    b_ptr,      # const  * [dim]            (float32) – β
    total_rows, # int32
    dim,        # int32
    eps,        # float32
    BLOCK_M: tl.constexpr,  # rows processed per program
    BLOCK_D: tl.constexpr,  # columns processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- accumulate sum & sum‑of‑squares ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, 1)
        sum_sq  += tl.sum(vals * vals, 1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalised values (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Batched GEMM kernel for the cubic update:
#     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#   left/right are provided as (B·H, N, N) in row‑major order.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,          # const * [batch, M, K] (fp16)
    b_ptr,          # const * [batch, K, N] (fp16)  – already transposed, i.e. Bᵀ
    c_ptr,          # mutable * [batch, M, N] (fp16)
    batch_stride,   # int32 – offset between successive batch matrices (M*K)
    M, N, K,        # int32 – matrix dimensions (M=N in our case)
    stride_am, stride_ak,                # for A: row stride, col stride
    stride_bk, stride_bn,                # for Bᵀ: row stride (k), col stride (n)
    stride_cm, stride_cn,                # for C: row stride, col stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # tiles per dimension
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    # decode pid -> (batch, tile_m, tile_n)
    batch_id = pid // (num_tiles_m * num_tiles_n)
    tile_id  = pid % (num_tiles_m * num_tiles_n)
    tile_m   = tile_id // num_tiles_n
    tile_n   = tile_id % num_tiles_n

    # starting offsets for this tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch
    a_batch = a_ptr + batch_id * batch_stride
    b_batch = b_ptr + batch_id * batch_stride

    # initialise accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Bᵀ tile (K x N) – remember B is transposed, therefore
        # row stride = 1 (move along K) and col stride = N (move along N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3️⃣ LayerNorm + output‑gate fusion (second LN)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,      # const  * [total_rows, dim] (fp16)
    gate_ptr,   # const  * [total_rows, dim] (fp16) – out_gate after sigmoid
    y_ptr,      # mutable* [total_rows, dim] (fp16) – result after LN * gate
    w_ptr,      # const  * [dim]            (fp32) – γ
    b_ptr,      # const  * [dim]            (fp32) – β
    total_rows, # int32
    dim,        # int32
    eps,        # float32
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f = vals.to(tl.float32)
        sum_val += tl.sum(vals_f, 1)
        sum_sq  += tl.sum(vals_f * vals_f, 1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalise, apply γ/β, multiply by gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)

        norm = (x - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = (norm * gamma[None, :] + beta[None, :]) * gate
        out_f16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_f16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator – fully
    Triton‑accelerated (except the large GEMM’s that stay in cuBLAS).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – optional
        - weights_dict : model weights (see key names below)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    ########################################
    # 0️⃣ unpack & sanity checks
    ########################################
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    total_rows = B * N * N        # rows for the first LN
    H = hidden_dim

    ########################################################
    # 1️⃣ first LayerNorm (row‑wise) – FP32 → FP16
    ########################################################
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 128
    grid_norm = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (FP32)
        weights["norm.bias"],     # β  (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    ########################################################
    # 2️⃣ fused projection + gate linear maps (4·H × dim)
    ########################################################
    # concatenate weights once (FP16)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H , dim]

    proj_gate = F.linear(x_norm, w_4h)       # [R, 4·H]  FP16
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # out‑gate (separate linear map)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # [R, H] FP16

    ########################################################
    # 3️⃣ optional mask, sigmoid gates and element‑wise gating
    ########################################################
    if not nomask:
        # mask: [B,N,N] → [R,1] (float16)
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # in‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply the two projection gates
    left  = left_proj_raw * left_gate          # [R, H] FP16
    right = right_proj_raw * right_gate        # [R, H] FP16

    # reshape to (B, N, N, H) for the cubic update
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)        # keep for later

    ########################################################
    # 4️⃣ cubic update via custom batched GEMM
    ########################################################
    # bring H into the batch dimension → (B·H, N, N) row‑major
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)   # fp16
    right_mat = right_perm.view(B * H, N, N)  # fp16 (will be logically transposed)

    # allocate output buffer (fp16)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_tiles_m = triton.cdiv(N, BLOCK_M)
    num_tiles_n = triton.cdiv(N, BLOCK_N)
    grid_gemm = (B * H * num_tiles_m * num_tiles_n,)

    batch_stride = N * N                       # distance between two matrices

    # Strides for A (left): row stride = N, col stride = 1
    stride_am = N
    stride_ak = 1
    # Strides for Bᵀ (rightᵀ): row stride = 1 (move along K), col stride = N (move along N)
    stride_bk = 1
    stride_bn = N
    # Strides for C (output)
    stride_cm = N
    stride_cn = 1

    batched_gemm_kernel[grid_gemm](
        left_mat,
        right_mat,
        out_mat,
        batch_stride,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    ########################################################
    # 5️⃣ second LayerNorm + output gate (fused)
    ########################################################
    out_flat = out.reshape(total_rows, H).contiguous()
    gated = torch.empty_like(out_flat, dtype=torch.float16)   # will hold LN(out) * out_gate

    grid_norm2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid_norm2](
        out_flat,
        out_gate.reshape(total_rows, H).contiguous(),
        gated,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    ########################################################
    # 6️⃣ final linear projection back to original dim
    ########################################################
    final = F.linear(
        gated,
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # [R, dim] fp16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
