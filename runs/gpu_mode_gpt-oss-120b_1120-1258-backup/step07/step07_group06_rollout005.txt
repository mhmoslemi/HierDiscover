"""
TriMul (outgoing) forward pass – high‑performance version.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim) → fp16.
2️⃣ Fuse the four linear maps (left/right projections + their gates) into a single
   batched GEMM:  X_norm @ W_concatᵀ  →  [R, 4·H] (R = B·N·N, H = hidden_dim).
   Slice the result into left_proj, right_proj, left_gate, right_gate.
   Compute the output gate with a second GEMM.
3️⃣ (Optional) apply the pair‑wise mask on the *projections*.
   Apply sigmoid to the gates and multiply projections by their gates.
4️⃣ Tri‑multiplicative update:
        out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   implemented as a batched half‑precision GEMM (torch.bmm) on the view
   (B·H, N, N).
5️⃣ Second LayerNorm over the hidden dimension → fp16.
6️⃣ Multiply by the output‑gate, project back to the original dimension,
   and cast to float32 (the required output dtype).

Only the two row‑wise LayerNorms are written in Triton; the heavy matrix
multiplications use cuBLAS (via torch.nn.functional.linear / torch.bmm),
which gives near‑optimal performance on an H100.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]   (float32)
    y_ptr,          # mutable *[total_rows, dim]   (float16)
    weight_ptr,     # const   *[dim]               (float32)   γ
    bias_ptr,       # const   *[dim]               (float32)   β
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / sqrt(σ²+eps)) * γ + β
    Output is fp16, accumulation in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over feature dimension in BLOCK_D sized chunks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        # address: [m, d] with stride dim on the feature axis
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                       # [BLOCK_M, BLOCK_D] fp32
        sum_val += tl.sum(vals, axis=1)        # sum over D
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized values γ·(x‑μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all linear / layer‑norm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : dict with keys
            * "dim"        – input feature dimension
            * "hidden_dim" – hidden dimension H
            * "nomask"     – bool, True if the mask should be ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total rows for row‑wise ops

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using Triton
    # ------------------------------------------------------------------
    # Flatten to (R, dim). view() is cheap (no copy) because the tensor is
    # already contiguous in the last two axes.
    x_flat = x.view(R, dim)
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 32           # feature chunk size
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection/gate weight matrices → [4*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16

    # (R, dim) @ (4*H, dim)ᵀ → (R, 4*H)   (half‑precision GEMM via cuBLAS)
    proj_gate = F.linear(x_norm, w_concat)  # fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )   # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any), sigmoid gates and apply gating
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask: [B, N, N] → [R, 1] and broadcast over hidden dim
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid to save memory
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate     # [R, H]  fp16
    right = right_proj_raw * right_gate   # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H) and move hidden dim into the batch axis.
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final projection back to `dim`, cast to fp32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate         # element‑wise, fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
