    b_ptr,        # const   * [dim]              (float32) – β
    total_rows,   # int
    dim,          # int
    eps,          # const float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise (per‑row) LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The reduction (mean / variance) is performed in fp32,
    the normalized output is written as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------- 1. compute mean & variance per row --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the hidden dimension
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [BLOCK_M, BLOCK_D]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # sum over d dimension
        sum_sq  += tl.sum(vals * vals, axis=1)   # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1 / sqrt(var + eps)

    # -------------------- 2. write normalized output --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd   ->   normalized value (fp32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β are fp32 vectors
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # [BLOCK_D]
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be ignored
        - weights_dict : dict of torch.Tensors (model parameters)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    device      = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total number of (b,i,k) rows

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()               # [R, dim] fp32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16,
                              device=device)

    BLOCK_M = 256                 # rows per program; good trade‑off for all test sizes
    BLOCK_D = 32                  # columns per tile (must divide dim ≤ 768 efficiently)
    grid = (triton.cdiv(R, BLOCK_M),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (fp32)
        weights["norm.bias"],        # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & gates into a single GEMM
    #    output shape: (R, 4·H)
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate the 4 linear weight matrices (each [H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,                           # → [4·H, dim]
    ).to(torch.float16)                  # fp16 for Tensor‑Core GEMM

    # fused linear: (R, dim) @ (4·H, dim).T → (R, 4·H)
    proj_gate = F.linear(x_norm_fp16, w_fused)    # bias = None, result fp16

    left_proj_raw   = proj_gate[:, 0:H]                # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]              # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]            # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]            # [R, H]

    # out‑gate (separate linear layer)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask is [B, N, N]; broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).view(R, 1)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # inplace sigmoid for speed (fp16)
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    # gating
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update  (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # bring hidden dim to batch axis for a single large batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)       # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)      # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],    # γ (fp32)
        weights["to_out_norm.bias"],      # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate   # element‑wise gating, fp16

    final_half = F.linear(
        out_norm.view(R, H),                # [R, H]
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
