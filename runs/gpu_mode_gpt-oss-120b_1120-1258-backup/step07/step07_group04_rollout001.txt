"""
custom_kernel - Outgoing TriMul (AlphaFold3)

Implements the forward pass of the “outgoing” TriMul operator (the
computational core of AlphaFold3, Chai, Protenix …).

Algorithm (all heavy math in FP16):
1️⃣ Row‑wise LayerNorm over the last channel (dim) → custom Triton kernel.
2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single GEMM:   X_norm @ W_fusedᵀ  →  [R, 4·H]   (R = B·N·N).
   Split the result into raw projections and raw gates.
3️⃣ Compute the output‑gate with a second GEMM.
4️⃣ (Optional) apply the binary mask to the raw projections.
5️⃣ Apply sigmoid to the three gate tensors and fuse them:
        left  = left_raw  * sigmoid(left_gate)
        right = right_raw * sigmoid(right_gate)
6️⃣ Reshape to [B, H, N, N] and perform the N³ multiplicative update as a
   batched matrix‑multiply:
        out[b,h] = left[b,h] @ right[b,h]ᵀ   (FP16 Tensor‑Core GEMM)
7️⃣ Row‑wise LayerNorm over the hidden dimension (H) → Triton kernel.
8️⃣ Multiply by the output‑gate, apply the final linear projection
   (FP16 GEMM) and cast the result back to FP32.

Both LayerNorms are executed by a single Triton kernel (`layernorm_kernel`);
all GEMMs use PyTorch/ATen (FP16 Tensor‑Cores) which is already optimal on
H100.  The implementation supports optional masks and works for any
configurable batch size, sequence length, and dimensionalities listed in the
task description.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)  γ
    bias_ptr,         # const   * [dim]               (float32)  β
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # const float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Input x  : float32
    Output y : float16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the channel dimension in tiles
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # address: row * dim + col
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)            # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized values: γ·(x−μ)·rstd + β  (as fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters
        - config_dict  : dict with keys
                         * "dim" (int)
                         * "hidden_dim" (int)
                         * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)   # True → no mask
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim                  # hidden dimension

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # flatten to [R, dim] (contiguous)
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256   # rows per program (tuned)
    BLOCK_D = 32    # columns per iteration (tuned)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,          # H100 has many SMs → use more warps
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM
    # --------------------------------------------------------------
    # concatenate weight matrices: [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # [4·H, dim]

    # GEMM: [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (FP16)
    fused = F.linear(x_norm_fp16, w_fused)     # fp16

    # split fused result
    left_raw       = fused[:, 0:H]                     # [R, H]
    right_raw      = fused[:, H:2*H]                 # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]               # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]               # [R, H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(dtype=torch.float16, device=device)  # [H, dim]
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates and combine
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_fp16 = mask.to(dtype=torch.float16, device=device).reshape(total_rows, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # in‑place sigmoid to keep memory low
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H]  fp16
    right = right_raw * right_gate        # [R, H]  fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] → permute to [B, H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # flatten batch & heads → [B·H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ   →  [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim H → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # elementwise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device)   # [dim, H]
    )   # [R, dim] fp16

    # Cast back to FP32 as the model expects
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
