"""
custom_kernel
=============
Highly‑optimized forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Key ideas
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the 4‑D input.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate) into a
   single half‑precision GEMM:     X_norm @ W_concatᵀ → [R, 4·H].
3️⃣ Compute the output‑gate projection with another half‑precision GEMM.
4️⃣ Apply optional mask **before** the gate, then inplace‑sigmoid on the three
   gate tensors.
5️⃣ Batched matrix multiplication:   (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
   implements `einsum('... i k d, ... j k d -> ... i j d')`.
6️⃣ Fuse hidden‑dim LayerNorm together with multiplication by the output‑gate.
7️⃣ Final linear projection back to the original `dim` (half‑precision GEMM) and
   cast to FP32.

All heavy per‑row work (the two LayerNorms) runs in Triton kernels; the
batched GEMM uses cuBLAS via `torch.bmm`, which is already optimal on H100.
The implementation works for `dim ∈ {128,384}`, `hidden_dim = 128`,
batch size 1‑2 and sequence lengths up to 1024, with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (FP32)
    y_ptr,               # mutable * [total_rows, dim]   (FP16)
    weight_ptr,          # const   * [dim]               (FP32) γ
    bias_ptr,            # const   * [dim]               (FP32) β
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # cols per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [BLOCK_M, BLOCK_D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                 # 1 / sqrt(var+eps)

    # ---------- write normalized (γ·(x-μ)·rstd + β) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (FP32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (FP32)

        out = norm * w[None, :] + b[None, :]                    # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused LayerNorm + multiplication by a gate (hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,                # input / output [total_rows, dim] (FP16)
    weight_ptr, bias_ptr,        # LN γ,β (FP32)
    gate_ptr,                    # gate tensor, same shape as x (FP16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------- mean / variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- normalized + bias + gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load input
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # load gate (FP16 → FP32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]    # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (FP32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (FP32)

        out = (norm * w[None, :] + b[None, :]) * g  # FP32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – optional
        - weights_dict : mapping name → torch.Tensor (all FP32)
        - config_dict  : must contain "dim" (int) and "hidden_dim" (int);
                         optional key "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype torch.float32
    """
    # --------------------------------------------------------------
    # Unpack -------------------------------------------------------
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H == 128 in all test configs
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                    # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # flatten to [R, dim] – keep as contiguous FP32
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # Choose a block size that fits both dim=128 and dim=384 efficiently.
    # BLOCK_D is a multiple of 32 for good memory alignment.
    BLOCK_D = 64 if dim > 128 else 32
    BLOCK_M = 256

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ  (FP32)
        weights["norm.bias"],            # β  (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (single GEMM)
    # --------------------------------------------------------------
    # Convert weights once to FP16
    w_left   = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16)  # [H, dim]
    w_lgate  = weights["left_gate.weight"].to(torch.float16)  # [H, dim]
    w_rgate  = weights["right_gate.weight"].to(torch.float16)  # [H, dim]

    # Concatenate along output (row) dimension → [4·H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)

    # Linear: [R, dim] (FP16) @ [4·H, dim].T → [R, 4·H] (FP16)
    proj_gate = F.linear(x_norm, w_concat)          # (bias=False)

    # Slice the four parts
    left_proj_raw  = proj_gate[:, :H]                # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate projection (another GEMM)
    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate_raw = F.linear(x_norm, w_out_gate)                # [R, H] (FP16)

    # --------------------------------------------------------------
    # 4️⃣ Apply optional mask, sigmoid gates, and element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] (FP16) for broadcasting
        mask_flat = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # In‑place sigmoid (half‑precision)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Element‑wise multiplication of projections by their gates
    left = left_proj_raw * left_gate          # [R, H] FP16
    right = right_proj_raw * right_gate       # [R, H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H] then permute to bring H into batch dim
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)                               # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                              # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_flat = out.view(total_rows, H)                               # [R, H]

    # --------------------------------------------------------------
    # 6️⃣ Fused hidden‑dim LayerNorm + multiplication by out_gate
    # --------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        out_gate.view(total_rows, H),    # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] FP16

    # Cast to FP32 as required by the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
