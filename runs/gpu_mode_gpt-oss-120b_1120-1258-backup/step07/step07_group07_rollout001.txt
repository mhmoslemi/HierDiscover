"""
custom_kernel – outgoing TriMul (AlphaFold‑3) forward pass

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (float32 → float16) using a
   Triton kernel (`layernorm_kernel`).  The LayerNorm parameters are
   stored in FP32 (as in the reference PyTorch model) and the result is
   kept in FP16 for the rest of the computation.

2️⃣ All four linear projections (left/right + left/right‑gate) are merged
   into a single GEMM by concatenating the weight matrices.  The gate
   for the output (`out_gate`) is computed with a second linear layer.
   All linear layers run in FP16 with `torch.nn.functional.linear`.

3️⃣ Optional mask (shape [B,N,N]) is applied to the two *proj* tensors
   before the sigmoid gates.

4️⃣ Element‑wise sigmoid gates are applied in‑place on the FP16 tensors.
   The gated tensors `left` and `right` are reshaped to
   `[B,N,N,hidden]`.

5️⃣ Cubic multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each head:
      left_mat  = left.permute(0,3,1,2).reshape(B*H,N,N)
      right_mat = right.permute(0,3,1,2).reshape(B*H,N,N)
   and `torch.bmm(left_mat, right_mat.transpose(1,2))` (FP16, Tensor‑Core).

6️⃣ Second LayerNorm + output‑gate multiplication are fused in a single
   Triton kernel (`layernorm_gate_kernel`).

7️⃣ Final linear projection back to `dim` (FP16 → FP32) with
   `torch.nn.functional.linear`.

All heavy arithmetic stays in FP16, only the final result is cast back
to FP32 to match the reference implementation.  The kernels are written
for Triton 3.3.1 and run efficiently on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # feature‑dim block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ compute per‑row mean / variance in fp32
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / √(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalised values (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Row‑wise LayerNorm + element‑wise gate (fusion)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    gate_ptr,         # const   * [total_rows, dim]   (float16)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load original x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate (fp16 → fp32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                        # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – optional
        - weights_dict : mapping ``name → torch.Tensor`` (float32)
        - config_dict  : must contain ``"dim"``, ``"hidden_dim"``,
                         and ``"nomask"`` (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape          # input shape: [B, N, N, dim]
    total_rows = B * N * N                    # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)                     # [R, dim] fp16

    BLOCK_M = 256          # tuneable: rows per program
    BLOCK_D = 128          # supports dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ   (float32)
        weights["norm.bias"],     # β   (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear layers
    #    Concatenate the four weight matrices (left/right + left/right_gate)
    # ------------------------------------------------------------------
    # each weight: [hidden_dim, dim] (float32)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [4*H, dim]

    # single GEMM → [R, 4*H] (fp16)
    proj_gate = F.linear(x_norm, w_4h)       # no bias

    # slice the concatenated result
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # separate linear for out_gate (still fp16)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask handling, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → flatten to [R, 1] and broadcast over H
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid (fp16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()               # [R, H] fp16

    # apply gates to projected tensors
    left = left_proj_raw * left_gate                 # [R, H]
    right = right_proj_raw * right_gate              # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensor layout
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)            # [B, N, N, H]   fp16
    right = right.view(B, N, N, H)          # [B, N, N, H]   fp16
    out_gate = out_gate.view(B, N, N, H)    # [B, N, N, H]   fp16

    # ------------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # bring hidden dimension into the batch axis
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B*H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # [B*H, N, N]

    # bmm returns fp16, uses Tensor‑Core on H100
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)          # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate fusion
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.reshape(total_rows, H), # gate tensor (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out_norm.reshape(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final.view(B, N, N, dim).to(torch.float32)   # cast back to fp32
    return final
