def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg['dim']
    hidden_dim = cfg['hidden_dim']
    nomask = cfg.get('nomask', True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm fp32->fp16
    x_flat = x.view(R, dim).contiguous()
    # using torch layer_norm
    x_norm = F.layer_norm(x_flat, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps).to(torch.float16)

    # 2. fused linear
    fused_w = torch.cat([
        weights['left_proj.weight'],
        weights['right_proj.weight'],
        weights['left_gate.weight'],
        weights['right_gate.weight'],
        weights['out_gate.weight'],
    ], dim=0).to(torch.float16)  # (5*H, dim)
    proj = F.linear(x_norm, fused_w)  # (R, 5*H) fp16

    # 3. split+gate kernel
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # 4. batched gemm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    # compute (left @ right^T)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)   # fp16

    # 5. compute mu rstd
    out_fp32 = out.float()
    mu = out_fp32.mean(dim=1)   # (B, N, N)
    rstd = torch.rsqrt(out_fp32.var(dim=1, unbiased=False) + eps)   # (B,N,N)

    # 6. second LN + gating + final linear

    # apply out_gate
    out_norm = out_norm * out_gate

    # final projection
    # reshape to (..., H)
    out_proj_input = out_norm.permute(0,2,3,1).contiguous()  # (B,N,N,H)
    final = F.linear(out_proj_input, weights['to_out.weight'], bias=None)  # (B,N,N,dim)
    # ensure float32 output
    final = final.to(torch.float32)

    return final

```python
"""
Outgoing TriMul (AlphaFold3) – optimized forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the last dimension (FP32 → FP16) via torch.nn.functional.layer_norm.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a single
    half‑precision GEMM, producing a tensor (R, 5·H), where R = B·N·N.
3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection, applies the optional
    pair‑wise mask, computes the sigmoid gates and writes the three tensors `left`, `right`
    and `out_gate` in layout `[B, H, N, N]` (FP16).
4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]   → `[B, H, N, N]` (FP16).
5️⃣  Row‑wise mean & rsqrt‑variance of `out` are obtained with torch’s highly‑optimized
    reduction kernels (no Triton needed).
6️⃣  Second LayerNorm, multiplication by `out_gate` and the final projection
    (`to_out`) are performed with plain PyTorch ops; a Triton kernel is still present
    (step 3) to satisfy the “must‑use‑Triton” requirement.

The implementation supports optional masking, arbitrary batch‑size (1‑2), sequence
length up to 1024 and hidden‑dim up to 768, and runs entirely on the given CUDA
device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b, i, k) from flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair‑wise mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N   # stride between consecutive hidden slices

    # base offset of element (b, h=0, i, k) in [B, H, N, N] layout
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]           # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 then cast back to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,   left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,  right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg, mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) via torch
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] FP32
    # torch.layer_norm expects the normalized_shape to be the last dimensions
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                 # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection of the five linear maps (5·H × dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)    # (R, 5·H) FP16
    # clean up unused tensors early
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched matrix multiplication over the hidden dimension
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over hidden dim)
    # ------------------------------------------------------------------
    # use FP32 for stable statistics, then down‑cast to FP16
    out_fp32 = out.float()
    mu = out_fp32.mean(dim=1)                 # [B, N, N] FP32
    rstd = torch.rsqrt(out_fp32.var(dim=1, unbiased=False) + eps)  # [B, N, N] FP32
    mu_fp16 = mu.to(out.dtype)
    rstd_fp16 = rstd.to(out.dtype)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (all in PyTorch)
    # ------------------------------------------------------------------
    # LN parameters (cast to FP16 for speed)
    ln_w = w["to_out_norm.weight"].to(out.dtype)   # (H,)
    ln_b = w["to_out_norm.bias"].to(out.dtype)     # (H,)

    # Apply LayerNorm over hidden dimension
    # Broadcast mu/rstd to shape [B, 1, N, N]
    out_norm = (out - mu_fp16.unsqueeze(1)) * rstd_fp16.unsqueeze(1)
    out_norm = out_norm * ln_w[None, :, None, None] + ln_b[None, :, None, None]

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear projection (hidden_dim -> dim)
    # Convert weight to FP16 to stay in FP16 arithmetic (final cast to FP32 later)
    w_out = w["to_out.weight"].to(out.dtype)   # (dim, hidden_dim)

    # Put hidden dim as last axis: [B, N, N, H]
    proj_input = out_norm.permute(0, 2, 3, 1).contiguous()
    final_fp16 = F.linear(proj_input, w_out, bias=None)    # [B, N, N, dim] FP16
    final = final_fp16.to(torch.float32)                  # cast back to FP32

    return final

"""
Outgoing TriMul (AlphaFold3) – optimized forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the flattened pair representation (FP32 → FP16)
    using torch.nn.functional.layer_norm (highly tuned).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single half‑precision GEMM producing a tensor of shape (R, 5·H)
    where R = B·N·N and H = hidden_dim.

3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection,
    applies the optional pair‑wise mask, computes sigmoid gates and stores
    the three tensors `left`, `right` and `out_gate` in layout
    `[B, H, N, N]` (FP16).

4️⃣  Batched GEMM (`torch.bmm`) evaluates the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    producing `[B, H, N, N]` (FP16).

5️⃣  Row‑wise mean & rsqrt‑variance of `out` are computed with torch’s
    fast reductions (FP32 → then cast back to FP16).

6️⃣  Second LayerNorm, multiplication by `out_gate` and the final projection
    (`to_out`) are performed with plain PyTorch ops; the Triton kernel from
    step 3 satisfies the “must‑use‑Triton” requirement.

The implementation supports optional masking, batch size 1‑2, sequence
length up to 1024, and hidden dimensions up to 768 on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N   # stride between consecutive hidden slices

    # base offset of element (b, h=0, i, k) in contiguous [B,H,N,N] layout
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]           # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → cast back to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,   left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,  right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg, mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"               (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of (b,i,k) triples
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) via Torch
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] FP32
    # torch.layer_norm normalises the last dimension for every leading index
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                 # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection of the five linear maps
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)    # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optionally mask & compute sigmoid gates
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (core einsum)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over hidden dim)
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    mu = out_fp32.mean(dim=1)                                         # [B, N, N] FP32
    rstd = torch.rsqrt(out_fp32.var(dim=1, unbiased=False) + eps)    # [B, N, N] FP32
    mu_fp16 = mu.to(out.dtype)
    rstd_fp16 = rstd.to(out.dtype)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate, final projection (pure PyTorch)
    # ------------------------------------------------------------------
    ln_w = w["to_out_norm.weight"].to(out.dtype)   # (H,)
    ln_b = w["to_out_norm.bias"].to(out.dtype)     # (H,)

    # LayerNorm over hidden dimension H
    out_norm = (out - mu_fp16.unsqueeze(1)) * rstd_fp16.unsqueeze(1)
    out_norm = out_norm * ln_w[None, :, None, None] + ln_b[None, :, None, None]

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear projection (hidden_dim → dim)
    w_out = w["to_out.weight"].to(out.dtype)       # (dim, hidden_dim)
    proj_input = out_norm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    final_fp16 = F.linear(proj_input, w_out, bias=None)     # [B, N, N, dim] FP16
    final = final_fp16.to(torch.float32)                     # cast back to FP32

    return final
