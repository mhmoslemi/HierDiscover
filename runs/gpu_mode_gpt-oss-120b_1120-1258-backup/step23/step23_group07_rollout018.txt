   This leverages cuBLAS Tensor‑Core kernels and is optimal for the
   N³ work.

5️⃣ A second LayerNorm (over the hidden dimension H) is applied to the
   contracted tensor using PyTorch `layer_norm`.  Its weight/bias are
   cast to FP16 so that the operation runs in mixed precision.

6️⃣ The output‑gate is multiplied element‑wise, the result is flattened
   to `(R, H)`, and a final half‑precision linear projection
   (`to_out`) maps the hidden dimension to `dim`.  The final tensor is
   cast back to FP32.

All heavy reductions are performed in FP32 inside the PyTorch
LayerNorm kernels, while the bulk of the work stays in FP16/Tensor
Core arithmetic.  The only custom Triton kernel is the split/fuse
kernel, which is cheap compared to the O(N³) GEMM.

The entry point is `custom_kernel`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 3️⃣ – split the fused projection, apply optional mask and gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,  # rows per program
    BLOCK_H: tl.constexpr,  # hidden channels per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode flat index r = b·N·N + i·N + k ----------
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # ---- optional scalar mask (broadcast over hidden) ----------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16 scalar per (b,i,k)
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- stride for the hidden plane (N×N) ----------
    hidden_stride = N * N
    # base offset of the first hidden channel for this (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k

    # ---- iterate over hidden dimension in tiles ----------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]                 # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # ----- sigmoid in FP32 for stability -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ----------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # ----- store into the 4‑D tensors ----------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or ``None``
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32)
        - config_dict  : must contain ``"dim"``, ``"hidden_dim"``,
          ``"nomask"`` (bool)

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Pair‑wise LayerNorm on the input (FP32 → FP16)
    # ------------------------------------------------------------------
    # The built‑in LayerNorm is heavily tuned for the GPU.
    # It works on the last dimension, exactly what we need.
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps).half()        # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (5·H × dim) into a single GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    # flatten to (R, dim) -> (R, 5·H)
    x_norm_flat = x_norm.view(R, dim)
    proj = F.linear(x_norm_flat, fused_w)              # [R, 5·H] FP16
    del fused_w, x_norm, x_norm_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates  → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_fp16
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # rows per program
    BLOCK_H = 64   # hidden channels per program
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM) – left @ rightᵀ
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N]  (contiguous memory layout)
    left_mat  = left.view(B * H, N, N)               # (B·H, N, N)
    right_mat = right.view(B * H, N, N)              # (B·H, N, N)

    # rightᵀ : (B·H, N, N) → (B·H, N, N) with swapped last two dims
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                              # (B, H, N, N) FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (per (i,j) pair)
    # ------------------------------------------------------------------
    # PyTorch LayerNorm expects the normalized dimension to be the *last* one,
    # so we permute to put H at the tail, apply LN, then permute back.
    out_perm      = out.permute(0, 2, 3, 1)            # (B, N, N, H)  FP16
    out_gate_perm = out_gate.permute(0, 2, 3, 1)      # (B, N, N, H)  FP16

    # cast LayerNorm parameters to FP16 to match the input dtype
    ln_w_h = w["to_out_norm.weight"].to(torch.float16)
    ln_b_h = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(out_perm,
                            normalized_shape=(H,),
                            weight=ln_w_h,
                            bias=ln_b_h,
                            eps=eps)                     # (B, N, N, H) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply the output gate and project back to `dim`
    # ------------------------------------------------------------------
    out_norm_gate = out_norm * out_gate_perm                     # (B, N, N, H) FP16

    out_norm_gate_flat = out_norm_gate.view(R, H)                # (R, H) FP16
    w_out_h = w["to_out.weight"].to(torch.float16)              # (dim, H)

    out_proj = F.linear(out_norm_gate_flat, w_out_h)             # (R, dim) FP16
    final = out_proj.view(B, N, N, dim).float()                  # (B, N, N, dim) FP32

    return final
```
