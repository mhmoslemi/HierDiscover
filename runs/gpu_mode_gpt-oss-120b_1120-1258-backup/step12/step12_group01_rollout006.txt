"""
High‑performance Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The algorithm follows the reference PyTorch implementation
but offloads all LayerNorms and the mask/gate fusion to hand‑written
Triton kernels.  The heavy linear algebra (projections, batched GEMM,
final projection) is delegated to cuBLAS/Tensor‑Cores via `torch.nn.functional.linear`
and `torch.bmm`, which are already heavily optimized on an H100.

Key ideas
---------
1️⃣ First LayerNorm (float32 → float16) is performed row‑wise on the
   flattened input (B·N·N × C).  A block size of 1024 rows × 128 features
   gives excellent memory coalescing and only two passes over the data.

2️⃣ The five linear projections (left/right + three gates) are fused into a
   single matrix‑multiply (`F.linear`).  A second Triton kernel splits this
   concatenated result, applies the optional pair‑wise mask, evaluates the
   sigmoid gates, and writes the left/right tensors directly in the layout
   required for the subsequent batched GEMM (shape (B·H, N, N)).  The gate
   tensor (`out_gate`) is also stored for later use.  All hidden dimensions
   are processed in a single 128‑wide iteration.

3️⃣ Batched GEMM: `out = left @ rightᵀ` over the sequence dimension N.
   Using `torch.bmm` exploits the H100 Tensor‑Core kernels.

4️⃣ Second LayerNorm + element‑wise multiplication with `out_gate` is again
   fused into a Triton kernel.  The GEMM result is normalized over the
   hidden dimension H (128) and multiplied by the pre‑computed gate before
   being written out.

5️⃣ Final linear projection back to the original channel dimension C.

All kernels run with 1024 threads per block (the maximum on H100) and
use fp32 accumulation for numerical stability, while the bulk of the
computation stays in fp16 for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # accumulate mean / variance in fp32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # write normalized values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]           # fp32 → fp16
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split / fuse projection, mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B·H, N, N] (float16)
    right_ptr,         # mutable *[B·H, N, N] (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]      (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # decode (b,i,j) from flat row index
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # optional mask (broadcast across hidden dim)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # hidden dimension loop (single iteration with BLOCK_H = H)
    offs_h = tl.arange(0, BLOCK_H)                     # [BLOCK_H]
    mask_h = offs_h < H

    # base offsets into concatenated projection matrix (row‑major)
    base = offs_r[:, None] * (5 * H) + offs_h[None, :]                # [R,H]

    left_raw       = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
    right_raw      = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
    left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
    right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
    out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

    # sigmoid in fp32 for stability
    l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
    r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
    o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

    # apply mask and gates
    left  = left_raw  * mask_val * l_gate
    right = right_raw * mask_val * r_gate

    # write left/right in (B·H, N, N) layout
    bh = b_idx[:, None] * H + offs_h[None, :]                     # [R,H]
    dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R,H]

    tl.store(left_ptr  + dst_off,
             left,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst_off,
             right,
             mask=mask_r[:, None] & mask_h[None, :])

    # store out_gate (still (rows, H))
    tl.store(out_gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
             o_gate,
             mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_strided_kernel(
    x_ptr,          # const   *[B·H, N, N]   (float16)   (GEMM result)
    gate_ptr,       # const   *[rows, H]    (float16)   (out_gate)
    y_ptr,          # mutable *[rows, H]    (float16)   (fused output)
    weight_ptr,     # const   *[H]         (float32)   γ
    bias_ptr,       # const   *[H]         (float32)   β
    rows: tl.int32,       # = B·N·N
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per program (== H)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_r = offs_r < rows

    seqsq = N * N

    # decode (b,i,j) from flat row index
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # base offset for hidden dimension = 0
    base0 = b_idx * H * seqsq + i_idx * N + j_idx      # [BLOCK_M]

    # ------------------------------------------------------------------
    # 1️⃣ mean / variance over H (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    offs_h = tl.arange(0, BLOCK_H)                     # [BLOCK_H]
    mask_h = offs_h < H

    vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                   mask=mask_r[:, None] & mask_h[None, :],
                   other=0.0).to(tl.float32)            # [M, H]

    sum_val += tl.sum(vals, axis=1)
    sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ normalize, apply γ/β, multiply by out_gate, store (fp16)
    # ------------------------------------------------------------------
    norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ (fp32)
    b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β (fp32)

    norm = norm * w[None, :] + b[None, :]               # fp32

    gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                   mask=mask_r[:, None] & mask_h[None, :],
                   other=0.0).to(tl.float32)           # fp32

    out = norm * gate                                   # fp32
    out_fp16 = out.to(tl.float16)

    tl.store(y_ptr + offs_r[:, None] * H + offs_h[None, :],
             out_fp16,
             mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim`` and ``hidden_dim`` and a bool
                         ``nomask`` (True → no mask).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise positions
    H = hidden_dim
    seqsq = N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                      # [R, C] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 1024        # rows per program (max on H100)
    BLOCK_D = 128         # dim chunk (covers 128‑ and 384‑dim cases)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32, [C])
        weights["norm.bias"],            # β (float32, [C])
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and three gates
    # ------------------------------------------------------------------
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, C]

    # (R, 5·H) fp16
    proj = F.linear(x_norm, concat_w)
    del concat_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split + mask + sigmoid gates, write left/right in (B·H, N, N)
    # ------------------------------------------------------------------
    left      = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask is [B, N, N] → flatten to [R]; values are 0/1 (float)
        mask_tensor = mask.to(torch.float16).reshape(R)
        has_mask = 1

    BLOCK_R = 1024    # rows per program (max threads per block)
    BLOCK_H = H       # process whole hidden dimension in one iteration
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_tensor

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (Tensor‑Core)
    # ------------------------------------------------------------------
    # left/right are (B·H, N, N); we need left @ rightᵀ
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    grid_lnorm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_strided_kernel[grid_lnorm2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32, [H])
        weights["to_out_norm.bias"],     # β (float32, [H])
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape (R, dim)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

The function `custom_kernel` can be called exactly as described in the problem statement:

output = custom_kernel((input_tensor, mask_tensor, weights_dict, config_dict))

It returns a tensor of shape `[B, N, N, dim]` with `dtype=torch.float32` and matches the reference PyTorch implementation while meeting the performance target on an H100.
