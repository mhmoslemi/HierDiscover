TriMul (outgoing) – highly‑fused Triton implementation
=====================================================

Algorithm (B = batch, N = sequence length, C = dim, H = hidden_dim)

1️⃣ Row‑wise LayerNorm on the 4‑D pair representation (B·N·N × C).  
   The kernel works in fp32 for the statistics, writes the normalized
   tensor directly in fp16.

2️⃣ One fused linear projection:
      X_norm @ W_fused.T  →  (B·N·N, 5·H)

   (max 640 × 384) and cuBLAS is already optimal for this GEMM.

3️⃣ Split the fused projection, apply the optional pair‑mask, evaluate the
   three sigmoid gates and write the three tensors *directly* in the

   temporaries.

4️⃣ Batched half‑precision GEMM (Tensor‑Core) over the hidden dim:
        out[b,i,j] = Σ_k left[b,i,k] · right[b,j,k]

    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # dim chunk size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- 1️⃣ Compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- 2️⃣ Write normalized values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#    write directly in (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) – ignored if HAS_MASK=0
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode (b,i,k) from flat row index ---------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask (if any)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # → [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between consecutive h

    # base offset (b·H·N·N + i·N + k) – common to all hidden slices of a row
    base_offset = b * (H * N * N) + i * N + k          # [BLOCK_R]

    # ---- Each program processes the whole hidden dim in one go (BLOCK_H == H) ----
    offs_h = tl.arange(0, BLOCK_H)                     # [BLOCK_H]
    mask_h = offs_h < H

    # offsets inside the fused projection matrix [R,5*H]
    proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

    # load the five blocks (left, right, left_gate, right_gate, out_gate)
    left_raw      = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
    right_raw     = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
    left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
    right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
    out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

    # sigmoid (stable fp32) – convert once
    left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
    out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

    # apply mask and gates
    left  = left_raw  * mask_val * left_gate
    right = right_raw * mask_val * right_gate

    # destination offsets for each hidden slice
    # dst = base_offset + h * hidden_stride
    dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

    # store results
    tl.store(left_ptr      + dst_offset, left,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr     + dst_offset, right,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr  + dst_offset, out_gate,
             mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, out_gate multiplication and final linear
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_fused_final_kernel(
    out_ptr,            # const   *[B, H, N, N] (fp16)
    gate_ptr,           # const   *[B, H, N, N] (fp16)
    to_out_norm_w_ptr,  # const   *[H]          (float32) γ
    to_out_norm_b_ptr,  # const   *[H]          (float32) β
    to_out_w_ptr,       # const   *[dim, H]      (float16)
    out_final_ptr,      # mutable *[B, N, N, dim] (float16)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # base offset for the hidden dimension (h = 0)
    base_offset = b * (H * nn) + i * N + j                # [BLOCK_R]

    # --------------------------------------------------------------
    # 1️⃣ Compute mean & variance over H (first pass)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * nn,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Second pass: normalize, apply γ/β, multiply by out_gate,
    #    and immediately accumulate the final linear projection.
    #    We accumulate over H for every output channel `d`.
    # --------------------------------------------------------------
    # accumulator for the final projection (fp32)
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)   # [row, dim]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # load tensors
        out_vals  = tl.load(out_ptr  + base_offset[:, None] + offs_h[None, :] * nn,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)          # [R, H_block]
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_h[None, :] * nn,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)          # [R, H_block]

        # γ/β for the hidden dim
        gamma = tl.load(to_out_norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # [H_block]
        beta  = tl.load(to_out_norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # [H_block]

        # normalize hidden slice
        norm = (out_vals - mean[:, None]) * rstd[:, None]       # fp32
        norm = norm * gamma[None, :] + beta[None, :]            # fp32

        # apply out‑gate
        norm = norm * gate_vals                                 # fp32

        # ---- final linear projection for this hidden slice -----------------
        # weight matrix: [dim, H] → we need row‑major for the dim axis
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            w = tl.load(to_out_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                        mask=mask_d[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)               # [BLOCK_D, H_block]

            # accumulate dim‑wise dot‑product
            # acc[row, d] += Σ_h norm[row, h] * w[d, h]
            acc[:, d_off * BLOCK_D : d_off * BLOCK_D + BLOCK_D] += tl.dot(norm, w)  # (R, BLOCK_D)

    # --------------------------------------------------------------
    # 3️⃣ Store final result (fp16) in (B, N, N, dim) layout
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # compute linear offset for the final tensor:
        # final layout: (B, N, N, dim) → stride = (N*N*dim, N*dim, dim, 1)
        final_offset = b * (N * N * dim) + i * (N * dim) + j * dim + offs_d   # [BLOCK_R, BLOCK_D]

        out_fp16 = acc[:, d_off * BLOCK_D : d_off * BLOCK_D + BLOCK_D].to(tl.float16)

        tl.store(out_final_ptr + final_offset,
                 out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (float32 output).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, [B, N, N, C]  (float32)
            - mask_tensor  : torch.Tensor, [B, N, N] (bool/float) or None
            - weights_dict : model weights (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]   # H (always 128 in the provided tests)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],      # γ (float32)
        w["norm.bias"],        # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=3,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (Linear)
    # --------------------------------------------------------------
    # Stack the five weight matrices (no bias) – shape [5*H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # (5*H, dim)

    proj = F.linear(x_norm, fused_w)                  # (R, 5*H) fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates (writes into (B, H, N, N))
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 256
    BLOCK_H = H    # process the whole hidden dimension in one go
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=3,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (Tensor‑Core)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) – contiguous for cuBLAS
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]
    out = out_mat.view(B, H, N, N)                             # (B, H, N, N)

    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate multiplication and final linear
    # --------------------------------------------------------------
    final = torch.empty(B, N, N, dim, dtype=torch.float16, device=device)

    BLOCK_R2 = 256
    BLOCK_H2 = H
    BLOCK_D2 = 128    # handle dim up to 384 in ≤ 3 steps
    grid_final = (triton.cdiv(R, BLOCK_R2),)

    layernorm_gate_fused_final_kernel[grid_final](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w["to_out.weight"].to(torch.float16),   # [dim, H] fp16
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
        num_stages=3,
    )

    del out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Cast back to float32 and return
    # --------------------------------------------------------------
    return final.to(torch.float32)
