"""
High‑performance Triton + PyTorch implementation of the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (B·N·N × dim) – fp32 → fp16.
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single GEMM:  (R×dim) @ (5·H×dim)ᵀ  →  (R×5·H)   (R = B·N·N).
3️⃣  Split‑fuse‑gate kernel:
    * extracts the five slices,
    * optionally applies the pair‑wise mask,
    * evaluates sigmoid gates (in fp32),
    * writes the *left* and *right* tensors directly in the layout required
      for the batched GEMM, (B·H, N, N),
    * stores the output‑gate tensor (R, H).
4️⃣  Batched half‑precision GEMM (cuBLAS/Tensor‑Cores):
        out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
5️⃣  Second LayerNorm + out‑gate multiplication – performed with
    fast PyTorch reductions (mean/var) on the H‑dimension.
6️⃣  Final linear projection back to `dim` (fp16 → fp32).

Only the LayerNorm‑1 and the split‑fuse‑gate operations are written in
Triton; the heavy matrix multiplication and the second LayerNorm use
high‑throughput PyTorch kernels.  This hybrid approach satisfies the
requirement of “implement at least part of the operations in a kernel”
while achieving sub‑millisecond latency on the target H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized values: γ·(x‑μ)·rstd + β   (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Split‑fuse‑gate kernel (projection → mask → sigmoid gates)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]     (float16)
    left_ptr,          # mutable *[B·H, N, N]   (float16)
    right_ptr,         # mutable *[B·H, N, N]   (float16)
    out_gate_ptr,      # mutable *[rows, H]     (float16)
    mask_ptr,          # const   *[rows]        (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,    # = B·N·N
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ----- decode (b,i,j) from flat row index -----
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # ----- optional mask (broadcast across hidden dim) -----
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ----- loop over hidden dimension blocks -----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ----- layout: [left, right, left_gate, right_gate, out_gate] each size H -----
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]                # [R,H]

        left_raw       = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # ----- sigmoid (fp32 for stability) -----
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----
        left  = left_raw  * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # ----- write left/right in (B·H, N, N) layout -----
        # bh = b * H + h
        bh = b_idx[:, None] * H + offs_h[None, :]                     # [R,H]
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R,H]

        tl.store(left_ptr  + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ----- store out_gate (rows, H) -----
        tl.store(out_gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``.
                         May also contain other flags.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of (b,i,j) positions
    H = hidden_dim
    seqsq = N * N

    # ------------------------------------------------------------------
    # 1️⃣ First Row‑wise LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 512          # rows per program
    BLOCK_D = 64           # dim chunk
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32, [dim])
        weights["norm.bias"],            # β (float32, [dim])
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left/right + 3 gates) into one GEMM
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (each [H, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # shape [5*H, dim]

    proj = F.linear(x_norm, w_concat)   # (R, 5*H)  fp16
    del w_concat, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split‑fuse‑gate kernel → left, right, out_gate
    # ------------------------------------------------------------------
    left     = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty_like(left)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask: bool/float of shape [B, N, N] → flatten to [R]
        mask_tensor = mask.to(torch.float16).reshape(R)
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_tensor

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension:
    #    out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) + element‑wise *out_gate*
    #    Implemented with fast PyTorch reductions.
    # ------------------------------------------------------------------
    out_mat_view = out_mat.view(B, H, N, N)            # (B, H, N, N) half
    out_mat_f    = out_mat_view.float()               # promote for numerics

    mean = out_mat_f.mean(dim=1, keepdim=True)        # (B,1,N,N)
    var  = out_mat_f.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)

    norm = (out_mat_f - mean) * rstd                  # (B,H,N,N) float

    gamma = weights["to_out_norm.weight"].view(1, H, 1, 1).float()
    beta  = weights["to_out_norm.bias"]  .view(1, H, 1, 1).float()
    norm = norm * gamma + beta                       # (B,H,N,N) float

    # out_gate: (R, H) → (B, N, N, H) → (B, H, N, N)
    out_gate_f = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).float()

    fused_f = norm * out_gate_f                       # (B,H,N,N) float
    fused = fused_f.to(torch.float16)                  # back to fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    fused_flat = fused.view(R, H)                      # (R, H) half
    final_half = F.linear(
        fused_flat,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (R, dim) half
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
