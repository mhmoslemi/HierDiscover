Let's outline the kernel step by step:

```

row_start = pid * BLOCK_R
offs_r = row_start + tl.arange(0, BLOCK_R)
mask_r = offs_r < rows

# decode batch, i, j
seqsq = N * N
b_idx = offs_r // seqsq
rem = offs_r - b_idx * seqsq
i_idx = rem // N
j_idx = rem - i_idx * N

# Compute mean/var
sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)
for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
    offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim
    ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
    vals = tl.load(ptr, mask=mask_r[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
    sum_val += tl.sum(vals, axis=1)
    sum_sq  += tl.sum(vals * vals, axis=1)
mean = sum_val / dim
var = sum_sq / dim - mean * mean
rstd = tl.rsqrt(var + eps)

# if mask needed: load mask once now for rows
if HAS_MASK:
    mask_val_row = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # shape [BLOCK_R]
    mask_val = mask_val_row[:, None]   # broadcast
else:
    mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

# Hidden tile loop
for h_off in range(0, tl.cdiv(H, BLOCK_H)):
    offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H
    # accumulators
    left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_acc = tl.zeros(...)
    left_gate_acc = tl.zeros(...)
    right_gate_acc = tl.zeros(...)
    out_gate_acc = tl.zeros(...)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load x block
        x_vals = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                         mask=mask_r[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        # normalize and apply gamma/beta
        ln_w = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0)
        ln_b = tl.load(norm_bias_ptr + offs_d, mask=mask_d, other=0.0)
        normed = ((x_vals - mean[:, None]) * rstd[:, None]) * ln_w[None, :] + ln_b[None, :]  # fp32

        # load weight slices for each projection
        left_w = tl.load(left_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                         mask=mask_h[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        right_w = tl.load(right_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :],
                          other=0.0).to(tl.float32)
        left_gate_w = tl.load(left_gate_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
        right_gate_w = tl.load(right_gate_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :],
                               other=0.0).to(tl.float32)
        out_gate_w = tl.load(out_gate_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)

        # accumulate
        left_acc += tl.dot(normed, left_w)
        right_acc += tl.dot(normed, right_w)
        left_gate_acc += tl.dot(normed, left_gate_w)
        right_gate_acc += tl.dot(normed, right_gate_w)
        out_gate_acc += tl.dot(normed, out_gate_w)
    # after dim loop: compute gates
    l_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)  # shape (BLOCK_R, BLOCK_H)
    r_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
    o_gate = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

    # cast left/right accumulators to fp16
    left_fp = left_acc.to(tl.float16)
    right_fp = right_acc.to(tl.float16)

    # Apply mask and gates
    left_out = left_fp * mask_val * l_gate
    right_out = right_fp * mask_val * r_gate

    # Store left/right
    bh = b_idx[:, None] * H + (h_off * BLOCK_H) + offs_h[None, :]  # shape (BLOCK_R, BLOCK_H)
    dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]
    tl.store(left_ptr + dst_off, left_out,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst_off, right_out,
             mask=mask_r[:, None] & mask_h[None, :])

    # Store out_gate
    gate_off = offs_r[:, None] * H + (h_off * BLOCK_H) + offs_h[None, :]
    tl.store(out_gate_ptr + gate_off, o_gate,
             mask=mask_r[:, None] & mask_h[None, :])

"""
High‑performance Triton implementation of the **outgoing** TriMul operator
(AlphaFold‑3).

Algorithm (merged kernels)
-------------------------
1️⃣  Row‑wise LayerNorm is fused with the 5 linear projections (left/right + 3
    gates).  For every position (b,i,j) we
    * compute mean / variance of the raw input vector (float32)
    * apply the LayerNorm affine transform (γ,β)
    * perform 5 matrix‑vector products against the transposed projection
      matrices (dim × H) → (H) using a tiny batched GEMM inside Triton.
    * materialise the *left* and *right* tensors and the *out‑gate* after
      applying the sigmoid gates and an optional pair‑wise mask.
    This replaces the separate LayerNorm + `F.linear` + split‑gate kernels.

2️⃣  Batched half‑precision GEMM (Tensor‑Core cuBLAS) on the hidden dimension:
        out(b,h,i,j) = Σₖ left(b,h,i,k)·right(b,h,j,k)

3️⃣  Second LayerNorm + element‑wise multiplication with the stored out‑gate.
    (still a Triton kernel.)

4️⃣  Final linear projection back to the original `dim` (FP16 GEMM via
    `torch.nn.functional.linear`).  The result is cast to float32.

All heavy linear / GEMM steps use fp16 Tensor‑Cores; the two LayerNorms and the
fused projection are hand‑written in Triton.  Block sizes have been tuned for
the H100 (8 warps, 64‑bit math).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Fused LayerNorm + 5 projections (left/right + 3 gates)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_proj_gate_kernel(
    x_ptr,                 # const   *[rows, dim]          (float32)
    left_ptr,              # mutable *[B·H, N, N]         (float16)
    right_ptr,             # mutable *[B·H, N, N]         (float16)
    out_gate_ptr,          # mutable *[rows, H]          (float16)
    mask_ptr,              # const   *[rows]               (float16) – ignored if HAS_MASK==0
    left_w_ptr,            # const   *[dim, H]             (float16)
    right_w_ptr,           # const   *[dim, H]             (float16)
    left_gate_w_ptr,       # const   *[dim, H]             (float16)
    right_gate_w_ptr,      # const   *[dim, H]             (float16)
    out_gate_w_ptr,        # const   *[dim, H]             (float16)
    norm_w_ptr,            # const   *[dim]                (float32) – γ
    norm_b_ptr,            # const   *[dim]                (float32) – β
    rows: tl.int32,        # = B·N·N
    H: tl.int32,
    dim: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim tile
    BLOCK_H: tl.constexpr,   # hidden dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # decode (b,i,j) from flat row index
    # ------------------------------------------------------------------
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # ------------------------------------------------------------------
    # 1️⃣ mean & variance (fp32) over the dim dimension
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]  # [R, D]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # optional pair‑wise mask (broadcast across hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                    # [BLOCK_R,1] fp16
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators for this tile (fp32)
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        #   loop over the dim axis and accumulate GEMM contributions
        # ------------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
            mask_d = offs_d < dim

            # ----- load raw input and apply LayerNorm + γ/β -----
            x_vals = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)               # [R, D]

            gamma = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)   # [D] fp32
            beta  = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)   # [D] fp32

            # (x - μ)·rstd * γ + β
            x_norm = ((x_vals - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]  # fp32

            # ----- load weight slices (transposed: [dim, H]) -----
            # shape (BLOCK_D, BLOCK_H)
            w_left = tl.load(left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            w_right = tl.load(right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_lgate = tl.load(left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_rgate = tl.load(right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_outgate = tl.load(out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                                mask=mask_d[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # ----- accumulate 5 GEMMs (R×D)·(D×H) -----
            left_acc      += tl.dot(x_norm, w_left)       # (R,Htile)
            right_acc     += tl.dot(x_norm, w_right)
            left_gate_acc += tl.dot(x_norm, w_lgate)
            right_gate_acc+= tl.dot(x_norm, w_rgate)
            out_gate_acc  += tl.dot(x_norm, w_outgate)

        # ------------------------------------------------------------------
        # 3️⃣ Apply sigmoid to gates, mask, and store results
        # ------------------------------------------------------------------
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        left_fp  = left_acc .to(tl.float16)
        right_fp = right_acc.to(tl.float16)

        left_out  = left_fp  * mask_val * l_gate
        right_out = right_fp * mask_val * r_gate

        # ---- destination offsets in (B·H, N, N) layout ----
        bh = b_idx[:, None] * H + (h_off * BLOCK_H) + offs_h[None, :]   # (R, Htile)
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]     # (R, Htile)

        tl.store(left_ptr  + dst_off, left_out,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right_out,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ---- store out_gate (R, H) ----
        gate_off = offs_r[:, None] * H + (h_off * BLOCK_H) + offs_h[None, :]
        tl.store(out_gate_ptr + gate_off, o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_strided_kernel(
    x_ptr,           # const   *[B·H, N, N]   (float16) – GEMM result
    gate_ptr,        # const   *[rows, H]    (float16) – out_gate
    y_ptr,           # mutable *[rows, H]    (float16) – fused output
    weight_ptr,      # const   *[H]         (float32) – γ
    bias_ptr,        # const   *[H]         (float32) – β
    rows: tl.int32,          # = B·N·N
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_r = offs_r < rows

    seqsq = N * N

    # decode (b,i,j) from row index
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # base offset for hidden‑slice h=0 in GEMM output
    base0 = b_idx * H * seqsq + i_idx * N + j_idx      # [BLOCK_M]

    # --------------------------------------------------------------
    # 1️⃣ compute mean / variance over H (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)           # [M, Hblk]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalize, apply γ/β, multiply by out_gate, store (fp16)
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)           # [M, Hblk]

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β fp32
        norm = norm * w[None, :] + b[None, :]                # fp32

        gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)           # fp32

        out = norm * gate                                   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # flatten input and mask
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()        # [R, dim] fp32

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()  # [R]

    # ------------------------------------------------------------------
    # allocate outputs of the first fused kernel
    # ------------------------------------------------------------------
    left      = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # prepare weight tensors (transpose projection weights for easier GEMM)
    # ------------------------------------------------------------------
    def _to_half_t(w):
        return w.t().contiguous().to(torch.float16)

    left_w      = _to_half_t(weights["left_proj.weight"])    # [dim, H]
    right_w     = _to_half_t(weights["right_proj.weight"])
    left_gate_w = _to_half_t(weights["left_gate.weight"])
    right_gate_w= _to_half_t(weights["right_gate.weight"])
    out_gate_w  = _to_half_t(weights["out_gate.weight"])

    norm_w = weights["norm.weight"].to(torch.float32).contiguous()
    norm_b = weights["norm.bias"  ].to(torch.float32).contiguous()

    # ------------------------------------------------------------------
    # launch fused LayerNorm + projection + gate kernel
    # ------------------------------------------------------------------
    BLOCK_R = 256   # rows per program
    BLOCK_D = 64    # dim tile
    BLOCK_H = 64    # hidden tile (H=128 ⇒ two tiles)
    grid = (triton.cdiv(R, BLOCK_R),)

    layernorm_proj_gate_kernel[grid](
        x_flat,
        left,
        right,
        out_gate,
        mask_flat,
        left_w,
        right_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        norm_w,
        norm_b,
        R,
        H,
        dim,
        N,
        eps,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # free intermediate tensors
    del x_flat, mask_flat

    # ------------------------------------------------------------------
    # 2️⃣ Batched GEMM over the hidden dimension (Tensor‑Core)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 3️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_R),)   # reuse BLOCK_R
    layernorm_mul_strided_kernel[grid2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ  fp32
        weights["to_out_norm.bias"],     # β  fp32
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_R,
        BLOCK_H=BLOCK_D,                 # reuse BLOCK_D as hidden‑tile size
        num_warps=8,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 4️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),    # (dim, hidden_dim)
    )   # fp16
    final = final.view(B, N, N, dim).to(torch.float32)
    return final
