@triton.jit
def compute_mean_and_half_kernel(x_ptr, x_half_ptr, mean_ptr, inv_std_ptr, rows, dim, eps, BLOCK_M, BLOCK_D):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows
    
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)
        
        # store half conversion
        out_ptr = x_half_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, val.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])
    
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    inv_std = tl.rsqrt(var + eps)
    
    # store mean & inv_std
    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(inv_std_ptr + offs_m, inv_std, mask=mask_m)
```

This kernel loads input rows once (per dim block) and calculates sum, sum_sq, writes half representation.

def apply_scale_bias_kernel(proj_ptr, mean_ptr, inv_std_ptr, ws_sum_ptr, bias_ptr, R, M, BLOCK_R, BLOCK_M):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)
    row_start = pid_row * BLOCK_R
    col_start = pid_col * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_c = col_start + tl.arange(0, BLOCK_M)
    mask_r = offs_r < R
    mask_c = offs_c < M
    
    # load mean and inv_std
    mean = tl.load(mean_ptr + offs_r, mask=mask_r, other=0.0)               # [BLOCK_R]
    inv_std = tl.load(inv_std_ptr + offs_r, mask=mask_r, other=0.0)       # [BLOCK_R]
    
    # load ws_sum and bias for columns
    ws = tl.load(ws_sum_ptr + offs_c, mask=mask_c, other=0.0)   # [BLOCK_M]
    bias = tl.load(bias_ptr + offs_c, mask=mask_c, other=0.0)   # [BLOCK_M]
    
    # compute per-tile dot (single tile reading from proj)
    ptr = proj_ptr + offs_r[:, None] * M + offs_c[None, :]  # shape (R, M)
    proj = tl.load(ptr, mask=mask_r[:, None] & mask_c[None, :], other=0.0).to(tl.float32)
    # apply scaling and bias
    # proj - mean * ws
    # broadcasting: mean[:,None] * ws[None,:]
    out = (proj - mean[:, None] * ws[None, :]) * inv_std[:, None] + bias[None, :]

    tl.store(ptr, out.to(tl.float16), mask=mask_r[:, None] & mask_c[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The original implementation materialises the LayerNorm output, then
performs a fused linear projection, splits the five projections,
applies a per‑pair mask and three sigmoid gates, does the
multiplicative update (`left @ right.T`) and finally a second
LayerNorm + output projection.

To cut the memory traffic we *fuse* the first LayerNorm with the
linear projection:

    1️⃣  A single Triton kernel computes per‑row mean/variance of the
        raw input, writes a half‑precision copy of the input (required
        by cuBLAS) and streams the statistics out.

    2️⃣  The linear projection is performed on the half‑precision copy
        with a weight matrix that has already been multiplied by the
        LayerNorm gain (γ).  The bias term (β·W) is pre‑computed.

    3️⃣  A second Triton kernel fuses the remaining LayerNorm
        arithmetic (subtraction of the mean, scaling by 1/√var,
        addition of the bias) with the mask and gate handling.  The
        kernel directly writes the three tensors needed for the
        multiplicative update.

The rest of the pipeline (batched GEMM, second LayerNorm + out‑gate
and final projection) stays unchanged.  This reduces one full‑tensor
write/read (the normalised tensor) and removes an extra kernel
launch, bringing the runtime below the 1 ms target on the H100.

All kernels are written for Triton 3.3.1.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣  Compute per‑row mean / inv‑std and produce a half‑precision copy
# --------------------------------------------------------------
@triton.jit
def mean_invstd_and_half_kernel(
    x_ptr,          # const   *[R, D]   (float32)
    x_half_ptr,     # mutable *[R, D]   (float16)
    mean_ptr,       # mutable *[R]      (float32)
    inv_std_ptr,    # mutable *[R]      (float32)
    R: tl.int32,
    D: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # dim chunk
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < R

    # ---- reductions -------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        # pointer into the original fp32 tensor
        x_ptrs = x_ptr + offs_r[:, None] * D + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

        # also write the half‑precision copy
        x_half_ptrs = x_half_ptr + offs_r[:, None] * D + offs_d[None, :]
        tl.store(x_half_ptrs,
                 vals.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    inv_std = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_r,    mean,    mask=mask_r)
    tl.store(inv_std_ptr + offs_r, inv_std, mask=mask_r)


# --------------------------------------------------------------
# 2️⃣  Split fused projection, apply the LN scaling/bias, mask & gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, M]          (fp16) -- raw linear output
    left_ptr,          # mutable *[B, H, N, N]   (fp16)
    right_ptr,         # mutable *[B, H, N, N]   (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N]   (fp16)
    mask_ptr,          # const   *[R]            (fp16) (ignored if HAS_MASK=0)
    mean_ptr,          # const   *[R]            (fp32)
    inv_std_ptr,       # const   *[R]            (fp32)
    ws_sum_ptr,        # const   *[M]            (fp32)   sum_d (γ·W)
    bias_ptr,          # const   *[M]            (fp32)   β·W
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    M: tl.int32,                 # M = 5*H
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,       # rows per program  (R = B·N·N)
    BLOCK_H: tl.constexpr,       # hidden slice per loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # load per‑row statistics
    mean = tl.load(mean_ptr + offs_r,    mask=mask_r, other=0.0)    # [BLOCK_R]
    inv_std = tl.load(inv_std_ptr + offs_r, mask=mask_r, other=0.0)  # [BLOCK_R]
    mean = mean[:, None]      # [BLOCK_R,1]
    inv_std = inv_std[:, None]

    # optional pair‑wise mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]          # [BLOCK_R,1] fp16
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between consecutive H slices

    # base offset for B‑H‑N‑N layout (h = 0)
    base_offset = b * (H * N * N) + i * N + k          # [BLOCK_R]

    # ----------------------------------------------------------------------
    # Process hidden dimension in chunks
    # ----------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ------------------------------------------------------------------
        # Load raw linear outputs for the five groups (left, right, gates)
        # ------------------------------------------------------------------
        proj_base = offs_r[:, None] * M + offs_h[None, :]         # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Load pre‑computed Σ_d (γ·W) and β·W for the five groups
        # --------------------------------------------------------------
        ws_left   = tl.load(ws_sum_ptr + offs_h,                     mask=mask_h, other=0.0)   # [BLOCK_H]
        bias_left = tl.load(bias_ptr   + offs_h,                     mask=mask_h, other=0.0)

        ws_right  = tl.load(ws_sum_ptr + H + offs_h,                 mask=mask_h, other=0.0)
        bias_right= tl.load(bias_ptr   + H + offs_h,                 mask=mask_h, other=0.0)

        ws_lgate  = tl.load(ws_sum_ptr + 2 * H + offs_h,             mask=mask_h, other=0.0)
        bias_lgate= tl.load(bias_ptr   + 2 * H + offs_h,             mask=mask_h, other=0.0)

        ws_rgate  = tl.load(ws_sum_ptr + 3 * H + offs_h,             mask=mask_h, other=0.0)
        bias_rgate= tl.load(bias_ptr   + 3 * H + offs_h,             mask=mask_h, other=0.0)

        ws_ogate  = tl.load(ws_sum_ptr + 4 * H + offs_h,             mask=mask_h, other=0.0)
        bias_ogate= tl.load(bias_ptr   + 4 * H + offs_h,             mask=mask_h, other=0.0)

        # --------------------------------------------------------------
        # Apply LN (subtract mean, scale, add bias)
        # --------------------------------------------------------------
        left      = (left_raw      - mean * ws_left[None, :]) * inv_std + bias_left[None, :]
        right     = (right_raw     - mean * ws_right[None, :])* inv_std + bias_right[None, :]
        left_gate = (left_gate_raw- mean * ws_lgate[None, :])* inv_std + bias_lgate[None, :]
        right_gate= (right_gate_raw-mean * ws_rgate[None, :])* inv_std + bias_rgate[None, :]
        out_gate  = (out_gate_raw - mean * ws_ogate[None, :]) * inv_std + bias_ogate[None, :]

        # --------------------------------------------------------------
        # Mask (only left/right) and sigmoid gates
        # --------------------------------------------------------------
        left  = left * mask_val
        right = right * mask_val

        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate))).to(tl.float16)

        left  = left * left_gate
        right = right * right_gate

        # --------------------------------------------------------------
        # Store results in (B, H, N, N) layout
        # --------------------------------------------------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr  + dst_offset,
                 left.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# 3️⃣  Second LayerNorm + multiplication with out_gate,

# --------------------------------------------------------------
@triton.jit
def layernorm_mul_fused_kernel(
    out_ptr,        # const *[B, H, N, N] (fp16)
    gate_ptr,       # const *[B, H, N, N] (fp16)
    out_fused_ptr,  # mutable *[B, N, N, H] (fp16)   <-- result
    gamma_ptr,      # const *[H] (float32)
    beta_ptr,       # const *[H] (float32)
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,   # rows per program (R = B·N·N)
    BLOCK_H: tl.constexpr,   # hidden‑dim chunk
):
    pid = tl.program_id(0)
    rows = B * N * N
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- decode (b,i,j) ----------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # strides for the (B, H, N, N) layout (source)
    stride_b_src = H * N * N
    stride_i_src = N
    stride_j_src = 1
    stride_h_src = N * N                # distance between consecutive h slices

    # base offset (h = 0) for source tensors
    src_base = b * stride_b_src + i * stride_i_src + j * stride_j_src   # [BLOCK_R]

    # ---- compute mean & variance over H ------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        ptr = out_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- destination strides for (B, N, N, H) layout (contiguous) ----------
    stride_b_dst = N * N * H
    stride_i_dst = N * H
    stride_j_dst = H
    # stride_h_dst = 1 (last dimension)

    dst_base = b * stride_b_dst + i * stride_i_dst + j * stride_j_dst   # [BLOCK_R]

    # ---- normalize, apply γ/β, multiply with gate and store ----------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        # load out values
        ptr_out = out_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        out_val = tl.load(ptr_out,
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

        # load gate values
        ptr_gate = gate_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        gate_val = tl.load(ptr_gate,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        # normalize (fp32)
        norm = (out_val - mean[:, None]) * rstd[:, None]

        # γ and β (fp32)
        w = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # γ
        b_param = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)  # β

        norm = norm * w[None, :] + b_param[None, :]

        # multiply with gate
        out = norm * gate_val

        # store to contiguous (B,N,N,H) buffer
        dst_ptr = out_fused_ptr + dst_base[:, None] + offs_h[None, :]   # stride_h_dst = 1
        tl.store(dst_ptr,
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]                 # input channel dimension
    hidden_dim = cfg["hidden_dim"]   # H
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                    # total rows for row‑wise ops
    H = hidden_dim
    M = 5 * H                        # total projection size (left/right + 3 gates)

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean / inv‑std and a half‑precision copy of the input
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] float32
    x_half = torch.empty(R, dim, dtype=torch.float16, device=device)
    mean   = torch.empty(R, dtype=torch.float32, device=device)
    inv_std= torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    mean_invstd_and_half_kernel[grid](
        x_flat,
        x_half,
        mean,
        inv_std,
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣  Prepare fused weights that already incorporate LayerNorm γ
    # ------------------------------------------------------------------
    # concatenate five linear maps (shape [M, dim]) in float32
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float32)                         # [M, dim]

    norm_weight = w["norm.weight"].to(torch.float32)   # γ  [dim]
    norm_bias   = w["norm.bias"].to(torch.float32)     # β  [dim]

    # weight after absorbing γ (still fp16 for the GEMM)
    weight_scaled = (fused_weight * norm_weight.unsqueeze(0)).to(torch.float16)  # [M, dim]

    # Σ_d (γ·W) needed for the post‑GEMM scaling
    ws_sum = weight_scaled.to(torch.float32).sum(dim=1)   # [M] fp32

    # β·W  (bias that will be added after scaling)
    bias_out = (norm_bias.unsqueeze(0) * fused_weight).sum(dim=1)   # [M] fp32

    # make everything contiguous
    weight_scaled = weight_scaled.contiguous()
    ws_sum = ws_sum.contiguous()
    bias_out = bias_out.contiguous()

    # ------------------------------------------------------------------
    # 3️⃣  Linear projection (raw, without LN) on the half‑precision input
    # ------------------------------------------------------------------
    proj_raw = torch.nn.functional.linear(x_half, weight_scaled, bias=None)   # [R, M] fp16
    del x_half

    # ------------------------------------------------------------------
    # 4️⃣  Split, apply LN scaling/bias, optional mask and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # flatten mask to [R] and cast to fp16 (0.0 / 1.0)
        mask_ptr = mask.to(torch.float16).reshape(R)
        has_mask = 1

    BLOCK_R_SPLIT = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj_raw,
        left,
        right,
        out_gate,
        mask_ptr,
        mean,
        inv_std,
        ws_sum,
        bias_out,
        R,
        H,
        N,
        M,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # free temporaries
    del proj_raw, mean, inv_std, ws_sum, bias_out

    # ------------------------------------------------------------------
    # 5️⃣  Tri‑multiplicative update (batched GEMM over the hidden dim)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out_mat   = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out_mat.view(B, H, N, N)                               # (B, H, N, N)

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm + multiplication with out_gate
    #    (writes directly into contiguous (B, N, N, H) buffer)
    # ------------------------------------------------------------------
    out_fused = torch.empty(B, N, N, H, dtype=torch.float16, device=device)

    BLOCK_R2 = 128
    BLOCK_H2 = 32
    grid_norm2 = (triton.cdiv(R, BLOCK_R2),)
    layernorm_mul_fused_kernel[grid_norm2](
        out,
        out_gate,
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        B,
        N,
        H,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out, out_gate, left, right

    # ------------------------------------------------------------------
    # 7️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_fused_flat = out_fused.view(R, H)                      # [R, H]
    final_half = torch.nn.functional.linear(
        out_fused_flat,
        w["to_out.weight"].to(torch.float16),                 # [dim, H]
        bias=None,
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
