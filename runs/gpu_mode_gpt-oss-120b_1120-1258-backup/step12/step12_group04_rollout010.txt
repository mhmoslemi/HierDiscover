def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; eps = cfg.get("eps",1e-5)
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm (first)
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # float32

    # Convert to fp16 for projection
    x_norm_half = x_norm.to(torch.float16).flatten(0,2) # shape (R, dim)

    # 2. Fused linear projection (5*H)
    w_concat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)  # (5*H, dim)

    proj = torch.nn.functional.linear(x_norm_half, w_concat)  # (R,5*H) fp16
    # (We could optionally free x_norm and x_norm_half.)

    # 3. Split and apply mask+gates
    left = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16).reshape(R)  # flatten
        has_mask = 1

    # kernel launch
    BLOCK_R = 512
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # 4. Batched GEMM
    out_mat = torch.bmm(left, right.transpose(1,2))   # (B*H, N, N) fp16

    # 5. Second LayerNorm + gating
    # reshape for layer_norm
    out_mat_reshaped = out_mat.view(B, H, N, N).permute(0,2,3,1)   # (B,N,N,H)
    out_mat_norm = torch.nn.functional.layer_norm(
        out_mat_reshaped,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_gate_reshaped = out_gate.view(B, N, N, H)
    out_mul = out_mat_norm * out_gate_reshaped   # (B,N,N,H) fp16

    # 6. Final linear projection
    out_mul_flat = out_mul.view(R, H)  # (R, H)
    final_half = torch.nn.functional.linear(out_mul_flat, weights["to_out.weight"].to(torch.float16))  # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

"""
High‑performance Triton implementation of the **outgoing** TriMul operator
(AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B,N,N,dim) – done with Torch’s
   highly‑optimised GPU LayerNorm (FP16 accumulation, FP16 output).

2️⃣ Fuse the five linear maps (left‑proj, right‑proj, three gates) into a
   single GEMM:  (B·N·N, dim) @ (5·H, dim)ᵀ → (B·N·N, 5·H).
   The weight matrix is built on‑the‑fly by concatenating the five
   weight tensors.

3️⃣ Triton kernel `split_fuse_gate_fused_kernel`:
   * extracts the five slices from the fused projection,
   * applies the optional pair‑wise mask,
   * evaluates the three sigmoid gates (FP32 for stability),
   * writes the gated *left* and *right* tensors directly in the
     layout required for a batched GEMM (shape (B·H, N, N)),
   * stores the *out_gate* (shape (B·N·N, H)) for the second LayerNorm.

4️⃣ Batched half‑precision GEMM (Tensor‑Core cuBLAS):
        out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)

5️⃣ Second LayerNorm (over the hidden dimension H) – again using Torch’s
   LayerNorm (FP16 input, FP32 accumulation).  The normalized tensor
   is multiplied element‑wise with the previously computed *out_gate*.

6️⃣ Final linear projection back to the original dimensionality `dim`
   (half‑precision weight, FP16 mat‑mul, cast to FP32 for the output).

Only step 3 is written in Triton; the remaining steps rely on
torch‑CUDA kernels which are already highly tuned on H100.  This mix
provides a fast end‑to‑end forward pass while satisfying the
“implement at least part of the operations in a kernel’’ requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split / fuse projection, apply mask, sigmoid gates,
#    write left/right (B·H,N,N) and out_gate (B·N·N,H)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[B·H, N, N] (fp16)
    right_ptr,         # mutable *[B·H, N, N] (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – ignored if HAS_MASK==0
    rows: tl.int32,    # = B·N·N
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Decode (b,i,j) from flat row index
    # ------------------------------------------------------------------
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # ------------------------------------------------------------------
    # Optional pair‑wise mask (broadcast across hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension blocks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ----- load 5 projected slices (layout: left, right, left_gate,
        #       right_gate, out_gate) -----
        # each slice has size H; all are concatenated along the last axis
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R,H]

        left_raw       = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_raw      = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # ----- sigmoid gates (computed in fp32 for numerical stability) -----
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----
        left  = left_raw  * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # ----- write left/right into (B·H, N, N) layout -----
        # hidden‑batch index: bh = b * H + h
        bh = b_idx[:, None] * H + offs_h[None, :]                     # [R,H]
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R,H]

        tl.store(left_ptr  + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ----- store out_gate (still (rows, H)) -----
        tl.store(out_gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 I/O, FP32 accumulation)
    # ------------------------------------------------------------------
    # Convert input and LayerNorm parameters to half‑precision.
    x_half = x.to(torch.float16)
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    x_norm = F.layer_norm(
        x_half,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )                                 # shape [B, N, N, dim], dtype fp16

    # Flatten to (R, dim) for the fused projection.
    x_norm_flat = x_norm.view(R, dim)                       # fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (left/right + 3 gates)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each of shape [H, dim]).
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                     # (5*H, dim)

    # (R, dim) @ (dim, 5*H)ᵀ → (R, 5*H)
    proj = F.linear(x_norm_flat, w_concat)                  # fp16
    del x_norm, x_norm_flat, w_concat

    # ------------------------------------------------------------------
    # 3️⃣ Split / fuse gates, apply optional mask, write left/right
    # ------------------------------------------------------------------
    left     = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty_like(left)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(R)       # flatten to [R]
        has_mask = 1

    BLOCK_R = 256      # rows per program (tuned for H100)
    BLOCK_H = 64       # hidden‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_fused_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (Tensor‑Core)
    #    out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
    # ------------------------------------------------------------------
    # left/right are (B·H, N, N) fp16
    out_mat = torch.bmm(left, right.transpose(1, 2))        # (B·H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) + element‑wise out_gate
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H) for torch.nn.functional.layer_norm
    out_mat_reshaped = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_mat_reshaped,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )                                                       # (B,N,N,H) fp16

    out_gate_reshaped = out_gate.view(B, N, N, H)          # (B,N,N,H) fp16
    gated = out_norm * out_gate_reshaped                    # (B,N,N,H) fp16
    del out_mat, out_gate, out_norm, out_gate_reshaped

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    gated_flat = gated.view(R, H)                           # (R, H) fp16
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)

    final_half = F.linear(gated_flat, to_out_w)             # (R, dim) fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
