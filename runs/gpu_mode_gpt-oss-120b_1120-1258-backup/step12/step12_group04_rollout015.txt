                         mask=mask_d[:, None] & mask_h[None, :], other=0.0) # fp16
        # similar for w_right, w_lgate, w_rgate, w_ogate

        acc_left += tl.dot(x_f16, w_left)
        acc_right += tl.dot(x_f16, w_right)
        acc_lgate += tl.dot(x_f16, w_lgate)
        acc_rgate += tl.dot(x_f16, w_rgate)
        acc_ogate += tl.dot(x_f16, w_ogate)

    # compute mean and rstd
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # load weight_sum and bias for groups
    weight_sum_left = tl.load(weight_sum_left_ptr + offs_h, mask=mask_h, other=0.0)
    bias_left = tl.load(bias_left_ptr + offs_h, mask=mask_h, other=0.0)
    # similar for other groups

    # adjust with mean, rstd, bias
    # broadcast mean, rstd to shape [BLOCK_R, BLOCK_H]
    mean_b = mean[:, None]
    rstd_b = rstd[:, None]

    left = (acc_left - mean_b * weight_sum_left[None, :]) * rstd_b + bias_left[None, :]
    right = (acc_right - mean_b * weight_sum_right[None, :]) * rstd_b + bias_right[None, :]
    lgate = (acc_lgate - mean_b * weight_sum_lgate[None, :]) * rstd_b + bias_lgate[None, :]
    rgate = (acc_rgate - mean_b * weight_sum_rgate[None, :]) * rstd_b + bias_rgate[None, :]
    ogate = (acc_ogate - mean_b * weight_sum_ogate[None, :]) * rstd_b + bias_ogate[None, :]

    # apply sigmoid to gate values
    lgate_sig = (1.0 / (1.0 + tl.exp(-lgate))).to(tl.float16)
    rgate_sig = (1.0 / (1.0 + tl.exp(-rgate))).to(tl.float16)
    ogate_sig = (1.0 / (1.0 + tl.exp(-ogate))).to(tl.float16)

    # mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0) # fp16
        mask_val = mask_val[:, None]  # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    left = left.to(tl.float16) * lgate_sig * mask_val
    right = right.to(tl.float16) * rgate_sig * mask_val

    # write left/right/out_gate
    # compute destination offsets
    seqsq = N * N
    bh = b_idx[:, None] * H + offs_h[None, :]   # [R,H]
    dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]

    mask_write = mask_r[:, None] & mask_h[None, :]

    tl.store(left_ptr + dst_off, left, mask=mask_write)
    tl.store(right_ptr + dst_off, right, mask=mask_write)
    tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :], ogate_sig, mask=mask_write)
```

def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # Precompute per-group weights
    gamma = weights["norm.weight"]   # shape [dim] fp32
    beta = weights["norm.bias"]     # shape [dim] fp32

    def preprocess_linear(weight):
        # weight shape [hidden_dim, dim] fp32 (bias=False)
        w = weight.to(torch.float32)  # ensure fp32
        w_scaled = w * gamma[None, :]            # [H, dim]
        bias = (beta[None, :] * w).sum(dim=1).to(torch.float16) # [H]
        weight_sum = w_scaled.sum(dim=1).to(torch.float16)         # [H]
        # transpose for faster access in kernel (dim, H)
        w_scaled_T = w_scaled.t().contiguous().to(torch.float16)   # [dim, H]
        return w_scaled_T, bias, weight_sum

    left_w_T, left_bias, left_sum = preprocess_linear(weights["left_proj.weight"])
    right_w_T, right_bias, right_sum = preprocess_linear(weights["right_proj.weight"])
    lgate_w_T, lgate_bias, lgate_sum = preprocess_linear(weights["left_gate.weight"])
    rgate_w_T, rgate_bias, rgate_sum = preprocess_linear(weights["right_gate.weight"])
    ogate_w_T, ogate_bias, ogate_sum = preprocess_linear(weights["out_gate.weight"])

    # allocate outputs
    left = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    # flatten mask
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(R)
        has_mask = 1

    # launch fused kernel
    BLOCK_R = 256   # rows per program
    BLOCK_D = 64    # dim tile
    BLOCK_H = 64    # hidden tile
    grid = (triton.cdiv(R, BLOCK_R),)
    norm_proj_gate_kernel[grid](
        x,
        left_w_T,
        right_w_T,
        lgate_w_T,
        rgate_w_T,
        ogate_w_T,
        left_sum,
        right_sum,
        lgate_sum,
        rgate_sum,
        ogate_sum,
        left_bias,
        right_bias,
        lgate_bias,
        rgate_bias,
        ogate_bias,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        dim,
        H,
        N,
        HAS_MASK=has_mask,
        eps=eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # Batched GEMM
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B*H,N,N) fp16
    del left, right

    # Second LN + out_gate multiplication
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)
    grid_norm2 = (triton.cdiv(R, BLOCK_R),)
    layernorm_mul_strided_kernel[grid_norm2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ
        weights["to_out_norm.bias"],     # β
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_R,
        BLOCK_H=BLOCK_D,
        num_warps=8,
    )
    del out_mat, out_gate

    # final linear projection
    final = F.linear(out_fused, weights["to_out.weight"].to(torch.float16)).view(B, N, N, dim).to(torch.float32)
    return final

```
@triton.jit
def norm_proj_gate_kernel(
    x_ptr,                 # const float32* [R, dim]
    w_left_ptr,            # const float16* [dim, H] (scaled)
    w_right_ptr,
    w_lgate_ptr,
    w_rgate_ptr,
    w_ogate_ptr,
    sum_left_ptr,          # const float16* [H]  weight_sum after scaling
    sum_right_ptr,
    sum_lgate_ptr,
    sum_rgate_ptr,
    sum_ogate_ptr,
    bias_left_ptr,         # const float16* [H]
    bias_right_ptr,
    bias_lgate_ptr,
    bias_rgate_ptr,
    bias_ogate_ptr,
    left_ptr,              # mutable float16* [B*H, N, N]
    right_ptr,
    out_gate_ptr,          # mutable float16* [R, H]
    mask_ptr,              # const float16* [R] optional
    R: tl.int32,
    dim: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,j from flat row index
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # allocate accumulators
    acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_ogate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x block (float32)
        x_ptrs = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0)   # fp32

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

        x_f16 = x.to(tl.float16)

        # load weight blocks (float16)
        w_left = tl.load(w_left_ptr + offs_d[:, None] * H + tl.arange(0, BLOCK_H)[None, :],
                         mask=mask_d[:, None] & (tl.arange(0, BLOCK_H)[None, :] < H),
                         other=0.0)
        w_right = tl.load(w_right_ptr + offs_d[:, None] * H + tl.arange(0, BLOCK_H)[None, :],
                          mask=mask_d[:, None] & (tl.arange(0, BLOCK_H)[None, :] < H),
                          other=0.0)
        w_lgate = tl.load(w_lgate_ptr + offs_d[:, None] * H + tl.arange(0, BLOCK_H)[None, :],

    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor

    """

    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]

    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of (b,i,j) positions
    H = hidden_dim

    # Pre‑compute scaled weights for the first fused kernel
    # ------------------------------------------------------------------
    gamma = weights["norm.weight"]    # [dim]  float32
    beta  = weights["norm.bias"]      # [dim]  float32

        w = weight.to(torch.float32)                # -> fp32
        w_scaled = w * gamma[None, :]               # γ·W

                bias.to(torch.float16),                        # [H]
                wsum.to(torch.float16))                        # [H]

    wL_T, bL, sL = preprocess(weights["left_proj.weight"])
    wR_T, bR, sR = preprocess(weights["right_proj.weight"])
    wLG_T, bLG, sLG = preprocess(weights["left_gate.weight"])
    wRG_T, bRG, sRG = preprocess(weights["right_gate.weight"])
    wOG_T, bOG, sOG = preprocess(weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # Allocate output buffers
    # ------------------------------------------------------------------
    left      = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Optional mask handling
    # ------------------------------------------------------------------
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(R)
        has_mask = 1

    # ------------------------------------------------------------------
    # 1️⃣  fused LayerNorm + projections + gates
    # ------------------------------------------------------------------
    BLOCK_R = 256   # rows per program – fits the typical batch sizes
    BLOCK_D = 64    # dim tile
    BLOCK_H = 64    # hidden tile (H ≤ 128)
    grid = (triton.cdiv(R, BLOCK_R),)

    norm_proj_gate_kernel[grid](
        x,
        wL_T, wR_T, wLG_T, wRG_T, wOG_T,
        sL, sR, sLG, sRG, sOG,
        bL, bR, bLG, bRG, bOG,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        dim,
        H,
        N,
        HAS_MASK=has_mask,
        eps=eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 2️⃣  Batched GEMM (Tensor‑Core cuBLAS)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 3️⃣  Second LayerNorm + element‑wise gate multiplication
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_R),)
    layernorm_mul_strided_kernel[grid2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32) of the second LN
        weights["to_out_norm.bias"],     # β (float32) of the second LN
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_R,
        BLOCK_H=BLOCK_D,                 # reuse BLOCK_D for the hidden‑dim tile
        num_warps=8,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 4️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    ).view(B, N, N, dim).to(torch.float32)

    return final
