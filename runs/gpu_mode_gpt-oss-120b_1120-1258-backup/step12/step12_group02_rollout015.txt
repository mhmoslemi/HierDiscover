import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,         # *[rows, 5*H]  fp16
    left_ptr,         # *[B*H, N, N] fp16
    right_ptr,        # *[B*H, N, N] fp16
    out_gate_ptr,     # *[rows, H]   fp16
    mask_ptr,         # *[rows]      fp16 (ignored if HAS_MASK=0)
    rows: tl.int32,   # = B*N*N
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < rows

    seqsq = N * N

    # decode (b,i,j) from flat row index
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # load mask once per row if present
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val_row = m[:, None]                               # broadcast
    else:
        mask_val_row = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # pointers to the 5 slices for this block
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]           # [R,Hblk]

        left_raw      = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw     = tl.load(proj_ptr + base + H,

    """
    High‑performance Triton‑accelerated forward pass of the *outgoing* TriMul
    operation (AlphaFold‑3).  The heavy linear algebra (projections, batched GEMM,
    final projection) are performed with cuBLAS in fp16, while the mask‑gate
    handling and the first split of the projected tensor are fused into a
    single Triton kernel.
    """
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of (b,i,j) entries
    H = hidden_dim
    seqsq = N * N

    # ------------------------------------------------------------------
    # 1️⃣  Layer‑norm over the last dimension (float32 -> fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()
    # PyTorch's native layer‑norm is highly optimised; we keep the result in fp32
    # and cast to fp16 for the later half‑precision kernels.
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Concatenated linear projection to (left, right, gates)
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # shape [5*H, dim]

    proj = F.linear(x_norm, w_concat)   # (R, 5*H)  fp16
    del x_norm, w_concat, x_flat

    # ------------------------------------------------------------------
    # 3️⃣  Split, apply mask & sigmoid gates, write left/right (B*H,N,N) and out_gate
    # ------------------------------------------------------------------
    left  = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16).reshape(R)   # flatten
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_tensor

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM over the hidden dimension:
    #     out(b,h,i,j) = Σₖ left(b,h,i,k)·right(b,h,j,k)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B*H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm (hidden_dim) + element‑wise multiply with out_gate
    # ------------------------------------------------------------------
    # reshape to (R, H) same ordering as out_gate
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).reshape(R, H).contiguous()
    del out_mat

    out_norm_fp32 = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # element‑wise multiply with out_gate (both fp16)
    fused = out_norm * out_gate
    del out_norm, out_gate, out_flat

    # ------------------------------------------------------------------
    # 6️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (R, dim) fp16
    del fused

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
