"""
High‑performance Triton implementation of the **outgoing** TriMul operator
(AlphaFold‑3).

The forward pass consists of:
1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16).
2️⃣ A single fused Linear that computes the 5 projections (left/right + 3 gates).
3️⃣ A Triton kernel that
   • splits the 5‑way projection,
   • applies the pair‑wise mask,
   • evaluates sigmoid gates,
   • writes the left/right tensors required for the batched GEMM,
   • stores the out‑gate for the second LayerNorm.
4️⃣ Batched half‑precision GEMM (Tensor‑Core cuBLAS) for the pairwise
   multiplication `out = left @ rightᵀ`.
5️⃣ A fused Triton kernel that
   • computes the hidden‑dimension LayerNorm (mean / variance),
   • applies its scale / bias,
   • multiplies by the previously stored out‑gate,
   • performs the final Linear projection back to `dim`,
   • writes the result directly in FP16.
   This kernel replaces the original “second LayerNorm + linear” two‑step
   implementation and therefore avoids an extra read/write of the intermediate
   hidden‑tensor.
6️⃣ Reshape to `[B, N, N, dim]` and cast to FP32.

Only the two Triton kernels (`split_fuse_gate_fused_kernel` and
`final_norm_linear_kernel`) are custom; all heavy matrix‑multiplications
use cuBLAS Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # const   *[rows, dim]   (float32)
    y_ptr,           # mutable *[rows, dim]   (float16)
    weight_ptr,      # const   *[dim]        (float32)
    bias_ptr,        # const   *[dim]        (float32)
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ mean & variance (accumulate in fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized values: γ·(x‑μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]          # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, sigmoid gates,
#    and write left/right in (B·H, N, N) layout.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B·H, N, N] (float16)
    right_ptr,         # mutable *[B·H, N, N] (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]      (float16) – ignored if HAS_MASK==0
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # decode (b,i,j) from flat row index
    # --------------------------------------------------------------
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # --------------------------------------------------------------
    # optional pair‑wise mask (broadcast across hidden dim)
    # --------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = m[:, None]                     # [R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    # loop over hidden dimension blocks
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---------- load 5 projected slices ----------
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]                # [R,H]

        left_raw       = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_raw      = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # ---------- sigmoid gates (fp32 for stability) ----------
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---------- apply mask and gates ----------
        left  = left_raw  * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # ---------- write left/right into (B·H, N, N) ----------
        # hidden‑batch index: bh = b * H + h
        bh = b_idx[:, None] * H + offs_h[None, :]                     # [R,H]
        # linear offset inside (B·H, N, N): bh * N² + i * N + j
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R,H]

        tl.store(left_ptr  + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ---------- store out_gate (still (rows, H)) ----------
        tl.store(out_gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fused LayerNorm (hidden dim) + out‑gate multiplication + final Linear
# ----------------------------------------------------------------------
@triton.jit
def final_norm_linear_kernel(
    out_ptr,          # const *[B·H, N, N]          (float16)
    gate_ptr,         # const *[rows, H]            (float16)
    weight_ptr,       # const *[dim, H]             (float16)
    norm_weight_ptr,  # const *[H]                  (float32) – γ of to_out_norm
    norm_bias_ptr,    # const *[H]                  (float32) – β of to_out_norm
    y_ptr,            # mutable *[rows, dim]        (float16)
    rows: tl.int32,   # = B·N·N
    N: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
    BLOCK_D: tl.constexpr,   # output dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_r = offs_r < rows

    seqsq = N * N

    # ------------------------------------------------------------------
    # decode (b,i,j) from flat row index
    # ------------------------------------------------------------------
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # base offset for hidden‑slice h = 0
    base0 = b_idx * H * seqsq + i_idx * N + j_idx      # [BLOCK_M]

    # ------------------------------------------------------------------
    # 1️⃣ mean & variance over H (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(out_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)           # [M, Hblk]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ fused normalize – out_gate – Linear
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # accumulator for this output‑dim tile
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
            mask_h = offs_h < H

            # ------- load hidden slice from out_mat -------
            vals = tl.load(out_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)           # [M, Hblk]

            # ------- load corresponding out_gate slice -------
            gate_vals = tl.load(gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)         # [M, Hblk]

            # ------- normalize (LayerNorm) -------
            norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

            # ------- apply to_out_norm weight & bias -------
            gamma = tl.load(norm_weight_ptr + offs_h,
                            mask=mask_h, other=0.0)           # fp32
            beta  = tl.load(norm_bias_ptr + offs_h,
                            mask=mask_h, other=0.0)           # fp32
            norm = norm * gamma[None, :] + beta[None, :]       # fp32

            # ------- multiply by out_gate -------
            norm = norm * gate_vals                           # fp32

            # ------- load weight slice (transposed) -------
            # weight layout: [dim, H] → offset = h * dim + d
            w = tl.load(weight_ptr + (offs_h[:, None] * dim + offs_d[None, :]),
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)           # [Hblk, Dblk]

            # ------- accumulate dot product -------
            acc += tl.dot(norm, w)                            # (M,H)@(H,D) → (M,D)

        # store this tile
        out_tile_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_tile_ptr, acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                      # [R, dim]  fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 512   # rows per program
    BLOCK_D = 64    # dim tile
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ (float32, [dim])
        weights["norm.bias"],      # β (float32, [dim])
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (left/right + 3 gates)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices: shape [5*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                      # [5*H, dim]

    proj = F.linear(x_norm, w_concat)                         # (R, 5*H) fp16
    del w_concat, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split/fuse gates, apply optional mask, write left/right
    # ------------------------------------------------------------------
    left = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16).reshape(R)
        has_mask = 1

    BLOCK_R = 256
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_tensor

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (Tensor‑Core)
    #    out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Fused hidden‑dim LayerNorm + out_gate * final Linear
    # ------------------------------------------------------------------
    out_final = torch.empty(R, dim, dtype=torch.float16, device=device)

    grid_final = (triton.cdiv(R, BLOCK_M),)
    final_norm_linear_kernel[grid_final](
        out_mat.flatten(),                                   # flatten for 1‑D indexing
        out_gate.flatten(),
        weights["to_out.weight"].to(torch.float16).flatten(),
        weights["to_out_norm.weight"],                       # γ (float32, [H])
        weights["to_out_norm.bias"],                         # β (float32, [H])
        out_final,
        R,
        N,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape back to [B, N, N, dim] and cast to float32
    # ------------------------------------------------------------------
    out = out_final.view(B, N, N, dim).to(torch.float32)
    return out
