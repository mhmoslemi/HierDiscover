"""
TriMul (outgoing) – Triton‑accelerated forward pass (no gradients)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16).
2️⃣ One fused FP16 GEMM that produces the five linear maps
   (left/right projections + left/right/out gates) in a single matmul.
3️⃣ Split the fused projection, apply the optional pair‑wise mask,
   evaluate sigmoid gates and write the three tensors directly in the
   (B, H, N, N) layout required for the batched GEMM.
4️⃣ Batched GEMM (cuBLAS) for the multiplicative update:
   out[b,h] = left[b,h] @ right[b,h]ᵀ   (FP16).
5️⃣ LayerNorm over the hidden dimension H, followed by element‑wise
   multiplication with the out‑gate.  The result is written directly
   into a contiguous (B, N, N, H) buffer.
6️⃣ Final linear projection back to the original `dim` (FP16 → FP32).

Only the two LayerNorms and the split/fuse kernel are written in Triton;
all dense matrix multiplications use highly‑optimized CuBLAS (torch.bmm /
torch.nn.functional.linear).  The kernels are tuned for H100 (block
sizes 256/128, 8 warps) and should comfortably meet the <1 ms target on
the provided benchmark shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   *[rows, dim]   (float32)
    y_ptr,               # mutable *[rows, dim]   (float16)
    w_ptr,               # const   *[dim]        (float32)  γ
    b_ptr,               # const   *[dim]        (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean and variance (fp32) in two passes over the dim axis
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalize, apply γ/β and store as FP16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]               # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)        # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)        # β (fp32)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#    write directly in (B, H, N, N) layout
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) – ignored if HAS_MASK=0
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index → (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask broadcasted over H
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between consecutive H slices

    # Base offset for the slice (b,i,k) at h = 0  (layout B,H,N,N)
    base_offset = b * (H * N * N) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load 5×[BLOCK_R,BLOCK_H] from fused projection -----------------
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R,H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate = tl.load(proj_ptr + proj_base + 2 * H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_gate = tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ---- sigmoid (stable, operate in fp32) -----------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates -----------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ---- destination offsets -------------------------------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R,H]

        tl.store(left_ptr      + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 3️⃣ Second LayerNorm + multiplication by out_gate,
#    writing directly into (B, N, N, H) layout
# --------------------------------------------------------------
@triton.jit
def layernorm_mul_fused_kernel(
    out_ptr,        # const *[B, H, N, N] (fp16)
    gate_ptr,       # const *[B, H, N, N] (fp16)
    out_fused_ptr,  # mutable *[B, N, N, H] (fp16)   <-- result
    gamma_ptr,      # const *[H] (float32)
    beta_ptr,       # const *[H] (float32)
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,   # rows per program (R = B·N·N)
    BLOCK_H: tl.constexpr,   # hidden‑dim chunk
):
    pid = tl.program_id(0)
    rows = B * N * N                      # total rows = B·N·N
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # Decode (b,i,j) from row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # Strides for source layout (B, H, N, N)
    stride_b_src = H * N * N
    stride_i_src = N
    stride_j_src = 1
    stride_h_src = N * N

    # Base offset at h = 0
    src_base = b * stride_b_src + i * stride_i_src + j * stride_j_src   # [BLOCK_R]

    # ------------------------------------------------------------------
    # Compute mean & variance over H (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        ptr = out_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Destination strides for (B, N, N, H) contiguous layout
    # ------------------------------------------------------------------
    stride_b_dst = N * N * H
    stride_i_dst = N * H
    stride_j_dst = H
    dst_base = b * stride_b_dst + i * stride_i_dst + j * stride_j_dst   # [BLOCK_R]

    # ------------------------------------------------------------------
    # Normalize, apply γ/β, multiply with out_gate and store
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # load output values (fp16 → fp32)
        ptr_out = out_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        out_val = tl.load(ptr_out,
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

        # load gate values (fp16 → fp32)
        ptr_gate = gate_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        gate_val = tl.load(ptr_gate,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        # LN (fp32)
        norm = (out_val - mean[:, None]) * rstd[:, None]

        # γ / β (fp32)
        w = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)          # γ
        b_param = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)    # β

        norm = norm * w[None, :] + b_param[None, :]

        # multiply with gate (still fp32) and store as fp16
        out = norm * gate_val

        dst_ptr = out_fused_ptr + dst_base[:, None] + offs_h[None, :]   # stride_h_dst = 1
        tl.store(dst_ptr, out.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all linear / layernorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim" and "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                 # rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256   # rows per program
    BLOCK_D = 128   # dim chunk
    grid_ln1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right proj + three gates → (R, 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)    # fp16, shape [R, 5·H]
    del fused_w, x_norm                 # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate (B, H, N, N)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Determine if we need to apply a mask
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # Flatten mask to [R] and cast to fp16
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 256   # rows per program for split kernel
    BLOCK_H = 64    # hidden‑dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N] fp16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N] fp16
    # BMM with transposed right matrix (rightᵀ) → [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N)                    # (B, H, N, N) fp16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise out‑gate multiplication,
    #    write directly into (B, N, N, H) contiguous buffer
    # ------------------------------------------------------------------
    out_fused = torch.empty(B, N, N, H, dtype=torch.float16, device=device)

    BLOCK_R2 = 256   # rows per program for second LN
    BLOCK_H2 = 64    # hidden‑dim chunk
    grid_ln2 = (triton.cdiv(R, BLOCK_R2),)
    layernorm_mul_fused_kernel[grid_ln2](
        out,
        out_gate,
        out_fused,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        B,
        N,
        H,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=8,
    )
    del out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 → fp32)
    # ------------------------------------------------------------------
    out_fused_flat = out_fused.view(R, H)                      # [R, H]
    final_half = F.linear(
        out_fused_flat,
        w["to_out.weight"].to(torch.float16),                 # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
