    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute mean and var per row
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

def custom_kernel(data):
    # unpack
    x, mask, w, cfg = data
    B,N,_,dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    nomask = cfg.get("nomask", True)

    # First layernorm in fp16 using torch
    # Cast input to fp16
    x_fp16 = x.to(torch.float16)
    norm_weight_fp16 = w["norm.weight"].to(torch.float16)
    norm_bias_fp16 = w["norm.bias"].to(torch.float16)
    x_norm = torch.nn.functional.layer_norm(
        x_fp16,
        normalized_shape=(dim,),
        weight=norm_weight_fp16,
        bias=norm_bias_fp16,
        eps=eps,
    )  # shape [B,N,N,dim], fp16

    # Linear transformations for left/right projection and gates: fused weight
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [5*H, dim]
    # note: torch.nn.functional.linear expects weight shape [out_features, in_features]
    proj = torch.nn.functional.linear(x_norm, fused_w)  # shape [B,N,N,5*H]?? Actually input shape (B,N,N,dim); but linear works on last dim.
    # will produce shape (B,N,N,5*H). Ensure contiguous.
    # we need to reshape to (R,5*H)
    R = B * N * N
    proj_flat = proj.view(R, 5*hidden_dim)  # [R,5H] fp16

    # Split / apply mask + gates via kernel
    left = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    if nomask:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        HAS_MASK = 0
    else:
        # mask shape [B,N,N] possibly bool or float; cast to float16 and flatten to rows
        mask_fp16 = mask.to(torch.float16).view(R)
        mask_ptr = mask_fp16
        HAS_MASK = 1
    # kernel launch as before (likely uses split_fuse_gate_kernel)
    BLOCK_R=128
    BLOCK_H=32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left, right ready, out_gate ready

    # multiplicative update
    left_mat = left.view(B * hidden_dim, N, N)  # shape [B*H,N,N]
    right_mat = right.view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, hidden_dim, N, N)  # shape (B,H,N,N) fp16

    # second layernorm in fp16 using torch
    # Need weight+bias for to_out_norm:
    to_out_norm_weight_fp16 = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias_fp16 = w["to_out_norm.bias"].to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight_fp16,
        bias=to_out_norm_bias_fp16,
        eps=eps,
    )  # shape (B,H,N,N) fp16

    # multiply with out_gate
    gated = out_norm * out_gate  # elementwise broadcast: (B,H,N,N)

    # final linear projection to dim
    to_out_weight_fp16 = w["to_out.weight"].to(torch.float16)  # [dim, H]
    # We need to move the hidden dim to last dimension for linear.
    # Input currently shape (B,N,N,H) required shape for linear expects last dim = in_features.
    # So first reshape
    gated_perm = gated.permute(0,2,3,1).contiguous()  # shape [B,N,N,H]

"""
TriMul (outgoing) – Triton‑accelerated forward pass (no gradients).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input (FP16) – performed by CuDNN
    via `torch.nn.functional.layer_norm`.  This is far faster than a
    hand‑written kernel on H100.

2️⃣  Fused linear projection that computes the five maps
    (left, right, left_gate, right_gate, out_gate) in a single
    `torch.nn.functional.linear` call (FP16).

3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * Decodes the flat `(b,i,k)` index.
    * Loads the five projected blocks.
    * (Optionally) applies the pair‑wise mask.
    * Applies the three sigmoid gates.
    * Writes `left`, `right` and `out_gate` directly in the
      `(B, H, N, N)` layout required for the batched GEMM.

4️⃣  Batched half‑precision GEMM (`torch.bmm`) for the multiplicative
    update:  out = left @ rightᵀ   → shape `(B, H, N, N)`.

5️⃣  Triton kernel `layernorm_mul_fused_kernel`:
    * Row‑wise LayerNorm over the hidden dimension `H` (the last axis of
      the output after a virtual transpose).
    * Multiplies the normalized values by `out_gate`.
    * Writes the result directly into a contiguous buffer of shape
      `(B, N, N, H)` (no extra permute/copy).

6️⃣  Final linear projection back to the original `dim` using a
    fast `torch.nn.functional.linear` (FP16 → FP32).

Only the two Triton kernels are required; all heavy matrix
multiplications are delegated to cuBLAS/cuDNN which gives a runtime
well below the 1 ms target on the provided benchmarks.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: split the fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b, i, k) from the flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between consecutive hidden slices

    # base offset for the (B,H,N,N) layout (h = 0)
    base_offset = b * (H * N * N) + i * N + k                    # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # each projection block lives consecutively in the fused matrix
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate  = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_gate = tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # stable sigmoid in fp32
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw   * mask_val * left_gate
        right = right_raw  * mask_val * right_gate

        # destination offsets for the 3 tensors (stride_h = hidden_stride)
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset,
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset,
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 2: second LayerNorm over hidden dim + multiplication with out_gate,
#           write directly into contiguous (B, N, N, H) layout.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_fused_kernel(
    out_ptr,        # const *[B, H, N, N] (fp16)
    gate_ptr,       # const *[B, H, N, N] (fp16)
    out_fused_ptr,  # mutable *[B, N, N, H] (fp16)   <-- result
    gamma_ptr,      # const *[H] (float32)
    beta_ptr,       # const *[H] (float32)
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,   # rows per program (R = B·N·N)
    BLOCK_H: tl.constexpr,   # hidden‑dim chunk
):
    pid = tl.program_id(0)
    rows = B * N * N
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # decode (b,i,j) for the source layout (B, H, N, N)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # source strides (B, H, N, N)
    stride_b_src = H * N * N
    stride_i_src = N
    stride_j_src = 1
    stride_h_src = N * N                               # distance between consecutive h slices

    src_base = b * stride_b_src + i * stride_i_src + j * stride_j_src   # [BLOCK_R]

    # -------------------- mean & variance over H -----------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H
        ptr = out_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------- destination layout (B, N, N, H) ---------------
    stride_b_dst = N * N * H
    stride_i_dst = N * H
    stride_j_dst = H
    # (stride_h_dst = 1)

    dst_base = b * stride_b_dst + i * stride_i_dst + j * stride_j_dst   # [BLOCK_R]

    # -------------------- normalize, apply γ/β, multiply gate ----------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)           # [BLOCK_H]
        mask_h = offs_h < H

        # load out values
        ptr_out = out_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        out_val = tl.load(ptr_out,
                          mask=mask_r[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

        # load gate values
        ptr_gate = gate_ptr + src_base[:, None] + offs_h[None, :] * stride_h_src
        gate_val = tl.load(ptr_gate,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        # normalize (fp32)
        norm = (out_val - mean[:, None]) * rstd[:, None]

        # γ and β (fp32)
        w = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # γ
        b_param = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)  # β

        norm = norm * w[None, :] + b_param[None, :]

        # multiply with gate (fp16 result later)
        out = norm * gate_val

        # store to contiguous (B,N,N,H) buffer
        dst_ptr = out_fused_ptr + dst_base[:, None] + offs_h[None, :]   # stride_h_dst = 1
        tl.store(dst_ptr,
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

# Custom forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] or None
        - weights_dict : dict of model parameters (float32 tensors)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16) using cuDNN – fast & stable
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)
    norm_weight_fp16 = w["norm.weight"].to(torch.float16)
    norm_bias_fp16   = w["norm.bias"].to(torch.float16)
    x_norm = torch.nn.functional.layer_norm(
        x_fp16,
        (dim,),
        weight=norm_weight_fp16,
        bias=norm_bias_fp16,
        eps=eps,
    )                                 # [B, N, N, dim]  (fp16)

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection for left/right + three gates
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # shape [5*H, dim]

    proj = torch.nn.functional.linear(x_norm, fused_w)   # [B, N, N, 5*H] (fp16)
    proj_flat = proj.view(R, 5 * H)                     # [R, 5*H]

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask (if any) and sigmoid gates (Triton)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy
    else:
        HAS_MASK = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)                   # [R]
        mask_ptr = mask_fp16

    BLOCK_R = 128   # rows per program
    BLOCK_H = 32    # hidden‑dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, proj_flat, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dim)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # (B·H, N, N)
    right_mat = right.view(B * H, N, N)         # (B·H, N, N)
    out_mat   = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)
    out = out_mat.view(B, H, N, N)                                # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate multiplication (writes contiguous)
    # --------------------------------------------------------------
    out_fused = torch.empty(B, N, N, H, dtype=torch.float16, device=device)

    BLOCK_R2 = 128
    BLOCK_H2 = 32
    grid_ln2 = (triton.cdiv(R, BLOCK_R2),)
    layernorm_mul_fused_kernel[grid_ln2](
        out,
        out_gate,
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        B,
        N,
        H,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out, out_gate, left, right

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (FP16 → FP32)
    # --------------------------------------------------------------
    # reshape to (R, H) where H is the innermost dimension of out_fused
    out_fused_flat = out_fused.view(R, H)          # [R, H]  fp16
    to_out_weight_fp16 = w["to_out.weight"].to(torch.float16)   # [dim, H]

    final_fp16 = torch.nn.functional.linear(out_fused_flat, to_out_weight_fp16)  # [R, dim]

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
