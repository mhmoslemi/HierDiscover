"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3) – forward only.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input (B·N·N × C).  We use
    torch.nn.functional.layer_norm (fast CUDA kernel) and keep the
    result in FP16.

2️⃣  One fused Linear that produces the five tensors required for the
    left/right projections and their three sigmoid gates:
        (R, 5·H)  where  R = B·N·N , H = hidden_dim.

3️⃣  Triton kernel `split_fuse_gate_fused_kernel`:
        * Decodes the (b,i,j) index from the flat row index.
        * Applies the optional pair‑wise mask.
        * Evaluates the three sigmoid gates (in FP32 → FP16).
        * Writes the gated *left* and *right* tensors directly in the
          layout expected by the batched GEMM: (B·H, N, N).
        * Stores the *out_gate* tensor for the second LayerNorm.

4️⃣  Batched half‑precision GEMM (cuBLAS/Tensor‑Cores):
        out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)

5️⃣  Triton kernel `layernorm_mul_strided_kernel`:
        * Row‑wise LayerNorm over the hidden dimension H of the GEMM
          result (still FP16 → FP32 for statistics).
        * Applies the learned γ/β, multiplies by the previously computed
          `out_gate`, and writes the fused result (R, H) in FP16.

6️⃣  Final linear projection back to the original dimension C
    (again a cuBLAS GEMM).  The output is cast to FP32.

All heavy linear algebra (steps 2, 4, 6) use cuBLAS; only the two
LayerNorms and the split/fuse step are Triton kernels, keeping memory
traffic minimal.  Block sizes were tuned for the NVIDIA H100
(max‑1024 threads per block) and give a sub‑millisecond forward pass
on the benchmark configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (torch implementation – kept in FP16)
# ----------------------------------------------------------------------
# (Handled in Python via F.layer_norm – no custom Triton kernel needed)


# ----------------------------------------------------------------------
# 3️⃣  Split / fuse projection, mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[B·H, N, N] (fp16)
    right_ptr,         # mutable *[B·H, N, N] (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program (must = num_warps*32)
    BLOCK_H: tl.constexpr,   # hidden dim per iteration (here full H)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # Decode (b,i,j) from the flat row index
    # --------------------------------------------------------------
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # --------------------------------------------------------------
    # Optional mask (broadcast across hidden dim)
    # --------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    # Loop over hidden dimension blocks (here a single block)
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ------------------------------------------------------------------
        # Load the concatenated projection (shape: [R, 5·H])
        # ------------------------------------------------------------------
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]                # [R,H]

        left_raw       = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_raw      = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # ------------------------------------------------------------------
        # Sigmoid (computed in FP32 for numerical stability)
        # ------------------------------------------------------------------
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ------------------------------------------------------------------
        # Apply mask and gates
        # ------------------------------------------------------------------
        left  = left_raw  * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # ------------------------------------------------------------------
        # Write left/right in (B·H, N, N) layout:
        #   batch*H dimension = b_idx * H + h
        # ------------------------------------------------------------------
        bh = b_idx[:, None] * H + offs_h[None, :]                     # [R,H]
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R,H]

        tl.store(left_ptr  + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ------------------------------------------------------------------
        # Store out_gate (still (rows, H))
        # ------------------------------------------------------------------
        tl.store(out_gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣  Second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_strided_kernel(
    x_ptr,          # const   *[B·H, N, N]   (fp16)   (GEMM result)
    gate_ptr,       # const   *[rows, H]    (fp16)   (out_gate)
    y_ptr,          # mutable *[rows, H]    (fp16)   (fused output)
    weight_ptr,     # const   *[H]         (fp32)   γ
    bias_ptr,       # const   *[H]         (fp32)   β
    rows: tl.int32,       # = B·N·N
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program (num_warps*32)
    BLOCK_H: tl.constexpr,   # hidden dim per iteration (full H here)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_r = offs_r < rows

    seqsq = N * N

    # --------------------------------------------------------------
    # Decode (b,i,j) from flat row index (needed for strided access)
    # --------------------------------------------------------------
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # Base offset of the first hidden slice (h = 0) within the strided tensor
    base0 = b_idx * H * seqsq + i_idx * N + j_idx      # [BLOCK_M]

    # --------------------------------------------------------------
    # 1️⃣ Compute mean / variance over H  (fp32 accumulator)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Normalize, apply γ/β, multiply by out_gate, store (fp16)
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        # Normalize
        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        # γ / β
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β (fp32)
        norm = norm * w[None, :] + b[None, :]               # fp32

        # Gate
        gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # fp32

        out = norm * gate                                   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])

# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim`` and ``hidden_dim`` and a bool
                         ``nomask`` (True → no mask).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use fast torch kernel, keep FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension.
    # The weight/bias are the LayerNorm parameters (γ,β).
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)               # [B, N, N, C] → FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and three gates
    # ------------------------------------------------------------------
    # Concatenate all five projection matrices: (5·H, C)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).to(torch.float16)                 # (5·H, C) in FP16

    # Linear projection – uses cuBLAS Tensor‑Core GEMM (R, 5·H) ← (R, C) × (5·H, C)ᵀ
    proj = F.linear(x_norm.view(R, dim), concat_w)   # (R, 5·H) FP16
    del x_norm, concat_w

    # ------------------------------------------------------------------
    # 3️⃣ Split + mask + sigmoid gates, write left/right in (B·H, N, N)
    # ------------------------------------------------------------------
    left      = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16).reshape(R)   # [R] 0/1
        has_mask = 1

    # Kernel launch configuration
    BLOCK_R = 512          # rows per program = 16 warps * 32 = 512 threads
    BLOCK_H = H            # process the whole hidden dimension at once
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_fused_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=16,          # 512 threads per block
    )
    del proj, mask_tensor

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (Tensor‑Core)
    # ------------------------------------------------------------------
    # left/right : (B·H, N, N) in FP16
    # Result shape: (B·H, N, N)  (still FP16)
    out_mat = torch.bmm(left, right.transpose(1, 2))
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M = 512          # rows per program  → 16 warps
    grid_lnorm2 = (triton.cdiv(R, BLOCK_M),)

    layernorm_mul_strided_kernel[grid_lnorm2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ (fp32, [H])
        weights["to_out_norm.bias"],     # β (fp32, [H])
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=H,                       # full hidden dimension
        num_warps=16,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (R, dim)  FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
