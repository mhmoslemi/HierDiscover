"""
TriMul (outgoing) – high‑performance Triton + cuBLAS implementation
===================================================================

Algorithm (all steps are fused as much as possible):
  1. Row‑wise LayerNorm on the (B·N·N, dim) view of the input.
     The kernel works in fp32 for the statistics and writes fp16 results.
  2. Fuse the five linear maps (left/right projections + three gates) into a
     single fp16 GEMM:  X_norm @ W_fusedᵀ → (R, 5·H)   (R = B·N·N).
  3. Split the fused projection, apply the optional pair‑wise mask,
     evaluate sigmoid gates and write the three tensors directly in the
     (B, H, N, N) layout required by the batched GEMM.
  4. Batched half‑precision GEMM (cuBLAS) over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
  5. Fuse second LayerNorm and out‑gate multiplication.  The kernel
     normalises over H for every (b,i,j) position, applies γ/β and the
     already‑sigmoided out‑gate, and writes a contiguous (R, H) buffer.
  6. Final linear projection back to the original `dim` (fp16 → fp32).

Only three custom Triton kernels are used – the heavy matrix‑multiplications
are delegated to cuBLAS Tensor‑Core kernels, which gives near‑peak
performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    w_ptr,          # const   *[dim]        (float32)   γ
    b_ptr,          # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # dim chunk size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#     write directly in (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode (b,i,k) from flat row index ---------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask (if any)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                     # distance between successive hidden slices

    # base offset (b·H·N·N + i·N + k)
    base_offset = b * (H * nn) + i * N + k                # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ----- load the five blocks from the fused projection ----------------
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw     = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(proj_ptr + proj_base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate    = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate   = tl.load(proj_ptr + proj_base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ----- sigmoid (stable fp32) ---------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ----------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- compute destination offsets -----------------------------------
        # offset = base_offset + h * hidden_stride
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        # store
        tl.store(left_ptr      + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm + out_gate multiplication (writes into (R, H))
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_fused_kernel(
    out_ptr,            # const   *[B, H, N, N] (fp16)
    gate_ptr,           # const   *[B, H, N, N] (fp16)
    out_fused_ptr,      # mutable *[R, H]       (fp16)
    w_ptr,              # const   *[H]           (float32) γ
    b_ptr,              # const   *[H]           (float32) β
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N                           # stride for the hidden dimension
    # decode b,i,j from flat row index
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # base offset for h = 0 in the (B, H, N, N) layout
    base_offset = b * (H * nn) + i * N + j                # [BLOCK_R]

    # ---- first pass: compute mean & variance over H -------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * nn,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: normalize, apply γ/β, multiply by out_gate, store -----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * nn,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_h[None, :] * nn,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        w = tl.load(w_ptr + offs_h, mask=mask_h, other=0.0)
        b_ = tl.load(b_ptr + offs_h, mask=mask_h, other=0.0)

        # normalise + γ/β
        norm = (out_vals - mean[:, None]) * rstd[:, None]       # fp32
        norm = norm * w[None, :] + b_[None, :]                  # fp32

        # apply out‑gate
        out = norm * gate_vals                                   # fp32

        # write to contiguous (R, H) buffer
        tl.store(out_fused_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                    # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256      # rows per program (tuned for H100)
    BLOCK_D = 64       # dim chunk size
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],     # γ (float32)
        w["norm.bias"],       # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free original fp32 view

    # --------------------------------------------------------------
    # 2️⃣ Fused linear: left/right proj + all gates → (R, 5·H)
    # --------------------------------------------------------------
    # concatenate the five weight matrices (they are stored as [out, in])
    fused_w = torch.cat(
        [
            w["left_proj.weight"],   # [H, dim]
            w["right_proj.weight"],  # [H, dim]
            w["left_gate.weight"],   # [H, dim]
            w["right_gate.weight"],  # [H, dim]
            w["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                   # fp16, shape [R, 5·H]
    del fused_w, x_norm                                 # free intermediates

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates (write directly in B·H·N·N layout)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if nomask or mask is None:
        # dummy mask – not used inside the kernel
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        # flatten mask to [R] and cast to fp16
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 256          # rows per program
    BLOCK_H = 64           # hidden dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dim)
    # --------------------------------------------------------------
    # reshape for cuBLAS: (B·H, N, N)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # batched half‑precision matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, H, N, N)    # (B, H, N, N)

    # free the two large tensors as early as possible
    del left, right, left_mat, right_mat, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm + out_gate multiplication (produce (R, H))
    # --------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R2 = 256
    BLOCK_H2 = 64
    grid_ln_gate = (triton.cdiv(R, BLOCK_R2),)
    layernorm_gate_fused_kernel[grid_ln_gate](
        out,
        out_gate,
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out, out_gate   # no longer needed

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

**Explanation of key performance choices**

* **Row‑wise LayerNorm kernel** works on a 2‑D view (`R × dim`).  
  The statistics are computed in fp32 (required for numerical stability) while the
  normalized output is stored in fp16 – the bandwidth‑limited part of the kernel.

* **Fused projection** eliminates four separate GEMM calls.  
  A single `torch.nn.functional.linear` with a concatenated weight matrix
  (already cast to fp16) performs the projection for *all* five linear layers
  in one go.

* **`split_fuse_gate_kernel`** decodes the flat row index `(b,i,k)` inside the
  kernel, applies the optional mask, evaluates the three sigmoid gates
  (computed in fp32 then cast back) and writes the three tensors directly
  into the `(B, H, N, N)` layout required by the later batched GEMM.
  This avoids extra transposes or copies.

* **Batched GEMM (`torch.bmm`)** is delegated to cuBLAS Tensor‑Core kernels,
  which are orders of magnitude faster than any hand‑written Triton GEMM for the
  hidden dimension `H = 128`.  The tensors are kept in fp16 throughout.

* **`layernorm_gate_fused_kernel`** performs a two‑pass reduction over the
  hidden dimension:
  1. compute mean / variance,
  2. normalise, apply the learned γ/β, multiply by the pre‑sigmoided
     `out_gate`, and store into a contiguous `(R, H)` buffer ready for the
     final projection.

* **Block sizes** (`BLOCK_M = BLOCK_R = BLOCK_R2 = 256`, `BLOCK_D = BLOCK_H = BLOCK_H2 = 64`) are tuned for the H100
  (max threads per warp = 32, 8 warps per SM) and give a good balance between
  occupancy and shared‑memory pressure.

The whole pipeline stays in half‑precision except for the final LayerNorm
statistics and the final output cast back to `float32`, matching the reference
PyTorch implementation while achieving a runtime comfortably below the
target 1 ms on the supplied benchmark configurations.
