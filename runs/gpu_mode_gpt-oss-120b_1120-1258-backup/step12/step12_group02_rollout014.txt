"""
High‑performance implementation of the *outgoing* TriMul operator
(AlphaFold‑3) using a mix of PyTorch and Triton.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D tensor (B·N·N × dim) – Torch
    implementation (highly optimized on H100).  Result cast to fp16.

2️⃣  Single Linear projection producing the five slices
    (left, right, left_gate, right_gate, out_gate) in one pass.

3️⃣  Apply the optional pair‑wise mask, sigmoid gates and reshape
    `left`/`right` directly into the layout required for the batched GEMM
    ((B·H, N, N)).  This is done with pure PyTorch tensor operations
    (no extra memory copies beyond the necessary views).

4️⃣  Batched GEMM (Tensor‑Core) :   out = left × rightᵀ
    → shape (B·H, N, N)  (fp16).

5️⃣  Second LayerNorm over the hidden dimension **and** element‑wise
    multiplication with `out_gate` – performed by a custom Triton
    kernel (`layernorm_mul_strided_kernel`).  The kernel reads the
    strided GEMM output, computes mean/var per (b,i,j) position,
    normalises, applies the learned γ/β and finally multiplies by the
    gate vector.

6️⃣  Final Linear projection back to `dim` (fp16 → fp32).

Only the second LayerNorm is implemented in Triton; all other heavy
arithmetic (projections, GEMM, first LayerNorm) uses the highly tuned
PyTorch / cuBLAS kernels, giving a fast overall runtime while keeping
the code simple and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm (over hidden) + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_strided_kernel(
    x_ptr,          # const   *[B·H, N, N]   (fp16)   (GEMM result)
    gate_ptr,       # const   *[rows, H]    (fp16)   (out_gate)
    y_ptr,          # mutable *[rows, H]    (fp16)   (fused output)
    weight_ptr,     # const   *[H]         (fp32)   γ
    bias_ptr,       # const   *[H]         (fp32)   β
    rows: tl.int32,       # = B·N·N
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_r = offs_r < rows

    seqsq = N * N                                     # stride between hidden slices

    # ---- Decode (b,i,j) from flat row index ----
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # base offset for the first hidden slice (h = 0)
    base0 = b_idx * H * seqsq + i_idx * N + j_idx      # [BLOCK_M]

    # --------------------------------------------------------------
    # 1️⃣ Compute mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Load a tile of the hidden dimension (strided)
        vals = tl.load(
            x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)          # [M, Hblk]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Normalise, apply γ/β, multiply with out_gate, store (fp16)
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(
            x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)          # [M, Hblk]

        # (v - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        # γ, β (fp32)
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # [Hblk]
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # [Hblk]

        norm = norm * w[None, :] + b[None, :]               # fp32

        # out_gate (fp16 → fp32)
        gate = tl.load(
            gate_ptr + offs_r[:, None] * H + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        out = norm * gate                                   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(
            y_ptr + offs_r[:, None] * H + offs_h[None, :],
            out_fp16,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim`` and ``hidden_dim``. May also
                         contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                    # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (torch – fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                              # [R, dim] fp32
    # torch.LayerNorm is heavily optimized; we keep everything in fp32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                               # [R, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Concatenated Linear projection (left/right + 3 gates)
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                          # shape [5*H, dim]

    proj = F.linear(x_norm, w_concat)            # (R, 5*H) fp16
    del w_concat, x_norm                         # free

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid gates, reshape left/right
    # ------------------------------------------------------------------
    proj = proj.view(R, 5, H)                    # [R, 5, H]

    left_raw   = proj[:, 0, :]                   # [R, H]
    right_raw  = proj[:, 1, :]                   # [R, H]
    left_gate  = torch.sigmoid(proj[:, 2, :])    # [R, H] fp16
    right_gate = torch.sigmoid(proj[:, 3, :])    # [R, H] fp16
    out_gate   = torch.sigmoid(proj[:, 4, :])    # [R, H] fp16
    del proj                                     # no longer needed

    # optional mask (broadcast across hidden dim)
    if cfg.get("nomask", True) or mask is None:
        mask_vec = None
    else:
        mask_vec = mask.view(R).to(torch.float16).unsqueeze(-1)  # [R,1]

    if mask_vec is not None:
        left  = left_raw  * mask_vec * left_gate
        right = right_raw * mask_vec * right_gate
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate
    # clean up temporaries
    del left_raw, right_raw, left_gate, right_gate, mask_vec

    # reshape to (B·H, N, N) layout expected by the batched GEMM
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left  = left.view(B * H, N, N)      # (B·H, N, N)
    right = right.view(B * H, N, N)     # (B·H, N, N)

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (Tensor‑Core)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16
    del left, right                                    # free memory

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden) + element‑wise multiply with out_gate
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M = 256                     # rows per program (larger than before)
    BLOCK_H = 32                      # hidden‑dim tile size (fits H=128 nicely)

    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_strided_kernel[grid](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32, [H])
        weights["to_out_norm.bias"],     # β (float32, [H])
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
