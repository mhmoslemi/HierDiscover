#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
TriMul (outgoing) – high‑performance Triton + cuBLAS implementation
===================================================================

The forward pass consists of six stages:

1️⃣ **Row‑wise LayerNorm** (float32 → fp16).  The input tensor of shape
   [B,N,N,dim] is reshaped to [R,dim] (R = B·N²).  The kernel computes the
   mean/variance per‑row in fp32, normalises, applies the learned γ/β
   and writes the result as fp16.

2️⃣ **Fused linear projection** (5·H rows).  All five linear maps
   (left‑proj, right‑proj, left‑gate, right‑gate, out‑gate) share the same
   weight matrix – we concatenate them and call ``torch.nn.functional.linear``.
   The result has shape [R,5·H] (half‑precision).

3️⃣ **Split & mask + sigmoid gates**.  A Triton kernel reads the fused
   projection, optionally multiplies by the pair‑wise mask, applies a stable
   sigmoid (computed in fp32) and writes the three tensors directly in the
   layout required by the batched GEMM:
   ``left``   – [B,H,N,N] (fp16)  
   ``right``  – [B,H,N,N] (fp16)  
   ``out_gate`` – [B,H,N,N] (fp16).

4️⃣ **Tri‑multiplicative update** – a batched half‑precision matrix
   multiplication performed with ``torch.bmm`` (cuBLAS Tensor‑Core kernel):
   ``out[b·H+i] = left[b·H+i] @ right[b·H+i].T``.

5️⃣ **Second LayerNorm + out‑gate fusion**.  A second Triton kernel
   normalises the intermediate tensor ``out`` along the hidden dimension H,
   applies the learned γ/β, multiplies by the already‑sigmoided ``out_gate``,
   and writes a contiguous buffer ``out_fused`` of shape [R,H] (fp16).

6️⃣ **Final linear projection** back to the model dimension ``dim`` with a
   half‑precision GEMM (again ``torch.nn.functional.linear``).  The result is
   cast back to float32 and reshaped to [B,N,N,dim].

Only three custom Triton kernels are needed; all heavy matrix multiplications
are handed to cuBLAS, which gives near‑peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # const   *[rows, dim]   (float32)
    y_ptr,        # mutable *[rows, dim]   (float16)
    w_ptr,        # const   *[dim]        (float32) gamma
    b_ptr,        # const   *[dim]        (float32) beta
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,    # number of rows processed per program
    BLOCK_D: tl.constexpr,    # chunk size along dim
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --- write normalised values -----------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#     write directly in (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask broadcast (if any)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                     # distance between consecutive h in the layout

    # base offset of the first hidden slice (h = 0)
    base_offset = b * (H * nn) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # proj layout: [R, 5*H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw     = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(proj_ptr + proj_base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate    = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate   = tl.load(proj_ptr + proj_base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # stable sigmoid in fp32 → back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offsets (B,H,N,N) layout
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm + out_gate multiplication
#     (writes into contiguous (R, H) buffer ready for the final linear)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_fused_kernel(
    out_ptr,            # const   *[B, H, N, N] (fp16)
    gate_ptr,           # const   *[B, H, N, N] (fp16)
    out_fused_ptr,      # mutable *[R, H]       (fp16)
    w_ptr,              # const   *[H]           (float32) gamma
    b_ptr,              # const   *[H]           (float32) beta
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # offset of the first hidden slice (h = 0) for this (b,i,j)
    base_offset = b * (H * nn) + i * N + j                 # [BLOCK_R]

    # ------- first pass: compute mean & variance over H --------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(out_ptr + base_offset[:, None] +
                       offs_h[None, :] * nn,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------- second pass: normalise, apply γ/β, multiply by gate -----------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        out_vals = tl.load(out_ptr + base_offset[:, None] +
                           offs_h[None, :] * nn,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        gate_vals = tl.load(gate_ptr + base_offset[:, None] +
                            offs_h[None, :] * nn,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        gamma = tl.load(w_ptr + offs_h, mask=mask_h, other=0.0)
        beta  = tl.load(b_ptr + offs_h, mask=mask_h, other=0.0)

        norm = (out_vals - mean[:, None]) * rstd[:, None]          # fp32
        norm = norm * gamma[None, :] + beta[None, :]                # fp32
        fused = norm * gate_vals                                    # fp32

        # store into contiguous buffer [R, H]
        tl.store(out_fused_ptr + offs_r[:, None] * H + offs_h[None, :],
                 fused.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])

# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # flat rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                    # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    # choose block sizes based on problem size (empirically good for H100)
    BLOCK_M = 128 if R > 1024 else 64
    BLOCK_D = 64 if dim > 64 else dim
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],   # gamma  (float32)
        w["norm.bias"],     # beta   (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat   # free original tensor

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H rows)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],   # [H, dim]
            w["right_proj.weight"],  # [H, dim]
            w["left_gate.weight"],   # [H, dim]
            w["right_gate.weight"],  # [H, dim]
            w["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                               # [5*H, dim]

    # Linear: x_norm (R, dim) @ fused_w.T (dim, 5*H) -> (R,5*H)
    proj = F.linear(x_norm, fused_w)                  # fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates  → (B, H, N, N) layout
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask is bool/float, broadcast over hidden dimension
        mask_flat = mask.to(torch.float16).reshape(R)                # [R]
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (tri‑multiplicative update) - cuBLAS Tensor Cores
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) contiguous for torch.bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, shape (B·H,N,N)

    out = out_mat.view(B, H, N, N)                            # back to (B,H,N,N)

    del left, right   # not needed any more

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate (writes (R, H) buffer)
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R2 = 128
    BLOCK_H2 = 32
    grid_ln_gate = (triton.cdiv(R, BLOCK_R2),)

    layernorm_gate_fused_kernel[grid_ln_gate](
        out,
        out_gate,
        out_fused,
        w["to_out_norm.weight"],   # gamma (float32)
        w["to_out_norm.bias"],     # beta  (float32)
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # fp16, shape (R, dim)

    final = final_half.view(B, N, N, dim).to(torch.float32)   # cast back to fp32

    return final

**Explanation of the performance choices**

* **Row‑wise LayerNorm** works on 1‑D rows, which yields perfect coalesced loads/stores.
  Mean/variance are accumulated in fp32 to retain numerical stability, then the
  normalised result is written in fp16 (the precision used by the rest of the
  network).

* **Fused projection** reduces five separate `torch.nn.Linear` calls to a **single**
  GEMM, decreasing memory traffic dramatically.

* **`split_fuse_gate_kernel`** reads the fused projection only once, applies the
  optional mask, the three sigmoids, and writes the three tensors directly in the
  layout required by the batched GEMM.  This eliminates an intermediate transpose
  and an extra kernel launch.

* **Batched GEMM** (`torch.bmm`) leverages cuBLAS Tensor‑Core kernels on the H100,
  which are far faster than a hand‑written Triton GEMM for the relatively large
  matrices (`N × N` with `N` up to 1024).

* **Second LayerNorm / out‑gate** is fused into a single kernel that performs both
  the normalisation (again in fp32) and the element‑wise multiplication by the
  already‑sigmoided gate.  The result is stored contiguously as `[R, H]`,
  ready for the final linear projection.

* **Final projection** is again a single half‑precision GEMM; the result is cast
  back to float32 only at the very end.

Only three Triton kernel launches are required, each operating on contiguous
memory regions.  All heavy linear algebra is delegated to cuBLAS, giving the
implementation well‑under the target 1 ms runtime on an H100 for the benchmark
configurations.
