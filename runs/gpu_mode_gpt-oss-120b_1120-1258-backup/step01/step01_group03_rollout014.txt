"""
TriMul “outgoing” kernel (AlphaFold‑3) – highly‑optimised forward pass.

Algorithm
---------
1. Row‑wise LayerNorm over the channel dimension `dim`.
   A single Triton kernel computes mean / variance in FP32
   and writes the normalised tensor in either FP32 or FP16
   (controlled by the compile‑time flag `IS_HALF`).

2. Fuse the five linear projections (left/right projection,
   three gates) into one GEMM:
        y = x_norm @ W_fused.T          # 5·hidden_dim outputs
   The weight matrix is built by concatenating the five
   weight tensors once and stored in FP16.  The result is split
   into `left_proj`, `right_proj`, `left_gate`, `right_gate`,
   `out_gate`.

3. Optional mask (broadcast over the hidden dimension) is applied
   to the two projections before gating.

4. Apply sigmoid gates and form the gated tensors `left` and `right`.

5. Batched matrix‑multiply:
        out[b, h] = left[b, h] @ right[b, h].T
   Implemented with `torch.bmm` on FP16 tensors (Tensor‑Core fast).

6. Second LayerNorm over `hidden_dim`.  The same Triton kernel
   (now with `IS_HALF=1`) operates directly on the FP16 result,
   eliminating extra FP32↔FP16 copies.

7. Multiply by the output‑gate (`out_gate`), then a final
   FP16 linear projection back to `dim`.

8. Cast the result to FP32 and return.

The implementation keeps the heavy reductions in Triton,
the large GEMMs in PyTorch (which uses cuBLAS/Tensor‑Cores),
and runs entirely in FP16 except for the two LayerNorm reductions,
meeting the required correctness while dramatically reducing runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 or FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # *const   float*   [total_rows, dim]
    y_ptr,            # *mut     float*   [total_rows, dim]
    weight_ptr,       # *const   float*   [dim]
    bias_ptr,         # *const   float*   [dim]
    total_rows,       # int (runtime)
    dim: tl.constexpr,      # int (compile‑time)
    eps: tl.constexpr,      # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per inner loop
    IS_HALF: tl.constexpr = 0,   # 0 → FP32 I/O, 1 → FP16 I/O
):
    """
    Normalises each row of a 2‑D tensor.  The kernel works for
    both FP32 (IS_HALF=0) and FP16 (IS_HALF=1).  Accumulations are
    always performed in FP32 for numerical stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass – compute sum and sum of squares
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    val_dtype = tl.float16 if IS_HALF else tl.float32

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=val_dtype,
        ).to(tl.float32)                                      # cast to FP32
        sum_val += tl.sum(vals, axis=1)                       # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)                # row‑wise sum‑of‑squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass – write normalised values with γ (weight) and β (bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=val_dtype,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0,
                    dtype=val_dtype).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0,
                    dtype=val_dtype).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :],
                 dtype=val_dtype)

# ----------------------------------------------------------------------

    # ------------------------------------------------------------------

    hidden_dim = config["hidden_dim"]

    w_fused_h = w_fused.to(torch.float16)

    # (total_rows, 5*hidden_dim) in FP16
    y = F.linear(x_norm_h, w_fused_h)

    # split into the five components
    y = y.view(total_rows, 5, hidden_dim)   # [R, 5, H]
    left_proj = y[:, 0, :]       # [R, H]
    right_proj = y[:, 1, :]
    left_gate  = y[:, 2, :]
    right_gate = y[:, 3, :]
    out_gate   = y[:, 4, :]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.to(torch.float16).contiguous().view(total_rows, 1)  # [R,1]
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 4) Apply sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)   # FP16
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left_proj * left_gate   # [R, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Reshape for batched GEMM (L @ Rᵀ) – each head is a separate batch
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → permute to [B, H, N, N]
    left  = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # collapse batch & head dimensions for a single bmm call
    BH = B * hidden_dim
    left_mat  = left.view(BH, N, N)          # [BH, N, N]
    right_mat = right.view(BH, N, N)

    # L @ Rᵀ for every head
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]  (FP16)

    # reshape back to [B, H, N, N]
    out = out_mat.view(B, hidden_dim, N, N).contiguous()        # FP16

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden_dim (operates on FP16 data)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    # bring to shape [B, N, N, H] → flatten to [R_h, H]
    out_flat_fp16 = out.permute(0, 2, 3, 1).contiguous().view(total_rows_h, hidden_dim)

    out_norm_flat = torch.empty_like(out_flat_fp16)   # FP16 output
    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat_fp16,
        out_norm_flat,
        weights["to_out_norm.weight"].to(torch.float16),
        weights["to_out_norm.bias"].to(torch.float16),
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M,
        BLOCK_D,
        IS_HALF=1,               # FP16 I/O
    )
    torch.cuda.synchronize()

    # reshape back to [B, H, N, N]
    out_norm = out_norm_flat.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # ------------------------------------------------------------------
    # 7) Apply the output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    out_norm = out_norm * out_gate                                              # element‑wise FP16

    # flatten to [R_h, H] for the final linear projection
    out_norm_flat2 = out_norm.permute(0, 2, 3, 1).contiguous().view(total_rows_h, hidden_dim)

    to_out_weight_h = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    final_flat = F.linear(out_norm_flat2, to_out_weight_h)       # [R_h, dim]  FP16

    final = final_flat.view(B, N, N, dim).to(torch.float32)      # required output dtype

    return final
```
