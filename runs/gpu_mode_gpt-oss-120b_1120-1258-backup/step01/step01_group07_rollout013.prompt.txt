import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B,N,N,C).
2. Project to a hidden dimension `H` with two linear layers (left/right)
   and optionally mask the result.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Apply the left/right gates element‑wise.
5. The core operation is a batched matrix multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batch of GEMMs: for each (b,h) compute
        L_{i,k} @ R_{k,j}ᵀ .
   This is performed by a fused Triton kernel that processes each
   (b·h) group as an independent matrix‑multiply.
6. Layer‑norm the result, multiply by the out‑gate and finish with a
   linear projection back to `C`.

The heavy O(N³) work is done in the Triton kernel, all other ops use
high‑performance PyTorch primitives.  Mixed‑precision (FP16) is used
for the large tensors, while the accumulation and final layers stay in
FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (M×K)·(K×N)→(M×N)
    stride_ag, stride_am, stride_ak,   # A: group, row, col strides
    stride_bg, stride_bk, stride_bn,   # B: group, row(K), col(N) strides
    stride_cg, stride_cm, stride_cn,   # C: group, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul:   C[g, i, j] = Σ_k A[g, i, k] * B[g, k, j]"""

    pid = tl.program_id(0)   # tile id within the (M,N) plane
    pid_g = tl.program_id(1) # batch/group id (g = b*H + h)

    # ------------------------------------------------------------------
    # Compute tile coordinates
    # ------------------------------------------------------------------
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_n_tiles    # block row index
    pid_n = pid %  num_n_tiles    # block col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [0..BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [0..BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current batch/group
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_g * stride_ag
    b_batch = b_ptr + pid_g * stride_bg
    c_batch = c_ptr + pid_g * stride_cg

    # ------------------------------------------------------------------
    # Accumulator (FP32 for precision)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M × K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16 → promotes to fp32 on dot

        # B tile (K × N)
        b = tl.load(
            b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate dot product
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # ------------------------------------------------------------------
    # Write result back (FP32)
    # ------------------------------------------------------------------
    c = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim].
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]            # input channel dimension C
    hidden_dim = config["hidden_dim"]  # projection dimension H
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # Layer‑norm on the input (FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,C]  FP32

    # ------------------------------------------------------------------
    # Linear projections (cast to FP16 for speed)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]

    left = F.linear(x, left_proj_w).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if use_mask:
        # mask: [B,N,N] -> [B,N,N,1] -> fp16
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))          # keep FP32

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM:
    #   left  : [B, H, N, K]  where K = N (the summed dimension)
    #   right : [B, H, K, N]  (transpose of the second N axis)
    # ------------------------------------------------------------------
    # left: (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,H) -> (B,H,N,N) with axes (k,i) swapped => (B,H,K,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    G = B * H                               # total groups for the kernel

    # Collapse batch & head dimensions to a single leading dimension
    A = left_perm.view(G, N, N)   # shape (G, M=N, K=N)   fp16
    B_ = right_perm.view(G, N, N) # shape (G, K=N, N)   fp16

    # Output buffer (FP32 accumulation)
    C = torch.empty_like(A, dtype=torch.float32, device=device)  # (G,N,N)

    # ------------------------------------------------------------------
    # Compute strides for the kernel (group, row, column)
    # ------------------------------------------------------------------
    stride_ag = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bg = B_.stride(0)
    stride_bk = B_.stride(1)   # row stride (over K)
    stride_bn = B_.stride(2)   # column stride (over N)

    stride_cg = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    M = N
    K = N

    # ------------------------------------------------------------------
    # Tile sizes – chosen to fit H100 shared memory & registers
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (tiles over M×N, number of groups)
    num_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, G)

    # Launch the kernel
    _batch_matmul_kernel[grid](
        A,
        B_,
        C,
        M,
        N,
        K,
        stride_ag,
        stride_am,
        stride_ak,
        stride_bg,
        stride_bk,
        stride_bn,
        stride_cg,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # Reshape back to [B,N,N,H]
    # ------------------------------------------------------------------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]  FP32

    # ------------------------------------------------------------------
    # Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                     # FP32

    out = out * out_gate                  # FP32

    to_out_w = weights["to_out.weight"]
    output = F.linear(out, to_out_w)      # [B,N,N,dim]   FP32

    return output
