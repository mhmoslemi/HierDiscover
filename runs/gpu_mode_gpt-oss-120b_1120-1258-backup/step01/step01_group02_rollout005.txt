    # Cast to fp16 for projection & gating
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)  # shape [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w)

"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm the input (float32 → float16 for the heavy work).
2. Linear projections → left / right tensors of shape [B,N,N,hidden_dim].
3. Optional mask (broadcast on the hidden dimension) and element‑wise gates.
4. Fuse the N³ contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a **batched GEMM** `C = A @ Bᵀ` for each hidden dimension.
   The kernel works on tiles (BLOCK_M×BLOCK_N) and streams K‑tiles
   (BLOCK_K) through the shared memory, using FP16 for the
   matrix elements and FP32 accumulation (Tensor‑cores).
5. Reshape back to [B,N,N,hidden_dim] and apply the output LayerNorm,
   out‑gate, and final linear projection to the original `dim`.

The kernel runs on H100 (AMPERE+ Tensor‑cores) with 8 warps,
BLOCK_M = BLOCK_N = 128 and BLOCK_K = 64 – a sweet spot for the
tested sequence lengths (up to 1024).  All other ops are done with
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C[b] = A[b] @ B[b]ᵀ  (A: M×K, B: N×K)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                    # fp16 *  (batch, M, K)
    b_ptr,                    # fp16 *  (batch, K, N)   ← will be accessed as (K,N)
    c_ptr,                    # fp16 *  (batch, M, N)
    M, N, K,                  # matrix dimensions (int32)
    stride_az, stride_am, stride_ak,  # strides for A
    stride_bz, stride_bk, stride_bn,  # strides for B
    stride_cz, stride_cm, stride_cn,  # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM with FP32 accumulation.
    The first program‑id dimension packs (batch, tile_m), the second is tile_n.
    """
    pid0 = tl.program_id(0)          # batch_id * tiles_M + tile_m
    pid1 = tl.program_id(1)          # tile_n

    # -------------------------------------------------------------
    # Decode batch index and tile coordinates
    # -------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id   = pid0 // num_tiles_m
    tile_m     = pid0 %  num_tiles_m
    tile_n     = pid1

    # -------------------------------------------------------------
    # Offsets inside the current tile
    # -------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the tile (without the k‑loop offset)
    a_ptrs = a_ptr + batch_id * stride_az \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # -------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Main K‑loop
    # -------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # [BLOCK_M, BLOCK_K] fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # [BLOCK_K, BLOCK_N] fp16

        # fp16⋅fp16 → fp32 accumulation (uses Tensor‑cores)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # Store the result (back to fp16)
    # -------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.
    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
                         {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32 for stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear / gating work
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Linear weights → FP16
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # -------------------------------------------------------------
    # Projections
    # -------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden]

    # -------------------------------------------------------------
    # Optional mask (broadcast on hidden dimension)
    # -------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------------------
    # Gating scalars (sigmoid)
    # -------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3) N³ contraction via a fused batched GEMM (A @ Bᵀ)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape   # shape: [B, N, N, hidden_dim]

    # Rearrange to (batch*hidden, M, K) / (batch*hidden, K, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP16, accumulation will be FP32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -------------------------------------------------------------
    # Triton launch configuration (tuned for H100)
    # -------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64          # larger K‑tile improves Tensor‑core utilisation

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)

    # pid0 = batch_id * grid_m + tile_m   → total first‑dim size = B*H*grid_m
    grid = (B * hidden_dim * grid_m, grid_n)

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 thrives with 8 warps per block
        num_stages=3,         # three stages for better latency hiding
    )

    # -------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and cast to FP32
    # -------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 4) Output LayerNorm, out‑gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the out‑gate (broadcasted on the hidden dimension)
    out = out * out_gate.to(torch.float32)

    # Final linear projection (hidden_dim → dim)
    out = F.linear(out, weights["to_out.weight"])   # bias = False

    return out
```

**Explanation of key choices**

* **Tiling** – 128 × 128 tiles for `M`/`N` match the H100 shared‑memory capacity while giving each warp enough work.  
* **BLOCK_K = 64** – larger K‑tiles improve Tensor‑core utilisation for the 1024‑long inner dimension.  
* **FP16 → FP32 accumulation** – retains the numerical quality of LayerNorms while letting the kernel exploit fast Tensor‑cores.  
* **8 warps / 3 stages** – empirically a good balance for Hopper GPUs, providing sufficient parallelism and pipeline depth.  
* **All non‑N³ work is done in PyTorch** – this keeps the kernel small and focused, yet the dominant O(hidden × N³) contraction is fully fused and runs entirely on the GPU.
