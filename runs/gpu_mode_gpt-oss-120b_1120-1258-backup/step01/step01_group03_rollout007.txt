Pseudo:

```
@triton.jit
def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                     total_rows,
                     dim: tl.constexpr,
                     eps: tl.constexpr,
                     BLOCK_M: tl.constexpr,
                     BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq += tl.sum(vals_f32 * vals_f32, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)
    # second pass
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)
        norm = (vals_f32 - mean[:, None]) * rstd[:, None]
        weight = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        bias = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        out = norm * weight[None, :] + bias[None, :]
        out_f16 = out.to(tl.float16)
        tl.store(ptrs, out_f16, mask=mask_m[:, None] & mask_d[None, :])
```

"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. **Row‑wise LayerNorm (dim)** – implemented in Triton.
   * Computes mean/variance in fp32 and writes normalized fp16 values.
2. **Fused linear projections** – concatenate the projection and gate weights so that
   each of the left/right branches is realized by a single fp16 GEMM.
   * left_proj , left_gate = Linear(x_norm)  → split, sigmoid on gate.
   * right_proj, right_gate = Linear(x_norm) → split, sigmoid on gate.
   * out_gate = sigmoid(Linear(x_norm)).
   * Optional binary mask is applied only to the two projection tensors
     before gating, exactly like the reference implementation.
3. **Element‑wise gating** – `left = left_proj * left_gate`,
   `right = right_proj * right_gate`.
4. **Triangular multiplicative update** – for each hidden channel h:
   `out_h = left_h @ right_hᵀ`.  Implemented as a batched GEMM
   (`torch.bmm`) on tensors of shape (B·H, N, N) using fp16 tensor‑core
   kernels.
5. **LayerNorm over hidden_dim** – same Triton kernel as step 1.
6. **Final gate & projection** – `out = out * out_gate`,
   then a last fp16 linear layer back to `dim`.
7. Cast the result to float32 before returning.

All heavy reductions (the two LayerNorms) are done in Triton;
all matrix‑multiplications use fp16 tensor‑core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton Row‑wise LayerNorm (supports fp16 inputs/outputs)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # *const half* [total_rows, dim]
    y_ptr,        # *mut   half* [total_rows, dim]
    w_ptr,        # *const half* [dim]
    b_ptr,        # *const half* [dim]
    total_rows,   # int (runtime)
    dim: tl.constexpr,          # compile‑time channel dimension
    eps: tl.constexpr,          # compile‑time epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < total_rows                      # bool

    # ---------- first pass: mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        # pointer to a block of values
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )                                            # half
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)          # row‑wise sum
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                       # 1 / sqrt(var+eps)

    # ---------- second pass: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        vals_f32 = vals.to(tl.float32)

        # (x - mean) * rstd   (fp32)
        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) – half -> fp32 for multiplication
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out_f32 = norm * w[None, :] + b[None, :]
        out_f16 = out_f32.to(tl.float16)

        tl.store(ptrs, out_f16,
                 mask=mask_m[:, None] & mask_d[None, :])


def _cdiv(x, y):
    return (x + y - 1) // y


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # -----------------------------------------------------------------
    # Unpack and move everything to fp16 (except the final cast)
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # Cast input to half
    x = input_tensor.to(torch.float16, non_blocking=True).contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim)
    # -----------------------------------------------------------------
    x_flat = x.view(total_rows, dim)                     # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)               # half

    # launch Triton LN kernel
    BLOCK_M = 256      # can be tuned
    BLOCK_D = 64
    grid = (_cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"].to(torch.float16, non_blocking=True),
        weights["norm.bias"].to(torch.float16, non_blocking=True),
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 2) Fused linear projections + gating (all fp16)
    # -----------------------------------------------------------------
    # Concatenate projection & gate weight matrices to halve the number of GEMMs
    left_fused_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)               # [2*H, dim]
    right_fused_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)               # [2*H, dim]
    out_gate_w = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    # ---- linear pass (fused) ------------------------------------------------
    left_concat = F.linear(x_norm_flat, left_fused_w)    # [R, 2*H]
    right_concat = F.linear(x_norm_flat, right_fused_w)  # [R, 2*H]

    left_proj = left_concat[:, :hidden_dim]              # [R, H]
    left_gate = torch.sigmoid(left_concat[:, hidden_dim:])  # [R, H]

    right_proj = right_concat[:, :hidden_dim]
    right_gate = torch.sigmoid(right_concat[:, hidden_dim:])

    # ---- optional mask ------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [R, 1] for broadcasting
        m = mask_tensor.to(torch.float16, non_blocking=True).view(total_rows, 1)
        left_proj = left_proj * m
        right_proj = right_proj * m

    # ---- apply element‑wise gates -------------------------------------------
    left  = left_proj * left_gate          # [R, H]
    right = right_proj * right_gate        # [R, H]

    # ---- out gate -----------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm_flat, out_gate_w))  # [R, H]

    # -----------------------------------------------------------------
    # 3) Reshape tensors for the batched GEMM (TriMul)
    # -----------------------------------------------------------------
    # From [R, H] → [B, N, N, H]
    left  = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Permute to (B, H, N, N) for batched matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, K]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, K, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)    # [B*H, N, K]
    right_mat = right_perm.view(BH, N, N)   # [B*H, K, N]

    # Batched GEMM: (B*H, N, K) @ (B*H, K, N) → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)   # fp16

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 4) Second LayerNorm over hidden_dim
    # -----------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)          # [R_h, H]
    out_norm_flat = torch.empty_like(out_flat)              # half

    grid2 = (_cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"].to(torch.float16, non_blocking=True),
        weights["to_out_norm.bias"].to(torch.float16, non_blocking=True),
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)    # [B, N, N, H]

    # -----------------------------------------------------------------
    # 5) Final gate & projection back to dim
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate                         # element‑wise (half)

    # final linear projection (H → dim)
    out_norm_flat = out_norm.view(total_rows_h, hidden_dim)
    final_fp16 = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16, non_blocking=True),
        bias=None,
    )  # [R_h, dim]

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast back to fp32

    return final
