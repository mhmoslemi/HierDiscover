"""
TriMul (outgoing) – fast Triton implementation
================================================

The module operates on a 4‑D tensor `x` of shape [B, N, N, C] and produces a
tensor of the same shape.  The forward pass consists of

1️⃣  Layer‑norm the input (FP32 for numerical stability).
2️⃣  Two linear projections (`left_proj`, `right_proj`) from C → H (H = hidden_dim)
    – we cast the result to fp16.
3️⃣  Optional masking (broadcast over H) and three sigmoid gates
    (`left_gate`, `right_gate`, `out_gate`).  The first two gates are applied
    element‑wise in fp16, the output gate stays in FP32.
4️⃣  Core “tri‑multiplicative” update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    which is a batched GEMM for each head `h`.  The computation is fused into
    a single Triton kernel that processes all groups `g = b * H` in parallel.
    The kernel works on fp16 inputs and accumulates in FP32 (tensor‑core
    friendly ``tl.dot``).
5️⃣  Layer‑norm over the hidden dimension, multiplication by the output gate,
    and a final linear projection back to C.

The heavy O(N³) work is therefore done by the Triton kernel while the
light‑weight surrounding operations use high‑performance PyTorch primitives.
Mixed‑precision (fp16 for large tensors, fp32 for the accumulators and final
layers) yields a good trade‑off between speed and accuracy on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to tensors A, B, C
    M, N, K,                           # matrix dimensions (M×K)·(K×N)→(M×N)
    stride_ag, stride_am, stride_ak,   # A: group, row, col strides
    stride_bg, stride_bk, stride_bn,   # B: group, row(K), col(N) strides
    stride_cg, stride_cm, stride_cn,   # C: group, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication:
    C[g, i, j] = Σ_k A[g, i, k] * B[g, k, j]   for all groups g.
    A and B are fp16, accumulation is done in fp32.
    """

    pid = tl.program_id(0)      # tile index inside the (M, N) plane
    pid_g = tl.program_id(1)    # group index (g = batch * hidden_dim)

    # -------------------------------------------------------------
    # Compute the tile coordinates (row block, column block)
    # -------------------------------------------------------------
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_n_tiles            # block row index
    pid_n = pid %  num_n_tiles            # block column index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # Pointers to the start of the current group
    # -------------------------------------------------------------
    a_batch = a_ptr + pid_g * stride_ag
    b_batch = b_ptr + pid_g * stride_bg
    c_batch = c_ptr + pid_g * stride_cg

    # -------------------------------------------------------------
    # Accumulator in fp32
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension in tiles
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16 → up‑cast to fp32 inside tl.dot

        # Load B tile (K × N)
        b = tl.load(
            b_batch
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # Accumulate dot product (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # Write the result (FP32)
    # -------------------------------------------------------------
    c = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (see docstring above)
        - config       : dict containing:
              * "dim"         – input channel dimension C
              * "hidden_dim" – projection dimension H
              * "nomask"      – bool, if True no mask will be applied

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  Input layer‑norm (FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,C]  FP32

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections – cast to fp16 for speed
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]

    left  = F.linear(x, left_proj_w).to(torch.float16)   # [B,N,N,H] fp16
    right = F.linear(x, right_proj_w).to(torch.float16) # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if use_mask:
        # mask: [B,N,N] → [B,N,N,1] → fp16
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid).  Gating tensors are kept in fp16 except
    #    out_gate which stays in FP32 for the later multiplication.
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # FP32

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Prepare tensors for the batched GEMM:
    #     left  → (B, H, N, N)   with axes (i,k)
    #     right → (B, H, N, N)   with axes (k,j)  (second N swapped)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B,H,N,N]

    B, H, N, _ = left_perm.shape          # N == K
    G = B * H                               # total groups for the kernel

    # Collapse batch & head dimensions → shape (G, N, N)
    A = left_perm.view(G, N, N)   # fp16
    B_ = right_perm.view(G, N, N) # fp16

    # Output buffer – accumulation in FP32
    C = torch.empty_like(A, dtype=torch.float32, device=device)  # (G,N,N)

    # ------------------------------------------------------------------
    # Compute strides (in elements) for the kernel
    # ------------------------------------------------------------------
    stride_ag = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bg = B_.stride(0)
    stride_bk = B_.stride(1)   # row stride (over K)
    stride_bn = B_.stride(2)   # column stride (over N)

    stride_cg = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # ------------------------------------------------------------------
    # Choose tile sizes – a simple heuristic based on N
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32

    M = N
    K = N

    # Grid dimensions: (tiles over M×N, number of groups)
    num_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, G)

    # ------------------------------------------------------------------
    # Launch the Triton kernel
    # ------------------------------------------------------------------
    _batched_matmul_kernel[grid](
        A,
        B_,
        C,
        M, N, K,
        stride_ag,
        stride_am,
        stride_ak,
        stride_bg,
        stride_bk,
        stride_bn,
        stride_cg,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps (256 threads) – good for H100
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H] (still FP32)
    # ------------------------------------------------------------------
    out_hidden = C.view(B, H, N, N).permute(0, 2, 3, 1)   # [B,N,N,H] fp32

    # ------------------------------------------------------------------
    # 6️⃣  Output layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                           # FP32

    out_hidden = out_hidden * out_gate           # broadcast over H

    to_out_w = weights["to_out.weight"]
    output = F.linear(out_hidden, to_out_w)      # [B,N,N,dim] FP32

    return output
```
