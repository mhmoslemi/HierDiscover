@triton.jit
def apply_mask_kernel(A_ptr, M_ptr, O_ptr, B, N, H,
                      stride_Ab, stride_Am, stride_Ak, stride_Ah,
                      stride_Mb, stride_Mm, stride_Mk,
                      stride_Ob, stride_Om, stride_Ok, stride_Oh,
                      BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = pid_j * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_i = offs_i < N
    mask_j = offs_j < N

    for off_h in range(0, H, BLOCK_H):
        offs_h = off_h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        a_ptrs = A_ptr + pid_b*stride_Ab + offs_i[:, None, None]*stride_Am + offs_j[None, :, None]*stride_Ak + offs_h[None, None, :]*stride_Ah
        m_ptrs = M_ptr + pid_b*stride_Mb + offs_i[:, None]*stride_Mm + offs_j[None, :]*stride_Mk  # mask broadcasted over H

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                    other=0.0)
        m = tl.load(m_ptrs,
                    mask=mask_i[:, None] & mask_j[None, :],
                    other=1.0)  # 1 for no mask

        out = a * m

"""
TriMul (outgoing) – fast forward pass
------------------------------------
The module follows the reference implementation but replaces the
N³‑inner‐product (the ``einsum('... i k d, ... j k d -> ... i j d')``) with a
high‑throughput batched GEMM performed by `torch.bmm`.  The heavy 3‑D reduction
remains the dominant cost, but using the cuBLAS‑based GEMM on the H100 gives a
large speed‑up over a pure‑Triton implementation.
A tiny Triton kernel is used to fuse the element‑wise multiplication of the
projected tensors with their sigmoid gates and the optional pair‑wise mask.
All other operations (LayerNorm, Linear, Sigmoid, final projection) are kept
in PyTorch because they are already highly optimized.

Key steps
---------
1. Layer‑norm over the channel dimension of the 4‑D input.
2. Linear projections (no bias) to the hidden dimension.
3. Compute three sigmoid gates from the same normalized input.
4. Fuse ``proj * gate * mask`` for the left and right branches with a
   custom Triton kernel (single‑pass, no extra Python loops).
5. Perform a batched half‑precision GEMM:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as ``torch.bmm(left_flat, right_flat.T)``.
6. Second Layer‑norm on the hidden dimension, apply the output gate,
   and a final linear projection back to the original channel size.

The kernel works on contiguous 1‑D views of the tensors, broadcasting the
mask over the hidden dimension by integer division (``offset // hidden_dim``).
This design keeps the Triton contribution non‑trivial while letting the
dominant N³ work run on the highly‑tuned cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication with mask
# Computes: out = a * b * m   (m is broadcasted over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul_mask_kernel(
    a_ptr,               # *ptr to left projection   (float16)
    b_ptr,               # *ptr to gate tensor       (float16)
    m_ptr,               # *ptr to mask (broadcast) (float16, length = B*N*N)
    out_ptr,             # *ptr to output tensor     (float16)
    total_elems,         # total number of elements = B*N*N*H
    hidden_dim,          # size of hidden dimension H
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load the two multiplicands
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)

    # broadcast mask: each mask element repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    m = tl.load(m_ptr + mask_idx, mask=mask)

    tl.store(out_ptr + offs, a * b * m, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, [B, N, N, C], float32
            - mask_tensor  : torch.Tensor, [B, N, N] (float32, bool or None)
            - weights_dict : mapping name → torch.Tensor (float32)
            - config_dict  : must contain "dim" and "hidden_dim";

    Returns:
        torch.Tensor of shape [B, N, N, C], float32
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. First LayerNorm over the channel dimension (C)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  (float32)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) to hidden dimension
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H] float32
    right_proj = F.linear(x, weights["right_proj.weight"])  # [B,N,N,H] float32

    # --------------------------------------------------------------
    # 3. Compute sigmoid gates (float32 -> half)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # keep float32 for later

    # --------------------------------------------------------------
    # 4. Prepare mask (broadcasted over hidden dim) and cast to fp16
    # --------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=input_tensor.dtype,
            device=device,
        )
    else:
        mask = mask_tensor.to(dtype=input_tensor.dtype, device=device)

    # mask shape [B,N,N] -> [B,N,N,1] -> flatten for kernel
    mask = mask.unsqueeze(-1)                # [B,N,N,1]
    mask_fp16 = mask.to(torch.float16)

    # --------------------------------------------------------------
    # 5. Fuse (proj * gate * mask) for left and right using Triton
    # --------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    total_elements = B * N * N * hidden_dim
    BLOCK = 1024  # reasonable tile size for the elementwise kernel
    grid = (triton.cdiv(total_elements, BLOCK),)

    # Make everything contiguous and flatten to 1‑D
    left_proj_h = left_proj.to(torch.float16).contiguous().view(-1)
    right_proj_h = right_proj.to(torch.float16).contiguous().view(-1)
    left_gate_h = left_gate.contiguous().view(-1)
    right_gate_h = right_gate.contiguous().view(-1)
    mask_flat = mask_fp16.contiguous().view(-1)          # length = B*N*N

    left_fused = torch.empty_like(left_proj_h)  # fp16 storage
    right_fused = torch.empty_like(right_proj_h)

    # Left branch
    fused_mul_mask_kernel[grid](
        left_proj_h,
        left_gate_h,
        mask_flat,
        left_fused,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    # Right branch
    fused_mul_mask_kernel[grid](
        right_proj_h,
        right_gate_h,
        mask_flat,
        right_fused,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to [B, N, N, H]
    left = left_fused.view(B, N, N, hidden_dim)
    right = right_fused.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6. Batched half‑precision GEMM using torch.bmm (much faster than a pure
    #    Triton implementation on H100)
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Convert to half‑precision tensors expected by torch.bmm
    left_h = left  # already fp16
    right_h = right

    # Rearrange to (B*H, N, N) where each head is a separate batch entry
    left_flat = left_h.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # [B*H, N, N]
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # [B*H, N, N]

    # Batched GEMM (left * rightᵀ)
    out_flat_fp16 = torch.bmm(left_flat, right_flat.transpose(1, 2))       # fp16, shape [B*H, N, N]

    # Cast back to fp32 for the remaining layers
    out_fp32 = out_flat_fp16.float()
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 7. Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H] float32

    # --------------------------------------------------------------
    # 8. Apply output gate (sigmoid) – still in float32
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9. Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, C] float32

    return out
