@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # tile along i
    pid_n = tl.program_id(1)  # tile along j
    pid_hb = tl.program_id(2)  # flatten over (batch, hidden): index = b * H + h

    b = pid_hb // H
    h = pid_hb % H

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Starting pointers for left and right and out:

    a_ptrs_base = left_ptr + b * stride_left_batch + h * stride_left_h
    # left dims: (i, k). We'll iterate k.
    b_ptrs_base = right_ptr + b * stride_right_batch + h * stride_right_h
    # right dims: (j, k). We'll read as B transposed (k,j). We'll compute pointer accordingly.
    c_ptrs_base = out_ptr + b * stride_out_batch + h * stride_out_h

    # compute load for left using i and k:
    # left[i,k] pointer: base + i*stride_left_i + k*stride_left_k
    # For each tile, we'll compute using same offsets.

    # compute output pointer: base + i*stride_out_i + j*stride_out_j
    c_ptrs = c_ptrs_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left load: A[i,k]
        a_ptrs = a_ptrs_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right load: B[k, j] (we need transpose)
        # Original right layout: (j,k). We need to read B[k,j] => index swapped.
        # right_ptr base + j*stride_right_j + k*stride_right_k
        # So for k as rows, j as cols: pointer = base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        b_ptrs = b_ptrs_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # dot
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write result
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operation.

Algorithm
---------
1. Layer‑norm over the input channel dimension (dim) – done with PyTorch.
2. Bias‑free linear projections → tensors `left` and `right` of shape
   [B, N, N, H] (H = hidden_dim).  Results are immediately cast to float16.
3. Optional masking (broadcasted over H) and sigmoid gating.
   The three gates (`left_gate`, `right_gate`, `out_gate`) are also cast to
   float16 and fused element‑wise.
4. Cubic contraction
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   is expressed as a batch of independent GEMMs C = A @ Bᵀ where each
   pair (b, h) is a distinct matrix multiplication of size N×N.
   A custom Triton kernel processes a tile of (i, j) and iterates over k.
   * No tensor permutation or copy – the kernel works directly on the
     original (B, N, N, H) layout using explicit strides.
   * Inputs are half‑precision; accumulation is float32 (native TF32/FP32
     on H100).  Tile sizes (128×128×64) fit comfortably in shared memory.
5. Layer‑norm over the hidden dimension, element‑wise multiplication with
   `out_gate`, and a final linear projection back to the original channel
   dimension `dim`.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    for a tile (BLOCK_M, BLOCK_N) of the (i, j) plane and a single
    (batch, hidden) pair indexed by pid_hb.
    """
    pid_i = tl.program_id(0)          # tile along i  (rows)
    pid_j = tl.program_id(1)          # tile along j  (cols)
    pid_hb = tl.program_id(2)         # flattened batch*hidden index

    # Decode batch and hidden indices
    b = pid_hb // H
    h = pid_hb % H

    # Starting coordinates of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Thread‑local i/j coordinates
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # Output pointer for the current tile
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left[b, i, k, h]  (row‑major)
        a_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # half → fp16

        # right[b, j, k, h] read as Bᵀ => right[k, j, h]
        b_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                     # half → fp16

        # Accumulate dot product (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # Store the result (fp32)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # unpack weights
    # ------------------------------------------------------------------ #
    norm_weight           = weights["norm.weight"]
    norm_bias             = weights["norm.bias"]
    left_proj_weight      = weights["left_proj.weight"]
    right_proj_weight     = weights["right_proj.weight"]
    left_gate_weight      = weights["left_gate.weight"]
    right_gate_weight     = weights["right_gate.weight"]
    out_gate_weight       = weights["out_gate.weight"]
    to_out_norm_weight    = weights["to_out_norm.weight"]
    to_out_norm_bias      = weights["to_out_norm.bias"]
    to_out_weight         = weights["to_out.weight"]

    # ------------------------------------------------------------------ #
    # 1) Layer‑norm over the input feature dimension
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------ #
    # 2) Linear projections (bias‑free), cast to float16 for speed
    # ------------------------------------------------------------------ #
    left = F.linear(x, left_proj_weight)          # [B, N, N, H]  float32
    right = F.linear(x, right_proj_weight)

    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3) Optional masking (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------ #
    if not nomask and mask is not None:
        mask_exp = mask.to(left.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 4) Gating (sigmoid) – also in half‑precision
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight)).to(torch.float16)

    left = left * left_gate                # [B, N, N, H]  half
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5) Triton kernel: batch GEMM A @ Bᵀ for each (b, h) pair
    # ------------------------------------------------------------------ #
    B, N, _, _ = left.shape                 # B, N, N, H
    M_total = B * hidden_dim                # flattened batch × hidden

    # Output tensor (float32 accumulator)
    out = torch.empty_like(left, dtype=torch.float32)   # [B, N, N, H]

    # Block sizes – tuned for H100 (shared‑memory friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides for the original layout (B, N, N, H)
    stride_left_batch = left.stride(0)
    stride_left_i     = left.stride(1)
    stride_left_k     = left.stride(2)
    stride_left_h     = left.stride(3)

    stride_right_batch = right.stride(0)
    stride_right_j     = right.stride(1)   # j dimension in original (B, N, N, H)
    stride_right_k     = right.stride(2)
    stride_right_h     = right.stride(3)

    stride_out_batch = out.stride(0)
    stride_out_i     = out.stride(1)
    stride_out_j     = out.stride(2)
    stride_out_h     = out.stride(3)

    # Grid:  (tiles along i, tiles along j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M_total,
    )
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_batch,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 6) Post‑processing: layer‑norm over hidden dim, gating, final linear
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )   # [B, N, N, H]  float32

    # out_gate is half – automatic up‑cast to float32 during multiplication
    out = out * out_gate

    output = F.linear(out, to_out_weight)   # [B, N, N, dim]  float32
    return output
