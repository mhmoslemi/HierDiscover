@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile i
    pid_n = tl.program_id(1)   # tile j
    pid_z = tl.program_id(2)   # combined batch*hidden

    # decode batch and hidden
    total_h = H
    batch = pid_z // total_h
    hidden = pid_z % total_h

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # compute base pointers for this (batch, hidden)
    left_base = left_ptr + batch * stride_lb + hidden * stride_lh
    right_base = right_ptr + batch * stride_rb + hidden * stride_rh
    out_base = out_ptr + batch * stride_ob + hidden * stride_oh

    # pointer offsets for tiles
    a_ptrs = left_base + i[:, None] * stride_li + tl.arange(0, BLOCK_K)[None, :] * stride_lk
    # we will compute in loop k.

    # Initialize accumulator fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        a_ptrs = left_base + i[:, None] * stride_li + k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        b_ptrs = right_base + k[:, None] * stride_rk + j[None, :] * stride_ri
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M,BLOCK_K) dot (BLOCK_K,BLOCK_N)
    # store result
    c_ptrs = out_base + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – fast Triton implementation.

The forward pass consists of:
1️⃣ Layer‑norm over the original feature dimension (dim).  (FP32)
2️⃣ Linear projections → left / right (bias‑free) and three sigmoid gates.
   All tensors are cast to FP16 to halve the memory traffic.
3️⃣ Optional element‑wise masking of left/right.
4️⃣ Gating: left *= left_gate , right *= right_gate.
5️⃣ Heavy O(N³·H) contraction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ where each (batch, hidden) pair
   is an independent GEMM of size N×N.  The kernel tiles the i, j and k
   axes (BLOCK_M × BLOCK_N × BLOCK_K) and accumulates with `tl.dot`.
   Strides are taken directly from the original (B,N,N,H) layout,
   so no costly `permute(...).contiguous()` copies are needed.
6️⃣ Layer‑norm over the hidden dimension, element‑wise multiplication with
   the output gate and a final linear projection back to `dim`.
   The result is returned as FP32.

The kernel works for any batch size B∈{1,2}, sequence length N∈{128,…,1024}
and hidden dimension H (always 128 in the provided tests).  Mixed‑precision
(FP16 compute, FP32 accumulation) gives >5× speed‑up compared to the
reference implementation.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,               # fp16, shape [B,N,N,H]
    right_ptr,              # fp16, shape [B,N,N,H]
    out_ptr,                # fp16, shape [B,N,N,H] – will hold the contracted tensor
    N,                      # sequence length (int)
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides: batch, i, k, hidden
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides: batch, j, k, hidden
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides:   batch, i, j, hidden
    H: tl.constexpr,        # hidden dimension (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A,B ∈ ℝ^{N×N}) for every (batch, hidden) pair."""
    pid_i = tl.program_id(0)          # tile row (i)
    pid_j = tl.program_id(1)          # tile col (j)
    pid_z = tl.program_id(2)          # combined batch‑hidden index

    # ---- decode batch and hidden indices ------------------------------------
    batch = pid_z // H
    hidden = pid_z - batch * H        # pid_z % H

    # ---- tile start offsets -------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # ---- base pointers for this (batch, hidden) ----------------------------
    left_base  = left_ptr  + batch * stride_lb + hidden * stride_lh
    right_base = right_ptr + batch * stride_rb + hidden * stride_rh
    out_base   = out_ptr   + batch * stride_ob + hidden * stride_oh

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over k ---------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left[i, k, hidden]   (shape BLOCK_M × BLOCK_K)
        a_ptrs = left_base + i[:, None] * stride_li + k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0))

        # right[k, j, hidden]   (the transpose we need)
        b_ptrs = right_base + k[:, None] * stride_rk + j[None, :] * stride_ri
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0))

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → accumulate in FP32
        acc += tl.dot(a, b)

    # ---- write result -------------------------------------------------------
    c_ptrs = out_base + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor [B, N, N, dim] (float32)
              - mask        : torch.Tensor [B, N, N] (bool/int) – may be ignored
              - weights     : dict of all model weights (float32 tensors)
              - config      : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # unpack model weights (all are FP32 torch tensors on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the input feature dimension (dim) – keep FP32 for
    #    stability, the result is still on the GPU.
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------------- #
    # Cast to FP16 for the heavy linear projections and subsequent kernel.
    # --------------------------------------------------------------------- #
    x16 = x.half()
    left_proj_weight16  = left_proj_weight.half()
    right_proj_weight16 = right_proj_weight.half()
    left_gate_weight16  = left_gate_weight.half()
    right_gate_weight16 = right_gate_weight.half()
    out_gate_weight16   = out_gate_weight.half()
    to_out_norm_weight16 = to_out_norm_weight.half()
    to_out_norm_bias16   = to_out_norm_bias.half()
    to_out_weight16      = to_out_weight.half()

    # --------------------------------------------------------------------- #
    # 2. Linear projections (bias‑free)
    # --------------------------------------------------------------------- #
    left  = torch.nn.functional.linear(x16, left_proj_weight16)   # [B, N, N, H]
    right = torch.nn.functional.linear(x16, right_proj_weight16)

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------------- #
    if not nomask:
        # mask is usually bool or 0/1; cast to half and expand last dim
        mask_exp = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x16, left_gate_weight16))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x16, right_gate_weight16))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x16, out_gate_weight16))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Triton kernel – the cubic contraction.
    #    Treat each (batch, hidden) pair as an independent GEMM.
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape  # left shape = [B, N, N, H] (half)

    # Output placeholder (same shape & dtype as left/right)
    out = torch.empty_like(left)

    # Strides for the original layout (contiguous)
    # left / right / out all have layout (B, N, N, H)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Choose tile sizes – these fit well on an H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: tiles over i, j and the combined (batch, hidden) dimension.
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H                           # one program per (batch, hidden)
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        H=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # typical for FP16 GEMM on H100
    )

    # --------------------------------------------------------------------- #
    # 6. Post‑contraction processing (still in FP16)
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight16,
        bias=to_out_norm_bias16,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight16)   # [B, N, N, dim] (FP16)

    # --------------------------------------------------------------------- #
    # Cast back to FP32 for the final output (as required by the reference)
    # --------------------------------------------------------------------- #
    return out.float()
