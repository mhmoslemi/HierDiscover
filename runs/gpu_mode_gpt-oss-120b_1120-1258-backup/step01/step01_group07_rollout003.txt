@triton.jit
def batch_matmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ag, stride_bg, stride_cg,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_g = tl.program_id(1)

    # compute block position, with group m for better L2 usage
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)

    # Determine group m
    group_id = pid // (num_m_blocks * num_n_blocks)
    pid_in_grid = pid % (num_m_blocks * num_n_blocks)
    num_pid_in_group = tl.cdiv(num_m_blocks, GROUP_M) * num_n_blocks
    group_m = pid_in_grid // num_n_blocks
    group_n = pid_in_grid % num_n_blocks

    pid_m = (group_id * GROUP_M + group_m) % num_m_blocks
    pid_n = group_n

    # offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers offset to current group
    a_ptrs = a_ptr + pid_g * stride_ag
    b_ptrs = b_ptr + pid_g * stride_bg
    c_ptrs = c_ptr + pid_g * stride_cg

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(a_ptrs + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c = c_ptrs + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – fast Triton implementation
================================================

The outgoing TriMul computes, for every hidden channel ``h``,

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where ``left`` and ``right`` are linear projections of a Layer‑Normed input
tensor ``x`` (shape ``[B,N,N,C]``).  The summation over the third index ``k``
is an O(N³) operation and dominates the runtime.  We implement this core
operation as a *batched* GEMM:

* each pair (batch, hidden) is treated as an independent matrix‑multiply
  ``A[g] @ B[g].T`` with ``A[g]`` and ``B[g]`` of shape ``(N,N)``.
* The kernel fuses the three outer loops (batch × head) into the second
  Triton program‑id dimension, giving a launch grid ``(tiles_M×tiles_N,
  G)`` where ``G = B * hidden_dim``.
* fp16 is used for the large operand tensors while accumulation is performed
  in fp32 (via ``tl.dot``), providing Tensor‑Core utilisation on H100.
* Block sizes ``(BLOCK_M, BLOCK_N, BLOCK_K) = (128,128,32)`` and
  ``num_warps=8`` follow the highest‑throughput Triton matmul pattern.
* All non‑cubic work (LayerNorm, linear projections, gating, final
  LayerNorm and projection) is performed with highly‑optimised PyTorch
  primitives.

The function ``custom_kernel`` follows the signature required by the test
harness and returns a ``float32`` tensor of shape ``[B,N,N,C]``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B.T) for the TriMul core
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (M×K)·(K×N)→(M×N)
    stride_ag, stride_am, stride_ak,         # strides for A (group, row, col)
    stride_bg, stride_bk, stride_bn,         # strides for B (group, row(K), col(N))
    stride_cg, stride_cm, stride_cn,         # strides for C (group, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul.

    For each group ``g`` (where ``g = batch * hidden_dim``) compute
        C[g] = A[g] @ B[g].T
    where A[g] and B[g] are ``(M,N)`` matrices (here M = N = K = seq_len).

    The kernel follows the classic Triton matmul pattern using Tensor‑Core
    friendly block sizes.
    """
    pid = tl.program_id(0)   # tile id in the (M,N) plane
    pid_g = tl.program_id(1) # group id (batch * head)

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_n_tiles
    pid_n = pid %  num_n_tiles

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the current group
    # ------------------------------------------------------------------
    a = a_ptr + pid_g * stride_ag
    b = b_ptr + pid_g * stride_bg
    c = c_ptr + pid_g * stride_cg

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_tile = tl.load(
            a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile: shape (BLOCK_K, BLOCK_N) – note B is already the
        # transpose we need (B[g,k,j] == right[b,j,k,h]).
        b_tile = tl.load(
            b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Tensor‑Core dot (fp16·fp16 → fp32 accumulation)
        acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    # Write result back to C (still FP32)
    # ------------------------------------------------------------------
    c_ptrs = c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, C] (float32)
        mask         : torch.Tensor [B, N, N] (bool/float) – may be unused
        weights      : dict of torch.Tensor – model parameters
        config       : dict with keys:
                       - "dim" (C)
                       - "hidden_dim" (H)
                       - "nomask" (bool, True ⇒ no mask)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # projection dimension H
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm the input (FP32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,C] – still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (cast to fp16 for the large tensors)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]     # (C, H)
    right_proj_w = weights["right_proj.weight"]   # (C, H)

    # torch.linear does (input @ weight.T); we keep fp16 for speed.
    left = F.linear(x, left_proj_w).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if use_mask:
        # mask: [B,N,N] → [B,N,N,1] → fp16
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating tensors (sigmoid) – keep gates in fp16, out‑gate stays fp32
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))                     # [B,N,N,H] fp32

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM
    #    left : (B,N,N,H) → (B,H,N,N) → view as (G,N,N) with G = B*H
    #    right: (B,N,N,H) → (B,H,N,N) with axes (k,i) swapped → (B,H,N,N)
    # ------------------------------------------------------------------
    #   left_perm  : [B, H, N, N] where axis 2 = i, axis 3 = k
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    #   right_perm : [B, H, N, N] where axis 2 = k, axis 3 = j
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    G = B * H                                   # total groups for the kernel

    # Collapse batch & head dimensions → (G, N, N)
    A = left_perm.view(G, N, N)    # fp16
    B_ = right_perm.view(G, N, N)  # fp16 (already transposed for matmul)

    # Output buffer – we keep fp32 accumulation (important for large N)
    C = torch.empty((G, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6️⃣ Compute strides (in elements, not bytes) for the kernel
    # ------------------------------------------------------------------
    stride_ag = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bg = B_.stride(0)
    stride_bk = B_.stride(1)    # row over K
    stride_bn = B_.stride(2)    # column over N

    stride_cg = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    M = N
    K = N

    # ------------------------------------------------------------------
    # 7️⃣ Triton kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, G)   # (tiles over M×N, groups)

    # Launch the kernel
    _tri_mul_gemm_kernel[grid](
        A,
        B_,
        C,
        M,
        N,
        K,
        stride_ag,
        stride_am,
        stride_ak,
        stride_bg,
        stride_bk,
        stride_bn,
        stride_cg,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B,N,N,H] (still fp32)
    # ------------------------------------------------------------------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 9️⃣ Output LayerNorm, out‑gate and final projection back to C
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # fp32

    out = out * out_gate                       # fp32

    to_out_w = weights["to_out.weight"]        # (C, H)
    output = F.linear(out, to_out_w)          # [B,N,N,C] – fp32

    return output
```
