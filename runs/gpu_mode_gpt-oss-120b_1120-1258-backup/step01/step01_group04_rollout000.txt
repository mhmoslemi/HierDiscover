"""
TriMul (outgoing) – fast Triton implementation
================================================

* Normalises the input (LayerNorm over `dim`).
* Projects to a hidden dimension (`hidden_dim`) with two linear layers.
* Optionally masks the projected tensors.
* Applies three element‑wise sigmoid gates.
* **Core cubic contraction**  
  out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]  

  The contraction is a batched matrix‑multiply `C = A @ Bᵀ` where each
  `(batch, hidden)` pair is treated as an independent GEMM of size
  `N × N`.  The GEMM is executed by a custom Triton kernel that
  * loads the operands in FP16 to reduce memory traffic,
  * accumulates in FP32 for numerical stability,
  * uses large tiles (128 × 128 × 64) that saturate an H100.
* Normalises the hidden result, multiplies by the output gate and
  projects back to the original feature dimension.

Only the heavy N³ contraction is hand‑written in Triton; all the
lighter linear / LayerNorm / gating steps stay in PyTorch.
"""
import math
import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  (M, N, N)   – FP16
    b_ptr,               # right (M, N, N)   – FP16, read as (k, j)
    c_ptr,               # output (M, N, N) – FP32
    N,                   # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B (M, j, k) – we read as (k, j)
    stride_cm, stride_ci, stride_cj,   # strides of C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for each batch‑hidden slice m,
        C[m, i, j] = Σₖ A[m, i, k] * B[m, j, k]
    where A and B are stored in FP16 and the accumulator C is FP32.
    """
    pid_m = tl.program_id(0)          # tile along i
    pid_n = tl.program_id(1)          # tile along j
    pid_b = tl.program_id(2)          # (batch * hidden) index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # pointer to output tile C[i, j]
    c_ptrs = (
        c_ptr
        + pid_b * stride_cm
        + i[:, None] * stride_ci
        + j[None, :] * stride_cj
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ---- load A[i, k] --------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_b * stride_am
            + i[:, None] * stride_ai
            + k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        ).to(tl.float32)                     # FP16 → FP32 for the dot

        # ---- load B[k, j] (B is stored as B[j, k]) -------------------
        b_ptrs = (
            b_ptr
            + pid_b * stride_bm
            + k[:, None] * stride_bk
            + j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0)
        ).to(tl.float32)                     # FP16 → FP32

        # ---- accumulate ------------------------------------------------
        acc += tl.dot(a, b)                  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ---- write back ---------------------------------------------------
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple `(input_tensor, mask, weights, config)`
            * `input_tensor` – torch.Tensor, shape [B, N, N, dim] (float32)
            * `mask`         – torch.Tensor, shape [B, N, N] (bool/float)
            * `weights`      – dict of all model parameters (already on GPU)
            * `config`       – dict with at least `dim`, `hidden_dim`,
                               `nomask` (bool) and optional `eps`.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------ #
    # unpack weights (all are torch.Tensor on the correct device)
    # ------------------------------------------------------------------ #
    norm_weight           = weights["norm.weight"]
    norm_bias             = weights["norm.bias"]
    left_proj_weight      = weights["left_proj.weight"]
    right_proj_weight     = weights["right_proj.weight"]
    left_gate_weight      = weights["left_gate.weight"]
    right_gate_weight     = weights["right_gate.weight"]
    out_gate_weight       = weights["out_gate.weight"]
    to_out_norm_weight    = weights["to_out_norm.weight"]
    to_out_norm_bias      = weights["to_out_norm.bias"]
    to_out_weight         = weights["to_out.weight"]

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------ #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------ #
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------ #
    # 3. Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------ #
    if not nomask:
        mask_exp = mask.to(left.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 4. Gating (sigmoid) – still in FP32 for stability
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Prepare tensors for the fused GEMM kernel
    #    - cast projection results to FP16 (dram bandwidth reduction)
    #    - reshape to (M, N, N) where M = B * hidden_dim
    # ------------------------------------------------------------------ #
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B, N, N, H) -> (B, H, N, N) -> view as (M, N, N)
    left_perm  = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat  = left_perm.view(M_total, N, N)   # [M, N, N]  FP16
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N]  FP16
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------ #
    # 6. Triton kernel launch – batched GEMM C = A @ Bᵀ
    # ------------------------------------------------------------------ #
    # Tile sizes – chosen to maximise occupancy on H100 while fitting
    #             into register file (~64 KiB per thread block).
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides (in elements) for the flattened tensors
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # original second dim (j)
    stride_bk = right_flat.stride(2)   # original third dim (k)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along i (rows)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along j (cols)
        M_total                         # one kernel per (batch, hidden) pair
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 7. Reshape back to the original 4‑D layout
    # ------------------------------------------------------------------ #
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] (FP32)

    # ------------------------------------------------------------------ #
    # 8. Hidden‑dim LayerNorm, gating, final projection back to `dim`
    # ------------------------------------------------------------------ #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim] (FP32)

    return output
