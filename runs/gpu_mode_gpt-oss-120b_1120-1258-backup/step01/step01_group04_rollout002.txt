"""
TriMul (outgoing) – fast forward pass.

The heavy cubic contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched matrix‑multiply C = A @ Bᵀ where
    A = left   (M, N, N)   with M = B·H,
    B = right  (M, N, N)   (the second operand is read transposed).

The kernel works on FP16 data (Tensor‑Core friendly) while accumulating in
FP32 for numerical stability.  Block sizes (128×128×64) are chosen to
maximise occupancy on an H100.  All other layers (LayerNorm, Linear,
sigmoids, masking) are performed with PyTorch in FP32; the result of the
kernel is cast back to FP32 before the final LayerNorm and projection.

A Triton kernel is required and is used for the N³ contraction – the
slowest part of the module – while the remaining operations remain in
PyTorch for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B are (M,N,N) FP16 tensors)
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,            # left  : (M, N, N)    FP16
    b_ptr,            # right : (M, N, N)    FP16 (read transposed)
    c_ptr,            # out   : (M, N, N)    FP16
    N,                # sequence length
    stride_am, stride_ai, stride_ak,   # strides for A : (batch, i, k)
    stride_bm, stride_bj, stride_bk,   # strides for B : (batch, j, k)
    stride_cm, stride_ci, stride_cj,   # strides for C : (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)   # tile along i (rows)
    pid_j = tl.program_id(1)   # tile along j (cols)
    pid_m = tl.program_id(2)   # batch * hidden index

    # block origin
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range of rows / cols handled by this block
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # masks for boundary handling
    mask_i = i < N
    mask_j = j < N

    # pointer to the output tile
    c_ptrs = c_ptr + pid_m * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # load A[i, k]
        a_ptrs = a_ptr + pid_m * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # (BLOCK_M, BLOCK_K)

        # load B[k, j]   (B is stored as B[j, k]; we read the transposed view)
        b_ptrs = b_ptr + pid_m * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)          # (BLOCK_K, BLOCK_N)

        # accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back the tile (auto‑cast to FP16)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------------- #
# Triton‑enabled forward pass
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors (already on device)
        - config       : dict with keys
            * "dim"        – input feature dimension
            * "hidden_dim" – hidden dimension (c_z)
            * "nomask"     – bool, true when mask is ignored

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # unpack weights (all are float32 tensors, on the correct device)
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # 1. LayerNorm over the input channel dimension
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------ #
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------ #
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]  FP32
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------ #
    # 3. Gating (sigmoid) – kept in FP32 for stability
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # 4. Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------ #
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] and cast to same dtype
        mask_exp = mask.to(left.dtype).unsqueeze(-1)    # float32
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 5. Apply gates
    # ------------------------------------------------------------------ #
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 6. Cast to FP16 – the contraction is the only part that benefits from
    #    Tensor‑Core acceleration.
    # ------------------------------------------------------------------ #
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 7. Fuse dimensions so that each (batch, hidden) pair is a separate
    #    matrix multiplication: (M, N, N) where M = B * H.
    # ------------------------------------------------------------------ #
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    M_total = B * H

    # reshape to (M, N, N)
    left_flat  = left_perm.view(M_total, N, N)   # FP16
    right_flat = right_perm.view(M_total, N, N)  # FP16
    out_flat   = torch.empty_like(left_flat)     # FP16, will hold the result

    # -------------------------------------------------------------- #
    # 8. Triton kernel launch (batched GEMM C = A @ Bᵀ)
    # -------------------------------------------------------------- #
    BLOCK_M = 128   # rows per block
    BLOCK_N = 128   # cols per block
    BLOCK_K = 64    # reduction tile size

    # strides in *elements* (not bytes)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # j dimension (second)
    stride_bk = right_flat.stride(2)   # k dimension (third)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along j
        M_total                         # one program per (batch, hidden)
    )

    # launch
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------- #
    # 9. Reshape back to [B, N, N, H] and convert to FP32 for the rest
    # -------------------------------------------------------------- #
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] FP16
    out = out.to(torch.float32)

    # -------------------------------------------------------------- #
    # 10. LayerNorm over the hidden dimension, gate, final projection
    # -------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gate (FP32)
    output = F.linear(out, to_out_weight)    # [B, N, N, dim]  FP32

    return output
```
