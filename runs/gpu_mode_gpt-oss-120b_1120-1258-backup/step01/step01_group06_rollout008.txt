@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid = tl.program_id(2)    # combined batch*hidden index

    total = B * H
    b = pid // H
    h = pid % H

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for left and right for this batch and hidden
    # left base: left_ptr + b*stride_l_b + h*stride_l_h
    lhs_base = left_ptr + b * stride_l_b + h * stride_l_h
    rhs_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # pointers for tiles: we want left[ i, k ] (i: M, k: K)
    # left offset: i * stride_l_i + k * stride_l_k
    # right offset: j * stride_r_i + k * stride_r_k

    a_ptrs = lhs_base + offs_m[:, None] * stride_l_i + tl.arange(0, BLOCK_K)[None, :] * stride_l_k
    b_ptrs = rhs_base + offs_n[None, :] * stride_r_i + tl.arange(0, BLOCK_K)[:, None] * stride_r_k

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_tiles_k):
        # Compute offsets for this tile
        k_offset = k * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        a = tl.load(lhs_base + offs_m[:, None] * stride_l_i + k_idx[None, :] * stride_l_k,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(rhs_base + offs_n[None, :] * stride_r_i + k_idx[:, None] * stride_r_k,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot produce (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b).to(tl.float32)

    # store out: out_base + i*stride_o_i + j*stride_o_j
    c_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – fast Triton implementation.

The core of the operator is a batched matrix multiplication per hidden
channel:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are the results of two linear projections on the
input tensor `x` (shape [B, N, N, C]).  The computation is realised as a
single Triton kernel that:

* works on the original layout [B,N,N,H] (no costly permute‑copy),
* uses fp16 for the bulk of the data and fp32 for the accumulation,
* stores the result in fp16 to halve memory traffic, then casts to fp32
  for the remaining LayerNorm / gating / output projection steps,
* exploits large tiles (128×128×64) and 8 warps per block on H100,
* respects an optional mask and the gating tensors that are applied
  before the kernel.

All auxiliary operations (LayerNorm, linear projections, sigmoids,
mask‑multiplication and the final projection) are performed with
standard PyTorch ops, which are already highly optimised.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    # tensor pointers
    left_ptr,                # fp16 * [B, N, N, H]
    right_ptr,               # fp16 * [B, N, N, H]
    out_ptr,                 # fp16 * [B, N, N, H]  (final result)
    # problem size
    B, N, H,                 # batch, seq_len, hidden_dim
    # strides for the 4‑D tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched mat‑mul C = A @ Bᵀ  over the hidden dimension.
    A and B have shape (B, N, N, H) and the result C has the same shape.
    The kernel loops over the K‑dimension (the second N axis) and
    accumulates in fp32, finally storing fp16.
    """
    pid_m = tl.program_id(0)          # block row  (i‑axis)
    pid_n = tl.program_id(1)          # block col  (j‑axis)
    pid   = tl.program_id(2)          # combined batch * hidden index

    # --------------------------------------------------------------
    # Decode combined index into batch b and hidden h
    # --------------------------------------------------------------
    b = pid // H
    h = pid %  H

    # --------------------------------------------------------------
    # Tile offsets for the i and j dimensions
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Compute base pointers for the current (b,h) slice
    # --------------------------------------------------------------
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b  + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b  + h * stride_o_h

    # --------------------------------------------------------------
    # Accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K (the second N axis)
    # --------------------------------------------------------------
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for tk in range(num_tiles_k):
        # global K indices for this tile
        k_base = tk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)   – left[b,i,k,h]
        a = tl.load(
            left_base
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )

        # Load B tile: shape (BLOCK_K, BLOCK_N)   – right[b,j,k,h]
        b_tile = tl.load(
            right_base
            + offs_j[None, :] * stride_r_i
            + offs_k[:, None] * stride_r_k,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0),
        )

        # Dot product – fp16 inputs, fp32 accumulation
        acc += tl.dot(a, b_tile, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Store result in fp16 (cast from fp32 accumulator)
    # --------------------------------------------------------------
    out_ptrs = out_base \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim        = config["dim"]                 # input channel dimension (C)
    hidden_dim = config["hidden_dim"]          # projection dimension (H)
    nomask     = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (float16 for the heavy tensors)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16) # [B,N,N,H]

    # --------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4) Gating (keep everything in fp16 to stay in the fast path)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Out‑gate (float32, applied after the kernel)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,H]

    # --------------------------------------------------------------
    # 6) Prepare the Triton kernel launch
    # --------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    assert H == hidden_dim, "hidden_dim mismatch"

    # Allocate output buffer (fp16) – will be cast to fp32 afterwards
    out_fp16 = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # Strides for the original layout (contiguous)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    # Choose tile size heuristically (good for the tested N ranges)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
        num_warps = 4
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑blocks
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑blocks
        B * H,                          # batch * hidden
    )

    # --------------------------------------------------------------
    # 7) Launch the batched mat‑mul kernel
    # --------------------------------------------------------------
    tri_mul_kernel[grid](
        left,
        right,
        out_fp16,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # Cast back to fp32 for the remaining layers
    out = out_fp16.to(torch.float32)   # [B,N,N,H]

    # --------------------------------------------------------------
    # 8) Final LayerNorm, out‑gate and projection back to `dim`
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # bias=False in the model

    return output
