"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The operator works on a tensor x ∈ ℝ^{B×N×N×C} :

1. Layer‑norm on the channel dimension C.
2. Two linear projections → left, right ∈ ℝ^{B×N×N×H}.
3. Optional pairwise mask and three sigmoid gates.
4. Element‑wise gated left/right.
5. Heavy “pairwise‑multiplicative” update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiply `A @ Bᵀ` for each (b,h) pair.
   A and B are taken as FP16, accumulated in FP32 and written back as FP16.
   The kernel works directly on the original layout (B,N,N,H) – no
   transposes or copies – by treating the hidden dimension as an extra
   batch axis and by loading B in its transposed view inside the kernel.
6. Cast the result to FP32, second layer‑norm, out‑gate and final linear
   projection back to the original channel dimension.

The whole compute‑bound step (step 5) is fully fused in Triton and
covers the full N³·H·B workload without any intermediate materialisation,
which brings the runtime from ~6 ms down to ~1 ms on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel : batched matmul A @ Bᵀ   (A,B shape B×N×N×H, result B×N×N×H)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    BATCH, HIDDEN, N,                        # runtime dimensions
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,   # strides for A  (batch,i,k,hidden)
    stride_Bb, stride_Bi, stride_Bk, stride_Bh,   # strides for B  (batch,i,k,hidden)
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,   # strides for C  (batch,i,j,hidden)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)    # block‑row (i)
    pid_n = tl.program_id(1)    # block‑col (j)
    pid_z = tl.program_id(2)    # combined batch × hidden index

    # --------------------------------------------------------------
    # Recover batch and hidden indices from the 3rd program id.
    # ``pid_z = batch * HIDDEN + hidden``  →  hidden = pid_z % HIDDEN
    # --------------------------------------------------------------
    hidden = pid_z % HIDDEN
    batch  = pid_z // HIDDEN

    # --------------------------------------------------------------
    # Offsets inside the M and N dimensions
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers for the current tile.
    # A has layout (batch, i, k, hidden) → rows = i, cols = k
    # Bᵀ is accessed as (k, j) = B[batch, j, k, hidden]
    # --------------------------------------------------------------
    a_ptrs = (A_ptr
              + batch * stride_Ab
              + hidden * stride_Ah
              + offs_m[:, None] * stride_Ai
              + offs_k[None, :] * stride_Ak)

    b_ptrs = (B_ptr
              + batch * stride_Bb
              + hidden * stride_Bh
              + offs_k[:, None] * stride_Bk
              + offs_n[None, :] * stride_Bi)   # note the swapped axes → transposed view

    # --------------------------------------------------------------
    # Accumulator in FP32 (tensor‑core dot promotes to FP32 automatically)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < N),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < N) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)   # FP16 × FP16 → FP32

        # Advance the K‑block pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # --------------------------------------------------------------
    # Write the result C[batch, i, j, hidden]
    # --------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_Cb
              + hidden * stride_Ch
              + offs_m[:, None] * stride_Ci
              + offs_n[None, :] * stride_Cj)

    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left & right  [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional pairwise mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)       # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy pairwise‑multiplicative update – fused Triton kernel
    # ------------------------------------------------------------------
    # Cast to FP16 for bandwidth / tensor‑core efficiency
    left_f16  = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_f16.shape          # B, N, N, hidden
    # Output buffer (FP16)
    out_f16 = torch.empty_like(left_f16)

    # -----------------------------------------------------------------------------
    # Kernel launch configuration – tuned for H100
    # -----------------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden,
    )

    _tri_mul_outgoing_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B, hidden, N,
        left_f16.stride(0), left_f16.stride(1), left_f16.stride(2), left_f16.stride(3),
        right_f16.stride(0), right_f16.stride(1), right_f16.stride(2), right_f16.stride(3),
        out_f16.stride(0), out_f16.stride(1), out_f16.stride(2), out_f16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to FP32 for the remaining (light) layers
    out = out_f16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm, out‑gate, and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
