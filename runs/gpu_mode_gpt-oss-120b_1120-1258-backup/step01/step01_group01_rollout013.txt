import torch
import triton
import triton.language as tl

@triton.jit
def _proj_gate_mul_kernel(proj_ptr, gate_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    out = (a * g).to(tl.float16)
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated “outgoing” TriMul forward pass.
    1. Layer‑norm the input.
    2. Fuse left/right projections and gating into two large linear layers.
    3. Fuse the element‑wise projection‑×‑gate using a tiny Triton kernel
       (produces fp16 tensors to cut bandwidth).
    4. Apply optional pair‑wise mask (broadcast on the hidden dimension).
    5. Batch‑matmul the left/right tensors (shape (B·H, N, N)) with
       torch.bmm – this follows the original einsum but uses cuBLAS
       (TF32/FP16 Tensor‑cores) and is far faster than a naïve Triton loop.
    6. Layer‑norm, out‑gate scaling and final linear projection back to ``dim``.
    The implementation stays fully on‑device, avoids extra copies and works
    for all the test configurations described in the statement.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input layer‑norm (channel dimension = dim)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Project + gate – we concatenate the four weight matrices
    #    (left_proj, right_proj, left_gate, right_gate) so that a
    #    single torch.nn.functional.linear does the heavy work.
    # --------------------------------------------------------------
    # Concatenate weight matrices in the order described above.
    # All weights are stored as (out_features, in_features).
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],      # (hidden, dim)
            weights["right_proj.weight"],     # (hidden, dim)
            weights["left_gate.weight"],      # (hidden, dim)
            weights["right_gate.weight"],     # (hidden, dim)
        ],
        dim=0,
    )  # (4*hidden, dim)

    combined = torch.nn.functional.linear(x, w_concat)  # (B, N, N, 4*hidden)

    # Split projected tensors and gates.
    left_proj = combined[..., :hidden]                       # (B,N,N,hidden)
    right_proj = combined[..., hidden:2 * hidden]            # (B,N,N,hidden)
    left_gate = combined[..., 2 * hidden:3 * hidden].sigmoid()   # (B,N,N,hidden)
    right_gate = combined[..., 3 * hidden:].sigmoid()           # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 3. Element‑wise proj × gate using a tiny Triton kernel.
    #    The kernel works on flattened 1‑D views (contiguous memory)
    #    and writes the result in fp16 (half) to reduce bandwidth.
    # --------------------------------------------------------------
    B, N, _, _ = x.shape
    numel = B * N * N * hidden

    left_fp16 = torch.empty_like(left_proj, dtype=torch.float16)
    right_fp16 = torch.empty_like(right_proj, dtype=torch.float16)

    # Ensure contiguous layout for a 1‑D view.
    lp = left_proj.contiguous().view(-1)
    lg = left_gate.contiguous().view(-1)
    rp = right_proj.contiguous().view(-1)
    rg = right_gate.contiguous().view(-1)
    lo = left_fp16.view(-1)
    ro = right_fp16.view(-1)

    BLOCK = 1024
    grid = ((numel + BLOCK - 1) // BLOCK,)

    _proj_gate_mul_kernel[grid](
        lp,
        lg,
        lo,
        numel,
        BLOCK_SIZE=BLOCK,
    )
    _proj_gate_mul_kernel[grid](
        rp,
        rg,
        ro,
        numel,
        BLOCK_SIZE=BLOCK,
    )

    # --------------------------------------------------------------
    # 4. Optional pair‑wise mask (broadcast on the hidden dimension).
    # --------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask shape [B, N, N] -> [B, N, N, 1] and cast to fp16
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left_fp16 = left_fp16 * mask_f16
        right_fp16 = right_fp16 * mask_f16

    # --------------------------------------------------------------
    # 5. Heavy multiplicative update: batched matmul.
    #    left: (B, N, N, hidden) -> (B·hidden, N, N)
    #    right needs a transpose on the two pair‑wise axes.
    # --------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) for easy view.
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()  # transpose on (i,k) <-> (k,i)

    BH = B * hidden
    left_mat = left_perm.view(BH, N, N)   # (BH, N, N)
    right_mat = right_perm.view(BH, N, N) # (BH, N, N)

    # torch.bmm uses cuBLAS (TF32/FP16 Tensor‑cores) and is very fast.
    out_mat = torch.bmm(left_mat, right_mat)  # (BH, N, N), fp16

    # Restore original layout: (B, N, N, hidden) in fp32 for the next LayerNorm.
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6. Final LayerNorm, out‑gate, and projection back to ``dim``.
    # --------------------------------------------------------------
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()  # shape (B,N,N,hidden)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) forward pass – highly‑optimised implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input on its channel dimension.
2. Fuse the four linear projections (left/right + their gates) into a single
   matrix‑multiply by concatenating the weight matrices.  Afterwards split the
   result and apply the sigmoid to the gating slices.
3. Fuse the element‑wise   proj × gate   operation into a tiny Triton kernel
   that writes the result in fp16 (halves the bandwidth for the next step).
4. Apply the optional pair‑wise mask (broadcast on the hidden dimension).
5. Heavy “pairwise‑multiplicative’’ update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication.  After transposing the right tensor
   we reshape both tensors to (B·H, N, N) and use `torch.bmm`, which calls
   cuBLAS (TF32/FP16 tensor‑cores) and is far faster than a naïve Triton loop.
6. Layer‑norm the result, multiply by the out‑gate and linearly project back
   to the original channel dimension.

The implementation stays fully on‑device, works for all test configurations
described in the statement and uses at least one Triton kernel (step 3).
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _proj_gate_mul_kernel(
    proj_ptr,          # fp32 tensor (flattened)
    gate_ptr,          # fp32 tensor (flattened)
    out_ptr,           # fp16 tensor (flattened)
    size,              # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise multiplication of two fp32 vectors with the result stored
    as fp16.  The kernel works on a 1‑D view of the tensors (contiguous memory)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(proj_ptr + offs, mask=mask, other=0.0)   # fp32
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)   # fp32
    out = (a * g).to(tl.float16)                        # fp16
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the “outgoing’’ TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, C] (float32)
        - mask         : torch.Tensor   [B, N, N] (bool/float) or None
        - weights      : dict of model weights
        - config       : dict with keys ``dim`` (C), ``hidden_dim`` (H) and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor : [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # input channel dimension C
    hidden = config["hidden_dim"]       # hidden dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Fuse left/right projections and their gates (single matmul)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices: left_proj, right_proj, left_gate,
    # right_gate (all have shape [hidden, dim]).
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    )  # (4*H, dim)

    # One large linear layer: (B,N,N,dim) @ (dim,4*H) -> (B,N,N,4*H)
    combined = torch.nn.functional.linear(x, w_concat)  # fp32

    # Split the concatenated result.
    left_proj  = combined[..., :hidden]                         # (B,N,N,H)
    right_proj = combined[..., hidden:2 * hidden]               # (B,N,N,H)
    left_gate  = combined[..., 2 * hidden:3 * hidden].sigmoid() # (B,N,N,H)
    right_gate = combined[..., 3 * hidden:].sigmoid()           # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Element‑wise (proj * gate) → fp16 via a tiny Triton kernel
    # ------------------------------------------------------------------
    total_elements = B * N * N * hidden

    # Allocate fp16 containers for the gated tensors.
    left_fp16  = torch.empty_like(left_proj, dtype=torch.float16)
    right_fp16 = torch.empty_like(right_proj, dtype=torch.float16)

    # Flatten everything to 1‑D contiguous views for the kernel.
    lp = left_proj.contiguous().view(-1)
    lg = left_gate.contiguous().view(-1)
    rp = right_proj.contiguous().view(-1)
    rg = right_gate.contiguous().view(-1)
    lo = left_fp16.view(-1)
    ro = right_fp16.view(-1)

    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    _proj_gate_mul_kernel[grid](
        lp, lg, lo, total_elements, BLOCK_SIZE=BLOCK,
    )
    _proj_gate_mul_kernel[grid](
        rp, rg, ro, total_elements, BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4. Optional pair‑wise mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_fp16 = left_fp16 * mask_f16
        right_fp16 = right_fp16 * mask_f16

    # ------------------------------------------------------------------
    # 5. Heavy multiplicative update – batched matmul
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) and flatten the (B,H) axes.
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    # Right needs a transpose on the two pairwise axes before flattening.
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N)

    BH = B * hidden
    left_mat  = left_perm.view(BH, N, N)   # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)  # (BH, N, N)

    # torch.bmm uses cuBLAS (TF32/FP16 tensor‑cores) and is extremely fast.
    out_mat = torch.bmm(left_mat, right_mat)   # (BH, N, N), fp16

    # Restore (B, N, N, H) layout and cast back to fp32 for the following LN.
    out = (
        out_mat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, out‑gate scaling and projection back to `dim`
    # ------------------------------------------------------------------
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()                                          # (B,N,N,H)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
