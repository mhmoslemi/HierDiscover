"""
TriMul (outgoing) – Triton‑accelerated forward pass.
--------------------------------------------------------------------
The heavy N³ contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix multiplication for every hidden dimension d:

    out_d = left_d @ right_dᵀ ,   left_d/right_d ∈ ℝ^{N×N}.

We fuse this GEMM into a single Triton kernel that works on half‑precision
matrices and accumulates in FP32 for numerical safety.  All other
operations (LayerNorm, Linear projections, sigmoid gating) are performed
with PyTorch; they are cheap compared to the O(N³) work.

The kernel is fully vector‑ised, uses two‑stage pipelining and a configurable
tile size (BLOCK_M/BLOCK_N/BLOCK_K).  The launch configuration adapts to the
sequence length, keeping the number of warps modest (4–8) for the H100.
The final output is returned in float32.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr,               # fp16 * left  : (B*H, M, K)
    b_ptr,               # fp16 * right : (B*H, K, N)   (right already transposed)
    c_ptr,               # fp16 * out   : (B*H, M, N)
    M, N, K,             # int32 dimensions (all == seq_len)
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bk, stride_bn,   # strides for B
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM   C = A @ Bᵀ
    A : (B*H, M, K)   B : (B*H, N, K)   →  we feed Bᵀ thus B : (B*H, K, N)
    The kernel loops over K in tiles of size BLOCK_K, accumulates in FP32
    and finally stores the result back as fp16.
    """
    pid0 = tl.program_id(0)      # combined batch * tile_m
    pid1 = tl.program_id(1)      # tile_n

    # -----------------------------------------------------------------
    # Decode batch index and tile coordinates
    # -----------------------------------------------------------------
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    batch_id   = pid0 // num_m_tiles
    tile_m     = pid0 %  num_m_tiles
    tile_n     = pid1

    # -----------------------------------------------------------------
    # Compute offsets for the current tile
    # -----------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointer arithmetic (all strides are in *elements*, not bytes)
    a_ptrs = a_ptr + batch_id * stride_az \
                + offs_m[:, None] * stride_am \
                + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                + offs_k[:, None] * stride_bk \
                + offs_n[None, :] * stride_bn

    # -----------------------------------------------------------------
    # Accumulator (FP32)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # K‑loop with double‑buffering style pre‑fetch (2 stages)
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_idx = cur_k + offs_k

        # Out‑of‑bounds handling
        mask_a = (offs_m[:, None] < M) & (k_idx[None, :] < K)
        mask_b = (k_idx[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)

        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Step forward in K
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -----------------------------------------------------------------
    # Store result (cast back to fp16)
    # -----------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                + offs_m[:, None] * stride_cm \
                + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (all float32)
        - config       : dict with at least {"dim": int, "hidden_dim": int,
                                            "nomask": bool}
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣  LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # -----------------------------------------------------------------
    # Cast to half‑precision before the linear layers – this reduces memory
    # traffic for the subsequent N³ contraction.  The weight tensors are
    # kept in float32; PyTorch will handle the mixed‑precision multiplication.
    x_half = x.to(torch.float16)

    left  = F.linear(x_half, weights["left_proj.weight"])
    right = F.linear(x_half, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] to broadcast over hidden_dim
        mask_f = mask.to(x_half.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣  Gating scalars (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_half, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_half, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_half, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣  Prepare tensors for the batched GEMM
    # -----------------------------------------------------------------
    # left  : [B, N, N, hidden_dim] → (B*hidden_dim, N, N) with layout (i,k)
    # right : [B, N, N, hidden_dim] → (B*hidden_dim, N, N) with layout (k,j)
    B, N, _, _ = left.shape

    # Permute to bring hidden_dim forward, then flatten batch*hidden_dim.
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (fp16, accumulation inside kernel is FP32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6️⃣  Launch configuration for the Triton kernel
    # -----------------------------------------------------------------
    # Choose tile sizes based on the sequence length.
    # For N ≤ 256 we can use smaller tiles to reduce shared‑memory pressure.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 4
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid_m = math.ceil(N / BLOCK_M)   # tiles in M per batch
    grid_n = math.ceil(N / BLOCK_N)   # tiles in N

    # pid0 combines (batch*hidden_dim) and tile_m
    grid = (B * hidden_dim * grid_m, grid_n)

    _batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,               # double‑buffered pipeline
    )

    # -----------------------------------------------------------------
    # 7️⃣  Reshape back to the original layout (B, N, N, hidden_dim)
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # cast back to FP32 for the final stages

    # -----------------------------------------------------------------
    # 8️⃣  Output LayerNorm, gating, and final projection back to `dim`
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                    # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # (hidden_dim → dim)

    return out

2. **Mask handling** – optional, broadcasted over the hidden dimension.

3. **Gating** – sigmoid on linear projections, fused by element‑wise multiplication.

4. **Batched GEMM kernel** – the only heavy part.  
   * `left` is stored as `(i,k)` and `right` as `(k,j)`.
