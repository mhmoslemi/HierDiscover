"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor `[B,N,N,C]`.
2. Two linear projections (`left_proj`, `right_proj`) map the normalized input to
   the hidden dimension `H`.  The optional binary mask is applied after the
   projection (broadcast over `H`).
3. Three sigmoidal gates (`left_gate`, `right_gate`, `out_gate`) are computed
   from the same normalized input.  The left/right gates are multiplied into
   the projected tensors element‑wise.
4. The heavy O(N³·H) work is a **batched matrix multiplication**:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   For each pair (batch, head) we have a GEMM `L_{i,k} @ R_{k,j}ᵀ`.
   This is implemented with a custom Triton kernel that processes a whole
   `(B·H)`‑group as an independent GEMM.  The kernel uses 128×128 tiles,
   FP16 arithmetic with FP32 accumulation (Tensor‑cores), 8 warps and 3
   pre‑fetch stages.
5. Layer‑norm the GEMM result, multiply by the per‑position `out_gate`,
   and finally linearly project back to the original channel dimension `C`.

All tensors are kept in FP16 for the large matrix work; the final
output is FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (M×K)·(K×N)→(M×N)
    stride_ag, stride_am, stride_ak,         # A: group, row, col strides
    stride_bg, stride_bk, stride_bn,         # B: group, row(K), col(N) strides
    stride_cg, stride_cm, stride_cn,         # C: group, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C[g,i,j] = Σ_k A[g,i,k] * B[g,k,j].

    Each program works on one (i‑tile, j‑tile) of a given group `g = batch*H+head`.
    """
    # ------------------------------------------------------------------
    # 0) program IDs
    # ------------------------------------------------------------------
    pid = tl.program_id(0)      # tile id within (M,N) plane
    pid_g = tl.program_id(1)    # group id (batch*head)

    # --------------------------------------------------------------
    # 1) compute tile coordinates (row‑tile, col‑tile)
    # --------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_tiles_n      # block row index
    pid_n = pid %  num_tiles_n      # block column index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # 2) pointers to the start of the current group
    # --------------------------------------------------------------
    a_batch = a_ptr + pid_g * stride_ag
    b_batch = b_ptr + pid_g * stride_bg
    c_batch = c_ptr + pid_g * stride_cg

    # --------------------------------------------------------------
    # 3) accumulator (FP32 for numerical stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 4) main loop over K dimension (blocked)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M x BLOCK_K)
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile: (BLOCK_K x BLOCK_N)
        b = tl.load(
            b_batch
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot products – tensor cores: fp16·fp16 → fp32
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # 5) write back the result
    # --------------------------------------------------------------
    c = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.BoolTensor or torch.FloatTensor of shape [B, N, N]
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys
              * "dim"          – input channel dimension C
              * "hidden_dim"   – projection dimension H
              * "nomask"       – if True, mask is ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B,N,N,C]  FP32

    # ------------------------------------------------------------------
    # 2) Linear projections  (cast to FP16 for speed)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if use_mask:
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # keep FP32 for later mul

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM:
    #    A[g,i,k] = left  (i rows, k cols)
    #    B[g,k,j] = right (k rows, j cols)  – note the transpose on the
    #                second spatial dimension.
    # ------------------------------------------------------------------
    #   left:  (B,N,N,H) -> (B,H,N,N)    (row i, col k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    #   right: (B,N,N,H) -> (B,H,N,N) with axes (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape                       # B, hidden, N, N
    G = B * H                                          # total groups

    # Collapse batch & head dimensions -> (G, N, N)
    A = left_perm.view(G, N, N)   # fp16
    B_ = right_perm.view(G, N, N) # fp16

    # Output buffer (FP32 accumulation)
    C = torch.empty_like(A, dtype=torch.float32, device=device)   # (G,N,N)

    # ------------------------------------------------------------------
    # 6) Strides for the kernel (group, row, col)
    # ------------------------------------------------------------------
    stride_ag = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bg = B_.stride(0)
    stride_bk = B_.stride(1)   # row (K) stride
    stride_bn = B_.stride(2)   # col (N) stride

    stride_cg = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    M = N
    K = N

    # ------------------------------------------------------------------
    # 7) Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # 32 works well with Tensor‑cores (16‑wise)

    num_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, G)   # (tiles over M×N, groups)

    _batched_matmul_kernel[grid](
        A,
        B_,
        C,
        M,
        N,
        K,
        stride_ag,
        stride_am,
        stride_ak,
        stride_bg,
        stride_bk,
        stride_bn,
        stride_cg,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per block
        num_stages=3,         # pipelined async copies
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B,N,N,H] (still FP32 because C is FP32)
    # ------------------------------------------------------------------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]  FP32

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                            # FP32

    out = out * out_gate                         # broadcast element‑wise, FP32

    to_out_w = weights["to_out.weight"]
    output = F.linear(out, to_out_w)             # [B,N,N,dim]  FP32

    return output
