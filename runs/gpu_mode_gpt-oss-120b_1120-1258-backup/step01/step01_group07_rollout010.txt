@triton.jit
def _trimul_kernel(a_ptr, b_ptr, c_ptr,
                    B, N, H,
                    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
                    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
                    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
                    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_mn = tl.program_id(0)
    pid_g = tl.program_id(1)

    # Determine batch and head
    batch = pid_g // H
    head = pid_g % H

    # Compute tile row, col
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid_mn // num_tiles_n
    pid_n = pid_mn % num_tiles_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute base pointers after batch/head offset
    a_batch_ptr = a_ptr + batch * stride_a_batch + head * stride_a_h
    b_batch_ptr = b_ptr + batch * stride_b_batch + head * stride_b_h
    c_batch_ptr = c_ptr + batch * stride_c_batch + head * stride_c_h

    # Load accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K)
        a = tl.load(
            a_batch_ptr + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K x N)
        b = tl.load(
            b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)
    # Store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def _trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # indices
    pid = tl.program_id(0)
    pid_g = tl.program_id(1)

    # Determine batch and head
    batch = pid_g // H
    head = pid_g % H

    # Compute tile coordinates
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_tiles_n
    pid_n = pid % num_tiles_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (batch,head)
    a_base = a_ptr + batch * stride_a_batch + head * stride_a_h
    b_base = b_ptr + batch * stride_b_batch + head * stride_b_h
    c_base = c_ptr + batch * stride_c_batch + head * stride_c_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The forward pass consists of

1. Layer‑norm on the 4‑D input  (B,N,N,C).
2. Two linear projections (left / right) to the hidden dimension H.
3. Optional element‑wise mask.
4. Three element‑wise sigmoid gates (left, right, out).
5. Core O(N³·H) update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a **batched GEMM** for every hidden head.
   A custom Triton kernel processes all (batch, head) groups together,
   reading the tensors in their native layout (no permute / copy) and
   using FP16 arithmetic with FP32 accumulation.
6. Layer‑norm, out‑gate and final linear projection back to C.

The kernel avoids the costly permutation that the reference code performs,
drastically cutting memory traffic and giving runtimes well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    a_ptr,                     # left  tensor   (B,N,N,H)  FP16
    b_ptr,                     # right tensor   (B,N,N,H)  FP16
    c_ptr,                     # output         (B,N,N,H)  FP32
    B, N, H,                   # scalars
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul:  C[g,i,j] = Σ_k A[g,i,k] * B[g,k,j]

    * g = batch * H + head  (total G = B*H groups)
    * A is indexed as (i,k)   using strides (stride_a_i, stride_a_k)
    * B is indexed as (k,j)   using strides (stride_b_k, stride_b_j)
    * C is indexed as (i,j)   using strides (stride_c_i, stride_c_j)
    """
    pid = tl.program_id(0)   # tile index in the (M,N) plane
    pid_g = tl.program_id(1) # combined (batch, head) index

    # ------------------------------------------------------------------
    # Decode (batch, head) from pid_g
    # ------------------------------------------------------------------
    batch = pid_g // H
    head  = pid_g %  H

    # ------------------------------------------------------------------
    # Compute 2‑D tile coordinates
    # ------------------------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_tiles_n
    pid_n = pid %  num_tiles_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # rows   (i)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # cols   (j)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch, head)
    # ------------------------------------------------------------------
    a_base = a_ptr + batch * stride_a_batch + head * stride_a_h
    b_base = b_ptr + batch * stride_b_batch + head * stride_b_h
    c_base = c_ptr + batch * stride_c_batch + head * stride_c_h

    # ------------------------------------------------------------------
    # Accumulator in FP32 (higher precision)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K (= N)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile   (M × K)  ->  left[i, k, h]
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile   (K × N)  ->  right[k, j, h]
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Multiply‑accumulate (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N]  (bool / int)
        - weights      : dict of model parameters
        - config       : dict containing "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # C
    hidden_dim = config["hidden_dim"]   # H
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (keep it FP32 for numerical safety)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B,N,N,C]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → hidden dimension (cast to FP16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dim)
    # ------------------------------------------------------------------
    if use_mask:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating tensors (sigmoid).  Gates for left/right stay in FP16.
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # FP32

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM via a fused Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # B,N,N,H
    # Output buffer (FP32 accumulation)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides of the three tensors (they are contiguous in the last dim)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tile sizes – tuned for H100 (shared‑mem & register budget)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid: (tiles over M×N, total groups = B*H)
    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (tiles_m * tiles_n, B * H)

    _trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_j,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps = 256 threads per block (good for 128×128 tiles)
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )                                   # FP32, shape [B,N,N,H]

    out = out * out_gate                # element‑wise gating (FP32)

    output = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return output
