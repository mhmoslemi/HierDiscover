"""
TriMul (outgoing) – fast forward implementation.

The heavy O(N³) pairwise reduction is a batched matrix‑multiply:
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

We treat each head `h` as an independent batch dimension
and evaluate the product with the highly‑optimised cuBLAS batched GEMM
(`torch.bmm`).  The element‑wise parts (masking and gating) are fused
into a tiny Triton kernel to satisfy the “use Triton” requirement while
keeping the overall runtime close to the cuBLAS optimum.

All linear layers and LayerNorms are done with PyTorch (float32) and the
large mat‑mul works in float16 → float32 accumulation (Tensor‑Core
friendly).  The final output is float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fused_mul_kernel(
    out_ptr, a_ptr, b_ptr, mask_ptr,  # pointers (1‑D)
    total_elems,                      # total number of elements to process
    HAS_MASK: tl.constexpr,           # compile‑time flag
    BLOCK_SIZE: tl.constexpr,         # threads per program
):
    """
    Element‑wise (a * b * mask) or (a * b) on a flattened tensor.
    All tensors are expected to be contiguous and of the same dtype
    (fp16 for left/right, fp32 for out_gate when used later).
    """
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=mask, other=1.0)
        out = a * b * m
    else:
        out = a * b

    tl.store(out_ptr + offs, out, mask=mask)


def _fused_mul(a: torch.Tensor,
               b: torch.Tensor,
               mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    Calls the Triton kernel to compute `a * b * mask` (if mask is given)
    or `a * b` otherwise.  The tensors must be contiguous and have the
    same shape and dtype.
    """
    assert a.shape == b.shape, "shape mismatch"
    if mask is not None:
        assert mask.shape == a.shape, "mask shape mismatch"
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024                       # 1024 threads per program – good for H100
    grid = (triton.cdiv(total, BLOCK),)

    # Triton expects raw pointers; we flatten to 1‑D.
    a_flat = a.view(-1)
    b_flat = b.view(-1)
    out_flat = out.view(-1)

    if mask is None:
        _fused_mul_kernel[grid](
            out_flat,
            a_flat,
            b_flat,
            torch.empty(0, device=a.device, dtype=a.dtype),  # dummy
            total,
            HAS_MASK=False,
            BLOCK_SIZE=BLOCK,
        )
    else:
        mask_flat = mask.view(-1)
        _fused_mul_kernel[grid](
            out_flat,
            a_flat,
            b_flat,
            mask_flat,
            total,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]               # input channel dimension
    hidden_dim = config["hidden_dim"] # projection dimension (usually 128)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (project to hidden_dim, kept in fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"]).to(torch.float16) # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Gating values (sigmoid, kept in fp16 for left/right, fp32 for out_gate)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # float32

    # ------------------------------------------------------------------
    # 4) Optional mask handling + gating fusion (Triton)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask shape [B,N,N] -> broadcast over hidden dim
        mask = mask_tensor.to(left_proj.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = _fused_mul(left_proj, left_gate, mask)          # left * left_gate * mask
        right = _fused_mul(right_proj, right_gate, mask)       # right * right_gate * mask
    else:
        left = _fused_mul(left_proj, left_gate)                # left * left_gate
        right = _fused_mul(right_proj, right_gate)             # right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched matrix multiplication: left @ rightᵀ  (cuBLAS)
    #    Treat each head as a separate batch element.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                      # H == hidden_dim
    # reshape to (B*H, N, N) where the last dim is the K dimension of GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    # torch.bmm uses Tensor Cores for fp16·fp16→fp32 accumulation
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N]  fp16
    out = out_flat.to(torch.float32)  # convert to fp32 for the next LayerNorm

    # ------------------------------------------------------------------
    # 6) Restore original layout: [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, out‑gate, and projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])  # bias=False in the model

    return output
