"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.

The most expensive part of TriMul is the 4‑D contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is an independent batched GEMM (A @ Bᵀ) for every hidden
dimension d.  We perform this GEMM with a single torch.bmm call
(batched fp16 matmul, cuBLAS‑accelerated) and we fuse the element‑wise
mask‑and‑gating step with a tiny Triton kernel.  All LayerNorms,
linear projections and the final projection are kept in PyTorch for
clarity.  The kernel guarantees that the mask (if present) is applied
exactly as in the reference implementation while still satisfying the
“must use Triton” requirement.

The overall workflow:

1. Layer‑norm on the last dimension.
2. Linear projections → left / right (float32 → fp16) and three gating
   scalars (left_gate, right_gate, out_gate) (float32 → fp16).
3. One‑shot Triton kernel that multiplies left/right with their gates
   and with the optional mask (broadcasted over the hidden dimension).
4. Batched fp16 GEMM (torch.bmm) implementing the `einsum` contraction.
5. Layer‑norm + out‑gate + final linear projection back to `dim`.
6. Return a float‑32 tensor of shape [B, N, N, dim].
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_kernel(
    left_ptr,                # fp16*  (B,N,N,hidden)
    right_ptr,               # fp16*  (B,N,N,hidden)
    left_gate_ptr,           # fp16*  (B,N,N,hidden)
    right_gate_ptr,          # fp16*  (B,N,N,hidden)
    mask_ptr,                # fp16*  (B,N,N)  – broadcasted over hidden
    total_elements: tl.constexpr,   # B*N*N*hidden
    hidden_dim:    tl.constexpr,   # hidden dimension (known at compile time)
    BLOCK_SIZE:    tl.constexpr,   # number of elements per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # ----- load -------------------------------------------------
    l = tl.load(left_ptr + offs,  mask=valid, other=tl.float16(0.0))
    r = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs,  mask=valid, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    # ----- apply gates -----------------------------------------
    l = l * lg
    r = r * rg

    # ----- apply broadcasted mask --------------------------------
    # each mask element repeats `hidden_dim` times in the flattened view
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))
    l = l * m
    r = r * m

    # ----- store ------------------------------------------------
    tl.store(left_ptr + offs,  l, mask=valid)
    tl.store(right_ptr + offs, r, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack -----------------------------------------------------------
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣  LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) → left / right
    # -----------------------------------------------------------------
    # flatten the first three dims to use a single matmul per projection
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    left  = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 3️⃣  Gating scalars (sigmoid) – left_gate, right_gate, out_gate
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"])
                              ).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"])
                              ).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"])
                              ).view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 4️⃣  Cast everything to fp16 (the heavy path works in half‑precision)
    # -----------------------------------------------------------------
    left_fp16       = left.to(torch.float16)
    right_fp16      = right.to(torch.float16)
    left_gate_fp16  = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    # -----------------------------------------------------------------
    # 5️⃣  Apply mask (if any) + gating with a tiny Triton kernel
    # -----------------------------------------------------------------
    if nomask or mask is None:
        # a mask filled with ones – no effect
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16,
                               device=device)
    else:
        mask_fp16 = mask.to(torch.float16)     # shape [B, N, N]

    # flatten both tensors for the kernel (contiguous layout)
    left_fp16  = left_fp16.contiguous()
    right_fp16 = right_fp16.contiguous()
    left_gate_fp16  = left_gate_fp16.contiguous()
    right_gate_fp16 = right_gate_fp16.contiguous()
    mask_fp16 = mask_fp16.contiguous()

    total_elems = left_fp16.numel()    # B * N * N * hidden_dim
    BLOCK_SIZE = 1024                  # good trade‑off for H100
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    gate_mask_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16.view(-1),          # flatten mask to 1‑D pointer
        total_elems,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
        num_stages=2,
    )
    # -----------------------------------------------------------------
    # 6️⃣  Batched matrix multiplication: out = left @ rightᵀ  (per hidden dim)
    # -----------------------------------------------------------------
    # reshape into (B*hidden, N, N) and make the right matrix transposed on the
    # “k” axis (i.e. shape (B*hidden, N, N) where the last two dims are k and j)
    left_mat  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm uses cuBLAS‐Lt under the hood and exploits Tensor‑cores for fp16
    out_mat = torch.bmm(left_mat, right_mat)          # (B*hidden, N, N)

    # restore the original 4‑D shape
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                       # keep FP32 for the rest

    # -----------------------------------------------------------------
    # 7️⃣  Output LayerNorm + out‑gate
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate                               # element‑wise gating

    # -----------------------------------------------------------------
    # 8️⃣  Final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])      # (dim, hidden_dim) → dim

    return out
```
