"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1. Layer‑norm over the input channel dimension.
2. Two linear projections (left / right) → shape [B, N, N, H].
   Optional mask is applied here.
3. Gating: element‑wise sigmoid(linear(x)) multiplied into the projected tensors.
4. Batched GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice h.
   It is implemented with a custom Triton kernel that:
   * works on FP16 tensors (half‑precision) for bandwidth,
   * accumulates in FP32,
   * tiles the computation (BLOCK_M × BLOCK_N × BLOCK_K) and
   * launches one program per (batch, hidden) slice.
5. Layer‑norm on the hidden dimension, out‑gate gating and final linear
   projection back to the original channel dimension.
All heavy N³ work lives in the Triton kernel; the surrounding
operations are cheap PyTorch calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (all = seq_len)
    stride_am, stride_ah, stride_ai, stride_ak,   # A strides: batch, hidden, i, k
    stride_ba, stride_bh, stride_bk, stride_bn,   # B strides: batch, hidden, k, j
    stride_cm, stride_ch, stride_ci, stride_cj,   # C strides: batch, hidden, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each (batch, hidden) slice.
    A : (batch, hidden, M, K)  – fp16
    B : (batch, hidden, K, N)  – fp16 (read as (K,N) from the original layout)
    C : (batch, hidden, M, N)  – fp32
    """
    pid_m = tl.program_id(0)            # tile row
    pid_n = tl.program_id(1)            # tile col
    pid_z = tl.program_id(2)            # flattened (batch * hidden)

    # Decode batch and hidden indices
    batch_idx = pid_z // hidden_dim
    hidden_idx = pid_z - batch_idx * hidden_dim

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_ba + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_ch

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- tiled K loop ------------------- #
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M × K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K × N) – note B is accessed as (k, j)
        b_ptrs = b_base + (cur_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # ------------------- write result ------------------- #
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, S, S, C] (float32)
            - mask_tensor: [B, S, S] (bool/float) or None
            - weights_dict: all required linear / ln parameters (float32)
            - config_dict: {'dim': C, 'hidden_dim': H, 'nomask': bool, ...}

    Returns:
        Tensor of shape [B, S, S, C] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # Weight shortcuts
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, dim]  float32

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_weight)   # [B, S, S, H]
    right = F.linear(x, right_proj_weight)  # [B, S, S, H]

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections) and element‑wise gating
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, S, S, H]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Batched GEMM (TriMul reduction) – Triton kernel
    # --------------------------------------------------------------------- #
    # Cast projections to half; keep the layout (B, S, S, H) – no permute!
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output buffer (FP32) in the same layout: [B, S, S, H]
    out_fp32 = torch.empty((B, S, S, hidden_dim),
                           dtype=torch.float32,
                           device=device)

    # --------------------------------------------------------------------- #
    # Kernel launch configuration
    # --------------------------------------------------------------------- #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(S, BLOCK_M),          # rows (i)
        triton.cdiv(S, BLOCK_N),          # cols (j)
        B * hidden_dim                    # batch * hidden
    )

    _trimul_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left_h.stride(0),   # batch stride
        stride_ah=left_h.stride(3),   # hidden stride (=1)
        stride_ai=left_h.stride(1),   # i (row) stride
        stride_ak=left_h.stride(2),   # k (col) stride
        stride_ba=right_h.stride(0),
        stride_bh=right_h.stride(3),
        stride_bk=right_h.stride(2),   # k stride
        stride_bn=right_h.stride(1),   # j stride
        stride_cm=out_fp32.stride(0),
        stride_ch=out_fp32.stride(3),   # hidden stride (=1)
        stride_ci=out_fp32.stride(1),   # i stride
        stride_cj=out_fp32.stride(2),   # j stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
        num_warps=8,                    # good occupancy on H100
    )

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm, out‑gate and final projection
    # --------------------------------------------------------------------- #
    # out_fp32 already has shape [B, S, S, H] (hidden last)
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H]

    out = out * out_gate                     # gating
    out = F.linear(out, to_out_weight)       # back to original dim → [B, S, S, dim]

    return out
