"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The dominant O(N³·D) workload is the batched contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]   (d = hidden_dim)

which is a batched GEMM (A @ Bᵀ) for every hidden channel d.
The kernel below implements a fused batched GEMM in FP16 with
FP32 accumulation.  All other operations (LayerNorm, Linear,
sigmoid gates) are performed with PyTorch – they are bandwidth‑
bound and cheap compared to the N³ contraction.

Key ideas
---------
*  Flatten the (batch, hidden) dimensions → a single batch axis for the GEMM.
*  Store `left` as (B, H, M, K) and `right` as (B, H, K, N) so the kernel can
  read A as (M,K) and B as (K,N) directly (no extra transposes inside the kernel).
*  Use block sizes 64×64×64 (tuned for H100) and 8 warps / 4 stages.
*  Load & store with masks to safely handle non‑divisible sizes.
*  All computations are done in half‑precision except the accumulator
  (fp32) to keep numerical quality; the final result is cast back
  to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ   (A: M×K, B: N×K)  – batch = B*H
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,               # fp16*  left matrix   (batch, M, K)
    b_ptr,               # fp16*  right matrix  (batch, K, N)
    c_ptr,               # fp16*  output matrix (batch, M, N)
    M, N, K,             # matrix dimensions (all == seq_len)
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bk, stride_bn,   # strides for B
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM with fp32 accumulation.
    A is (M, K), B is (K, N) – the kernel therefore computes A @ B.
    """
    pid0 = tl.program_id(0)            # batch * tile_m
    pid1 = tl.program_id(1)            # tile_n

    # ---------------------------------------------------------------
    # Decode batch and tile coordinates
    # ---------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)   # tiles in the M dimension per batch
    batch_id   = pid0 // num_tiles_m
    tile_m     = pid0 %  num_tiles_m
    tile_n     = pid1

    # ---------------------------------------------------------------
    # Offsets within the current tile
    # ---------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this tile (no k‑offset yet)
    a_ptrs = a_ptr + batch_id * stride_az \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # ---------------------------------------------------------------
    # Accumulator (fp32)
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over K dimension
    # ---------------------------------------------------------------
    for cur_k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] + cur_k * BLOCK_K < K)
        mask_b = (offs_k[:, None] + cur_k * BLOCK_K < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N) fp16

        # fp16⋅fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---------------------------------------------------------------
    # Write back result (cast to fp16)
    # ---------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        - mask         : torch.Tensor [B, N, N] or None (bool/float)
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys
                         {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32) – result of the TriMul forward.
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # (1) LayerNorm on the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # (2) Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # -> [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # (3) Optional mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # (4) Gating scalars (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # (5) Prepare tensors for the N³ contraction
    # -----------------------------------------------------------------
    # Cast to fp16 for the heavy GEMM
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape   # shape: [B, N, N, hidden_dim]

    # Rearrange to (B, hidden_dim, M, K) and (B, hidden_dim, K, N)
    #   left  -> A : (M=i, K=k)
    #   right -> B : (K=k, N=j)   (i.e. we transpose the last two axes)
    left_mat  = left_fp16.permute(0, 3, 1, 2).contiguous()          # [B, H, M, K]
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous()         # [B, H, K, N]

    # Collapse (B, H) into a single batch dimension for the kernel
    left_mat  = left_mat.view(B * hidden_dim, N, N)
    right_mat = right_mat.view(B * hidden_dim, N, N)

    # Allocate output buffer (fp16, will be cast to fp32 later)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # (6) Launch the batched GEMM kernel
    # -----------------------------------------------------------------
    # Tuned block sizes for H100 – 64×64×64 gives good occupancy.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)   # tiles along M per batch
    grid_n = triton.cdiv(N, BLOCK_N)   # tiles along N
    # program_id(0) encodes (batch * tile_m), program_id(1) = tile_n
    grid = (B * hidden_dim * grid_m, grid_n)

    batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # H100 benefits from 8 warps per block
        num_stages=4,       # double‑buffering + extra stage for latency hiding
    )

    # -----------------------------------------------------------------
    # (7) Reshape back to [B, N, N, hidden_dim] and cast to fp32
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # (8) Output LayerNorm, gating and final projection back to `dim`
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                      # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])  # (dim, hidden_dim) → dim

    return out
