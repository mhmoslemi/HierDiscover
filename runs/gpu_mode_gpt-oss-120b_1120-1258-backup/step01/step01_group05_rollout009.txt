def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    eps = 1e-5

    # first layernorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # combine weights into one for projection/gates
    proj_weights = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0)  # shape (5*H, dim)

    # linear projection all at once
    proj = F.linear(x, proj_weights)  # [B,N,N,5*H]

    # slice
    H = hidden_dim
    left_proj = proj[..., :H]
    right_proj = proj[..., H:2*H]
    left_gate_raw = proj[..., 2*H:3*H]
    right_gate_raw = proj[..., 3*H:4*H]
    out_gate_raw = proj[..., 4*H:]

    # sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # mask handling
    if mask_tensor is None:
        mask = torch.ones_like(left_proj[..., :1])  # shape (B,N,N,1)
    else:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)  # shape (B,N,N,1)

    # Use Triton kernel to apply mask and gating in-place
    # flatten tensors:
    total = left_proj.numel()
    # Ensure contiguous
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()
    # In-place gating
    BLOCK = 8192
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
    fused_mask_gate_kernel[grid](
        left_proj, mask, left_gate, left_proj,
        total,
        BLOCK_SIZE=BLOCK
    )
    fused_mask_gate_kernel[grid](
        right_proj, mask, right_gate, right_proj,
        total,
        BLOCK_SIZE=BLOCK
    )
    # Now left_proj and right_proj are gated.

    # Convert to half for matmul
    left_h = left_proj.to(torch.float16)
    right_h = right_proj.to(torch.float16)

    # reshape for batched matmul: (B,H,N,N) -> (B*H,N,N)
    B = left_h.shape[0]
    N = left_h.shape[1]
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul (left @ right.T)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to B,N,N,H
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # second layernorm
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    # apply out gate
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) – fast forward implementation.
The heavy pair‑wise contraction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed with an extremely efficient batched GEMM from cuBLAS (torch.bmm),
while the element‑wise mask‑gate fusion is done with a tiny Triton kernel.
All remaining operations (LayerNorm, Linear, Sigmoid) use the highly‑optimised
PyTorch kernels.  The implementation respects the original paper semantics
and works for any of the test configurations (with/without mask, fp32
inputs, variable batch/sequence dimensions).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gate (three‑way elementwise product)
# ----------------------------------------------------------------------
@triton.jit
def fused_mask_gate_kernel(
    proj_ptr,          # *ptr to the projection tensor (float32)
    mask_ptr,          # *ptr to the broadcasted mask   (float32)
    gate_ptr,          # *ptr to the sigmoid‑gate tensor (float32)
    out_ptr,           # *ptr to output (same as proj_ptr for in‑place)
    size,              # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    m = offs < size

    # Load three operands, multiply and store back in‑place
    a = tl.load(proj_ptr + offs, mask=m, other=0.0)
    b = tl.load(mask_ptr + offs, mask=m, other=0.0)
    c = tl.load(gate_ptr + offs, mask=m, other=0.0)
    tl.store(out_ptr + offs, a * b * c, mask=m)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: mapping from weight name to torch.Tensor (float32)
            - config_dict : must contain keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣  First LayerNorm over the channel dimension C
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]

    # --------------------------------------------------------------
    # 2️⃣  Fuse the five linear projections (left, right, three gates)
    # --------------------------------------------------------------
    # weights for the projections/gates have shape (hidden_dim, dim)
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # (5*H, C)
    )
    # One large matmul → shape [B, N, N, 5*H]
    proj = F.linear(x, proj_weight)

    H = hidden_dim
    # Slice the concatenated tensor
    left_proj   = proj[..., 0:H]                     # [B,N,N,H]
    right_proj  = proj[..., H:2 * H]
    left_gate   = torch.sigmoid(proj[..., 2 * H:3 * H])
    right_gate  = torch.sigmoid(proj[..., 3 * H:4 * H])
    out_gate    = torch.sigmoid(proj[..., 4 * H:5 * H])

    # --------------------------------------------------------------
    # 3️⃣  Apply the pair‑wise mask (if any) and the two gates *in‑place*
    #     via a tiny Triton kernel for (proj * mask * gate)
    # --------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones_like(left_proj[..., :1], dtype=left_proj.dtype, device=left_proj.device)
    else:
        mask = mask_tensor.to(left_proj.dtype).unsqueeze(-1)   # [B,N,N,1]

    # Broadcast mask to hidden dimension (no extra copy in memory)
    mask_exp = mask.expand(-1, -1, -1, H).contiguous()

    # Ensure contiguous tensors for the kernel
    left_proj   = left_proj.contiguous()
    right_proj  = right_proj.contiguous()
    left_gate   = left_gate.contiguous()
    right_gate  = right_gate.contiguous()
    mask_exp    = mask_exp.contiguous()

    total_elems = left_proj.numel()
    BLOCK = 8192
    grid = (triton.cdiv(total_elems, BLOCK),)

    # left  = left_proj * mask * left_gate   (in‑place)
    fused_mask_gate_kernel[grid](
        left_proj,
        mask_exp,
        left_gate,
        left_proj,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )
    # right = right_proj * mask * right_gate (in‑place)
    fused_mask_gate_kernel[grid](
        right_proj,
        mask_exp,
        right_gate,
        right_proj,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )
    # ------------------------------------------------------------------
    # 4️⃣  Batched matrix multiplication:  out_{b,h} = left_{b,h} @ right_{b,h}ᵀ
    #    (shape per batch: N × N)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape  # B, N, N, H

    # (B, N, N, H) → (B*H, N, N)
    left_mat  = left_proj.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Core O(N³) work – cuBLAS batch‑GEMM (FP32, very fast on H100)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # Restore original layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm (over the hidden dimension)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 6️⃣  Apply the output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 7️⃣  Final linear projection back to the original channel dimension C
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
