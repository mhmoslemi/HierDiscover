import torch
import triton
import triton.language as tl

@triton.jit
def apply_gate_kernel(proj_ptr, gate_ptr, out_ptr, total_elems, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # sigmoid approximation: 1/(1+exp(-x))
    sigmoid = 1.0 / (1.0 + tl.exp(-gate))
    out = proj * sigmoid
    tl.store(out_ptr + offs, out, mask=mask)

    Triton‑accelerated “outgoing” TriMul forward pass.
    ------------------------------------------------
    1. Layer‑norm on the channel dimension (float32).
    2. Fuse the five linear projections (left/right proj & gates, out_gate) into a single half‑precision matmul.
    3. Apply sigmoid gates with a tiny Triton kernel (elementwise prod with sigmoid).
    4. Optional pairwise mask (broadcast on the hidden dimension).
    5. Batched half‑precision matrix multiplication `C = A @ Bᵀ` using torch.bmm (leverages cuBLAS/Tensor‑cores).
    6. Final layer‑norm, out‑gate, and linear projection back to the original dimension.
    The kernel is used for the gate‑application step; the heavy `N³` work is handled by cuBLAS.
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Fuse the five linear layers into a single half‑precision matmul
    # ------------------------------------------------------------------
    # Cast to half for the large matmul
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape
    M = B * N * N  # flatten dimension

    # Concatenate the five weight matrices (all are (hidden, dim) and bias‑free)

        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden, dim)

    # Linear projection: (M, dim) @ (dim, 5*hidden).T -> (M, 5*hidden)
    proj_gate = torch.nn.functional.linear(x_h.view(M, dim), W_fused)  # (M, 5*hidden)

    # Reshape to (B, N, N, 5, hidden)
    proj_gate = proj_gate.view(B, N, N, 5, hidden)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_proj = proj_gate[..., 0, :].contiguous()
    left_gate_raw = proj_gate[..., 1, :].contiguous()
    right_proj = proj_gate[..., 2, :].contiguous()
    right_gate_raw = proj_gate[..., 3, :].contiguous()
    out_gate = proj_gate[..., 4, :].contiguous().sigmoid()   # we keep out_gate in FP16 for now

    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Triton kernel: gated = proj * sigmoid(gate)
    total = left_proj.numel()

        right,

    )

    left_gate = left_gate_raw.sigmoid()   # not used further; we already applied via kernel
    right_gate = right_gate_raw.sigmoid() # same

    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:

        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    #    We use torch.bmm (cuBLAS/Tensor‑cores) on half‑precision tensors.
    # ------------------------------------------------------------------

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.float()
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input on its channel dimension (float32 for numerical stability).
2️⃣  Fuse the five linear layers (left_proj, left_gate, right_proj, right_gate,

4️⃣  If a pair‑wise mask is supplied, broadcast it over the hidden dimension

5️⃣  Perform the heavy N³ update `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`

6️⃣  Reshape back to `[B, N, N, hidden]`, apply a second layer‑norm, multiply by

7️⃣  Return the result in float32.

Only the gate‑application (step 3) is implemented in Triton; the dominant
