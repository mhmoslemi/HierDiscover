"""
custom_kernel – outgoing TriMul (AlphaFold‑3)

A fast forward implementation that fuses the expensive linear‑projection + gating
step with a tiny Triton kernel and lets the highly‑optimised PyTorch / cuBLAS
back‑ends handle everything else:

1️⃣  Layer‑norm on the input (float32 → float16) using torch.nn.functional.
2️⃣  One fused linear (5·hidden_dim → dim) to produce left/right projections
    and three gates.
3️⃣  Triton kernel ``split_fuse_gate_kernel``:
        * splits the fused projection into the three tensors,
        * applies the pair‑wise mask,
        * computes sigmoid gates,
        * writes ``left``, ``right`` and ``out_gate`` in the layout
          (B, H, N, N) (H = hidden_dim) as fp16.
4️⃣  Batched half‑precision GEMM (cuBLAS tensor‑cores):
        ``out = left @ rightᵀ``   →   (B·H, N, N) fp16.
5️⃣  Second LayerNorm (over the hidden dimension) with torch.nn.functional.
6️⃣  ``out = out_norm * out_gate``   (element‑wise fp16).
7️⃣  Final linear projection (H → dim) performed with a single
    ``torch.nn.functional.linear`` on the flattened tensor.

All heavy arithmetic stays in fp16 (except the LayerNorm statistics which are
computed in FP32 inside the PyTorch kernels).  Only the split/gate step is
implemented in Triton, satisfying the “use at least one kernel” requirement
while minimizing launch overhead for maximal speed on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]  (fp16) fused projection
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R] (fp16) – ignored if HAS_MASK=0
    R: tl.int32,       # total positions = B·N·N
    H: tl.int32,       # hidden dimension
    N: tl.int32,       # sequence length
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode linear index (b,i,k) – layout of the flattened input
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑position mask (scalar, broadcast over H)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # → (BLOCK_R, 1)
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride of one hidden plane within the (B, H, N, N) layout
    hidden_stride = N * N
    # flat offset of the first hidden plane for this (b,i,k)
    base_offset = b * (H * N * N) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # each block reads contiguous 5·H values for the current (b,i,k)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in fp32 → fp16 (more stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                 # out_gate is NOT masked

        # destination offset for the current hidden slice
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr    + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm  (float32 → float16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                       # (R, dim) float32
    # torch's LayerNorm is highly tuned on H100
    x_norm = F.layer_norm(x_flat,
                          (dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps).to(torch.float16)           # (R, dim) fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim → dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                         # (5*H, dim)

    proj = F.linear(x_norm, fused_w)                            # (R, 5*H) fp16
    del x_norm, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid gates (layout B·H·N·N)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # flatten mask to (R,) and cast to fp16
        mask_ptr = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS Tensor‑Core) : left @ rightᵀ
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                         # (B·H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # (B·H, N, N) fp16
    out = out.view(B, H, N, N)                                 # (B, H, N, N) fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16 → fp16)
    # --------------------------------------------------------------
    # Move hidden dim to the last axis because torch.layer_norm normalises
    # over the trailing dimensions.
    out_perm = out.permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)
    out_norm = F.layer_norm(out_perm,
                            (H,),
                            weight=w["to_out_norm.weight"],
                            bias=w["to_out_norm.bias"],
                            eps=eps)                     # fp16
    # --------------------------------------------------------------
    # 6️⃣ Multiply by out_gate and final linear projection (H → dim)
    # --------------------------------------------------------------
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    gated = out_norm * out_gate_perm                           # (B, N, N, H) fp16

    # flatten (B·N·N, H) and apply final linear (H → dim)
    gated_flat = gated.view(-1, H)                              # (R, H)
    w_out = w["to_out.weight"].to(torch.float16)                # (dim, H)
    out_final = F.linear(gated_flat, w_out)                     # (R, dim) fp16

    # reshape to the required output layout and cast to float32
    out_final = out_final.view(B, N, N, dim).to(torch.float32)

    return out_final
```
The implementation follows the six‑step strategy described in the docstring, uses a single Triton kernel for the split‑and‑gate phase, and relies on highly‑optimised PyTorch / cuBLAS kernels for all matrix‑heavy work, delivering a fast forward pass for the outgoing TriMul operator.
