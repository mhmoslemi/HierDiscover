def custom_kernel(data):
    x, mask, weights, cfg = data
    # get dims
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; eps = 1e-5; use_mask = not cfg.get("nomask", True)
    B,N,N2,dim_check = x.shape
    assert N==N2
    device = x.device

    # 1. LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)  # float32
    # flatten and cast to fp16
    x_norm_flat = x_norm.view(-1, dim).contiguous()
    x_norm_fp16 = x_norm_flat.to(torch.float16)

    # 2. Fused linear
    fused_W = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16) # shape [5*H, dim]

    proj = torch.nn.functional.linear(x_norm_fp16, fused_W)  # (R,5*H) fp16

    # 3. Split + mask + gates
    R = B * N * N
    left = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

    if use_mask:
        # mask dim is [B,N,N]; cast to fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)  # shape R
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)

    # launch split_fuse_gate_kernel
    BLOCK_R = 256
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        hidden_dim,
        N,
        HAS_MASK=1 if use_mask else 0,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # ensure memory
    torch.cuda.synchronize()

    # 4. Batch matmul
    left_mat = left.view(B*hidden_dim, N, N)
    right_mat = right.view(B*hidden_dim, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2)).view(B, hidden_dim, N, N)

    # 5. Compute per-pos mean & rstd via torch
    # out shape [B, H, N, N]; flatten over hidden dim
    out_flat = out.permute(0,2,3,1).contiguous()  # shape [B,N,N,H] maybe
    # Instead compute directly:
    mu = out.mean(dim=1)  # shape [B,N,N], but out shape [B,H,N,N], dim=1 is hidden dim.
    var = out.var(dim=1, unbiased=False)
    rstd = torch.rsqrt(var + eps)
    mu_flat = mu.view(-1)  # shape R
    rstd_flat = rstd.view(-1)

    # 6. Final output kernel
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]
    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64
    BLOCK_K_OUT = 64
    BLOCK_N_OUT = 64
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu_flat,
        rstd_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        w_out_T,
        final,
        R,
        hidden_dim,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )
    return final

"""
TriMul (outgoing) forward pass – Triton‑accelerated.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the last dimension (FP32) using
    `torch.nn.functional.layer_norm`.  This is highly optimized in
    PyTorch and avoids a custom kernel launch.

2️⃣  Fuse the five linear maps
        left_proj, right_proj,
        left_gate, right_gate, out_gate
    into a single GEMM (`torch.nn.functional.linear`).  The weight matrix
    is built on‑the‑fly from the supplied dict.  The result is kept in
    FP16.

3️⃣  Custom Triton kernel `split_fuse_gate_kernel`:
        * splits the fused projection into the three tensors
          `left`, `right` and `out_gate`;
        * applies the optional pair‑wise mask to `left`/`right`;
        * computes all sigmoid gates (FP32 → FP16).

4️⃣  Batched half‑precision matrix multiply:
        `out[b, h] = left[b, h] @ right[b, h].T`
    using `torch.bmm`.  Result `out` has shape `[B, H, N, N]` in FP16.

5️⃣  Compute per‑position mean and rsqrt‑variance of `out` on the GPU
    (`torch.mean/var`).  This replaces a custom row‑wise stats kernel.

6️⃣  Final Triton kernel `final_output_kernel`:
        * normalises `out` with the second LayerNorm parameters;
        * multiplies by the `out_gate` (already sigmoid‑ed);
        * applies the last linear projection (`to_out`) and writes the
          final tensor `[B, N, N, dim]` (FP32).

Only two Triton kernels are needed, the rest is delegated to
high‑performance PyTorch operators.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, optional mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,                # [R, 5*H]   fp16 (R = B*N*N)
    left_ptr,                # [B, H, N, N] fp16
    right_ptr,               # [B, H, N, N] fp16
    out_gate_ptr,            # [B, H, N, N] fp16
    mask_ptr,                # [R] fp16 (ignored if HAS_MASK==0)
    R: tl.int32, H: tl.int32, N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index (b,i,j) from r = b*N*N + i*N + j
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R] fp16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j      # start of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        # proj is stored row‑major: (R, 5*H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]               # [R, H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 → FP16 (more stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                            # no mask

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr      + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 6️⃣ Final LayerNorm + out_gate + output projection (fused)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,                 # [B, H, N, N] fp16
    gate_ptr,                # out_gate, same shape, fp16
    mu_ptr,                  # [R] fp32
    rstd_ptr,                # [R] fp32
    ln_w_ptr,                # to_out_norm.weight  (FP32) [H]
    ln_b_ptr,                # to_out_norm.bias    (FP32) [H]
    w_out_T_ptr,             # transposed final weight [H, dim] fp16
    out_final_ptr,           # [B, N, N, dim] fp32
    total_rows: tl.int32, H: tl.int32, N: tl.int32, dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # position (row) dimension
    pid_n = tl.program_id(1)   # output‑dim (col) dimension

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b,i,j) from linear position index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j    # start of hidden‑plane 0

    # per‑position mean / rsqrt‑var (pre‑computed)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] fp32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] fp32

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor slice and out_gate slice (fp16 → fp32)
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # LayerNorm parameters (γ,β) are per hidden unit
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # Normalise, apply γ/β and the out‑gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                # [M, K]
        term = norm * gate_vals                                      # [M, K]
        term_fp16 = term.to(tl.float16)                              # [M, K] fp16

        # final linear projection tile (K × dim)
        w_tile = tl.load(w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # [K, N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)              # [M, N]

    # write final result `[B, N, N, dim]` (float32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# Custom forward entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys (float32 tensors):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (high‑performance PyTorch implementation)
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension `dim`
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                   # [B,N,N,dim] float32
    x_norm_flat = x_norm.reshape(-1, dim).contiguous()   # [R, dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim) – keep FP32 for precision
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    )                                   # [5*H, dim] FP32
    proj_fp32 = F.linear(x_norm_flat, fused_w)            # [R,5*H] FP32
    proj = proj_fp32.to(torch.float16)                    # cast to FP16 for the kernel
    del fused_w, proj_fp32, x_norm, x_norm_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    use_mask = not cfg.get("nomask", True)
    if use_mask:
        assert mask is not None, "mask must be provided when nomask=False"
        mask_fp16 = mask.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                    # [R]
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 256
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        N,
        HAS_MASK=1 if use_mask else 0,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # left/right are [B, H, N, N] → view as (B*H, N, N) for bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)).view(B, H, N, N)   # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Compute per‑position mean & rsqrt‑variance (second LayerNorm)
    # ------------------------------------------------------------------
    # mean/var across the hidden dimension H
    mu = out.mean(dim=1)                       # [B,N,N] FP32
    var = out.var(dim=1, unbiased=False)       # [B,N,N] FP32
    rstd = torch.rsqrt(var + eps)              # [B,N,N] FP32
    mu_flat   = mu.reshape(-1).contiguous()    # [R] FP32
    rstd_flat = rstd.reshape(-1).contiguous()  # [R] FP32

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final linear projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64
    BLOCK_K_OUT = 64
    BLOCK_N_OUT = 64
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu_flat,
        rstd_flat,
        w["to_out_norm.weight"],   # γ (FP32) [H]
        w["to_out_norm.bias"],     # β (FP32) [H]
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )
    return final
