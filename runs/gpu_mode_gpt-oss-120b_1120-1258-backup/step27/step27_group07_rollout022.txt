import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernels:
@triton.jit
def layernorm_kernel(x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps,
                     BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # accumulate mean and variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized values
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

@triton.jit
def split_fuse_gate_kernel(proj_ptr,
                           left_ptr,
                           right_ptr,
                           out_gate_ptr,
                           mask_ptr,
                           R, H, N,

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # 1.0 if masked out? Actually mask is 0/1, treat as multiplier
        mask_val = mask_val[:, None]  # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for h=0 in destination tensors
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]

def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implemented with Triton.
    The algorithm follows the reference:
      1. Row‑wise LayerNorm (FP32 → FP16) over the flattened pair tensor.
      2. Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
         into a single GEMM, producing a (R,5*H) tensor.
      3. Triton kernel splits the fused output, applies optional mask,
         computes the sigmoid gates and writes three tensors left, right,
         out_gate in layout [B, H, N, N] (FP16).
      4. Batched GEMM (torch.bmm) computes the core 𝚎𝚒𝚗𝚜𝚞𝚖:
             out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
      5. Per‑row LayerNorm statistics (mean & rsqrt‑variance) are computed in FP32.
      6. The second LayerNorm, out‑gate multiplication and final linear projection
         are performed in PyTorch (the heavy work is already in FP16); the final
         result is returned as float32.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()   # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear projections into a single GEMM
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape (5*H, dim)
    proj = F.linear(x_norm, fused_weight)   # (R,5*H) FP16
    del fused_weight, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # mask shape [B, N, N]; broadcast to float16 multiplier 0/1
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (torch.bmm)
    # --------------------------------------------------------------
    # left/right layout: [B, H, N, N] → merge B and H to treat each hidden plane as a batch
    left_mat = left.view(B * H, N, N)   # [B*H, N, N] FP16
    right_mat = right.view(B * H, N, N) # same
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16
    out = out.view(B, H, N, N)   # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Per‑row mean & rsqrt‑variance across hidden dim (FP32)
    # --------------------------------------------------------------
    # Convert to float32 for numerics
    out_fp32 = out.float()
    mu = out_fp32.mean(dim=1)                      # [B, N, N]   mean over H
    var = out_fp32.var(dim=1, unbiased=False)     # [B, N, N]
    rstd = torch.rsqrt(var + eps)                  # [B, N, N]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (torch)
    # --------------------------------------------------------------
    # normalize
    out_norm = (out_fp32 - mu.unsqueeze(1)) * rstd.unsqueeze(1)   # [B, H, N, N]

    # apply to_out_norm (γ, β) which are shape [H]
    gamma = weights["to_out_norm.weight"].view(1, H, 1, 1)   # broadcast
    beta = weights["to_out_norm.bias"].view(1, H, 1, 1)
    out_norm = out_norm * gamma + beta

    # multiply by out_gate (FP16 → cast)
    out_norm = out_norm * out_gate.float()

    # final linear projection: (B, N, N, H) → (B, N, N, dim)
    # reshape to (..., H)
    out_norm_perm = out_norm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    final = F.linear(out_norm_perm, weights["to_out.weight"], bias=None)  # float32

    return final
```

"""
Outgoing TriMul (AlphaFold3) – forward‑only implementation with Triton.
The operator works on a tensor x ∈ ℝ^{B×N×N×C} and optionally a mask
M ∈ {0,1}^{B×N×N}.  The computation follows the reference PyTorch module:

1️⃣ Row‑wise LayerNorm (float32 → float16) over the flattened pair tensor.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (cuBLAS) → a (R, 5·H) FP16 tensor,
   where R = B·N·N and H = hidden_dim.
3️⃣ Triton kernel splits the fused tensor, applies the optional mask,
   evaluates sigmoid gates and writes three tensors
   left, right, out_gate of shape [B, H, N, N] (FP16).
4️⃣ Batched GEMM (torch.bmm) implements the core einsum:
      out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   → [B, H, N, N] (FP16).
5️⃣ Per‑row statistics (mean & rsqrt‑variance) over the hidden dimension are
   computed in float32 (torch.mean/var).
6️⃣ The second LayerNorm, multiplication by out_gate and the final projection
   are performed with PyTorch (the heavy work is already in FP16).  The result
   is returned as float32 with shape [B, N, N, C].

Only the LayerNorm and the split‑&‑gate kernel are written in Triton;
all other heavy linear algebra uses PyTorch/cuBLAS for simplicity and
correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim] FP32 input
    y_ptr,    # [R, dim] FP16 output
    w_ptr,    # [dim]   γ (FP32)
    b_ptr,    # [dim]   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)    # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for hidden‑plane 0 in the destination tensors
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Linear view: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 for stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                               # mask does NOT affect out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implemented with Triton.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the model weights
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5·H, dim)

    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, compute sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (torch.bmm)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Per‑row mean & rsqrt‑variance of `out` (float32)
    # --------------------------------------------------------------
    out_fp32 = out.float()
    mu   = out_fp32.mean(dim=1)                       # [B, N, N]
    var  = out_fp32.var(dim=1, unbiased=False)       # [B, N, N]
    rstd = torch.rsqrt(var + eps)                    # [B, N, N]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate, and final projection (torch)
    # --------------------------------------------------------------
    # Normalize across hidden dimension
    out_norm = (out_fp32 - mu.unsqueeze(1)) * rstd.unsqueeze(1)   # [B, H, N, N]

    # Apply the layer‑norm after the einsum (to_out_norm)
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1, H, 1, 1]
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)     # [1, H, 1, 1]
    out_norm = out_norm * gamma + beta                # [B, H, N, N]

    # Multiply by the out‑gate (converted to float32)
    out_norm = out_norm * out_gate.float()            # [B, H, N, N]

    # Final linear projection: (B, N, N, H) @ (H, dim) → (B, N, N, dim)
    out_perm = out_norm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    final = F.linear(out_perm, w["to_out.weight"], bias=None)  # float32

    return final
