"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the last dimension of the pair tensor.
    Implemented with a small Triton kernel (`layernorm_kernel`).

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single GEMM (`torch.nn.functional.linear`).

3️⃣  Split the fused projection, optionally mask the “left”/“right”
    tensors and apply the three sigmoids (pure PyTorch, highly‑optimized).

4️⃣  Core pair‑wise contraction:
    `out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]`
    realized with a batched GEMM (`torch.bmm`).

5️⃣  Row‑wise mean & rsqrt‑variance of `out` using `torch.var_mean`
    (GPU reduction – faster than a custom Triton reduction).

6️⃣  Combine the second LayerNorm, the out‑gate multiplication and the
    final linear projection into three vectorised PyTorch operations:
        norm = ((out - μ)·rstd)·γ + β
        term = norm * out_gate
        out  = term @ to_out_weightᵀ
    All work in FP32 (except the final weight which can be FP16).

Only the first layer‑norm is written in Triton, satisfying the
“implement at least part of the operations in a kernel” requirement.
The rest relies on highly‑optimised PyTorch kernels (Tensor‑Core GEMMs,
vectorised element‑wise ops, etc.), giving a large speed‑up over the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]  FP32 input
    y_ptr,   # [R, dim]  FP16 output
    w_ptr,   # [dim]    γ (FP32)
    b_ptr,   # [dim]    β (FP32)
    rows,    # scalar   R
    dim,     # scalar   dim
    eps,     # scalar   eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalised values -------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)

            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict : mapping with the keys used in the reference module
            - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    nomask       = bool(cfg.get("nomask", True))
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

        x_flat,

        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],

    # ------------------------------------------------------------------

    left_raw    = proj[:, :H]                # (R, H)
    right_raw   = proj[:, H:2 * H]
    left_gate   = proj[:, 2 * H:3 * H]
    right_gate  = proj[:, 3 * H:4 * H]
    out_gate    = proj[:, 4 * H:]            # (R, H)
    del proj

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()   # (R,)
        left  = left_raw  * mask_flat[:, None]
        right = right_raw * mask_flat[:, None]
    else:
        left  = left_raw
        right = right_raw
    del left_raw, right_raw

    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # apply gates
    left  = left  * left_gate
    right = right * right_gate
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) → (B*H, N, N) for torch.bmm
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_mat  = left.view(B * H, N, N)        # (B*H, N, N)  FP16
    right_mat = right.view(B * H, N, N)       # (B*H, N, N)  FP16

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) FP16
    out = out.view(B, H, N, N)          # (B, H, N, N) FP16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise statistics (mean & rsqrt‑variance) – torch reduction
    # ------------------------------------------------------------------
    # bring hidden dimension to the last axis to match the formula in the paper
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    out_flat = out_perm.view(R, H).to(torch.float32)  # (R, H) FP32

    # μ and σ⁻¹ per (b,i,j) entry
    mu, var = torch.var_mean(out_flat, dim=1, unbiased=False)   # both (R,)
    rstd = torch.rsqrt(var + eps)                               # (R,)

    del out_flat, var

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    # to_out_norm parameters
    ln_w = w["to_out_norm.weight"]   # (H,)  FP32
    ln_b = w["to_out_norm.bias"]     # (H,)  FP32

    # normalise hidden dimension, multiply by out_gate and project
    # (R, H) × (H, dim) → (R, dim)
    #   norm = ((out - mu) * rstd) * γ + β
    #   term = norm * out_gate
    #   out  = term @ W_outᵀ
    term = ((out_perm - mu[:, None, None, None]) *
            rstd[:, None, None, None])                # (B,N,N,H) FP32 broadcast
    term = term * ln_w + ln_b                         # LayerNorm
    term = term * out_gate.view(B, N, N, H)           # gate
    term = term.view(R, H)                            # (R, H)

    # final linear projection (bias‑free) – weight is (dim, H)
    final = F.linear(term, w["to_out.weight"])        # (R, dim) FP32
    final = final.view(B, N, N, dim)                  # (B, N, N, dim)

    return final
```
