import torch.nn.functional as F
import triton

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (AlphaFold‑3).
    This implementation fuses:
      - LayerNorm on the input (using torch.nn.functional.layer_norm).
      - A single linear projection that produces left, right and three gates.
      - A Triton kernel that splits the projection, applies the optional mask,
        computes sigmoids and writes left/right/out_gate tensors.

      - A Triton kernel computing per‑pair mean / rsqrt‑variance of the
        contracted tensor.

        and the final projection to the output dimension.
    All heavy memory‑bound operations are performed in half‑precision,
    while statistics are accumulated in single‑precision.
    The returned tensor is float32 with shape [B, N, N, dim].

    Args:
        data (tuple): (x, mask, weights, cfg)
            x: Tensor[B, N, N, dim] (float32)
            mask: Tensor[B, N, N] (bool/float) or None
            weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            cfg: dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1️⃣ Input LayerNorm (use torch's highly‑optimized implementation)
    # -----------------------------------------------------------------
    # Normalizes over the last dim (dim)

        x,

        bias=w["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,

    proj = F.linear(x_norm, fused_w)  # (R, 5*H) fp16
    # free memory we no longer need
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, apply optional pair mask, and compute sigmoid gates
    # -----------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    @triton.jit
    def split_fuse_gate_kernel(
        proj_ptr, left_ptr, right_ptr, out_gate_ptr,
        mask_ptr, R, H, N,
        HAS_MASK: tl.constexpr,
        BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_R
        offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
        mask_r = offs_r < R

        nn = N * N
        b = offs_r // nn
        rem = offs_r - b * nn
        i = rem // N
        k = rem - i * N

        if HAS_MASK:
            m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
            m_val = m_val[:, None]  # broadcast to hidden dim
        else:
            m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

        # hidden stride for left/right/out_gate layout [B, H, N, N]
        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + k   # start index for h=0

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
            mask_h = offs_h < H

            # each row contains concatenated projections: [left|right|lg|rg|og]
            proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

            left_raw = tl.load(
                proj_ptr + proj_base,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )
            right_raw = tl.load(
                proj_ptr + proj_base + H,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )
            left_gate_raw = tl.load(
                proj_ptr + proj_base + 2 * H,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )
            right_gate_raw = tl.load(
                proj_ptr + proj_base + 3 * H,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )
            out_gate_raw = tl.load(
                proj_ptr + proj_base + 4 * H,
                mask=mask_r[:, None] & mask_h[None, :],
                other=0.0,
            )

            # sigmoid in FP32 → back to FP16
            left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
            right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
            out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

            left = left_raw * m_val * left_gate
            right = right_raw * m_val * right_gate
            outg = out_gate_val

            dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

            tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
            tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
            tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])

    # launch split/fuse kernel
    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Core contraction (batched matmul)
    # -----------------------------------------------------------------
    # reshape to [B*H, N, N] and compute left @ rightᵀ
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] fp16
    out = out.view(B, H, N, N)  # [B, H, N, N] fp16

    # -----------------------------------------------------------------
    # 5️⃣ Row‑wise mean / rsqrt‑variance of the contracted tensor
    # -----------------------------------------------------------------
    @triton.jit
    def rowwise_stats_kernel(
        out_ptr, mu_ptr, rstd_ptr,
        total_rows, H, N,
        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < total_rows

        nn = N * N
        b = offs_m // nn
        rem = offs_m - b * nn
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + j

        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
            vals = tl.load(
                ptr,
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
            ).to(tl.float32)

            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / H
        var  = sum_sq / H - mean * mean
        rstd = tl.rsqrt(var + eps)

        tl.store(mu_ptr + offs_m, mean, mask=mask_m)
        tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = H   # whole hidden dim in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate, and final projection
    # -----------------------------------------------------------------
    @triton.jit
    def final_output_kernel(
        out_ptr, gate_ptr,
        mu_ptr, rstd_ptr,
        ln_w_ptr, ln_b_ptr,
        w_out_T_ptr, out_final_ptr,
        total_rows, H, N, dim,
        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
    ):
        pid_m = tl.program_id(0)  # rows (R)
        pid_n = tl.program_id(1)  # output dim tiles

        row_start = pid_m * BLOCK_M
        col_start = pid_n * BLOCK_N

        offs_m = row_start + tl.arange(0, BLOCK_M)
        offs_n = col_start + tl.arange(0, BLOCK_N)

        mask_m = offs_m < total_rows
        mask_n = offs_n < dim

        nn = N * N
        b = offs_m // nn
        rem = offs_m - b * nn
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + j

        mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)      # [BLOCK_M]
        rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)  # [BLOCK_M]

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        # Since BLOCK_K == H we will have a single iteration.
        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # Load hidden values and gate
            ptr_out = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
            ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

            vals_out = tl.load(
                ptr_out,
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
            ).to(tl.float32)   # [M, K]

            vals_gate = tl.load(
                ptr_gate,
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
            ).to(tl.float32)   # [M, K]

            # LayerNorm parameters for this hidden slice
            gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
            beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [K]

            # (out - µ) * rstd → γ· + β → * out_gate
            diff = (vals_out - mu[:, None]) * rstd[:, None]              # [M, K]
            norm = diff * gamma[None, :] + beta[None, :]                 # [M, K]
            term = norm * vals_gate                                      # [M, K]

            # Cast to fp16 for the matrix‑multiply with final projection weight
            term_fp16 = term.to(tl.float16)  # [M, K]

            # Load a tile of the final linear weight (H × dim) transposed
            w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]  # [K, TILE_N]
            w_tile = tl.load(
                w_ptr,
                mask=mask_k[:, None] & mask_n[None, :],
                other=0.0,
            )  # [K, TILE_N]

            # (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
            acc += tl.dot(term_fp16, w_tile).to(tl.float32)

        # Store final result
        dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
        tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    # Prepare final projection weight transposed [H, dim]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = H   # full hidden dimension
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final

Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Input LayerNorm (torch‑optimised) over the last dimension.
2️⃣  Fuse five linear maps (left, right, left_gate, right_gate, out_gate) into
    a single GEMM → tensor (R, 5·H) where R = B·N·N.
3️⃣  Triton kernel `split_fuse_gate_kernel`:
    - splits the fused projection,
    - applies the optional pair mask,
    - computes three sigmoid gates,
    - writes the three tensors `left`, `right` and `out_gate` in layout
      [B, H, N, N] (FP16).
4️⃣  Batched GEMM (cuBLAS) computes the core contraction
    `out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]` → [B, H, N, N] (FP16).
5️⃣  Triton kernel `rowwise_stats_kernel` computes per‑pair mean and
    rsqrt‑variance of `out` (FP32 statistics, FP16 data).
6️⃣  Triton kernel `final_output_kernel` fuses the second LayerNorm
    (`to_out_norm`), the `out_gate` multiplication and the final projection
    (`to_out`) and writes the final tensor `[B, N, N, dim]` (float32).

    left_ptr,                     # [B, H, N, N] FP16

):

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]

    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # BLOCK_K equals H → a single iteration
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # hidden‑plane values
        ptr_out  = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]          # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]             # [M,K]
        term = norm * vals_gate                                   # [M,K]

        # cast to fp16 for the dot with final projection weight
        term_fp16 = term.to(tl.float16)                           # [M,K]

        # final linear weight (H × dim) transposed → [K, TILE_N]
        w_ptr  = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                               # [K, TILE_N]

        # (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store final tensor (flattened first three dims → row “R”)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch implementation – highly optimised)
    # --------------------------------------------------------------
    # Normalises over the last dimension (dim)
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    # reshape to [R, dim] and cast to half‑precision for the fused GEMM
    x_norm = x_norm.view(R, dim).to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and three gates
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]
    proj = F.linear(x_norm, fused_w)        # (R, 5·H)  fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, and compute sigmoid gates
    # --------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core contraction via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # FP16
    right_mat = right.view(B * H, N, N)               # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise statistics of the contracted tensor
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = H                     # whole hidden dimension at once
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    # to_out.weight : [dim, H] → we need its transpose [H, dim]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = H            # full hidden dim → single iteration
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
