import torch.nn.functional as F

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradients).

    Algorithm:
    1. Apply LayerNorm to input tensor (FP32 → FP16).
    2. Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
       into a single GEMM.
    3. Split the fused projection, apply optional pair‑mask, compute sigmoid gates,
       and write `left`, `right` and `out_gate` tensors (FP16) using a Triton kernel.
    4. Batched GEMM (torch.bmm) contracts `left` and `right` → `out` (FP16).
    5. Row‑wise mean / variance of `out` (FP32) using Torch reductions.
    6. Fuse LayerNorm (γ,β), `out_gate` multiplication and final linear projection
       into a single large GEMM:   X = ((out‑μ)·rstd·γ + β)·out_gate
                                   output = X @ W_out.
    7. Cast the result back to float32 and reshape to `[B, N, N, dim]`.
    """
    # unpack inputs
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    # torch.nn.functional.layer_norm operates on the last dimension.
    # We keep the tensor in FP32 for numerical stability then cast.
    x_ln = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32
    x_fp16 = x_ln.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fused projection (5×hidden_dim)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5·H, dim]

    proj = F.linear(x_fp16, fused_w)          # [R, 5·H] FP16
    del fused_w, x_fp16

    # -----------------------------------------------------------------
    # 3️⃣ Split projection, optional mask and sigmoid gates
    # -----------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
    else:
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)   # dummy tensor
        has_mask = 0

    BLOCK_R = 64
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_fp16,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_fp16

    # -----------------------------------------------------------------
    # 4️⃣ BMM over the hidden dimension (left ⊗ rightᵀ)
    # -----------------------------------------------------------------
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)).view(B, H, N, N)  # [B, H, N, N] FP16

    # -----------------------------------------------------------------
    # 5️⃣ Row‑wise statistics (mean, variance) in FP32
    # -----------------------------------------------------------------
    out_flat = out.view(R, H).float()               # [R, H] FP32
    mu = out_flat.mean(dim=1)                       # [R]
    var = out_flat.var(dim=1, unbiased=False)      # [R]
    rstd = torch.rsqrt(var + eps)                   # [R]

    # -----------------------------------------------------------------
    # 6️⃣ Fuse LN (γ/β), out_gate and final projection into a single GEMM
    # -----------------------------------------------------------------
    # Prepare tensors for the final GEMM: X = ((out - μ)·rstd·γ + β)·out_gate
    gate_flat = out_gate.view(R, H).float()         # [R, H] FP32
    gamma = w["to_out_norm.weight"]                 # [H] FP32

"""
Outgoing TriMul (forward‑only) – high‑performance implementation.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D pair tensor (`torch.nn.functional.layer_norm` – FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (`F.linear`) producing a `[R, 5·H]` tensor.
3️⃣ A Triton kernel (`split_fuse_gate_kernel`) splits the fused projection,
   applies the optional pair‑mask, computes the three sigmoid gates and writes
   the three tensors `left`, `right` and `out_gate` in layout `[B, H, N, N]`
   (FP16).
4️⃣ Batched GEMM (`torch.bmm`) contracts the two projected tensors:
      out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]   → `[B, H, N, N]` (FP16).
5️⃣ Row‑wise statistics (mean & variance) of `out` are computed with cheap
   Torch reductions (FP32).
6️⃣ The second LayerNorm, the out‑gate multiplication and the final projection
   are fused in a single large GEMM:
      X = ((out – μ)·rstd·γ + β) · out_gate   (FP32 → FP16)
      output = X @ W_out                     (FP16 GEMM, final cast to FP32)
7️⃣ Reshape the result to `[B, N, N, dim]`.

Only step 3 is written in Triton; the remaining work leverages highly‑optimised
PyTorch/CUDA kernels, dramatically reducing the overall runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N]  FP16
    right_ptr,     # [B, H, N, N]  FP16
    out_gate_ptr,  # [B, H, N, N]  FP16
    mask_ptr,      # [R]           FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R, 1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for hidden‑plane 0 inside the destination tensors
    base = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Linear view: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Sigmoid (FP32 for stability, back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Triton entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the model weights (FP32)
                keys:
                    "norm.weight", "norm.bias",
                    "left_proj.weight", "right_proj.weight",
                    "left_gate.weight", "right_gate.weight", "out_gate.weight",
                    "to_out_norm.weight", "to_out_norm.bias",
                    "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of pair tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # Torch's layer_norm is heavily tuned; we keep the tensor in FP32 for
    # numerical stability and cast to FP16 afterwards.
    x_ln = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                         # [B, N, N, dim]  FP32
    x_fp16 = x_ln.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)      # [5·H, dim]

    proj = F.linear(x_fp16, fused_w)   # (R, 5·H)  FP16
    del fused_w, x_fp16

    # --------------------------------------------------------------
    # 3️⃣ Split + mask + sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
    else:
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)   # dummy
        has_mask = 0

    BLOCK_R = 64
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_fp16,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (left ⊗ rightᵀ) → out
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & variance (FP32)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).float()          # [R, H] FP32
    mu = out_flat.mean(dim=1)                  # [R]
    var = out_flat.var(dim=1, unbiased=False) # [R]
    rstd = torch.rsqrt(var + eps)              # [R]

    # --------------------------------------------------------------
    # 6️⃣ Fuse LN (γ,β), out_gate and final projection
    # --------------------------------------------------------------
    #   X = ((out - μ)·rstd·γ + β) · out_gate   (FP32 → FP16)
    #   output = X @ W_out   (large GEMM)
    gate_flat = out_gate.view(R, H).float()   # [R, H] FP32
    gamma = w["to_out_norm.weight"]           # [H] FP32
    beta  = w["to_out_norm.bias"]             # [H] FP32

    # Normalisation + affine (still FP32)
    norm = (out_flat - mu[:, None]) * rstd[:, None]       # [R, H]
    norm_affine = norm * gamma[None, :] + beta[None, :]   # [R, H]

    # Apply the out‑gate and cast back to FP16
    X = (norm_affine * gate_flat).to(torch.float16)       # [R, H] FP16

    # Final projection: X @ W_outᵀ  (uses cuBLAS Tensor‑cores)
    w_out = w["to_out.weight"].to(torch.float16)          # [dim, H]
    out_fp16 = F.linear(X, w_out)                         # [R, dim] FP16
    out_fp32 = out_fp16.to(torch.float32).view(B, N, N, dim)  # [B, N, N, dim]

    return out_fp32
