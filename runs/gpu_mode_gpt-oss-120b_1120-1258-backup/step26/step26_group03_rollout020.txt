"""
Outgoing TriMul (AlphaFold3) – forward‑only implementation.

Algorithm
---------
1️⃣ Layer‑norm over the pair representation (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (R × 5·H).
3️⃣ Decode the fused projection, apply the (optional) mask, compute the
   sigmoid gates and store `left`, `right` and `out_gate` (FP16)
   in layout [B, H, N, N].
4️⃣ Batched GEMM (cuBLAS) computes the core einsum
      out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
   → `[B, H, N, N]` (FP16).
5️⃣ One Triton kernel fuses the second LayerNorm, the `out_gate` multiplication,
   and the final projection.  It reads `out` and `out_gate` once, stores them
   in shared memory, computes the per‑row mean/variance, normalises,
   multiplies by the gate, and finally performs the matrix‑multiply with
   `to_out.weight`.  The result is written as `[B, N, N, dim]` (FP32).

All kernels target Triton‑3.3.1, run on an H100 and use mixed‑precision
(arithmetic in FP32, storage in FP16 where possible).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, dim] FP32 input
    y_ptr,       # [R, dim] FP16 output
    w_ptr,       # [dim]    LN weight (γ) FP32
    b_ptr,       # [dim]    LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat row → (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional masking (broadcast over H)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mval = mval[:, None]                                        # [BLOCK_R, 1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # destination layout: [B, H, N, N] → offset = b*(H*nn) + h*nn + i*N + k
    hidden_stride = nn
    base_dest = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # slice inside fused projection (5 consecutive H‑sized blocks)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate
        outg  = out_gate

        dst_off = base_dest[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5+6️⃣ Fuse per‑row statistics, second LayerNorm, out_gate and final proj.
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_proj_kernel(
    out_ptr,        # [B, H, N, N] FP16 (contracted tensor)
    gate_ptr,       # [B, H, N, N] FP16
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [R, dim] FP32  (R = B·N·N)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H_MAX: tl.constexpr, BLOCK_N: tl.constexpr,
):
    # ------------------- row tile -------------------
    pid_row = tl.program_id(0)               # rows
    pid_col = tl.program_id(1)               # output‑dim tiles

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b, i, j) from flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    # offset of hidden‑plane 0 for each (b,i,j)
    base_offset = b * (H * hidden_stride) + i * N + j

    # ------------------------------------------------------------------
    # Shared memory buffers for `out` and `gate` (size = BLOCK_M × BLOCK_H_MAX)
    # ------------------------------------------------------------------
    s_out = tl.shared(tl.float16, BLOCK_M * BLOCK_H_MAX)
    s_gate = tl.shared(tl.float16, BLOCK_M * BLOCK_H_MAX)

    # ----- first pass: load into SMEM + compute sum / sum‑sq -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, H, BLOCK_H_MAX):
        cur_h = h_off + tl.arange(0, BLOCK_H_MAX)               # [BLOCK_H_MAX]
        mask_h = cur_h < H

        # global loads (strided by hidden_stride)
        out_tile = tl.load(
            out_ptr + base_offset[:, None] + cur_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        gate_tile = tl.load(
            gate_ptr + base_offset[:, None] + cur_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )

        # store to shared memory for later reuse
        s_out_ptr = s_out + offs_m[:, None] * BLOCK_H_MAX + cur_h[None, :]
        s_gate_ptr = s_gate + offs_m[:, None] * BLOCK_H_MAX + cur_h[None, :]
        tl.store(s_out_ptr, out_tile, mask=mask_m[:, None] & mask_h[None, :])
        tl.store(s_gate_ptr, gate_tile, mask=mask_m[:, None] & mask_h[None, :])

        # accumulate statistics (FP32)
        out_fp32 = out_tile.to(tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)
        sum_sq  += tl.sum(out_fp32 * out_fp32, axis=1)

    # ----- compute μ and rsqrt‑variance (rstd) -----
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: normalise, apply gate and final projection -----
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for h_off in range(0, H, BLOCK_H_MAX):
        cur_h = h_off + tl.arange(0, BLOCK_H_MAX)
        mask_h = cur_h < H

        # read from shared memory (already in FP16)
        s_out_ptr  = s_out  + offs_m[:, None] * BLOCK_H_MAX + cur_h[None, :]
        s_gate_ptr = s_gate + offs_m[:, None] * BLOCK_H_MAX + cur_h[None, :]
        out_tile  = tl.load(s_out_ptr,
                            mask=mask_m[:, None] & mask_h[None, :]).to(tl.float32)
        gate_tile = tl.load(s_gate_ptr,
                            mask=mask_m[:, None] & mask_h[None, :]).to(tl.float32)

        # layer‑norm parameters (γ, β) for this hidden slice
        gamma = tl.load(ln_w_ptr + cur_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + cur_h, mask=mask_h, other=0.0)   # FP32

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (out_tile - mu[:, None]) * rstd[:, None]            # [M, H]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, H]
        term = norm * gate_tile                                    # [M, H]

        # cast to FP16 for the matmul with W_out
        term_fp16 = term.to(tl.float16)

        # weight tile: [H, dim] → slice (cur_h, offs_n)
        w_ptr = w_out_T_ptr + cur_h[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_h[:, None] & mask_n[None, :],
                         other=0.0)                               # [H, N_out]

        # accumulate (M,H)·(H,N_out) -> (M,N_out) in FP32
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final tensor (flattened over B,N,N)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : "norm.weight", "norm.bias",
                            "left_proj.weight", "right_proj.weight",
                            "left_gate.weight", "right_gate.weight", "out_gate.weight",
                            "to_out_norm.weight", "to_out_norm.bias",
                            "to_out.weight" (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H == 128 in all test cases
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                          # flattened pair tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Single GEMM for the five linear maps
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core einsum
    # ------------------------------------------------------------------
    # view as [B·H, N, N] for torch.bmm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5+6️⃣ Fused second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # Choose block sizes that respect shared‑memory limits:
    BLOCK_M_OUT = 64                     # rows per program
    BLOCK_H_MAX = 128                    # equals max hidden_dim (compile‑time)
    BLOCK_N_OUT = 128                    # output‑dim tiles

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_norm_proj_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_H_MAX=BLOCK_H_MAX,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
