def mask_gate_kernel(

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    B, N, _, _ = input_tensor.shape
    # 1. LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # 2. Linear projections
    # weights are stored as (out_features, in_features)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate_pre = torch.nn.functional.linear(x, weights["left_gate.weight"])
    right_gate_pre = torch.nn.functional.linear(x, weights["right_gate.weight"])
    out_gate_pre = torch.nn.functional.linear(x, weights["out_gate.weight"])
    # 3. mask apply to left and right
    if not nomask and mask is not None:
        # flatten for kernel
        B, N, _, H = left.shape  # left shape [B,N,N,hidden_dim]? Actually left shape [B,N,N,hidden_dim] because linear output hidden_dim (out_features). So H = hidden_dim.
        # Reshape to 1D for kernel
        total = B * N * N * hidden_dim
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        mask_flat = mask.view(-1).to(left.dtype)  # float32
        # Expand mask across hidden dim using kernel
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        apply_mask_kernel[grid](
            left_flat, right_flat,
            mask_flat,
            total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_flat.view(B, N, N, hidden_dim)
        right = right_flat.view(B, N, N, hidden_dim)
    # 4. Gating
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)
    left = left * left_gate
    right = right * right_gate
    # 5. core einsum via batched GEMM
    # Permute to bring hidden as separate batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # Compute out (B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous() # shape [B, N, N, H]
    # 6. Second LayerNorm across hidden dim
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # 7. Multiply by out_gate
    out = out * out_gate
    # 8. Final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

    left = torch.empty_like(left_raw)

    else:

        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        mask_f,
        left,
        right,
        total,
        hidden_dim,
        mask_total,
        HAS_MASK=1 if has_mask else 0,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # free raw tensors?
    del left_raw, right_raw, left_gate_raw, right_gate_raw

    # Step 4: out_gate sigmoid
    out_gate = torch.sigmoid(out_gate_raw)
    del out_gate_raw

    # Step 5: core einsum via batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # Step 6: second LayerNorm
    out = torch.nn.functional.layer_norm(out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps)

    # Step 7: multiply by out_gate
    out = out * out_gate

    # Step 8: final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑Norm over the input channel dim (dim) – torch.nn.functional.layer_norm.
2️⃣  Five linear projections of the normalized tensor:
      * left  , right  – hidden_dim output.
      * left_gate, right_gate, out_gate – hidden_dim output.
3️⃣  Fuse mask‑application and sigmoid‑gates for *left* and *right* with a tiny
    Triton kernel (mask is broadcast over the hidden dimension).
4️⃣  Core “tri‑multiplicative” contraction:
      out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h].
    Implemented as a batched GEMM (torch.bmm) on tensors of shape
    [B·H, N, N].
5️⃣  Second Layer‑Norm over the hidden dimension H (torch.nn.functional.layer_norm),
    followed by element‑wise multiplication with the out‑gate.
6️⃣  Final linear projection back to the original channel dim (dim).
All heavy linear algebra is done with cuBLAS (via torch.nn.functional.linear /
torch.bmm).  The only Triton kernel is the mask‑+‑gate fusion (≈ 1 µs).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: apply mask (if any) and sigmoid gates to left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_raw_ptr,     # fp32,   [B,N,N,H] flattened
    right_raw_ptr,    # fp32,   [B,N,N,H] flattened
    left_gate_ptr,    # fp32,   [B,N,N,H] flattened (pre‑sigmoid)
    right_gate_ptr,   # fp32,   [B,N,N,H] flattened (pre‑sigmoid)
    mask_ptr,         # fp32,   [B,N,N]   flattened, optional
    left_out_ptr,     # fp32,   output left  (same shape as left_raw)
    right_out_ptr,    # fp32,   output right (same shape as right_raw)
    total_elems,      # int32  = B*N*N*H
    hidden_dim,       # int32  = H
    mask_total,       # int32  = B*N*N
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """Flattened kernel:
       - mask is broadcast over the hidden dimension,
       - sigmoid is computed in fp32 for stability.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # [BLOCK]
    valid = offs < total_elems

    # ------------------------------------------------------------------
    # Load raw values
    # ------------------------------------------------------------------
    left_raw  = tl.load(left_raw_ptr  + offs, mask=valid, other=0.0)
    right_raw = tl.load(right_raw_ptr + offs, mask=valid, other=0.0)
    left_gate_raw  = tl.load(left_gate_ptr  + offs, mask=valid, other=0.0)
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # ------------------------------------------------------------------
    # Load (or synthesize) the mask value.
    #     mask is stored per (b,i,k) -> we take the floor division by H.
    # ------------------------------------------------------------------
    if HAS_MASK:
        pair_idx = offs // hidden_dim                     # index into mask_flat
        mask_val = tl.load(mask_ptr + pair_idx,
                           mask=pair_idx < mask_total,
                           other=1.0)                     # fp32 scalar per element
    else:
        mask_val = tl.full([BLOCK], 1.0, dtype=tl.float32)

    # ------------------------------------------------------------------
    # Sigmoid in fp32 (more stable than fp16)
    # ------------------------------------------------------------------
    left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # ------------------------------------------------------------------
    # Apply mask and gates
    # ------------------------------------------------------------------
    left_out  = left_raw  * mask_val * left_gate
    right_out = right_raw * mask_val * right_gate

    # ------------------------------------------------------------------
    # Store results back to the destination buffers
    # ------------------------------------------------------------------
    tl.store(left_out_ptr  + offs, left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    eps = 1e-5

    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Five linear projections (all weight matrices are [out, in])
    # ------------------------------------------------------------------
    left_raw      = F.linear(x, w["left_proj.weight"])      # [B,N,N,H]
    right_raw     = F.linear(x, w["right_proj.weight"])     # [B,N,N,H]
    left_gate_raw = F.linear(x, w["left_gate.weight"])      # [B,N,N,H]
    right_gate_raw= F.linear(x, w["right_gate.weight"])     # [B,N,N,H]
    out_gate_raw  = F.linear(x, w["out_gate.weight"])       # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask (if present) + sigmoid gates for left/right via Triton
    # ------------------------------------------------------------------
    left  = torch.empty_like(left_raw, device=device)
    right = torch.empty_like(right_raw, device=device)

    total_elements = B * N * N * H          # flatten size of left/right tensors
    mask_elements  = B * N * N               # flatten size of the mask

    has_mask = (not nomask) and (mask is not None)
    if has_mask:
        # flatten mask to [B*N*N] and make it float32
        mask_flat = mask.to(torch.float32).reshape(mask_elements).contiguous()
    else:
        # dummy tensor – won't be read because HAS_MASK=0
        mask_flat = torch.empty(0, dtype=torch.float32, device=device)

    # Triton grid: one program per BLOCK-sized chunk of the flattened tensors
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    mask_gate_fuse_kernel[grid](
        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        mask_flat,
        left,
        right,
        total_elements,
        H,
        mask_elements,
        HAS_MASK=1 if has_mask else 0,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # Free raw (pre‑gate) tensors – they are no longer needed
    del left_raw, right_raw, left_gate_raw, right_gate_raw

    # ------------------------------------------------------------------
    # 4️⃣ out‑gate sigmoid (no mask involvement)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)   # [B,N,N,H]
    del out_gate_raw

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction: out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    #    Implemented as a batched GEMM on tensors of shape [B·H, N, N]
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] (contiguous layout required for torch.bmm)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication (cuBLAS)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate
    # ------------------------------------------------------------------
    out = out * out_gate   # broadcast element‑wise

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original channel dim (dim)
    # ------------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B,N,N,dim] – float32

    return out
