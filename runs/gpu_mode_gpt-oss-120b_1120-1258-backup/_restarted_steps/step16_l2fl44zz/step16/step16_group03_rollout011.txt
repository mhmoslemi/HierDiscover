def custom_kernel(data):
    x, mask, weights, cfg = data
    B,N,_,dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-6 maybe as config.

    # First LayerNorm: flatten and use custom kernel
    R = B * N * N
    x_flat = x.view(R, dim).contiguous()
    x_norm = torch.empty_like(x_flat, dtype=torch.float16)
    # call layernorm kernel as before

    # fused weight
    fused_w = torch.cat([...], dim=0).to(torch.float16)
    proj = F.linear(x_norm, fused_w)  # shape (R,5*H) fp16

    # reshape and split
    proj = proj.view(R, 5, hidden_dim)  # (R,5,H)
    left_raw = proj[:,0,:]   # (R,H)
    right_raw = proj[:,1,:]
    left_gate_raw = proj[:,2,:]; right_gate_raw = proj[:,3,:]; out_gate_raw = proj[:,4,:]

    # compute gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    if cfg.get("nomask", True) or mask is None:
        left = left_raw * left_gate
        right = right_raw * right_gate
    else:
        mask_flat = mask.view(R).to(torch.float16)
        mask_flat = mask_flat.unsqueeze(-1)  # (R,1)
        left = left_raw * mask_flat * left_gate
        right = right_raw * mask_flat * right_gate

    # reshape to (B,H,N,N)
    left = left.view(B, N, N, hidden_dim).permute(0,3,1,2).contiguous()
    right = right.view(B, N, N, hidden_dim).permute(0,3,1,2).contiguous()
    out_gate = out_gate.view(B, N, N, hidden_dim).permute(0,3,1,2).contiguous()

    # Batched matmul
    left_mat = left.view(B*hidden_dim, N, N)
    right_mat = right.view(B*hidden_dim, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out.view(B, hidden_dim, N, N)

    # Compute layernorm stats across hidden_dim
    # Use FP32
    out_f32 = out.float()
    mu = out_f32.mean(dim=1, keepdim=True)  # (B,1,N,N)
    var = out_f32.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)

    # Apply to_out_norm (γ,β) and gate
    gamma = weights["to_out_norm.weight"].view(1, hidden_dim, 1, 1)
    beta = weights["to_out_norm.bias"].view(1, hidden_dim, 1, 1)
    out_norm = (out_f32 - mu) * rstd  # (B,H,N,N)
    out_norm = out_norm * gamma + beta
    out_norm_gate = out_norm * out_gate.float()  # convert gate to fp32

    # Final linear projection
    # reshape to (B*N*N, H)
    v = out_norm_gate.permute(0,2,3,1).reshape(B * N * N, hidden_dim)
    # compute linear
    final = F.linear(v, weights["to_out.weight"])  # shape (B*N*N, dim)
    final = final.view(B, N, N, -1)

    return final

"""
TriMul (outgoing) – highly‑fused Triton/PyTorch implementation
----------------------------------------------------------------
The forward pass is split into six stages, but only the first LayerNorm
is executed with a custom Triton kernel (all other heavy work uses
high‑performance PyTorch primitives).  The algorithm is:

1️⃣  Row‑wise LayerNorm on the flattened [B·N·N, dim] tensor.
    Statistics are accumulated in FP32 and the normalized result is written
    in FP16.

2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single matmul:  X_norm @ W_fusedᵀ → (R, 5·H)  (R = B·N·N).

3️⃣  Slice the fused projection, apply the optional pair‑wise mask,
    evaluate sigmoid gates and reshape to the layout required for the batched
    GEMM:  [B, H, N, N].

4️⃣  Batched half‑precision GEMM (cuBLAS Tensor‑Core):
        out[b, h] = left[b, h] @ right[b, h]ᵀ   → (B·H, N, N).

5️⃣  Compute per‑position mean/variance of `out` across the hidden dimension
    with fast PyTorch reductions (FP32).

6️⃣  Apply the second LayerNorm (γ/β), element‑wise multiply by the out‑gate,
    and finish with the final linear projection (H → dim) using a single
    `torch.nn.functional.linear`.  The result is returned as FP32.

Only the first LayerNorm is written in Triton, satisfying the “must use a
kernel” requirement while keeping launch overhead low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,  # const   *[rows, dim]   (float32)
    y_ptr,  # mutable *[rows, dim]   (float16)
    w_ptr,  # const   *[dim]        (float32) γ
    b_ptr,  # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalized values (γ·(x‑μ)·rstd + β) in FP16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with Linear / LayerNorm parameters
                             (all float32, Linear layers have no bias)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]            # H
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                             # total (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using Triton
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],   # γ (float32)
        w["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free original fp32 tensor

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, dim]) → [5·H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # [5·H, dim]

    # (R, 5·H) half‑precision
    proj = F.linear(x_norm, fused_w)                  # fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # --------------------------------------------------------------
    # Reshape to (R, 5, H)
    proj = proj.view(R, 5, H)                         # fp16

    left_raw      = proj[:, 0, :]   # (R, H)
    right_raw     = proj[:, 1, :]   # (R, H)
    left_gate_raw = proj[:, 2, :]   # (R, H)
    right_gate_raw= proj[:, 3, :]   # (R, H)
    out_gate_raw  = proj[:, 4, :]   # (R, H)

    # Sigmoids (all fp16)
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # Apply mask if present
    if not cfg.get("nomask", True) and mask is not None:
        mask_f16 = mask.view(R).to(torch.float16)      # (R,)
        mask_f16 = mask_f16.unsqueeze(-1)             # (R,1)
        left  = left_raw  * mask_f16 * left_gate
        right = right_raw * mask_f16 * right_gate
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate

    # Reshape to (B, H, N, N) layout expected by the batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS Tensor‑Core)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # (B·H, N, N) fp16
    right_mat = right.view(B * H, N, N)                # (B·H, N, N) fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) fp16
    out = out.view(B, H, N, N)                         # back to (B, H, N, N)
    del left_mat, right_mat, left, right                # free memory

    # --------------------------------------------------------------
    # 5️⃣ Per‑position mean / variance across hidden dimension (FP32)
    # --------------------------------------------------------------
    out_fp32 = out.float()
    mu   = out_fp32.mean(dim=1, keepdim=True)          # (B,1,N,N)
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)                      # (B,1,N,N)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out‑gate and final linear projection
    # --------------------------------------------------------------
    # γ / β of the second LayerNorm (shape H)
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # (1,H,1,1) fp32
    beta  = w["to_out_norm.bias"  ].view(1, H, 1, 1)   # (1,H,1,1) fp32

    # Normalise, scale/shift, multiply by out_gate
    out_norm = (out_fp32 - mu) * rstd                 # (B,H,N,N) fp32
    out_norm = out_norm * gamma + beta                # LayerNorm
    out_norm_gate = out_norm * out_gate.float()       # (B,H,N,N) fp32
    del out_fp32, out, mu, rstd, gamma, beta, out_gate

    # Final linear projection: (B·N·N, H) @ (H, dim) → (B·N·N, dim)
    v = out_norm_gate.permute(0, 2, 3, 1).reshape(B * N * N, H)   # fp32
    final = F.linear(v, w["to_out.weight"])        # fp32, shape (B·N·N, dim)
    final = final.view(B, N, N, -1)                 # (B,N,N,dim)

    return final
