"""
TriMul (outgoing) forward pass – Triton implementation.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×D}.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last dimension (FP32 → FP16).
2️⃣ Fuse the five linear projections (left, right, left‑gate, right‑gate, out‑gate)
   into a single GEMM and split the result with a single Triton kernel.
   The kernel also applies the optional pair‑wise mask and the sigmoid gates.
3️⃣ Batched half‑precision matrix multiplication
   left @ rightᵀ (shape (B·H,N,N) → (B·H,N,N)).
4️⃣ Row‑wise mean / variance of the contracted tensor `out`
   over the hidden dimension H.
5️⃣ Fuse the second LayerNorm, the out‑gate and the final Linear projection
   into one Triton kernel that produces the float‑32 output.

All intermediate tensors are kept in fp16 except the statistics
(mean / rstd) and the final result (fp32).  The implementation works for
any batch size (1‑2), sequence length up to 1024 and arbitrary
`dim` / `hidden_dim` (the test suite uses 128‑ and 384‑dimensional
features).  The mask is optional; when omitted a mask of all‑ones is
used.

The code below defines all required Triton kernels and the
`custom_kernel` entry point.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,          # input / output / scale / bias
    rows: tl.int32, dim: tl.int32, eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm over the last dimension.
    x_ptr : [rows, dim] (fp32)
    y_ptr : [rows, dim] (fp16)  <-- normalized result
    w_ptr, b_ptr : [dim] (fp32) (γ,β)
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- accumulate mean / variance (FP32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_tiles = tl.cdiv(dim, BLOCK_D)
    for t in range(num_tiles):
        offs_d = t * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a tile (fp32)
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- write normalized values (γ·(x‑μ)·rstd + β) -----
    for t in range(num_tiles):
        offs_d = t * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # [BLOCK_M, BLOCK_D]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β (fp32)

        out = norm * w[None, :] + b[None, :]               # fp32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,                # [R, 5*H] (fp16)
    left_ptr,                # [B, H, N, N] (fp16)
    right_ptr,               # [B, H, N, N] (fp16)
    out_gate_ptr,            # [B, H, N, N] (fp16)
    mask_ptr,                # [R] (fp16) – ignored if HAS_MASK=0
    R: tl.int32, H: tl.int32, N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Fused split of the (5*H) projection, optional mask and sigmoid gates."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index (b,i,k)
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # Optional scalar mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Layout within fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Sigmoid (computed in FP32 for stability)
        left_gate   = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate  = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate    = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows: tl.int32, H: tl.int32, N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute per‑position mean and rstd over the hidden dimension."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start of hidden slice 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    total_rows: tl.int32, H: tl.int32, N: tl.int32, dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """Second LayerNorm + out_gate + Linear(to_out) fused into one kernel."""
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‐dim blocks

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start of hidden slice 0

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out and gate (fp16 → fp32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm parameters for the hidden dimension
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # fp32

        # Normalise, apply γ/β and the out‑gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                # [M,K]
        term = norm * gate_vals                                     # [M,K]
        term_fp16 = term.to(tl.float16)                             # fp16 for dot

        # Load a tile of the final weight matrix (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                   # [K, N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final float‑32 result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor : [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, D] (fp32)
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 × hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, D]

    proj = F.linear(x_norm, fused_w)                       # (R, 5*H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        has_mask = 1
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM (left @ rightᵀ) over the hidden dim
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B*H, N, N] (fp16)
    right_mat = right.view(B * H, N, N)         # [B*H, N, N] (fp16)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] fp16
    out = out.view(B, H, N, N)                             # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over H)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    # to_out.weight has shape [dim, H]; we need its transpose (H, dim) in fp16
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, D]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64    # rows per program (tunable)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 64    # output‑dim chunk
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
```

**Explanation of the implementation**

* **LayerNorm** – performed row‑wise on the flattened `[R, D]` tensor with
  FP32 accumulation for numerical stability; results are stored as FP16.

* **Fused projection** – the five linear layers (`left_proj`, `right_proj`,
  `left_gate`, `right_gate`, `out_gate`) share the same input, so their
  weights are concatenated and a single `torch.nn.functional.linear` is used.
  The subsequent `split_fuse_gate_kernel` splits the result, applies the optional
  mask and sigmoid gates, and writes the three tensors (`left`, `right`,
  `out_gate`) directly in the required `[B, H, N, N]` layout.

* **Batched GEMM** – `left @ rightᵀ` is carried out with `torch.bmm`; each
  batch corresponds to one hidden channel, staying in FP16.

* **Statistics** – a Triton kernel computes the mean and the reciprocal
  standard deviation of the contracted tensor `out` over the hidden dimension.

* **Final fused kernel** – combines the second `LayerNorm`, multiplication
  by `out_gate`, and the final linear projection (`to_out`).  All operations
  are fused, minimizing memory traffic.  The kernel works on blocks of rows,
  hidden‑dim chunks and output‑dim chunks, accumulating with FP32 precision
  and emitting the final float‑32 result.

The code follows the API required by the evaluation harness:
`custom_kernel((input, mask, weights, config)) → torch.Tensor` with shape
`[B, N, N, dim]`.  It runs entirely on the GPU, uses mixed‑precision for
speed, and respects all optional‑mask and configuration settings.
