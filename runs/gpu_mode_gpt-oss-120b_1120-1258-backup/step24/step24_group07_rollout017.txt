def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B,N,_,_ = x.shape
    # Input layernorm (FP32 -> FP16)
    # reshape
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)   # shape [B,N,N,dim] fp16

    # Fuse projection to 5*H
    fused_w = torch.cat([
        w["left_proj.weight"], w["right_proj.weight"], w["left_gate.weight"], w["right_gate.weight"], w["out_gate.weight"]
    ], dim=0).to(torch.float16)
    # Compute projection on flattened input
    R = B * N * N
    x_flat = x_norm.reshape(R, dim)
    proj = torch.nn.functional.linear(x_flat, fused_w)  # [R, 5*H] fp16

    # Allocate output tensors
    left = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

    # Create mask for split kernel if needed
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # Launch split_fuse_gate kernel
    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # Batched GEMM: left @ right^T
    # reshape for torch.bmm
    left_mat = left.view(B * hidden_dim, N, N)  # [B*H, N, N]
    right_mat = right.view(B * hidden_dim, N, N)  # [B*H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] fp16
    out = out.view(B, hidden_dim, N, N)  # [B, H, N, N] fp16

    # Second LayerNorm over hidden_dim
    # reshape to (B*N*N, hidden_dim)
    out_perm = out.permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out_flat = out_perm.view(-1, hidden_dim)  # [B*N*N, H]
    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    # reshape back to [B, H, N, N]
    out_norm = out_norm.view(B, N, N, hidden_dim).permute(0, 3, 1, 2)  # [B,H,N,N]

    # Apply out_gate
    out_gated = out_norm * out_gate  # fp16 elementwise

    # Final linear projection hidden_dim -> dim
    w_out_fp16 = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

def custom_kernel(data):
    """
    Outgoing TriMul forward pass accelerated with Triton.
    ...
    """
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. Input LayerNorm (float32 -> keep as float32)
    x_norm = F.layer_norm(x, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps)
    x_norm_fp16 = x_norm.to(torch.float16)

    # 2. Fused projection (5 * hidden_dim)
    fused_w = torch.cat([
        w["left_proj.weight"],
        w["right_proj.weight"],
        w["left_gate.weight"],
        w["right_gate.weight"],
        w["out_gate.weight"]
    ], dim=0).to(torch.float16)

    proj = F.linear(x_norm_fp16.view(R, dim), fused_w)  # [R, 5*H] fp16

    # 3. Split + mask + sigmoid gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # 4. Batched GEMM (hidden_dim copies). Use fp16 matmul then cast to fp32.
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    # Compute in fp16 (Tensor cores), then upcast for LN.
    out_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))         # [B*H, N, N] fp16
    out = out_fp16.view(B, H, N, N)                                 # [B, H, N, N] fp16
    out_fp32 = out.float()                                          # [B, H, N, N] fp32

    # 5. LayerNorm over hidden_dim in fp32
    # reshape to (B*N*N, H)
    out_perm = out_fp32.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]
    out_flat = out_perm.view(-1, H)                                 # [B*N*N, H]
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    # reshape back to [B, H, N, N]
    out_norm = out_norm.view(B, N, N, H).permute(0, 3, 1, 2)       # [B, H, N, N] fp32

    # 6. Apply out_gate (cast to fp32) and final projection
    out_gate_fp32 = out_gate.float()
    out_gated = out_norm * out_gate_fp32                               # [B, H, N, N] fp32

    # Linear projection hidden -> dim (fp32 weight)
    w_out_fp32 = w["to_out.weight"]                                    # [dim, H] fp32
    # flatten
    out_in = out_gated.permute(0, 2, 3, 1).contiguous().view(-1, H)   # [B*N*N, H] fp32
    out_final_fp32 = F.linear(out_in, w_out_fp32, bias=None)           # [B*N*N, dim] fp32
    out_final = out_final_fp32.view(B, N, N, dim)                       # [B, N, N, dim] fp32

    return out_final

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and an optional pair‑wise mask
M ∈ {0,1}^{B×N×N}.  The computation consists of:

1️⃣  Row‑wise LayerNorm over the feature dimension C (float32 → float16).
2️⃣  A *fused* linear projection that generates the five weight matrices
    (left, right, left_gate, right_gate, out_gate) with a single GEMM.
3️⃣  Split the fused projection, apply the optional mask and compute sigmoid
    gates – all done in one Triton kernel.
4️⃣  Batched GEMM over the hidden dimension H:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
   (implemented with torch.bmm for maximal cuBLAS performance).
5️⃣  LayerNorm over the hidden dimension H (float32) using the built‑in
    torch.nn.functional.layer_norm (no custom kernel needed here).
6️⃣  Multiply by the out‑gate, then a final linear projection
    (hidden → C) using float32 weights.  The final tensor is returned as
    float32.

Only the split‑+‑gate kernel is written in Triton, satisfying the
“must use a kernel” rule while letting the heavy linear‑algebra kernels
run in highly‑optimized cuBLAS/Triton‑C++ code.  Mixed‑precision is used
where it does not affect numerical fidelity: all LayerNorms and the final
linear are performed in float32, everything else in float16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   fp16 fused projection
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index: r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # Optional per‑position mask (scalar, broadcast over H)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Destination layout base offset for hidden‑plane h = 0
    hidden_stride = N * N                     # stride between successive h‑planes
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # Iterate over hidden dim in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Projection layout = [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Stable sigmoid in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                     # no mask for out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,       left,       mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,      right,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,   outg,       mask=mask_r[:, None] & mask_h[None, :])

        R,
        H,

        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,

    # --------------------------------------------------------------

    w_out = w["to_out.weight"]                                 # [dim, H] fp32
    out_in  = out_gated.permute(0, 2, 3, 1).contiguous().view(-1, H)   # [B·N·N, H]

    out_final_fp32 = F.linear(out_in, w_out, bias=None)        # [B·N·N, dim] fp32
    out_final = out_final_fp32.view(B, N, N, dim)              # [B, N, N, dim]

    return out_final
```
