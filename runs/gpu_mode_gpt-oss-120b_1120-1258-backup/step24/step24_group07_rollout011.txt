"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16).
2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single GEMM; split the result, apply the optional mask,
    and evaluate the sigmoid gates – all in one Triton kernel.
3️⃣  Batched matrix‑multiply:  left[b,h] @ right[b,h].T   (torch.bmm, fp16).
4️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the final
    linear projection (hidden_dim → dim) into a single Triton kernel.
    The kernel first reduces over the hidden dimension to obtain μ and
    σ⁻¹, then computes the normalized values, applies the LN parameters,
    multiplies by the out‑gate, and finally projects to the output dimension.
    No intermediate “out” or statistics tensors are materialised, reducing
    memory traffic dramatically.

The implementation works for any batch size B∈{1,2}, sequence length N,
feature dimension dim (e.g. 128, 384, 768) and hidden dimension hidden_dim
(e.g. 128).  Masks are optional.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim] float32
    y_ptr,    # [R, dim] float16
    w_ptr,    # [dim]    float32 (γ)
    b_ptr,    # [dim]    float32 (β)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] fp16 fused projection
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # Optional mask (scalar per (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Strides for destination tensors
    hidden_stride = N * N                     # stride between successive hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # start offset for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout = [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Sigmoid (FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])

        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_output_kernel(
    out_ptr,          # [B, H, N, N] fp16 (contracted left·right)
    gate_ptr,         # [B, H, N, N] fp16 (out_gate sigmoid)
    ln_w_ptr,         # [H] fp32 (to_out_norm.weight)
    ln_b_ptr,         # [H] fp32 (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim] fp16 (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows = B·N·N
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # Decode (b,i,j) from flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of h=0 for each (b,i,j)

    # ------------------------------------------------------------------
    # First pass: compute mean and rstd across hidden dimension
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: LayerNorm → out_gate → projection
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load tensors for this hidden‑plane tile
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)         # [M, K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd  →  γ· + β  →  * out_gate
        norm = (out_vals - mean[:, None]) * rstd[:, None]           # [M, K]
        norm = norm * gamma[None, :] + beta[None, :]                # [M, K]
        term = norm * gate_vals                                      # [M, K]

        term_fp16 = term.to(tl.float16)     # back to fp16 for dot

        # Load a tile of the final linear weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                   # [K, TILE_N]

        # Accumulate matrix‑multiply (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)             # [M, TILE_N]

    # Store the result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape            # x ∈ [B, N, N, dim]
    R = B * N * N                    # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()            # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    triton.cdiv(R, BLOCK_M)  # for readability
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection producing 5·hidden_dim outputs
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H) fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (torch.bmm)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out.view(B, H, N, N)                             # [B, H, N, N] fp16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate multiplication and final projection
    # --------------------------------------------------------------
    # Transpose the final weight for easier column‑wise loads
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    out_final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64   # rows (R) per program
    BLOCK_K_OUT = 32   # hidden‑dim tile
    BLOCK_N_OUT = 128  # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (fp32)
        w["to_out_norm.bias"],     # β (fp32)
        w_out_T,
        out_final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # Cleanup intermediate tensors
    del out, out_gate, w_out_T

    return out_final
```
This implementation follows the reference PyTorch logic while dramatically reducing memory traffic by fusing the second LayerNorm, out‑gate scaling and the final linear projection into a single Triton kernel. The only remaining heavy work (the batched matrix multiply) is performed by `torch.bmm`, which already delivers near‑optimal performance on H100 GPUs.
