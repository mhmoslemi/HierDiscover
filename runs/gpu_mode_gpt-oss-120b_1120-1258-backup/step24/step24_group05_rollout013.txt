    right = right * right_gate

    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()

"""
Outgoing TriMul (AlphaFold3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the channel dimension *dim* (FP32 → FP16) with a tiny
    Triton kernel (no bias to reduce register pressure).

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into ONE half‑precision GEMM:  
    `proj = x_norm @ W_fused.T`  → shape (R, 5·H),  R = B·N·N.

3️⃣  Split the fused projection, apply the optional pair‑wise mask, evaluate the
    three sigmoid gates and obtain the tensors *left*, *right* and *out_gate*
    (all FP16, layout [B, H, N, N]).

4️⃣  Batched matrix multiplication over the sequence dimension using cuBLAS:  
    `out[b,h] = left[b,h] @ right[b,h].T`  → shape (B, H, N, N) (FP16).

5️⃣  Row‑wise LayerNorm over the hidden dimension *H* (FP32) using the
    built‑in torch reductions, apply the second LayerNorm parameters, multiply
    by *out_gate* and finally a half‑precision linear projection to the original
    *dim*.

All heavy reductions and the final projection use highly‑optimised PyTorch
kernels; only the first LayerNorm is written in Triton.  The implementation
covers both “nomask” and masked variants and works for the full set of test
configs described in the prompt.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm kernel (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]   FP32 input
    y_ptr,          # [R, dim]   FP16 output
    w_ptr,          # [dim]      γ (FP32)
    b_ptr,          # [dim]      β (FP32)
    rows, dim, eps,                     # scalar arguments
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: accumulate sum & sum‑sq ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)
    # ---------- compute mean / rstd ----------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )

    # 1️⃣ Row‑wise LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    # flatten to [R, dim] (FP32)
    x_flat = x.reshape(R, dim).contiguous()
    # allocate FP16 output
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256    # rows per program
    BLOCK_D = 64     # channel tile
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],          # FP32
        w["norm.bias"],            # FP32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # `x` is no longer needed
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H outputs)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # shape (5·H, dim)

    # linear:  (R, dim) @ (dim, 5·H).T  → (R, 5·H)
    proj = F.linear(x_norm, fused_w)   # FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates
    # ------------------------------------------------------------------
    proj = proj.view(R, 5, H)          # (R, 5, H)

    left      = proj[:, 0, :]          # (R, H)
    right     = proj[:, 1, :]          # (R, H)
    left_gate = proj[:, 2, :]
    right_gate= proj[:, 3, :]
    out_gate  = proj[:, 4, :]

    if not nomask and mask is not None:
        mask_flat = mask.view(R).to(torch.float16).unsqueeze(-1)   # (R,1)
        left  = left  * mask_flat
        right = right * mask_flat

    # sigmoid – use FP32 for numerical stability, cast back to FP16
    left_gate  = left_gate.float().sigmoid().to(torch.float16)
    right_gate = right_gate.float().sigmoid().to(torch.float16)
    out_gate   = out_gate.float().sigmoid().to(torch.float16)

    # apply gates
    left  = left  * left_gate
    right = right * right_gate

    # reshape to [B, H, N, N]
    left  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the sequence dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left .view(B * H, N, N)               # (B·H, N, N)
    right_mat = right.view(B * H, N, N)               # (B·H, N, N)

    # out = left @ rightᵀ   → (B·H, N, N)  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                       # (B, H, N, N)  FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise hidden‑dim LayerNorm, out‑gate multiplication,
    #    and final projection – everything in (mostly) torch kernels
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    # mean / variance across the hidden dimension (dim=1)
    mu   = out_f32.mean(dim=1, keepdim=True)                 # (B,1,N,N)
    var  = out_f32.var(dim=1, unbiased=False, keepdim=True) # (B,1,N,N)
    rstd = (var + eps).rsqrt()                               # (B,1,N,N)

    # apply the second LayerNorm (γ,β are of size H)
    ln_w = w["to_out_norm.weight"].view(1, H, 1, 1)   # (1,H,1,1)
    ln_b = w["to_out_norm.bias"  ].view(1, H, 1, 1)   # (1,H,1,1)

    out_norm = (out_f32 - mu) * rstd * ln_w + ln_b    # (B,H,N,N)  FP32

    # gate with out_gate (cast it to FP32)
    gated = out_norm * out_gate.to(torch.float32)     # (B,H,N,N)  FP32

    # flatten for the final linear projection
    gated_flat = gated.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R, H)

    # final projection weight (dim, H) → cast to FP16 for TensorCore GEMM
    w_out = w["to_out.weight"].to(torch.float16)      # (dim, H)

    # half‑precision matmul → half output, then up‑cast to FP32
    final_fp16 = F.linear(gated_flat.to(torch.float16), w_out)   # (R, dim)
    final = final_fp16.to(torch.float32).view(B, N, N, dim)      # (B,N,N,dim)

    return final
```
