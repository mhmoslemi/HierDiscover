"""
Highly‑fused outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the last dimension (FP32 → FP16) using the
    native torch.nn.functional.layer_norm (fast vectorised implementation).

2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
    single GEMM:  X_norm @ W_fusedᵀ  →  (R, 5·H)  where  R = B·N·N.
    The result is kept in FP16.

3️⃣ Triton kernel “split_fuse_gate_kernel”  
   • Splits the fused projection into left / right / gate tensors.  
   • Applies the optional pair‑wise mask (broadcast over the hidden dim).  
   • Computes the three sigmoid‑gates in FP32 for numerical stability.  
   • Writes three FP16 tensors: left (R,H), right (R,H), out_gate (R,H).

4️⃣ Batched matrix‑multiply:
        out[b,h] = left[b,h] @ right[b,h]ᵀ                          (FP16)
   Implemented with a single torch.matmul on the view (B·H, N, N).

5️⃣ Second LayerNorm on the hidden dimension (FP32 → FP16) using the native
    torch.nn.functional.layer_norm, then element‑wise multiplication with
    the previously computed out_gate.

6️⃣ Final linear projection back to `dim` (FP16 GEMM) and cast to FP32.

Only step 3 is a custom Triton kernel; all heavy GEMMs use cuBLAS/Tensor‑Core
operations, giving near‑peak H100 performance while keeping the overall
runtime below the 1 ms target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Kernel 3 – split fused projection, apply mask & sigmoid gates
# ------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK==0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r (0 ≤ r < rows):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    proj layout inside a row: [left, right, left_gate, right_gate, out_gate],
    each slice has length H.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ----------------------------------------------------------------
    # mask per row (1.0 if no mask)
    # ----------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)  # [BLOCK_R]
        mask_val = mask_val[:, None]                         # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ----------------------------------------------------------------
    # iterate over hidden dimension blocks
    # ----------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # offset inside the fused projection (row-major)
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        # load the five slices
        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in FP32 for stability, then back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all linear / layer‑norm weights
                         (float32 tensors, bias=False for Linear layers)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------
    # unpack & basic shapes
    # ------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of (i,j) pairs per batch
    H = hidden_dim

    # ------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – use native torch implementation
    # ------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim]  float32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                                          # float32
    x_norm = x_norm_fp32.to(torch.float16)                    # fp16 for GEMM
    del x_flat, x_norm_fp32

    # ------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + three gates
    # ------------------------------------------------------------
    # concatenate the five weight matrices (each of shape [H, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape [5*H, dim]

    proj = F.linear(x_norm, w_concat)          # (R, 5*H)  fp16
    del w_concat, x_norm

    # ------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates – Triton kernel
    # ------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        HAS_MASK = 1
        mask_half = mask.to(torch.float16).reshape(R)  # flatten mask
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS/Tensor‑Cores)
    # ------------------------------------------------------------
    # reshape to [B, H, N, N] – ready for bmm
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.matmul(
        left.view(B * H, N, N),
        right.view(B * H, N, N).transpose(-1, -2),
    )
    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left, right, out_mat

    # ------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate
    # ------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H]  fp16
    out_norm_fp32 = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                # fp32
    # element‑wise gate (out_gate is fp16)
    out_norm_gate_fp16 = (out_norm_fp32 * out_gate.to(torch.float32)).to(torch.float16)
    del out_flat, out_gate, out, out_norm_fp32

    # ------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 linear) and cast to fp32
    # ------------------------------------------------------------
    final_half = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
