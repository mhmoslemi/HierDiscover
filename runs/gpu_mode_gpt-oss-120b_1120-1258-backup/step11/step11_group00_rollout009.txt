    pre‑computed `out_gate`.

6️⃣  **Final linear projection** back to the original `dim` (FP16 GEMM) and
    cast to FP32.

Only the split‑gate step is Triton‑based; the rest uses PyTorch primitives
which already exploit Tensor‑Core kernels on the H100.  This yields a fast
implementation that meets the correctness requirements while staying
well below the target 1 ms runtime on the supplied benchmarks.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split‑fuse‑gate kernel
#    Input   : proj   – [rows, 5*H]   (FP16)   – fused linear output
#    Outputs : left   – [B, H, N, N]  (FP16)
#              right  – [B, H, N, N]  (FP16)
#              out_gate – [rows, H]   (FP16)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]  (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,    # total rows = B * N * N
    H: tl.int32,       # hidden dimension
    B: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                     # N²
    stride_h = NN                  # distance between two consecutive hidden slices
    stride_b = H * NN              # distance between two consecutive batches

    # ------------------------------------------------------------------
    # Decode flat row index → (b, i, k)
    # ------------------------------------------------------------------
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process hidden dimension in tiles of size BLOCK_H
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside the fused projection (contiguously stored)
        # proj is laid out as [..., 5*H] → we stride by 5*H for each row
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # Sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ------------------------------------------------------------------
        # Store left/right in [B, H, N, N] layout
        #   offset = b * stride_b + i * N + k  (position of first hidden slice)
        # ------------------------------------------------------------------
        base0 = b * stride_b + i * N + k                     # [BLOCK_R]

        left_dst_ptr  = left_ptr  + base0[:, None] + offs_h[None, :] * stride_h
        right_dst_ptr = right_ptr + base0[:, None] + offs_h[None, :] * stride_h

        tl.store(left_dst_ptr,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ------------------------------------------------------------------
        # Store out_gate in flat [rows, H] buffer
        # ------------------------------------------------------------------
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst_ptr,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        weights_dict : mapping of strings to torch.Tensor with keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32
    """
    # -------------------------
    # Unpack inputs
    # -------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    device = x.device
    B, N, _, _ = x.shape
    rows = B * N * N            # B * N²
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # torch's layer_norm is highly optimized; we keep it in FP32 and cast later.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection: produces [rows, 5*H] in FP16
    # ------------------------------------------------------------------
    # Concatenate all five projection matrices (each [H, dim]) → [5*H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # reshape input to [rows, dim] for the linear
    x_flat = x_norm.view(rows, dim)          # contiguous view
    proj = F.linear(x_flat, w_fused)         # [rows, 5*H] FP16
    del w_fused, x_norm, x_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split‑fuse‑gate kernel (Triton)
    # ------------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(rows, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        HAS_MASK = 1
        # flatten mask to [rows] and convert to FP16 (0.0 / 1.0)
        mask_fp16 = mask.to(torch.float16).reshape(rows)
        mask_ptr = mask_fp16

    # Triton launch configuration
    BLOCK_R = 128      # rows per program
    BLOCK_H = 32       # hidden tiles per iteration
    grid = (triton.cdiv(rows, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        rows,
        H,
        B,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B*H, N, N] for batched matrix multiplication
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out = left @ rightᵀ  → shape [B*H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (torch)
    # ------------------------------------------------------------------
    # torch.layer_norm on the hidden dimension
    out_norm = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # multiply by out_gate (reshape to [B, N, N, H])
    out_gate_4d = out_gate.view(B, N, N, H)
    gated = out_norm * out_gate_4d
    del out, out_norm, out_gate, out_gate_4d

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (FP16 GEMM) and cast to FP32
    # ------------------------------------------------------------------
    # flatten to [rows, H] for the linear
    gated_flat = gated.view(rows, H)

    final_half = F.linear(
        gated_flat,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [rows, dim] (FP16)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
