"""
TriMul (outgoing) – fast Triton‑fused implementation
===================================================

Algorithm (forward only, no gradients)
-------------------------------------
1️⃣ Row‑wise LayerNorm over the last dimension (float32 → float16) – custom Triton.
2️⃣ Fuse the five linear maps (left/right projections + three gates) into a single
   GEMM:   X_norm @ W_fusedᵀ  →  (R, 5·H)   where R = B·N·N.
3️⃣ One Triton kernel splits the fused result, applies the optional pair‑wise mask,
   evaluates sigmoid gates (in fp32 for stability) and writes three tensors:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
4️⃣ Batched fp16 GEMM (cuBLAS/Tensor‑Cores) over the hidden dimension:
        out[b, h] = left[b, h] @ right[b, h]ᵀ   (B·H independent N×N matmuls)
5️⃣ Second LayerNorm + element‑wise multiplication with out_gate – custom Triton.
6️⃣ Final fp16 linear projection back to `dim` (no bias) and cast to fp32.

Only three kernels are custom; the heavy matrix multiplications use cuBLAS
through `torch.nn.functional.linear` / `torch.matmul`, which gives near‑peak
performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Kernel 1 – row‑wise LayerNorm (fp32 → fp16)
# ------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]      (float32)
    y_ptr,          # mutable *[rows, dim]      (float16)
    weight_ptr,     # const   *[dim]           (float32)  γ
    bias_ptr,       # const   *[dim]           (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean / variance (accumulate in fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values (γ·(x‑μ)·rstd + β) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]           # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ------------------------------------------------------------
# Kernel 2 – split fused projection, apply mask & sigmoid gates
# ------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r (r ∈ [0,R)):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are FP16 except the sigmoid (computed in FP32 for stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # Load per‑row mask (1.0 if no mask)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)      # [BLOCK_R]
        mask_val = mask_val[:, None]                         # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # Base offset for the first segment (proj_left)
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # Sigmoid in fp32
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Store results
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ------------------------------------------------------------
# Kernel 3 – second LayerNorm + multiplication with out_gate
# ------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16) – out_gate
    y_ptr,            # mutable *[rows, dim]   (float16) – result
    weight_ptr,       # const   *[dim]        (float32) γ
    bias_ptr,         # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean / variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, apply γ/β, multiply with gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)
        norm = norm * w[None, :] + b[None, :]                     # fp32

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32

        out = norm * gate                                        # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all linear / layer‑norm weights
                         (float32 tensors, bias=False for Linear layers)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack and basic shapes
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # rows = B·N·N
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (fp32 → fp16) – Triton
    # -----------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # -----------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + three gates
    # -----------------------------------------------------------------
    # concatenate the five weight matrices (order matches split_fuse_gate_kernel)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)          # shape: [5*H, dim]

    # (R, 5·H) = X_norm (R, dim) @ W_fusedᵀ
    proj = F.linear(x_norm, w_concat)   # fp16
    del w_concat, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates – Triton
    # -----------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        HAS_MASK = 1
        # flatten mask (bool/float) → half, shape [R]
        mask_half = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # -----------------------------------------------------------------
    # reshape to [B, H, N, N] for a single batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)  (fp16)
    out_mat = torch.matmul(left, right.transpose(-1, -2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate – Triton
    # -----------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid_ln2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid_ln2](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out

    # -----------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 linear) and cast to fp32
    # -----------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
