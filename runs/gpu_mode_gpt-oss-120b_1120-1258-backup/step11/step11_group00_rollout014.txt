"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
The kernel hierarchy follows the exact forward pass of the reference PyTorch
implementation but removes all unnecessary transposes and intermediate
allocations:

1️⃣  Row‑wise LayerNorm on the 4‑D input (FP32 → FP16) – performed with the
    highly optimised ATen `torch.nn.functional.layer_norm`.

2️⃣  A single cuBLAS GEMM produces the fused projection of shape
    `[rows, 5*H]` (five linear layers concatenated), where
    `rows = B * N * N` and `H = hidden_dim`.

3️⃣  `split_fuse_gate_kernel` (Triton) splits the fused projection,
    applies the optional mask, the three sigmoid gates and stores the
    left / right tensors directly in the layout required for the batch GEMM:
    `[B, H, N, N]`.  This avoids the expensive `transpose` + `view` steps
    present in the reference code.

4️⃣  Batched GEMM (`torch.bmm`) computes the true tri‑multiplicative update:
    `out = left @ rightᵀ`  with half‑precision Tensor Cores.

5️⃣  `layernorm_gate_kernel` (Triton) fuses the second LayerNorm,
    its affine parameters and the element‑wise multiplication by `out_gate`.

6️⃣  Final linear projection back to `dim` (cuBLAS GEMM) and cast to FP32.

The kernels are tuned for H100 (block sizes chosen to keep register pressure
low while exposing enough parallelism).  The implementation works for all
configurations listed in the problem statement and supports both masked and
unmasked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates, write `left`/`right`
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # const   *[rows, 5*H]   (fp16)
    left_ptr,      # mutable *[B, H, N, N] (fp16)
    right_ptr,     # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,  # mutable *[rows, H]   (fp16)
    mask_ptr,      # const   *[rows]      (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,      # rows = B * N * N
    H: tl.int32,
    B: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                 # N²
    stride_h = NN              # offset between two consecutive hidden slices
    stride_b = H * NN          # offset between two consecutive batches

    # decode (b, i, k) from the flat row index
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process the hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # address for every (row, hidden) pair inside the fused projection
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        # load the five slices (left, right, 3 gates)
        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        out_g_raw   = tl.load(proj_ptr + base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)

        # sigmoid (use triton helper for speed & stability)
        left_gate  = tl.sigmoid(left_g_raw.to(tl.float32)).to(tl.float16)
        right_gate = tl.sigmoid(right_g_raw.to(tl.float32)).to(tl.float16)
        out_gate   = tl.sigmoid(out_g_raw.to(tl.float32)).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # compute flat offset for first hidden slice of this row:
        #   offset = b * stride_b + i * N + k
        base0 = b * stride_b + i * N + k                # [BLOCK_R]

        # store left / right in [B, H, N, N] layout
        left_dst_ptr  = left_ptr  + base0[:, None] + offs_h[None, :] * stride_h
        right_dst_ptr = right_ptr + base0[:, None] + offs_h[None, :] * stride_h

        tl.store(left_dst_ptr,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # store out_gate (still flat [rows, H])
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst_ptr,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Row‑wise LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (fp16)
    gate_ptr,       # const   *[rows, dim]   (fp16)
    y_ptr,          # mutable *[rows, dim]   (fp16)
    weight_ptr,     # const   *[dim]        (fp32) γ
    bias_ptr,       # const   *[dim]        (fp32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # Compute mean & variance (fp32 accumulator)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, apply LN weight/bias and multiply by gate
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals       # fp32
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    rows = B * N * N                           # total pair‑wise rows
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using ATen kernel (fast)
    # --------------------------------------------------------------
    # shape [rows, dim]
    x_flat = x.view(rows, dim).contiguous()
    # built‑in LayerNorm – operates in FP32, then cast
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fused Linear → [rows, 5*H]  (FP16)
    # --------------------------------------------------------------
    # concatenate the five projection matrices once (FP16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]

    proj = F.linear(x_norm, w_fused)        # [rows, 5*H]  FP16
    del w_fused, x_norm                     # free memory early

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask & gates, write `left` / `right`
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(rows, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask broadcasted over the hidden dimension
        mask_fp16 = mask.to(torch.float16).reshape(rows)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 256          # rows per program
    BLOCK_H = 64           # hidden tiles per program
    grid_split = (triton.cdiv(rows, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        rows,
        H,
        B,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM, FP16 TensorCores)
    # --------------------------------------------------------------
    # reshape to [B*H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # right needs to be transposed for the inner product
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # reshape back to [B, N, N, H] for the next LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused Triton kernel)
    # --------------------------------------------------------------
    out_flat = out.view(rows, H).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 512          # rows per program (larger block reduces launch overhead)
    BLOCK_D = 32           # dimension tile
    grid_ln2 = (triton.cdiv(rows, BLOCK_M),)
    layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (FP16 GEMM) + cast to FP32
    # --------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [rows, dim]  fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
