"""
Highly‑optimized forward pass of the *outgoing* TriMul operator
used in AlphaFold‑3, Chai, Protenix, ….

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (fp32 → fp16).
2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single half‑precision GEMM.
3️⃣ Triton kernel `split_fuse_gate_kernel`:
      * decodes the flat row index (b,i,k)  →  (batch, query, key)
      * applies the optional pair‑wise mask,
      * evaluates the three σ‑gates,
      * writes **left**, **right** and **out_gate** directly in layout
        `[B, H, N, N]` (no extra transposes).
4️⃣ Batched matrix multiplication (cuBLAS Tensor‑cores):
        out[b·H+i, j] = Σₖ left[b·H,i,k] * right[b·H,j,k]ᵀ
5️⃣ Row‑wise LayerNorm on the result + element‑wise multiplication with
   `out_gate` (fused in `layernorm_mul_kernel`).
6️⃣ Final linear projection back to the original **dim**.

All kernels target Triton 3.3.1 and an NVIDIA H100.  If a GPU is not
available the implementation falls back to a pure‑PyTorch reference
(which is still functionally correct, albeit slower).
"""

import torch
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Try to import Triton; if it fails (e.g. CPU only) we will use a fallback.
# ----------------------------------------------------------------------
try:
    import triton
    import triton.language as tl
    _HAS_TRITON = True
except Exception:  # pragma: no cover
    _HAS_TRITON = False


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
if _HAS_TRITON:
    @triton.jit
    def layernorm_kernel(
        x_ptr,          # const   *[rows, dim]   (float32)
        y_ptr,          # mutable *[rows, dim]   (float16)
        weight_ptr,     # const   *[dim]        (float32)   γ
        bias_ptr,       # const   *[dim]        (float32)   β
        rows: tl.int32,
        dim: tl.int32,
        eps: tl.float32,
        BLOCK_M: tl.constexpr,   # rows per program
        BLOCK_D: tl.constexpr,   # cols per iteration
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
        mask_m = offs_m < rows

        # ---- mean / variance (accumulate in fp32) ----
        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)
            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / dim
        var  = sum_sq / dim - mean * mean
        rstd = tl.rsqrt(var + eps)

        # ---- write normalized values (fp16) ----
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

            norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

            w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
            b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

            out = norm * w[None, :] + b[None, :]                    # fp32
            out_fp16 = out.to(tl.float16)

            out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
            tl.store(out_ptr, out_fp16,
                     mask=mask_m[:, None] & mask_d[None, :])

    # ----------------------------------------------------------------------
    # 2️⃣ Split fused projection, apply mask & σ‑gates, write in [B,H,N,N]
    # ----------------------------------------------------------------------
    @triton.jit
    def split_fuse_gate_kernel(
        proj_ptr,          # const   *[rows, 5*H]   (float16)
        left_ptr,          # mutable *[B, H, N, N] (float16)
        right_ptr,         # mutable *[B, H, N, N] (float16)
        out_gate_ptr,      # mutable *[B, H, N, N] (float16)
        mask_ptr,          # const   *[rows]      (float16) – ignored if HAS_MASK=0
        rows: tl.int32,    # R = B·N·N
        N: tl.int32,       # seq len
        H: tl.int32,       # hidden dim
        HAS_MASK: tl.constexpr,
        BLOCK_R: tl.constexpr,
        BLOCK_H: tl.constexpr,
    ):
        """Produce left/right/out_gate from fused projections."""
        pid = tl.program_id(0)
        row_start = pid * BLOCK_R
        offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
        mask_r = offs_r < rows

        # ---- decode (b,i,k) from linear row id ----
        seq_sq = N * N
        b = offs_r // seq_sq
        rem = offs_r - b * seq_sq
        i = rem // N
        k = rem - i * N

        # ---- optional pair‑wise mask (scalar per row) ----
        if HAS_MASK:
            # mask values are stored as 0/1 (float16)
            m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
        else:
            m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

        # ---- loop over hidden dimension in chunks ----
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
            mask_h = offs_h < H

            # layout inside proj: [left, right, left_gate, right_gate, out_gate]
            base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, Hc]

            left_raw_ptr    = proj_ptr + base
            right_raw_ptr   = proj_ptr + base + H
            left_gate_ptr   = proj_ptr + base + 2 * H
            right_gate_ptr  = proj_ptr + base + 3 * H
            out_gate_raw_ptr= proj_ptr + base + 4 * H

            left_raw    = tl.load(left_raw_ptr,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)
            right_raw   = tl.load(right_raw_ptr,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)
            left_g_raw  = tl.load(left_gate_ptr,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)
            right_g_raw = tl.load(right_gate_ptr,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)
            out_g_raw   = tl.load(out_gate_raw_ptr,
                                  mask=mask_r[:, None] & mask_h[None, :],
                                  other=0.0)

            # ---- sigmoid (in fp32 for stability) ----
            left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
            right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
            out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

            # ---- apply mask and gates ----
            left  = left_raw  * m_val * left_gate
            right = right_raw * m_val * right_gate

            # ---- compute flat destination offsets for [B, H, N, N] ----
            # flat = ((b * H + h) * N + i) * N + k
            b_exp = b[:, None]          # [R,1]
            i_exp = i[:, None]
            k_exp = k[:, None]
            h_exp = offs_h[None, :]     # [1,Hc]

            dst_off = ((b_exp * H + h_exp) * N + i_exp) * N + k_exp   # [R,Hc]

            # ---- store ----
            tl.store(left_ptr      + dst_off, left,      mask=mask_r[:, None] & mask_h[None, :])
            tl.store(right_ptr     + dst_off, right,     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(out_gate_ptr  + dst_off, out_gate,  mask=mask_r[:, None] & mask_h[None, :])

    # ----------------------------------------------------------------------
    # 3️⃣ Row‑wise LayerNorm + multiplication with out_gate (fused)
    # ----------------------------------------------------------------------
    @triton.jit
    def layernorm_mul_kernel(
        x_ptr,            # const   *[rows, dim]   (float16)
        gate_ptr,         # const   *[rows, dim]   (float16)
        y_ptr,            # mutable *[rows, dim]   (float16)
        weight_ptr,       # const   *[dim]        (float32) γ
        bias_ptr,         # const   *[dim]        (float32) β
        rows: tl.int32,
        dim: tl.int32,
        eps: tl.float32,
        BLOCK_M: tl.constexpr,
        BLOCK_D: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
        mask_m = offs_m < rows

        # ---- mean / variance (fp32) ----
        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)
            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / dim
        var  = sum_sq / dim - mean * mean
        rstd = tl.rsqrt(var + eps)

        # ---- normalize, apply γ/β, multiply with gate, store (fp16) ----
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                             mask=mask_m[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)

            norm = (x_vals - mean[:, None]) * rstd[:, None]        # fp32

            w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
            b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

            norm = norm * w[None, :] + b[None, :]                     # fp32

            gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

            out = norm * gate                                        # fp32
            out_fp16 = out.to(tl.float16)

            tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                     out_fp16,
                     mask=mask_m[:, None] & mask_d[None, :])

# Helper: pick modest block sizes (tuned for typical shapes)
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    """Return (BLOCK_M, BLOCK_D) adapted to the problem size."""
    # Heuristics: more rows → larger BLOCK_M, while BLOCK_D stays modest.
    BLOCK_M = 256 if rows >= 8192 else 128
    BLOCK_D = 64 if dim >= 64 else dim
    return BLOCK_M, BLOCK_D

# Fallback pure‑PyTorch implementation (used when Triton is unavailable)

def _tri_mul_reference(x, mask, weights, cfg):
    """Reference implementation matching the semantics of the Triton version."""
    eps = 1e-5
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    B, N, _, _ = x.shape

    # 1️⃣ LayerNorm (float32)
    x = (x - x.mean(-1, keepdim=True)) / torch.sqrt(x.var(-1, unbiased=False, keepdim=True) + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # 3️⃣ Apply mask (if present)
    if mask is not None:
        mask = mask.unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # 4️⃣ Gate the projections
    left = left * left_gate
    right = right * right_gate

    # 5️⃣ Outer product across the third dimension (k)
    out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)  # (B,N,N,hidden)

    # 6️⃣ Second LayerNorm + out_gate multiplication
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out = out * out_gate

    # 7️⃣ Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors; Linear layers are bias‑free)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    # ------------------------------------------------------------------
    # Quick fallback if Triton is unavailable or CUDA is missing
    # ------------------------------------------------------------------
    if not _HAS_TRITON or not x.is_cuda:
        # Note: the reference implementation is still correct.
        return _tri_mul_reference(x, mask if not nomask else None, weights, cfg)

    # ------------------------------------------------------------------
    # Shape bookkeeping
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    R = B * N * N                # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()            # [R, dim] (fp32)
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M, BLOCK_D = _choose_block_sizes(R, dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],     # γ (float32)
        weights["norm.bias"],       # β (float32)
        R, dim, eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim outputs)
    # ------------------------------------------------------------------
    # Stack weights: [left_proj; right_proj; left_gate; right_gate; out_gate]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5*H, dim)

    proj = F.linear(x_norm, w_fused)          # (R, 5*H) fp16
    del w_fused, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, σ‑gates → left / right / out_gate in [B,H,N,N]
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if nomask:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        HAS_MASK = 0
    else:
        # flatten mask to [R] with 0/1 values (float16)
        mask_flat = mask.to(torch.float16).reshape(R)
        HAS_MASK = 1

    BLOCK_R = 128   # rows per program (tunable)
    BLOCK_H = 32    # hidden‑dim chunk size (tunable)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        N,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension: left @ rightᵀ
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) – no transpose needed for left,
    # but right must be transposed on the second axis.
    left_mat = left.view(B * H, N, N)                     # (B·H, N, N)
    right_mat = right.view(B * H, N, N)                   # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    out = out_mat.view(B, H, N, N)                     # (B, H, N, N)
    del left_mat, right_mat, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate (fused)
    # ------------------------------------------------------------------
    # Rearrange to [R, H] where R = B·N·N
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)           # (R, H)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R, H)

    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2, BLOCK_D2 = _choose_block_sizes(R, H)
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out_flat, out_gate_flat, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # (dim, H) fp16
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
