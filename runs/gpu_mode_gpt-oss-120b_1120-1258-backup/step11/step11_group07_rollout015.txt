@triton.jit
def fused_proj_gate_kernel(
    x_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    weight_ptr,
    rows: tl.int32, dim: tl.int32,
    H: tl.constexpr, OUT_FEAT: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # accumulator
    acc = tl.zeros((BLOCK_R, OUT_FEAT), dtype=tl.float32)

    col_range = tl.arange(0, OUT_FEAT)

    for k_start in range(0, tl.cdiv(dim, BLOCK_K)):
        k_off = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < dim

        # load x block
        x = tl.load(x_ptr + offs_r[:, None] * dim + k_off[None, :],
                    mask=mask_r[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # load weight block (BLOCK_K, OUT_FEAT)
        w = tl.load(weight_ptr + k_off[:, None] * OUT_FEAT + col_range[None, :],
                    mask=mask_k[:, None],
                    other=0.0).to(tl.float32)

        acc += tl.dot(x, w)

    # split accumulator
    left_raw = acc[:, 0:H].to(tl.float16)
    right_raw = acc[:, H:2*H].to(tl.float16)
    left_gate_raw = acc[:, 2*H:3*H].to(tl.float16)
    right_gate_raw = acc[:, 3*H:4*H].to(tl.float16)
    out_gate_raw = acc[:, 4*H:5*H].to(tl.float16)

    # sigmoid gates
    left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
    out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]  # broadcast to [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    left = left_raw * mask_val * left_gate
    right = right_raw * mask_val * right_gate

    # store results
    col_range_H = tl.arange(0, H)
    tl.store(left_ptr + offs_r[:, None] * H + col_range_H[None, :],

             right,
             mask=mask_r[:, None])

    tl.store(out_gate_ptr + offs_r[:, None] * H + col_range_H[None, :],
             out_gate,
             mask=mask_r[:, None])

"""
Highly‑optimized Triton implementation of the *outgoing* Triangular‑Multiplicative
update (TriMul) used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (float32) → fp16.
2️⃣ Fused projection‑gate kernel:
   * Computes the GEMM  X (R×dim)  ×  Wᵀ (dim×5·H)  in‑place.
   * Splits the 5·H results into
        left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw.
   * Applies the pairwise mask (if any) and the three sigmoid gates,
     writing directly the three tensors required for the triangular update:
        left   = left_raw   * mask * sigmoid(left_gate_raw)
        right  = right_raw  * mask * sigmoid(right_gate_raw)
        out_gate = sigmoid(out_gate_raw)
   This eliminates the large intermediate (R,5·H) tensor of the reference code.
3️⃣ Batched GEMM (Tensor‑Core) over the hidden dimension:
        out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]
   performed with `torch.bmm`.
4️⃣ Row‑wise LayerNorm on the result, element‑wise multiplication by `out_gate`.
5️⃣ Final linear projection back to `dim` (Tensor‑Core GEMM) and cast to float32.

Only the two row‑wise LayerNorm passes and the fused projection‑gate kernel are
written in Triton; all heavy matrix multiplications use cuBLAS/Tensor‑Core for
peak performance on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,               # const *[rows, dim] (fp32)
    y_ptr,               # mutable *[rows, dim] (fp16)
    weight_ptr,          # const *[dim] (fp32)   γ
    bias_ptr,            # const *[dim] (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- mean / variance (fp32) ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (fp16) ----------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]               # fp32 -> fp16
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gate kernel (produces left, right, out_gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,                # const *[rows, dim] (fp16)
    left_ptr,             # mutable *[rows, H] (fp16)
    right_ptr,            # mutable *[rows, H] (fp16)
    out_gate_ptr,         # mutable *[rows, H] (fp16)
    mask_ptr,             # const *[rows]    (fp16) – ignored if HAS_MASK=0
    weight_ptr,           # const *[dim, OUT_FEAT] (fp16); OUT_FEAT = 5*H
    rows: tl.int32,
    dim: tl.int32,
    H: tl.constexpr,              # hidden dimension
    OUT_FEAT: tl.constexpr,      # =5*H
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    For each row:
        left   = (proj_left   * mask) * sigmoid(proj_left_gate)
        right  = (proj_right  * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All work is done in fp16, sigmoids are evaluated in fp32 for stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # accumulator for the whole (R x 5*H) projection
    acc = tl.zeros((BLOCK_R, OUT_FEAT), dtype=tl.float32)

    col_range = tl.arange(0, OUT_FEAT)                  # compile‑time column ids

    for k_start in range(0, tl.cdiv(dim, BLOCK_K)):
        k_off = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < dim

        # X block (BLOCK_R x BLOCK_K)
        x = tl.load(x_ptr + offs_r[:, None] * dim + k_off[None, :],
                    mask=mask_r[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # weight block (BLOCK_K x OUT_FEAT)
        w = tl.load(weight_ptr + k_off[:, None] * OUT_FEAT + col_range[None, :],
                    mask=mask_k[:, None],
                    other=0.0).to(tl.float32)

        acc += tl.dot(x, w)        # (BLOCK_R, OUT_FEAT)

    # ---- split the accumulated result ---------------------------------
    left_raw      = acc[:, 0:H]          .to(tl.float16)
    right_raw     = acc[:, H:2*H]        .to(tl.float16)
    left_gate_raw = acc[:, 2*H:3*H]      .to(tl.float16)
    right_gate_raw= acc[:, 3*H:4*H]      .to(tl.float16)
    out_gate_raw  = acc[:, 4*H:5*H]      .to(tl.float16)

    # ---- sigmoids (fp32 for stability) --------------------------------
    left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
    out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw  .to(tl.float32)))).to(tl.float16)

    # ---- optional mask -------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r,
                    mask=mask_r,
                    other=1.0).to(tl.float16)      # [BLOCK_R]
        m = m[:, None]                                 # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- apply mask & gates --------------------------------------------
    left  = left_raw  * m * left_gate
    right = right_raw * m * right_gate

    # ---- store ---------------------------------------------------------
    col_h = tl.arange(0, H)    # column ids for the three output tensors

    tl.store(left_ptr     + offs_r[:, None] * H + col_h[None, :],
             left,
             mask=mask_r[:, None])
    tl.store(right_ptr    + offs_r[:, None] * H + col_h[None, :],
             right,
             mask=mask_r[:, None])
    tl.store(out_gate_ptr + offs_r[:, None] * H + col_h[None, :],
             out_gate,
             mask=mask_r[:, None])


# ----------------------------------------------------------------------
# 4️⃣ Row‑wise LayerNorm + multiplication by out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16) – out_gate
    y_ptr,            # mutable *[rows, dim]   (fp16)
    weight_ptr,       # const   *[dim]        (fp32) γ
    bias_ptr,         # const   *[dim]        (fp32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- mean / variance (fp32) --------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- normalize, apply γ/β, multiply with gate --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)
        norm = norm * w[None, :] + b[None, :]                     # fp32

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                          # fp32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Helpers for block size selection
# ----------------------------------------------------------------------
def _choose_block_sizes_rows(rows: int, dim: int):
    BLOCK_M = 128 if rows >= 128 else max(1, rows)
    BLOCK_D = 32 if dim >= 32 else dim
    return BLOCK_M, BLOCK_D


def _choose_proj_block_sizes(rows: int, dim: int, hidden: int):
    # keep registers modest (≈80 KB) → BLOCK_R ≤ 32, BLOCK_K = 32 (or dim)
    BLOCK_R = 32 if rows >= 32 else rows
    BLOCK_K = 32 if dim >= 32 else dim
    return BLOCK_R, BLOCK_K


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the model weights (float32, bias=False for Linear)
        - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor  shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    nomask     = cfg.get("nomask", True)
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of rows for row‑wise ops
    H = hidden_dim
    OUT_FEAT = 5 * H                  # left, right, three gates

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=weights["norm.weight"],
                          bias=weights["norm.bias"],
                          eps=eps)                      # still fp32

    x_norm = x_norm.view(R, dim).contiguous()                # [R, dim] fp32
    x_norm_h = x_norm.to(torch.float16)                      # fp16 for the kernels

    BLOCK_M1, BLOCK_D1 = _choose_block_sizes_rows(R, dim)
    y_norm = torch.empty_like(x_norm_h, dtype=torch.float16, device=device)

    grid_norm = (triton.cdiv(R, BLOCK_M1),)
    rowwise_layernorm_kernel[grid_norm](
        x_norm, y_norm,
        weights["norm.weight"],     # γ
        weights["norm.bias"],       # β
        R, dim, eps,
        BLOCK_M=BLOCK_M1, BLOCK_D=BLOCK_D1,
        num_warps=4,
    )
    del x_norm  # free fp32 version

    # ------------------------------------------------------------------
    # 2️⃣ fused projection + gate (produces left, right, out_gate)
    # ------------------------------------------------------------------
    # fuse the five linear maps into a single weight matrix (transposed)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # [5·H, dim]

    w_fused_T = w_fused.t().contiguous()            # [dim, 5·H]

    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_fp16 = mask.to(torch.float16).contiguous().view(R)   # [R]
        mask_ptr = mask_fp16

    BLOCK_R, BLOCK_K = _choose_proj_block_sizes(R, dim, H)

    grid_proj = (triton.cdiv(R, BLOCK_R),)
    fused_proj_gate_kernel[grid_proj](
        y_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        w_fused_T,
        R,
        dim,
        H,
        OUT_FEAT,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    del y_norm, w_fused_T, w_fused, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ triangular multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape for batched matmul: [B, N, N, H] → [B·H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 4️⃣ second LayerNorm + multiplication by out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_gate_flat = out_gate.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2, BLOCK_D2 = _choose_block_sizes_rows(R, H)
    grid_norm2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid_norm2](
        out_flat,
        out_gate_flat,
        out_fused,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out_flat, out_gate_flat, out

    # ------------------------------------------------------------------
    # 5️⃣ final projection back to `dim` (fp16) → cast to fp32
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
