"""
custom_kernel
-------------

Highly‑optimized forward implementation of the **outgoing** triangular
multiplicative update (TriMul) used in AlphaFold‑3 and related protein‑folding
models.

Algorithm
~~~~~~~~~
1️⃣  Row‑wise LayerNorm on the input tensor (B·N·N × C) – performed with
    PyTorch’s native `layer_norm` (fast fused kernel on H100).

2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single GEMM:
        X_norm @ W_fusedᵀ → (R, 5·H)   where R = B·N·N, H = hidden_dim.
    The fused weight matrix `W_fused` is built from the model’s linear weights.

3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * Splits the fused projection into `left`, `right` and `out_gate`.
    * Applies an optional pairwise mask (broadcasted) to `left`/`right`.
    * Evaluates three sigmoid gates (fp32 → fp16 for stability).

4️⃣  Batched half‑precision GEMM (`torch.bmm`) over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
    This leverages cuBLAS Tensor‑Core kernels.

5️⃣  Second LayerNorm + element‑wise multiplication by `out_gate`.
    Implemented with PyTorch’s `layer_norm` (fp16) followed by a simple
    element‑wise product.

6️⃣  Final linear projection back to the original channel dimension `C`
    (half‑precision GEMM) and cast the result to float32.

Only step 3 is written in Triton, satisfying the “must use a kernel” rule,
while the remaining heavy operations use highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flattened pairwise entry (row), compute:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    All arithmetic in fp16; the sigmoid is evaluated in fp32 for numerical
    stability.  The layout of `proj_ptr` is
        [left, right, left_gate, right_gate, out_gate] each of size H.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # Load (or create) per‑row mask value
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # broadcast on H
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Process the hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)            # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets for the five fused slices
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw_ptr     = proj_ptr + base
        right_raw_ptr    = proj_ptr + base + H
        left_gate_ptr    = proj_ptr + base + 2 * H
        right_gate_ptr   = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        left_raw      = tl.load(left_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(right_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(left_gate_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(right_gate_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(out_gate_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Sigmoid in fp32, cast back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # Store results
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


def _grid_split(rows: int, block_r: int = 128) -> Tuple[int]:
    """Simple helper to compute launch grid for the split/fuse kernel."""
    return (triton.cdiv(rows, block_r),)


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all linear / layer‑norm parameters
                        (float32 tensors, bias=False for Linear layers)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣  unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32 input → FP16 normalized)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)                                 # [R, dim] float32
    x_norm_fp32 = F.layer_norm(x_flat,
                               normalized_shape=(dim,),
                               weight=weights["norm.weight"],
                               bias=weights["norm.bias"],
                               eps=eps)                     # still float32
    x_norm = x_norm_fp32.to(torch.float16)                  # [R, dim] fp16
    del x_flat, x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection -> (R, 5·H)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                    # [5·H, dim]
    proj = F.linear(x_norm, w_fused)                       # [R, 5·H] fp16
    del w_fused, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split into left / right / out_gate, apply mask & sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask shape [B, N, N] → flatten to [R]; values are 0/1
        mask_half = mask.to(torch.float16).contiguous().view(R)
        HAS_MASK = 1
        mask_ptr = mask_half

    grid = _grid_split(R, block_r=128)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=128,
        BLOCK_H=32,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Tri‑multiplicative update (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into batch dimension
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)        # [B·H, N, N]
    right_mat = right.view(B * H, N, N)       # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm + element‑wise multiply with out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                 # [R, H] fp16
    out_norm = F.layer_norm(out_flat,
                            normalized_shape=(H,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=eps)                     # fp16
    out_fused = (out_norm * out_gate).to(torch.float16)    # [R, H] fp16
    del out_flat, out_norm, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣  Final projection back to original dimension `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(out_fused,
                          weights["to_out.weight"].to(torch.float16))   # [R, dim] fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)                # fp32

    return final
