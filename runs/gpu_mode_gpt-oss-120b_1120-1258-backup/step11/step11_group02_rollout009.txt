def layernorm_gate_linear_kernel(
    x_ptr,            # const [rows, H] (fp16)
    gate_ptr,         # const [rows, H] (fp16)
    out_ptr,          # mutable [rows, D] (fp16)
    w_norm_ptr,       # const [H] (fp32)   gamma
    b_norm_ptr,       # const [H] (fp32)   beta
    w_out_ptr,        # const [D, H] (fp16)
    rows, H, D,
    eps,
    BLOCK_M: tl.constexpr,  # rows per block
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,  # output dim per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute mean/var of x across H:
    # similar to layernorm kernel but we could compute during loop chunk-wise.
    # We'll do a two-pass: first compute sum, sumsq.
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        # load chunk of x (raw)
        offs_h = h_off*BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        x_chunk = tl.load(x_ptr + offs_m[:, None]*H + offs_h[None,:],
                         mask=mask_m[:, None] & mask_h[None,:],
                         other=0.0).to(tl.float32)
        sum_val += tl.sum(x_chunk, axis=1)
        sum_sq += tl.sum(x_chunk * x_chunk, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # now compute normalized x * gate and accumulate into output
    # allocate accumulator for output dims chunk(s)

    # will combine multiple blocks of output dims across loops over H
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):

        mask_d = offs_d < D

        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        # loop over h again to compute contribution
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):

"""
TriMul (outgoing) – fast Triton+Torch forward pass.
---------------------------------------------------
Algorithm (all tensors on the same CUDA device)

1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) → fp16.
2️⃣ Fuse the five linear maps (left/right proj + three gates) into one
   GEMM:  X_norm @ W_fusedᵀ  →  (R, 5·H)   with  R = B·N·N .
3️⃣ Triton kernel → splits the fused result, applies the optional
   pair‑wise mask, evaluates the three sigmoid gates and writes three
   tensors:
        * left  : (B·H, N·N)   (flattened ‹i,k› matrix)
        * right : (B·H, N·N)
        * out_gate : (R, H)   (kept in the original row layout)
   The split kernel writes directly into the flattened left/right
   buffers, avoiding the expensive view + permute copy required by the
   reference implementation.
4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out_mat = left @ rightᵀ      (torch.bmm, Tensor‑Core)
   out_mat : (B·H, N, N)  → flatten → (R, H) for the next step.
5️⃣ Row‑wise LayerNorm + element‑wise multiplication with `out_gate`
   (single Triton kernel).  Result → (R, H) fp16.
6️⃣ Final linear projection back to `dim`:
        y = (normed·out_gate) @ W_outᵀ    (torch.nn.functional.linear)
   Reshape to (B, N, N, dim) and cast back to float32.

All heavy element‑wise work is realised in Triton kernels, while the
dense matrix multiplications (steps 2, 4, 6) are delegated to cuBLAS
(Tensor‑Core) via ``torch.nn.functional.linear`` / ``torch.bmm``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   *[rows, dim]   (float32)
    y_ptr,               # mutable *[rows, dim]   (float16)
    w_ptr,               # const   *[dim]        (float32)   γ
    b_ptr,               # const   *[dim]        (float32)   β
    rows,                # tl.int32
    dim,                 # tl.int32
    eps,                 # tl.float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalised values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2 – split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]       (float16)
    left_ptr,          # mutable *[B*H, N*N]      (float16)
    right_ptr,         # mutable *[B*H, N*N]      (float16)
    out_gate_ptr,      # mutable *[rows, H]       (float16)
    mask_ptr,          # const   *[rows]          (float16)  (ignored if HAS_MASK=0)
    rows,              # tl.int32   = B*N*N
    H,                 # tl.int32   = hidden_dim
    N,                 # tl.int32   = seq_len
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For every (b,i,k) → flattened row `r` and each hidden index `h`:
        left  = (proj_left  * mask[r]) * sigmoid(proj_left_gate)
        right = (proj_right * mask[r]) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    Results are written directly into the flattened left/right buffers of
    shape (B·H, N·N) so that a later batched GEMM can be performed without
    any view/permute copy.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                                           # total positions per matrix

    # b = offs_r // NN   ;  pos = offs_r % NN
    b = offs_r // NN
    pos = offs_r % NN

    # load scalar mask (0/1) if present
    if HAS_MASK:
        mask_val_raw = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val_raw[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load the five projected blocks --------------------------------
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # (R, BH)

        left_raw = tl.load(proj_ptr + base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)
        right_raw = tl.load(proj_ptr + base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # ---- sigmoid (stable fp32) ----------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates -----------------------------------------
        left_val  = left_raw  * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate

        # ---- destination offsets for left/right ----------------------------
        # batch index = b * H + h
        batch_idx = b[:, None] * H + offs_h[None, :]                # [BLOCK_R, BLOCK_H]
        dst_offset = batch_idx * NN + pos[:, None]                  # flatten position

        # store left/right
        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ---- out_gate (kept in original row layout) -----------------------
        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 3 – row‑wise LayerNorm + element‑wise multiplication with `gate`
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)   (out_gate)
    y_ptr,            # mutable *[rows, dim]   (float16)
    w_ptr,            # const   *[dim]        (float32)   γ
    b_ptr,            # const   *[dim]        (float32)   β
    rows,             # tl.int32
    dim,              # tl.int32
    eps,              # tl.float32
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm on `x` (fp16) followed by element‑wise
    multiplication with `gate` (also fp16).  The result is stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean & variance (fp32) -----------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β, multiply with gate -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_val - mean[:, None]) * rstd[:, None]           # fp32

        # γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)      # fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)      # fp32
        norm = norm * w[None, :] + b[None, :]                    # fp32

        # gate value (fp16 → fp32)
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

        out = norm * gate_val                                    # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

        out_gate,
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out_mat

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape (R, dim)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
