"""
TriMul (outgoing) – fast Triton+Torch forward pass.
---------------------------------------------------
Algorithm (all tensors live on the same CUDA device)

1️⃣ Row‑wise LayerNorm on the input 4‑D tensor (B·N·N × dim)
   – computes μ,σ² in fp32, writes normalized fp16 values.

2️⃣ Fuse the five linear maps (left/right projections + three gates)
   into a single GEMM:
        X_norm @ W_fusedᵀ  →  (R, 5·H)    where R = B·N·N.
   The result stays in fp16.

3️⃣ Triton kernel splits the fused matrix, applies the optional
   pair‑wise mask, evaluates the three sigmoid gates and writes
   directly into the flattened left/right buffers (shape (B·H, N·N))
   and the out‑gate matrix (shape (R, H)).  No view/permute copies.

4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out_mat = left @ rightᵀ      (torch.bmm, Tensor‑Core)

5️⃣ Row‑wise LayerNorm on `out_mat` (dim = H) + element‑wise
   multiplication with `out_gate` (single Triton kernel).

6️⃣ Final linear projection back to the original channel dimension:
        y = out_fused @ W_outᵀ       (torch.nn.functional.linear)

The heavy element‑wise work is fully vectorised in Triton while the
dense matrix multiplications are delegated to cuBLAS (Tensor‑Core).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   *[rows, dim]   (float32)
    y_ptr,            # mutable *[rows, dim]   (float16)
    w_ptr,            # const   *[dim]        (float32) γ
    b_ptr,            # const   *[dim]        (float32) β
    rows,             # tl.int32
    dim,              # tl.int32
    eps,              # tl.float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- compute mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    d_iter = tl.cdiv(dim, BLOCK_D)
    for d_off in range(d_iter):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # (BLOCK_M, BLOCK_D)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalised values (γ·(x‑μ)·rstd + β) ----------
    for d_off in range(d_iter):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β fp32

        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2 – split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]       (float16)
    left_ptr,          # mutable *[B*H, N*N]      (float16)
    right_ptr,         # mutable *[B*H, N*N]      (float16)
    out_gate_ptr,      # mutable *[rows, H]       (float16)
    mask_ptr,          # const   *[rows]          (float16)  (ignored if HAS_MASK=0)
    rows,              # tl.int32   = B*N*N
    H,                 # tl.int32   = hidden_dim
    N,                 # tl.int32   = seq_len
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flattened row r = (b,i,k) and each hidden index h:
        left      = (proj_left      * mask) * sigmoid(proj_left_gate)
        right     = (proj_right     * mask) * sigmoid(proj_right_gate)
        out_gate  = sigmoid(proj_out_gate)

    Results are written directly into the flattened left/right buffers
    of shape (B*H, N*N) so that a later batched GEMM can be performed
    without any view/permute copy.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                                           # positions per matrix

    # (b,i,k) → (b, pos) where pos = i*N + k
    b = offs_r // NN
    pos = offs_r % NN

    # load scalar mask (1.0 if not present)
    if HAS_MASK:
        mask_val_raw = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val_raw[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate hidden dimension in chunks
    h_iter = tl.cdiv(H, BLOCK_H)
    for h_off in range(h_iter):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ----- load the five projected blocks (left, right, 3 gates) -----
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # (R, BH)

        left_raw      = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_raw     = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # ----- sigmoid (stable fp32) -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----
        left_val  = left_raw  * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate

        # ----- destination offsets for left/right (flattened B*H × N*N) -----
        # row index in left/right buffers = b*H + h
        batch_idx = b[:, None] * H + offs_h[None, :]                 # (R, BH)
        dst_offset = batch_idx * NN + pos[:, None]                  # (R, BH)

        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ----- write out_gate (kept in original row layout) -----
        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 3 – row‑wise LayerNorm + element‑wise multiplication with gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)   (out_gate)
    y_ptr,            # mutable *[rows, dim]   (float16)
    w_ptr,            # const   *[dim]        (float32) γ
    b_ptr,            # const   *[dim]        (float32) β
    rows,             # tl.int32
    dim,              # tl.int32
    eps,              # tl.float32
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    d_iter = tl.cdiv(dim, BLOCK_D)
    for d_off in range(d_iter):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalise, apply γ/β, multiply with gate ----------
    for d_off in range(d_iter):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_val - mean[:, None]) * rstd[:, None]          # fp32

        # γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)     # fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)     # fp32
        norm = norm * w[None, :] + b[None, :]                   # fp32

        # gate value (fp16 → fp32)
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

        out = norm * gate_val                                   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 “outgoing”).
    All heavy element‑wise work is realised in Triton while the three
    dense matrix multiplications are performed by cuBLAS (Tensor‑Core).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]          # input / output channel dimension
    hidden_dim = cfg["hidden_dim"]   # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total rows for row‑wise ops
    H = hidden_dim
    NN = N * N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 1024   # rows per program (larger => fewer program launches)
    BLOCK_D = 128    # dim chunk size (aligns with fp16 SIMD)
    grid1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid1](
        x_flat,
        x_norm,
        w["norm.weight"],          # γ  (float32)
        w["norm.bias"],            # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear: 5 projections (left, right, gates) → (R, 5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)           # fp16, shape (R, 5·H)
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates – write directly to flattened buffers
    # --------------------------------------------------------------
    left_flat  = torch.empty(B * H, NN, dtype=torch.float16, device=device)   # (B·H, N·N)
    right_flat = torch.empty_like(left_flat)

    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)          # (R, H)

    # mask handling
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy
    else:
        has_mask = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)                  # flatten
        mask_ptr = mask_fp16

    BLOCK_R = 1024   # rows per program (flattened (b,i,k))
    BLOCK_H = 32    # hidden‑dim chunk
    grid2 = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid2](
        proj,
        left_flat,
        right_flat,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # reshape for the batched GEMM
    left  = left_flat.view(B * H, N, N)       # (B·H, N, N)
    right = right_flat.view(B * H, N, N)      # (B·H, N, N)

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # --------------------------------------------------------------
    # reshape to (R, H) row layout required by the kernel
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().view(R, H)
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid3 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid3](
        out_flat,
        out_gate,
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out_mat

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, (R, dim)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
