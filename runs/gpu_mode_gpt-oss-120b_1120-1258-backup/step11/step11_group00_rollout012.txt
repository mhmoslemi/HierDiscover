import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Fusion kernel for projection, gating and mask application.
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[rows, H] (fp16)
    mask_ptr,          # const   *[rows] (fp16) - ignored if HAS_MASK==0
    rows: tl.int32,
    H: tl.int32,
    B: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)       # [BLOCK_R]
    mask_r = offs_r < rows

    # decode (b,i,k) from flat row index
    NN = N * N
    stride_h = NN          # distance between hidden slices
    stride_b = H * NN      # distance between batches

    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # optional mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast to hidden
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Process hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # address within fused projection matrix
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # compute flat offset for [B, H, N, N] storage
        base0 = b * stride_b + i * N + k               # [BLOCK_R]

        left_dst = left_ptr + base0[:, None] + offs_h[None, :] * stride_h
        right_dst = right_ptr + base0[:, None] + offs_h[None, :] * stride_h

        tl.store(left_dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # store out_gate as flat rows
        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# Fused LayerNorm + gate multiplication (second LN)
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim] (fp16)
    gate_ptr,       # const   *[rows, dim] (fp16)
    y_ptr,          # mutable *[rows, dim] (fp16)
    weight_ptr,     # const   *[dim] (fp32) γ
    bias_ptr,       # const   *[dim] (fp32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # Compute per‑row mean & variance in fp32
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Normalize, apply LN weight/bias and multiply by gate
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    Input:

        mask   : [B, N, N]   torch.bool or torch.float (may be None)
        weights: dict of tensors (see docstring above)
        cfg    : dict with keys "dim", "hidden_dim", "nomask"

            weights["right_proj.weight"],

            weights["right_gate.weight"],
            weights["out_gate.weight"],

    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)            # [B*H, N, N]
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    BLOCK_D = 32

        weights["to_out_norm.bias"],
        rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`  (fp16 GEMM) → float32
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # [rows, dim] fp16

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
The implementation follows the exact forward described in the reference
PyTorch module but replaces the costly full‑tensor transposes with a
few carefully‑written Triton kernels:

1️⃣  First LayerNorm is performed with cuDNN (`torch.nn.functional.layer_norm`)
    because it is already highly tuned and eliminates a custom kernel launch.

2️⃣  A *single* fused linear projection creates the five tensors needed for
    left/right projections, their sigmoid gates and the out‑gate (all in
    fp16).  A Triton kernel (`split_fuse_gate_kernel`) writes the left/right
    tensors directly into the `[B, H, N, N]` layout, applies the optional mask,
    and stores the out‑gate in a flat `[rows, H]` buffer.

3️⃣  The true tri‑multiplicative update is a batched GEMM:
        out = left @ rightᵀ
    (`torch.bmm` on fp16 tensors automatically uses Tensor‑Cores).

4️⃣  The second LayerNorm + element‑wise multiplication with the out‑gate is
    fused into another Triton kernel (`layernorm_gate_kernel`).

5️⃣  Finally a linear projection brings the hidden dimension back to `dim`
    (fp16 GEMM) and the result is cast to float32.

The kernels are deliberately simple, use fp32 accumulators for numerical
stability, and keep memory traffic low by avoiding intermediate transposes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: split projection, apply mask & sigmoid gates, write left/right
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]     (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    B: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)       # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # Decode flat index (b, i, k) from the row number.
    # --------------------------------------------------------------
    NN = N * N                           # N²
    stride_h = NN                        # distance between hidden slices
    stride_b = H * NN                    # distance between batches

    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # --------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    # Process the hidden dimension in tiles of size BLOCK_H.
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Base address for the fused projection matrix (5*H columns)
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # Sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw  .to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ------------------------------------------------------------------
        # Store left/right into the [B, H, N, N] layout
        # ------------------------------------------------------------------
        # flat offset of the first hidden slice of the current row:
        #   offset = b * stride_b + i * N + k
        base0 = b * stride_b + i * N + k                     # [BLOCK_R]

        left_dst  = left_ptr  + base0[:, None] + offs_h[None, :] * stride_h
        right_dst = right_ptr + base0[:, None] + offs_h[None, :] * stride_h

        tl.store(left_dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ------------------------------------------------------------------
        # Store out_gate as a flat [rows, H] matrix
        # ------------------------------------------------------------------
        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 2: second LayerNorm + multiplication with out_gate (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim] (fp16)
    gate_ptr,       # const   *[rows, dim] (fp16)
    y_ptr,          # mutable *[rows, dim] (fp16)
    weight_ptr,     # const   *[dim]      (fp32) γ
    bias_ptr,       # const   *[dim]      (fp32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # Compute mean & variance (fp32 accumulator)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, apply LN weight/bias, and multiply by gate.
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals       # fp32
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])
