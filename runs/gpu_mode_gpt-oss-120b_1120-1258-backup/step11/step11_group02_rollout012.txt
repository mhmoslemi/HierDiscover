"""
TriMul (outgoing) – fast Triton + torch forward pass
===================================================

The outgoing TriMul block performs the following steps on a 4‑D tensor
`x ∈ ℝ^{B×N×N×C}` (C = dim) :

1️⃣ Row‑wise LayerNorm over the last dimension (C) → fp16.
2️⃣ Fuse the five linear maps (left/right projection + three gates) into a
   single GEMM:   X_norm @ W_fusedᵀ   →   (R, 5·H)   with R = B·N·N.
3️⃣ Triton kernel splits the fused result, applies the optional pair‑wise mask,
   evaluates the three sigmoid gates and writes three buffers:
        * left   : (B·H, N·N)  – flattened ‹i,k› matrix
        * right  : (B·H, N·N)  – flattened ‹j,k› matrix
        * out_gate : (R, H)    – kept in the original row layout
4️⃣ Batched half‑precision GEMM (Tensor‑Core) computes the multiplicative
   update:  out_mat = left @ rightᵀ   → (B·H, N, N).
5️⃣ Row‑wise LayerNorm on `out_mat` + element‑wise multiplication with
   `out_gate` (single Triton kernel).
6️⃣ Final linear projection back to `dim` (cuBLAS GEMM) and reshape to
   (B, N, N, dim).

All heavy element‑wise work is realised in Triton kernels; the three dense
matrix multiplications are delegated to cuBLAS (Tensor‑Core).  The
implementation is fully vectorised, allocates the minimum number of
temporary tensors and works for both masked / un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # *[rows, dim]   (float32)
    y_ptr,               # *[rows, dim]   (float16)
    w_ptr,               # *[dim]         (float32)   γ
    b_ptr,               # *[dim]         (float32)   β
    rows,                # tl.int32
    dim,                 # tl.int32
    eps,                 # tl.float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # -------------------------------------------------
    # mean & variance (fp32)
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------
    # normalise, apply γ/β and store as fp16
    # -------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2 – split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # *[rows, 5*H]   (float16)
    left_ptr,          # *[B*H, N*N]   (float16)
    right_ptr,         # *[B*H, N*N]   (float16)
    out_gate_ptr,      # *[rows, H]    (float16)
    mask_ptr,          # *[rows]       (float16)  (ignored if HAS_MASK=0)
    rows,              # tl.int32   = B*N*N
    H,                 # tl.int32   = hidden_dim
    N,                 # tl.int32   = seq_len
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flattened row r = (b, i, k) and each hidden index h:
        left   = (proj_left   * mask) * sigmoid(proj_left_gate)
        right  = (proj_right  * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    The left/right values are written to buffers of shape (B*H, N*N)
    such that a later batched GEMM can be performed without any view /
    permute copies.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N

    # decode batch index and flattened (i,k) position
    b   = offs_r // NN               # [BLOCK_R]
    pos = offs_r % NN                # [BLOCK_R]  = i*N + k

    # -----------------------------------------------------------------
    # optional scalar mask (0/1)
    # -----------------------------------------------------------------
    if HAS_MASK:
        mask_val_raw = tl.load(mask_ptr + offs_r,
                               mask=mask_r,
                               other=1.0)               # [BLOCK_R]
        mask_val = mask_val_raw[:, None]                 # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # -----------------------------------------------------------------
    # iterate over hidden dimension in chunks of BLOCK_H
    # -----------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # base offset for the current (r, h) pair inside the fused matrix
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # (R, BH)

        # ----- load the five projection blocks -------------------------
        left_raw      = tl.load(proj_ptr + base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_raw     = tl.load(proj_ptr + base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_gate_raw= tl.load(proj_ptr + base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(proj_ptr + base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)

        # ----- sigmoid (stable fp32) ---------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----------------------------------
        left_val  = left_raw  * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate

        # ----- write to flattened buffers -----------------------------
        # row index inside the (B*H, N*N) buffers
        batch_idx = b[:, None] * H + offs_h[None, :]   # [BLOCK_R, BLOCK_H]
        dst_offset = batch_idx * NN + pos[:, None]    # flatten position

        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ----- out_gate stays in row‑major layout --------------------
        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

        norm = (x_val - mean[:, None]) * rstd[:, None]           # fp32

        # γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)      # fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)      # fp32
        norm = norm * w[None, :] + b[None, :]                    # fp32

        # gate (fp16 → fp32)
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

        out = norm * gate_val                                    # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 "outgoing" variant).
    Arguments
    ---------
    data : tuple  (input, mask, weights, config)
        input   – Tensor[bs, seq_len, seq_len, dim] (float32)
        mask    – Tensor[bs, seq_len, seq_len]      (float32 or bool) or None
        weights – dict of model parameters (all tensors are on the same CUDA device)
        config  – dict with keys:
                  * dim         – input / output channel dimension
                  * hidden_dim  – inner hidden dimension H
                  * nomask      – bool, whether the mask should be ignored

    Returns
    -------
    Tensor[bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]          # C
    hidden_dim = cfg["hidden_dim"]   # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total rows for row‑wise ops
    H = hidden_dim
    NN = N * N                # positions per flattened matrix

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    # reshape to (R, dim) – contiguous memory layout required by the kernel
    x_flat = x.view(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 1024      # more rows per program → fewer kernel launches
    BLOCK_D = 128       # dim chunk (power‑of‑2, fits fp16 SIMD)
    grid1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid1](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ  (float32)
        weights["norm.bias"],            # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection: five linear maps → (R, 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # shape (5·H, dim)

    proj = F.linear(x_norm, fused_w)               # fp16, shape (R, 5·H)
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    left_flat  = torch.empty(B * H, NN, dtype=torch.float16, device=device)
    right_flat = torch.empty_like(left_flat)
    out_gate   = torch.empty(R, H, dtype=torch.float16, device=device)

    # mask handling ----------------------------------------------------
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # flatten mask to (R,) and cast to fp16
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 512          # rows per program
    BLOCK_H = 32           # hidden‑dim chunk
    grid2 = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid2](
        proj,
        left_flat,
        right_flat,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # reshape buffers for the batched GEMM
    left  = left_flat.view(B * H, N, N)      # (B·H, N, N)
    right = right_flat.view(B * H, N, N)     # (B·H, N, N)

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dim)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    # re‑arrange to row‑major layout (R, H) required by the kernel
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().view(R, H)
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid3 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid3](
        out_flat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out_mat

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # fp16, shape (R, dim)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
