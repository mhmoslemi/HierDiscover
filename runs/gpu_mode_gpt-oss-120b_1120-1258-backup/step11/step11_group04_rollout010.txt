import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# split_fuse_gate kernel same as above but maybe adjust block sizes.
@triton.jit
def split_fuse_gate_kernel(...):
    ...

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Algorithm:
    1. Row-wise LayerNorm on input (torch.nn.functional.layer_norm) in fp32,
       then cast to fp16.
    2. Fuse linear projections and gates into one fp16 GEMM (W_fused).
    3. Triton kernel splits projection into left, right, out_gate tensors
       and applies mask and sigmoid gates, writing directly in (B, H, N, N)
       layout.
    4. Batched matrix multiplication left @ right^T via torch.bmm.
    5. Apply second LayerNorm (torch.nn.functional.layer_norm) across hidden dim,
       multiply elementwise with out_gate.
    6. Final linear projection back to dim via fp16 GEMM.
    Returns float32 tensor of shape (B,N,N,dim).
    """
    # Unpack
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. First layernorm (fp32) then cast to fp16
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps).to(torch.float16)

    # 2. Fuse linear: combine all five weight matrices (dim x hidden_dim each)
    # weight shapes: [hidden_dim, dim] each (since Linear without bias). torch.nn.Linear weight shape (out_features, in_features).
    # The provided weights dict likely have shape [hidden_dim, dim] for left_proj, etc.
    # Considering we need to multiply: x_norm (R, dim) @ left_proj_weight.T -> (R, hidden_dim)
    # In F.linear the weight shape is (out_features, in_features). We'll follow earlier code.
    fused_w = torch.cat([weights["left_proj.weight"],
                         weights["right_proj.weight"],
                         weights["left_gate.weight"],
                         weights["right_gate.weight"],
                         weights["out_gate.weight"]], dim=0).to(torch.float16)  # [5*H, dim]

    proj = F.linear(x_norm, fused_w)  # shape [R, 5*H]

    # 3. Split, mask, sigmoid gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32  # if H up to 384, this covers
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # 4. Batched GEMM of (B*H, N, N)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]
    out = out_mat.view(B, H, N, N)

    # 5. Second LN across hidden dim (H) using torch.layer_norm
    # permute to (B,N,N,H)
    out_perm = out.permute(0, 2, 3, 1).contiguous()
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    # Apply LayerNorm with weight and bias
    out_ln = F.layer_norm(out_perm, (H,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16), eps=eps)
    out_ln_gate = out_ln * gate_perm

    # 6. Final projection back to dim
    final_half = F.linear(out_ln_gate, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k from row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask if needed
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # fp16
        mask_val = mask_val[:, None]   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N  # stride between consecutive hidden slices

    # base offset for output tensors (b*H*N*N + i*N + k)
    base_offset = b * (H * N * N) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # offsets into proj matrix for the five blocks
        proj_row = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + proj_row,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_row + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate  = tl.load(proj_ptr + proj_row + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_gate = tl.load(proj_ptr + proj_row + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_row + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offset = base_offset + h * hidden_stride
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradients).

    Algorithm:
    1. LayerNorm on input (fp32) -> cast to fp16.
    2. Fuse left/right projections and three gates into a single GEMM.
    3. Triton kernel splits the fused projection, applies optional pairwise mask and
       sigmoid gates, writing directly in (B, H, N, N) layout.
    4. Batched GEMM over hidden dimension (left @ rightᵀ) via torch.bmm.
    5. LayerNorm across hidden dimension, element‑wise multiply by out_gate.
    6. Final linear projection back to original dim.

    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    import math
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # -------------------------------------------------
    # 1️⃣ Input LayerNorm (fp32) → fp16
    # -------------------------------------------------
    norm_w_f16 = weights["norm.weight"].to(torch.float16)
    norm_b_f16 = weights["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(x, (dim,), weight=norm_w_f16, bias=norm_b_f16, eps=eps)
    x_norm = x_norm.to(torch.float16)   # [B, N, N, dim]

    # -------------------------------------------------
    # 2️⃣ Fuse linear projections + gates
    # -------------------------------------------------
    # Concatenate five weight matrices (each shape [hidden_dim, dim])
    fused_w = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)  # [5*H, dim]

    proj = F.linear(x_norm, fused_w)          # [B, N, N, 5*H] fp16
    proj_flat = proj.view(R, 5 * H)

    # -------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (Triton)
    # -------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # Release memory no longer needed
    del proj, proj_flat, mask_ptr

    # -------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension
    # -------------------------------------------------
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]
    out = out_mat.view(B, H, N, N)   # [B, H, N, N]

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm (across H) + out_gate multiplication
    # -------------------------------------------------
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    to_out_norm_w_f16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_f16 = weights["to_out_norm.bias"].to(torch.float16)

    out_ln = F.layer_norm(out_perm, (H,), weight=to_out_norm_w_f16,
                          bias=to_out_norm_b_f16, eps=eps)
    out_ln_gate = out_ln * gate_perm   # element‑wise fp16

    # -------------------------------------------------
    # 6️⃣ Final projection back to dim
    # -------------------------------------------------
    final_half = F.linear(out_ln_gate, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
Outgoing TriMul (TriMul‑out) – fast GPU forward.

Algorithm (all‑fp16 where possible)
----------------------------------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B,N,N,dim) using
    :func:`torch.nn.functional.layer_norm`.  The result is cast to fp16.
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single GEMM:  X_norm @ W_fusedᵀ  →  (B,N,N,5·H)  (H = hidden_dim).
3️⃣  Triton kernel ``split_fuse_gate_kernel``:
    * decodes the flat row index (b,i,k)  →  positions (i,k) of the pair‑wise
      mask,
    * loads the five ‑block slices of the fused projection,
    * applies the optional mask,
    * evaluates sigmoid gates in fp32 (for numerical stability),
    * writes ``left``, ``right`` and ``out_gate`` directly in the layout
      (B, H, N, N) required by the upcoming batched GEMM.
4️⃣  Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
    performed with :func:`torch.bmm` (cuBLAS Tensor‑Core kernels).
5️⃣  LayerNorm across the hidden dimension (H) using
    :func:`torch.nn.functional.layer_norm`, then element‑wise multiplication
    with ``out_gate``.
6️⃣  Final half‑precision linear projection back to ``dim`` with another GEMM.

Only step 3 is a custom Triton kernel – the rest uses highly‑optimised
torch/cuBLAS kernels.  The implementation works for all test‑case shapes
and for both masked / un‑masked inputs.
"""

# Triton kernel: split fused projection, apply mask & sigmoid gates,
#                write directly in (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,       # total rows = B * N * N
    H: tl.int32,       # hidden dimension
    N: tl.int32,       # sequence length
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flat row r = b·N·N + i·N + k (0 ≤ r < R) the kernel extracts the
    five ‑blocks belonging to that position from ``proj_ptr``:

        left_raw, right_raw, left_gate, right_gate, out_gate_raw

    applies the (optional) pair‑wise mask and sigmoid gates, and stores the
    three results in the (B, H, N, N) layout required by the batched GEMM.
    All arithmetic is fp16 except the sigmoid (computed in fp32 for stability).
    """

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # --------------------------------------------------------------
    # Decode (b, i, k) from the flat row index
    # --------------------------------------------------------------

    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # --------------------------------------------------------------

        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:

    hidden_stride = N * N

    # = b·(H·N·N) + i·N + k   →  shape [BLOCK_R]
    base_offset = b * (H * N * N) + i * N + k

    # --------------------------------------------------------------
    # Loop over hidden dimension in tiles of size BLOCK_H
    # --------------------------------------------------------------

        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets into the fused projection (five blocks of size H)
        proj_row = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + proj_row,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_row + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate  = tl.load(proj_ptr + proj_row + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_gate = tl.load(proj_ptr + proj_row + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_row + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ----------------------------------------------------------
        # Sigmoid in fp32 → cast back to fp16
        # ----------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----------------------------------------------------------
        # Apply mask and gates
        # ----------------------------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Destination offsets for the three output tensors
        #   offset = base_offset + h * hidden_stride
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        # Store
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping of all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N               # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    norm_w_fp16 = w["norm.weight"].to(torch.float16)
    norm_b_fp16 = w["norm.bias"].to(torch.float16)

    # torch.nn.functional.layer_norm works on the last dimension (dim)
    x_norm = F.layer_norm(x,
                          (dim,),
                          weight=norm_w_fp16,
                          bias=norm_b_fp16,
                          eps=eps)               # still float32
    x_norm = x_norm.to(torch.float16)                # [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],   # [H, dim]
            w["right_proj.weight"],  # [H, dim]
            w["left_gate.weight"],   # [H, dim]
            w["right_gate.weight"],  # [H, dim]
            w["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                               # [5*H, dim]

    # Project (B,N,N,dim) → (B,N,N,5*H) in fp16
    proj = F.linear(x_norm, fused_w)                 # fp16
    proj_flat = proj.view(R, 5 * H)                  # flatten to [R, 5*H]

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (Triton)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128    # rows per program (tuned for H100)
    BLOCK_H = 32     # hidden‑dim tile (covers hidden_dims up to 384)
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, proj_flat, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (B*H independent matmuls)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # [B*H, N, N]
    right_mat = right.view(B * H, N, N)              # [B*H, N, N]

    # out_mat[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]
    out = out_mat.view(B, H, N, N)                     # [B, H, N, N]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm across hidden dimension + out_gate multiplication
    # --------------------------------------------------------------
    # Rearrange to (B, N, N, H) so that H is the last (contiguous) dimension
    out_perm  = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H]

    to_out_norm_w_fp16 = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_fp16 = w["to_out_norm.bias"].to(torch.float16)

    # LayerNorm across the hidden dimension H (fp16)
    out_ln = F.layer_norm(out_perm,
                          (H,),
                          weight=to_out_norm_w_fp16,
                          bias=to_out_norm_b_fp16,
                          eps=eps)

    # Element‑wise gate
    out_ln_gate = out_ln * gate_perm                     # fp16

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to original `dim`
    # --------------------------------------------------------------
    final_half = F.linear(out_ln_gate,
                          w["to_out.weight"].to(torch.float16))  # [B,N,N,dim] fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
