"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The operator works on a 4‑D tensor `x` of shape `[B, N, N, C]`.  The
computation consists of

1️⃣  Row‑wise LayerNorm over the last dimension (`C` → fp16).
2️⃣  A *single* GEMM that produces the five linear projections required
    for ``left``, ``right`` and the three gates (``left_gate``,
    ``right_gate``, ``out_gate``).  This is an ``(R, C) @ (5·H, C).T``
    where ``R = B·N·N``.
3️⃣  A Triton kernel that
    * splits the fused projection,
    * multiplies by the optional pair‑wise mask,
    * evaluates the three sigmoid gates (sigmoid in fp32 for stability) and
      writes the three tensors ``left``, ``right`` and ``out_gate``.
4️⃣  A batched FP16 GEMM (cuBLAS/Tensor‑Cores) that computes the
    outer product ``out[b,i,j,d] = Σ_k left[b,i,k,d]·right[b,j,k,d]``.
    The layout after reshaping is ``[B·H, N, N]``.
5️⃣  **Fused** second LayerNorm, gating by ``out_gate`` and the final
    linear projection back to ``dim`` are performed by a single Triton
    kernel.  This eliminates an extra memory round‑trip and saves a
    kernel launch.

All heavy matrix multiplications use cuBLAS FP16 Tensor‑Core kernels,
while the three custom kernels (layer‑norm, split+gate, fused LN+proj)
are fully vectorised and compute‑bound.

The implementation works for the full test‑matrix (``B∈{1,2}``,
``N∈{128,256,512,1024}`` etc.) and respects the ``nomask`` flag.
"""

import torch.nn.functional as F

# ----------------------------------------------------------------------
# Kernel 1 – Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim] (float32)
    y_ptr,          # mutable *[rows, dim] (float16)
    weight_ptr,     # const   *[dim]      (float32) γ
    bias_ptr,       # const   *[dim]      (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values (fp16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Kernel 2 – Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]      (float16) – optional
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are in FP16; sigmoid evaluated in FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ----- mask value per row (broadcast over H) -----
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)          # [BLOCK_R]
        mask_val = mask_val[:, None]                         # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ----- iterate over hidden dimension -----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw    = tl.load(proj_ptr + base,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_raw   = tl.load(proj_ptr + base + H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        left_g_raw  = tl.load(proj_ptr + base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        out_g_raw   = tl.load(proj_ptr + base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # sigmoid in fp32
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        tl.store(left_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 3 – Fused second LayerNorm + out_gate * linear proj
# ----------------------------------------------------------------------
@triton.jit
def final_norm_linear_kernel(
    x_ptr,               # const   *[rows, H]      (float16) – out of left×right
    gate_ptr,            # const   *[rows, H]      (float16) – out_gate
    w_out_ptr,           # const   *[D, H]        (float16) – to_out.weight
    ln_weight_ptr,       # const   *[H]           (float32) – to_out_norm.weight
    ln_bias_ptr,         # const   *[H]           (float32) – to_out_norm.bias
    out_ptr,             # mutable *[rows, D]      (float16) – final output
    rows: tl.int32,
    H: tl.int32,
    D: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_N: tl.constexpr,   # output‑dim tiles
    BLOCK_K: tl.constexpr,   # hidden‑dim tiles
):
    pid_m = tl.program_id(0)          # row block
    pid_n = tl.program_id(1)          # output‑dim block

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < rows
    mask_n = offs_n < D

    # ------------ 1️⃣ mean & variance over H ------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)    # [BLOCK_K]
        mask_k = offs_k < H

        # load x (fp16 → fp32)
        x = tl.load(x_ptr + offs_m[:, None] * H + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # [M,K]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------ 2️⃣ accumulate output ------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)    # [BLOCK_K]
        mask_k = offs_k < H

        # load x and gate (fp16 → fp32)
        x = tl.load(x_ptr + offs_m[:, None] * H + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # [M,K]
        g = tl.load(gate_ptr + offs_m[:, None] * H + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # [M,K]

        # LN weight & bias (γ,β) for this hidden slice
        gamma = tl.load(ln_weight_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta  = tl.load(ln_bias_ptr   + offs_k, mask=mask_k, other=0.0)  # [K]

        # Normalize, apply LN affine, then gate
        # shape: [M,K]
        norm = ((x - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        gated = norm * g

        # Load a tile of the output weight: shape (K,N)
        # w_out[d, h] laid out row‑major → offset = d*H + h
        w = tl.load(w_out_ptr + offs_k[:, None] + offs_n[None, :] * H,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)          # [K,N]

        # Accumulate GEMM contribution: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(gated, w)

    # ------------ 3️⃣ write result ------------
    out_fp16 = acc.to(tl.float16)
    tl.store(out_ptr + offs_m[:, None] * D + offs_n[None, :],
             out_fp16,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------- unpack ---------------------
    x, mask, w, cfg = data
    dim      = cfg["dim"]
    hidden   = cfg["hidden_dim"]
    eps      = 1e-5
    device   = x.device
    B, N, _, _ = x.shape
    R = B * N * N                     # rows = B·N·N

    # ---------------- 1️⃣ first LayerNorm ----------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],   # γ  (float32)
        w["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ---------------- 2️⃣ fused projection ----------------
    # concatenate the five weight matrices once
    w_fused = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # shape (5·H, dim)

    proj = F.linear(x_norm, w_fused)                 # (R, 5·H), fp16
    del w_fused, x_norm

    # ---------------- 3️⃣ split, mask, sigmoid gates ----------------
    left      = torch.empty(R, hidden, dtype=torch.float16, device=device)
    right     = torch.empty(R, hidden, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, hidden, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_ptr = mask.to(torch.float16).reshape(R)   # flatten mask

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ---------------- 4️⃣ batched GEMM (left × right) ----------------
    # reshape to [B, H, N, N] → [B·H, N, N]
    left = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()
    left_mat  = left.reshape(B * hidden, N, N)        # [B·H, N, N]
    right_mat = right.reshape(B * hidden, N, N)       # [B·H, N, N]

    out_mat = torch.matmul(left_mat, right_mat.transpose(-1, -2))   # fp16
    del left, right, left_mat, right_mat

    # flatten to (R, hidden) – this will be the input to the final kernel
    out_flat = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out_flat = out_flat.view(R, hidden)                # [R, H]

    # ---------------- 5️⃣ fused LN + out_gate * final linear ----------------
    final_out = torch.empty(R, dim, dtype=torch.float16, device=device)

    # kernel tiling
    BLOCK_M = 128   # rows per program
    BLOCK_N = 64    # output‑dim tiles
    BLOCK_K = 32    # hidden‑dim tiles

    grid = (triton.cdiv(R, BLOCK_M), triton.cdiv(dim, BLOCK_N))
    final_norm_linear_kernel[grid](
        out_flat,
        out_gate.view(R, hidden),
        w["to_out.weight"].to(torch.float16).contiguous(),   # [dim, hidden]
        w["to_out_norm.weight"],   # γ  (float32)
        w["to_out_norm.bias"],     # β  (float32)
        final_out,
        R,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    del out_flat, out_gate, out_mat

    # ---------------- 6️⃣ reshape & cast back to fp32 ----------------
    out = final_out.view(B, N, N, dim).to(torch.float32)
    return out
