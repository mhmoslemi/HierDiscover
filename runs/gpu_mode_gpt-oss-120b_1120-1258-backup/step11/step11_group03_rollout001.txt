"""
custom_kernel – highly‑optimized forward of the *outgoing* TriMul operator
(AlphaFold‑3, Chai, Protenix).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (fp32 → fp16).
2️⃣ Fuse the five linear maps (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single GEMM
   (R×dim  ×  dim×5H → R×5H,   R = B·N·N, H = hidden_dim).
3️⃣ Triton kernel `split_fuse_gate`:
   * extracts the five slices from the fused GEMM,
   * applies the optional pair‑wise mask,
   * computes σ‑gates in fp32,
   * writes the three tensors **left**, **right**, **out_gate**
     directly in layout `[B, H, N, N]` (no extra transposes).
4️⃣ Batched matrix multiplication (cuBLAS Tensor‑cores):
       out[b·H+i, j] = Σₖ left[b·H, i, k] · right[b·H, j, k]ᵀ
   implemented with a single `torch.bmm`.
5️⃣ Row‑wise LayerNorm on the result + element‑wise × out_gate
   (fused in `layernorm_mul_kernel`).
6️⃣ Final linear projection (fp16 → fp32) back to `dim`.

All kernels are written for Triton 3.3.1 and tuned for an NVIDIA H100.
Only the forward pass is required; gradients are not computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- accumulate mean / variance in fp32 ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (fp16) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask, σ‑gates → left/right/out_gate
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B, H, N, N] (float16)
    right_ptr,         # mutable *[B, H, N, N] (float16)
    out_gate_ptr,      # mutable *[B, H, N, N] (float16)
    mask_ptr,          # const   *[rows]      (float16) – ignored if HAS_MASK=0
    rows: tl.int32,    # R = B·N·N
    N: tl.int32,       # sequence length
    H: tl.int32,       # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each (b,i,k) row (R = B·N·N):
        left_raw  = proj[:, :H]
        right_raw = proj[:, H:2H]
        left_gate_raw  = proj[:, 2H:3H]
        right_gate_raw = proj[:, 3H:4H]
        out_gate_raw   = proj[:, 4H:5H]

        left  = left_raw  * mask * σ(left_gate_raw)
        right = right_raw * mask * σ(right_gate_raw)
        out_gate = σ(out_gate_raw)

    Results are stored as [B, H, N, N] (no transposes).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- decode linear row → (b,i,k) ----
    seq_sq = N * N
    b = offs_r // seq_sq
    rem = offs_r - b * seq_sq
    i = rem // N
    k = rem - i * N

    # ---- optional pair‑wise mask (scalar per row) ----
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- iterate over hidden dim in chunks ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, Hc]

        left_raw    = tl.load(proj_ptr + base,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_raw   = tl.load(proj_ptr + base + H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # ----- sigmoid in fp32 for stability -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # ----- destination offsets for [B, H, N, N] -----
        # flat = ((b * H + h) * N + i) * N + k
        dst_off = ((b[:, None] * H + offs_h[None, :]) * N + i[:, None]) * N + k[:, None]   # [R,Hc]

        tl.store(left_ptr      + dst_off, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, out_gate,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise LayerNorm + element‑wise × gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)
    y_ptr,            # mutable *[rows, dim]   (float16)
    weight_ptr,       # const   *[dim]        (float32) γ
    bias_ptr,         # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean / variance (fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, γ/β, gate, store (fp16) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]        # fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32
        norm = norm * w[None, :] + b[None, :]                     # fp32

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                          # fp32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Helper: pick moderately sized blocks (tuned for H100)
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    """Return (BLOCK_M, BLOCK_D) suited for the current shape."""
    # rows are usually much larger than dim (dim ≤ 384). Use a larger
    # row block for good occupancy, a moderate column block for good
    # memory coalescing.
    if rows >= 2**16:
        BLOCK_M = 512
    elif rows >= 2**14:
        BLOCK_M = 256
    else:
        BLOCK_M = 128

    BLOCK_D = 64 if dim >= 64 else dim
    return BLOCK_M, BLOCK_D

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear layers)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M, BLOCK_D = _choose_block_sizes(R, dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R, dim, eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ fused projection (5 linear maps → single GEMM)
    # ------------------------------------------------------------------
    # Build fused weight matrix with shape [5*H, dim] (half‑precision)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # Linear: (R, dim) @ (dim, 5*H).T → (R, 5*H)
    proj = F.linear(x_norm, w_fused)        # (R, 5*H) fp16
    del w_fused, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ split, apply mask, compute σ‑gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if nomask:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)   # dummy
        HAS_MASK = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)                    # [R] in {0,1}
        HAS_MASK = 1

    # Tuned block sizes: 128 rows × 32 hidden per iteration gives good
    # memory bandwidth on H100.
    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        N,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,           # more warps improve occupancy for hidden=128
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ batched GEMM: left @ rightᵀ   (B*H, N, N) × (B*H, N, N)ᵀ
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # (B·H, N, N)
    right_mat = right.view(B * H, N, N)               # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Back to [B, H, N, N]
    out = out_mat.view(B, H, N, N)
    del left_mat, right_mat, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ second LayerNorm + element‑wise × out_gate (fused)
    # ------------------------------------------------------------------
    # Rearrange to [R, H] where each row corresponds to an (b,i,j) triple
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # (B,N,N,H) → (R,H)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)

    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2, BLOCK_D2 = _choose_block_sizes(R, H)
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out_flat, out_gate_flat, out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H] fp16
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
