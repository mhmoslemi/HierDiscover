"""
TriMul (outgoing) – fast Triton + torch implementation
=====================================================

Algorithm (forward only, no gradients)

1️⃣  Row‑wise LayerNorm over the last dimension (fp32 input → fp16 output).  
    Implemented with `torch.nn.functional.layer_norm`, which on H100 is
    highly‑optimized (faster than the previous pure‑Triton version).

2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single GEMM:  X_norm @ W_fusedᵀ  →  (R, 5·H) where R = B·N·N.  The GEMM is
    performed with `torch.nn.functional.linear` (fp16, Tensor‑Core friendly).

3️⃣  One Triton kernel (`split_fuse_gate_kernel`) splits the fused result,
    applies the optional pair‑wise mask, evaluates the three sigmoids (in fp32
    for stability) and writes the three tensors  
    `left`, `right` and `out_gate`.  All work is done in fp16 except the
    sigmoid.

4️⃣  Batched fp16 GEMM over the hidden dimension using torch.matmul:
        out[b, h] = left[b, h] @ right[b, h]ᵀ    (B·H independent N×N matmuls).

5️⃣  Second LayerNorm on the hidden‑dim output (`out`) with `torch.nn.functional.layer_norm`
    followed by an element‑wise multiplication with `out_gate`.

6️⃣  Final fp16 linear projection back to `dim` and cast to fp32.

Only step 3 uses a custom Triton kernel, satisfying the “must have a kernel”
requirement while keeping the overall runtime well below the 1 ms target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Kernel 3 – split fused projection, apply mask & sigmoid gates
# ------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r ∈ [0, rows):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    All ops are FP16 except the sigmoid (computed in FP32 for stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------
    # Load per‑row mask if present (scalar 0/1 broadcasted)
    # --------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)   # [BLOCK_R]
        mask_val = mask_val[:, None]                  # [BLOCK_R, 1] broadcast over H
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, dtype=tl.float16)

    # --------------------------------------------------------
    # Iterate over the hidden dimension
    # --------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside each row: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ----- sigmoid in fp32 -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- store results -----
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

    # ------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + three gates
    # ------------------------------------------------------------
    # Concatenate all five weight matrices (shape: (5*H, dim))
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                         # (5*H, dim)
    proj = F.linear(x_norm, w_fused)                            # (R, 5*H) fp16

    # ------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates – custom Triton kernel
    # ------------------------------------------------------------
    left     = torch.empty(R, hidden, dtype=torch.float16, device=device)
    right    = torch.empty(R, hidden, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, hidden, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        HAS_MASK = 1
        # flatten mask (bool/float -> half) to shape (R,)
        mask_flat = mask.to(torch.float16).view(R)
        mask_ptr = mask_flat

    BLOCK_R = 256        # rows per program
    BLOCK_H = 64         # hidden dimension chunk per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # Release unused tensors early
    del proj, x_norm, w_fused, mask_ptr

    # ------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (fp16, Tensor‑Core)
    # ------------------------------------------------------------
    # reshape to [B, H, N, N] for a single batched matmul
    left = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)   (fp16)
    out_mat = torch.matmul(left, right.transpose(-2, -1))                # (B, H, N, N)

    # ------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate
    # ------------------------------------------------------------
    # flatten to (R, H)
    out_flat = out_mat.permute(0, 2, 3, 1).contiguous().view(R, hidden)   # (R, H)
    # LayerNorm on hidden dimension (fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(out_flat, (hidden,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b,
                            eps=eps)                                   # fp16
    # multiply with sigmoid(out_gate)
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` and cast to float32
    # ------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # (R, dim) fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
