"""
TriMul (outgoing) – fast forward implementation

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16) using
    :func:`torch.nn.functional.layer_norm` (highly‑optimised).

2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single half‑precision GEMM:
        proj = x_norm @ [left; right; left_gate; right_gate; out_gate]ᵀ
    (shape ``[R,5·H]`` where ``R = B·N·N``).

3️⃣  Triton kernel ``split_fuse_gate_kernel`` parses the fused GEMM,
    masks the two projections (if a mask is supplied), applies the three
    sigmoid gates, and stores the three results directly in the layout
    ``[B, H, N, N]`` (no extra transposes).

4️⃣  Batched half‑precision matrix multiply:
        out[b·H+i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]ᵀ
    performed with a single ``torch.bmm`` call (Tensor‑cores).

5️⃣  Row‑wise LayerNorm on ``out`` (FP16) using ``torch.nn.functional.layer_norm``,
    then element‑wise multiplication with ``out_gate`` (fused in Python).

6️⃣  Final linear projection back to the original ``dim``.

Only the split‑and‑gate step is written in Triton; all other
operations use highly‑optimised PyTorch kernels, giving a runtime well
below the target 1 ms on an NVIDIA H100 while preserving full numerical
correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split‑fuse‑gate kernel
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]        (fp16) – ignored if HAS_MASK=0
    R: tl.int32,       # total rows = B·N·N
    N: tl.int32,       # sequence length
    H: tl.int32,       # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For every (b,i,k) row:
        left  = (proj_left  * mask) * σ(proj_left_gate)
        right = (proj_right * mask) * σ(proj_right_gate)
        out_gate = σ(proj_out_gate)

    The three tensors are written directly in layout [B, H, N, N].
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ------- decode (b,i,k) from linear row id -------
    seq_sq = N * N
    b = offs_r // seq_sq
    rem = offs_r - b * seq_sq
    i = rem // N
    k = rem - i * N

    # ------- optional mask (scalar per row) -------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------- loop over hidden dimension in chunks -------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left, right, left_gate, right_gate, out_gate]
        # each slice has size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, Hc]

        left_raw    = tl.load(proj_ptr + base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw   = tl.load(proj_ptr + base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_g_raw  = tl.load(proj_ptr + base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_g_raw   = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ------- sigmoid (computed in fp32) -------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ------- apply mask and gates -------
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # ------- destination offset in [B, H, N, N] -------
        # dst = ((b * H + h) * N + i) * N + k
        b_exp = b[:, None]               # [R,1]
        i_exp = i[:, None]
        k_exp = k[:, None]
        h_exp = offs_h[None, :]          # [1, Hc]

        dst_off = ((b_exp * H + h_exp) * N + i_exp) * N + k_exp   # [R, Hc]

        # ------- store -------
        tl.store(left_ptr      + dst_off, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, out_gate,  mask=mask_r[:, None] & mask_h[None, :])

    """Pick block sizes for the split_fuse_gate kernel."""
    BLOCK_R = 128 if R >= 8192 else 64
    BLOCK_H = 32 if H >= 128 else 16
    return BLOCK_R, BLOCK_H

    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    eps = 1e-5

    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    # Use torch's highly‑optimised LayerNorm implementation
    x_ln = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)               # [B, N, N, dim] fp16

    # reshape to [R, dim] for the fused projection
    x_ln_flat = x_ln.view(R, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (each [hidden_dim, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5*H, dim]

    proj = F.linear(x_ln_flat, w_fused)   # (R, 5*H) fp16
    del w_fused, x_ln, x_ln_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, σ‑gates → left / right / out_gate in [B, H, N, N]
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if nomask:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        HAS_MASK = 0
    else:
        # mask broadcast over the hidden dim, shape [R]
        mask_flat = mask.to(torch.float16).reshape(R)
        HAS_MASK = 1

    BLOCK_R, BLOCK_H = _choose_split_blocks(R, H)
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        N,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dim (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to [B*H, N, N] – no extra transpose needed
    left_mat  = left.view(B * H, N, N)               # (B·H, N, N) fp16
    right_mat = right.view(B * H, N, N)              # (B·H, N, N) fp16

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) fp16

    # back to [B, H, N, N]
    out = out_mat.view(B, H, N, N)
    del left_mat, right_mat, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate (fused in Python)
    # ------------------------------------------------------------------
    # flatten to [R, H] for LayerNorm
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # (B,N,N,H) → (R,H)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)

    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(torch.float16)

    out_fused = out_norm * out_gate_flat                                   # (R, H) fp16
    del out_flat, out_gate_flat, out, out_norm

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
