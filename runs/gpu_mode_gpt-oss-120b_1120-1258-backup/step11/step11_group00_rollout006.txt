"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
The implementation follows exactly the reference PyTorch module but
removes all intermediate transposes and reduces the number of memory
passes:

1️⃣  Row‑wise LayerNorm on the 4‑D input (FP32 → FP16) – Triton.
2️⃣  One fused Linear that projects the normalized tensor into the
    five required heads (left, right, three gates).  This is a single
    cuBLAS GEMM on FP16.
3️⃣  A Triton kernel that
    • splats the optional pair‑wise mask,
    • applies the three sigmoid gates,
    • writes `left` and `right` directly into the layout `[B, H, N, N]`,
    • stores `out_gate` in a flat `[rows, H]` buffer.
4️⃣  Batched GEMM (`torch.bmm`) computes the true tri‑multiplicative
    update: `out = left @ rightᵀ`  (shape `[B·H, N, N]`).  cuBLAS/Tensor‑Core
    handles this efficiently.
5️⃣  A fused LayerNorm + element‑wise multiplication with `out_gate`
    (Triton).  The gate is already in half‑precision, so we keep the
    whole operation in FP16.
6️⃣  Final Linear projection back to `dim` (FP16 GEMM) and cast to FP32.

Only three Triton kernels are used (two LayerNorms and the split‑gate
kernel).  All heavy matrix multiplications stay on cuBLAS, which on an
H100 gives the best possible throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim] (float16)
    weight_ptr,     # const *[dim]         (float32) γ
    bias_ptr,       # const *[dim]         (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # dim processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------- mean / var (float32 accumulator) -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # ------------------- write normalized values -------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates, write `left`/`right`
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B, H, N, N] (float16)
    right_ptr,         # mutable *[B, H, N, N] (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const *[rows] (float16) – ignored if HAS_MASK=0
    rows: tl.int32,    # rows = B * N * N
    H: tl.int32,
    B: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                     # N²
    stride_h = NN                  # distance between two hidden slices
    stride_b = H * NN              # distance between two batches

    # ---- decode (b,i,k) from the flat row index ------------------------
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # ---- optional mask -------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- process hidden dimension in tiles ------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # base address for the fused projection of shape [rows, 5*H]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [R,H]

        # ----- load raw slices (no bias – already baked into linear) -----
        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ----- sigmoid (fp32 for numeric stability) --------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- store left/right into [B, H, N, N] layout -----------------
        # linear offset for the first hidden slice of the (b,i,k) row:
        #   offset = b * stride_b + i * N + k
        base0 = b * stride_b + i * N + k                     # [BLOCK_R]

        left_dst_ptr  = left_ptr  + base0[:, None] + offs_h[None, :] * stride_h
        right_dst_ptr = right_ptr + base0[:, None] + offs_h[None, :] * stride_h

        tl.store(left_dst_ptr,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ----- store out_gate (still flat [rows, H]) --------------------
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst_ptr,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const *[rows, dim]   (float16)
    gate_ptr,       # const *[rows, dim]   (float16)
    y_ptr,          # mutable *[rows, dim] (float16)
    weight_ptr,     # const *[dim]         (float32) γ
    bias_ptr,       # const *[dim]         (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------- mean / var (float32) -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- normalization + gate multiplication ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals         # fp32
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape                # x: [B, N, N, dim]
    rows = B * N * N                     # total “pair‑wise” rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(rows, dim).contiguous()                     # [rows, dim] FP32
    x_norm = torch.empty(rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # dim per iteration (must be a multiple of 8 for coalescing)
    grid_ln1 = (triton.cdiv(rows, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection producing the five heads (FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each: [H, dim]) → [5*H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # Linear:   out = x_norm @ w_fused.T   → shape [rows, 5*H] (FP16)
    proj = F.linear(x_norm, w_fused)        # [rows, 5*H] FP16
    del w_fused, x_norm                     # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid gates, write `left/right`/`out_gate`
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(rows, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask: [B, N, N] → flatten to [rows] and cast to half
        mask_fp16 = mask.to(torch.float16).reshape(rows)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 256        # rows per program (tuned for H100)
    BLOCK_H = 64         # hidden chunks per program
    grid_split = (triton.cdiv(rows, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        rows,
        H,
        B,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B*H, N, N] without copy
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out_mat = left @ rightᵀ   (Tensor‑Core GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # Back to [B, N, N, H] (needed for the second LN)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(rows, H).contiguous()          # [rows, H] FP16
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2 = 128
    BLOCK_D2 = 64
    grid_ln2 = (triton.cdiv(rows, BLOCK_M2),)
    layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        rows,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (FP16 GEMM) + cast to FP32
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [rows, dim]  FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
