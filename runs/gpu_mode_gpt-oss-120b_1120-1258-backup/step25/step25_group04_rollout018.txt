"""
Outgoing TriMul (AlphaFold3) – forward‑only implementation.
================================================================
Algorithm (high‑level)
---------------------
1️⃣  Input LayerNorm (FP32 → FP16) using PyTorch’s highly‑optimised
    `torch.nn.functional.layer_norm`.  This avoids a custom reduction kernel
    and gives the best possible mean/variance computation on H100.

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM:
        proj = X_norm @ W_fused.T   where   W_fused ∈ ℝ^{5·H×C}
    The GEMM runs in FP16 on Tensor‑Cores.

3️⃣  A Triton kernel `split_fuse_gate_kernel`:
       * decodes the flat index  r = b·N·N + i·N + k
       * extracts the five blocks (left, right, left_gate, right_gate, out_gate)
         from *proj*,
       * applies the optional pair mask only to *left* and *right*,
       * applies the three sigmoid gates,
       * writes the three tensors in layout **[B, H, N, N]** (contiguous).

4️⃣  Batched matrix multiplication (cuBLAS) computes the core contraction
    ``out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k]``.
    This is performed as a single `torch.bmm` on the view
    ``[B·H, N, N]`` → ``[B·H, N, N]``.

5️⃣  Final processing – all done with PyTorch kernels:
       * bring the hidden dimension to the last axis → shape (B,N,N,H),
       * `torch.nn.functional.layer_norm` over the hidden dimension H
         using *to_out_norm* parameters,
       * element‑wise multiplication with the *out_gate* (already sigmoid‑ed),
       * final linear projection back to *dim* with `torch.nn.functional.linear`.

Only the gate‑splitting kernel is written in Triton, satisfying the
requirement that “part of the operations” be implemented in a kernel while
keeping the whole forward pass fast on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply optional mask and
# three sigmoid gates, and store the three tensors in [B, H, N, N] layout.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  fp16 fused projection
    left_ptr,      # [B, H, N, N] fp16
    right_ptr,     # [B, H, N, N] fp16
    out_gate_ptr,  # [B, H, N, N] fp16
    mask_ptr,      # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset for one hidden‑plane (stride = N·N)
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    # iterate over hidden dimension tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # index into the fused projection (layout: [R, 5·H])
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R,BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in fp32, cast back to fp16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        # out_gate is NOT masked, as in the reference implementation
        og    = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R,BLOCK_H]

        tl.store(left_ptr + dst, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,    mask=mask_r[:, None] & mask_h[None, :])

            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5·H, dim)
    proj = F.linear(x_norm, fused_w)            # (R, 5·H)  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128          # rows per program (tuned for H100)
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core contraction
    # ------------------------------------------------------------------
    # reshape for bmm: [B·H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                           # (B, H, N, N) FP16

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm + out_gate + projection (all in PyTorch)
    # ------------------------------------------------------------------
    # bring hidden dimension to the last axis → (B, N, N, H)
    out_perm = out.permute(0, 2, 3, 1).contiguous()
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()

    # LayerNorm over hidden dimension H
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # (B, N, N, H) FP16

    # Apply the output gate (sigmoid already applied in the kernel)
    out_norm = out_norm * out_gate_perm                # (B, N, N, H) FP16

    # Final linear projection back to `dim`
    # Flatten the first three axes -> (R, H)
    out_final = F.linear(
        out_norm.view(R, H),
        w["to_out.weight"].to(torch.float16),  # (dim, H)
    )  # (R, dim) FP16

    # Reshape to the original layout and cast to FP32
    out_final = out_final.view(B, N, N, dim).to(torch.float32)

    return out_final
```
