@triton.jit
def proj_gate_kernel(
    x_ptr,               # [R, dim] fp16
    left_ptr,            # [B, H, N, N] fp16
    right_ptr,           # [B, H, N, N] fp16
    out_gate_ptr,        # [B, H, N, N] fp16
    mask_ptr,            # [R] fp16
    # weights for each projection
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    # Decode b,i,k from offs_r (R = B*N*N)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Compute base offset for hidden dim start (h=0)
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k      # shape [BLOCK_R]

    # Load mask if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]  # broadcast to hidden dimension
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Loop over hidden dimension blocks

"""
Outgoing TriMul (AlphaFold3) – forward‑only, highly‑fused Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pair tensor.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single Triton kernel that directly computes the three tensors `left`, `right`
   and `out_gate` from the normalized input, applying the optional mask and the
   sigmoid gates (no intermediate large matrix is materialised).
3️⃣ Batched GEMM (`torch.bmm`) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   producing a tensor `[B, H, N, N]` (FP16).
4️⃣ Row‑wise statistics (mean & rsqrt‑variance) of the contracted tensor.
5️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the final
   projection (`to_out`) into a single Triton kernel, writing the final
   `[B, N, N, dim]` tensor in FP32.

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: compute mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_tiles = tl.cdiv(dim, BLOCK_D)
    for d_tile in range(num_d_tiles):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_tile in range(num_d_tiles):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Fuse five linear maps and gates (no intermediate projection)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,                      # [R, dim]               FP16 (norm‑ed input)
    left_ptr, right_ptr, out_gate_ptr,   # [B, H, N, N]   FP16
    mask_ptr,                   # [R] (optional)         FP16
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,   # each [H, dim] FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode flat index r = b·N·N + i·N + k ----
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start for h=0

    # ---- mask (broadcast over hidden) ----
    if HAS_MASK:
        mask_row = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0)                 # [BLOCK_R] FP16
    else:
        mask_row = tl.full([BLOCK_R], 1.0, dtype=tl.float16)

    # ---- loop over hidden dimension in blocks ----
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        h_start = h_tile * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)        # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators in FP32
        acc_left       = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_left_gate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_out_gate   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ---- reduction over the feature dimension ----
        num_d_tiles = tl.cdiv(dim, BLOCK_D)
        for d_tile in range(num_d_tiles):
            offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # load a slice of the input row
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)            # [BLOCK_R, BLOCK_D]

            # load weight slices for the five linear maps
            w_left   = tl.load(left_w_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :],
                               other=0.0).to(tl.float32)
            w_right  = tl.load(right_w_ptr  + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :],
                               other=0.0).to(tl.float32)
            w_lgate  = tl.load(left_gate_w_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :],
                               other=0.0).to(tl.float32)
            w_rgate  = tl.load(right_gate_w_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :],
                               other=0.0).to(tl.float32)
            w_outgate = tl.load(out_gate_w_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                               mask=mask_h[:, None] & mask_d[None, :],
                               other=0.0).to(tl.float32)

            # dot‑product (sum over D)
            acc_left       += tl.dot(x, w_left)
            acc_right      += tl.dot(x, w_right)
            acc_left_gate  += tl.dot(x, w_lgate)
            acc_right_gate += tl.dot(x, w_rgate)
            acc_out_gate   += tl.dot(x, w_outgate)

        # ---- sigmoid gates (computed in FP32, cast back) ----
        left_gate   = (1.0 / (1.0 + tl.exp(-acc_left_gate))).to(tl.float16)
        right_gate  = (1.0 / (1.0 + tl.exp(-acc_right_gate))).to(tl.float16)
        out_gate    = (1.0 / (1.0 + tl.exp(-acc_out_gate))).to(tl.float16)

        # ---- apply mask & gates, cast to FP16 ----
        left_val  = acc_left.to(tl.float16) * mask_row[:, None] * left_gate
        right_val = acc_right.to(tl.float16) * mask_row[:, None] * right_gate

        # ---- store results ----
        # off‑set for hidden planes in the output tensors
        hidden_offset = offs_h * hidden_stride                     # [BLOCK_H]
        dst_offset = base_offset[:, None] + hidden_offset[None, :] # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)

# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr, gate_ptr,
    mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics (mean & rstd)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]             # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                # [M, K]
        term = norm * vals_gate                                     # [M, K]
        term_fp16 = term.to(tl.float16)                             # FP16 for dot

        # Load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                     # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask_input, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projections & gates (no intermediate large tensor)
    # ------------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # optional mask handling
    if (not nomask) and (mask_input is not None):
        mask_flat = mask_input.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # bring weight tensors to FP16 and make them contiguous
    left_w      = w["left_proj.weight"].to(torch.float16).contiguous()
    right_w     = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w= w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w  = w["out_gate.weight"].to(torch.float16).contiguous()

    BLOCK_R = 64
    BLOCK_H = 64
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        left_w,
        right_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, mask_ptr, left_w, right_w, left_gate_w, right_gate_w, out_gate_w

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # view as [B·H, N, N] for bmm
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Row‑wise statistics (mean & rsqrt‑variance) of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128          # H ≤ 128 in all test configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # gamma FP32
        w["to_out_norm.bias"],     # beta  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
