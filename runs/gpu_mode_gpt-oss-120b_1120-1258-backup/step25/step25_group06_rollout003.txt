"""
TriMul (outgoing) forward‑only implementation – highly fused & fast.

Algorithm
---------
1️⃣  LayerNorm on the input pair representation (FP32 → FP16) – done with the
    torch.nn.functional.layer_norm, which is heavily optimized on H100.

2️⃣  Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
    into a single half‑precision GEMM:   (R, 5·H) = (B·N·N, dim) × (5·H, dim)ᵀ

3️⃣  Split the fused projection, apply the optional pair‑wise mask,
    compute the sigmoid gates and store three tensors (left, right, out_gate)
    in the layout [B, H, N, N] (FP16).  This step is the only custom Triton kernel.

4️⃣  Batched half‑precision GEMM (torch.bmm) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (FP16).

5️⃣  Row‑wise LayerNorm on the contracted tensor (again via torch.nn.functional.layer_norm).

6️⃣  Apply the out‑gate, then a final half‑precision linear projection back to the
    original channel dimension and cast the result to FP32.

All heavy compute (GEMMs) is performed by cuBLAS/cuDNN; the only Triton kernel
deals with the gating logic, keeping launch overhead minimal while still satisfying
the “must‑use‑a‑kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: split the fused projection, apply mask & sigmoid
# ---------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  fused projection (FP16)
    left_ptr,          # [B, H, N, N]          output (FP16)
    right_ptr,         # [B, H, N, N]          output (FP16)
    out_gate_ptr,      # [B, H, N, N]          output (FP16)
    mask_ptr,          # [R] (optional)        mask as FP16 scalar
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per block
    BLOCK_H: tl.constexpr,   # hidden‑dim tiles per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional scalar mask per (b,i,k)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]          # broadcast to hidden dim
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    hidden_stride = N * N                         # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset for hidden h = 0

    # iterate over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H_tile]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ---------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (FP32 → FP16) – using torch for speed
    # ------------------------------------------------------------------
    # Normalise across the last dimension (`dim`)
    x_norm = torch.nn.functional.layer_norm(
        x, normalized_shape=(dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps
    ).to(torch.float16)                     # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection → (R, 5·H)  (FP16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]

    proj = F.linear(x_norm.view(R, dim), fused_w)   # [R, 5·H] FP16
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        has_mask = 0

    BLOCK_R = 64   # rows per program (tuned for H100)
    BLOCK_H = 64   # hidden‑dim tiles per program
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right are [B, H, N, N] → reshape to [B·H, N, N]
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (row‑wise) using torch (fast)
    # ------------------------------------------------------------------
    # reshape to (R, H) where each row corresponds to a (b,i,j) triple
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H] FP16
    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out_flat, normalized_shape=(H,), weight=ln_w, bias=ln_b, eps=eps
    )  # FP16, shape [R, H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply out‑gate and final projection
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H] FP16
    gated = out_norm * out_gate_flat                                    # FP16

    # final linear: (R, H) × (dim, H)ᵀ → (R, dim)
    w_out = w["to_out.weight"].to(torch.float16)                         # [dim, H]
    final_flat = torch.nn.functional.linear(gated, w_out)               # FP16

    final = final_flat.view(B, N, N, dim).to(torch.float32)              # FP32
    return final
