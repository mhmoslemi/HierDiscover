"""
TriMul (outgoing) – Triton‑fused forward pass

The operator works on a 4‑D pairwise tensor X ∈ ℝ^{B×N×N×C}
and produces Y ∈ ℝ^{B×N×N×C}.  The implementation follows the
AlphaFold‑3 “outgoing” formulation:

1️⃣  Row‑wise LayerNorm over the last dimension C (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate,
    out_gate) into a single half‑precision GEMM:
        P = X̂ @ W_fusedᵀ      (shape [R, 5·H],  R = B·N·N).
3️⃣  Split the fused projection, apply the optional pairwise mask,
    compute sigmoid gates and store three tensors:
        L, R, G ∈ ℝ^{B·H·N·N}  (FP16).
4️⃣  Batched half‑precision GEMM (cuBLAS):
        O_{b,h,i,j} = Σ_k L_{b,h,i,k} · R_{b,h,j,k}
   giving a tensor O ∈ ℝ^{B·H·N·N}.
5️⃣  Row‑wise mean / variance of O across the hidden dimension H
    (second LayerNorm).
6️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the
    final projection (H → C) into a single Triton kernel that writes
    Y directly (FP32).

All heavy element‑wise work is done in Triton; the only “library”
call is a single `torch.bmm` for the hidden‑dimension contraction.
The final result is returned as a float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, C]  FP32 input (contiguous)
    y_ptr,      # [R, C]  FP16 output
    w_ptr,      # [C]    gamma (FP32)
    b_ptr,      # [C]    beta  (FP32)
    rows,       # R = B·N·N
    dim,        # C
    eps,        # float
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- accumulate sum & sum‑of‑squares --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # read tile [M, D]
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values ---------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split the fused projection, apply optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]        FP16 (only if HAS_MASK)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode flat index r → (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        m_val = m_val[:, None]        # shape [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + k   # offset of hidden = 0

    for off_h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = off_h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw  = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        lgate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        rgate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        ogate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        # sigmoid on FP32, cast back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate_raw.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate_raw.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * lgate
        right = right_raw * m_val * rgate
        og    = ogate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 og,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows,   # R = B·N·N
    H, N,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # hidden = 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out‑gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] γ  (float32)
    ln_b_ptr,       # [H] β  (float32)
    w_out_T_ptr,    # [H, dim] FP16 (transposed projection)
    out_final_ptr,  # [B, N, N, dim] float32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # row dimension
    pid_n = tl.program_id(1)   # output‑dim dimension

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # hidden = 0

    # per‑row statistics (μ, rstd) – FP32
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor and out‑gate (FP16 → FP32)
        out_tile  = tl.load(out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_tile = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # second LayerNorm params
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β
        diff = (out_tile - mu[:, None]) * rstd[:, None]          # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]             # [M, K]

        # multiply with out‑gate
        term = norm * gate_tile                                   # [M, K]
        term_fp16 = term.to(tl.float16)                           # back to fp16 for dot

        # slice of final weight matrix   (K, dim_tile)
        w_tile = tl.load(w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                               # [K, dim_tile]

        # dot (fp16·fp16 → fp32) and accumulate
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

    Args:

            - mask_tensor  : torch.Tensor[B, N, N] (bool/float) or None

    Returns:
        torch.Tensor[B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                       # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32

    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection to 5·H features
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # shape [5·H, dim]
    proj = F.linear(x_norm, fused_w)         # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, gates → left, right, out_gate (FP16)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # mask handling

        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat
        has_mask = 1
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        has_mask = 0

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] and perform (i,k) · (j,k)ᵀ
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------

    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](

        num_warps=4,
    )

    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128

    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)

        final,

        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
