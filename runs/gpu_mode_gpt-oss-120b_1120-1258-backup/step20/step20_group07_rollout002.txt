"""
TriMul (outgoing) – Fast Triton‑fused forward pass
=================================================

This implementation follows the “outgoing” TriMul operator from AlphaFold‑3
but trades the three custom post‑multiplication kernels for highly‑optimized
PyTorch primitives (cuBLAS Tensor‑Core GEMMs and reductions).  Only the
element‑wise heavy work – the initial LayerNorm and the projection/gating
step – is performed in Triton.  All matrix multiplications, reductions and
the final linear projection use Torch’s native kernels, which on an H100
are considerably faster than a hand‑written Triton kernel.

Algorithm
---------
1️⃣ **LayerNorm (FP32 → FP16)** – a row‑wise kernel normalises the flattened
   input `[B·N·N, C]` and writes the result in FP16.

2️⃣ **Fused projection + gating** – a single linear layer computes the five
   linear maps (left/right projections and three gates).  A Triton kernel
   reshapes this tensor to `[B, H, N, N]`, applies the optional pairwise
   mask, evaluates the three sigmoids and stores
   `left, right, out_gate` (all FP16).

3️⃣ **Batched GEMM** – `left @ rightᵀ` is performed with `torch.bmm`,
   yielding `out ∈ FP16` of shape `[B, H, N, N]`.

4️⃣ **Row‑wise statistics** – mean and variance over the hidden dimension
   are obtained with `torch.mean` / `torch.var` (FP32).

5️⃣ **Second LayerNorm + out‑gate** – the result is normalised, affine‑
   transformed with `to_out_norm.{weight,bias}` and multiplied by
   `out_gate`.

6️⃣ **Final linear projection** – the tensor is reshaped to `[B·N·N, H]`
   and multiplied with the transposed output weight (`to_out.weightᵀ`).
   The matmul is executed in FP16; the final result is cast back to
   FP32 and reshaped to `[B, N, N, C]`.

Only two Triton kernels are needed, which drastically reduces launch
overhead while still keeping the element‑wise work fully fused.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm kernel (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # const   *[rows, dim] (float32)
    y_ptr,    # mutable *[rows, dim] (float16)
    w_ptr,    # const   *[dim]      (float32)   γ
    b_ptr,    # const   *[dim]      (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    n_blocks_d = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, n_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                     # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalised values (γ·(x‑μ)·rstd + β) in FP16
    # ------------------------------------------------------------------
    for d_off in range(0, n_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const   *[R, 5*H]   (fp16)   fused projection
    left_ptr,       # mutable *[B, H, N, N] (fp16)
    right_ptr,      # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,   # mutable *[B, H, N, N] (fp16)
    mask_ptr,       # const   *[R]        (fp16)   (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index (b,i,k) → destination layout (B,H,N,N)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask handling (scalar per (b,i,k))
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    # Base offset for hidden plane 0 in the destination tensors
    base_offset = b * (H * N * N) + i * N + k   # [BLOCK_R]

    n_blocks_h = tl.cdiv(H, BLOCK_H)
    for h_off in range(0, n_blocks_h):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in fp32 for stability, cast back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                         # out_gate is not masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with Linear / LayerNorm parameters
                             (all float32, Linear layers have no bias)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total (b,i,k) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32 if dim <= 128 else 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],   # γ (float32)
        w["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]

    # Linear: (R, dim_fp16) @ (dim, 5*H).T -> (R, 5*H)  (FP16)
    proj = F.linear(x_norm, fused_w)        # (R, 5*H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and evaluate gates (layout B,H,N,N)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ) – Tensor‑Core cuBLAS
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # (B·H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                            # (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean / rsqrt‑variance over the hidden dimension
    # ------------------------------------------------------------------
    # Compute in FP32 for numerical stability.
    out_f32 = out.float()
    mu   = out_f32.mean(dim=1)                       # (B, N, N)
    var  = out_f32.var(dim=1, unbiased=False)       # (B, N, N)
    rstd = (var + eps).rsqrt()                       # (B, N, N)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    # Normalise and apply affine parameters (to_out_norm)
    ln_w = w["to_out_norm.weight"]   # (H) float32
    ln_b = w["to_out_norm.bias"]     # (H) float32
    # Broadcast to [B, H, N, N]
    ln_w_b = ln_w.view(1, H, 1, 1)
    ln_b_b = ln_b.view(1, H, 1, 1)

    # (out - mu) * rstd  →  (B, H, N, N)  (FP32)
    mu_b = mu.unsqueeze(1)          # (B,1,N,N)
    rstd_b = rstd.unsqueeze(1)      # (B,1,N,N)
    out_norm = (out_f32 - mu_b) * rstd_b

    # Affine transform + out‑gate (gate is FP16 → cast to FP32)
    out_ln = out_norm * ln_w_b + ln_b_b                     # (B, H, N, N) FP32
    out_gate_f32 = out_gate.float()
    out_mod = out_ln * out_gate_f32                         # (B, H, N, N) FP32

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (hidden → dim)
    # ------------------------------------------------------------------
    # Reshape to (B·N·N, H) and cast to FP16 for Tensor‑Core matmul
    out_mod_fp16 = out_mod.view(B * N * N, H).to(torch.float16)

    # Weight matrix: (H, dim) in FP16 (transposed from Linear weight)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)

    # Matmul in FP16, then cast back to FP32 and reshape
    out_fp16 = torch.matmul(out_mod_fp16, w_out_T)                    # (R, dim) FP16
    out_final = out_fp16.float().view(B, N, N, dim)                   # (B,N,N,dim) FP32

    return out_final
