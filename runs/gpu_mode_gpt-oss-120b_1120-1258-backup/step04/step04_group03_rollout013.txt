@triton.jit
def layernorm_stats_kernel(x_ptr, mean_ptr, rstd_ptr, total_rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < total_rows

   sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

   for d in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim

       ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
       vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
       sum_val += tl.sum(vals, axis=1)
       sum_sq += tl.sum(vals * vals, axis=1)

   # compute mean+variance
   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   rstd = tl.rsqrt(var + eps)

   # store
   tl.store(mean_ptr + offs_m, mean, mask=mask_m)
   tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

"""
High‑performance Triton implementation of the “outgoing” TriMul (AlphaFold‑3) forward
pass.

Algorithm
---------
1️⃣  Compute per‑row mean and 1/√variance (rstd) of the input tensor → two
    vectors `mean_in` / `rstd_in` (float32) with a lightweight Triton kernel.
    The kernel performs a single pass (no second write‑back pass like a
    full LayerNorm).

2️⃣  Fuse the first LayerNorm affine transform (γ,β) together with the five
    linear projections (left/right projections + three gates):
        * pre‑multiply the concatenated weight matrix W∈ℝ^{5·H×C} with the
          LayerNorm gain γ (element‑wise) → W′.
        * pre‑compute the bias term  b_i = Σ_d β[d]·W_i[d]  and the row‑wise
          sums  w_sum_i = Σ_d W′_i[d] .
    The normalized input is never materialised.  Using the statistics from
    (1) we obtain the projected tensors with the closed‑form:
        proj = rstd_in·(X·W′ᵀ – mean_in·w_sum) + b
    where X is the half‑precision copy of the original tensor.
    This yields the five tensors (left_proj, right_proj, left_gate,
    right_gate, out_gate) in a single GEMM call.

3️⃣  Optional mask is applied to the two projection tensors (left/right).

4️⃣  Apply sigmoid to the three gate tensors and element‑wise gate the
    projections.

5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    realised as a batched matrix multiplication on tensors of shape
    (B·H, N, N).

6️⃣  Compute mean/rstd of the result `out` with another Triton kernel
    (identical to (1) but works on half‑precision data).

7️⃣  Fuse the second LayerNorm affine transform (γ₂,β₂) and the output‑gate
    multiplication in a single Triton kernel:
        gated = ((out – mean_out)·rstd_out·γ₂ + β₂) * out_gate
    The kernel writes the final gated tensor directly in half‑precision.

8️⃣  Final linear projection back to the original channel dimension `dim`
    with a regular cuBLAS GEMM, then cast the result to float32.

All heavy reductions are done in Triton; dense GEMMs use Tensor‑Core‑friendly
FP16 kernels from PyTorch.  The implementation works for every supported
batch‑size, sequence‑length and channel configuration.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel 1: per‑row mean & rstd (used for both LayerNorms)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_stats_kernel(
    x_ptr,          # *const   dtype: any (fp32 or fp16)   [total_rows, dim]
    mean_ptr,       # *mut float32  [total_rows]
    rstd_ptr,       # *mut float32  [total_rows]
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of the input (any dtype) and promote to fp32
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# Triton kernel 2: second LayerNorm + output gate (affine + gate)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_affine_gate_kernel(
    out_ptr,            # *const half  [total_rows, dim_h]
    gate_ptr,           # *const half  [total_rows, dim_h] (out_gate)
    gated_ptr,          # *mut   half  [total_rows, dim_h] (result)
    mean_ptr,           # *const float32 [total_rows]
    rstd_ptr,           # *const float32 [total_rows]
    weight_ptr,         # *const float32 [dim_h]   (to_out_norm.weight)
    bias_ptr,           # *const float32 [dim_h]   (to_out_norm.bias)
    total_rows,         # int
    dim_h: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim_h

    # Load raw out (half -> fp32)
    out_vals = tl.load(
        out_ptr + offs_m[:, None] * dim_h + offs_d[None, :],
        mask=mask_m[:, None] & mask_d[None, :],
        other=0.0,
    ).to(tl.float32)

    # Load gate (half -> fp32)
    gate_vals = tl.load(
        gate_ptr + offs_m[:, None] * dim_h + offs_d[None, :],
        mask=mask_m[:, None] & mask_d[None, :],
        other=0.0,
    ).to(tl.float32)

    # Load per‑row statistics
    mean_vals = tl.load(mean_ptr + offs_m, mask=mask_m, other=0.0)[:, None]   # [BLOCK_M,1]
    rstd_vals = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)[:, None]

    # Load affine parameters (broadcast across rows)
    w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # [BLOCK_D]
    b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # [BLOCK_D]

    # Normalisation + affine
    normed = (out_vals - mean_vals) * rstd_vals            # fp32
    affine = normed * w + b                                 # fp32
    gated = affine * gate_vals                              # fp32

    # Store result back in FP16
    tl.store(
        gated_ptr + offs_m[:, None] * dim_h + offs_d[None, :],
        gated.to(tl.float16),
        mask=mask_m[:, None] & mask_d[None, :],
    )


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – returns a float32 tensor of shape
    [bs, seq_len, seq_len, dim].
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, C = input_tensor.shape   # C should equal dim
    assert C == dim, f"dim mismatch: {C} vs {dim}"
    total_rows = B * N * N              # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean & rstd of the *raw* input (first LayerNorm statistics)
    # ------------------------------------------------------------------
    mean_in = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd_in = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_M = 256    # rows per program (tuned)
    BLOCK_D = 64     # columns per tile
    grid_stats = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_stats_kernel[grid_stats](
        input_tensor.contiguous().view(total_rows, dim),
        mean_in,
        rstd_in,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Fuse first LayerNorm (γ,β) with the five linear projections
    # ------------------------------------------------------------------
    # Concatenated weight matrix of the five projections (no bias)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).float()  # (5*H, C)

    # Apply the gain γ of the first LayerNorm (element‑wise per input channel)
    norm_weight = weights["norm.weight"]   # (C,)
    norm_bias   = weights["norm.bias"]     # (C,)

    W_prime = concat_w * norm_weight.unsqueeze(0)          # (5H, C)
    bias_term = (concat_w * norm_bias.unsqueeze(0)).sum(dim=1)  # (5H,)

    # Row‑wise sums of the γ‑scaled weights (needed for the mean subtraction)
    w_sum = W_prime.sum(dim=1)                            # (5H,)

    # Convert input to FP16 for the GEMM
    x_fp16 = input_tensor.view(total_rows, dim).contiguous().to(torch.float16)

    # Perform the GEMM with the γ‑scaled weights
    W_prime_fp16 = W_prime.half()
    proj_fp16 = F.linear(x_fp16, W_prime_fp16)            # (total_rows, 5H)

    # Apply the normalisation scaling and add the bias term
    # All arithmetic in FP32 for numerical stability
    proj_fp32 = (
        rstd_in.unsqueeze(1) * (proj_fp16.float() - mean_in.unsqueeze(1) * w_sum)
        + bias_term
    )
    proj_fp16 = proj_fp32.half()                           # (total_rows, 5H)

    # ------------------------------------------------------------------
    # 3️⃣  Split back to the five tensors
    # ------------------------------------------------------------------
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_fp16, hidden_dim, dim=1
    )  # each (total_rows, H)

    # ------------------------------------------------------------------
    # 3a️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 3b️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # Element‑wise gating of the projection tensors
    left_fp16  = left_proj * left_gate   # (total_rows, H)
    right_fp16 = right_proj * right_gate

    # Restore 4‑D layout
    left  = left_fp16.view(B, N, N, hidden_dim)    # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣  Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for a batched matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B_H = B * hidden_dim
    left_mat  = left_perm.view(B_H, N, N)
    right_mat = right_perm.view(B_H, N, N)

    # Batched matrix multiplication – uses cuBLAS/Tensor Cores
    out_mat = torch.matmul(left_mat, right_mat)          # (B*H, N, N), fp16

    # Back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm statistics (over `out`)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    mean_out = torch.empty(total_rows_h, dtype=torch.float32, device=device)
    rstd_out = torch.empty(total_rows_h, dtype=torch.float32, device=device)

    grid_stats2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_stats_kernel[grid_stats2](
        out.view(total_rows_h, hidden_dim).contiguous(),
        mean_out,
        rstd_out,
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6️⃣  Fuse second LayerNorm (γ₂,β₂) with the output‑gate multiplication
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (H,)

    gated_fp16 = torch.empty_like(out.view(total_rows_h, hidden_dim), dtype=torch.float16, device=device)

    grid_gate = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_affine_gate_kernel[grid_gate](
        out.view(total_rows_h, hidden_dim).contiguous(),
        out_gate.view(total_rows_h, hidden_dim).contiguous(),
        gated_fp16,
        mean_out,
        rstd_out,
        to_out_norm_weight,
        to_out_norm_bias,
        total_rows_h,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()          # (dim, H)
    final_fp16 = F.linear(gated_fp16, to_out_w)         # (total_rows_h, dim)

    # Cast back to FP32 as required by the API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
