"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Row‑wise LayerNorm on the input (shape [B,N,N,dim]).
   A custom Triton kernel computes (x‑μ)/√(σ²+ε)·γ+β in fp16.

2. Fuse the five linear layers (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single fp16 GEMM:
       Y = LN(x) @ W_concat.T            # [R,5*H]
   where R = B·N·N, H = hidden_dim.
   The result is sliced into projected tensors and gates.

3. (Optional) mask the projected tensors (broadcasted over H).

4. Apply sigmoid gates and reshape to [B,N,N,H].

5. Triangular multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM of shape (B·H, N, N) using
   torch.bmm (cuBLAS), which is far faster than a Python loop.

6. Second LayerNorm over the hidden dimension (fp16 Triton kernel).

7. Element‑wise output‑gate, final linear projection back to `dim`,
   and cast the result to float32.

All heavy arithmetic runs in fp16, exploiting Tensor‑Core throughput.
Only the two LayerNorms are written in Triton; the rest uses highly
optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (support fp16 input → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim] (fp16 input)
    y_ptr,          # mutable *[total_rows, dim] (fp16 output)
    w_ptr,          # const   *[dim]           (fp32 weight γ)
    b_ptr,          # const   *[dim]           (fp32 bias   β)
    total_rows,     # int (runtime)
    dim,            # int (compile‑time)
    eps,            # fp32 scalar (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Computes LayerNorm over the last dimension of a 2‑D tensor
    (row‑wise). The result is written as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------- first pass: compute mean & variance -----------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # load a tile (BLOCK_M x BLOCK_D)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: write normalized values --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x-μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β are fp32 → keep computation in fp32, cast back to fp16
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]

        # store fp16
        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Tensors (float32)
        - config_dict  : {"dim": int, "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32, on the same device.
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)   # True → mask not applied
    eps        = 1e-5
    device     = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N                         # total rows for the flattened view

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – input is float32, output fp16
    # --------------------------------------------------------------
    x_flat = inp.view(R, dim).contiguous()               # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16,
                         device=device)                # fp16 output

    # launch Triton LN kernel
    BLOCK_M = 256            # rows per program (tunable)
    BLOCK_D = 64             # cols per tile   (tunable, works for dim≤384)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                         # fp32 input pointer
        x_norm,                         # fp16 output pointer
        weights["norm.weight"],         # γ  (fp32)
        weights["norm.bias"],           # β  (fp32)
        R, dim, eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse the five dense layers (proj + 3 gates) into one GEMM
    # --------------------------------------------------------------
    # Concatenate weights: shape [5*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                 # fp16

    # Linear: [R, dim] @ [dim, 5*H].T → [R, 5*H]  (fp16)
    y_concat = F.linear(x_norm, w_concat)   # bias is zero (None)

    H = hidden_dim
    left_proj   = y_concat[:, 0:H]                # [R, H]
    right_proj  = y_concat[:, H:2*H]              # [R, H]
    left_gate_p = y_concat[:, 2*H:3*H]            # [R, H]
    right_gate_p= y_concat[:, 3*H:4*H]            # [R, H]
    out_gate_p  = y_concat[:, 4*H:5*H]            # [R, H]

    # --------------------------------------------------------------
    # 3) Optional mask (broadcasted over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → flatten to [R,1] (fp16)
        mask_h = mask.to(torch.float16).view(R, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4) Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_p)   # fp16
    right_gate  = torch.sigmoid(right_gate_p) # fp16
    out_gate    = torch.sigmoid(out_gate_p)   # fp16

    # --------------------------------------------------------------
    # 5) Apply gates and reshape to [B,N,N,H]
    # --------------------------------------------------------------
    left  = (left_proj * left_gate).view(B, N, N, H)   # fp16
    right = (right_proj * right_gate).view(B, N, N, H)# fp16
    out_gate = out_gate.view(B, N, N, H)              # fp16, used later

    # --------------------------------------------------------------
    # 6) Triangular multiplicative update via batched GEMM
    #    Compute: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Bring hidden dim into the batch axis: (B*H, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (cuBLAS, fp16)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # Restore shape [B, N, N, H]
    out = out_b.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension (row‑wise)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()            # [R, H] fp16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R, H, eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, H)              # fp16

    # --------------------------------------------------------------
    # 8) Apply output gate, final linear projection, cast to float32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                      # fp16 element‑wise

    # Final linear layer: [R, H] → [R, dim] (fp16)
    final_fp16 = F.linear(out_norm.view(R, H),
                          weights["to_out.weight"].to(torch.float16))  # [R, dim]

    # Cast back to float32 and reshape
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
