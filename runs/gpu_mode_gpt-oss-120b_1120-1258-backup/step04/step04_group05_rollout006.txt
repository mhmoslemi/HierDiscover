"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the input channels (C) – Triton kernel,
    fp32 accumulation, fp16 output.

2️⃣  Fuse the five linear projections (left/right + three gates) into a
    single half‑precision GEMM by concatenating their weight matrices.

3️⃣  (Optional) apply the pair‑wise mask *before* the left/right gates.

4️⃣  Apply sigmoid gates and element‑wise gating (still fp16).

5️⃣  Triangular multiplicative update (outgoing version):
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   → realized as a batched GEMM `left @ rightᵀ` on tensors of shape
     (B·H, N, N).  The heavy O(N³) work is delegated to cuBLAS Tensor‑Core
     kernels (`torch.bmm`).

6️⃣  Second LayerNorm over the hidden dimension (H) – same Triton kernel.

7️⃣  Multiply by the output gate and apply the final linear projection
   back to the original channel dimension (C).

All heavy linear algebra uses fp16 Tensor‑Core kernels, the only Triton
kernels are the two LayerNorms.  The implementation works for the full
range of test configurations (B∈{1,2}, N∈{128…1024}, C∈{128,384},
H=128) and returns a float‑32 tensor as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,           # *const  float*   [total_rows, dim]   (input, fp32 or fp16)
    y_ptr,           # *mut    float16* [total_rows, dim]   (output)
    weight_ptr,      # *const  float*   [dim]              (γ)
    bias_ptr,        # *const  float*   [dim]              (β)
    total_rows,      # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a block of the input
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom kernel entry‑point
# ----------------------------------------------------------------------

    """
    Forward pass of the outgoing TriMul operator.

        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Shape preparation
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                 # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output) – Triton
    # --------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)          # fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)                     # fp16

    # Choose block size based on dim (max dim = 384)

    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right + three gates)
    # --------------------------------------------------------------
    # Each weight: [hidden_dim, dim] (no bias as per reference)
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    concat_w = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).half().contiguous()   # shape (5*H, C)

    # Fused GEMM – fp16
    proj_all = F.linear(x_norm_fp16, concat_w)   # (total_rows, 5*H)

    # Split back
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden_dim, dim=1
        )                                            # each (total_rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied BEFORE gating)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> broadcast over hidden dim
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16) and element‑wise gating
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)      # (total_rows, H)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape to 4‑D layout for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)          # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Bring the hidden dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)                # (BH, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N), fp16

    # Restore 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (fp16 output)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # (total_rows, H)
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,                     # input (fp16) – cast to fp32 inside kernel
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                      # (B, N, N, H), fp16

    # Final projection (fp16 GEMM)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)   # (total_rows, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16) # (total_rows, C)

    # --------------------------------------------------------------
    # 8️⃣ Cast to fp32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
