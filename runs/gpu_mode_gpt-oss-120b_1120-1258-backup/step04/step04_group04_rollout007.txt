"""
TriMul (outgoing) – fast forward pass implemented in Triton + PyTorch.

Algorithm
---------
1️⃣ Layer‑norm over the input channel dimension `dim` using a custom
   Triton kernel (`layernorm_fp16_kernel`). The kernel works on the
   flattened `[B·N·N, dim]` tensor, accumulates statistics in FP32 and
   writes the normalized result in FP16.

2️⃣ Fuse the five linear projections (left/right, three gates) into one
   FP16 GEMM by concatenating their weight matrices (`concat_weight`).
   The result `[R, 5·H]` (R = B·N·N) is split into five tensors:
   `left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`.

3️⃣ Optional pair‑wise mask (broadcast over the hidden dimension) is
   applied directly to the projection tensors.

4️⃣ Apply sigmoid gates (still in FP16) and element‑wise gating:
   `left = left_proj * sigmoid(left_gate)`,
   `right = right_proj * sigmoid(right_gate)`,
   `out_gate = sigmoid(out_gate)`.

5️⃣ Triangular multiplicative update:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   Implemented with a single `torch.einsum('bikd,bjkd->bijd', left, right)`,
   avoiding any expensive `permute + contiguous` copies required for a
   batched `bmm`. The operation utilizes cuBLAS Tensor‑Cores (FP16
   with FP32 accumulation).

6️⃣ Second layer‑norm over the hidden dimension `H` (`to_out_norm`) via
   the same Triton kernel.

7️⃣ Fuse the output gate and the final projection:
   `gated = out_norm * out_gate`,
   `final_fp16 = F.linear(gated_flat, to_out_weight)`.
   The result is cast back to FP32 as required by the API.

All heavy reductions (two LayerNorms) are performed by Triton,
dense linear algebra uses cuBLAS.  Memory copies are minimised
by keeping tensors in FP16 and by removing the explicit
`permute/contiguous` steps that dominated the previous
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                # *const   fp32/fp16   [total_rows, dim]
    y_ptr,                # *mut     fp16        [total_rows, dim]
    weight_ptr,           # *const   fp32        [dim]   (γ)
    bias_ptr,             # *const   fp32        [dim]   (β)
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – mean & variance (FP32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – write normalized result (FP16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.FloatTensor   [B, N, N, C]
        - mask_tensor   : torch.BoolTensor / FloatTensor   [B, N, N]   (optional)
        - weights_dict  : dict of model parameters (see code)
        - config_dict   : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.FloatTensor
        Tensor of shape [B, N, N, C] (float32, same device as inputs)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # input / output channel dimension C
    hidden_dim = config["hidden_dim"]   # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)   # when True, ignore mask

    B, N, _, _ = input_tensor.shape          # [B, N, N, C]
    total_rows = B * N * N                    # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm  (FP16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, C] FP32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    # Tune block sizes – 128 rows (BLOCK_M) and 64 columns (BLOCK_D) work well
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5 linear heads) – stay in FP16
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, C]) → [5H, C]
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5·H, C)

    # Linear:   [R, C] @ [C, 5·H]^T → [R, 5·H]   (FP16 GEMM)
    proj_all = F.linear(x_norm_fp16, concat_weight)   # (R, 5·H) FP16

    # Split into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )   # each [R, H] FP16

    # ------------------------------------------------------------------
    # 2a️⃣ Optional pair‑wise mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Apply sigmoid gates and element‑wise gating (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16  = left_proj * left_gate      # [R, H]
    right_fp16 = right_proj * right_gate    # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Reshape to 4‑D tensors (no extra copies)
    # ------------------------------------------------------------------
    left  = left_fp16.view(B, N, N, hidden_dim)      # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)    # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update – einsum avoids permute+copy
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('bikd,bjkd->bijd', left, right)   # FP16, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # FP16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Output gating + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                               # FP16 element‑wise

    # Final linear layer (H → C) – FP16 weight, no bias
    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)          # [R_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)      # [R_h, C] FP16

    # Cast back to FP32 as required by the interface
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
