    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for out-of-bounds M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # add batch offset to pointers
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

"""
custom_kernel: optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the 4‑D tensor [B,N,N,D].
2️⃣  Fuse projection + gate for the left and right paths:
    * a single matmul with concatenated weights produces
      (proj, pre‑gate) for each hidden dimension H.
3️⃣  Optional pair‑wise mask (broadcast over H) and sigmoid‑gates.
4️⃣  Tri‑multiplicative update:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    This is a *batched* matrix multiplication of H·B independent
    N×N @ N×Nᵀ products.  We implement it with a custom Triton kernel
    (one kernel launch handles all batches, avoiding the overhead of
    cuBLAS’s batched GEMM).
5️⃣  Second LayerNorm (on the hidden dimension) – again FP32 → FP16.
6️⃣  Apply the output‑gate (sigmoid) and a final linear projection
    (H → D).  The final result is cast back to FP32 as required by the
    reference PyTorch module.
All heavy compute is performed in FP16 with FP32 accumulation for
numerical stability, while the two LayerNorms are hand‑written Triton
kernels to keep memory traffic low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    y_ptr,          #       half*  [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- pass 1: mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the Tri‑multiplicative update
# Computes (B*H) independent N×N @ N×Nᵀ products.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # which matrix in the batch
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current matrix
    a = a_ptr + pid_batch * stride_a_batch
    b = b_ptr + pid_batch * stride_b_batch
    c = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a_tile = tl.load(a_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0).to(tl.float32)

        # Load b tile (transposed): shape (BLOCK_K, BLOCK_N)
        # b is stored as (batch, row=j, col=k); we need (k, j)
        b_ptrs = b + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b_tile = tl.load(b_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float32)

        acc += tl.dot(a_tile, b_tile)

    # Write back (FP16)
    c_ptrs = c + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, D] (float32)
        mask_tensor  : torch.Tensor [B, N, N] (float32) – may be unused
        weights_dict : {
            "norm.weight":        Tensor[D],
            "norm.bias":          Tensor[D],
            "left_proj.weight":   Tensor[H, D],
            "right_proj.weight":  Tensor[H, D],
            "left_gate.weight":   Tensor[H, D],
            "right_gate.weight":  Tensor[H, D],
            "out_gate.weight":    Tensor[H, D],
            "to_out_norm.weight": Tensor[H],
            "to_out_norm.bias":   Tensor[H],
            "to_out.weight":      Tensor[D, H],
        }
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True),
        }

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, D).contiguous()          # [rows, D] fp32

    x_norm_h = torch.empty((total_rows, D), dtype=torch.float16,
                           device=device)                         # fp16 output buffer

    BLOCK_M = 256                 # rows per program (tuned)
    BLOCK_D = 64                  # columns per inner loop (covers D ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        D,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape back to [B,N,N,D] (fp16)
    x_norm_h = x_norm_h.view(B, N, N, D)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (four matmuls in two fused calls)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights (shape: 2H x D)
    left_cat_w = torch.cat([weights["left_proj.weight"],
                            weights["left_gate.weight"]], dim=0).to(
        torch.float16, non_blocking=True)                         # [2H, D]

    right_cat_w = torch.cat([weights["right_proj.weight"],
                             weights["right_gate.weight"]], dim=0).to(
        torch.float16, non_blocking=True)                         # [2H, D]

    rows = total_rows
    # Linear: (rows, D) @ (2H, D).T -> (rows, 2H)
    left_cat = F.linear(x_norm_h.view(rows, D), left_cat_w)      # fp16
    right_cat = F.linear(x_norm_h.view(rows, D), right_cat_w)    # fp16

    # split into projection and pre‑gate tensors
    left_proj = left_cat[:, :H]          # (rows, H)
    left_gate_pre = left_cat[:, H:]      # (rows, H)
    right_proj = right_cat[:, :H]
    right_gate_pre = right_cat[:, H:]

    # Output gate (shared across all positions)
    out_gate_pre = F.linear(
        x_norm_h.view(rows, D),
        weights["out_gate.weight"].to(torch.float16, non_blocking=True)
    )   # (rows, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left_proj * left_gate     # (rows, H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring H into the batch dimension → [BH, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_mat = left_perm.reshape(BH, N, N)    # (BH, N, N)
    right_mat = right_perm.reshape(BH, N, N)  # (BH, N, N)

    # allocate output buffer (fp16)
    out_mat = torch.empty_like(left_mat)

    # launch Triton batched GEMM: out = left @ rightᵀ
    BLOCK_M = 64    # tuned for H100, works up to N=1024
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (BH, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                   # M, N, K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        # right matrix is accessed transposed inside the kernel
        right_mat.stride(0), right_mat.stride(2), right_mat.stride(1),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back → [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (on hidden dim H)
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_h.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # gated output (fp16)

    final_h = F.linear(
        out_norm.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # (rows_h, D) fp16

    # Cast to FP32 as in the reference implementation
    final = final_h.view(B, N, N, D).to(torch.float32)

    return final
