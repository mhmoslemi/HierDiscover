import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing**
TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Input LayerNorm (row‑wise) – native Triton kernel that reads each
    row once, computes mean/var in FP32 and writes the normalized
    FP16 result.

2️⃣  Fuse the five linear projections (left/right + 3 gates) into a
    single FP16 GEMM (`torch.nn.functional.linear`).  The weight matrix
    is the concatenation of the five `[hidden, dim]` matrices.

3️⃣  (Optional) apply the binary mask to the left/right projections.

4️⃣  Apply sigmoid gates (still FP16) and element‑wise gating.

5️⃣  Triangular multiplicative update – a batched matrix‑multiply:
    for each batch and hidden channel we compute
    ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.
    This is performed with a single `torch.bmm` on the reshaped tensors,
    which lets cuBLAS/Tensor‑Core handle the whole batch efficiently.

6️⃣  Fused LayerNorm + output‑gate (both FP16) – another Triton kernel.
    It normalises the triangular‑update tensor, applies the learned
    per‑channel weight & bias and finally multiplies by the pre‑computed
    `out_gate`.

7️⃣  Final linear projection `hidden → dim` (FP16 GEMM) and cast back
    to FP32 for the module output.

Only the two LayerNorm steps are written in Triton; all dense linear
algebra uses cuBLAS/Tensor‑Core which is already optimal on an H100.
The implementation avoids intermediate synchronisations and extra
memory copies, giving a sizable speed‑up over the reference code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (input FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (output FP16)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static dimension (≤ 384)
    eps: tl.constexpr,   # LayerNorm epsilon
    BLOCK_M: tl.constexpr,  # rows per program (e.g. 64)
    BLOCK_D: tl.constexpr,  # columns processed per iteration (e.g. 64)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, dim] (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, dim] (post‑norm⋅gate)
    weight_ptr,          # *const float* [dim]          (γ)
    bias_ptr,            # *const float* [dim]          (β)
    gate_ptr,            # *const half*  [total_rows, dim] (out_gate, already sigmoid’ed)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        # multiply by the already‑sigmoided out_gate (half → FP32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]     (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32 / bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {
              "dim"        : int   (C),
              "hidden_dim" : int   (H),
              "nomask"    : bool,
              "eps"       : float (optional)
          }

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    # Tune block sizes: 64 rows × 64 columns fits comfortably in shared memory
    BLOCK_M = 64
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No explicit sync – the next kernel will automatically wait for the
    # previous one due to CUDA stream ordering.

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five projection / gate weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # (5*H, C)

    # Linear projection of the normalised input (R,5*H) FP16
    proj_all = F.linear(x_norm_fp16, concat_w)   # (R, 5*H)

    # Split into the individual tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                            # each (R, H)

    # ------------------------------------------------------------------
    # 2a) Apply mask if required (mask is broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (R,1) → half
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates (still FP16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)        # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)      # (R, H)
    out_gate = torch.sigmoid(out_gate_fp16)          # (R, H) – will be used later

    left_fp16 = left_proj_fp16 * left_gate           # (R, H)
    right_fp16 = right_proj_fp16 * right_gate        # (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Reshape for the triangular multiplicative update
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden) # (B, N, N, H)

    # Prepare batched matrices:
    #   left_mat  : (B*H, N, N)  where rows = i, cols = k
    #   right_mat : (B*H, N, N)  where rows = k, cols = j
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    # Batched matrix multiplication (Tensor‑Core, FP16)
    out_mat = torch.bmm(left_mat, right_mat)                        # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + final output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat)        # (R, H) FP16

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                          # already sigmoid’ed, FP16
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()                # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)       # (R, dim) FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)        # cast to FP32
    return final
