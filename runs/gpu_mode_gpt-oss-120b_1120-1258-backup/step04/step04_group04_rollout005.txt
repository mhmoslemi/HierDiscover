"""
TriMul (outgoing) – fused Triton implementation
================================================

Algorithm
---------
1️⃣ Layer‑norm over the last channel (dim) → fp16 (custom Triton kernel).
2️⃣ Fuse the five linear projections (left/right & three gates) into one
   half‑precision GEMM (torch.nn.functional.linear).
3️⃣ Optional pairwise mask (broadcasted over the hidden dimension).
4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣ Triangular multiplicative update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   realised by a *batched* GEMM kernel written in Triton
   (batch = B·hidden_dim).  This kernel is a Tensor‑Core‑friendly
   MatMul that avoids the many small `torch.bmm` launches.
6️⃣ Second layer‑norm over the hidden dimension (fp16, Triton).
7️⃣ Multiply by the output gate, then a final half‑precision linear
   projection back to `dim`.  Cast the result to float32 for the API.

All heavy reductions (the two LayerNorms) and the N³ reduction (TriMul)
are performed in Triton; the remaining dense linear layers use cuBLAS
via `torch.nn.functional.linear`, which already runs on Tensor Cores.
The implementation works for the full range of test configurations
(batches 1‑2, sequence lengths 128‑1024, dim up to 384,
hidden_dim = 128, with or without masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm (fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # *const  fp32/fp16* [total_rows, dim]
    y_ptr,            # *mut    fp16*    [total_rows, dim]
    weight_ptr,       # *const  fp32*    [dim]   (γ)
    bias_ptr,         # *const  fp32*    [dim]   (β)
    total_rows,       # int (runtime)
    dim: tl.constexpr,                # compile‑time
    eps: tl.constexpr,                # compile‑time
    BLOCK_M: tl.constexpr,            # rows per program
    BLOCK_D: tl.constexpr,            # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance (fp32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values (fp16)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Batched GEMM kernel for the N³ TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,                # *const half* [BATCH, M, K]
    B_ptr,                # *const half* [BATCH, K, N]
    C_ptr,                # *mut   half* [BATCH, M, N]
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(2)          # which batch (B·H)
    pid_m   = tl.program_id(0)           # output tile row
    pid_n   = tl.program_id(1)           # output tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile
    a_ptrs = A_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + batch_id * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)                         # fp32 accumulation
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    c = acc.to(tl.float16)
    c_ptrs = C_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Triton‑powered TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C] (float32)
            - mask_tensor : [B, N, N] (bool/float, optional)
            - weights_dict : model parameters (float32)
            - config_dict  : {"dim": C,
                               "hidden_dim": H,
                               "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H (fixed to 128 in the tests)
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape          # [B, N, N, C]
    total_rows = B * N * N                    # flatten over (B,N,N)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, C] fp32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64            # enough for dim ≤ 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gating (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, C])
    cat_W = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Linear projection: (R, C) @ (C, 5H)^T → (R, 5H)
    proj_all = F.linear(x_norm_fp16, cat_W)    # fp16, bias = 0

    # Split into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: [R, H]

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N] → flatten → [R,1]
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates & element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16 = left_proj * left_gate      # [R, H]
    right_fp16 = right_proj * right_gate   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Reshape for the triangular update
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # Convert to (batch, hidden, i, k) / (batch, hidden, k, j) layout
    left_mat = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N] (i,k)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    # Collapse the batch & hidden dimensions → batched GEMM inputs
    BH = B * hidden_dim
    left_mat = left_mat.view(BH, N, N)                  # [B·H, N, N]
    right_mat = right_mat.view(BH, N, N)                # [B·H, N, N]

    # ------------------------------------------------------------------
    # 4️⃣ N³ triangular multiplicative update (batched GEMM in Triton)
    # ------------------------------------------------------------------
    out_mat = torch.empty_like(left_mat)               # fp16 output buffer

    # Strides for the three tensors (batch, M, K) / (batch, K, N) / (batch, M, N)
    stride_Ab = left_mat.stride(0)
    stride_Am = left_mat.stride(1)
    stride_Ak = left_mat.stride(2)

    stride_Bb = right_mat.stride(0)
    stride_Bk = right_mat.stride(1)
    stride_Bn = right_mat.stride(2)

    stride_Cb = out_mat.stride(0)
    stride_Cm = out_mat.stride(1)
    stride_Cn = out_mat.stride(2)

    # Tunable block sizes – these work well on H100 for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32    # N is always a multiple of 32 in the test suite

    grid = (
        triton.cdiv(N, BLOCK_M),     # program_id(0)
        triton.cdiv(N, BLOCK_N),     # program_id(1)
        BH,                          # program_id(2) = batch dimension
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH, N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back‑reshape to [B, N, N, H] (still fp16)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output gating + final linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                           # fp16 element‑wise

    # Final linear layer (H → C) – half‑precision weight, no bias
    final_fp16 = F.linear(gated.view(total_rows_h, hidden_dim),
                          weights["to_out.weight"].half())   # [R_h, C] fp16

    # Cast back to float32 as required by the specification
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
