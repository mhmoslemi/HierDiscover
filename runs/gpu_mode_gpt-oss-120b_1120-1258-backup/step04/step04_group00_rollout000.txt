2️⃣ Fuse **all** linear projections/gates in a **single** GEMM:
        concat( left_proj, left_gate, right_proj, right_gate, out_gate )
   – Output shape [B·N·N, 5·H] (H = hidden_dim).  Slices give the
     individual tensors.  Everything stays in FP16.
3️⃣ Optional pair‑wise mask (broadcast over H) applied to left/right
   projection tensors before gating.
4️⃣ Apply sigmoid‑gates and element‑wise product:
        left = left_proj * σ(left_gate)
        right = right_proj * σ(right_gate)
5️⃣ Tri‑multiplicative update (batched Gram‑matrix):
        out_h(i,j) = Σ_k left(i,k,h) · right(j,k,h)
   – Implemented as a batched GEMM on tensors of shape
        [B·H, N, N]  (left)  ×  [B·H, N, N]ᵀ (right) → [B·H, N, N].
6️⃣ Second LayerNorm (on the hidden dimension) – same Triton kernel.
7️⃣ Output‑gate (σ) and final linear projection (H → C) in FP16,
   cast to FP32 for the returned tensor.

Only the two LayerNorms are written in Triton; all dense linear layers
and the N×N GEMMs are delegated to PyTorch’s highly‑optimised Tensor‑Core
kernels.  This drastically reduces kernel‑launch overhead while keeping
the memory‑bound parts (LayerNorm) on‑GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Compute LayerNorm over the last dimension of a 2‑D tensor.
    The mean/variance are accumulated in FP32, the normalized result
    is written as FP16 (γ,β are FP32).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------------- pass 1: mean & variance -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ----------------- pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]          # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, C] (float32)
        mask_tensor  : torch.Tensor [B, N, N] (float32) – may be unused
        weights_dict : {
            "norm.weight":        Tensor[C],
            "norm.bias":          Tensor[C],
            "left_proj.weight":   Tensor[H, C],
            "right_proj.weight":  Tensor[H, C],
            "left_gate.weight":   Tensor[H, C],
            "right_gate.weight":  Tensor[H, C],
            "out_gate.weight":    Tensor[H, C],
            "to_out_norm.weight": Tensor[H],
            "to_out_norm.bias":   Tensor[H],
            "to_out.weight":      Tensor[C, H],
        }
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True),
        }

    Returns:
        torch.Tensor [B, N, N, C] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    total_rows = B * N * N                               # rows of the 2‑D view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim),
                              dtype=torch.float16,
                              device=device)

    # launch
    BLOCK_M = 512                # rows per program (tuned)
    BLOCK_D = 128                # columns per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape back to 4‑D (still FP16)
    x_norm_fp16 = x_norm_fp16.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse ALL linear projections + gates (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate weights along the output‑feature dimension:
    # order = [left_proj, left_gate, right_proj, right_gate, out_gate]
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # (5*H, dim)
    ).to(torch.float16, non_blocking=True)

    # Linear: (total_rows, dim) @ (5*H, dim).T -> (total_rows, 5*H)
    x_norm_flat = x_norm_fp16.view(total_rows, dim)
    proj_out = F.linear(x_norm_flat, combined_weight)   # FP16

    # Slice the concatenated result
    left_proj    = proj_out[:, 0:H]                           # [rows, H]
    left_gate_pre = proj_out[:, H:2 * H]                      # [rows, H]
    right_proj   = proj_out[:, 2 * H:3 * H]                    # [rows, H]
    right_gate_pre = proj_out[:, 3 * H:4 * H]                 # [rows, H]
    out_gate_pre = proj_out[:, 4 * H:5 * H]                   # [rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16,
                                                   non_blocking=True)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)   # [rows, H]
    right_gate = torch.sigmoid(right_gate_pre)  # [rows, H]
    out_gate   = torch.sigmoid(out_gate_pre)    # [rows, H]

    left_h  = left_proj * left_gate   # [rows, H]
    right_h = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # bring H into batch axis → [B*H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_mat  = left_perm.view(BH, N, N)          # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)         # (BH, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N)

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)      # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)          # FP16
    out_norm = out_norm * out_gate                 # gated output (FP16)

    # Final linear: (hidden_dim → dim)
    final_fp16 = F.linear(
        out_norm.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )                                            # [rows_h, dim] (FP16)

    # Cast to FP32 as in the reference implementation
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```

The implementation follows the reference PyTorch TriMul module while
fusing all linear layers (projections, gates, and output‑gate) into a
single matrix multiplication and using a custom Triton kernel for the
two LayerNorms.  All heavy arithmetic (the N×N Gram‑matrix update and
the dense projections) run in FP16 on Tensor‑Core kernels, keeping the
total runtime well below the target for the prescribed test cases.
