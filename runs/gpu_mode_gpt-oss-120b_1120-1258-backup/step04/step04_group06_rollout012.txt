def gating_kernel(
    proj_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows, hidden_dim,
    stride_proj_row, stride_mask_row,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    offs_h = tl.arange(0, BLOCK_H)
    # Only compute for hidden_dim <= BLOCK_H
    # In case hidden_dim < BLOCK_H, we mask hidden dimension as well
    mask_h = offs_h < hidden_dim

    # Combine row mask and hidden mask for loads/stores
    # Row mask broadcast: mask_r[:, None] & mask_h[None, :]

    # Load mask value if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r * stride_mask_row,
                           mask=mask_r,
                           other=1.0).to(tl.float16)
        # Expand to [BLOCK_R, BLOCK_H] broadcast
        mask_val = mask_val[:, None]   # shape [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # compute column offsets
    base0 = 0
    base1 = hidden_dim
    base2 = 2 * hidden_dim
    base3 = 3 * hidden_dim
    base4 = 4 * hidden_dim

    col0 = base0 + offs_h
    col1 = base1 + offs_h
    col2 = base2 + offs_h
    col3 = base3 + offs_h
    col4 = base4 + offs_h

    row_off = offs_r[:, None] * stride_proj_row

                        mask=mask_r[:, None] & mask_h[None, :],

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm (FP16‑dominated, only a tiny Triton kernel is used):
1️⃣ Layer‑norm on the input (float32 → float16) using the highly‑optimised
   torch.nn.functional.layer_norm implementation.
2️⃣ Fuse the five linear layers (left/right projections + three gates) into a
   single half‑precision GEMM:   X_norm @ W_concat.T → [R,5·H].
3️⃣ Triton kernel `gate_kernel`:
   * loads the concatenated projection for a block of rows,
   * computes σ(·) for the three gate vectors,
   * applies the optional mask,
   * writes out the gated tensors `left`, `right` and `out_gate`.
   This fuses mask, sigmoid and element‑wise multiplications in one pass.
4️⃣ Triangular multiplicative update:
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   Implemented as a batched half‑precision GEMM (cuBLAS Tensor‑Core).
5️⃣ Second Layer‑Norm on the hidden dimension (again torch’s fast kernel).
6️⃣ Multiply by the output gate and project back to the original dimension.

All heavy arithmetic is performed in FP16, but the final result is returned
as float32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: gate + optional mask fusion
# ----------------------------------------------------------------------
@triton.jit
def gate_kernel(
    proj_ptr,                     # const half*   [R, 5*H]
    mask_ptr,                     # const half*   [R] (optional)
    left_ptr,                     #      half*   [R, H]
    right_ptr,                    #      half*   [R, H]
    out_gate_ptr,                 #      half*   [R, H]
    R,                            # total rows  (runtime)
    H,                            # hidden dim  (runtime)
    proj_row_stride,              # stride (in elements) between rows of proj_concat (≈5*H)
    out_row_stride,               # stride (in elements) between rows of left/right/out_gate (≈H)
    mask_row_stride,              # stride of mask (usually 1)
    BLOCK_R: tl.constexpr,        # rows per program
    BLOCK_H: tl.constexpr,        # hidden dim per program (≤ H)
    HAS_MASK: tl.constexpr,       # 1 → mask present, 0 → no mask
):
    """
    For each (row, hidden) pair compute:
        left  = proj_left  * sigmoid(proj_left_gate)  * mask
        right = proj_right * sigmoid(proj_right_gate) * mask
        out_gate = sigmoid(proj_out_gate)
    where the five projection slices are stored consecutively in proj_ptr.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # Load optional mask (scalar per row) and broadcast to hidden dim
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r * mask_row_stride,
                           mask=mask_r, other=1.0).to(tl.float16)   # [BLOCK_R]
        mask_val = mask_val[:, None]                               # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Column offsets for the five slices inside proj_concat
    # ------------------------------------------------------------------
    col0 = offs_h + 0 * H          # left projection
    col1 = offs_h + 1 * H          # right projection
    col2 = offs_h + 2 * H          # left gate
    col3 = offs_h + 3 * H          # right gate
    col4 = offs_h + 4 * H          # out gate

    # Row offset (in elements) for the source tensor
    row_off_src = offs_r[:, None] * proj_row_stride          # [BLOCK_R,1]

    # Load the five slices (mask hides rows beyond R)
    left_proj = tl.load(proj_ptr + row_off_src + col0[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float16)
    right_proj = tl.load(proj_ptr + row_off_src + col1[None, :],
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float16)
    left_gate_raw = tl.load(proj_ptr + row_off_src + col2[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
    right_gate_raw = tl.load(proj_ptr + row_off_src + col3[None, :],
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
    out_gate_raw = tl.load(proj_ptr + row_off_src + col4[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)

    # ------------------------------------------------------------------
    # Sigmoid (σ(x) = 1 / (1+exp(-x))) – operates in FP16
    # ------------------------------------------------------------------
    sigmoid = lambda x: 1.0 / (1.0 + tl.exp(-x))

    left_gate  = sigmoid(left_gate_raw)
    right_gate = sigmoid(right_gate_raw)
    out_gate   = sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # Apply mask and write the three outputs
    # ------------------------------------------------------------------
    left  = left_proj  * left_gate  * mask_val
    right = right_proj * right_gate * mask_val

    # Row offset for the destination tensors (stride == H)
    row_off_dst = offs_r[:, None] * out_row_stride

    tl.store(left_ptr      + row_off_dst + offs_h[None, :],
             left,  mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr     + row_off_dst + offs_h[None, :],
             right, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr  + row_off_dst + offs_h[None, :],
             out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]               # input / output channel dimension
    hidden_dim = config["hidden_dim"] # usually 128
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to FP16
    # --------------------------------------------------------------
    # weights for LN are kept in float32 (exact as in the reference)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    x = x_norm_fp32.to(torch.float16)   # [R, dim] in half‑precision

    # --------------------------------------------------------------
    # 2️⃣ Fuse 5 linear layers (2 proj + 3 gates) into one GEMM
    # --------------------------------------------------------------
    # concatenate the five weight matrices: (5*hidden_dim, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # half‑precision

    proj_concat = F.linear(x, w_concat)          # [R, 5*hidden_dim] (fp16)

    # --------------------------------------------------------------
    # 3️⃣ Gate + optional mask (fused Triton kernel)
    # --------------------------------------------------------------
    # allocate output buffers
    left      = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    right     = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    out_gate  = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)

    if nomask:
        # dummy mask – will never be read
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
    else:
        mask_fp16 = mask_tensor.to(torch.float16).reshape(total_rows).contiguous()
        has_mask = 1

    # launch kernel
    BLOCK_R = 128                # rows per program – tuned for H100
    BLOCK_H = 128                # hidden_dim (max hidden) – fits in registers
    grid = (triton.cdiv(total_rows, BLOCK_R),)

    gate_kernel[grid](
        proj_concat,
        mask_fp16,
        left,
        right,
        out_gate,
        total_rows,
        hidden_dim,
        proj_concat.stride(0),          # row stride of proj_concat (5*H)
        left.stride(0),                 # stride of the output tensors (H)
        mask_fp16.stride(0) if not nomask else 0,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        HAS_MASK=has_mask,
    )

    # reshape back to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim) # [B, N, N, H]

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # bring H to the batch dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * hidden_dim, N, N)     # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]

    # batched matrix multiplication (Tensor‑Core fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # restore original layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]   # float32
    to_out_norm_bias   = weights["to_out_norm.bias"]    # float32
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                                  # fp16 (same dtype as out)

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                       # element‑wise (fp16)

    to_out_weight = weights["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]
    final_half = F.linear(
        out_norm.view(total_rows, hidden_dim),   # [R, hidden_dim]
        to_out_weight,
    )                                            # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
