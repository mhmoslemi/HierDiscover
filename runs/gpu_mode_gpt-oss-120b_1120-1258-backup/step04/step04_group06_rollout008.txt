"""
Fast outgoing TriMul (AlphaFold‑3) forward pass – Triton + PyTorch.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the 4‑D input.
2️⃣ Fuse the 5 linear layers (2 projections + 3 gates) into a single
   half‑precision GEMM:  [R, dim] @ [5·hidden, dim]ᵀ → [R, 5·hidden].
3️⃣ Split the result, apply the optional mask and the 3 sigmoid gates.
4️⃣ Triangular multiplicative update:
   out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d].
   Implemented with a single ``torch.einsum`` which avoids the two large
   ``permute‑contiguous`` copies required for a batched GEMM.
5️⃣ Second LayerNorm (FP16) over the hidden dimension.
6️⃣ Apply the output gate and a final half‑precision linear projection
   back to ``dim``.  Cast the result to ``float32`` as in the reference.

All heavy per‑row normalisations are performed by a custom Triton kernel.
The expensive N³‑step uses a highly‑optimised PyTorch kernel
(`torch.einsum`) that works directly on the original layout,
thereby eliminating the costly memory copies of the previous
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const *   [total_rows, dim] (float32 or fp16)
    y_ptr,          # mutable * [total_rows, dim] (float16 output)
    weight_ptr,     # const *   [dim] (float32)
    bias_ptr,       # const *   [dim] (float32)
    total_rows,     # int (runtime)
    dim,            # tl.constexpr
    eps,            # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = (x - μ) / √(σ²+eps) * γ + β
    The result is stored as FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute sum and sum‑of‑squares per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    device      = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows  = B * N * N                # rows for the row‑wise LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    # Choose block sizes that work well for the tested dimensions
    BLOCK_M = 1024
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (2 projections + 3 gates)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices -> [5·hidden, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Single GEMM: [R, dim] @ [5·hidden, dim]ᵀ → [R, 5·hidden]
    proj = F.linear(x_norm, w_concat)                 # fp16, shape [R, 5·hidden]

    H = hidden_dim
    left_proj     = proj[:, 0:H]                       # [R, H]
    right_proj    = proj[:, H:2*H]                     # [R, H]
    left_gate_pre = proj[:, 2*H:3*H]                   # [R, H]
    right_gate_pre= proj[:, 3*H:4*H]                   # [R, H]
    out_gate_pre  = proj[:, 4*H:5*H]                   # [R, H]

    # Sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied only to the projected tensors)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Element‑wise gating
    left  = left_proj * left_gate          # [R, H]
    right = right_proj * right_gate        # [R, H]

    # Reshape back to 4‑D tensors for the N³ update
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (einsum, no extra copies)
    # --------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to ``dim``
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate          # fp16 element‑wise

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
