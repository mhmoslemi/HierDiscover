"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input tensor (shape B·N·N × dim).  
   Implemented in Triton (`row_layernorm`) and written in FP16.

2️⃣ Fuse the five linear layers (left/right projections + three gates) into a
   single half‑precision GEMM:
   `[R, dim] @ [5·hidden, dim].T → [R, 5·hidden]`
   where `R = B·N·N`.  The result is sliced into left_proj, right_proj,
   left_gate, right_gate and out_gate.

3️⃣ Apply the optional mask (only to the two projection tensors) and the
   three sigmoid gates (still FP16).

4️⃣ Triangular multiplicative update:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.  
   This is exactly a batched matrix multiplication of shape
   `(B·hidden, N, N) @ (B·hidden, N, N)^T`.  We use `torch.bmm`,
   which calls cuBLAS and exploits Tensor‑Cores on H100.

5️⃣ Second LayerNorm over the hidden dimension (FP16, Triton).

6️⃣ Multiply by the output gate, project back to `dim` with a final half‑precision
   linear layer and cast the result to FP32 (as in the reference module).

The implementation supports both masked and unmasked inputs and works for all
test configurations in the problem statement.  Only the layer‑normalisations
are performed in custom Triton kernels; all large matrix multiplications use
highly‑optimised cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (output FP16)
# ----------------------------------------------------------------------
@triton.jit
def row_layernorm(
    x_ptr,            # const *  [total_rows, dim]  (float32 or fp16)
    y_ptr,            # mutable *[total_rows, dim] (float16 output)
    weight_ptr,       # const * [dim]               (float32)
    bias_ptr,         # const * [dim]               (float32)
    total_rows,       # runtime int
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    y = (x - μ) / sqrt(σ²+eps) * γ + β   (output FP16)
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the feature dimension
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load as float32 irrespective of input precision
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values γ·(x−μ)·rstd + β  (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor : [B, N, N] (float32 or bool)
            - weights_dict: mapping of weight names to torch tensors
            - config_dict : contains "dim", "hidden_dim", and optionally "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape               # B, N, N, dim
    total_rows = B * N * N                         # rows for row‑wise LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_flat = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    row_layernorm[grid](
        x_flat,                     # input pointer (float32)
        x_norm_flat,                # output pointer (fp16)
        weights["norm.weight"],     # γ (float32)
        weights["norm.bias"],       # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (2 projections + 3 gates)
    # --------------------------------------------------------------
    # Concatenate all five weight matrices: [5*hidden, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden, dim)
            weights["right_proj.weight"],  # (hidden, dim)
            weights["left_gate.weight"],   # (hidden, dim)
            weights["right_gate.weight"],  # (hidden, dim)
            weights["out_gate.weight"],    # (hidden, dim)
        ],
        dim=0,
    ).to(torch.float16)   # half‑precision for Tensor‑Cores

    # Single GEMM: [R, dim] @ [5*hidden, dim].T  →  [R, 5*hidden]
    proj_concat = F.linear(x_norm_flat, w_concat)   # (R, 5*hidden) fp16

    H = hidden_dim
    left_proj   = proj_concat[:, 0:H]               # (R, H)
    right_proj  = proj_concat[:, H:2*H]             # (R, H)
    left_gate_p = proj_concat[:, 2*H:3*H]           # (R, H)
    right_gate_p = proj_concat[:, 3*H:4*H]          # (R, H)
    out_gate_p  = proj_concat[:, 4*H:5*H]           # (R, H)

    # Sigmoid on gates (still fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied only to left/right projections)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → flatten to (R, 1) and cast to fp16
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Apply element‑wise gates
    left  = left_proj * left_gate          # (R, H) fp16
    right = right_proj * right_gate        # (R, H) fp16

    # Reshape back to 4‑D tensors for the N³ update
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) → treat each head as a batch dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Collapse batch & head dimensions → (B*H, N, N)
    left_mat  = left_perm.view(B * H, N, N)              # (BH, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (BH, N, N)

    # Batched matrix multiplication: (BH, N, N) @ (BH, N, N)^T
    # Uses cuBLAS Tensor‑Core kernels (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original order: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (FP16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    row_layernorm[grid_ln2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_flat.view(B, N, N, H)   # [B, N, N, H] FP16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise (FP16)

    # Final linear projection: [R, H] → [R, dim]   (half precision)
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)    # (dim, H)
    )   # (R, dim) FP16

    # Cast to FP32 to match reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
