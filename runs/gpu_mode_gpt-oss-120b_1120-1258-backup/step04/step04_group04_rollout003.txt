def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Cases: flatten rows for layernorm
    total_rows = B * N * N

    # 1st layernorm
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # choose block sizes
    BLOCK_M = 128
    BLOCK_D = max(32, 2 ** (dim - 1).bit_length() // 2) # not needed

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # projections
    concat_weight = torch.cat([weights["left_proj.weight"],
                               weights["right_proj.weight"],
                               weights["left_gate.weight"],
                               weights["right_gate.weight"],
                               weights["out_gate.weight"]], dim=0).to(torch.float16).contiguous()
    proj_all = F.linear(x_norm_fp16, concat_weight)  # [R, 5H]
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(proj_all, hidden_dim, dim=1)

    if not nomask:
        # mask shape [B,N,N] cast to fp16 and flatten
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16 = left_proj * left_gate
    right_fp16 = right_proj * right_gate

    # reshape for GEMM
    left = left_fp16.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()  # B,H,N,N (i,k)
    right = right_fp16.view(B, N, N, hidden_dim).permute(0, 3, 2, 1).contiguous()  # B,H,N,N (k,j)

    BH = B * hidden_dim
    left_mat = left.view(BH, N, N)
    right_mat = right.view(BH, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # fp16

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # B,N,N,H

    # second layernorm over hidden_dim
    out_flat = out.view(total_rows, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)

    gated = out_norm * out_gate.view(B, N, N, hidden_dim)

    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16)

    gated_flat = gated.view(total_rows, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # no bias

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```

"""
TriMul (outgoing) – fast forward pass in Triton + PyTorch.

Algorithm
---------
1️⃣ Layer‑norm on the last channel (`dim`) → fp16 output (custom Triton kernel).
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single GEMM by concatenating their weight matrices.
3️⃣ (Optional) apply the pair‑wise mask to the left/right projections.
4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣ Triangular multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   realized as a batched fp16 GEMM (`torch.bmm`) on tensors of shape
   (B·H, N, N), where H = hidden_dim.
6️⃣ Second layer‑norm over the hidden dimension (fp16 output, same Triton kernel).
7️⃣ Multiply by the output gate and project back to `dim` with a final fp16 linear.
8️⃣ Cast the result to float32 (required by the spec).

All heavy reductions (the two LayerNorms) are performed by a custom
Triton kernel; dense linear algebra uses fp16‐Tensor‑Core‑friendly
operations. The implementation works for the full range of test
configurations (various batch sizes, sequence lengths, channel sizes,
and optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel – works for fp32 or fp16 input, writes fp16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const   X*   [total_rows, dim]   (fp32 or fp16 input)
    y_ptr,          # *mut     half* [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const float*   [dim]            (γ)
    bias_ptr,       # *const float*   [dim]            (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    """
    Row‑wise LayerNorm:
      μ = mean(x, dim)
      σ² = var(x, dim)
      y = ((x-μ)/sqrt(σ²+eps))*γ + β
    The kernel processes a tile of rows (BLOCK_M) and a tile of columns
    (BLOCK_D). Statistics are accumulated in fp32 for numerical stability,
    while the final normalized result is stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: compute per‑row mean and variance (fp32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a tile of shape [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # cast to fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # fp32 raw data

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        # γ·norm + β  (still fp32), then down‑cast to fp16
        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping of weight names to torch.Tensors
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # Config
    dim = config["dim"]                # C  (input / output channel size)
    hidden_dim = config["hidden_dim"]  # H  (inner hidden dimension)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape helpers
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape          # [B, N, N, C]
    total_rows = B * N * N                    # flatten (B,N,N)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (input -> fp16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, C] fp32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    # Choose reasonable tile sizes (covers dim up to 384)
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gating (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five linear weight matrices (each [H, C]) → [5H, C]
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()   # (5*H, C)

    # Perform one batched GEMM: [R, C] @ [C, 5H]^T → [R, 5H] (fp16)
    proj_all = F.linear(x_norm_fp16, concat_weight)  # fp16

    # Split back into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each [R, H]

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates & element‑wise gating (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16 = left_proj * left_gate      # [R, H]
    right_fp16 = right_proj * right_gate   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Reshape for the triangular update
    # ------------------------------------------------------------------
    # Shape back to [B, N, N, H]
    left = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Permute so that the reduction axis (k) is the innermost dimension
    # left_perm : [B, H, i, k]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right_perm: [B, H, k, j]  (swap the two N dimensions)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # [B*H, i, k]
    right_mat = right_perm.view(BH, N, N)  # [B*H, k, j]

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update via batched GEMM (fp16)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, i, j] fp16

    # Reshape back to 4‑D tensor [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating with the output gate + final linear projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # fp16 element‑wise

    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16)

    gated_flat = gated.view(total_rows, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # [R, C] fp16, no bias

    # Cast back to float32 as required by the API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
