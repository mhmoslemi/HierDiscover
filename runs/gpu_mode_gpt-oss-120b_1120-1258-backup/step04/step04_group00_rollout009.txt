"""
custom_kernel: optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ First LayerNorm (FP32 mean/var, FP16 output) – pure Triton kernel.
2️⃣ Fuse the four linear projections (left_proj, left_gate, right_proj,
   right_gate) and the out_gate linear into **one** GEMM (rows × dim  × 5·H).
   This removes two kernel launches and reduces intermediate memory traffic.
3️⃣ Apply the optional pair‑wise mask (broadcast over the hidden dimension)
   and the sigmoid‑gates (still in FP16).
4️⃣ Tri‑multiplicative update:
       out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
   performed as a batched GEMM (B·H matrices of size N×N) with torch.bmm.
5️⃣ Second LayerNorm (on the hidden dimension) – Triton kernel (FP32 stats,
   FP16 output).
6️⃣ Apply the output‑gate (sigmoid) and the final linear projection
   (hidden_dim → dim) with a cuBLAS FP16 GEMM.
7️⃣ Cast the result back to FP32.

Only the two LayerNorms are written in Triton; all dense GEMMs use the
high‑throughput cuBLAS kernels on the H100.  The fused projection reduces
launch overhead, and everything runs in FP16 except the statistics of the
LayerNorms, giving a large speed‑up over the reference implementation while
preserving numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 reduction, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const half*   [total_rows, dim]   (input)
    y_ptr,          #       half*   [total_rows, dim]   (output)
    w_ptr,          # const float* [dim]               (γ)
    b_ptr,          # const float* [dim]               (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0,
                       dtype=tl.float16).to(tl.float32)    # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)               # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values (FP16)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0,
                       dtype=tl.float16).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β (float32)
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * gamma[None, :] + beta[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 stats → FP16 output) – Triton
    # ------------------------------------------------------------------
    total_rows = B * N * N
    # flatten to 2‑D, keep float32 for accurate stats
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    # allocate FP16 buffer for the normalized tensor
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program (empirically good)
    BLOCK_D = 128          # cover dim up to 384 in ≤3 iterations
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],            # FP32
        weights["norm.bias"],              # FP32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # reshape back to 4‑D tensor (still FP16)
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all projections (left/right + gates) + out_gate into one GEMM
    # ------------------------------------------------------------------
    # concatenate weights: [left_proj; left_gate; right_proj; right_gate; out_gate]
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)          # shape (5*H, dim)

    rows = total_rows
    # Linear: (rows, dim) @ (5*H, dim).T → (rows, 5*H)
    combined = F.linear(x_norm_h.view(rows, dim), combined_weight)   # FP16

    # split the concatenated output
    left_proj_h      = combined[:, 0:H]                # (rows, H)
    left_gate_pre_h  = combined[:, H:2*H]              # (rows, H)
    right_proj_h     = combined[:, 2*H:3*H]            # (rows, H)
    right_gate_pre_h = combined[:, 3*H:4*H]            # (rows, H)
    out_gate_pre_h   = combined[:, 4*H:5*H]            # (rows, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    out_gate_h   = torch.sigmoid(out_gate_pre_h)

    left_h  = left_proj_h * left_gate_h   # (rows, H)
    right_h = right_proj_h * right_gate_h # (rows, H)

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # bring H into the batch axis → [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * H                                 # effective batch size
    left_mat  = left_perm.view(BH, N, N)       # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)      # (BH, N, N)

    # Core operation: Σ_k left[i,k] * right[j,k]   → (N×N) @ (N×N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N)

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (on hidden dimension) – Triton
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],   # FP32
        weights["to_out_norm.bias"],     # FP32
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm_h.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection and cast back to FP32
    # ------------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, H)   # FP16
    out_norm = out_norm * out_gate            # gated, still FP16

    # final linear: hidden_dim → dim
    final_h = F.linear(
        out_norm.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True),
    )   # (rows_h, dim) FP16

    # cast to FP32 as required by the original module
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
```
