The TriMul operation:

Pseudo-code:

- Input x shape (B,N,N,dim)
- LayerNorm on last dim (dim).
- Linear projections: left_proj = x_norm @ left_proj_weight^T (dim->hidden) ; same for right_proj
- Gates: left_gate = sigmoid(x_norm @ left_gate_weight^T) ; right_gate = sigmoid(x_norm @ right_gate_weight^T) ; out_gate = sigmoid(x_norm @ out_gate_weight^T)
- Mask applied to left_proj and right_proj before gating: left_proj = left_proj * mask (broadcast)
- left = left_proj * left_gate; right = right_proj * right_gate
- TriMul: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d], which is effectively for each batch b and channel d: left[b,:, :, d] @ right[b,:, :, d]^T   (where left is (N,N,hidden) as [i, k, d]; right is (N,N,hidden) as [j,k,d]; we compute for each d a product over k). That's effectively per batch-channel: (N,N) @ (N,N)^T gives (N,N) with O(N^3). But can implement using batched matrix multiplication: left_perm = left.permute(0,3,1,2) shape (B, hidden, N, N), right_perm = right.permute(0,3,1,2); out = torch.bmm(left_perm_reshape, right_perm_reshape.transpose(-2,-1)).
- Then to_out_norm: layernorm over hidden dim (the intermediate hidden_dim)
- Multiply by out_gate (which matches hidden dim)
- final linear: (B,N,N,hidden) -> from hidden to dim using to_out weight.

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)    # combined batch*hidden index
    pid_m = tl.program_id(0)    # i tile
    pid_n = tl.program_id(1)    # j tile

    # decode batch and hidden
    bh = pid_z
    b = bh // H
    h = bh % H

    # offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + b * stride_bl + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl + h * stride_hl
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile: shape (BLOCK_K, BLOCK_N) transposed
        b_ptrs = right_ptr + b * stride_br + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_ir + h * stride_hr

        b_tile = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b_tile)

    # write accumulator to output
    c_ptrs = out_ptr + b * stride_bo + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo + h * stride_ho
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

Algorithm
---------
1. Row‑wise LayerNorm on the input tensor (shape [B,N,N,D]) – custom Triton
   kernel writes the normalized result directly in fp16.

2. Fuse the five dense layers (left_proj, right_proj and three gates) into a
   single half‑precision GEMM by concatenating their weight matrices.
   The result is sliced into left/right projections and gate pre‑activations,
   followed by a sigmoid.

3. Optional mask (broadcast over the hidden dimension) is applied to the
   left/right projections before gating.

4. Triangular multiplicative update:
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication (A @ Bᵀ) for every hidden
   channel using a custom Triton kernel.  The kernel works on the original
   layout [B,N,N,H] – no costly permute+contiguous copy is required.

5. Second LayerNorm over the hidden dimension (fp16).

6. Apply the output gate and a final half‑precision linear projection back to
   the original dimension.  The final tensor is cast to float32.

The heaviest work (LayerNorms, fused projection, and the O(N³) update) runs
in fp16, exploiting Tensor‑Core throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (fp32 stats → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const *[total_rows, dim]   (fp32 input)
    y_ptr,          # mutable *[total_rows, dim] (fp16 output)
    weight_ptr,     # const *[dim]               (fp32 γ)
    bias_ptr,       # const *[dim]               (fp32 β)
    total_rows,     # int (runtime)
    dim,            # int (compile‑time)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program (tile over dim)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1st pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # ---------- 2nd pass: write normalized result ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) Batched triangular multiplication (A @ Bᵀ) – custom Triton GEMM
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,   # const *[B, N, N, H]  (fp16)
    right_ptr,  # const *[B, N, N, H]  (fp16)
    out_ptr,    # mutable *[B, N, N, H] (fp16)
    B, N, H,    # runtime dimensions
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_ir, stride_kr, stride_hr,   # right strides
    stride_bo, stride_io, stride_jo, stride_ho,   # out strides
    BLOCK_M: tl.constexpr,   # tile size over i
    BLOCK_N: tl.constexpr,   # tile size over j
    BLOCK_K: tl.constexpr,   # tile size over k
):
    # program IDs: (i_tile, j_tile, batch*hidden)
    pid_i = tl.program_id(0)          # tile over output rows   (i)
    pid_j = tl.program_id(1)          # tile over output cols   (j)
    pid_z = tl.program_id(2)          # combined batch & hidden index

    # decode batch index b and hidden index h from pid_z
    bh = pid_z
    b = bh // H
    h = bh % H

    # ------------------- tile offsets -------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile: shape (BLOCK_M, BLOCK_K) ----
        a_ptrs = (left_ptr
                  + b * stride_bl
                  + offs_i[:, None] * stride_il
                  + offs_k[None, :] * stride_kl
                  + h * stride_hl)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- right tile (transposed): shape (BLOCK_K, BLOCK_N) ----
        # we load right as (k, j) so that tl.dot(a, b) performs A @ Bᵀ
        b_ptrs = (right_ptr
                  + b * stride_br
                  + offs_k[:, None] * stride_kr
                  + offs_j[None, :] * stride_ir
                  + h * stride_hr)
        b_tile = tl.load(b_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b_tile)      # (M,K) @ (K,N) -> (M,N)

    # ------------------- write result -------------------
    c_ptrs = (out_ptr
              + b * stride_bo
              + offs_i[:, None] * stride_io
              + offs_j[None, :] * stride_jo
              + h * stride_ho)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1) first LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2) fused projection + three gates (5 * hidden_dim)
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, D]
            weights["right_proj.weight"],  # [H, D]
            weights["left_gate.weight"],   # [H, D]
            weights["right_gate.weight"],  # [H, D]
            weights["out_gate.weight"],    # [H, D]
        ],
        dim=0,
    ).to(torch.float16)                     # fp16 for Tensor‑Core speed

    # fused linear (R x 5H)
    out_concat = F.linear(x_norm, w_concat)      # fp16

    H = hidden_dim
    left_proj   = out_concat[:, 0:H]
    right_proj  = out_concat[:, H:2 * H]
    left_gate_p = out_concat[:, 2 * H:3 * H]
    right_gate_p = out_concat[:, 3 * H:4 * H]
    out_gate_p   = out_concat[:, 4 * H:5 * H]

    # sigmoid gates (still fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # ------------------------------------------------------------------
    # 3) optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj   = left_proj * mask_h
        right_proj  = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4) apply gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate          # [R, H]
    right = right_proj * right_gate        # [R, H]

    # reshape to [B, N, N, H] – layout (b,i,k,h)
    left_view  = left.view(B, N, N, H)
    right_view = right.view(B, N, N, H)

    # allocate output for the triangular multiplication
    out_view = torch.empty_like(left_view)   # fp16

    # strides (in element count, not bytes)
    stride_bl, stride_il, stride_kl, stride_hl = left_view.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right_view.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_view.stride()

    # ------------------------------------------------------------------
    # 4) triangular update: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid_mul = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H,                     # batch × hidden (each runs its own GEMM)
    )
    trimul_outgoing_kernel[grid_mul](
        left_view,
        right_view,
        out_view,
        B,
        N,
        H,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5) second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out_flat = out_view.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # reshape back to [B, N, N, H]
    out_norm = out_norm.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 6) apply output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 7) final projection back to original dimension
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.view(total_rows, H)
    final_half = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16)   # [D, H]
    )                                                # fp16, shape [R, D]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
