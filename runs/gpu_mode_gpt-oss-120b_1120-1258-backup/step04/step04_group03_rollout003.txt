"""
Highly‑optimized TriMul (outgoing) forward pass.

Algorithm (FP16‑centric)
------------------------
1️⃣ First LayerNorm over the last channel (dim).  The reduction is done in FP32
   but the normalized output is written as FP16 by a custom Triton kernel.
2️⃣ Fuse the five linear projections (left, right and three gates) into a single
   GEMM by concatenating the weight matrices.  All arithmetic stays in FP16,
   which lets cuBLAS/Tensor‑Cores take over.
3️⃣ Optional pair‑wise mask is applied after the projections (before the gates).
4️⃣ Apply sigmoid to the three gate tensors and perform element‑wise gating.
5️⃣ Triangular multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realized as a batched matrix‑multiply on tensors shaped (B·H,N,N) using
   `torch.matmul` (cuBLAS), i.e. a dense N×N × N×N GEMM for each head.
6️⃣ Second LayerNorm over the hidden dimension (again via Triton, FP16 output).
7️⃣ Multiply by the output gate and project back to the original `dim` with a
   final FP16 linear layer.  The result is cast to FP32 as required by the API.

All heavy reductions (the two LayerNorms) are performed by a bespoke Triton
kernel; dense linear algebra (PROJ + GEMM) uses cuBLAS/Tensor‑Core for maximal
throughput on H100.  The implementation works for any supported batch‑size,
sequence‑length and channel dimensions, with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm that reads FP32, reduces in FP32, writes FP16
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (input FP32)
    y_ptr,          # *mut   half*    [total_rows, dim]   (output FP16)
    weight_ptr,     # *const float*   [dim]               (γ)
    bias_ptr,       # *const float*   [dim]               (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1 – compute row‑wise mean and variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of x (fp32)
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2 – write normalized + affine transformation (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # fp32 raw values

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ, fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β, fp32

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – returns a float32 tensor of shape
    [bs, seq_len, seq_len, dim].

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, [B, N, N, C], float32
        * mask_tensor  : torch.Tensor, [B, N, N] (bool/float) or None
        * weights_dict : mapping of all layer‑norm and linear weights (float32)
        * config_dict  : "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor, dtype float32, same device as the inputs.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack & constants
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape          # [B, N, N, C]
    total_rows = B * N * N                    # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    # Heuristic block sizes (tuned for H100)
    BLOCK_M = 256          # rows per program
    BLOCK_D = 64 if dim > 128 else 32   # channel tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections (left, right, left_gate,
    #     right_gate, out_gate) into a single GEMM.
    # ------------------------------------------------------------------
    # Concatenate weight matrices: shape (5*H, C)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device).contiguous()  # (5*H, C)

    # Linear projection in FP16
    proj_all = F.linear(x_norm_fp16, concat_w)  # (total_rows, 5*H)

    # Split back into the five tensors (all FP16)
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B, N, N] → [total_rows, 1] in half
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates + element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # (total_rows, H)
    right_gate = torch.sigmoid(right_gate) # (total_rows, H)
    out_gate = torch.sigmoid(out_gate)     # (total_rows, H)

    left_fp16  = left_proj * left_gate     # (total_rows, H)
    right_fp16 = right_proj * right_gate   # (total_rows, H)

    # ------------------------------------------------------------------
    # 3️⃣ Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    # Shape: (B, N, N, H)
    left  = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) where the summed‑over dimension k is the inner
    # dimension for the matrix multiply.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # View as a huge batch of (B*H) matrix multiplications.
    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)      # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)     # (BH, N, N)

    # cuBLAS/Tensor‑Core matmul (FP16)
    out_mat = torch.matmul(left_mat, right_mat)  # (BH, N, N), fp16

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (FP16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, N, N, H) fp16

    to_out_w_fp16 = weights["to_out.weight"].to(dtype=torch.float16, device=device)

    gated_flat = gated.view(total_rows_h, hidden_dim)   # (total_rows_h, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # (total_rows_h, C)

    # Cast to FP32 to satisfy the API contract
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
