@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)  # batch*hidden index

    # Compute b and h
    h = pid_hb % H
    b = pid_hb // H

    # Compute base pointer for this batch and hidden
    left_base = left_ptr + b * stride_b + h * stride_h
    right_base = right_ptr + b * stride_b + h * stride_h
    out_base = out_ptr + b * stride_b + h * stride_h

    # Offsets for i and j tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets within tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile from right (need transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + offs_j[None, :] * stride_i + offs_k[:, None] * stride_k
        b = tl.load(b_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0).to(tl.float32)

        # Compute dot product for this k block
        acc += tl.dot(a, b)   # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write result
    out_ptrs = out_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_i  # Wait we need correct stride for out j dimension.

"""
TriMul (outgoing) – Triton‑accelerated implementation.

Algorithm
---------
1. **LayerNorm (first)** – use native PyTorch `layer_norm` (FP16) which is
   heavily optimized in cudnn on H100.
2. **Fused projection + gates** – concatenate the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) and apply a
   single half‑precision linear layer.
3. **Mask & sigmoid gates** – optional mask is applied before gating,
   gates are computed with `torch.sigmoid` (FP16).
4. **Triangular multiplicative update** – custom Triton kernel computes
   for every batch‑hidden pair the matrix product

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   i.e. a batched GEMM where the hidden dimension is treated as an extra
   batch dimension.  The kernel works on the original `[B,N,N,H]`
   layout, thus avoiding the costly `permute/contiguous` copies of the
   reference implementation.
5. **LayerNorm (second)** – again use native `layer_norm` on the hidden
   dimension.
6. **Output gate + final projection** – element‑wise gate, a final half‑
   precision linear layer back to `dim`, and cast the result to `float32`.

All heavy arithmetic runs in FP16, while accumulators are FP32 to keep
numerical stability.  Only the O(N³) triangular update is written in
custom Triton code; everything else uses highly‑optimized PyTorch
operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the triangular multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                         # const *[B, N, N, H]  (fp16)
    right_ptr,                        # const *[B, N, N, H]  (fp16)
    out_ptr,                          # mutable *[B, N, N, H] (fp16)
    B, N, H,                          # runtime constants
    stride_b, stride_i, stride_k, stride_h,   # strides of the tensors
    BLOCK_M: tl.constexpr,            # tile size for i dimension
    BLOCK_N: tl.constexpr,            # tile size for j dimension
    BLOCK_K: tl.constexpr,            # tile size for reduction (k)
):
    """Batched GEMM on the layout [B, N, N, H] (i, k) × (j, k)."""
    pid_i = tl.program_id(0)          # tile over the i‑axis
    pid_j = tl.program_id(1)          # tile over the j‑axis
    pid_hb = tl.program_id(2)         # combined batch & hidden index

    # ------------------------------------------------------------------
    # Decode batch (b) and hidden channel (h) from pid_hb
    # ------------------------------------------------------------------
    h = pid_hb % H
    b = pid_hb // H

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_b + h * stride_h
    right_base = right_ptr + b * stride_b + h * stride_h
    out_base   = out_ptr   + b * stride_b + h * stride_h

    # ------------------------------------------------------------------
    # Offsets for the current i / j tile
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  : [BLOCK_M, BLOCK_K]  (i, k)
        a_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                # fp16

        # right tile : [BLOCK_K, BLOCK_N]  (k, j)
        b_ptrs = right_base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                # fp16

        # GEMM of the two sub‑tiles
        acc += tl.dot(a, b)                # fp32 accumulation

    # ------------------------------------------------------------------
    # Store the result back (still fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    out_fp16 = acc.to(tl.float16)
    tl.store(out_ptrs,
             out_fp16,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Triton‑based TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping str → torch.Tensor (float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1) First LayerNorm – native PyTorch (fast, cuDNN)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=eps).half()
    x_norm_flat = x_norm.view(total_rows, dim)    # [R, dim]

    # ------------------------------------------------------------------
    # 2) Fuse the five linear layers (proj + three gates) into one GEMM
    # ------------------------------------------------------------------
    # concatenate weights: [5*hidden, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16

    # Single half‑precision linear layer
    proj = F.linear(x_norm_flat, w_concat)   # [R, 5*hidden] (fp16)

    H = hidden
    left_proj   = proj[:, 0:H]               # [R, H]
    right_proj  = proj[:, H:2*H]
    left_gate_p = proj[:, 2*H:3*H]
    right_gate_p= proj[:, 3*H:4*H]
    out_gate_p  = proj[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask: [B, N, N] → flatten → [R, 1]
        mask_h = mask.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4) Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # ------------------------------------------------------------------
    # 5) Gated projected tensors and reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    left  = (left_proj * left_gate).view(B, N, N, H)
    right = (right_proj * right_gate).view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)      # used after the second LN

    # ------------------------------------------------------------------
    # 6) Triangular multiplicative update – custom Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)               # fp16, shape [B,N,N,H]

    # strides of the tensors (all share the same layout)
    stride_b, stride_i, stride_k, stride_h = left.stride()

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),              # i‑tiles
        triton.cdiv(N, BLOCK_N),              # j‑tiles
        B * H,                                 # combined batch‑hidden dimension
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 8) Output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                         # element‑wise fp16
    final = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].t().to(torch.float16)     # [dim, hidden] → transposed inside F.linear
    )                                                       # [R, dim] fp16

    # Cast to float32 and reshape to original 4‑D layout
    final = final.view(B, N, N, dim).to(torch.float32)
    return final
```
