"""
Optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Custom Triton LayerNorm (mean/var in fp32, output fp16).
2️⃣  Fuse the five linear layers (left/right projections + 3 gates) by
    concatenating their weight matrices and performing one half‑precision GEMM.
3️⃣  Apply the optional binary mask only to the two projections (left/right).
4️⃣  Compute sigmoid gates and element‑wise gating.
5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    performed with a single `torch.einsum` which is compiled to a
    high‑performance Triton kernel (no costly permute+contiguous copies).
6️⃣  Second LayerNorm (same Triton kernel as step 1).
7️⃣  Multiply by the output gate and project back to `dim` with a final
    half‑precision linear layer.  The result is cast to float32 as required.

Only the two LayerNorms are written in Triton; all dense linear algebra
uses cuBLAS/Tensor‑Cores via regular PyTorch ops.  This reduces memory
traffic (no extra permute copies) and stays well‑within the H100 memory
budget while keeping the overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm (fp16 output, fp32 reduction)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut   half*    [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const float*   [dim]              (γ)
    bias_ptr,       # *const float*   [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1 – row‑wise mean & variance (fp32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of x (fp32)
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2 – write normalized + affine (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # fp32 raw input

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β (fp32)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : dict of layer‑norm and linear weight tensors (float32)
        * config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C], dtype float32, on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # C
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape helpers
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                # [B, N, N, C]
    total_rows = B * N * N                         # rows for flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim),
                               dtype=torch.float16,
                               device=device)

    # Tune block sizes for the H100 (empirically good values)
    BLOCK_M = 256          # rows per program
    BLOCK_D = 64           # channel tile size
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + three gates (still FP16)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape (5*H, C)) and cast to half
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                      # (5*H, C)

    # Single GEMM: (total_rows, C) @ (5*H, C).T → (total_rows, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)   # fp16, no bias

    # Split back into the five tensors, each (total_rows, H)
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 2a️⃣ Optional binary mask (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating of the projections
    left_fp16 = left_proj * left_gate          # (total_rows, H)
    right_fp16 = right_proj * right_gate       # (total_rows, H)

    # Reshape back to 4‑D tensors required by the triangular update
    left = left_fp16.view(B, N, N, hidden_dim)     # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched contraction)
    # ------------------------------------------------------------------
    # Einsum contracts over the shared index k:
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # PyTorch compiles this pattern to a highly‑optimized Triton kernel,
    # avoiding the expensive permute+contiguous copies used previously.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N                     # B*N*N rows for hidden dim
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, N, N, H) fp16

    # Final linear projection H → C (still fp16)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)    # (B*N*N, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)    # (B*N*N, C)

    # Cast back to float32 for the public API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
