import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm + five linear projections (left/right + three gates)
# --------------------------------------------------------------------------
@triton.jit
def _ln_proj_gate_kernel(
    # ------------ inputs ----------------------------------------------------
    x_ptr,               # const   *  [R, D]   (fp32)
    mask_ptr,            # const   *  [R]      (fp16) – 1.0 for no mask
    # ------------ outputs ---------------------------------------------------
    left_ptr,            # mutable *  [R, H]   (fp16)
    right_ptr,           # mutable *  [R, H]   (fp16)
    out_gate_ptr,        # mutable *  [R, H]   (fp16)
    # ------------ LayerNorm parameters --------------------------------------
    ln_weight_ptr,       # const   *  [D]      (fp32)
    ln_bias_ptr,         # const   *  [D]      (fp32)
    # ------------ projection / gate weight matrices (transposed) ------------
    w_lp_ptr,            # const   *  [D, H]   (fp16) left_proj.T
    w_rp_ptr,            # const   *  [D, H]   (fp16) right_proj.T
    w_lg_ptr,            # const   *  [D, H]   (fp16) left_gate.T
    w_rg_ptr,            # const   *  [D, H]   (fp16) right_gate.T
    w_og_ptr,            # const   *  [D, H]   (fp16) out_gate.T
    # ------------ sizes ----------------------------------------------------
    total_rows: tl.int32,
    dim: tl.constexpr,            # model dimension (e.g. 128 or 384)
    hidden: tl.constexpr,         # hidden dimension (always 128 in the tests)
    eps: tl.constexpr,
    # ------------ tiling ----------------------------------------------------
    BLOCK_M: tl.constexpr,        # rows per program
    BLOCK_D: tl.constexpr,        # dim‑wise tile
    BLOCK_H: tl.constexpr,        # hidden‑wise tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------------------------------------------------------------------
    # First pass – row‑wise mean / variance
    # ----------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # (BLOCK_M, BLOCK_D)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ----------------------------------------------------------------------
    # Second pass – joint LayerNorm + all five linear layers
    # ----------------------------------------------------------------------
    # Loop over hidden dimension tiles
    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < hidden

        # output accumulators for the current hidden tile
        acc_lp = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)   # left proj
        acc_lg = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)   # left gate
        acc_rp = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)   # right proj
        acc_rg = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)   # right gate
        acc_og = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)   # out   gate

        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
            mask_d = offs_d < dim

            # ---- load raw input ------------------------------------------------
            x_raw = tl.load(
                x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                               # (BLOCK_M, BLOCK_D)

            # ---- LayerNorm (affine) -------------------------------------------
            ln_w = tl.load(
                ln_weight_ptr + offs_d,
                mask=mask_d,
                other=0.0,
            ).to(tl.float32)                               # (BLOCK_D)
            ln_b = tl.load(
                ln_bias_ptr + offs_d,
                mask=mask_d,
                other=0.0,
            ).to(tl.float32)                               # (BLOCK_D)

            norm = ((x_raw - mean[:, None]) * rstd[:, None]) * ln_w[None, :] + ln_b[None, :]

            # ---- load weight tiles (already transposed: [D, H]) ---------------
            wp_lp = tl.load(
                w_lp_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)                               # (BLOCK_D, BLOCK_H)
            wp_rp = tl.load(
                w_rp_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            wp_lg = tl.load(
                w_lg_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            wp_rg = tl.load(
                w_rg_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            wp_og = tl.load(
                w_og_ptr + offs_d[:, None] * hidden + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # ---- GEMM steps ----------------------------------------------------
            acc_lp += tl.dot(norm, wp_lp)   # (BLOCK_M, BLOCK_H)
            acc_rp += tl.dot(norm, wp_rp)
            acc_lg += tl.dot(norm, wp_lg)
            acc_rg += tl.dot(norm, wp_rg)
            acc_og += tl.dot(norm, wp_og)

        # ---- apply sigmoid gates ---------------------------------------------
        gate_l = 1.0 / (1.0 + tl.exp(-acc_lg))
        gate_r = 1.0 / (1.0 + tl.exp(-acc_rg))
        gate_o = 1.0 / (1.0 + tl.exp(-acc_og))

        # ---- optional pairwise mask (already fp16) ----------------------------
        mask_val = tl.load(
            mask_ptr + offs_m,
            mask=mask_m,
            other=1.0,
        ).to(tl.float32)[:, None]          # (BLOCK_M, 1)

        left_tile  = (acc_lp * gate_l) * mask_val
        right_tile = (acc_rp * gate_r) * mask_val
        out_g_tile = gate_o                     # no mask on out_gate

        # ---- store results (fp16) --------------------------------------------
        tl.store(
            left_ptr + offs_m[:, None] * hidden + offs_h[None, :],
            left_tile.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + offs_m[:, None] * hidden + offs_h[None, :],
            right_tile.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
            out_g_tile.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )


# --------------------------------------------------------------------------
# 2️⃣  Second LayerNorm + out‑gate + final linear (all fused)
# --------------------------------------------------------------------------
@triton.jit
def _ln2_gate_linear_kernel(
    # ------------ inputs ----------------------------------------------------
    x_ptr,                # const * [R, H]   (fp16) – result of TriMul
    gate_ptr,             # const * [R, H]   (fp16) – out_gate (sigmoid)
    # ------------ LayerNorm parameters ------------------------------------
    ln_weight_ptr,        # const * [H]      (fp32)
    ln_bias_ptr,          # const * [H]      (fp32)
    # ------------ final linear weight (already transposed) ----------------
    w_out_ptr,            # const * [H, D]   (fp16) – to_out.weight.T
    # ------------ output ---------------------------------------------------
    out_ptr,              # mutable * [R, D] (fp32)
    # ------------ sizes ----------------------------------------------------
    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    # ------------ tiling ---------------------------------------------------
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------------------------------------------------------------------
    # Compute row‑wise mean / variance over the hidden dimension
    # ----------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < hidden

        x = tl.load(
            x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)                               # (BLOCK_M, BLOCK_H)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var + eps)

    # ----------------------------------------------------------------------
    # Second pass – LayerNorm (affine) + gate + final linear
    # ----------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for h in range(0, tl.cdiv(hidden, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
            mask_h = offs_h < hidden

            # load hidden activation and its gate
            x = tl.load(
                x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)                               # (BLOCK_M, BLOCK_H)

            g = tl.load(
                gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)                               # (BLOCK_M, BLOCK_H)

            # --- LayerNorm (affine) -----------------------------------------
            ln_w = tl.load(
                ln_weight_ptr + offs_h,
                mask=mask_h,
                other=0.0,
            ).to(tl.float32)                               # (BLOCK_H)
            ln_b = tl.load(
                ln_bias_ptr + offs_h,
                mask=mask_h,
                other=0.0,
            ).to(tl.float32)                               # (BLOCK_H)

            x_norm = ((x - mean[:, None]) * rstd[:, None]) * ln_w[None, :] + ln_b[None, :]
            x_scaled = x_norm * g                            # gate multiplication

            # --- final linear weight tile (H × D) ----------------------------
            w = tl.load(
                w_out_ptr + offs_h[:, None] * dim + offs_d[None, :],
                mask=mask_h[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                               # (BLOCK_H, BLOCK_D)

            acc += tl.dot(x_scaled, w)                     # (BLOCK_M, BLOCK_D)

        # store the result (fp32)
        tl.store(
            out_ptr + offs_m[:, None] * dim + offs_d[None, :],
            acc,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# --------------------------------------------------------------------------
# 3️⃣  Public entry‑point – the outgoing TriMul forward pass
# --------------------------------------------------------------------------
def custom_kernel(data):
    """
    Fast Triton implementation of the outgoing TriMul operator (AlphaFold‑3).

    The kernel fuses:
        • First LayerNorm over the [B·N·N, dim] tensor,
        • Five dense layers (two projections + three gates),
        • Optional pairwise mask,
        • Second LayerNorm, out‑gate multiplication,
        • Final linear projection back to the original dimension.

    All heavy tensor‑wise operations run in fp16 on Tensor‑Core friendly kernels.
    The result is returned as FP32.
    """
    # ----------------------------------------------------------------------
    # Unpack arguments
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # model dimension (e.g. 128 or 384)
    hidden = config["hidden_dim"]           # hidden dimension (always 128 in tests)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                  # R = B·N·N

    # ----------------------------------------------------------------------
    # Prepare tensors
    # ----------------------------------------------------------------------
    # flatten input to [R, D]
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    # mask: 1.0 everywhere → no effect; otherwise use the supplied mask
    if config.get("nomask", True):
        mask_flat = torch.ones(total_rows, dtype=torch.float16, device=device)
    else:
        mask_flat = mask_tensor.to(torch.float16).view(total_rows)

    # pre‑transpose all weight matrices required by the fused kernels
    w_lp_T = weights["left_proj.weight"].T.contiguous().to(torch.float16)      # [D, H]
    w_rp_T = weights["right_proj.weight"].T.contiguous().to(torch.float16)     # [D, H]
    w_lg_T = weights["left_gate.weight"].T.contiguous().to(torch.float16)      # [D, H]
    w_rg_T = weights["right_gate.weight"].T.contiguous().to(torch.float16)     # [D, H]
    w_og_T = weights["out_gate.weight"].T.contiguous().to(torch.float16)       # [D, H]

    # output buffers for the first fused kernel
    left  = torch.empty(total_rows, hidden, dtype=torch.float16, device=device)
    right = torch.empty(total_rows, hidden, dtype=torch.float16, device=device)
    out_gate = torch.empty(total_rows, hidden, dtype=torch.float16, device=device)

    # ----------------------------------------------------------------------
    # 1️⃣  Fuse LayerNorm + projections + gates + optional mask
    # ----------------------------------------------------------------------
    BLOCK_M = 32          # rows per program (tuned for H100 register budget)
    BLOCK_D = 32          # dim‑wise tile
    BLOCK_H = 32          # hidden‑wise tile

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    _ln_proj_gate_kernel[grid](
        x_ptr=x_flat,
        mask_ptr=mask_flat,
        left_ptr=left,
        right_ptr=right,
        out_gate_ptr=out_gate,
        ln_weight_ptr=weights["norm.weight"],
        ln_bias_ptr=weights["norm.bias"],
        w_lp_ptr=w_lp_T,
        w_rp_ptr=w_rp_T,
        w_lg_ptr=w_lg_T,
        w_rg_ptr=w_rg_T,
        w_og_ptr=w_og_T,
        total_rows=total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
    )

    # reshape to the 4‑D form expected by the triangular update
    left  = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)
    out_gate = out_gate.view(B, N, N, hidden)

    # ----------------------------------------------------------------------
    # 2️⃣  Triangular multiplicative update (batched matmul)
    # ----------------------------------------------------------------------
    # bring hidden dimension to the batch axis – a copy is cheap compared to the
    # O(N³) GEMM that follows
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out = torch.matmul(
        left_perm,
        right_perm.transpose(-1, -2)
    )
    # back to [B, N, N, H]
    out = out.permute(0, 2, 3, 1).contiguous()
    out_flat = out.view(total_rows, hidden)                # [R, H]
    out_gate_flat = out_gate.view(total_rows, hidden)      # [R, H]

    # ----------------------------------------------------------------------
    # 3️⃣  Fuse second LayerNorm, out‑gate, and final linear projection
    # ----------------------------------------------------------------------
    # final linear weight: to_out.weight ∈ ℝ^{dim × hidden}
    # we need the transpose so that the inner dimension matches hidden
    w_out_T = weights["to_out.weight"].T.contiguous().to(torch.float16)  # [H, D]

    final = torch.empty(total_rows, dim, dtype=torch.float32, device=device)

    BLOCK_M2 = 32      # rows per program for the second stage
    BLOCK_H2 = 32      # hidden‑wise tile
    BLOCK_D2 = 32      # output‑dim tile

    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    _ln2_gate_linear_kernel[grid2](
        x_ptr=out_flat,
        gate_ptr=out_gate_flat,
        ln_weight_ptr=weights["to_out_norm.weight"],
        ln_bias_ptr=weights["to_out_norm.bias"],
        w_out_ptr=w_out_T,
        out_ptr=final,
        total_rows=total_rows,
        hidden=hidden,
        dim=dim,
        eps=eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_H=BLOCK_H2,
        BLOCK_D=BLOCK_D2,
    )

    # reshape to the original 4‑D layout
    final = final.view(B, N, N, dim)
    return final
