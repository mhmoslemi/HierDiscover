def final_fused_kernel(
    out_ptr, out_gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    to_out_weight_ptr,
    output_ptr,
    total_rows, hidden_dim, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # first pass: sum, sumsq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, hidden_dim, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim
        a = tl.load(out_ptr + (offs_m[:, None] * hidden_dim + offs_k[None, :]),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(a, axis=1)
        sum_sq += tl.sum(a * a, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute final output via accumulation over hidden_dim
    acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

    for k in range(0, hidden_dim, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        # load out and out_gate slices
        a = tl.load(out_ptr + (offs_m[:, None] * hidden_dim + offs_k[None, :]),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32) # shape [BLOCK_M, BLOCK_K]

        g = tl.load(out_gate_ptr + (offs_m[:, None] * hidden_dim + offs_k[None, :]),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32) # same shape

        # normalize a
        a_norm = (a - mean[:, None]) * rstd[:, None]   # shape [BLOCK_M, BLOCK_K]

        # apply layernorm weight and bias (gamma, beta)
        wnorm = tl.load(norm_weight_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)
        bn = tl.load(norm_bias_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)
        a_norm = a_norm * wnorm[None, :] + bn[None, :]  # broadcast

        # gating
        a_gated = a_norm * g

        # load to_out weight block
        for d in range(0, dim, BLOCK_D):
            offs_d = d + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            w_out = tl.load(to_out_weight_ptr + (offs_k[:, None] * dim + offs_d[None, :]),
                            mask=mask_k[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)  # shape [BLOCK_K, BLOCK_D]

            # accumulate
            acc += tl.dot(a_gated, w_out)  # shape [BLOCK_M, BLOCK_D]

def proj_gate_fused_kernel(
    proj_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows, hidden_dim,
    nomask: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_M
    offs_m = start + tl.arange(0, BLOCK_M)
    mask = offs_m < total_rows

    if nomask:
        mask_val = tl.full([BLOCK_M], 1.0, dtype=tl.float16)
    else:
        mask_val = tl.load(mask_ptr + offs_m, mask=mask, other=0.0)

    for h_start in range(0, hidden_dim, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        # compute ptr offsets for each slice
        # row stride = 5 * hidden_dim
        row_stride = hidden_dim * 5

        # left_proj slice
        left_proj = tl.load(proj_ptr + offs_m[:, None] * row_stride + offs_h[None, :],
                            mask=mask[:, None] & mask_h[None, :], other=0.0)

        # left_gate slice offset
        left_gate = tl.load(proj_ptr + offs_m[:, None] * row_stride + (hidden_dim + offs_h)[None, :],
                            mask=mask[:, None] & mask_h[None, :], other=0.0)

        # right_proj slice offset
        right_proj = tl.load(proj_ptr + offs_m[:, None] * row_stride + (2*hidden_dim + offs_h)[None, :],
                             mask=mask[:, None] & mask_h[None, :], other=0.0)

        # right_gate slice
        right_gate = tl.load(proj_ptr + offs_m[:, None] * row_stride + (3*hidden_dim + offs_h)[None, :],
                             mask=mask[:, None] & mask_h[None, :], other=0.0)

        # out_gate slice
        out_gate_raw = tl.load(proj_ptr + offs_m[:, None] * row_stride + (4*hidden_dim + offs_h)[None, :],
                               mask=mask[:, None] & mask_h[None, :], other=0.0)

        # apply sigmoid
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
        out_gate_sig = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # apply mask on left_proj and right_proj before gating
        if not nomask:
            # mask_val is broadcasted across BLOCK_M rows; M dims not to broadcast across H directly.
            left_val = left_proj * mask_val[:, None] * left_gate_sig
            right_val = right_proj * mask_val[:, None] * right_gate_sig
        else:
            left_val = left_proj * left_gate_sig
            right_val = right_proj * right_gate_sig

        # store outputs
        tl.store(left_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                 left_val, mask=mask[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                 right_val, mask=mask[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                 out_gate_sig, mask=mask[:, None] & mask_h[None, :])
```

"""
TriMul (outgoing) – fused Triton implementation
================================================
The forward pass consists of four stages

1️⃣ First LayerNorm over the last dimension (C) – custom Triton kernel.
2️⃣ One large GEMM (FP16) from the normalized tensor to the five
   projection / gate weight matrices, followed by a fused kernel that
   * applies the pairwise mask, * computes the three sigmoid gates,
   * multiplies left/right projections with their gates and writes the
   * three result tensors (left, right, out_gate) directly.
3️⃣ Triangular multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented with a single batched matrix‑multiply (torch.bmm) on
   tensors of shape (B·H, N, N).
4️⃣ Final fused kernel – per‑row LayerNorm over the hidden dimension,
   element‑wise output‑gate, and the final projection back to `dim`.  All
   three operations are performed in one pass, producing a Float32
   output tensor of shape [B, N, N, dim].

Only three custom kernels are used (two LayerNorms, one fused projection‑gate,
one fused final‑norm‑linear).  All heavy matrix multiplications are handled
by cuBLAS (FP16 Tensor‑Core friendly), while the kernels eliminate precious
intermediate memory allocations and reduce global‑memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ LayerNorm (FP16 output) – two‑pass implementation
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const   fp32*   [total_rows, dim]   (input)
    y_ptr,          # *mut     half*  [total_rows, dim]   (output)
    weight_ptr,     # *const   float* [dim]               (γ)
    bias_ptr,       # *const   float* [dim]               (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # compile‑time
    eps: tl.constexpr,          # compile‑time
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: compute mean / variance (fp32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptr = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fuse projection + gating + (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fused_kernel(
    proj_ptr,            # *const   half*  [total_rows, 5*H]
    mask_ptr,            # *const   half*  [total_rows] (unused when nomask)
    left_ptr,            # *mut     half*  [total_rows, H]
    right_ptr,           # *mut     half*  [total_rows, H]
    out_gate_ptr,        # *mut     half*  [total_rows, H]
    total_rows: tl.constexpr,
    hidden_dim: tl.constexpr,
    nomask: tl.constexpr,        # 1 → skip mask, 0 → use mask
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_row = offs_m < total_rows

    # mask broadcast (0/1) – only needed when nomask==0
    if nomask == 0:
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_row, other=0.0).to(tl.float16)
    else:
        mask_val = tl.full([BLOCK_M], 1.0, dtype=tl.float16)

    row_stride = hidden_dim * 5            # distance between rows in proj

    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        # ----- load the five slices -----
        base = offs_m[:, None] * row_stride

        left_proj   = tl.load(proj_ptr + base + offs_h[None, :],
                              mask=mask_row[:, None] & mask_h[None, :], other=0.0)
        left_gate   = tl.load(proj_ptr + base + (hidden_dim + offs_h)[None, :],
                              mask=mask_row[:, None] & mask_h[None, :], other=0.0)

        right_proj  = tl.load(proj_ptr + base + (2*hidden_dim + offs_h)[None, :],
                              mask=mask_row[:, None] & mask_h[None, :], other=0.0)
        right_gate  = tl.load(proj_ptr + base + (3*hidden_dim + offs_h)[None, :],
                              mask=mask_row[:, None] & mask_h[None, :], other=0.0)

        out_gate_raw = tl.load(proj_ptr + base + (4*hidden_dim + offs_h)[None, :],
                               mask=mask_row[:, None] & mask_h[None, :], other=0.0)

        # ----- sigmoid gates -----
        left_gate_sig  = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
        out_gate_sig   = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # ----- apply mask and gating -----
        left_val  = left_proj * mask_val[:, None] * left_gate_sig
        right_val = right_proj * mask_val[:, None] * right_gate_sig

        # ----- store results -----
        tl.store(left_ptr   + offs_m[:, None] * hidden_dim + offs_h[None, :],
                 left_val,  mask=mask_row[:, None] & mask_h[None, :])
        tl.store(right_ptr  + offs_m[:, None] * hidden_dim + offs_h[None, :],
                 right_val, mask=mask_row[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                 out_gate_sig, mask=mask_row[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Final fused kernel (LayerNorm → out‑gate → final Linear)
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,           # *const   half*   [total_rows, H]   (triangular result)
    out_gate_ptr,      # *const   half*   [total_rows, H]
    norm_weight_ptr,   # *const   float*  [H]               (to_out_norm.weight)
    norm_bias_ptr,     # *const   float*  [H]               (to_out_norm.bias)
    linear_weight_ptr, # *const   half*   [H, C]            (to_out.weight.T)
    out_ptr_fp32,      # *mut     float*  [total_rows, C]
    total_rows: tl.constexpr,
    hidden_dim: tl.constexpr,
    out_dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1) load the entire hidden‑dim vector (H ≤ 128) for each row
    # --------------------------------------------------------------
    K = hidden_dim
    offs_k = tl.arange(0, K)
    mask_k = offs_k < K

    # raw tensors (fp16 → fp32)
    out_fp16 = tl.load(out_ptr + offs_m[:, None] * K + offs_k[None, :],
                       mask=mask_m[:, None] & mask_k[None, :], other=0.0)
    out_gate_fp16 = tl.load(out_gate_ptr + offs_m[:, None] * K + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :], other=0.0)

    out_fp32 = out_fp16.to(tl.float32)          # [M, K]
    out_gate_fp32 = out_gate_fp16.to(tl.float32)

    # --------------------------------------------------------------
    # 2) per‑row LayerNorm over H
    # --------------------------------------------------------------
    sum_val = tl.sum(out_fp32, axis=1)          # [M]
    sum_sq  = tl.sum(out_fp32 * out_fp32, axis=1)
    mean = sum_val / K
    var  = sum_sq / K - mean * mean
    rstd = tl.rsqrt(var + eps)

    # load LN weight/bias (γ,β) – shape = H
    w_norm = tl.load(norm_weight_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)   # [K]
    b_norm = tl.load(norm_bias_ptr   + offs_k, mask=mask_k, other=0.0).to(tl.float32)   # [K]

    # normalized tensor (M,K)
    normed = (out_fp32 - mean[:, None]) * rstd[:, None]   # (x-μ)/σ
    normed = normed * w_norm[None, :] + b_norm[None, :]   # apply γ,β

    # gating
    gated = normed * out_gate_fp32                         # [M, K]

    # --------------------------------------------------------------
    # 3) final linear projection: gated @ W_out   (W_out : [H, C])
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(out_dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < out_dim

        # weight slice [K, BLOCK_D]
        w = tl.load(linear_weight_ptr + offs_k[:, None] * out_dim + offs_d[None, :],
                    mask=mask_k[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # acc = gated (M,K) @ w (K,DB) → (M,DB)
        acc = tl.dot(gated, w)   # fp32 accumulator

        # store result
        out_ptrs = out_ptr_fp32 + offs_m[:, None] * out_dim + offs_d[None, :]
        tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Arguments
    ---------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, C] (float32)
        - mask_tensor  : [B, N, N]    (bool/float, optional)
        - weights_dict : model weights
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # number of (b,i,k) triples

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 output)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()  # [R, C]  (fp32)
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Concatenated linear projection (FP16) → [R, 5·H]
    # --------------------------------------------------------------
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5·H, C)

    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16, shape [R,5·H]

    # --------------------------------------------------------------
    # 2a️⃣ Fuse mask, sigmoid gates and produce left/right/out_gate
    # --------------------------------------------------------------
    left_buf      = torch.empty((total_rows, hidden_dim), dtype=torch.float16, device=device)
    right_buf     = torch.empty((total_rows, hidden_dim), dtype=torch.float16, device=device)
    out_gate_buf  = torch.empty((total_rows, hidden_dim), dtype=torch.float16, device=device)

    if nomask:
        dummy_mask = torch.empty((0,), dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_flat = mask_tensor.view(total_rows).to(torch.float16)
        mask_ptr = mask_flat

    # launch fused projection+gate kernel
    BLOCK_H = 64
    grid_gate = (triton.cdiv(total_rows, BLOCK_M),)
    proj_gate_fused_kernel[grid_gate](
        proj_all,
        mask_ptr,
        left_buf,
        right_buf,
        out_gate_buf,
        total_rows,
        hidden_dim,
        nomask=int(nomask),          # 1 → skip mask, 0 → apply mask
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to 4‑D tensors
    left = left_buf.view(B, N, N, hidden_dim)          # [B,N,N,H]
    right = right_buf.view(B, N, N, hidden_dim)        # [B,N,N,H]
    out_gate = out_gate_buf.view(B, N, N, hidden_dim)  # [B,N,N,H]

    # prepare tensors for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]  (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B,H,N,N]  (k,j)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)      # [B·H, N, N]
    right_mat = right_perm.view(BH, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat)   # fp16, shape [B·H, N, N]
    # reshape back to 4‑D tensor
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                            # [B,N,N,H]

    # --------------------------------------------------------------
    # 4️⃣ Final fused kernel: LayerNorm (H) → out_gate → final Linear
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(total_rows, hidden_dim).contiguous()

    # transpose final linear weight once (to get shape [H, C])
    to_out_weight_T = weights["to_out.weight"].transpose(0, 1).contiguous().half()

    out_final_fp32 = torch.empty((total_rows, dim), dtype=torch.float32, device=device)

    BLOCK_D_FINAL = 64
    grid_final = (triton.cdiv(total_rows, BLOCK_M),)

    final_fused_kernel[grid_final](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],      # γ (float32, [H])
        weights["to_out_norm.bias"],        # β (float32, [H])
        to_out_weight_T,                    # [H, C] (fp16)
        out_final_fp32,
        total_rows,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D_FINAL,
    )

    # reshape to [B, N, N, dim] (float32) and return
    final = out_final_fp32.view(B, N, N, dim)
    return final
