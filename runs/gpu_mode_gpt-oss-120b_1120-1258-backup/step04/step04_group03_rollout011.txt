3️⃣  Optional pair‑wise mask is applied to the two projection streams only
    (as in the reference implementation).

4️⃣  Sigmoid gates are computed in half‑precision and the gated tensors are
    kept in fp16 to stay Tensor‑Core friendly.

5️⃣  The triangular multiplicative update
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is expressed as an `einsum`.  This avoids the costly
    `permute(...).contiguous()` copies that were required for the
    batched `torch.matmul` version while still letting cuBLAS/Tensor‑Core
    handle the underlying matrix‑multiply.

6️⃣  Second LayerNorm (over the hidden dimension) re‑uses the same Triton
    kernel.  The result is multiplied *in‑place* by the output‑gate to
    save an extra temporary tensor.

7️⃣  Final linear projection back to the model dimension is a half‑precision
    GEMM (`F.linear`).  The API contract requires a float32 output, so the
    result is cast back at the end.

All heavy‑weight arithmetic stays in fp16 (Tensor‑Core friendly); only the
normalisation statistics are accumulated in fp32.  The implementation works
for every configuration listed in the problem statement and respects the
“no‑mask” / “mask” options.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm (fp16 output, mean/var in fp32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # mut   half*    [total_rows, dim]   (fp16 output)
    weight_ptr,     # const float*   [dim]               (γ)
    bias_ptr,       # const float*   [dim]               (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1 – compute mean & variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2 – write normalized + affine (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # fp32 raw data

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β (fp32)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – returns a float32 tensor of shape
    [bs, seq_len, seq_len, dim].

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : mapping of all layer‑norm / linear weights (float32)
        * config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Float32 tensor of shape [B, N, N, dim] on the same device as the inputs.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape helpers
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                # [B, N, N, C]
    total_rows = B * N * N                         # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Tuned launch configuration (good for H100)
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (still fp16)
    # ------------------------------------------------------------------
    # concatenate all five weight matrices (shape (5*H, C))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                         # (5*H, C)

    # Linear: (total_rows, C) @ (C, 5*H).T → (total_rows, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)    # (total_rows, 5*H) fp16

    # Split back to the five tensors (each total_rows × H)
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape (B, N, N) → (total_rows, 1) broadcast over H
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Gates (sigmoid) + element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # (total_rows, H)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16  = left_proj * left_gate     # (total_rows, H)
    right_fp16 = right_proj * right_gate   # (total_rows, H)

    # Reshape back to 4‑D tensors for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM via einsum)
    # ------------------------------------------------------------------
    # einsum avoids the costly permute/contiguous copies required for
    # torch.matmul on (B*H, N, N) tensors.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden_dim (fp16 output)
    # ------------------------------------------------------------------
    out_flat = out.view(B * N * N, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(B * N * N, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B * N * N,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # In‑place multiplication by the output‑gate (saves a temporary)
    out_norm_fp16 *= out_gate.view(B * N * N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(out_norm_fp16, to_out_w_fp16)   # (B*N*N, C) fp16

    # Cast back to FP32 as required by the API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
