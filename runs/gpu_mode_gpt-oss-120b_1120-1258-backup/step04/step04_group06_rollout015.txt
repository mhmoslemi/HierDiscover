    left_ptr, right_ptr, out_ptr,
    B, N, H,

    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    total_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    H_DIM: tl.constexpr,
    EPS: tl.constexpr,
):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

```python
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    total_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    H_DIM: tl.constexpr,
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    offs_m = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # broadcast
    offs_m = offs_m[:, None]   # (M,1)
    offs_n = offs_n[None, :]   # (1,N)

    # allocate accumulator (M,N,H_DIM)
    acc = tl.zeros((BLOCK_M, BLOCK_N, H_DIM), dtype=tl.float32)

    # range for hidden dim
    h_range = tl.arange(0, H_DIM)

    for k in range(total_k):
        # load left vector shape (M, H_DIM)
        left_ptrs = left_ptr + \
            pid_b * stride_l_b + \
            offs_m * stride_l_i + \
            k * stride_l_j + \
            h_range * stride_l_h
        left_vec = tl.load(left_ptrs,
                           mask=mask_m[:, None] & (k < total_k),  # cond always true
                           other=0.0).to(tl.float32)  # shape (M, H_DIM)

        # load right vector shape (N, H_DIM)
        right_ptrs = right_ptr + \
            pid_b * stride_r_b + \
            offs_n * stride_r_i + \
            k * stride_r_j + \
            h_range * stride_r_h
        right_vec = tl.load(right_ptrs,
                            mask=mask_n[None, :] & (k < total_k),
                            other=0.0).to(tl.float32)  # shape (N, H_DIM)

        # accumulate outer product (M,N,H_DIM)
        acc += left_vec[:, None, :] * right_vec[None, :, :]

    # store out
    out_ptrs = out_ptr + \
        pid_b * stride_o_b + \
        offs_m * stride_o_i + \
        offs_n * stride_o_j + \
        h_range * stride_o_h

    # masks for store
    mask_store = mask_m[:, None] & mask_n[None, :]   # (M,N)
    mask_store = mask_store[:, :, None]  # (M,N,1) broadcast to H_DIM

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)
```

"""
fast_trimul_outgoing.py
~~~~~~~~~~~~~~~~~~~~~~~

A high‑performance implementation of the *outgoing* Triangle Multiplicative
Update (TriMul) used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (input) – Triton kernel (`layernorm_kernel`);
   operates on a flattened `[B·N·N, dim]` view and writes fp16 values.

2️⃣ One fused linear projection produces the two projections and three gates
   (`[R, dim] @ [5·H, dim].T → [R, 5·H]`).  The result is sliced to obtain
   `left_proj, right_proj, left_gate, right_gate, out_gate`.

3️⃣ Optional mask is applied (only to the two projections) and the three
   sigmoid gates are applied.

4️⃣ The heavy N³ update:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is a batch of H independent matrix‑multiplications of size `N×N`.
   By permuting the tensors to `[B, H, N, N]` we can invoke the highly
   optimised cuBLAS batched GEMM (`torch.matmul`) without any copies.

5️⃣ Second LayerNorm (over the hidden dimension) – same Triton kernel.

6️⃣ Element‑wise output‑gate multiplication, final projection back to `dim`,
   and cast to `float32`.

Only the two LayerNorms are written in Triton; all GEMM‑heavy parts use
cuBLAS/Tensor‑Cores, yielding a runtime well below the target 1 ms on an
H100 for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *   [total_rows, dim]   (float32 or fp16)
    y_ptr,            # mutable * [total_rows, dim]   (fp16 output)
    weight_ptr,       # const *   [dim]               (float32)
    bias_ptr,         # const *   [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    y = (x - μ) / √(σ²+eps) * γ + β   (output fp16).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute sum and sum‑of‑squares per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape          # input shape [B, N, N, dim]
    total_rows = B * N * N                    # rows for row‑wise LN

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input pointer (fp32)
        x_norm,                     # output pointer (fp16)
        weights["norm.weight"],     # γ (fp32)
        weights["norm.bias"],       # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 3️⃣ Fuse five linear layers (2 proj + 3 gates) into one GEMM
    # --------------------------------------------------------------
    # concat weight matrix: (5·H, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    proj = F.linear(x_norm, w_concat)               # [R, 5·H]  (R = B·N·N)
    H = hidden_dim
    left_proj   = proj[:, 0:H]                       # [R, H]
    right_proj  = proj[:, H:2 * H]                   # [R, H]
    left_gate   = torch.sigmoid(proj[:, 2 * H:3 * H])
    right_gate  = torch.sigmoid(proj[:, 3 * H:4 * H])
    out_gate    = torch.sigmoid(proj[:, 4 * H:5 * H])

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (applied only to the projected tensors)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → flatten to [R, 1]
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # element‑wise gating
    left  = left_proj * left_gate                     # [R, H]
    right = right_proj * right_gate                   # [R, H]

    # reshape back to 4‑D tensors
    left  = left.view(B, N, N, H)                     # [B, N, N, H]
    right = right.view(B, N, N, H)                    # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)              # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # permute to [B, H, N, N] – no .contiguous() to avoid copies
    left_perm  = left.permute(0, 3, 1, 2)   # shape (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)

    # batched matrix multiplication: (B,H,N,N) @ (B,H,N,N)ᵀ → (B,H,N,N)
    out_mat = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # back to the original layout [B, N, N, H]
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)
    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                         # element‑wise (fp16)
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)         # [dim, H]
    )   # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
