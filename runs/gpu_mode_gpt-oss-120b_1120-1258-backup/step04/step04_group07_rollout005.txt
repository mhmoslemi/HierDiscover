def trigate_kernel(
    out_concat_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows, hidden_dim, 
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):

    row_start = row * BLOCK_R
    offs_m = row_start + tl.arange(0, BLOCK_R)
    mask_m = offs_m < total_rows

    # For each column block (hidden dim)
    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        # Compute pointers offsets
        # Base stride for out_concat columns: total_cols = hidden_dim * 5
        total_cols = hidden_dim * 5
        # Row base offset
        row_offset = offs_m[:, None] * total_cols

        left_proj = tl.load(
            out_concat_ptr + row_offset + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],

        )
        # right_proj slice offset + hidden_dim
        right_proj = tl.load(
            out_concat_ptr + row_offset + hidden_dim + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        # left_gate raw offset + 2*hidden_dim
        left_gate_raw = tl.load(
            out_concat_ptr + row_offset + 2 * hidden_dim + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0

        # right_gate raw offset + 3*hidden_dim
        right_gate_raw = tl.load(
            out_concat_ptr + row_offset + 3 * hidden_dim + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0

        # out_gate raw offset + 4*hidden_dim
        out_gate_raw = tl.load(

   * A single GEMM (`F.linear`) computes the five linear projections
     (left_proj, right_proj, left_gate, right_gate, out_gate) in one shot.
   * A custom Triton kernel reads the 5‑block result, applies
     `sigmoid` to the three gate blocks, multiplies the gates with the
     corresponding projections, and (optionally) applies the pair‑wise mask.
   * The kernel emits three tensors in fp16: `left` (B,N,N,H), `right`
     (B,N,N,H) and `out_gate` (B,N,N,H).

3. **Triangular multiplicative update** (the “outgoing” step):
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This is a batched matrix‑multiply `left_batch @ right_batch^T` where
   `left_batch = left.permute(0,3,1,2).view(B*H,N,N)` and similarly for
   `right_batch`.  The operation uses the highly‑optimized cuBLAS GEMM
   (`torch.bmm`).

4. **Second LayerNorm** over the hidden dimension `H` – same Triton kernel as
   step 1.

5. **Output gate & final projection**:
   `out = (norm(out) * out_gate) @ to_out.weight` (all in fp16) and finally
   cast back to `float32`.

All heavy arithmetic (LayerNorm, the fused projection/gating, and the
tri‑multiplicative update) runs in half precision on H100 Tensor‑Cores,
while the kernel launches are kept minimal (only two Triton kernels).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,            # const *[total_rows, dim]   (fp32 input)
    y_ptr,            # mutable *[total_rows, dim] (fp16 output)
    weight_ptr,       # const   *[dim]            (fp32)
    bias_ptr,         # const   *[dim]            (fp32)
    total_rows,
    dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------- first pass: mean & variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: normalize & apply γ,β -------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def gated_projection_kernel(
    proj_ptr,           # const *[R, 5*H]   (fp16, result of fused linear)
    mask_ptr,           # const *[R]        (fp16, 0/1 mask, may be all‑ones)
    left_ptr,           # mutable *[R, H]   (fp16)
    right_ptr,          # mutable *[R, H]   (fp16)
    out_gate_ptr,       # mutable *[R, H]   (fp16)
    total_rows,
    hidden_dim,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < total_rows

    # total columns in the fused projection tensor
    cols_total = hidden_dim * 5

    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_c = h * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_c = offs_c < hidden_dim

        # base offsets for the current rows
        row_off = offs_r[:, None] * cols_total

        # ---- load the five blocks ------------------------------------
        left_proj    = tl.load(proj_ptr + row_off + offs_c[None, :],
                               mask=mask_r[:, None] & mask_c[None, :],
                               other=0.0)                       # [R, H]
        right_proj   = tl.load(proj_ptr + row_off + hidden_dim + offs_c[None, :],
                               mask=mask_r[:, None] & mask_c[None, :],
                               other=0.0)
        left_gate_raw = tl.load(proj_ptr + row_off + 2*hidden_dim + offs_c[None, :],
                               mask=mask_r[:, None] & mask_c[None, :],
                               other=0.0)
        right_gate_raw = tl.load(proj_ptr + row_off + 3*hidden_dim + offs_c[None, :],
                               mask=mask_r[:, None] & mask_c[None, :],
                               other=0.0)
        out_gate_raw = tl.load(proj_ptr + row_off + 4*hidden_dim + offs_c[None, :],
                               mask=mask_r[:, None] & mask_c[None, :],
                               other=0.0)

        # ---- sigmoid in fp32 for stability ----
        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))

        # back to fp16
        left_gate  = left_gate .to(tl.float16)
        right_gate = right_gate.to(tl.float16)
        out_gate   = out_gate  .to(tl.float16)

        # ---- apply gates ------------------------------------------------
        left  = left_proj  * left_gate
        right = right_proj * right_gate

        # ---- optional mask (broadcasted over hidden dim) ----------------
        if mask_ptr is not None:
            mask_val = tl.load(mask_ptr + offs_r,
                               mask=mask_r,
                               other=1.0)                # scalar per row
            mask_val = mask_val[:, None]                 # broadcast
            left  = left  * mask_val
            right = right * mask_val

        # ---- store -------------------------------------------------------
        tl.store(left_ptr     + row_off + offs_c[None, :],
                 left, mask=mask_r[:, None] & mask_c[None, :])
        tl.store(right_ptr    + row_off + offs_c[None, :],
                 right, mask=mask_r[:, None] & mask_c[None, :])
        tl.store(out_gate_ptr + row_off + offs_c[None, :],
                 out_gate, mask=mask_r[:, None] & mask_c[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping str → torch.Tensor (float32)
        - config_dict  : {"dim": int, "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – fp16 output
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (fp32)
        weights["norm.bias"],     # β  (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2) Fused linear projection (5 heads) – fp16 result
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape [5*H, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)

    # out_concat : [R, 5*H] (fp16)
    out_concat = F.linear(x_norm, w_concat)

    # ------------------------------------------------------------------
    # 3) Gate & mask fusion – produce left, right, out_gate (fp16)
    # ------------------------------------------------------------------
    if nomask:
        # dummy mask of ones (no effect)
        mask_flat = torch.ones(total_rows, dtype=torch.float16, device=device)
    else:
        mask_flat = mask_tensor.to(torch.float16).view(total_rows)

    left  = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    right = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    out_gate = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 64
    grid_gate = (triton.cdiv(total_rows, BLOCK_R),)

    gated_projection_kernel[grid_gate](
        out_concat,                     # proj_ptr
        mask_flat,                      # mask_ptr
        left,                           # left_ptr
        right,                          # right_ptr
        out_gate,                       # out_gate_ptr
        total_rows,
        hidden_dim,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
    )

    # reshape tensors back to [B, N, N, H]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4) Triangular multiplicative update (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for a single bmm call
    left_batch  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiply with transposed right operand
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

    # restore to [B, N, N, H]
    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Second LayerNorm over hidden dimension H (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6) Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise fp16
    out_norm_flat = out_norm.view(total_rows, hidden_dim)

    final_half = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16)    # [dim, hidden_dim]
    )                                                   # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
