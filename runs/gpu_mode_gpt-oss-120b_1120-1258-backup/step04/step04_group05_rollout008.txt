def layernorm_fp16_fused_kernel(
    x_ptr,         # *fp32* [total_rows, dim]
    y_ptr,         # *fp16* [total_rows, dim]
    weight_ptr,    # *fp32* [dim]
    bias_ptr,      # *fp32* [dim]
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # offsets for loading
    x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])  # shape [BLOCK_M, BLOCK_D]

    # Load x (some threads may be out-of-bounds)
    x = tl.load(x_ptrs,
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0)  # fp32

    # compute mean and var per row (BLOCK_M)
    mean = tl.sum(x, axis=1) / dim
    sum_sq = tl.sum(x * x, axis=1)
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Broadcast mean, rstd to shape [BLOCK_M, BLOCK_D]
    # Use tl.reshape? Actually we can broadcast via indexing: mean[:, None], rstd[:, None]
    norm = (x - mean[:, None]) * rstd[:, None]  # fp32

    # Load weight and bias per column
    w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # shape [BLOCK_D]
    b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)   # shape [BLOCK_D]

    # Store output
    y_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])

def layernorm_fp16_fused_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # compute global offsets to load
    x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
    x = tl.load(x_ptrs,
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0).to(tl.float32)

    # row sum and sum_sq
    sum_val = tl.sum(x, axis=1)
    sum_sq = tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean

    # normalize

    y_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])

             mask=mask_m[:, None] & mask_d[None, :])

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key optimisations
-----------------
1️⃣ Fused row‑wise LayerNorm written in Triton:
   – Each program processes a tile of (BLOCK_M rows × dim columns).
   – The whole row is loaded once (no second global‑memory read).
   – Mean / variance are computed in‑place with fp32 accumulation,
     then the normalized result is written back in fp16.
   – Works for any dim ≤ 384 (the maximum hidden size in the tests).

2️⃣ Five linear projections (left/right + three gates) are concatenated
   into a single `torch.nn.functional.linear` call (fp16 GEMM).

3️⃣ The heavy triangular update is performed as a batched GEMM
   `left @ rightᵀ` using `torch.bmm` on fp16 tensors (Tensor‑Core).

4️⃣ The second LayerNorm (on the hidden dimension) re‑uses the same fused
   Triton kernel.

All other operations (masking, sigmoids, final gating + linear) are kept
in PyTorch but stay in fp16 to minimise memory traffic.
The final tensor is cast back to float32 as required by the API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton fused LayerNorm (fp16 output, fp32 stability)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_fused_kernel(
    x_ptr,          # *const fp32/ fp16*   [total_rows, dim]   (input)
    y_ptr,          # *mut    fp16*       [total_rows, dim]   (output)
    weight_ptr,     # *const fp32*        [dim]                (γ)
    bias_ptr,       # *const fp32*        [dim]                (β)
    total_rows,     # i32
    dim: tl.constexpr,        # channel dimension (compile‑time)
    eps: tl.constexpr,        # numeric stabiliser
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # columns per program (== dim)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    offs_d = tl.arange(0, BLOCK_D)                      # [BLOCK_D]
    mask_d = offs_d < dim

    # -----------------------------------------------------------------
    # Load a tile of the input (fp32 or fp16 – cast to fp32 for math)
    # -----------------------------------------------------------------
    x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
    x = tl.load(
        x_ptrs,
        mask=mask_m[:, None] & mask_d[None, :],
        other=0.0,
    ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

    # -----------------------------------------------------------------
    # Compute mean and variance (fp32) per row
    # -----------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)                         # [BLOCK_M]
    sum_sq  = tl.sum(x * x, axis=1)                     # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Normalise, apply γ (weight) and β (bias), cast to fp16
    # -----------------------------------------------------------------
    # broadcast mean / rstd over the column dimension
    norm = (x - mean[:, None]) * rstd[:, None]          # fp32

    w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # [BLOCK_D]
    b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # [BLOCK_D]

    y = (norm * w[None, :] + b[None, :]).to(tl.float16)       # fp16

    # -----------------------------------------------------------------
    # Store the normalised result
    # -----------------------------------------------------------------
    y_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
    tl.store(
        y_ptrs,
        y,
        mask=mask_m[:, None] & mask_d[None, :],
    )

# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – may be all‑ones
            - weights_dict : mapping of weight names to tensors (float32)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Shape preparation
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # number of (b,i,j) triples

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output, fused kernel)
    # --------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)   # fp32
    x_norm_fp16 = torch.empty(
        total_rows, dim, dtype=torch.float16, device=device
    )                                                          # fp16 output buffer

    BLOCK_M = 64                      # rows per program – fits the grid nicely
    BLOCK_D = dim                     # one thread per channel
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_fused_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # --------------------------------------------------------------
    # Each weight is [hidden_dim, dim]; concatenate along the output axis.
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                 # (5*H, C)

    # Fused GEMM – produces [total_rows, 5*hidden_dim] in fp16
    proj_all = F.linear(x_norm_fp16, concat_w)            # fp16

    # Split the fused projection back to the individual tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )                                                     # each [total_rows, H] (fp16)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [total_rows, 1] → broadcast over H
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16) and element‑wise gating
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)       # [total_rows, H]
    right_gate  = torch.sigmoid(right_gate_fp16)     # [total_rows, H]
    out_gate    = torch.sigmoid(out_gate_fp16)       # [total_rows, H]

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape back to 4‑D tensors for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)      # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)    # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Move hidden dimension into the batch dimension to obtain a true GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim                                 # batch * hidden
    left_mat   = left_perm.view(BH, N, N)                # [BH, N, N]
    right_mat  = right_perm.view(BH, N, N)               # [BH, N, N]

    # Batched matrix multiplication (fp16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fused kernel)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)               # [total_rows, H]
    out_norm_fp16 = torch.empty_like(
        out_flat, dtype=torch.float16, device=device
    )                                                        # fp16 buffer

    BLOCK_M2 = BLOCK_M                                        # reuse same tile size
    BLOCK_D2 = hidden_dim                                     # tile width = hidden_dim
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    layernorm_fp16_fused_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                               # fp16 element‑wise

    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)           # [total_rows, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)         # [total_rows, C]

    # --------------------------------------------------------------
    # 8️⃣ Cast to fp32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
