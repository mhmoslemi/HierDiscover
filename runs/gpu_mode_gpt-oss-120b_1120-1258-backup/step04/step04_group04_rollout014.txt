Inside kernel:

- Compute per row mean and var: sum and sum_sq across C.
- Compute normalized values: norm = (x - mean) * inv_std (float32)
- Optionally apply mask later after scaling

- For each hidden index h, compute left_proj, left_gate, right_proj, right_gate, out_gate accumulations.

- Compute left_gate_sigmoid = 1/(1+exp(-left_gate)), etc.
- left_scaled = left_proj * left_gate_sigmoid
- Similarly for right_scaled
- if mask: left_scaled *= mask, right_scaled *= mask

- Write results.

@triton.jit
def layernorm_linear_gate_fusion_kernel(
    x_ptr,         # *const fp32* [R, C]
    norm_weight_ptr, norm_bias_ptr,  # *const fp32* [C]
    left_proj_weight_ptr,
    right_proj_weight_ptr,
    left_gate_weight_ptr,
    right_gate_weight_ptr,
    out_gate_weight_ptr,
    mask_ptr,               # *const fp16* [R] (or fp32)
    left_out_ptr, right_out_ptr, out_gate_out_ptr,  # *mut fp16* [R, H]
    R, C, H,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # Compute per-row mean & var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(C, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < C
        x = tl.load(x_ptr + (offs_m[:, None] * C + offs_d[None, :]),
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0)  # fp32
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Loop over hidden dimension blocks
    for h_start in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_start * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Initialize accumulators for each projection/gate
        left_proj_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_proj_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        # Iterate over columns of x
        for d in range(0, tl.cdiv(C, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < C

            # Load raw x
            x = tl.load(x_ptr + (offs_m[:, None] * C + offs_d[None, :]),
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0)   # fp32

            # Normalize
            norm = (x - mean[:, None]) * rstd[:, None]   # fp32

            # Apply LN weight and bias
            w_lnorm = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
            b_lnorm = tl.load(norm_bias_ptr + offs_d, mask=mask_d, other=0.0)     # fp32
            norm = norm * w_lnorm[None, :] + b_lnorm[None, :]   # broadcasting

            # Load weight tiles (transposed: [BLOCK_D, BLOCK_H])
            # left_proj_weight
            w_left = tl.load(left_proj_weight_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            w_right = tl.load(right_proj_weight_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_lgate = tl.load(left_gate_weight_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_rgate = tl.load(right_gate_weight_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_outgate = tl.load(out_gate_weight_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                                mask=mask_d[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # Dot products
            left_proj_acc += tl.dot(norm, w_left)   # (M, D)*(D, H) -> (M, H)
            left_gate_acc += tl.dot(norm, w_lgate)
            right_proj_acc += tl.dot(norm, w_right)
            right_gate_acc += tl.dot(norm, w_rgate)
            out_gate_acc += tl.dot(norm, w_outgate)

        # Apply sigmoid to gate accumulators
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate_acc))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate_acc))
        out_gate_sig = 1.0 / (1.0 + tl.exp(-out_gate_acc))

        # Scale projections with gates
        left_scaled = left_proj_acc * left_gate_sig
        right_scaled = right_proj_acc * right_gate_sig

        # Apply mask if provided (mask is fp16 or fp32)
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0)   # fp32
        left_scaled = left_scaled * mask_val[:, None]
        right_scaled = right_scaled * mask_val[:, None]

        # Store results
        left_ptrs = left_out_ptr + (offs_m[:, None] * H + offs_h[None, :])
        right_ptrs = right_out_ptr + (offs_m[:, None] * H + offs_h[None, :])
        out_gate_ptrs = out_gate_out_ptr + (offs_m[:, None] * H + offs_h[None, :])

        tl.store(left_ptrs, left_scaled.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptrs, right_scaled.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptrs, out_gate_sig.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
```

"""
TriMul (outgoing) – fused first LayerNorm + all five linear projections +
sigmoid gating + optional mask.  The heavy‑weight reduction (mean/var) and the
five GEMMs are performed in a single Triton kernel, dramatically reducing the
traffic on the input tensor (the original implementation reads the input three
times).  The triangular update itself stays as a batched GEMM (torch.bmm) which
uses Tensor‑Core‑accelerated cuBLAS.  The second LayerNorm is kept in a
custom Triton kernel (required “at‑least‑one‑kernel” constraint).  All
intermediate tensors are FP16; the final output is cast back to FP32.

Inputs:
    input_tensor : [B, N, N, C]       (float32)
    mask_tensor  : [B, N, N]          (bool/float, optional)
    weights      : dict with the nine model parameters
    config       : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float}

Outputs:
    Tensor of shape [B, N, N, C] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# LayerNorm kernel (used for the second LayerNorm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const  fp32* [rows, dim]
    y_ptr,          # *mut    fp16* [rows, dim]
    weight_ptr,     # *const  fp32* [dim]   (γ)
    bias_ptr,       # *const  fp32* [dim]   (β)
    rows,           # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- pass 1: mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                     # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: normalize, scale and bias ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                     # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Fused kernel: LayerNorm + 5 linear projections + sigmoid gates + mask
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_proj_gate_kernel(
    x_ptr,                # *const  fp32*   [R, C]
    ln_weight_ptr,        # *const  fp32*   [C]
    ln_bias_ptr,          # *const  fp32*   [C]
    left_proj_w_ptr,      # *const  fp16*   [H, C]
    right_proj_w_ptr,     # *const  fp16*   [H, C]
    left_gate_w_ptr,      # *const  fp16*   [H, C]
    right_gate_w_ptr,     # *const  fp16*   [H, C]
    out_gate_w_ptr,       # *const  fp16*   [H, C]
    mask_ptr,             # *const  fp16*   [R]   (all‑ones when no mask)
    left_out_ptr,         # *mut    fp16*   [R, H]
    right_out_ptr,        # *mut    fp16*   [R, H]
    out_gate_out_ptr,     # *mut    fp16*   [R, H]
    R, C, H,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # columns per tile (C‑dimension)
    BLOCK_H: tl.constexpr,    # hidden block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # ----- first pass: compute mean / variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(C, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < C
        x = tl.load(x_ptr + (offs_m[:, None] * C + offs_d[None, :]),
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0)                     # fp32
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- loop over hidden‑dim blocks -----
    for h_start in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_start * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Accumulators for the five projections / gates
        left_proj_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_proj_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        out_gate_acc   = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        for d in range(0, tl.cdiv(C, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < C

            # Load raw input
            x = tl.load(x_ptr + (offs_m[:, None] * C + offs_d[None, :]),
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0)                      # fp32

            # Layer‑norm (including γ,β)
            norm = (x - mean[:, None]) * rstd[:, None]   # fp32
            w_ln = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
            b_ln = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32
            norm = norm * w_ln[None, :] + b_ln[None, :]

            # Load weight tiles (transposed: [BLOCK_D, BLOCK_H])
            w_lproj = tl.load(left_proj_w_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_rproj = tl.load(right_proj_w_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_lgate = tl.load(left_gate_w_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_rgate = tl.load(right_gate_w_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_outgate = tl.load(out_gate_w_ptr + (offs_d[:, None] + offs_h[None, :] * C),
                                mask=mask_d[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # GEMM tiles (M×D)·(D×H) → (M×H)
            left_proj_acc += tl.dot(norm, w_lproj)
            left_gate_acc += tl.dot(norm, w_lgate)
            right_proj_acc += tl.dot(norm, w_rproj)
            right_gate_acc += tl.dot(norm, w_rgate)
            out_gate_acc   += tl.dot(norm, w_outgate)

        # ----- sigmoid gates -----
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate_acc))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate_acc))
        out_gate_sig   = 1.0 / (1.0 + tl.exp(-out_gate_acc))

        # ----- apply gates -----
        left_scaled  = left_proj_acc * left_gate_sig
        right_scaled = right_proj_acc * right_gate_sig

        # ----- optional mask (mask already FP16, broadcast over H) -----
        mask_val = tl.load(mask_ptr + offs_m,
                           mask=mask_m,
                           other=1.0).to(tl.float32)   # shape [BLOCK_M]
        left_scaled  = left_scaled * mask_val[:, None]
        right_scaled = right_scaled * mask_val[:, None]

        # ----- store results -----
        left_ptrs  = left_out_ptr  + (offs_m[:, None] * H + offs_h[None, :])
        right_ptrs = right_out_ptr + (offs_m[:, None] * H + offs_h[None, :])
        out_ptrs   = out_gate_out_ptr + (offs_m[:, None] * H + offs_h[None, :])

        tl.store(left_ptrs,  left_scaled .to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptrs, right_scaled.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_ptrs,   out_gate_sig.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a fused
    LayerNorm‑Projection‑Gate kernel.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # R = B·N·N

    # ------------------------------------------------------------------
    # Prepare mask (flat, fp16). If no mask -> all‑ones.
    # ------------------------------------------------------------------
    if nomask:
        mask_flat = torch.ones(total_rows,
                               dtype=torch.float16,
                               device=device)
    else:
        mask_flat = mask_tensor.view(total_rows).to(torch.float16)

    # ------------------------------------------------------------------
    # Allocate output buffers for the fused kernel
    # ------------------------------------------------------------------
    left_scaled  = torch.empty((total_rows, hidden_dim),
                               dtype=torch.float16,
                               device=device)
    right_scaled = torch.empty_like(left_scaled)
    out_gate     = torch.empty_like(left_scaled)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 32          # rows per program (≤ 1024 / BLOCK_H)
    BLOCK_D = 64          # intra‑column tile (covers dim up to 384)
    BLOCK_H = 32          # hidden‑dim tile (fits 5× blocks for H≤384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)

    # ------------------------------------------------------------------
    # Run fused LayerNorm + 5 linear projections + sigmoid gates + mask
    # ------------------------------------------------------------------
    fused_ln_proj_gate_kernel[grid](
        input_tensor.view(total_rows, dim).contiguous(),               # x_ptr (fp32)
        weights["norm.weight"],                                         # ln weight (fp32)
        weights["norm.bias"],                                           # ln bias   (fp32)
        weights["left_proj.weight"].half().contiguous(),               # left_proj weight (fp16)
        weights["right_proj.weight"].half().contiguous(),              # right_proj weight
        weights["left_gate.weight"].half().contiguous(),               # left_gate weight
        weights["right_gate.weight"].half().contiguous(),              # right_gate weight
        weights["out_gate.weight"].half().contiguous(),                # out_gate weight
        mask_flat,                                                      # mask (fp16)
        left_scaled,                                                    # left_out (fp16)
        right_scaled,                                                   # right_out (fp16)
        out_gate,                                                       # out_gate_out (fp16)
        total_rows,
        dim,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # Reshape tensors back to [B, N, N, H]
    # ------------------------------------------------------------------
    left = left_scaled.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_scaled.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate_4d = out_gate.view(B, N, N, hidden_dim)     # [B, N, N, H]

    # ------------------------------------------------------------------
    # Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # (B, H, N, N) where axes are (i,k) and (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # [B*H, N, N]   (i,k)
    right_mat = right_perm.view(BH, N, N)    # [B*H, N, N]   (k,j)

    out_mat = torch.bmm(left_mat, right_mat)    # fp16, result [B*H, N, N]

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # fp16

    # ------------------------------------------------------------------
    # Gating with out_gate and final projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate_4d                     # fp16 element‑wise

    to_out_weight_fp16 = weights["to_out.weight"].half().contiguous()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [R_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)   # [R_h, C] fp16

    # Cast back to float32 as required by the specification
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
