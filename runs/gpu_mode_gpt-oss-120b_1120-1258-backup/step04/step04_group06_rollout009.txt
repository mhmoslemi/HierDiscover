"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (B·N·N×dim) → fp16 via a tiny Triton
   kernel (the only mandatory kernel in the solution).

2️⃣ Fuse the two projections (left/right) together with the two gate
   linear layers into ONE half‑precision GEMM:
   `[R, dim] @ [4*H, dim].T → [R, 4*H]`
   where  `R = B·N·N` and `H = hidden_dim`.
   The result is split into `left_proj, right_proj, left_gate, right_gate`.

3️⃣ Compute the output gate with a second GEMM (`[R, dim] @ [H, dim].T`).

4️⃣ Apply the optional mask *only* to the two projected tensors,
   apply the three sigmoid gates (in‑place) and obtain the gated
   tensors `left` and `right`.

5️⃣ Triangular multiplicative update:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is a batched half‑precision GEMM
   ` (B·H, N, N) @ (B·H, N, N)ᵀ ` performed with `torch.bmm`
   (cuBLAS Tensor‑Cores on H‑100).

6️⃣ Second LayerNorm over the hidden dimension (again Triton).

7️⃣ Multiply by the output gate and linearly project back to `dim`
   (half‑precision).  Cast the result to `float32` to match the reference
   implementation.

The implementation reduces memory traffic by:
   * avoiding the large concatenated `[R,5*H]` tensor of the naïve version,
   * using in‑place sigmoid, and
   * re‑using the same normalized tensor for all linear layers.
All heavy per‑row normalisations are done with a custom Triton kernel,
while large matrix multiplications leverage cuBLAS for maximum speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (float32)
    y_ptr,               # mutable * [total_rows, dim]   (float16)
    weight_ptr,          # const   * [dim]               (float32)
    bias_ptr,            # const   * [dim]               (float32)
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Output is fp16, accumulation is fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and "nomask" flag

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [R, dim] float32
    x_norm_half = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their two gates (4*H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four weight matrices needed for left/right + gates
    w_concat_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [4*H, dim]

    # one GEMM: [R, dim] @ [4*H, dim].T → [R, 4*H]
    proj_gate = F.linear(x_norm_half, w_concat_4h)        # fp16

    # split the concatenated result
    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]              # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]            # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]            # [R, H]

    # output gate (separate linear)
    out_gate_raw = F.linear(x_norm_half,
                            weights["out_gate.weight"].to(torch.float16))  # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates, and apply them
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B,N,N] → flatten to [R,1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid to avoid extra buffers
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # element‑wise gating
    left  = left_proj_raw * left_gate        # [R, H] fp16
    right = right_proj_raw * right_gate       # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # bring the head dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(B * H, N, N)  # [B*H, N, N]

    # batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # [B,N,N,H] fp16

    final_half = F.linear(out_norm.view(total_rows, H),
                          weights["to_out.weight"].to(torch.float16))   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
