def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # 1. First LayerNorm using Triton kernel
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    del x_flat  # free original input

    # 2. Fused projections + gates + out_gate in one linear
    rows = total_rows
    # build combined weight (5*H, dim)
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device, non_blocking=True)  # shape (5*H, dim)

    # Linear: rows x dim @ (5*H, dim).T -> rows x 5*H
    combined = torch.nn.functional.linear(
        x_norm_h.view(rows, dim), weight_concat
    )  # bias = None

    # slices
    H = hidden_dim
    left_proj_pre = combined[:, :H]
    left_gate_pre = combined[:, H:2 * H]
    right_proj_pre = combined[:, 2 * H:3 * H]
    right_gate_pre = combined[:, 3 * H:4 * H]
    out_gate_pre = combined[:, 4 * H:]

    # optional mask for left & right projections
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(dtype=torch.float16, device=device, non_blocking=True)
        left_proj_pre = left_proj_pre * mask_h
        right_proj_pre = right_proj_pre * mask_h

    # compute gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # apply gates to projections
    left = left_proj_pre * left_gate
    right = right_proj_pre * right_gate

    # free unused tensors to free memory
    del combined, left_proj_pre, left_gate_pre, right_proj_pre, right_gate_pre, out_gate_pre, x_norm_h

    # 3. Tri-multiplicative update (batched GEMM over hidden dim)
    # reshape left/right to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # permute to [B, H, N, N] for batch GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch+head
    BH = B * H
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # compute out = left_mat @ right_mat.transpose(1, 2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    del left_mat, right_mat, left_perm, right_perm, left, right

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del out_mat

    # 4. Second LayerNorm on hidden dimension
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    del out_flat

    out_norm = out_norm_h.view(B, N, N, H)
    del out_norm_h

    # 5. Apply output gate and final linear projection
    # out_gate currently shape rows, H -> reshape
    out_gate = out_gate.view(B, N, N, H)
    gated = out_norm * out_gate

    # final linear
    final = torch.nn.functional.linear(
        gated.view(rows_h, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device, non_blocking=True)
    ).view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul “outgoing” forward pass (AlphaFold‑3) – highly‑optimised version.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the 4‑D input tensor.
2️⃣  Fuse *all* linear projections and gates (left/right proj, left/right gate,
    output gate) into a single GEMM:  x_norm @ W_concat.T   (W_concat has
    shape (5·hidden_dim, dim)).  This reduces three passes over the large
    [B·N², dim] matrix to a single pass, halving memory traffic.
3️⃣  Apply the optional pair‑wise mask (broadcast over hidden_dim) and the
    sigmoid‑gates.  Left and right tensors are multiplied element‑wise with
    their gates.
4️⃣  Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
    which is a batched matrix‑multiply per hidden dimension:
        out = (left_mat @ right_matᵀ)  with shape (B·H, N, N).
5️⃣  Second LayerNorm (on the hidden dimension) using the same Triton kernel.
6️⃣  Apply the output‑gate (sigmoid) and the final linear projection
        hidden_dim → dim.  The result is cast back to FP32.

All heavy linear operations run in FP16 (Tensor‑Core friendly).  Two Triton
kernels implement the LayerNorms; everything else uses fused PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input FP32, output FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per inner loop
):
    """Row‑wise LayerNorm over the last dimension of a flattened 2‑D view.
    Computes mean/variance in FP32, writes the normalized result in FP16."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------------- mean & variance (FP32) -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)         # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+eps)

    # ----------------- write normalized values (FP16) ------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # FP32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask_tensor  : torch.Tensor [B, N, N]      (float32, 0/1)
        weights_dict : dict with the following keys (all float32):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        config_dict : dict with keys
            "dim" (int), "hidden_dim" (int), optional "nomask" (bool)

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) using Triton
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()  # [rows, dim]

    # Allocate FP16 buffer for normalized tensor
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Triton launch
    BLOCK_M = 256            # rows per program (tuned for H100)
    BLOCK_D = 64             # cols per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # Free the original FP32 view – not needed any more
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates + out_gate (single GEMM)
    # ------------------------------------------------------------------
    rows = total_rows
    # Concatenate all five weight matrices (5·H × dim) and cast to FP16
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device, non_blocking=True)  # (5H, dim)

    # Linear: rows × dim  @  (5H, dim).T   → rows × 5H   (no bias)
    combined = F.linear(x_norm_h.view(rows, dim), weight_concat)  # FP16

    H = hidden_dim
    # Slice the concatenated output
    left_proj_pre = combined[:, :H]                 # [rows, H]
    left_gate_pre = combined[:, H:2 * H]
    right_proj_pre = combined[:, 2 * H:3 * H]
    right_gate_pre = combined[:, 3 * H:4 * H]
    out_gate_pre = combined[:, 4 * H:]              # [rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim) & sigmoid gates
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [rows, 1]
        mask_h = mask_tensor.view(rows, 1).to(dtype=torch.float16, device=device,
                                            non_blocking=True)
        left_proj_pre = left_proj_pre * mask_h
        right_proj_pre = right_proj_pre * mask_h

    left_gate = torch.sigmoid(left_gate_pre)        # [rows, H]
    right_gate = torch.sigmoid(right_gate_pre)      # [rows, H]
    out_gate = torch.sigmoid(out_gate_pre)          # [rows, H]

    # Apply gates to the projected values
    left = left_proj_pre * left_gate                # [rows, H]
    right = right_proj_pre * right_gate

    # Release tensors we no longer need (frees GPU memory early)
    del combined, left_proj_pre, left_gate_pre, right_proj_pre, right_gate_pre, out_gate_pre, x_norm_h

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Bring hidden dim into the batch axis for a single bmm call
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * H                                           # effective batch size
    left_mat = left_perm.view(BH, N, N)                  # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)                # [BH, N, N]

    # out_mat[b*h] = left_mat[b*h] @ right_mat[b*h].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [BH, N, N]

    # Cleanup intermediates
    del left_mat, right_mat, left_perm, right_perm, left, right

    # Reshape back to [B, N, N, H] (still FP16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (on hidden dimension)
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    del out_flat

    out_norm = out_norm_h.view(B, N, N, H)   # FP16
    del out_norm_h

    # ------------------------------------------------------------------
    # 6️⃣ Gating with out_gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)    # FP16
    gated = out_norm * out_gate             # element‑wise gate

    # Final linear: hidden_dim → dim
    final_h = F.linear(
        gated.view(rows_h, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device, non_blocking=True)
    )                                         # [rows_h, dim] (FP16)

    # Cast to FP32 as required by the original module
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
