"""
TriMul (outgoing) forward pass – Triton‑accelerated LayerNorms with fused
FP16 linear pipeline.

Algorithm
---------
1️⃣  Custom Triton LayerNorm (mean/var computed in FP32, output in FP16).
2️⃣  Fuse the five linear layers (left/right projections + three gates) into a
    single FP16 GEMM by concatenating their weight matrices.
3️⃣  Optional pairwise mask applied to the left/right projections (before gating).
4️⃣  Apply sigmoid to the three gate tensors.
5️⃣  Triangular multiplicative update:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    realized as a batched matrix‑multiply on tensors (B·H, N, N).
6️⃣  Second FP16 LayerNorm over the hidden dimension.
7️⃣  Multiply by the output gate and project back to the original dimension.
    Result is cast to FP32 to follow the API contract.

All heavy reductions (the two LayerNorms) are performed by a custom Triton
kernel; dense matrix algebra is delegated to cuBLAS/Tensor‑Core via standard
PyTorch ops.  The implementation works for any supported batch‑size, sequence
length, and channel dimensions, with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (FP32 accumulation, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim] (FP32 input)
    y_ptr,          # *mut half*     [total_rows, dim] (FP16 output)
    w_ptr,          # const float*   [dim]            (γ)
    b_ptr,          # const float*   [dim]            (β)
    total_rows,     # int (runtime)
    DIM: tl.constexpr,          # dim (compile‑time)
    EPS: tl.constexpr,          # eps (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – compute per‑row mean & variance (FP32 reduction)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < DIM

        # Load a tile (FP32)
        ptrs = x_ptr + (offs_m[:, None] * DIM + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / DIM
    var  = sum_sq / DIM - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # --------------------------------------------------------------
    # Pass 2 – write normalized + affine (FP16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM

        ptrs = x_ptr + (offs_m[:, None] * DIM + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        # Normalize
        norm = (vals - mean[:, None]) * rstd[:, None]

        # Load γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * DIM + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated TriMul forward (outgoing) – returns a float32 tensor
    of shape [B, N, N, C] on the same device as the inputs.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # C
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shapes
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                # [B, N, N, C]
    total_rows = B * N * N                          # rows for flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim),
                              dtype=torch.float16,
                              device=device)

    # Launch params – tuned for typical H100 configs
    BLOCK_M = 256          # rows per program (adjustable)
    BLOCK_D = 128          # dim tile (covers up to dim=384 efficiently)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # 2️⃣ Fused linear projections + gates (still FP16)
    # ------------------------------------------------------------------

        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()          # (5*H, C)

    # Linear: (total_rows, C) @ (C, 5*H).T → (total_rows, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)   # FP16 output

    # Split back into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )   # each: (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional pairwise mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B, N, N] → [total_rows, 1]
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # (total_rows, H)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating
    left_fp16  = left_proj * left_gate
    right_fp16 = right_proj * right_gate

    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    left  = left_fp16.view(B, N, N, hidden_dim)    # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)  # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim) # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # bring summed‑over dimension (k) to innermost axis for GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B_H = B * hidden_dim
    left_mat  = left_perm.view(B_H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B_H, N, N)  # (B*H, N, N)

    # batched matrix‑multiply – uses cuBLAS/Tensor Cores (FP16)
    out_mat = torch.matmul(left_mat, right_mat)   # (B*H, N, N) FP16

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension (FP16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # FP16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, N, N, H) FP16

    # Final linear projection H → C (still FP16)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # (total_rows_h, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # (total_rows_h, C)

    # Cast back to FP32 as required by the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
