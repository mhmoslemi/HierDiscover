def proj_gate_fuse_kernel(
    x_ptr,        # half [total_rows, dim]
    mask_ptr,     # half [total_rows] (or 1) – broadcast value
    left_ptr,     # half [total_rows, hidden] output
    right_ptr,    # half [total_rows, hidden] output
    out_gate_ptr, # half [total_rows, hidden] output (sigmoid)
    w_left_ptr,
    w_right_ptr,
    w_left_gate_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_rows = offs_m < total_rows

    for h_start in range(0, hidden, BLOCK_H):
        cur_H = tl.minimum(hidden - h_start, BLOCK_H)
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # Initialize accumulators for each projection
        acc_left = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_left_gate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_right_gate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_out_gate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        for d_start in range(0, dim, BLOCK_D):
            cur_D = tl.minimum(dim - d_start, BLOCK_D)
            offs_d = d_start + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # Load x tiles: [BLOCK_M, BLOCK_D]
            x = tl.load(x_ptr + (offs_m[:, None] * dim + offs_d[None, :]),
                        mask=mask_rows[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)  # shape (block_m, block_d)

            # Load weight tiles for each projection: shape (BLOCK_D, BLOCK_H) after transposition
            # compute base offsets for each weight matrix, using stride_h = dim, stride_d = 1
            w_left = tl.load(w_left_ptr + (offs_d[:, None] * 1 + (offs_h[None, :] * dim)),
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32) # shape (block_d, block_h)
            w_right = tl.load(w_right_ptr + (offs_d[:, None] * 1 + (offs_h[None, :] * dim)),
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            w_left_gate = tl.load(w_left_gate_ptr + (offs_d[:, None] * 1 + (offs_h[None, :] * dim)),
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            w_right_gate = tl.load(w_right_gate_ptr + (offs_d[:, None] * 1 + (offs_h[None, :] * dim)),
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            w_out_gate = tl.load(w_out_gate_ptr + (offs_d[:, None] * 1 + (offs_h[None, :] * dim)),
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            # Compute matrix multiplication: x (M,D) dot w (D,H) => (M,H)
            acc_left += tl.dot(x, w_left)          # (M,H)
            acc_right += tl.dot(x, w_right)
            acc_left_gate += tl.dot(x, w_left_gate)
            acc_right_gate += tl.dot(x, w_right_gate)
            acc_out_gate += tl.dot(x, w_out_gate)

        left_gate_sig = 1.0 / (1.0 + tl.exp(-acc_left_gate))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-acc_right_gate))
        out_gate_sig = 1.0 / (1.0 + tl.exp(-acc_out_gate))

        mask_val = tl.load(mask_ptr + offs_m, mask=mask_rows, other=1.0)

        right_out = (acc_right * right_gate_sig) * mask_val

        # Compute store pointers: each output has shape (total_rows, hidden)
        left_ptrs = left_ptr + (offs_m[:, None] * hidden + offs_h[None, :])
        right_ptrs = right_ptr + (offs_m[:, None] * hidden + offs_h[None, :])
        out_gate_ptrs = out_gate_ptr + (offs_m[:, None] * hidden + offs_h[None, :])

        tl.store(left_ptrs, left_out.to(tl.float16),
                 mask=mask_rows[:, None] & mask_h[None, :])
        tl.store(right_ptrs, right_out.to(tl.float16),
                 mask=mask_rows[:, None] & mask_h[None, :])
        tl.store(out_gate_ptrs, out_gate_sig.to(tl.float16),
                 mask=mask_rows[:, None] & mask_h[None, :])
```

Implementation:

- Compute mean/var per row over hidden dimension.

- Compute normalized values into accumulator maybe per hidden block.

- Multiply by gamma and add beta to get normalized_fp32.

- Multiply by gate (in half, convert to float32). This produce an intermediate vector (float32) of length hidden.

- Then the linear projection: compute dot product of this vector (hidden) with weight_out (dim x hidden). Could compute as GEMM: intermediate vector (row) times weight_out.T.

- While performing linear projection, we can compute: out = (normed_gated) @ weight_out.T ; This can be done using dot with weight_out transposed. Use loops over dim blocks.

Thus we need nested loops for hidden dim (BLOCK_H) and output dim (BLOCK_D) to compute accumulate.

Pseudo:

```
pid = tl.program_id(0)
row_start = pid * BLOCK_M
offs_m = row_start + tl.arange(0, BLOCK_M)
mask_m = offs_m < R

# First pass: mean and var across hidden dimension
sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
for h in range(0, hidden, BLOCK_H):
    cur_H = tl.minimum(hidden - h, BLOCK_H)
    offs_h = h + tl.arange(0, BLOCK_H)
    mask_h = offs_h < hidden
    vals = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)
    sum_val += tl.sum(vals, axis=1)
    sum_sq += tl.sum(vals * vals, axis=1)
mean = sum_val / hidden
var = sum_sq / hidden - mean * mean
rstd = tl.rsqrt(var + eps)

# Second pass: combine LN + gate + linear projection into final out
# We will compute the final out per dim block using accumulation.
for d_start in range(0, dim, BLOCK_D):
    cur_D = tl.minimum(dim - d_start, BLOCK_D)
    offs_d = d_start + tl.arange(0, BLOCK_D)

    # Initialize accumulator for this dim block
    acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

    # Loop over hidden dimension in BLOCK_H segments
    for h_start in range(0, hidden, BLOCK_H):
        cur_H = tl.minimum(hidden - h_start, BLOCK_H)
        offs_h = h_start + tl.arange(0, BLOCK_H)

        # Load x values for this hidden block
        x_vals = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                         mask=mask_m[:, None] & (offs_h < hidden)[None, :],
                         other=0.0).to(tl.float32)  # [M,H_block]

        # Compute normalized values (x - mean) * rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # Load gamma and beta for this hidden block
        gamma = tl.load(weight_ln_ptr + offs_h, mask=offs_h < hidden, other=0.0)
        beta = tl.load(bias_ln_ptr + offs_h, mask=offs_h < hidden, other=0.0)
        # Apply LN affine
        norm = norm * gamma[None, :] + beta[None, :]  # still float32

        # Load gate values for this hidden block
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                            mask=mask_m[:, None] & (offs_h < hidden)[None, :],
                            other=0.0).to(tl.float32)   # half->float32

        # Apply gate
        norm = norm * gate_vals   # float32

        # Load output weight slice for this hidden block and dim block: w_out_T shape (BLOCK_H, BLOCK_D)
        # weight_out shape (dim, hidden); we need its transpose (hidden, dim) -> we want w_out_T = weight_out.T (hidden, dim)
        # So pointer for w_out_T blocks:
        w_out_T = tl.load(weight_out_ptr + (offs_h[:, None] * dim + offs_d[None, :]),
                          mask=(offs_h[:, None] < hidden) & (offs_d[None, :] < dim),
                          other=0.0).to(tl.float32)  # shape (block_H, block_D)

        # Accumulate
        acc += tl.dot(norm, w_out_T) # (M, D)
    # store acc to output
    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             acc.to(tl.float16),
             mask=mask_m[:, None] & (offs_d[None, :] < dim))
```

Define:

```
@triton.jit
def proj_gate_fuse_kernel(
    x_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    w_left_ptr, w_right_ptr, w_left_gate_ptr, w_right_gate_ptr, w_out_gate_ptr,
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows
    # Load mask values per row
    mask_vals = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0).to(tl.float32)  # [M]

    for h_start in range(0, hidden, BLOCK_H):
        cur_h = tl.minimum(hidden - h_start, BLOCK_H)
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        acc_left = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_left_gate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_right_gate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_out_gate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        for d_start in range(0, dim, BLOCK_D):
            cur_d = tl.minimum(dim - d_start, BLOCK_D)
            offs_d = d_start + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # Load x block
            x = tl.load(x_ptr + (offs_m[:, None] * dim + offs_d[None, :]),
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)   # shape [M,D]

            # Load weight blocks for each projection as [D,H]
            # use transposed version: weight_T shape (D,H)
            w_left = tl.load(w_left_ptr + (offs_d[:, None] * 1 + offs_h[None, :] * dim),
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            w_right = tl.load(w_right_ptr + (offs_d[:, None] * 1 + offs_h[None, :] * dim),
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            w_left_gate = tl.load(w_left_gate_ptr + (offs_d[:, None] * 1 + offs_h[None, :] * dim),
                                  mask=mask_d[:, None] & mask_h[None, :],
                                  other=0.0).to(tl.float32)
            w_right_gate = tl.load(w_right_gate_ptr + (offs_d[:, None] * 1 + offs_h[None, :] * dim),
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0).to(tl.float32)
            w_out_gate = tl.load(w_out_gate_ptr + (offs_d[:, None] * 1 + offs_h[None, :] * dim),
                                 mask=mask_d[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float32)

            # Compute dot products
            acc_left += tl.dot(x, w_left)          # (M,H)
            acc_right += tl.dot(x, w_right)
            acc_left_gate += tl.dot(x, w_left_gate)
            acc_right_gate += tl.dot(x, w_right_gate)
            acc_out_gate += tl.dot(x, w_out_gate)

        # Apply sigmoid on gates
        left_gate = 1.0 / (1.0 + tl.exp(-acc_left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-acc_right_gate))
        out_gate = 1.0 / (1.0 + tl.exp(-acc_out_gate))

        # Apply gating to projections
        left = acc_left * left_gate
        right = acc_right * right_gate

        # Apply mask (broadcast)
        # mask_vals shape [M]; broadcast across hidden dim
        mask_vals_expand = mask_vals[:, None]   # [M,1]
        left = left * mask_vals_expand
        right = right * mask_vals_expand

        # Store results
        # compute pointers
        left_ptrs = left_ptr + (offs_m[:, None] * hidden + offs_h[None, :])
        right_ptrs = right_ptr + (offs_m[:, None] * hidden + offs_h[None, :])
        out_gate_ptrs = out_gate_ptr + (offs_m[:, None] * hidden + offs_h[None, :])

        tl.store(left_ptrs, left.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptrs, right.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptrs, out_gate.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
```

Make sure to adjust strides for weight loading: w_left_ptr + offs_d[:, None] * 1 + offs_h[None, :] * dim. Since weight shape [hidden, dim] row stride = dim, col stride =1; transposed view weight_T has stride 1 across hidden and dim across dim. The pointer expression is base + h_offset * stride_h + d_offset * stride_d where stride_h = 1, stride_d = dim gives shape (D,H). That's correct.

                           mask=mask_h,
                           other=0.0)

                                mask=mask_m[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            norm = norm * gate_vals  # [M, H_block]

            # load out weight block (hidden block, dim block) from transposed view
            w_out_T = tl.load(out_weight_ptr + (offs_h[:, None] * 1 + offs_d[None, :] * hidden),
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)   # shape (H_block, D_block)

            # accumulate dot product
            acc += tl.dot(norm, w_out_T)  # (M,D)

        # store final outputs
        out_ptrs = out_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])
```

    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]

    nomask = cfg.get("nomask", True)
    B, N, _, _ = inp.shape

    inp_flat = inp.view(total_rows, dim).contiguous()

    BLOCK_M = 64
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # Prepare mask_flat
    if nomask:
        mask_flat = torch.ones(total_rows, dtype=torch.half, device=device)
    else:
        mask_flat = mask.view(total_rows).to(torch.half)

    left_proj_w_T = weights["left_proj.weight"].t().contiguous().half()
    right_proj_w_T = weights["right_proj.weight"].t().contiguous().half()
    left_gate_w_T = weights["left_gate.weight"].t().contiguous().half()
    right_gate_w_T = weights["right_gate.weight"].t().contiguous().half()
    out_gate_w_T = weights["out_gate.weight"].t().contiguous().half()

        right_proj_w_T,

        right_gate_w_T,

"""
Highly‑optimized Triton forward for the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 reduction → FP16 output) – same kernel as before.
2️⃣  Fused projection + gating + optional mask:
    * Input: LN‑output `[R, C]` (FP16) where `R = B·N·N`.
    * Compute in one pass the five linear maps
          (left, right, left‑gate, right‑gate, out‑gate)
      needed for the TriMul update.
    * Apply sigmoid to the three gates, multiply left/right by their
      respective gates and by the (binary) mask.
    * Outputs: `left`, `right`, `out_gate` all `[R, H]` (FP16).
3️⃣  Triangular multiplicative update – batched `torch.bmm`
    on reshaped tensors (fully tensor‑core‑accelerated).
4️⃣  Fused second LayerNorm + out‑gate multiplication + final projection:
    * Input: the raw TriMul result `[R, H]` (FP16) and `out_gate` (FP16).
    * Compute LayerNorm over `H`, apply the learned γ/β,
      multiply by `out_gate`, then a single GEMM with the
      `to_out` weight (`[D, H]`) → final tensor `[R, D]` (FP16).
5️⃣  Cast final result back to FP32 and reshape to `[B, N, N, D]`.

The implementation removes the large intermediate `(R,5·H)` tensor used in
the reference code and merges two LayerNorm + Linear stages, cutting
memory traffic and kernel launches while keeping the same mathematical
semantics.
"""

# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim]  (input FP32)
    y_ptr,               # *mut   half*   [total_rows, dim]  (output FP16)
    weight_ptr,          # *const float*  [dim]             (γ)
    bias_ptr,            # *const float*  [dim]             (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static dimension (≤ 384)
    eps: tl.constexpr,   # layer‑norm epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # Pass 1 – mean / variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Pass 2 – normalise, apply γ / β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]          # FP32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2) Fused projection + gates + mask (produces left/right/out_gate)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fuse_kernel(
    x_ptr,                # *const half*  [R, C]   (input from LN)
    mask_ptr,             # *const half*  [R]      (1.0 if no mask)
    left_ptr,             # *mut   half*  [R, H]   (output)
    right_ptr,            # *mut   half*  [R, H]   (output)
    out_gate_ptr,         # *mut   half*  [R, H]   (sigmoid)
    w_left_ptr,           # *const half*  [C, H]   (transposed weight)
    w_right_ptr,
    w_left_gate_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    R: tl.int32,
    C: tl.constexpr,
    H: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # broadcast mask value per row (scalar)
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0).to(tl.float32)
    mask_val = mask_val[:, None]          # [M,1]

    # -----------------------------------------------------------------
    # Loop over hidden dimension in blocks
    # -----------------------------------------------------------------
    for hb in range(0, tl.cdiv(H, BLOCK_H)):
        h_start = hb * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators for the five linear maps
        acc_left = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_left_gate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_right_gate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        acc_out_gate = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        # -----------------------------------------------------------------
        # Loop over input channel dimension (C)
        # -----------------------------------------------------------------
        for db in range(0, tl.cdiv(C, BLOCK_D)):
            d_start = db * BLOCK_D
            offs_d = d_start + tl.arange(0, BLOCK_D)
            mask_d = offs_d < C

            # Load a block of the normalized input
            x = tl.load(
                x_ptr + offs_m[:, None] * C + offs_d[None, :],
                mask=mask_m[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                     # [M, D]

            # Load the five weight blocks (shape D×H after transposition)
            w_left = tl.load(
                w_left_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)                     # [D, H]

            w_right = tl.load(
                w_right_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_left_gate = tl.load(
                w_left_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_right_gate = tl.load(
                w_right_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            w_out_gate = tl.load(
                w_out_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # Matrix‑multiply the input block with each weight block
            acc_left += tl.dot(x, w_left)           # (M, H)
            acc_right += tl.dot(x, w_right)
            acc_left_gate += tl.dot(x, w_left_gate)
            acc_right_gate += tl.dot(x, w_right_gate)
            acc_out_gate += tl.dot(x, w_out_gate)

        # ---------------------------------------------------------------
        # Apply sigmoid gates and mask, then store
        # ---------------------------------------------------------------
        left_gate = 1.0 / (1.0 + tl.exp(-acc_left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-acc_right_gate))
        out_gate = 1.0 / (1.0 + tl.exp(-acc_out_gate))

        left = acc_left * left_gate * mask_val
        right = acc_right * right_gate * mask_val

        # store
        left_ptrs = left_ptr + offs_m[:, None] * H + offs_h[None, :]
        right_ptrs = right_ptr + offs_m[:, None] * H + offs_h[None, :]
        out_gate_ptrs = out_gate_ptr + offs_m[:, None] * H + offs_h[None, :]

        tl.store(
            left_ptrs,
            left.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptrs,
            right.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptrs,
            out_gate.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )

# ----------------------------------------------------------------------
@triton.jit
def final_ln_gate_linear_kernel(
    x_ptr,                # *const half*  [R, H]   (TriMul output)
    gate_ptr,             # *const half*  [R, H]   (out_gate sigmoid)
    ln_weight_ptr,        # *const float* [H]      (γ of to_out_norm)
    ln_bias_ptr,          # *const float* [H]      (β of to_out_norm)
    out_weight_ptr,       # *const half*  [D, H]   (to_out weight)
    R: tl.int32,
    H: tl.constexpr,
    D: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # -----------------------------------------------------------------
    # First pass: compute mean / variance over hidden dim (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for hb in range(0, tl.cdiv(H, BLOCK_H)):
        h_start = hb * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        vals = tl.load(
            x_ptr + offs_m[:, None] * H + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: normalise, apply LN affine, gate, then linear proj
    # -----------------------------------------------------------------
    for db in range(0, tl.cdiv(D, BLOCK_D)):
        d_start = db * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # accumulator for this output‑dim block
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for hb in range(0, tl.cdiv(H, BLOCK_H)):
            h_start = hb * BLOCK_H
            offs_h = h_start + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # load raw value
            vals = tl.load(
                x_ptr + offs_m[:, None] * H + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # layer‑norm normalisation
            norm = (vals - mean[:, None]) * rstd[:, None]

            # apply LN affine parameters (γ,β) over H
            gamma = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)
            beta = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0)
            norm = norm * gamma[None, :] + beta[None, :]   # [M, H_block]

            # multiply by previously computed out_gate
            gate = tl.load(
                gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            norm = norm * gate

            # load slice of the final linear weight (shape H×D after transposition)
            # out_weight shape = [D, H]; we need a (H_block, D_block) slice.
            w = tl.load(
                out_weight_ptr + offs_h[:, None] * 1 + offs_d[None, :] * H,
                mask=mask_h[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)                     # [H_block, D_block]

            # accumulate contribution
            acc += tl.dot(norm, w)                # (M, D_block)

        # store final projected values
        out_ptrs = out_weight_ptr.parent + offs_m[:, None] * D + offs_d[None, :]  # reinterpret as out buffer
        tl.store(
            out_weight_ptr.parent + offs_m[:, None] * D + offs_d[None, :],
            acc.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )

    """

        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : model parameters (float32)

    -------

    """

    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------

    BLOCK_D = 64

        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # 2️⃣ Prepare mask (broadcasted) – 1.0 if no mask
    # ------------------------------------------------------------------

        mask_flat = torch.ones(total_rows, dtype=torch.half, device=device)
    else:
        mask_flat = mask.view(total_rows).to(torch.half)

    # 2️⃣ Transpose projection/gate weights for fast Triton access
    # ------------------------------------------------------------------

    w_right_T = weights["right_proj.weight"].t().contiguous().half()
    w_left_gate_T = weights["left_gate.weight"].t().contiguous().half()
    w_right_gate_T = weights["right_gate.weight"].t().contiguous().half()
    w_out_gate_T = weights["out_gate.weight"].t().contiguous().half()

    # Allocate buffers for fused projection output
    # ------------------------------------------------------------------
    left_fp16 = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    right_fp16 = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    out_gate_fp16 = torch.empty(total_rows, hidden, dtype=torch.half, device=device)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating + mask
    # ------------------------------------------------------------------
    BLOCK_H = 32
    proj_grid = (triton.cdiv(total_rows, BLOCK_M),)
    proj_gate_fuse_kernel[proj_grid](
        x_norm_fp16,
        mask_flat,
        left_fp16,
        right_fp16,
        out_gate_fp16,
        w_left_T,
        w_right_T,
        w_left_gate_T,
        w_right_gate_T,
        w_out_gate_T,
        total_rows,
        dim,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden)             # (B,N,N,H)
    right = right_fp16.view(B, N, N, hidden)           # (B,N,N,H)

    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                        # (B*H, N, N)

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16
    final_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    final_grid = (triton.cdiv(total_rows, BLOCK_M),)
    final_ln_gate_linear_kernel[final_grid](
        out_flat,
        out_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        weights["to_out.weight"].t().contiguous().half(),   # transpose -> [D, H] already
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # Cast back to FP32 and reshape
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
