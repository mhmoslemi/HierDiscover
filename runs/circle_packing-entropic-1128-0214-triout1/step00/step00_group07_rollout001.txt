```
def compute_optimal_radii_and_sum(centers):
    n = len(centers)
    # distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.linalg.norm(diff, axis=2)
    # Build A_ub, b_ub
    rows = []
    b = []
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            b.append(dists[i, j])
    # boundary constraints
    for i in range(n):
        xi, yi = centers[i]
        # left
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(xi)
        # right
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(1 - xi)
        # bottom
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(yi)
        # top
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(1 - yi)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback maybe set radii zero
        radii = np.zeros(n)
        sum_r = 0.0
    else:
        radii = res.x
        sum_r = -res.fun
    return radii, sum_r
```

```
def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    np.random.seed(12345)

    n = 26

    # helper: base solve for radii & sum
    def solve_radii(centers):
        n = centers.shape[0]
        # compute distances
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        # pairwise constraints
        pair_indices = [(i, j) for i in range(n) for j in range(i+1, n)]
        m_pair = len(pair_indices)
        m_total = m_pair + 4*n
        A = np.zeros((m_total, n))
        b = np.zeros(m_total)

        # fill pair constraints
        for idx, (i, j) in enumerate(pair_indices):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = dists[i, j]
        # boundary constraints
        offset = m_pair
        for i in range(n):
            x, y = centers[i]
            # left
            A[offset, i] = 1
            b[offset] = x
            offset += 1
            # right
            A[offset, i] = 1
            b[offset] = 1 - x
            offset += 1
            # bottom
            A[offset, i] = 1
            b[offset] = y
            offset += 1
            # top
            A[offset, i] = 1
            b[offset] = 1 - y
            offset += 1

        c = -np.ones(n)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            # fallback radii are zeros
            radii = np.zeros(n)
            sum_r = 0.0
        else:
            radii = np.maximum(res.x, 0.0)
            sum_r = float(-res.fun)
        return radii, sum_r

    # Heuristic optimization
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Number of restarts
    for restart in range(5):
        # initial arrangement: a perturbed grid
        xs = np.linspace(0.1, 0.9, 5)
        ys = np.linspace(0.1, 0.9, 5)
        grid = np.array([(x, y) for y in ys for x in xs])
        # add extra random point in interior
        extra = np.random.rand(1,2) * 0.8 + 0.1  # within [0.1,0.9]
        centers = np.vstack([grid, extra])

        # apply small random jitter
        centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
        centers = np.clip(centers, 0.0, 1.0)

        radii, cur_sum = solve_radii(centers)

        cur_centers = centers.copy()
        cur_radii = radii.copy()
        cur_sum = cur_sum

        # keep track best
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers.copy()
            best_radii = cur_radii.copy()

        # simulated annealing
        step = 0.15
        temperature = 0.02
        decay = 0.9995
        iters = 3000
        for it in range(iters):
            # propose move
            i = np.random.randint(n)
            new_center = cur_centers[i] + np.random.uniform(-step, step, size=2)
            new_center = np.clip(new_center, 0.0, 1.0)
            new_centers = cur_centers.copy()
            new_centers[i] = new_center

            new_radii, new_sum = solve_radii(new_centers)

            delta = new_sum - cur_sum
            if delta > 0 or np.random.rand() < np.exp(delta / max(temperature, 1e-12)):

                cur_sum = new_sum

                    best_radii = cur_radii.copy()
```

```
import numpy as np
from scipy.optimize import linprog

def optimal_radii_and_sum(centers):
    n = centers.shape[0]
    # distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # pair constraints
    pair_list = [(i, j) for i in range(n) for j in range(i+1, n)]
    m_pair = len(pair_list)
    m_total = m_pair + 4*n
    A = np.zeros((m_total, n))
    b = np.zeros(m_total)
    # fill pair constraints
    for idx, (i, j) in enumerate(pair_list):
        A[idx, i] = 1
        A[idx, j] = 1
        b[idx] = dists[i, j]
    # boundary constraints
    offset = m_pair
    for i in range(n):
        x, y = centers[i]
        A[offset, i] = 1
        b[offset] = x           # left
        offset += 1
        A[offset, i] = 1
        b[offset] = 1 - x       # right
        offset += 1
        A[offset, i] = 1
        b[offset] = y           # bottom
        offset += 1
        A[offset, i] = 1
        b[offset] = 1 - y       # top
        offset += 1

    c = -np.ones(n)   # maximize sum, so minimize negative sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zeros
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = np.maximum(res.x, 0.0)
        total = float(-res.fun)
    return radii, total
```

```
    np.random.seed(0)

    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    max_restarts = 5
    for restart in range(max_restarts):
        centers = initial_grid_centers()

        centers += np.random.uniform(-0.02, 0.02, size=centers.shape)

        radii, cur_sum = optimal_radii_and_sum(centers)
        cur_centers = centers.copy()

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers.copy()
            best_radii = cur_radii.copy()

        # simulated annealing / hill climbing
        step = 0.15
        temperature = 0.02
        decay = 0.9995
        iterations = 4000
        for it in range(iterations):
            i = np.random.randint(n)
            proposal = cur_centers[i] + np.random.uniform(-step, step, size=2)
            proposal = np.clip(proposal, 0.0, 1.0)
            new_centers = cur_centers.copy()
            new_centers[i] = proposal
            new_radii, new_sum = optimal_radii_and_sum(new_centers)

            delta = new_sum - cur_sum
            if delta > 0 or np.random.rand() < np.exp(delta / max(temperature, 1e-12)):
                cur_centers = new_centers
                cur_radii = new_radii
                cur_sum = new_sum
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = cur_centers.copy()
                    best_radii = cur_radii.copy()

            step *= decay
            temperature *= decay

    # final refinement: re-optimize radii for best centers
    best_radii, best_sum = optimal_radii_and_sum(best_centers)

    return best_centers, best_radii, best_sum
```

The objective is  

\[

\]

subject to  

* circles stay inside the unit square  

  `x_i – r_i ≥ 0 ,  x_i + r_i ≤ 1`  
  `y_i – r_i ≥ 0 ,  y_i + r_i ≤ 1`

* circles do not overlap  

  \(\sqrt{(x_i-x_j)^2+(y_i-y_j)^2}\; \ge\; r_i+r_j\) for alli<j.

For a *fixed* set of centre positions the radii appear only linearly
in the constraints, therefore the sub‑problem “maximise the sum of
radii for given centres” is a **linear program (LP)**.
The complete packing problem is tackled with a very simple
meta‑heuristic:

1. start from a reasonable arrangement (a \(5\times5\) square grid,
   plus one extra point in the interior);
2. repeatedly move a randomly chosen centre a little, solve the LP
   for the new positions and keep the move if it improves the total
   radius (or with a small probability to escape local minima);
3. perform a few independent restarts and keep the best solution
   ever found.

All heavy lifting (the LP) is performed by `scipy.optimize.linprog`
with the high‑performance HiGHS backend, which solves every LP
instantly for 26 variables.

```python
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Linear programme: for a given set of centres compute the
# maximal radii and the corresponding sum of radii.
# ------------------------------------------------------------
def optimal_radii_and_sum(centres: np.ndarray):
    """
    Solve the LP
        maximise   sum(r_i)
        subject to r_i >= 0,
                   centre_i +/- r_i stay inside the unit square,
                   distance(centre_i, centre_j) >= r_i + r_j .
    Returns the radii (array of length n) and the achieved sum (float).
    """
    n = centres.shape[0]

    # pairwise distances, needed for the constraints r_i + r_j <= d_ij
    diff = centres[:, None, :] - centres[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # indices of the upper‑triangular part (i<j)
    i_idx, j_idx = np.triu_indices(n, k=1)
    n_pairs = i_idx.shape[0]

    # total number of inequality constraints:
    #   one for each pair  + 4 per circle (left, right, bottom, top)
    m = n_pairs + 4 * n

    # build the constraint matrix A_ub * r <= b_ub
    A = np.zeros((m, n), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)

    # pair constraints: r_i + r_j <= d_ij
    rows = np.arange(n_pairs)
    A[rows, i_idx] = 1.0
    A[rows, j_idx] = 1.0
    b[rows] = dists[i_idx, j_idx]

    # boundary constraints
    off = n_pairs
    for i in range(n):
        x, y = centres[i]

        A[off, i] = 1.0          # x - r >= 0   ->  r <= x
        b[off] = x
        off += 1

        A[off, i] = 1.0          # x + r <= 1   ->  r <= 1 - x
        b[off] = 1.0 - x
        off += 1

        A[off, i] = 1.0          # y - r >= 0   ->  r <= y
        b[off] = y
        off += 1

        A[off, i] = 1.0          # y + r <= 1   ->  r <= 1 - y
        b[off] = 1.0 - y
        off += 1

    # linear objective (maximise sum r_i  <=> minimise -sum r_i)
    c = -np.ones(n, dtype=np.float64)

    # variable bounds: radii are non‑negative, no explicit upper bound
    bounds = [(0.0, None) for _ in range(n)]

    # solve with the HiGHS backend (very fast for this size)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    if not res.success:
        # fallback – all radii zero (should never happen)
        radii = np.zeros(n, dtype=np.float64)
        total = 0.0
    else:
        radii = np.maximum(res.x, 0.0)   # numerical safety
        total = float(-res.fun)

    return radii, total


# ------------------------------------------------------------
# Simple initializer – a regular 5×5 square grid plus one random point.
# ------------------------------------------------------------
def initial_centres():
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    grid = np.array([(x, y) for y in ys for x in xs])   # 25 points
    extra = np.random.rand(1, 2) * 0.8 + 0.1            # somewhere inside [0.1,0.9]^2
    return np.vstack([grid, extra])


# ------------------------------------------------------------
# Main routine required by the problem statement.
# ------------------------------------------------------------
def run_packing():
    """
    Returns (centres, radii, sum_of_radii) for a packing of 26 circles.
    The routine uses a small simulated‑annealing loop together with the
    linear programme defined above.
    """
    np.random.seed(0)          # deterministic result

    n = 26
    best_sum = -1.0
    best_centres = None
    best_radii = None

    # ---- several independent restarts ---------------------------------
    for restart in range(5):
        centres = initial_centres()
        # small initial jitter, still inside the square
        centres += np.random.uniform(-0.02, 0.02, size=centres.shape)
        centres = np.clip(centres, 0.0, 1.0)

        radii, cur_sum = optimal_radii_and_sum(centres)
        cur_centres = centres.copy()

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centres = cur_centres.copy()
            best_radii = radii.copy()

        # ---- simple simulated annealing --------------------------------
        step = 0.25            # maximal move in each coordinate
        temperature = 0.5
        decay = 0.9997
        iterations = 6000

        for it in range(iterations):
            i = np.random.randint(n)                     # which circle to move
            trial = cur_centres[i] + np.random.uniform(-step, step, size=2)
            trial = np.clip(trial, 0.0, 1.0)               # stay inside the square

            new_centres = cur_centres.copy()
            new_centres[i] = trial

            new_radii, new_sum = optimal_radii_and_sum(new_centres)
            delta = new_sum - cur_sum

            # Metropolis acceptance
            if delta > 0.0 or np.random.rand() < np.exp(delta / max(temperature, 1e-12)):
                cur_centres = new_centres
                cur_sum = new_sum
                # keep global best
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centres = cur_centres.copy()
                    best_radii = new_radii.copy()

            # gradual cooling
            step *= decay
            temperature *= decay

    # final LP evaluation to guarantee that the radii correspond exactly
    # to the best centre positions we have stored
    best_radii, best_sum = optimal_radii_and_sum(best_centres)

    return best_centres, best_radii, best_sum
```
