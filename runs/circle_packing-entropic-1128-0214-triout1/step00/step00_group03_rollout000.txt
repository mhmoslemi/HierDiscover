```
def solve_radii(centers):
    n = centers.shape[0]
    # compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]  # shape (n,n,2)
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # For each pair i<j
    pair_indices = np.triu_indices(n, k=1)
    d_ij = dists[pair_indices]
    m = len(d_ij)
    # Build A_ub: m + n rows, n cols
    A = np.zeros((m + n, n))
    # pair constraints: r_i + r_j <= d_ij
    rows = np.arange(m)
    i_idx = pair_indices[0]
    j_idx = pair_indices[1]
    A[rows, i_idx] = 1
    A[rows, j_idx] = 1
    # boundary constraints: r_i <= b_i
    b = np.concatenate([d_ij, np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])])
    for i in range(n):
        A[m + i, i] = 1
    # Objective
    c = -np.ones(n)
    # Solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # Possibly due to numerical; fallback radius = zeros
        radii = np.zeros(n)
        sum_r = 0.0
    else:
        radii = res.x
        sum_r = np.sum(radii)
    return radii, sum_r
```

```
def optimize_positions(n, max_iter=20000, temp0=0.1, seed=0):
    np.random.seed(seed)
    # initial grid: 5x5 grid (25), plus 1 extra random
    grid_n = int(np.floor(np.sqrt(n)))   # maybe 5
    # we need exactly 26 points: generate grid positions for first 25
    coords = []
    for i in range(grid_n):
        for j in range(grid_n):
            coords.append([(i+0.5)/grid_n, (j+0.5)/grid_n])
    # extra point randomly:
    extra = np.random.rand(2)
    coords.append(extra)
    centers = np.array(coords[:n])   # shape (26,2)
    # Solve radii
    radii, best_sum = solve_radii(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    temp = temp0
    for it in range(max_iter):
        # pick random index
        i = np.random.randint(n)
        # propose new center
        new_center = centers[i].copy()
        # step size decays
        sigma = 0.1 * (1 - it / max_iter) + 0.01
        new_center += np.random.randn(2) * sigma
        # clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        # keep others
        new_centers = centers.copy()
        new_centers[i] = new_center
        # solve
        radii_new, sum_new = solve_radii(new_centers)
        # accept if better or SA
        if sum_new > best_sum or np.random.rand() < np.exp((sum_new - best_sum) / (temp+1e-12)):
            centers = new_centers
            if sum_new > best_sum:
                best_sum = sum_new
                best_centers = new_centers.copy()
                best_radii = radii_new.copy()
        # update temperature
        temp = temp0 * (1 - it / max_iter)
    return best_centers, best_radii, best_sum
```

```
import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    pair_i, pair_j = np.triu_indices(n, k=1)
    d_ij = np.sqrt(((centers[pair_i] - centers[pair_j]) ** 2).sum(axis=1))
    m = d_ij.shape[0]
    # Build A_ub
    A = np.zeros((m + n, n))
    A[np.arange(m), pair_i] = 1
    A[np.arange(m), pair_j] = 1
    A[m:, np.arange(n)] = 1
    # Build b_ub
    b_boundary = np.minimum.reduce([centers[:,0], 1-centers[:,0],
                                    centers[:,1], 1-centers[:,1]])
    b = np.concatenate([d_ij, b_boundary])
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        radii = res.x
        sum_r = radii.sum()
    else:
        radii = np.zeros(n)
        sum_r = 0.0
    return radii, sum_r
```

```python
import numpy as np
from scipy.optimize import linprog
from typing import Tuple


def _solve_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    For a fixed set of centre coordinates solve the linear program that
    maximises the sum of the radii.
    Returns the optimal radii and their sum.
    """
    n = centers.shape[0]
    # pairwise distances (upper triangular)
    i_idx, j_idx = np.triu_indices(n, k=1)
    dij = np.linalg.norm(centers[i_idx] - centers[j_idx], axis=1)

    # ----- build A_ub and b_ub -----
    m = dij.shape[0]                     # number of pair constraints
    A = np.zeros((m + n, n))
    # r_i + r_j ≤ d_ij
    rows = np.arange(m)
    A[rows, i_idx] = 1.0
    A[rows, j_idx] = 1.0
    # boundary constraints r_i ≤ b_i
    A[m:, np.arange(n)] = 1.0

    # b_ub vector
    b_boundary = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0],
         centers[:, 1], 1.0 - centers[:, 1]]
    )
    b = np.concatenate([dij, b_boundary])

    # objective: maximise Σ r_i  → minimise -Σ r_i
    c = -np.ones(n)

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0.0, None)] * n,
                  method='highs')
    if not res.success:
        # fallback – everything zero (still feasible)
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = res.x
        total = radii.sum()
    return radii, total


def _initial_grid() -> np.ndarray:
    """5×5 regular grid (25 points) plus one random point → 26 points."""
    k = 5
    step = 1.0 / k
    coords = (np.arange(k) + 0.5) * step
    xv, yv = np.meshgrid(coords, coords)
    grid = np.column_stack([xv.ravel(), yv.ravel()])          # 25×2
    extra = np.random.rand(1, 2)                             # one random point
    centres = np.vstack([grid, extra])
    return centres[:26]


def _initial_two_big() -> np.ndarray:
    """Two large circles at (0.25,0.5) and (0.75,0.5) + 24 random points."""
    big = np.array([[0.25, 0.5],
                    [0.75, 0.5]])
    rest = np.random.rand(24, 2)
    centres = np.vstack([big, rest])
    return centres[:26]


def _initial_random() -> np.ndarray:
    """Completely random placement."""
    return np.random.rand(26, 2)


def _make_initial(seed: int) -> np.ndarray:
    """Choose an initial layout based on the seed."""
    np.random.seed(seed)
    mode = seed % 3
    if mode == 0:
        return _initial_grid()
    elif mode == 1:
        return _initial_two_big()
    else:
        return _initial_random()


def _optimize_from(start_centers: np.ndarray,
                   max_iter: int = 2000,
                   seed: int = 0) -> Tuple[np.ndarray, np.ndarray, float]:
    """Simulated‑annealing optimisation of the centre positions."""
    rng = np.random.RandomState(seed)
    n = start_centers.shape[0]

    centres = start_centers.copy()
    radii_cur, cur_sum = _solve_radii(centres)

    best_centres = centres.copy()
    best_radii = radii_cur.copy()
    best_sum = cur_sum

    temp0 = 0.1

    for it in range(max_iter):
        # temperature decreasing linearly
        temp = temp0 * (1.0 - it / max_iter)

        # ------- propose a move (single centre) -------
        i = rng.randint(n)
        sigma = 0.05 * (1.0 - it / max_iter) + 0.005
        trial = centres[i] + rng.randn(2) * sigma
        trial = np.clip(trial, 0.0, 1.0)

        new_centres = centres.copy()
        new_centres[i] = trial
        radii_new, sum_new = _solve_radii(new_centres)

        # ------- acceptance test -------
        accept = False
        if sum_new > cur_sum:
            accept = True
        else:
            # Metropolis criterion
            if temp > 0 and rng.rand() < np.exp((sum_new - cur_sum) / temp):
                accept = True

        if accept:
            centres = new_centres
            radii_cur = radii_new
            cur_sum = sum_new
            if sum_new > best_sum:
                best_sum = sum_new
                best_centres = new_centres.copy()
                best_radii = radii_new.copy()

        # occasional global jitter to keep diversity
        if (it + 1) % 500 == 0:
            jitter = (rng.rand(n, 2) - 0.5) * sigma
            centres = np.clip(centres + jitter, 0.0, 1.0)
            radii_cur, cur_sum = _solve_radii(centres)
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centres = centres.copy()
                best_radii = radii_cur.copy()

    return best_centres, best_radii, best_sum


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Main entry point required by the evaluation harness.
    Returns (centres, radii, sum_of_radii) for a packing of 26 circles
    inside the unit square.
    """
    n_circles = 26
    overall_best_sum = -1.0
    overall_best_centres = None
    overall_best_radii = None

    # a few diversified starts
    seeds = [42, 123, 987, 2021, 0]
    for s in seeds:
        init = _make_initial(s)
        centres_opt, radii_opt, sum_opt = _optimize_from(init,
                                                        max_iter=2000,
                                                        seed=s + 1)
        if sum_opt > overall_best_sum:
            overall_best_sum = sum_opt
            overall_best_centres = centres_opt
            overall_best_radii = radii_opt

        # early exit if we already beat the advertised target
        if overall_best_sum > 2.63:
            break

    # safety margin to avoid any numerical violation
    eps = 1e-9
    final_radii = np.maximum(overall_best_radii - eps, 0.0)
    final_sum = float(final_radii.sum())

    return overall_best_centres, final_radii, final_sum
```
