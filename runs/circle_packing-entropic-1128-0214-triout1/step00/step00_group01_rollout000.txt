```
def solve_lp(centers):
    n = centers.shape[0]
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]  # shape (n,n,2)
    dists = np.sqrt(np.sum(diff**2, axis=2))
    A = []
    b = []

    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(dists[i,j])
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(x)
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(1 - x)
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(y)
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(1 - y)
    A = np.array(A)
    b = np.array(b)
    c = -np.ones(n)  # maximize sum r_i
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback return zeros
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = radii.sum()
    return radii, sum_r
```

    best_radii = None

    # set parameters
    min_dist = 0.15

def solve_lp(centers):
    n = centers.shape[0]
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    A = []
    b = []
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            b.append(dists[i, j])
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # left
        row = np.zeros(n)
        row[i] = 1.0
        A.append(row)
        b.append(x)
        # right
        row = np.zeros(n)
        row[i] = 1.0
        A.append(row)
        b.append(1.0 - x)
        # bottom
        row = np.zeros(n)
        row[i] = 1.0
        A.append(row)
        b.append(y)
        # top
        row = np.zeros(n)
        row[i] = 1.0
        A.append(row)
        b.append(1.0 - y)
    A = np.array(A, dtype=float)
    b = np.array(b, dtype=float)
    c = -np.ones(n, dtype=float)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')

        # fallback all zeros
        radii = np.zeros(n)

import numpy as np
from scipy.optimize import linprog
from typing import Tuple

def _solve_lp(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Solve the linear programme that maximises the sum of radii for a fixed
    set of circle centres.
    Returns the radii and the total sum.
    """
    n = centers.shape[0]

    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    rows = []
    rhs = []

    # r_i + r_j <= d_ij  for i < j
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dists[i, j])

    # boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]

        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(x)               # left wall

        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - x)         # right wall

        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(y)               # bottom wall

        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - y)         # top wall

    A = np.array(rows, dtype=float)
    b = np.array(rhs, dtype=float)

    c = -np.ones(n, dtype=float)  # maximise sum -> minimise negative sum
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – should never happen because r_i = 0 is always feasible
        radii = np.zeros(n, dtype=float)
        total = 0.0
    else:
        radii = res.x
        total = float(radii.sum())
    return radii, total


def _generate_small_centres(num: int,
                            min_dist: float,
                            wall_clearance: float,
                            big_centres: np.ndarray,
                            big_radius: float,
                            rng: np.random.Generator) -> np.ndarray:

    Poisson‑disk style sampler for the small circles.
    Points are required to be:
       • at least `wall_clearance` away from each side of the unit square,
       • at least `big_radius + wall_clearance` away from each big centre,
       • at least `min_dist` away from any previously generated small centre.
    """
    points = []
    attempts = 0
    max_attempts = 20000
    # required distance from a big centre so that a small radius of size
    # `wall_clearance` is still feasible
    min_dist_to_big = big_radius + wall_clearance

    while len(points) < num and attempts < max_attempts:
        cand = rng.random(2)                     # uniform in [0,1)
        # keep away from the four walls
        if (cand < wall_clearance).any() or (cand > 1.0 - wall_clearance).any():
            attempts += 1
            continue
        # keep away from the big circles
        if any(np.linalg.norm(cand - bc) < min_dist_to_big for bc in big_centres):
            attempts += 1
            continue
        # keep away from already placed small centres
        if points:
            pts = np.asarray(points)
            if np.min(np.linalg.norm(pts - cand, axis=1)) < min_dist:
                attempts += 1
                continue
        points.append(cand)
        attempts = 0  # reset after a successful placement
    return np.asarray(points, dtype=float)


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, sum_radii) for 26 circles.
    The packing is constructed as described in the solution explanation.
    """
    rng = np.random.default_rng(12345)   # deterministic randomness

    # ---- 1. the four large circles (fixed) ---------------------------------
    big_radius = 0.25
    big_centres = np.array([[0.25, 0.25],
                            [0.75, 0.25],
                            [0.25, 0.75],
                            [0.75, 0.75]], dtype=float)

    N_TOTAL = 26
    N_BIG = big_centres.shape[0]
    N_SMALL = N_TOTAL - N_BIG          # 22

    # Parameters for the small circles
    wall_clearance = 0.09              # we would like them to reach radius ~0.09
    min_pairwise = 0.18                # guarantees radius ≤ min_pairwise/2

    best_sum = -1.0
    best_centres = None
    best_radii = None

    # ---- 2. random search for a good set of small centres ------------------
    for _ in range(300):                 # number of independent trials
        small = _generate_small_centres(N_SMALL,
                                        min_pairwise,
                                        wall_clearance,
                                        big_centres,
                                        big_radius,
                                        rng)
        if small.shape[0] != N_SMALL:
            # could not place enough points in this trial – skip it
            continue

        centres = np.vstack([big_centres, small])
        radii, total = _solve_lp(centres)

        if total > best_sum:
            best_sum = total
            best_centres = centres.copy()
            best_radii = radii.copy()
            if best_sum >= 2.636:       # early exit if we already beat the target
                break

    # ---- 3. local refinement (tiny moves of the small centres) ------------
    if best_centres is None:  # extremely unlikely, but keep the function safe
        # fallback: just the four big circles + zeros
        radii = np.zeros(N_TOTAL)
        radii[:N_BIG] = big_radius
        return np.vstack([big_centres,
                          np.zeros((N_SMALL, 2))]), radii, radii.sum()

    # only tweak the small circles (indices N_BIG .. 25)
    small_indices = np.arange(N_BIG, N_TOTAL)
    centres = best_centres.copy()
    radii = best_radii.copy()
    current_sum = best_sum

    for _ in range(200):                 # hill‑climbing steps
        i = rng.choice(small_indices)    # pick a small circle
        old_pos = centres[i].copy()
        # propose a tiny move, keep inside the unit square
        proposal = old_pos + rng.normal(scale=0.02, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)
        centres[i] = proposal

        new_radii, new_sum = _solve_lp(centres)
        if new_sum > current_sum:
            # keep the improvement
            radii = new_radii
            current_sum = new_sum
        else:
            # revert
            centres[i] = old_pos

    return centres, radii, float(current_sum)
