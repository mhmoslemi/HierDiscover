def solve_lp(centers):
    n = centers.shape[0]
    # distances upper bound matrix
    dists = np.linalg.norm(centers[:,None,:] - centers[None,:,:], axis=2)
    # Build constraints
    rows = []
    cols = []
    data = []
    b = []
    # wall constraints
    for i in range(n):
        # r <= x
        rows.append(len(b)); cols.append(i); data.append(1); b.append(centers[i,0])
        # r <= 1-x
        rows.append(len(b)); cols.append(i); data.append(1); b.append(1 - centers[i,0])
        # r <= y
        rows.append(len(b)); cols.append(i); data.append(1); b.append(centers[i,1])
        # r <= 1-y
        rows.append(len(b)); cols.append(i); data.append(1); b.append(1 - centers[i,1])
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            rows.append(len(b)); cols.extend([i,j]); data.extend([1,1]); b.append(dists[i,j])

import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    # distances matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build A_ub and b_ub
    num_cons = 4*n + n*(n-1)//2
    A = np.zeros((num_cons, n))
    b = np.empty(num_cons)
    row = 0
    for i in range(n):
        A[row, i] = 1; b[row] = centers[i,0]; row+=1
        A[row, i] = 1; b[row] = 1 - centers[i,0]; row+=1
        A[row, i] = 1; b[row] = centers[i,1]; row+=1
        A[row, i] = 1; b[row] = 1 - centers[i,1]; row+=1
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dists[i,j]
            row+=1
    c = -np.ones(n)  # maximize sum -> minimize negative sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero radii
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = res.x
        total = -res.fun
    return radii, total

**Goal** – Find 26 circles inside the unit square `[0,1]×[0,1]` that maximise the sum of their radii.

**Idea** –  
For a *fixed* set of centre points the optimal radii are obtained by the linear program  

```
max   Σ r_i
s.t.  r_i ≥ 0
      r_i ≤ x_i               (distance to the left side)
      r_i ≤ 1‑x_i             (distance to the right side)
      r_i ≤ y_i               (distance to the bottom side)
      r_i ≤ 1‑y_i             (distance to the top side)
      r_i + r_j ≤ dist(i,j)   (no overlap)
```

All constraints are linear in the radii, therefore the LP can be solved quickly with `scipy.optimize.linprog`.  
If the centres are well‑spread, the radii become ≈ ½ · (minimum distance to a neighbour or to a wall).  
The larger the pairwise distances, the larger the radii and the larger the total sum.

**Search strategy**

1. **Generate many candidate centre sets** –  
   * a far‑thest‑point sampling (a simple greedy “max‑min” construction) that produces a fairly uniform point set,  
   * random jittered subsets of a coarse grid (to explore slightly different geometries).  

2. **For each candidate** solve the LP above, obtain the radii and the total sum.

3. **Keep the best solution** (largest Σ r_i).

4. **Local refinement** – starting from the best set, repeatedly pick a random circle, move its centre a tiny amount, re‑solve the LP and keep the move only if the total radius sum increases. This hill‑climbing step can raise the sum a little further.

The whole procedure runs in a few hundred LP solves (the LP is very small – 26 variables, ≤ 429 constraints) and therefore finishes quickly.  
Finally the function `run_packing` returns the centres, radii and the achieved sum.

import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Linear programming: optimal radii for a fixed set of centres
# ------------------------------------------------------------
def solve_optimal_radii(centres: np.ndarray):
    """
    Solve the LP that maximises the sum of radii for the given centres.
    Returns (radii, total_sum).  If the LP fails, all radii are set to 0.
    """
    n = centres.shape[0]

    # pairwise Euclidean distances
    diff = centres[:, None, :] - centres[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # number of linear inequality constraints:
    # 4 per circle for the four walls + one per unordered pair
    n_pair = n * (n - 1) // 2
    n_cons = 4 * n + n_pair

    # Build constraint matrix A_ub @ r <= b_ub
    A = np.zeros((n_cons, n), dtype=np.float64)
    b = np.empty(n_cons, dtype=np.float64)

    row = 0
    # wall constraints
    for i in range(n):
        # r_i <= x_i
        A[row, i] = 1.0
        b[row] = centres[i, 0]
        row += 1
        # r_i <= 1 - x_i
        A[row, i] = 1.0
        b[row] = 1.0 - centres[i, 0]
        row += 1
        # r_i <= y_i
        A[row, i] = 1.0
        b[row] = centres[i, 1]
        row += 1
        # r_i <= 1 - y_i
        A[row, i] = 1.0
        b[row] = 1.0 - centres[i, 1]
        row += 1

    # non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # Objective: maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Solve with the HiGHS interior‑point method (fast and reliable)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback – all zeros (still a feasible point)
        radii = np.zeros(n, dtype=np.float64)
        total = 0.0
    else:
        radii = res.x
        total = -res.fun   # because we minimised the negative sum
    return radii, float(total)


# ------------------------------------------------------------
# Point set generators
# ------------------------------------------------------------
def farthest_point_set(num: int, rng: np.random.Generator, cand_per_iter: int = 3000):
    """
    Greedy max‑min (farthest‑point) construction.
    Starts with the centre of the square and repeatedly adds the
    candidate that maximises the minimal distance to the current set.
    """
    points = [np.array([0.5, 0.5])]
    while len(points) < num:
        cand = rng.random((cand_per_iter, 2))  # uniform points in [0,1]²
        existing = np.asarray(points)
        # distance from each candidate to the current set
        d = np.linalg.norm(cand[:, None, :] - existing[None, :, :], axis=2)
        min_d = np.min(d, axis=1)
        best_idx = np.argmax(min_d)
        points.append(cand[best_idx])
    return np.vstack(points)


def jittered_grid_set(num: int, rng: np.random.Generator):
    """
    Produce a regular grid (6×5 = 30 points) inside the square,
    keep a random subset of size `num` and add a tiny jitter.
    """
    rows, cols = 5, 6
    margin = 0.08  # distance from the walls
    xs = np.linspace(margin, 1.0 - margin, cols)
    ys = np.linspace(margin, 1.0 - margin, rows)
    xv, yv = np.meshgrid(xs, ys)
    grid = np.column_stack([xv.ravel(), yv.ravel()])          # 30 points
    idx = rng.choice(grid.shape[0], size=num, replace=False) # choose 26 of them
    chosen = grid[idx]
    # tiny random jitter so that the points are not perfectly aligned
    jitter = rng.uniform(-0.02, 0.02, size=chosen.shape)
    chosen = np.clip(chosen + jitter, 0.0, 1.0)
    return chosen


# ------------------------------------------------------------
# Local hill‑climbing refinement
# ------------------------------------------------------------
def local_refine(centres: np.ndarray, radii: np.ndarray,
                 total: float, rng: np.random.Generator,
                 steps: int = 150):
    """
    Simple stochastic hill climbing:
    move a random centre a little, recompute the LP, keep move if it improves Σr.
    """
    n = centres.shape[0]
    best_c = centres.copy()
    best_r = radii.copy()
    best_t = total

    for _ in range(steps):
        i = rng.integers(n)                       # which circle to move
        delta = rng.uniform(-0.015, 0.015, size=2) # small perturbation
        new_center = best_c[i] + delta
        new_center = np.clip(new_center, 0.0, 1.0)  # stay inside the square
        cand = best_c.copy()
        cand[i] = new_center
        cand_r, cand_t = solve_optimal_radii(cand)
        if cand_t > best_t + 1e-12:               # accept only true improvements
            best_c = cand
            best_r = cand_r
            best_t = cand_t
    return best_c, best_r, best_t

# ------------------------------------------------------------
def run_packing():
    """
    Returns (centers, radii, sum_of_radii) for 26 circles.
    The routine repeatedly generates candidate centre sets,
    solves the associated LP and keeps the best result.
    """
    n = 26
    rng = np.random.default_rng(123456)  # deterministic seed
    best_sum = -1.0
    best_centres = None
    best_radii = None

    # ------------------------------------------------------------------
    # 1. Global search – a mixture of far‑thest‑point and jittered grids
    # ------------------------------------------------------------------
    for it in range(200):
        if it % 5 == 0:                           # every 5th iteration use farthest‑point
            centres = farthest_point_set(n, rng)
        else:
            centres = jittered_grid_set(n, rng)

        radii, s = solve_optimal_radii(centres)

        if s > best_sum:
            best_sum = s
            best_centres = centres.copy()
            best_radii = radii.copy()

        # early stop once the target is met
        if best_sum >= 2.636:
            break

    # ------------------------------------------------------------------
    # 2. Local refinement – try to push the sum a little higher
    # ------------------------------------------------------------------
    if best_centres is not None:
        refined_c, refined_r, refined_t = local_refine(best_centres, best_radii,
                                                      best_sum, rng, steps=200)
        if refined_t > best_sum:
            best_sum = refined_t
            best_centres = refined_c
            best_radii = refined_r

    # Final sanity check – the validator will also verify everything.
    # The function must return numpy arrays of the correct shape.
    return best_centres, best_radii, float(best_sum)
