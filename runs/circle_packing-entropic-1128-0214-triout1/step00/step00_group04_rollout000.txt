import numpy as np
from scipy.optimize import linprog
from math import sqrt

def compute_optimal_radii(centers):
    n = centers.shape[0]
    # upper bounds from sides
    bound = np.minimum(np.minimum(centers[:,0], 1 - centers[:,0]),
                       np.minimum(centers[:,1], 1 - centers[:,1]))
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt((diff**2).sum(axis=2))
    # Build constraints
    A = []
    b = []
    # r_i <= bound_i
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A.append(row)
        b.append(bound[i])
    # r_i + r_j <= d_ij for i<j
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            b.append(dists[i,j])
    A = np.array(A)
    b = np.array(b)
    c = -np.ones(n)  # maximize sum => minimize -sum
    # Solve LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: zero radii
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = res.x
        total = -res.fun  # since objective is -sum
    return radii, total

import numpy as np
from scipy.optimize import linprog
from typing import Tuple

# ----------------------------------------------------------------------
# Linear programme: for a given set of centres compute the optimal radii
# ----------------------------------------------------------------------
def optimal_radii(centres: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Solve the LP that maximises the sum of radii for the supplied centres.

    Parameters
    ----------
    centres : np.ndarray, shape (n, 2)
        The (x, y) coordinates of the circles.

    Returns
    -------
    radii : np.ndarray, shape (n,)
        The optimal radii (non‑negative, respects all constraints).
    total : float
        Sum of the radii (the objective value).
    """
    n = centres.shape[0]

    # Upper bound for each radius due to the four sides of the square
    side_bounds = np.minimum.reduce(
        [centres[:, 0], 1.0 - centres[:, 0],
         centres[:, 1], 1.0 - centres[:, 1]]
    )

    # Pairwise Euclidean distances
    diff = centres[:, np.newaxis, :] - centres[np.newaxis, :, :]
    dists = np.sqrt((diff ** 2).sum(axis=2))

    # Build the inequality matrix A_ub @ r <= b_ub
    rows = []
    rhs = []

    # r_i <= side bound
    for i in range(n):
        row = np.zeros(n, dtype=float)
        row[i] = 1.0
        rows.append(row)
        rhs.append(side_bounds[i])

    # r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n, dtype=float)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dists[i, j])

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs, dtype=float)

    # Objective: maximise sum(r)  -> minimise -sum(r)
    c = -np.ones(n, dtype=float)

    # Solve the LP (HiGHS is fast and robust for this size)
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0.0, None)] * n,
        method='highs',
        options={'presolve': True}
    )

    if not res.success:
        # In the very unlikely event of failure fall back to zeros
        radii = np.zeros(n, dtype=float)
        total = 0.0
    else:
        radii = res.x
        total = -res.fun   # because we minimised the negative sum

    return radii, float(total)


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber that searches for good centre locations
# ----------------------------------------------------------------------
def hill_climb(
    n: int = 26,
    n_iter: int = 5000,
    seed: int = 0
) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Perform a hill‑climbing search on the centre positions.
    Returns the best configuration found.
    """
    rng = np.random.default_rng(seed)

    # ----- initial layout -------------------------------------------------
    # start from a regular grid and keep four points near the corners
    grid_side = int(np.ceil(np.sqrt(n * 1.1)))          # a little denser than sqrt(N)
    xs = np.linspace(0.05, 0.95, grid_side)
    ys = np.linspace(0.05, 0.95, grid_side)
    X, Y = np.meshgrid(xs, ys)
    grid_pts = np.column_stack((X.ravel(), Y.ravel()))

    if grid_pts.shape[0] >= n:
        idx = rng.choice(grid_pts.shape[0], size=n, replace=False)
        centres = grid_pts[idx].astype(float)
    else:
        centres = rng.random((n, 2))

    # enforce a few points close to the four corners – they usually become the
    # large circles in a good packing
    if n >= 4:
        corner_targets = np.array(
            [[0.25, 0.25],
             [0.75, 0.25],
             [0.25, 0.75],
             [0.75, 0.75]]
        )
        centres[:4] = corner_targets

    # ----- evaluate the initial placement ---------------------------------
    radii, best_total = optimal_radii(centres)
    best_centres = centres.copy()
    best_radii = radii.copy()

    # ----- hill‑climbing loop --------------------------------------------
    no_improve = 0
    max_no_improve = 1500

    for it in range(n_iter):
        # pick a random circle and propose a small move
        i = rng.integers(n)
        proposal = best_centres[i] + rng.normal(scale=0.05, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)

        trial_centres = best_centres.copy()
        trial_centres[i] = proposal

        trial_radii, trial_total = optimal_radii(trial_centres)

        # accept only if the total radius sum increases
        if trial_total > best_total:
            best_total = trial_total
            best_centres = trial_centres
            best_radii = trial_radii
            no_improve = 0
        else:
            no_improve += 1

        # occasional random restart of the *non‑corner* points if stuck
        if no_improve >= max_no_improve:
            # keep the four corner points (they are often valuable) and
            # re‑sample the remaining ones
            if n > 4:
                inner_cnt = n - 4
                new_inner = rng.random((inner_cnt, 2))
                best_centres[4:] = new_inner
            best_radii, best_total = optimal_radii(best_centres)
            no_improve = 0

    return best_centres, best_radii, best_total


# ----------------------------------------------------------------------
# Public entry point required by the task
# ----------------------------------------------------------------------
def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Compute a packing of 26 circles in the unit square that maximises the
    sum of radii.

    Returns
    -------
    centres : np.ndarray, shape (26, 2)
        Circle centre coordinates.
    radii : np.ndarray, shape (26,)
        Corresponding radii (all non‑negative, non‑overlapping, inside the square).
    total : float
        Sum of the radii.
    """
    # perform a few independent hill‑climbs and keep the best result
    best_total = -1.0
    best_centres = None
    best_radii = None

    for trial_seed in range(3):
        centres, radii, total = hill_climb(
            n=26,
            n_iter=6000,
            seed=trial_seed
        )
        if total > best_total:
            best_total = total
            best_centres = centres
            best_radii = radii

    return best_centres, best_radii, float(best_total)
