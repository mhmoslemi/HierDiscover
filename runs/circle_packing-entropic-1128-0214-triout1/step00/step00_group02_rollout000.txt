def solve_lp(centers):
    N = centers.shape[0]
    # compute boundary bounds
    B = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # compute distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # shape N,N,2
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # build A_ub
    m = N*(N-1)//2
    A_ub = np.zeros((m, N))
    b_ub = np.empty(m)
    idx = 0
    for i in range(N):
        for j in range(i+1,N):
            A_ub[idx,i] = 1
            A_ub[idx,j] = 1
            b_ub[idx] = dists[i,j]
            idx += 1
    c = -np.ones(N)
    bounds = [(0, B[i]) for i in range(N)]
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        radii = result.x
        sum_r = np.sum(radii)
        return sum_r, radii
    else:
        # fallback: zero radii
        return 0.0, np.zeros(N)

def compute_pair_matrix(N):
    m = N*(N-1)//2
    A = np.zeros((m,N))
    pair_idx = []
    idx = 0
    for i in range(N):
        for j in range(i+1,N):
            A[idx,i] = 1.0
            A[idx,j] = 1.0
            pair_idx.append((i,j))
            idx += 1
    return A, np.array(pair_idx, dtype=int)

def greedy_pack(N, samples_per_step=2000, seed=0):
    np.random.seed(seed)
    centers = []
    radii = []
    for k in range(N):
        # generate candidate points
        cand = np.random.rand(samples_per_step, 2)
        # compute boundary distances:
        b = np.minimum.reduce([cand[:,0], 1-cand[:,0], cand[:,1], 1-cand[:,1]])
        # compute distance to existing circles
        if len(centers) == 0:
            max_r = b
        else:
            # compute distances to each existing center
            C = np.array(centers)  # shape (m,2)
            # compute distances for each candidate to all existing centers: shape (samples_per_step, m)
            # Use broadcasting
            diff = cand[:, None, :] - C[None, :, :]  # shape (samples,m,2)
            d = np.linalg.norm(diff, axis=2)  # shape (samples,m)
            # radius limited by d - radii_i
            # radii_i is array shape (m)
            limit = d - np.array(radii)  # broadcast
            # For each candidate, radius is min of b and min_i(limit_i)
            max_r = np.minimum(b, np.min(limit, axis=1))
            max_r = np.maximum(max_r, 0.0)  # nonnegative
        # select best candidate
        idx = np.argmax(max_r)
        best_center = cand[idx]
        best_radius = max_r[idx]
        centers.append(best_center)
        radii.append(best_radius)
    return np.array(centers), np.array(radii)

The task is to place 26 circles inside the unit square `[0,1]×[0,1]` and to make the sum of their radii as large as possible.
For a *fixed* set of centre positions the optimal radii are the solution of a very simple
linear program:

```
maximise   Σ r_i
subject to r_i ≥ 0
           r_i ≤ distance to the four sides   (boundary constraint)
           r_i + r_j ≤ distance( centre_i , centre_j )   (non‑overlap)
```

All constraints are linear in the radii, therefore for any given placement of the
centres the LP gives the best possible radii instantly.

Consequently the whole problem reduces to finding a good set of centre positions.
A global stochastic optimiser (Differential Evolution) is used for this purpose.
For every candidate of the optimiser the LP is solved (with `scipy.optimize.linprog`);
the optimiser receives the *negative* total radius because it minimises the objective.
After the evolutionary phase a short simulated‑annealing hill‑climb is performed
to polish the best solution found.

All helper functions are defined at module top‑level (no closures, no lambdas) and
the final `run_packing` function returns

```python
centres : np.ndarray  shape (26, 2)
radii   : np.ndarray  shape (26, )
sum_r   : float
```

which satisfy the validation routine.

```python
import numpy as np
from scipy.optimize import linprog, differential_evolution

# ----------------------------------------------------------------------
# Global data
# ----------------------------------------------------------------------
N = 26                                            # number of circles


def _pair_matrix(n):
    """
    Build the matrix that encodes the constraints  r_i + r_j ≤ d_ij  .
    Returns
        A_ub : (m, n) array,   m = n·(n‑1)/2
        pairs: (m, 2) int array with the indices (i, j) for each row.
    """
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    pairs = np.array(pairs, dtype=int)
    m = pairs.shape[0]
    A = np.zeros((m, n))
    rows = np.arange(m)
    A[rows, pairs[:, 0]] = 1.0
    A[rows, pairs[:, 1]] = 1.0
    return A, pairs


# constant matrices used by every LP call
A_UB_STATIC, PAIRS = _pair_matrix(N)   # shape (325, 26)


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine
# ----------------------------------------------------------------------
def solve_radii(centres: np.ndarray) -> tuple[float, np.ndarray]:
    """
    For a given array of centre positions (N × 2) compute the optimal radii
    by solving a linear program.
    Returns (sum_of_radii, radii_array).
    """
    # distance of each centre to the four sides of the square
    bnd = np.minimum.reduce([centres[:, 0],
                             1.0 - centres[:, 0],
                             centres[:, 1],
                             1.0 - centres[:, 1]])          # shape (N,)

    # pairwise Euclidean distances for the non‑overlap constraints
    diff = centres[PAIRS[:, 0]] - centres[PAIRS[:, 1]]
    dists = np.sqrt(np.einsum('ij,ij->i', diff, diff))           # shape (325,)

    # bounds for every radius variable
    bounds = [(0.0, b) for b in bnd]

    # maximise Σ r_i  →  minimise -Σ r_i
    c = -np.ones(N)

    res = linprog(c, A_ub=A_UB_STATIC, b_ub=dists,
                  bounds=bounds, method='highs')
    if not res.success:
        # fallback – all radii zero
        return 0.0, np.zeros(N)

    radii = res.x
    # tiny safety margin (avoids floating‑point edge cases in the validator)
    eps = 1e-9
    radii = np.where(radii > eps, radii - eps, 0.0)

    return float(radii.sum()), radii


# ----------------------------------------------------------------------
# Objective function for the optimiser (negative total radius)
# ----------------------------------------------------------------------
def _objective(flat_coords: np.ndarray) -> float:
    """
    DE works with a flat vector of length 2·N.
    The function returns the *negative* sum of radii because DE minimises.
    """
    centres = flat_coords.reshape(N, 2)
    total, _ = solve_radii(centres)
    return -total


# ----------------------------------------------------------------------
# Small hill‑climbing post‑processor
# ----------------------------------------------------------------------
def _local_refine(best_c: np.ndarray, best_val: float,
                  steps: int = 500,
                  start_sigma: float = 0.04,
                  start_temp: float = 0.01) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Simple simulated‑annealing style refinement.
    """
    centres = best_c.copy()
    value = best_val
    radii = None

    sigma = start_sigma
    temp = start_temp
    for _ in range(steps):
        # propose a new set of centres by jittering every point a little
        cand = centres + np.random.normal(scale=sigma, size=centres.shape)
        cand = np.clip(cand, 0.0, 1.0)

        cand_val, cand_r = solve_radii(cand)
        delta = cand_val - value

        if delta > 0.0 or np.random.rand() < np.exp(delta / temp):
            centres = cand
            value = cand_val
            radii = cand_r

        sigma *= 0.995
        temp *= 0.995

    return centres, value, radii


# ----------------------------------------------------------------------
# Main entry point – required signature
# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, sum_of_radii) for a packing of 26 circles.
    The implementation uses Differential Evolution to optimise the centre
    positions and a final hill‑climb to polish the result.
    """
    rng_seed = 0                     # deterministic behaviour
    np.random.seed(rng_seed)

    # bounds for each coordinate (all must stay inside the unit square)
    bounds = [(0.0, 1.0)] * (2 * N)

    # Differential Evolution – relatively few generations to keep runtime modest
    de_result = differential_evolution(
        _objective,
        bounds,
        maxiter=40,               # 40 generations
        popsize=5,                # 5 × N individuals ≈ 260 per generation
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=rng_seed,
        polish=False,             # we will polish manually
        init='latinhypercube',
        updating='deferred'       # a tiny speed‑up
    )

    # Best centre set obtained by DE
    best_centres = de_result.x.reshape(N, 2)
    best_sum, best_radii = solve_radii(best_centres)

    # Local refinement (simulated‑annealing style)
    refined_centres, refined_sum, refined_radii = _local_refine(
        best_centres, best_sum)

    # If the refinement happened to improve the total, use it
    if refined_sum > best_sum:
        best_centres, best_sum, best_radii = refined_centres, refined_sum, refined_radii

    # Final safety cut‑off (ensures no tiny negatives)
    best_radii = np.where(best_radii > 0, best_radii, 0.0)
    best_sum = float(best_radii.sum())

    return best_centres, best_radii, best_sum
```
