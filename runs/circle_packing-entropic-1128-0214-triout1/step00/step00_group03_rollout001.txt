```python
def solve_lp(centers):
    n = centers.shape[0]
    # A_ub shape (num_constraints, n)
    rows = []
    b = []

    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        rows.append(np.eye(1, n, i)[0])   # r_i <= x
        b.append(x)
        rows.append(np.eye(1, n, i)[0])   # r_i <= 1 - x
        b.append(1 - x)
        rows.append(np.eye(1, n, i)[0])   # r_i <= y
        b.append(y)
        rows.append(np.eye(1, n, i)[0])   # r_i <= 1 - y
        b.append(1 - y)
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            coeff = np.zeros(n)
            coeff[i] = 1.0
            coeff[j] = 1.0
            rows.append(coeff)
            b.append(d)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)

    # objective
    c = -np.ones(n)  # maximize sum
    # bounds
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: maybe return zeros
        return np.zeros(n)
    return res.x
```

```python
import numpy as np
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Create a packing of 26 circles inside the unit square.

    - 25 circles are placed on a regular 5×5 grid with centre spacing 0.2.
      Each circle touches its neighbours and the four sides of the square,
      giving a radius of 0.1.
    - One additional degenerate circle with radius 0.0 is added to meet the
      required count of 26 circles.

    Returns
    -------
    centers : np.ndarray, shape (26, 2)
        (x, y) coordinates of the circle centres.
    radii : np.ndarray, shape (26,)
        Radii of the circles.
    total_radius : float
        Sum of all radii (used for scoring).
    """
    # Parameters for the regular grid
    r = 0.1                     # radius of the grid circles
    spacing = 2 * r             # centre‑to‑centre distance (touching)

    # Generate the 5×5 grid of centres
    coords = []
    for i in range(5):
        for j in range(5):
            x = r + i * spacing
            y = r + j * spacing
            coords.append((x, y))
    centers_grid = np.array(coords)                # shape (25, 2)

    # Add the 26th “zero‑radius” circle (any point inside the square works)
    extra_center = np.array([[0.0, 0.0]])           # placed at the origin
    centers = np.vstack([centers_grid, extra_center])

    # Radii: 0.1 for the 25 grid circles, 0.0 for the extra one
    radii = np.concatenate([np.full(25, r), np.array([0.0])])

    total_radius = float(radii.sum())
    return centers, radii, total_radius
```
