def compute_radius(candidate, selected_centers, selected_radii):
    # compute max radius for candidate given selected circles (with fixed radii)
    # radius bound by walls
    x,y = candidate
    r_wall = min(x, 1-x, y, 1-y)

    r_nb = np.min(dists - selected_radii)

import numpy as np
from scipy.optimize import linprog

def generate_hex_candidates(step: float = 0.02) -> np.ndarray:
    """
    Generate a dense set of candidate centres on a hexagonal lattice.
    The lattice covers the whole unit square.
    """
    # Horizontal spacing
    dx = step
    # Vertical spacing for a hexagonal pattern
    dy = step * np.sqrt(3) / 2.0

    points = []
    y = 0.0
    row = 0
    while y <= 1.0:
        # Offset every second row by half a step
        offset = 0.0 if row % 2 == 0 else dx / 2.0
        x = offset
        while x <= 1.0:
            points.append((x, y))
            x += dx
        y += dy
        row += 1
    return np.array(points, dtype=float)


def max_allowed_radius(point: np.ndarray,
                       centers: np.ndarray,
                       radii: np.ndarray) -> float:
    """
    Return the largest radius that can be assigned to `point` without
    overlapping the circles defined by `centers` and `radii` and while staying
    inside the unit square.
    """
    x, y = point
    # distance to the four walls
    r_wall = min(x, 1.0 - x, y, 1.0 - y)

    if centers.shape[0] == 0:
        return r_wall

    # distances to already placed centres
    dists = np.sqrt(((centers - point) ** 2).sum(axis=1))
    # radius limited by each neighbour
    r_neighbour = np.min(dists - radii)
    return min(r_wall, r_neighbour)


def optimise_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the linear programme

        maximise   sum r_i
        subject to r_i >= 0,
                   r_i <= w_i,
                   r_i + r_j <= d_ij

    where w_i is the distance of centre i to the closest wall and d_ij is the
    Euclidean distance between centres i and j.
    """
    n = centers.shape[0]

    # distance to the closest wall for each centre
    w = np.minimum.reduce([centers[:, 0],
                           1.0 - centers[:, 0],
                           centers[:, 1],
                           1.0 - centers[:, 1]])

    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt((diff ** 2).sum(axis=2))

    # Build A_ub * r <= b_ub
    # Wall constraints
    A = []
    b = []
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A.append(row)
        b.append(w[i])

    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            b.append(dists[i, j])

    A = np.array(A)
    b = np.array(b)

    # maximise sum(r)  ==> minimise -sum(r)
    c = -np.ones(n)

    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError('Linear programming failed: ' + res.message)

    # Numerical noise can produce tiny negative values
    radii_opt = np.maximum(res.x, 0.0)
    return radii_opt


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Build a packing of 26 circles inside the unit square.
    Returns:
        centres : (26, 2) array of (x, y) coordinates
        radii   : (26,) array of radii
        total   : sum of radii (float)
    """
    # ---------- 1. start with four corner circles ----------
    corner_centres = np.array([
        [0.25, 0.25],
        [0.75, 0.25],
        [0.25, 0.75],
        [0.75, 0.75]
    ], dtype=float)
    corner_radii = np.full(4, 0.25, dtype=float)

    # ---------- 2. generate many candidate points ----------
    candidates = generate_hex_candidates(step=0.02)

    # Remove points that are already occupied (distance zero)
    # (won’t happen, but keep the code tidy)
    # ---------- 3. greedy insertion ----------
    centres = corner_centres.copy()
    radii = corner_radii.copy()

    while centres.shape[0] < 26:
        best_idx = -1
        best_r = -1.0

        # Scan all remaining candidates
        for idx, pt in enumerate(candidates):
            r = max_allowed_radius(pt, centres, radii)
            if r > best_r:
                best_r = r
                best_idx = idx

        # If no positive radius can be placed, exit the loop
        if best_idx == -1 or best_r <= 0.0:
            break

        # Add the best candidate
        new_center = candidates[best_idx]
        centres = np.vstack([centres, new_center])
        radii = np.append(radii, best_r)

        # Remove the used candidate
        candidates = np.delete(candidates, best_idx, axis=0)

    # ---------- 4. pad with zero‑radius circles if needed ----------
    needed = 26 - centres.shape[0]
    if needed > 0:
        # Use the origin (wall distance = 0) – guarantees radius 0
        pad = np.zeros((needed, 2), dtype=float)
        centres = np.vstack([centres, pad])
        radii = np.append(radii, np.zeros(needed, dtype=float))

    # ---------- 5. optimise radii for the final set of centres ----------
    radii_opt = optimise_radii(centres)

    total = float(radii_opt.sum())
    return centres, radii_opt, total
