def propose_candidate(seed=42, budget_s=1000, **kwargs):
    start = time.time()
    end = start + budget_s - 2.0
    rng = np.random.default_rng(seed)
    best_seq = None
    best_score = np.inf

    # Load previous best if any
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None
    # Phase 1: quick random search
    while time.time() < start + 0.1*budget_s:
        # choose n (like 2000-8000)
        n = int(rng.integers(1500, 8000))
        a0 = _initial_vector(n, rng, base_seq)
        peak = _max_convolution(a0)
        if peak < best_score:
            best_score = peak
            best_seq = a0
    # Phase 2: scaling and multiple optimizations
    # create list of target lengths - start with best length, up to larger lengths
    lengths = [len(best_seq)] if best_seq is not None else [2000]
    # add expansions
    for factor in [1.5, 2, 3, 4]:
        cand = int(lengths[0] * factor)
        if cand <= 30000:
            lengths.append(cand)
    lengths = sorted(set(lengths))
    # iterate over lengths
    for target_n in lengths:
        # allocate per length time slice: shrink after each.
        remaining = end - time.time()
        if remaining < 20:
            break
        slice_time = min(remaining * 0.4, 250.0)   # could be 250 secs for long run
        # generate scaled best, then do multiple restarts
        scaled = _scale_sequence(best_seq, target_n) if best_seq is not None else _initial_vector(target_n, rng, None)
        # Normalize sum to sqrt(2n)
        S = np.sqrt(2.0 * target_n)
        scaled = np.clip(scaled, 0.0, None)
        scaled = scaled * (S / scaled.sum())
        # We'll run several restarts within slice_time
        restarts = 3
        per_restart = slice_time / restarts
        for i in range(restarts):
            # random perturbation of scaled to create start
            start_a = scaled.copy()
            # add some noise and renorm
            noise = rng.normal(scale=0.05, size=target_n)
            start_a = np.maximum(start_a + noise, 0.0)
            start_a = start_a * (S / start_a.sum())
            a_opt, peak = _adam_optimize_improved(start_a, time.time() + per_restart, rng)
            if peak < best_score:

"""
Adversarial step‑function optimizer – enhanced version.

The algorithm consists of three phases.

1️⃣ **Fast random exploration** – generate many feasible candidates from a
   diverse set of parametric families (power‑law, exponential, linear,
   gamma, uniform, Dirichlet and a sparse‑gamma variant).  Each candidate
   is normalised so that its ℓ₁‑norm equals √(2 n) and its self‑convolution
   peak is measured with an FFT.  The best candidate seen so far is kept.

2️⃣ **Iterative scaling + Adam‑refinement** – repeatedly double the current
   best length (up to a safe cap) and optimise the new vector on the
   simplex with an Adam optimiser.  The optimiser works on the log‑softmax
   parametrisation of the simplex; for the first few thousand iterations
   a smooth τ‑softmax approximation of the max‑convolution is used, after
   which an exact hard‑max sub‑gradient is employed.  Several random
   restarts are performed for each length, and a short stochastic hill‑climb
   is applied after the Adam run.

3️⃣ **Final sanity check** – the best sequence found is evaluated with the
   provided ``evaluate_sequence`` function and stored in the global variable
   ``height_sequence_1`` (required by the harness).

The routine respects the 1000 s time budget, dynamically allocating time
between the phases and printing progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _hard_gradient(a: np.ndarray):
    """
    Exact sub‑gradient of the (hard) max‑convolution.
    Returns (grad, max_b) where grad = ∂ max(b)/∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()
    mask = (b >= max_b - 1e-12).astype(float)   # tolerance for floating‑point noise

    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize_improved(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    - For the first 2000 iterations a smooth τ‑softmax gradient is used.
    - Afterwards the exact hard‑max sub‑gradient is employed.
    Returns the best sequence encountered and its (hard) max‑convolution.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)           # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.25                       # slightly smaller initial step size

    # initialise temperature for soft‑max
    tau = max(_max_convolution(a0), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- gradient & hard peak -----
        if it <= 2000:
            g, peak = _softmax_gradient(a, tau)
            tau *= 0.9997                # slowly reduce smoothing
        else:
            g, peak = _hard_gradient(a)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        lr *= 0.9999

        # occasional jitter to escape plateaus
        if it % 600 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)          # exponent
    offset = rng.uniform(0.0, 15.0)    # shift of the denominator
    mix = rng.uniform(0.0, 1.0)        # mixing weight
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)   # decay rate
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)                     # height at the edge
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))  # controls width of the plateau
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n) * rng.uniform(0.0, 0.2)
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like start (raw random numbers)."""
    raw = rng.random(n)
    return raw


def _init_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Very sparse vector – a few random positions get random mass."""
    k = max(1, int(rng.integers(1, max(2, n // 10))))   # at most n/10 non‑zero entries
    vals = rng.exponential(scale=1.0, size=k)
    seq = np.zeros(n)
    idx = rng.choice(n, size=k, replace=False)
    seq[idx] = vals
    # optionally blend with a small uniform background
    mix = rng.uniform(0.0, 0.5)
    base = np.full(n, 0.1)
    raw = mix * seq + (1.0 - mix) * base
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best and otherwise samples
    from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'sparse'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:   # sparse
        raw = _init_sparse(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.

    The routine respects the time budget and stores the final construction
    in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 15 % of the total budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + min(0.15 * budget_s, 150.0)
    attempt = 0
    while time.time() < explore_deadline:
        attempt += 1
        # choose a length roughly between 1500 and 8000
        n = int(rng.integers(1500, 8000))
        cand = _initial_vector(n, rng, base_seq)
        peak = _max_convolution(cand)
        if peak < best_score:
            best_score = peak
            best_seq = cand.copy()
            best_n = n
            print(f"[explore] attempt={attempt:4d} n={n:5d}  new best={best_score:.6f}")

    # fallback in the unlikely case that nothing was generated
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        print("[fallback] using uniform start")

    # ------------------------------------------------------------------
    # Phase 2 – scaling and intensive Adam refinement
    # ------------------------------------------------------------------
    cur_n = best_n
    while True:
        remaining = end_time - time.time()
        if remaining < 30.0:
            break

        # propose the next length (doubling, capped at 30000)
        target_n = min(cur_n * 2, 30000)
        if target_n == cur_n:
            break

        # Resize the current best to the new length and normalise
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # Allocate a time slice for this length (up to 40 % of the remaining)
        time_slice = min(remaining * 0.4, 300.0)
        restarts = 3
        per_restart = time_slice / restarts

        for r in range(restarts):
            # jitter the start a bit before each restart
            a0 = a_scaled + rng.normal(scale=0.02, size=target_n)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 * (S / a0.sum())

            a_opt, peak = _adam_optimize_improved(a0,
                                                   time.time() + per_restart,
                                                   rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_opt.copy()
                cur_n = target_n
                print(f"[refine] length={target_n:5d} restart={r}  new best={best_score:.6f}")

        # ------------------------------------------------------------------
        # Short stochastic hill‑climb on the best vector of this length
        # ------------------------------------------------------------------
        hill_deadline = time.time() + min(remaining * 0.1, 20.0)
        a_cur = best_seq.copy()
        while time.time() < hill_deadline:
            # random subset of indices to perturb
            k = max(1, int(0.01 * target_n))
            idx = rng.choice(target_n, size=k, replace=False)
            delta = rng.normal(scale=0.05, size=k)
            a_new = a_cur.copy()
            a_new[idx] += delta
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * (S / a_new.sum())
            peak = _max_convolution(a_new)
            if peak < best_score:
                best_score = peak
                best_seq = a_new.copy()
                a_cur = a_new
                print(f"[hill] length={target_n:5d} improved to {best_score:.6f}")

        # Prepare for possible next scaling step
        cur_n = target_n
        if cur_n >= 30000:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
