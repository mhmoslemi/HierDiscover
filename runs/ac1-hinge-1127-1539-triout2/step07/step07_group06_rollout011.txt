"""
Propose a step‑function sequence that minimises the provided evaluation.

The algorithm consists of three stages:

1. **Warm‑start** – if a previous best sequence (`height_sequence_1`) is
   available it is projected onto the simplex {a ≥ 0, Σa = √(2·n)}.

2. **Random analytic search** – families of analytically defined sequences
   (pure power‑law, shifted power‑law and a simple two‑segment mixture)
   are sampled for several lengths.  Each candidate is evaluated with an
   FFT based max‑convolution (O(n log n)).  The best candidate encountered
   is retained.

3. **Gradient polishing** – the current best candidate is refined by a
   projected gradient descent on a smooth approximation of the
   max‑convolution (soft‑max with increasing temperature).  This is the
   same “soft‑max” gradient routine used in the reference solution, but
   we allocate a larger fraction of the remaining budget to it.

All steps respect the global time budget (default 1000 s).  The returned
list contains non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (no normalisation)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Fast objective: max entry of self‑convolution (Σa already √(2n))."""
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Analytic families of sequences
# ----------------------------------------------------------------------


def _power_law_seq(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+shift)^(-p)  –  normalised to Σa = √(2n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0 + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    a *= target / a.sum()
    return a


def _mixed_power_law_seq(
    n: int,
    p1: float,
    p2: float,
    cutoff_frac: float,
    shift: float = 0.0,
) -> np.ndarray:
    """
    Two‑segment power‑law:
        first part (0 … cutoff)  : (i+1+shift)^(-p1)
        second part (cutoff … n) : (i+1+shift)^(-p2)
    Normalised to Σa = √(2n).
    """
    i = np.arange(n, dtype=np.float64)
    cutoff = int(np.clip(cutoff_frac * n, 1, n - 1))
    a = np.empty_like(i)
    a[:cutoff] = (i[:cutoff] + 1.0 + shift) ** (-p1)
    a[cutoff:] = (i[cutoff:] + 1.0 + shift) ** (-p2)
    target = np.sqrt(2.0 * n)
    a *= target / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient descent (reference polishing routine)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)                # soft‑max weights

    # convolution of reversed a with the soft‑max weights
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # gradient w.r.t. a (only the central n entries matter)
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, conv


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β starts at 1 and is multiplied by 1.5 periodically
    up to 4096.  The routine stops when the deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g

        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """

    The routine combines a warm‑start, a fast random analytic search and
    a final gradient‑descent polish.  It respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

            p2 = rng.uniform(0.45, 0.65)

            shift = rng.uniform(0.0, 10.0)
            a = _mixed_power_law_seq(n, p1, p2, cutoff_frac, shift)

        val = _score(a)

        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.1f}] new best n={n} score={val:.6f}")

        # Occasionally polish the current best to accelerate improvement.
        # The probability is low to keep overhead small.
        if rng.random() < 0.015 and time.time() < deadline - 5.0:
            # use a modest fraction of the remaining time
            remaining = deadline - time.time()
            polish_deadline = time.time() + remaining * 0.30
            cand, cand_score = _annealed_gradient_descent(best_seq, polish_deadline)
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"[{time.time():.1f}] polished during search score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final intensive gradient‑descent polishing.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        final_deadline = time.time() + remaining * 0.60
        best_seq, best_score = _annealed_gradient_descent(best_seq, final_deadline)
        print(f"[{time.time():.1f}] final polishing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Strip trailing zeros (they do not influence the objective).
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:

        else:
            best_seq = np.empty(0, dtype=float)

    print(f"[{time.time():.1f}] returning best ≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
