def gradient_softmax(a, beta):
    n = a.size
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = b.max()
    exp_shifted = np.exp(beta * (b - max_b))
    # avoid overflow: if beta large, exp may underflow for negative differences; fine.
    p = exp_shifted / exp_shifted.sum()
    # compute convolution a_rev * p using FFT
    a_rev = a[::-1]
    conv_len = n + (2*n-1) - 1
    size2 = 1 << (conv_len.bit_length())
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    conv_full = np.fft.irfft(a_rev_fft * p_fft, n=size2)[:conv_len]
    # gradient slice
    start = n-1
    g = 2.0 * conv_full[start:start+n]
    return g, b

def _subgradient_opt(a0, deadline, step0=0.05, decay=0.9995):
    a = a0.copy()
    n = a.size
    target = math.sqrt(2.0 * n)
    step = step0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution
        b = _max_convolution_fft(a)  # Actually we need full vector b for max and index.
        # we need full convolution vector, not just max. Let's compute full conv via FFT:
        size = 1 << ((2*n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv_full = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        k = int(np.argmax(conv_full))
        max_val = conv_full[k]
        # gradient of max entry
        # compute g as described:
        g = np.zeros_like(a)
        # compute j = k - i for each i
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        # using vectorized operations:
        idx = i[mask]
        j_idx = j[mask]
        # For each idx, if idx != j_idx: g[idx] = 2*a[j_idx]; else g[idx] = a[idx]
        diff = idx != j_idx

        # update
        a -= step * g
        # projection
        a = _project_to_simplex(a, target)
        step *= decay
        # optional jitter occasionally
        if rng.random() < 0.01:
            # add small random noise
            a += rng.normal(scale=target*1e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
    return a

def _permutation_annealing(a0, deadline):
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)
    # temperature schedule: start high proportionally to best_score
    T = best_score * 0.05 + 1e-6  # larger than before
    cooling = 0.9998
    rng = np.random.default_rng()
    while time.time() < deadline:
        move_type = rng.integers(0, 3)
        if move_type == 0:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
        elif move_type == 1:
            # reverse subarray
            start = rng.integers(0, n-1)
            length = rng.integers(2, max(2, n//10))
            end = min(n, start+length)
            a[start:end] = a[start:end][::-1]
        else:
            # rotate subarray
            start = rng.integers(0, n-1)
            length = rng.integers(2, max(2, n//10))
            end = min(n, start+length)
            sub = a[start:end]
            shift = rng.integers(1, sub.size)
            a[start:end] = np.concatenate((sub[shift:], sub[:shift]))
        cur_score = _max_convolution_fft(a)
        delta = cur_score - best_score
        if cur_score <= best_score or rng.random() < np.exp(-delta / T):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert the move
            # For swap, we already swapped; need to revert.
            # Since we didn't keep previous state, easiest: revert by performing same move again.
            if move_type == 0:
                a[i], a[j] = a[j], a[i]  # swap back
            elif move_type == 1:
                a[start:end] = a[start:end][::-1]  # reverse back
            else:
                # rotate back by opposite direction
                sub = a[start:end]
                # reverse rotation: shift left by (sub.size - shift)
                # compute opposite shift
                inv_shift = sub.size - shift
                a[start:end] = np.concatenate((sub[inv_shift:], sub[:inv_shift]))
        T *= cooling
        # occasional random reshuffle of a tiny fraction
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                part = a[mask]
                rng.shuffle(part)
                a[mask] = part
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
    return best

"""
Propose a step‑function (non‑negative sequence) that minimises
 ``evaluate_sequence`` (i.e. the maximal entry of its self‑convolution).
The algorithm combines:

* a large random parametric search over several families and lengths,
  including a Dirichlet‑type random draw;
* a fast FFT‑based evaluation of max‑convolution;
* a two‑stage continuous optimisation:
    – Adam optimiser on a log‑sum‑exp smooth approximation of the max,
    – a pure sub‑gradient descent on the true max entry;
* an aggressive permutation simulated‑annealing that uses swaps,
  sub‑array reversals and rotations.

All optimisation respects the ℓ¹‑normalisation Σ a = √(2 n) by projection
onto the simplex after every update.  The routine obeys the supplied
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returned as a 1‑d numpy array)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# 2. Projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 3. Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a  (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient = 2 * (a_rev ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# 4. Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 0.5,
                   beta_end: float = 5000.0) -> np.ndarray:
    """Projected Adam on the log‑sum‑exp surrogate of the max."""
    a = a0.copy()
    n = a.size
    target = math.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    beta = beta_start
    it = 1
    # increase beta multiplicatively; after ~5000 it reaches beta_end
    beta_factor = 1.02
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size proportional to the target sum (scale‑invariant)
        step = 0.3 * target / (np.linalg.norm(m_hat) + 1e-12)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasibility
        a = _project_to_simplex(a, target)

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            jitter = rng.normal(scale=target * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # anneal beta
        if beta < beta_end:
            beta = min(beta * beta_factor, beta_end)

        it += 1

    return a


# ----------------------------------------------------------------------
# 5. Sub‑gradient descent on the true maximum entry
# ----------------------------------------------------------------------
def _subgradient_opt(a0: np.ndarray,
                     deadline: float,
                     step0: float = 0.04,
                     decay: float = 0.9995) -> np.ndarray:
    """Projected sub‑gradient descent on the max entry of a*a."""
    a = a0.copy()
    n = a.size
    target = math.sqrt(2.0 * n)
    step = step0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # full convolution (FFT is fast enough)
        conv = _full_convolution_fft(a)
        k = int(np.argmax(conv))          # index of the current maximum
        # build gradient of b_k w.r.t. a
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)

        idx = i[mask]
        jdx = j[mask]
        diff = idx != jdx
        g[idx[diff]] = 2.0 * a[jdx[diff]]
        g[idx[~diff]] = a[idx[~diff]]

        a -= step * g
        a = _project_to_simplex(a, target)

        step *= decay

        # tiny random jitter every now and then
        if rng.random() < 0.01:
            a += rng.normal(scale=target * 5e-5, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# 6. Permutation‑based simulated annealing (swap / reverse / rotate)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray,
                           deadline: float) -> np.ndarray:
    """Improves the ordering of a fixed multiset by simulated annealing."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature roughly proportional to current objective
    T = best_score * 0.04 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        move_type = rng.integers(0, 3)  # 0=swap, 1=reverse, 2=rotate

        if move_type == 0:  # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            revert = lambda: (a.__setitem__(i, a[j]), a.__setitem__(j, a[i]))
        elif move_type == 1:                # sub‑array reversal
            start = rng.integers(0, n - 1)
            length = rng.integers(2, max(2, n // 10))
            end = min(n, start + length)
            a[start:end] = a[start:end][::-1]
            revert = lambda: a.__setitem__(slice(start, end), a[start:end][::-1])
        else:                               # sub‑array rotation
            start = rng.integers(0, n - 1)
            length = rng.integers(2, max(2, n // 10))
            end = min(n, start + length)
            sub = a[start:end]
            shift = rng.integers(1, sub.size)
            a[start:end] = np.concatenate((sub[shift:], sub[:shift]))
            # opposite rotation restores the original order
            rev_shift = sub.size - shift
            revert = lambda: a.__setitem__(slice(start, end),
                                          np.concatenate((a[start:end][rev_shift:], a[start:end][:rev_shift])))

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < math.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # undo move
            revert()

        T *= cooling

        # occasional tiny reshuffle of a random 2 % of entries
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                part = a[mask]
                rng.shuffle(part)
                a[mask] = part
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# 7. Families of parametric sequences (all ensure Σa = √(2 n))
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ¹‑norm equals *target*."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, math.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, math.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, math.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, math.sqrt(2.0 * n))


def _dirichlet_sequence(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point from a Dirichlet distribution (α controls concentration)."""
    raw = rng.gamma(shape=alpha, scale=1.0, size=n)
    return _normalize_to_sum(raw, math.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# 8. Helper to keep the global best and an elite list
# ----------------------------------------------------------------------
def _consider(seq: np.ndarray,
              best_seq_ref: list,
              best_obj_ref: list,
              elite_ref: list):
    """Update global best and elite list with a new candidate."""
    obj = _max_convolution_fft(seq)
    if obj < best_obj_ref[0]:
        best_obj_ref[0] = obj
        best_seq_ref[0] = seq.copy()
        print(f"[{time.time():.1f}] New global best obj={obj:.6f} len={seq.size}")
    # keep at most 12 elite items
    if len(elite_ref) < 12 or obj < elite_ref[-1][0]:
        elite_ref.append((obj, seq.copy()))
        elite_ref.sort(key=lambda x: x[0])
        del elite_ref[12:]


# ----------------------------------------------------------------------
# 9. Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective (equivalently the maximal entry
    of its self‑convolution).  The routine respects the allotted time
    budget, explores several parametric families, refines promising
    candidates with Adam + sub‑gradient optimisation and finally
    performs a global permutation‑annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = [None]      # mutable container for the best sequence
    best_obj = [np.inf]    # mutable container for the best objective
    elite = []             # list of (obj, array)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        n0 = init.size
        init = _project_to_simplex(init, math.sqrt(2.0 * n0))
        _consider(init, best_seq, best_obj, elite)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Random parametric generation (≈ 60 % of the time budget)
    # ------------------------------------------------------------------
    lengths = [5000, 8000, 10000, 12000, 15000, 18000, 20000]
    families = ['power', 'exp', 'linear', 'quadratic', 'mixed', 'dirichlet']
    gen_deadline = time.time() + 0.60 * (deadline - time.time())

    while time.time() < gen_deadline:
        n = int(rng.choice(lengths))
        target_sum = math.sqrt(2.0 * n)
        family = rng.choice(families)

        if family == 'power':
            p = rng.uniform(0.2, 1.0)
            shift = rng.uniform(0.0, 5.0)
            base = _powerlaw_sequence(n, p, shift)

        elif family == 'exp':
            rate = rng.uniform(5e-05, 0.025)
            shift = rng.uniform(0.0, 5.0)
            base = _exponential_sequence(n, rate, shift)

        elif family == 'linear':
            slope = rng.uniform(0.0, 0.015)
            intercept = rng.uniform(0.5, 5.0)
            base = _linear_sequence(n, slope, intercept)

        elif family == 'quadratic':
            a0 = rng.uniform(0.5, 6.0)
            b = rng.uniform(0.0, 0.015)
            c = rng.uniform(0.0, 0.0001)
            base = _quadratic_sequence(n, a0, b, c)

        elif family == 'mixed':
            w = rng.uniform(0.3, 0.7)
            p = rng.uniform(0.3, 1.0)
            shift_p = rng.uniform(0.0, 5.0)
            pow_seq = _powerlaw_sequence(n, p, shift_p)

            rate = rng.uniform(5e-05, 0.025)
            shift_e = rng.uniform(0.0, 5.0)
            exp_seq = _exponential_sequence(n, rate, shift_e)

            mix = w * pow_seq + (1.0 - w) * exp_seq
            base = _project_to_simplex(mix, target_sum)

        else:  # dirichlet
            alpha = rng.uniform(0.1, 2.0)
            base = _dirichlet_sequence(n, alpha, rng)

        # ------------------------------------------------------------------
        # Try a few deterministic orderings + one random shuffle
        # ------------------------------------------------------------------
        for order in ['desc', 'asc', 'rand']:
            if order == 'desc':
                cand = np.sort(base)[::-1]
            elif order == 'asc':
                cand = np.sort(base)
            else:
                cand = base.copy()
                rng.shuffle(cand)

            _consider(cand, best_seq, best_obj, elite)

        # ------------------------------------------------------------------
        # Occasionally run a quick permutation anneal on the current best
        # ------------------------------------------------------------------
        if rng.random() < 0.04 and best_seq[0] is not None:
            tmp = _permutation_annealing(best_seq[0],
                                         time.time() + 0.2)
            _consider(tmp, best_seq, best_obj, elite)

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (remaining time)
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size

        # a) Adam optimiser on smooth surrogate
        sub_dead = min(deadline, time.time() + 18.0)
        refined = _adam_optimize(seed_seq, sub_dead)

        # b) Sub‑gradient descent on the true max
        sub_dead = min(deadline, time.time() + 6.0)
        refined = _subgradient_opt(refined, sub_dead)

        # c) Permutation annealing to the very end
        refined = _permutation_annealing(refined, deadline)

        _consider(refined, best_seq, best_obj, elite)

    # ------------------------------------------------------------------
    # 3. Final global permutation annealing if any time left
    # ------------------------------------------------------------------
    if best_seq[0] is not None and time.time() < deadline:
        best_seq[0] = _permutation_annealing(best_seq[0], deadline)
        best_obj[0] = _max_convolution_fft(best_seq[0])

    # ------------------------------------------------------------------
    # 4. Fallback in the (unlikely) case no sequence was produced
    # ------------------------------------------------------------------
    if best_seq[0] is None:
        n = 8000
        best_seq[0] = np.full(n, math.sqrt(2.0 * n) / n)

    # Trim negligible tail entries (purely cosmetic)
    nz = np.where(best_seq[0] > 1e-12)[0]
    if nz.size:
        best_seq[0] = best_seq[0][: nz[-1] + 1]

    # Optional verification with the provided evaluator
    try:
        val = evaluate_sequence(best_seq[0].tolist())
        print(f"[{time.time():.1f}] verify evaluate_sequence = {val:.6f}")
    except Exception as e:
        print(f"[{time.time():.1f}] verification error: {e}")

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj[0]:.6f} "
          f"length = {best_seq[0].size}")

    return best_seq[0].tolist()
