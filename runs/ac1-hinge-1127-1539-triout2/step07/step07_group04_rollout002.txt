"""
Propose a step‑function that (approximately) minimizes ``evaluate_sequence``.

Algorithm – multi‑stage optimisation
------------------------------------
* **Initial seed** – use the previously found construction (global variable
  ``height_sequence_1``) when present, otherwise start from a random
  distribution on the ℓ¹‑simplex.

* **Stage schedule** – we run the optimisation for a sequence of lengths
  (2000, 5000, 10000, 20000, 40000).  At each stage the current best
  vector is up‑sampled to the new length and refined.

* **Sub‑gradient descent on the true max** – instead of the smooth
  log‑sum‑exp surrogate we compute the exact sub‑gradient of the
  ``max(convolution)`` (the indicator of the worst convolution index).
  The gradient is obtained by a single FFT‑based convolution per iteration,
  and we perform a decreasing‑step‑size projected (Euclidean) sub‑gradient
  descent on the simplex.

* **Mass‑transfer hill‑climbing** – after the sub‑gradient phase we run a
  simulated‑annealing hill‑climber that updates the full linear
  self‑convolution in O(n) time per move.  The temperature is set
  proportionally to the current worst entry and cooled very slowly
  (``cooling=0.9998``) to allow many uphill moves early on.

* **Final polishing** – any remaining time is spent in a last
  sub‑gradient run with a very small step size.

All vectors stay non‑negative and are projected onto the ℓ¹‑simplex
{aᵢ≥0, Σaᵢ = √(2 n)} after each move, so the official evaluator receives a
valid step function.  Progress and intermediate scores are printed for
monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                         # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                       # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                       # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                       # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                       # sine half‑wave
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                       # Gaussian bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

# Sub‑gradient descent on the true maximum of the convolution
# ----------------------------------------------------------------------


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent that directly minimises
    max(conv(a,a)).  The sub‑gradient is computed from the indicator of the
    worst convolution index (exact, not a smooth approximation).
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # step‑size schedule (initial step proportional to target)
    step0 = target * 0.6
    it = 0

    while time.time() < deadline:
        # full convolution via FFT
        size = _next_pow2(2 * n - 1)
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = b.max()
        # indicator (soft distribution) on the worst entries
        mask = (b == max_b).astype(float)
        cnt = mask.sum()
        if cnt == 0:
            break
        p = mask / cnt                     # probability vector on the worst positions

        # gradient = 2·(a_rev ⋆ p)
        a_rev = a[::-1]
        conv_len = n + (2 * n - 1) - 1      # = 3n‑2
        size2 = _next_pow2(conv_len)
        a_rev_fft = np.fft.rfft(a_rev, n=size2)
        p_fft = np.fft.rfft(p, n=size2)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size2)[:conv_len]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # decreasing step size
        step = step0 / (1.0 + it * 0.02)
        a -= step * grad

        # project back onto the simplex (and keep non‑negative)
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

        it += 1

    return a

# Mass‑transfer hill climbing (exact O(n) convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray,
                            b0: np.ndarray,
                            deadline: float,
                            temp_factor: float = 0.5,
                            cooling: float = 0.9998,
                            dest_bias: float = 0.6) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing hill climbing with a slower cooling schedule and
    a larger admissible mass transfer.  Returns the best vector found and
    its convolution.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * temp_factor + 1e-12

    while time.time() < deadline:
        # choose one of the current worst indices uniformly
        worst_idxs = np.flatnonzero(b == cur_score)
        if worst_idxs.size == 0:
            break
        M = rng.choice(worst_idxs)

        # source index – weighted by contribution to the worst entry
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index – biased towards small masses
        if rng.random() < dest_bias:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move – allow moving up to the whole source mass
        max_delta = a[i]
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (linear terms)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic corrections
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        # temperature decay
        T *= cooling

    return best_a, b

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = globals()["height_sequence_1"]
        if isinstance(prev, (list, tuple, np.ndarray)):
            a0 = np.asarray(prev, dtype=float)
            n0 = a0.size
            target0 = np.sqrt(2.0 * n0)
            a0 = _project_to_simplex(a0, target0)
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – increasingly larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000]

    for n in schedule:
        if time.time() >= deadline:
            break

        target = np.sqrt(2.0 * n)
        print(f"[{time.time():.1f}] -------- Stage n={n} --------")

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target)

        # --------------------------------------------------------------
        # Allocate a stage budget (dynamic split)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        stage_budget = min(remaining * 0.9, 250.0)      # keep a little for final polishing
        sub_budget = stage_budget * 0.55                # sub‑gradient phase
        hc_budget = stage_budget * 0.40                 # hill‑climbing phase
        # final polishing inside the stage (very short)
        final_sub_budget = stage_budget * 0.05

        # --------------------------------------------------------------
        # Phase A – sub‑gradient descent with several restarts
        # --------------------------------------------------------------
        sub_deadline = time.time() + sub_budget
        n_restarts = 6 if n <= 10000 else 4
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= sub_deadline:
                break
            # allocate a modest slice of the sub‑gradient budget per restart
            now = time.time()
            per_restart_deadline = now + (sub_deadline - now) / (n_restarts - r)
            a_start = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _subgradient_descent_opt(a_start, per_restart_deadline)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()
                print(f"[{time.time():.1f}] Sub‑grad restart {r+1}/{n_restarts} score={score:.6f}")

        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time():.1f}] Phase‑A improvement for n={n} score={local_best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – hill‑climbing refinement
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        hc_deadline = time.time() + min(hc_budget, remaining * 0.8)

        # make sure we are working with length n
        if best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = best_seq.copy()
        a_current = _project_to_simplex(a_current, target)
        b_full = _full_convolution(a_current)

        a_hc, b_hc = _hill_climbing_improved(a_current, b_full, hc_deadline,
                                            temp_factor=0.5,
                                            cooling=0.9998,
                                            dest_bias=0.6)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Optional tiny final sub‑gradient polishing for this stage
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0 and final_sub_budget > 0.1:
            refine_deadline = time.time() + min(final_sub_budget, remaining * 0.4)
            a_refine = _subgradient_descent_opt(best_seq.copy(), refine_deadline)
            score_ref = _max_convolution_fft(a_refine)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_refine.copy()
                print(f"[{time.time():.1f}] Phase‑A refinement score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        n = best_seq.size
        target = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target)
        final_deadline = deadline
        a_final = _subgradient_descent_opt(a, final_deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: trim trailing near‑zeros for nicer output
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:               # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
