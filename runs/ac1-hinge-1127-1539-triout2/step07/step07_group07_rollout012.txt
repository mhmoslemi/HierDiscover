def _alternating_order(base):
    asc = np.sort(base)
    n = asc.size
    alt = np.empty_like(base)
    lo, hi = 0, n-1
    for i in range(n):
        if i%2==0:
            alt[i] = asc[lo]; lo+=1
        else:
            alt[i] = asc[hi]; hi-=1
    return alt
```

```
def _phi_permutation(base, rng):
    n = base.size
    # choose step coprime to n near golden ratio
    step = int(round(n / ((1+np.sqrt(5))/2)))
    # ensure gcd(step,n)=1
    while np.gcd(step, n) != 1:
        step = (step + 1) % n
        if step == 0:
            step = 1
    perm = np.empty_like(base)
    asc = np.sort(base)
    pos = 0
    for v in asc:
        perm[pos] = v
        pos = (pos + step) % n
    return perm

def _weight_perturbation(a, deadline, rng):
    a = a.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)
    # temperature
    T0 = best_score * 0.1
    T = T0
    cooling = 0.9999
    avg = np.mean(a)
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # choose delta relative to avg
        delta = rng.uniform(-avg*0.02, avg*0.02)
        if a[i] - delta < 0: 
            continue
        a[i] -= delta
        a[j] += delta
        # renormalize to keep sum constant? The move conserves sum, so sum unchanged.
        # no need to project.
        cur = _max_convolution_fft(a)
        dE = cur - best_score
        if cur < best_score or rng.random() < np.exp(-dE / T):
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
        T *= cooling
    return best

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.  The strategy focuses on a single long sequence
(≈20 000 entries) because the lower bound of the objective is roughly
S²/(2n‑1) ≈ 1, so larger n offers more room for improvement.  The pipeline
is:

1.  Initialise a shape (power‑law, exponential or truncated linear) and
    normalise it to the required ℓ¹‑norm Σa = √(2 n).
2.  Apply a few deterministic orderings (ascending, descending,
    alternating, a golden‑ratio step permutation) and keep the best.
3.  Refine the multiset of heights with a projected Adam optimiser that
    minimises a smooth log‑sum‑exp surrogate of ``max(a*a)``; the
    temperature β is annealed to a large value (≈10⁹) so the surrogate
    approaches the true max.
4.  Optimise the ordering of the refined multiset by simulated‑annealing
    swaps with a very slow cooling schedule.
5.  Perform a final local “mass‑transfer’’ hill‑climb: repeatedly move a
    tiny amount of mass from one index to another and keep the move if it
    reduces the maximum convolution entry.
6.  Return the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s , x ≥ 0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0 , Σ x_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ_exp(β·b_k) where b = a*a (full convolution).
    Returns (grad, b) where grad has the same shape as a.
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)          # shape (2n‑1,)

    # gradient = 2·(a ⋆ p)   (⋆ = convolution with reversed a)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation of the soft‑max surrogate of max(a*a).
    Runs until the supplied deadline (unix time) is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e9
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate that keeps the ℓ¹‑norm stable
        lr = 0.5 * target_sum / ((beta + 1.0) * (np.sqrt(v_hat).mean() + 1e-8))

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex Σa = √(2 n)
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Permutation annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing on the ordering of a fixed multiset.
    Swaps are proposed; the L¹‑norm is invariant.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9          # initial temperature
    cooling = 0.99985                     # very slow cooling
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny scramble of a random 2 % of entries
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                tmp = a[mask].copy()
                rng.shuffle(tmp)
                a[mask] = tmp
                cur = _max_convolution_fft(a)
                if cur < best_score:
                    best_score = cur
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Local mass‑transfer hill‑climb (preserves Σa)
# ----------------------------------------------------------------------


def _weight_perturbation(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Small random transfers of mass between two coordinates.
    Accepts every improving move and occasional uphill moves (simulated annealing).
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.1 + 1e-9
    cooling = 0.9999
    avg = np.mean(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.uniform(-avg * 0.02, avg * 0.02)
        if a[i] - delta < 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur < best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True
        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
        T *= cooling
    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponential, truncated linear)
# ----------------------------------------------------------------------


def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ¹‑norm = target."""
    s = np.sum(v)
    if s == 0.0:
        return np.full_like(v, target / v.size)
    a = v * (target / s)
    return np.maximum(a, 0.0)


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = (i + shift) ** (-p)
    return _scale_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = np.exp(-rate * (i + shift))
    return _scale_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * i)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _scale_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------


def _alternating_order(base: np.ndarray) -> np.ndarray:
    """Small → large → small → large … (starting with small)."""
    asc = np.sort(base)
    n = asc.size
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for k in range(n):
        if k % 2 == 0:
            alt[k] = asc[lo]
            lo += 1
        else:
            alt[k] = asc[hi]
            hi -= 1
    return alt


def _phi_permutation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Permutation based on a step close to the golden‑ratio.
    Guarantees a coprime step for (almost) all n.
    """
    n = base.size
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    step = int(round(n / phi))
    # make sure step is coprime to n
    while math.gcd(step, n) != 1:
        step = (step + 1) % n
        if step == 0:
            step = 1
    asc = np.sort(base)
    perm = np.empty_like(base)
    pos = 0
    for v in asc:
        perm[pos] = v
        pos = (pos + step) % n
    return perm
