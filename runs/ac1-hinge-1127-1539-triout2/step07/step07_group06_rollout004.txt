def _adam_softmax_descent(a0, deadline, rng):
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0*n)
    best_a = a.copy()
    max_b = _max_convolution(a)
    best_val = max_b

    beta = 1.0
    max_beta = 4096.0
    beta_mul = 1.5
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    # step size base
    base_lr = target * 0.12

    while time.time() < deadline:
        t += 1
        # compute gradient and b
        g, b = _gradient_softmax(a, beta)
        cur_val = float(b.max())
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = base_lr / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat
        # projection onto simplex
        a = _project_to_simplex(a, target)

        # optional jitter occasionally
        if t % 300 == 0:
            jitter_scale = target * 0.001 / (beta + 1.0)
            a += rng.normal(scale=jitter_scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # evaluate current max via b (since we computed)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam GD improvement β={beta:.1f} val={best_val:.6f}")

        # increase beta every some iterations
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * beta_mul, max_beta)

    return best_a, best_val

"""
Improved step‑function optimizer.

The evaluation metric (after the ℓ¹‑normalisation Σa = √(2·n))
is exactly the maximum entry of the self‑convolution a*a.
The algorithm consists of three phases:

1.  Fast incremental local search with a temperature schedule and
    a donor/receiver choice that explicitly avoids the current peak.
    Moves are cheap (O(n)) because the convolution is updated
    incrementally; occasional smoothing and jitter moves are applied
    with full FFT recomputation.

2.  Polarised gradient descent on a soft‑max approximation of the peak.
    The gradient is obtained via FFT (same formula as the reference)
    and an Adam optimiser is used; after each step the vector is
    projected onto the simplex {a ≥ 0, Σa = √(2·n)}.

3.  The best candidate from all attempts (different lengths, warm start)
    is returned after stripping trailing zeros.

The whole routine respects the 1000 s budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create (i+1)^(-p) of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (Σa = √(2·n) assumed)."""
    return float(_conv_fft(a).max())


def _apply_transfer(
    a: np.ndarray,
    conv: np.ndarray,
    donor: int,
    receiver: int,
    delta: float,
) -> None:
    """
    Transfer `delta` from `donor` to `receiver` and update the convolution
    exactly (O(n) work).  The function mutates both `a` and `conv`.
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    a_before = a.copy()

    # linear part
    conv[donor : donor + n] -= 2.0 * delta * a_before
    conv[receiver : receiver + n] += 2.0 * delta * a_before

    # quadratic part
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # coefficients
    a[donor] -= delta
    a[receiver] += delta

    # protect against tiny numerics
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)

# 1) Fast incremental local search (improved)
# ----------------------------------------------------------------------


def _improved_local_search(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Incremental simulated‑annealing local search.
    Returns the best sequence found and its max‑convolution value.
    """
    target = np.sqrt(2.0 * n)

    # initialise ---------------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), mode="constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)
    conv = _conv_fft(a)

    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total = deadline - start

    # scheduling parameters ------------------------------------------------
    start_temp = 1.0
    final_temp = 1e-4
    start_frac = 0.8
    final_frac = 0.005

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total

        # cooling & step‑size schedules
        temp = start_temp * (final_temp / start_temp) ** tau
        max_frac = start_frac * (final_frac / start_frac) ** tau

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass‑transfer move (≈80 % of the steps)
        # --------------------------------------------------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)

            # donors that actually contribute to the peak
            w = a[idx] * a[k_max - idx]
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            # receiver: prefer indices that contribute little to the current peak
            contrib = np.zeros(n, dtype=np.float64)
            mask = (k_max - np.arange(n) >= 0) & (k_max - np.arange(n) < n)
            contrib[mask] = a[k_max - np.arange(n)[mask]]
            inv = 1.0 / (contrib + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] new best max={best_max:.6f} (n={n})")
            else:
                # revert
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing window (≈10 % of the steps)
        # --------------------------------------------------------------
        elif move_type < 0.95:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            win_len = rng.integers(2, max_window + 1)
            end_win = min(start_win + win_len, n)

            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) Gaussian jitter (≈10 % of the steps)
        # --------------------------------------------------------------
        else:
            scale = target * 0.002 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved max={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# 2) Adam‑based gradient descent on a soft‑max approximation
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # softmax weights over convolution indices

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]  # gradient w.r.t a
    return g, b


def _adam_softmax_descent(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max proxy of the peak.
    Returns the best sequence encountered and its max‑convolution value.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    beta = 1.0
    max_beta = 4096.0
    beta_factor = 1.5

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    base_lr = target * 0.12

    while time.time() < deadline:
        t += 1
        g, b = _gradient_softmax(a, beta)
        cur_val = float(b.max())

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = base_lr / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target)

        # occasional jitter to escape shallow plateaus
        if t % 300 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam improvement β={beta:.1f} val={best_val:.6f}")

        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine runs an improved local‑search, followed by an Adam
    soft‑max polish, and finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.size
        if time.time() < deadline - 5.0:
            sub_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.1)
            seq, val = _improved_local_search(n0, sub_deadline, rng, init_seq=seq0)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] warm‑start n={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Run a few independent searches with increasing lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000]  # longer sequences give more freedom
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        # allocate at most 250 s per size, but honour the global deadline
        chunk = min(remaining * 0.3, 250.0)
        sub_deadline = time.time() + chunk
        seq, val = _improved_local_search(n, sub_deadline, rng)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final Adam‑based polish (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        extra_deadline = time.time() + (deadline - time.time()) * 0.3
        seq, val = _adam_softmax_descent(best_seq, extra_deadline, rng)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] final polish score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Remove trailing zeros (they do not affect the objective)
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(
        f"[{time.time():.1f}] final best approx max={best_score:.6f}, length={len(best_seq)}"
    )
    return best_seq.tolist()
