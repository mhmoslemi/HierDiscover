"""
Propose a step function minimizing max convolution (score == max(b)).
Uses Adam-accelerated projected subgradient descent on a soft‑max
approximation of the max entry, several random restarts with power‑law
initialisations, and a final simulated‑annealing polish.
"""

import time
import math
import numpy as np

def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x>=0, sum x = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # avoid overflow/underflow
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   max_beta: float = 1 << 20,
                   base_lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation of the soft‑max smoothed
    max‑convolution. Returns the best sequence seen and its max
    convolution value.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.astype(np.float64), target_sum)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 1.0
    while time.time() < deadline:
        t += 1
        # increase β gradually
        if t % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        grad, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * target_sum / (np.sqrt(v_hat) + eps)  # element‑wise
        a -= step * m_hat

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter

        # projection onto simplex (keeps ℓ¹‑norm fixed)
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:6d} β={beta:.0f} best={best_val:.6f}")

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 5.0,
                         end_temp: float = 1e-4,
                         max_frac_start: float = 0.4,
                         max_frac_end: float = 0.01,
                         max_window: int = 80) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves mass between indices and locally smooths.
    The temperature and allowed transfer fraction decay exponentially.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.size
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    conv = np.fft.irfft(np.fft.rfft(a, n=size_fft) ** 2, n=size_fft)[: 2 * n - 1]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # pick move type
        move = rng.random()
        if move < 0.65:   # mass transfer
            # index contributing most to the current maximum
            k = int(np.argmax(conv))
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            sens = a[k - idx]  # contribution of a[idx] to b[k]
            # donor – largest product a[i]*sens[i]
            weight = a[idx] * sens
            donor_idx = idx[np.argmax(weight)] if weight.size else rng.integers(n)
            # receiver – a location with small contribution
            low_thr = np.percentile(weight, 20) if weight.size else 0.0
            candidates = idx[weight <= low_thr]
            if candidates.size:
                receiver_idx = rng.choice(candidates)
            else:
                receiver_idx = rng.integers(n)

            if donor_idx != receiver_idx:
                delta = a[donor_idx] * rng.random() * max_frac
                if delta > 0:
                    a[donor_idx] -= delta
                    a[receiver_idx] += delta
                    move_type = "transfer"
                else:
                    move_type = None
            else:
                move_type = None

        elif move < 0.90:  # smoothing window
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            old_seg = a[start:end].copy()
            avg = old_seg.mean()
            a[start:end] = avg
            move_type = "smooth"
            saved = (start, end, old_seg)

        else:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score for the tentative state
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _max_convolution_fft(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor_idx] += delta
                a[receiver_idx] -= delta
            elif move_type == "smooth":
                s, e, seg = saved
                a[s:e] = seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (target_sum / cur_sum)
            conv = np.fft.irfft(np.fft.rfft(a, n=size_fft) ** 2, n=size_fft)[: 2 * n - 1]
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with minimal max convolution.
    The method consists of:
    1. a handful of power‑law initialisations,
    2. Adam‑accelerated projected optimisation on a soft‑max surrogate,
    3. a short simulated‑annealing polish.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    # problem size – a bit larger than previous runs gives more flexibility
    if budget_s > 800:
        n = 15000
    elif budget_s > 400:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial candidates (power‑law exponents and a random one)
    # ------------------------------------------------------------------
    exponents = [0.45, 0.5, 0.55, 0.6, 0.65, 0.7]
    inits = [_power_law_sequence(n, p, target_sum) for p in exponents]

    rng = np.random.default_rng()
    random_vec = rng.random(n)
    random_vec = _project_to_simplex(random_vec, target_sum)
    inits.append(random_vec)

    # if a previous best is present, add it (resized)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        prev_resized = np.interp(np.linspace(0, 1, n),
                                 np.linspace(0, 1, prev.size),
                                 prev)
        inits.append(_project_to_simplex(prev_resized, target_sum))

    best_seq = None
    best_val = float("inf")

    # allocate roughly equal time to each restart
    remaining = deadline - time.time()
    per_restart = remaining / max(len(inits), 1)

    for idx, init in enumerate(inits):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_restart
        a_opt, val_opt = _adam_optimize(init, target_sum, sub_deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Restart {idx} improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing refinement (uses the remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        a_refined, val_refined = _simulated_annealing(best_seq, deadline)
        if val_refined < best_val:
            best_seq = a_refined
            best_val = val_refined
            print(f"[{time.time():.2f}] SA polish improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and convert to plain list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final exact evaluation (if evaluate_sequence is available)
    try:
        exact = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Exact benchmark score = {exact:.6f}")
    except Exception:
        pass

    return final_seq

Search for a non‑negative step function with minimal max‑convolution.

The algorithm consists of three stages:
1. A small pool of power‑law initialisations (and a random one) is built.
2. Each candidate is refined by an Adam‑accelerated projected optimisation
   of a soft‑max approximation of the max‑convolution.  β for the soft‑max
   is increased exponentially so the surrogate becomes the true max
   towards the end of the run.
3. The best candidate is polished with a brief simulated‑annealing phase
   that moves mass between indices, smooths short windows and performs
   occasional swaps.

All updates keep the ℓ¹‑norm fixed (∑a = √(2·n)) by projection onto the
simplex, therefore the benchmark score equals the achieved maximum
convolution entry.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create (i+1)^(-p) of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max probabilities (numerically stable)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   max_beta: float = 1 << 20,
                   base_lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation of the soft‑max surrogate of
    max‑convolution.  Returns the best vector seen and its max‑convolution.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.astype(np.float64), target_sum)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 1.0
    while time.time() < deadline:
        t += 1
        if t % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        grad, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * target_sum / (np.sqrt(v_hat) + eps)  # element‑wise
        a -= step * m_hat

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter

        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:6d} β={beta:.0f} best={best_val:.6f}")

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         start_temp: float = 5.0,
                         end_temp: float = 1e-4,
                         max_frac_start: float = 0.4,
                         max_frac_end: float = 0.01,
                         max_window: int = 80) -> tuple[np.ndarray, float]:
    """
    Simple SA that transfers mass, smooths short windows and swaps entries.
    Temperature and allowed transfer fraction decay exponentially.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.size
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    conv = np.fft.irfft(np.fft.rfft(a, n=size_fft) ** 2, n=size_fft)[: 2 * n - 1]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        move_type = None
        # 0‑65 % : mass transfer
        if rng.random() < 0.65:
            k = int(np.argmax(conv))
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            sens = a[k - idx]                     # contribution of a[idx] to b[k]
            weight = a[idx] * sens                # product a_i * a_{k-i}
            donor = idx[np.argmax(weight)] if weight.size else rng.integers(n)

            low_thr = np.percentile(weight, 20) if weight.size else 0.0
            cand = idx[weight <= low_thr]
            receiver = rng.choice(cand) if cand.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"
        # 65‑90 % : local smoothing
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
        # remaining : swap
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score for the tentative state
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _max_convolution_fft(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_score:.6f}")
        else:
            # revert rejected move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[start:end] = old_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to shake the system
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (target_sum / cur_sum)
            conv = np.fft.irfft(np.fft.rfft(a, n=size_fft) ** 2, n=size_fft)[: 2 * n - 1]
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence with a low benchmark score.
    The workflow is:
      1. build several power‑law starting points,
      2. run Adam‑projected optimisation on each (time‑shared),
      3. polish the best result with simulated annealing,
    finally return the sequence as a plain Python list.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin for the runner

    # Choose a problem size that fits comfortably into the time budget
    if budget_s > 800:
        n = 15000
    elif budget_s > 400:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Construct initial candidates
    # ------------------------------------------------------------------
    exponents = [0.45, 0.5, 0.55, 0.6, 0.65, 0.7]
    inits = [_power_law_sequence(n, p, target_sum) for p in exponents]

    rng = np.random.default_rng()
    random_vec = rng.random(n)
    random_vec = _project_to_simplex(random_vec, target_sum)
    inits.append(random_vec)

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        # linear interpolation to the new length
        prev_resized = np.interp(np.linspace(0, 1, n),
                                 np.linspace(0, 1, prev.size),
                                 prev)
        inits.append(_project_to_simplex(prev_resized, target_sum))

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Adam optimisation on each initialisation (time‑shared)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    per_candidate = remaining / max(len(inits), 1)

    for idx, init in enumerate(inits):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_candidate
        a_opt, val_opt = _adam_optimize(init, target_sum, sub_deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Restart {idx} improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        a_ref, val_ref = _simulated_annealing(best_seq, target_sum, deadline)
        if val_ref < best_val:
            best_seq = a_ref
            best_val = val_ref
            print(f"[{time.time():.2f}] SA polish improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and convert to a plain list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional exact evaluation (the grader will also call evaluate_sequence)
    try:
        exact = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Exact benchmark score = {exact:.6f}")
    except Exception:
        pass

    return final_seq
