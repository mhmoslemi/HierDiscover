    v = rng.gamma(1.0, size=n)
    target = np.sqrt(2.0 * n)

def _exponential_family(n, rng):
    rate = rng.uniform(0.0001, 0.02)
    shift = rng.uniform(0, 5)
    idx = np.arange(n)
    v = np.exp(-rate * (idx + shift))
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(v, target)
def _powerlaw_family(n, rng):
    p = rng.uniform(0.2, 1.0)
    shift = rng.uniform(0,5)
    idx = np.arange(n)
    v = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(v, target)
def _mixed_family(n, rng):
    w = rng.uniform(0.2, 0.8)
    pow_seq = _powerlaw_family(n, rng)
    exp_seq = _exponential_family(n, rng)
    mix = w * pow_seq + (1-w) * exp_seq
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(mix, target)

def _simulated_annealing(a, deadline):
    n = a.size
    cur = a.copy()
    cur_obj = _max_convolution_fft(cur)
    best = cur.copy()
    best_obj = cur_obj
    T = cur_obj * 0.1
    cooling = 0.9997  # can be tuned
    step_scale = np.sqrt(2.0 * n) / n  # typical magnitude
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.8:
            # transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # propose delta
            delta = rng.normal(scale=step_scale)
            # limit delta
            delta = max(-cur[i], min(delta, cur[j]))
            if delta == 0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:
            # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = cur[j], cur[i]

        new_obj = _max_convolution_fft(new)
        dE = new_obj - cur_obj
        accept = False
        if dE <= 0:
            accept = True

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The objective equals the maximum entry of the self‑
convolution after normalising the L¹‑norm to √(2 n).  Our strategy is a
large‑scale stochastic search:

1.  Randomly generate many candidate multiset of heights from several
    simple parametric families (exponential, power‑law, mixtures, Dirichlet).
2.  For each candidate run a short **simulated‑annealing** on the
    continuous values: a random transfer of mass between two positions
    (keeps the L¹‑norm fixed) together with occasional swaps of positions.
    The temperature and the transfer step are annealed automatically.
3.  Keep the best few candidates (elite list) and finally polish the
    current global best with a projected gradient descent on a smooth
    log‑sum‑exp surrogate of the max‑convolution, followed by a dedicated
    permutation‑annealing phase.
4.  All computation of the objective uses a fast FFT based convolution.
   The whole routine respects the supplied time budget (default 1000 s).

The code contains only top‑level functions (no closures) and prints
progress information.  It can be started from a previously discovered
construction if the global variable ``height_sequence_1`` is present.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximal entry of the linear convolution a*a.
    Uses a power‑of‑two FFT length for speed.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Duchi et al. (2008) Euclidean projection onto the simplex
    {x ≥ 0, Σx = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for random initialisation
# ----------------------------------------------------------------------
def _exponential_family(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.00005, 0.02)
    shift = rng.uniform(0.0, 5.0)
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    target = math.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)


def _powerlaw_family(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 1.2)
    shift = rng.uniform(0.0, 5.0)
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = math.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)


def _dirichlet_family(n: int, rng: np.random.Generator) -> np.ndarray:
    # Gamma(1) = exponential → Dirichlet(1,…,1) (uniform on simplex)
    raw = rng.gamma(1.0, size=n)
    target = math.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)


def _mixed_family(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.uniform(0.2, 0.8)
    pow_seq = _powerlaw_family(n, rng)
    exp_seq = _exponential_family(n, rng)
    mix = w * pow_seq + (1.0 - w) * exp_seq
    target = math.sqrt(2.0 * n)
    return _project_to_simplex(mix, target)


_FAMILY_FUNCS = [_exponential_family,
                 _powerlaw_family,
                 _dirichlet_family,
                 _mixed_family]


# ----------------------------------------------------------------------
# Simulated annealing on the continuous values (mass transfers + swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a fast SA that preserves the L¹‑norm by transferring mass between
    two coordinates (or swapping them).  Returns the best sequence seen.
    """
    a = start.copy()
    n = a.shape[0]
    cur_obj = _max_convolution_fft(a)
    best = a.copy()
    best_obj = cur_obj

    # temperature initialisation – a fraction of the current objective
    T = max(cur_obj * 0.05, 1e-6)
    cooling = 0.9995
    # typical magnitude of a single entry (average)
    avg = math.sqrt(2.0 * n) / n
    step = avg * 0.5          # initial transfer scale
    it = 0
    accept_cnt = 0

    while time.time() < deadline:
        # ----- choose move -----
        if rng.random() < 0.85:        # transfer move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # propose a signed transfer delta; keep both entries non‑negative
            delta = rng.normal(scale=step)
            # clip to feasible interval
            delta = max(-a[i], min(delta, a[j]))
            if abs(delta) < 1e-12:
                continue
            new = a.copy()
            new[i] -= delta
            new[j] += delta
        else:                           # pure swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = a.copy()
            new[i], new[j] = a[j], a[i]

        new_obj = _max_convolution_fft(new)
        dE = new_obj - cur_obj
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < math.exp(-dE / T):
                accept = True

        if accept:
            a = new
            cur_obj = new_obj
            accept_cnt += 1
            if new_obj < best_obj:
                best_obj = new_obj
                best = new.copy()
                print(f"[{time.time():.1f}] SA improvement: obj={best_obj:.6f} n={n}")

        # ----- annealing schedule -----
        T *= cooling
        it += 1
        # adapt step size every few thousand iterations
        if it % 2000 == 0:
            # simple heuristic: shrink step if acceptance ratio low
            if accept_cnt < it * 0.15:
                step *= 0.9
            elif accept_cnt > it * 0.35:
                step *= 1.1
            accept_cnt = 0

    return best, best_obj


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max and projected gradient descent
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where
        conv = a*a (full linear convolution)
        grad = ∇_a ( (1/β)·log Σ exp(β·conv) )
              = 2·(a ⋆ p) ,  with p_k ∝ exp(β·conv_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # log‑sum‑exp probabilities
    max_conv = np.max(conv)
    exp_shifted = np.exp(beta * (conv - max_conv))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


def _gradient_descent_opt(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> np.ndarray:
    """
    Projected gradient descent on the log‑sum‑exp surrogate.
    The L¹‑norm is enforced by a simplex projection after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta (make surrogate sharper)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only) – useful for final polishing
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current best
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < math.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny random shuffle to shake the system
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask].copy()
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget, explores several
    random families, refines promising candidates with simulated annealing,
    gradient descent and a final permutation‑annealing phase, and finally
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf

    # Helper to update the global best
    # ------------------------------------------------------------------
    def _update_best(seq: np.ndarray, obj: float):
        nonlocal best_seq, best_obj
        if obj < best_obj:
            best_obj = obj
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST obj={best_obj:.6f} len={seq.size}")

    # ------------------------------------------------------------------
    # Optional warm start from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        n0 = init.size
        target = math.sqrt(2.0 * n0)
        init = _project_to_simplex(init, target)
        _update_best(init, _max_convolution_fft(init))
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Random generation + short SA (majority of the budget)
    # ------------------------------------------------------------------
    lengths = [8000, 10000, 12000, 15000, 18000, 20000]
    # give each length a rough probability (larger lengths a bit more)
    length_weights = [1, 1, 1, 1, 1, 1]

    # Approximate time we spend in this phase (70 % of total)
    gen_deadline = time.time() + 0.70 * (deadline - time.time())

    while time.time() < gen_deadline:

        # pick a random family

    # ------------------------------------------------------------------
    # 2. Gradient‑descent refinement of the current best (if any)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        gd_dead = min(deadline, time.time() + 20.0)   # up to ~20 s
        grads = _gradient_descent_opt(best_seq, gd_dead, rng)
        _update_best(grads, _max_convolution_fft(grads))

    # ------------------------------------------------------------------
    # 3. Final permutation‑annealing on the best solution
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

        best_obj = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 4. Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Final verification using the provided evaluate_sequence (optional)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
