def _adam_gradient_descent(a0, deadline, target_sum):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_score = _score(a)
    # hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 1.0
    max_beta = 65536.0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute gradient on softmax
        g, _ = _gradient_softmax(a, beta)
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size schedule: base_lr = target_sum * 0.5 / (beta+1) maybe
        base_lr = target_sum * 0.3 / (beta + 1.0)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)
        # jitter occasionally
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        # evaluate
        score = _score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t} β={beta:.1f} score={best_score:.6f}")
        # increase beta occasionally
        if t % 1000 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
    return best_a, best_score

"""
Optimiser for the step‑function benchmark.

The algorithm proceeds in four stages:
1. **Exploration** – a few short Adam‑optimised runs from random
   power‑law initialisations.  The best candidate is kept.
2. **Intensive Adam descent** – a long projected‑Adam optimisation on the
   soft‑max approximation of the max‑convolution.
3. **Simulated annealing** – a stochastic refinement that moves mass from
   high‑sensitivity positions to low‑sensitivity ones, smoothes short
   windows and occasionally swaps entries.
4. **Final polish** – a brief Adam pass to clean up any remaining noise.

All updates project onto the ℓ¹‑simplex {x≥0, Σx = √(2·n)}.
The objective is 2·n·max(a∗a)/(∑a)², approximated by a smooth soft‑max
during gradient steps.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact benchmark score for a."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create (i+1)^(-p) of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float) -> tuple[np.ndarray, float]:
    """Projected Adam descent on the soft‑max of max‑convolution."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = 1.0, 1.0
    t = 0

    # soft‑max temperature schedule
    beta = 1.0
    max_beta = 65536.0

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # gradient of the smoothed max
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate scaled inversely with current temperature
        base_lr = target_sum * 0.3 / (beta + 1.0)
        step = base_lr / (np.sqrt(v_hat) + eps)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                               size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:6d} β={beta:7.1f} score={best_val:.6f}")

        # temperature annealing (geometric)
        if t % 1000 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 10.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.5,
                         max_frac_end: float = 0.01,
                         max_window: int = 100) -> tuple[np.ndarray, float]:
    """Stochastic refinement that moves mass guided by sensitivities."""
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current maximum‑convolution index
        k_max = int(np.argmax(conv))

        # sensitivities of the active max entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None
        # ------------------- mass transfer -------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_score:.6f}")
        else:
            # rollback
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (math.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low benchmark score.
    The routine mixes Adam gradient descent, several short random restarts,
    and a final simulated‑annealing polish, all within the allotted time budget.
    """
    start = time.time()
    deadline = start + budget_s - 0.5       # small safety margin

    # problem size – larger n gives a richer search space but costs more time
    if budget_s >= 900:
        n = 18000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 300:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – exploration with a few short Adam runs (≈20% of the budget)
    # ------------------------------------------------------------------
    exploration_deadline = time.time() + (deadline - time.time()) * 0.20
    while time.time() < exploration_deadline:
        # random power‑law exponent for diversity
        p = rng.uniform(0.40, 0.70)
        a0 = _power_law_sequence(n, p, target_sum)
        a0 += rng.normal(scale=target_sum * 0.01, size=n)   # light jitter
        a0 = np.maximum(a0, 0.0)
        a0 = _project_to_simplex(a0, target_sum)

        # short Adam run (takes a few seconds)
        a_gd, val_gd = _adam_gradient_descent(a0, exploration_deadline, target_sum)

        # brief SA to possibly improve the candidate further
        if time.time() < exploration_deadline:
            a_sa, val_sa = _simulated_annealing(
                a_gd,
                exploration_deadline,
                start_temp=5.0,
                end_temp=1e-4,
                max_frac_start=0.5,
                max_frac_end=0.01,
                max_window=80,
            )
            if val_sa < best_score:
                best_score = val_sa
                best_seq = a_sa.copy()
        else:
            if val_gd < best_score:
                best_score = val_gd
                best_seq = a_gd.copy()

    # ------------------------------------------------------------------
    # Phase 2 – long Adam optimisation on the current best (≈60% of remaining time)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: deterministic power law if no candidate survived
        best_seq = _power_law_sequence(n, 0.55, target_sum)

    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.60
    a_gd, val_gd = _adam_gradient_descent(best_seq, gd_deadline, target_sum)
    if val_gd < best_score:
        best_score = val_gd
        best_seq = a_gd.copy()

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing on the rest of the budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            deadline,
            start_temp=10.0,
            end_temp=1e-5,
            max_frac_start=0.3,
            max_frac_end=0.01,
            max_window=120,
        )
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()

    # ------------------------------------------------------------------
    # Phase 4 – final quick Adam polish (a few seconds)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_final, val_final = _adam_gradient_descent(best_seq, deadline, target_sum)
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()

    # Trim insignificant trailing zeros (purely cosmetic)
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]

        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
