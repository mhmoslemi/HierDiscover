def propose_candidate(seed=..., budget_s=1000):

    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 5

    # base seq
    base_seq = globals().get('height_sequence_1', None)
    if base_seq is not None:
        base_seq = np.asarray(base_seq, dtype=float)
    else:
        base_seq = np.random.random(5000)
    # Ensure sum normalized to sqrt(2*n)
    def normalize(a):
        n = a.shape[0]
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s <= 0:
            a = np.full(n, 1.0)
            s = n
        return a / s * S

    # Phase 1: random candidates upsampled and scaled; also produce some families of longer length.
    # We'll maintain best_seq, best_score.

    # We'll generate candidates while time left > some threshold.

    best_seq = None
    best_score = np.inf

    # Determine max length we attempt maybe 30000
    max_len = 30000  # limit, you could adjust.
    # Main loop
    while time.time() < deadline - 30:  # keep 30 secs for final refinements
        # pick a length: sometimes upsample base, some random chosen
        if rng.random() < 0.6 and base_seq is not None:
            scale = rng.uniform(0.8, 2.0)
            n = int(base_seq.shape[0] * scale)
            n = max(1000, min(max_len, n))
            a = _upsample_seq(base_seq, n)
        else:
            n = rng.integers(2000, max_len+1)
            fam = rng.integers(0,7)
            if fam == 0:
                a = rng.random(n)
            elif fam == 1:
                a = np.sin(np.pi * (np.arange(n)+0.5) / n) ** rng.uniform(0.5,4.0)
            elif fam == 2:
                p = rng.uniform(0.2,4.0)
                idx = np.arange(n)
                a = ((idx+1)*(n-idx)) ** p
            elif fam == 3:
                rate = rng.uniform(0.9, 0.9999)
                left = rate ** np.arange(n)
                a = (left + left[::-1]) / 2.0
            elif fam == 4:
                p1 = rng.uniform(0.1,5.0)
                p2 = rng.uniform(0.1,5.0)
                x = np.linspace(0,1,n)
                a = (x**p1) * ((1-x)**p2)
            else:
                a = np.full(n, 0.0)
                # create block shape: high plateau middle, low tails
                mid_start = n//4
                mid_end = 3*n//4
                a[mid_start:mid_end] = 1.0
        # clip negatives etc.
        a = np.clip(a, 0.0, None)
        a = normalize(a)
        # random symmetrisation with prob 0.5
        if rng.random() < 0.5:
            a = _symmetrize(a)
        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] n={n} score={cur_score:.6f}")
    
    # Now best_seq has decent start.

    # Phase 2: Adam refinement with longer time budget.
    # We'll use _optimize_adam but modify hyperparameters for larger n
    # maybe more iterations.
    if best_seq is None:
        # fallback
        n = 5000
        best_seq = normalize(rng.random(n))
        best_score = np.max(_full_convolution(best_seq))

    # Provide a variant optimizer that uses a larger temperature schedule.
    def _optimize_adam_extended(a0, deadline, rng, enforce_sym=True):
        a = a0.copy()
        n = a.shape[0]
        S = np.sqrt(2.0 * n)
        lr = 0.6
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        alpha = 0.05
        alpha_mul = 1.06  # faster increase
        max_alpha = 400.0

        best_a = a.copy()
        best_score = np.max(_full_convolution(a))

        it = 0
        while time.time() < deadline:
            it += 1
            conv = _full_convolution(a)
            cur_max = conv.max()
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

            w = _smooth_weights(conv, alpha)
            grad = _conv_with_weights(a, w)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a -= lr * m_hat / (np.sqrt(v_hat) + eps)

            if enforce_sym:
                a = _symmetrize(a)

            a = _project_simplex(a, S)

            # occasional jitter
            if rng.random() < 0.02:
                i, j = rng.integers(0, n, size=2)
                if i != j and a[i] > 0:
                    delta = 0.07 * a[i] * rng.random()
                    a[i] -= delta
                    a[j] += delta
                    a = _project_simplex(a, S)

            alpha = min(alpha * alpha_mul, max_alpha)
            lr *= 0.99999
        return best_a, best_score

    # allocate time: half of remaining for Adam
    now = time.time()
    remaining = deadline - now
    # allocate 60% for Adam, 30% subgrad, 10% final hillclimb
    t_adam_deadline = now + 0.6 * remaining
    a_adam, sc_adam = _optimize_adam_extended(best_seq, t_adam_deadline, rng, enforce_sym=True)
    if sc_adam < best_score:
        best_score = sc_adam
        best_seq = a_adam.copy()
        print(f"[adam-extended] improved to {best_score:.6f}")

    # subgrad stage
    now = time.time()
    if now < deadline:
        t_sub_deadline = now + 0.3 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad(best_seq, t_sub_deadline, rng, enforce_sym=True)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_score:.6f}")

    # final hillclimb
    now = time.time()
    if now < deadline:
        a_loc, sc_loc = _local_swap_search(best_seq, deadline, rng, enforce_sym=True)
        if sc_loc < best_score:
            best_score = sc_loc
            best_seq = a_loc.copy()
            print(f"[final hillclimb] improved to {best_score:.6f}")

    # Return
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} length={len(final_seq)}")
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

"""
Improved step‑function optimisation.

The search consists of three phases:

1. **Exploratory sampling** – generate many candidate families (uniform,
   sinusoidal, power‑law, beta, geometric, block‑shaped) with lengths up to
   30 000.  If a previous best sequence is available it is up‑sampled to a
   larger size and used as a seed.  All candidates are normalised to
   Σa = √(2·n) so that the external objective equals the convolution peak.

2. **Projected Adam optimisation** – minimise a soft‑max approximation of the
   convolution peak with an increasing temperature (α up to 400).  The update
   uses Adam, respects the simplex constraint via Euclidean projection and
   optionally enforces exact mirror symmetry (which empirically helps).

3. **Refinement** – a sub‑gradient descent on the true max convolution entry,
   followed by a cheap random‑swap hill‑climb.  Both stages keep the simplex
   projection and optional symmetry.

The routine is time‑aware (budget – 5 s safety margin) and stores the best
sequence globally as ``height_sequence_1`` for reuse by later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices extracting the needed slice
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Optimisers used in the later phases
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the soft‑max approximation of the convolution peak."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max (more aggressive than before)
    alpha = 0.05
    alpha_mul = 1.06
    max_alpha = 400.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99999

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true maximum convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.14
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient (valid one for max entry)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random mass‑swap occasionally
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early stop if nothing improves for a while
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Tiny random‑swap hill‑climbing on the true max."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Helper for normalisation to Σa = √(2·n)
    # ------------------------------------------------------------------
    def _normalise(a: np.ndarray) -> np.ndarray:
        n = a.shape[0]
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0)
            s = n
        return a / s * S

    # ------------------------------------------------------------------
    # Retrieve previous best if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap random families (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1

        # length selection
        if base_seq is not None and rng.random() < 0.6:
            # up‑sample the existing construction to a new size
            scale = rng.uniform(0.8, 2.0)
            n = int(base_seq.shape[0] * scale)
            n = max(1000, min(30000, n))
            a = _upsample_seq(base_seq, n)
        else:
            n = int(rng.integers(2000, 30001))

        # pick a parametric family
        fam = rng.integers(0, 7)
        if fam == 0:
            a = rng.random(n)                                 # uniform
        elif fam == 1:
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** rng.uniform(0.5, 4.0)
        elif fam == 2:
            p = rng.uniform(0.2, 4.0)
            idx = np.arange(n, dtype=float)
            a = ((idx + 1) * (n - idx)) ** p
        elif fam == 3:
            rate = rng.uniform(0.9, 0.9999)                   # symmetric geometric
            left = rate ** np.arange(n)
            a = (left + left[::-1]) * 0.5
        elif fam == 4:
            p1 = rng.uniform(0.1, 5.0)
            p2 = rng.uniform(0.1, 5.0)
            x = np.linspace(0.0, 1.0, n)
            a = (x ** p1) * ((1 - x) ** p2)
        elif fam == 5:
            # block‑shaped: high plateau in the middle, zeros near edges
            a = np.zeros(n)
            left = int(0.25 * n)
            right = int(0.75 * n)
            a[left:right] = 1.0
        else:
            # pure power‑law decreasing from centre
            idx = np.arange(n, dtype=float) - (n - 1) / 2.0
            a = 1.0 / (1.0 + np.abs(idx) ** rng.uniform(0.3, 2.5))

        a = np.clip(a, 0.0, None)
        a = _normalise(a)

        # occasionally enforce symmetry (helps the landscape)
        if rng.random() < 0.5:
            a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case Phase 1 found nothing useful
    # ------------------------------------------------------------------

        best_score = np.max(_full_convolution(best_seq))

    # Phase 2 – Adam optimisation (≈60 % of remaining time)
    # ------------------------------------------------------------------

        best_seq = a_adam.copy()
        print(f"[adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent (≈30 % of remaining time)
    # ------------------------------------------------------------------

    return final_seq
