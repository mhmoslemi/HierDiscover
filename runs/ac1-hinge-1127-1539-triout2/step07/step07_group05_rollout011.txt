    rows = []
    cols = []
    data = []
    for row, p in enumerate(idxs):

        i_end = min(n-1, p)
        i = np.arange(i_start, i_end+1)

        return f.copy()

    sum_g = g.sum()
    if sum_g == 0:
        g = np.full_like(g, target_sum / n)  # fallback
    else:

def _max_convolution_fft(a):
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
    return float(conv.max())

    One LP refinement step.
    Returns new f.
    """

    b = np.convolve(f, f)
    M = b.max()

    raw = rng.exponential(scale=1.0, size=n)
    target = np.sqrt(2.0 * n)

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improves the ordering of a fixed multiset by random swaps."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current best
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[i], a[j] = a[j], a[i]                     # revert

        T *= cooling

        # occasional tiny global jitter
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Linear‑programming refinement (follows the “active‑set” idea)
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, idxs: np.ndarray) -> csr_matrix:
    """
    Build the matrix A (|idxs| × n) representing the linear
    constraints (f * g)[p] ≤ M for the selected indices p.
    Row `r` corresponds to index `p = idxs[r]` and contains the
    coefficients f[i] for all i with 0 ≤ i < n and 0 ≤ p‑i < n.
    """
    n = f.shape[0]
    rows, cols, data = [], [], []
    for r, p in enumerate(idxs):
        i_start = max(0, p - (n - 1))
        i_end = min(n - 1, p)
        i = np.arange(i_start, i_end + 1)
        j = p - i
        rows.extend([r] * len(i))
        cols.extend(j.tolist())
        data.extend(f[i].tolist())
    return csr_matrix((data, (rows, cols)), shape=(len(idxs), n), dtype=np.float64)


def _lp_step(f: np.ndarray, K: int, t: float, target_sum: float) -> np.ndarray:
    """
    One LP refinement step.
    * K   – number of active convolution entries (largest ones) that become constraints
    * t   – convex‑combination weight for the LP solution
    * target_sum – the prescribed ℓ¹‑norm ( √(2·n) )
    Returns a new sequence.
    """
    n = f.size
    b = np.convolve(f, f)
    M = b.max()

    # choose the K indices with the largest convolution value
    if K >= b.size:
        idxs = np.arange(b.size)
    else:
        idxs = np.argpartition(b, -K)[-K:]

    A = _build_constraint_matrix(f, idxs)

    # maximise Σ g_j   ⇔   minimise –Σ g_j
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(len(idxs), M),
                  bounds=bounds, method='highs')
    if not res.success:
        # if the LP fails we simply keep the current vector
        return f.copy()

    g = res.x
    sum_g = g.sum()
    if sum_g == 0.0:
        g = np.full(n, target_sum / n)
    else:
        g = g * (target_sum / sum_g)

    new_f = (1.0 - t) * f + t * g
    new_f = np.maximum(new_f, 0.0)
    new_f = _project_to_simplex(new_f, target_sum)
    return new_f


# ----------------------------------------------------------------------
# Helper families for random initialisation
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng) -> np.ndarray:
    """Draw a positive vector from an exponential (Dirichlet) distribution."""
    raw = rng.exponential(scale=1.0, size=n)
    target = np.sqrt(2.0 * n)
    return raw * (target / raw.sum())


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective (i.e. the maximal entry of its
    self‑convolution).  The routine proceeds in three stages:

    1. Random or warm‑start initialisation.
    2. A handful of linear‑programming refinement steps that push
       mass into regions where the current convolution is small.
    3. Simulated‑annealing permutation optimisation (swap moves).

    All work respects the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5           # safety margin

    best_seq = None
    best_obj = np.inf

    def _record(seq: np.ndarray):
        nonlocal best_seq, best_obj
        obj = _max_convolution_fft(seq)
        if obj < best_obj:
            best_obj = obj
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New global best obj={best_obj:.6f} n={seq.size}")

    # ------------------------------------------------------------------
    # 0. Warm start if a previous construction is available
    # ------------------------------------------------------------------
    candidates = []
    try:
        init = np.array(height_sequence_1, dtype=float)
        n0 = init.size
        init = _project_to_simplex(init, np.sqrt(2.0 * n0))
        _record(init)
        candidates.append(init)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Generate a few random starts (different lengths & families)
    # ------------------------------------------------------------------
    lengths = [6000, 8000, 10000, 12000]
    while time.time() < deadline - 20.0 and len(candidates) < 4:
        n = int(rng.choice(lengths))
        target = np.sqrt(2.0 * n)

        if rng.random() < 0.5:
            a = _random_dirichlet(n, rng)
        else:
            p = rng.uniform(0.30, 0.90)
            shift = rng.uniform(0.0, 5.0)
            a = _powerlaw_sequence(n, p, shift)

        rng.shuffle(a)                 # random permutation
        _record(a)
        candidates.append(a)

    # ------------------------------------------------------------------
    # 2. Refine each candidate (LP steps + permutation annealing)
    # ------------------------------------------------------------------
    for init_seq in candidates:
        if time.time() >= deadline:
            break
        n = init_seq.size
        target_sum = np.sqrt(2.0 * n)
        f = init_seq.copy()

        # ---- LP refinement -------------------------------------------------
        lp_iters = 8                     # a few fast interior‑point solves
        lp_K = max(150, int(0.02 * n))   # size of active set
        t = 0.25                         # convex‑combination coefficient
        for _ in range(lp_iters):
            if time.time() >= deadline - 5.0:
                break
            f = _lp_step(f, K=lp_K, t=t, target_sum=target_sum)
            _record(f)

        # ---- permutation annealing ----------------------------------------
        if time.time() < deadline - 5.0:
            f = _permutation_annealing(f, deadline)
            _record(f)

    # ------------------------------------------------------------------
    # 3. Final global annealing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _permutation_annealing(best_seq, deadline)
        best_obj = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 4. Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing negligible entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional verification using the official evaluator
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
