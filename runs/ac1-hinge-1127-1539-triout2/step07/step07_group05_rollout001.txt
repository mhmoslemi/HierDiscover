"""
propose_candidate
=================

Search for a non‑negative step function (a list of heights) that minimises
``evaluate_sequence`` which is exactly the maximal entry of the linear
self‑convolution (the L¹‑norm is fixed to √(2 n) inside the evaluator).

The algorithm consists of four stages:

1. **Wide random sampling** – generate many candidates from several
   parametric families (power‑law, exponential, linear, quadratic,
   mixed).  For each multiset we try a few deterministic orderings
   (descending, ascending, alternating) and keep the best few
   candidates (the *elite* list).

2. **Continuous refinement** – each elite candidate is refined by a
   projected Adam optimiser on a smooth log‑sum‑exp surrogate of the
   maximal convolution entry.  The optimiser respects the ℓ¹‑simplex
   constraint (sum = √(2 n)) after each step.

3. **Targeted mass‑transfer annealing** – starting from the refined
   vectors we perform a specialised simulated‑annealing move that
   transfers a small amount of mass away from the pair of indices that
   contribute most to the current maximal convolution entry.  The move
   is accepted with the usual Metropolis rule.  This directly attacks
   the peak of the convolution.

4. **Permutation annealing** – finally we improve the ordering of the
   best sequence by a standard swap‑based simulated annealing.

The whole routine respects the supplied time budget (default 1000 s)
and prints progress information.  It returns the best sequence found
as a plain list of floats.

import time

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_surrogate(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where

        conv = a*a (full convolution)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·conv_k) )
               = 2·(a ⋆ p)   with   p_k ∝ exp(β·conv_k)

    The gradient is computed via FFT for speed.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities (log‑sum‑exp trick)
    max_c = np.max(conv)
    exp_shifted = np.exp(beta * (conv - max_c))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient = 2·(a ⋆ p)  (linear convolution)
    # Build convolution a_rev * p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    tmp = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * tmp[n - 1: n - 1 + n]
    return grad, conv

# 2.  Projected Adam optimiser on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser (projected onto the ℓ¹‑simplex) for the smooth max
    surrogate.  Parameter β is annealed exponentially.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr0 = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta = 1.0               # start with a gentle smoothness
    max_beta = 1e6
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _softmax_surrogate(a, beta)

        # Adam update on the *unprojected* vector
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # step size normalisation: keep the ℓ¹‑norm close to the target
        a -= step
        a = _project_to_simplex(a, target_sum)

        # small random jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (make the surrogate sharper)
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        it += 1

    return a

def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    The L¹‑norm stays unchanged, only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # Initial temperature proportional to current best
    T = best_score * 0.02 + 1e-9
    cooling = 0.9994
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        if cur_score <= best_score:
            best_score = cur_score
            best = a.copy()
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(cur_score - best_score) / T):
                # keep the worse state (allowing uphill moves)
                pass
            else:
                # revert swap
                a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny random shuffle of a small fraction of entries
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best

def _mass_transfer_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Focused simulated annealing that moves a small amount of mass away
    from the pair of indices that contribute most to the current maximal
    convolution entry.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    rng = np.random.default_rng()
    T = best_score * 0.015 + 1e-9
    cooling = 0.9996

    while time.time() < deadline:
        # locate the shift with maximal convolution
        conv = np.fft.irfft(
            np.fft.rfft(a, n=1 << ((2 * n - 1).bit_length())) ** 2,
            n=1 << ((2 * n - 1).bit_length())
        )[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # indices that form the k_max shift
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        idx = np.arange(i_min, i_max + 1)
        jdx = k_max - idx

        # contribution of each pair (i, j)
        contrib = a[idx] * a[jdx]
        # give larger weight to pairs with larger product
        probs = contrib / (contrib.sum() + 1e-18)

        # pick a pair (i, j) proportionally to its contribution
        chosen = rng.choice(idx.size, p=probs)
        i = int(idx[chosen])
        j = int(jdx[chosen])

        # pick two distinct target locations
        t1, t2 = rng.choice(n, size=2, replace=False)

        # amount to transfer – a modest fraction of the smaller donor
        max_transfer = min(a[i], a[j]) * 0.08
        if max_transfer <= 0.0:
            # nothing to move; skip
            continue
        delta = rng.random() * max_transfer

        # perform the mass move
        a[i] -= delta
        a[j] -= delta
        a[t1] += delta
        a[t2] += delta

        # keep non‑negativity by projecting tiny negatives to 0
        a = np.maximum(a, 0.0)

        # recompute objective
        cur_score = _max_convolution_fft(a)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(cur_score - best_score) / T):
                # keep the worse state
                pass
            else:
                # revert move
                a[i] += delta
                a[j] += delta
                a[t1] -= delta
                a[t2] -= delta
                a = np.maximum(a, 0.0)

        T *= cooling

    return best

def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _mixed_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.uniform(0.2, 0.8)

    p = rng.uniform(0.30, 0.90)
    shift_p = rng.uniform(0.0, 5.0)
    pow_seq = _powerlaw_sequence(n, p, shift_p)

    rate = rng.uniform(0.0001, 0.02)
    shift_e = rng.uniform(0.0, 5.0)
    exp_seq = _exponential_sequence(n, rate, shift_e)

    mix = w * pow_seq + (1.0 - w) * exp_seq
    return _project_to_simplex(mix, np.sqrt(2.0 * n))

        # ----- orderings -----
        asc = np.sort(base)          # small → large
        desc = asc[::-1]             # large → small

        # alternating (small, large, small, …)
        alt = np.empty_like(asc)
        lo, hi = 0, n - 1
        for i in range(n):
            if i % 2 == 0:
                alt[i] = asc[lo]
                lo += 1
            else:
                alt[i] = asc[hi]
                hi -= 1

        # single random shuffle
        shuf = asc.copy()
        rng.shuffle(shuf)

        for cand in (desc, asc, alt, shuf):
            _consider(cand)

        # occasional cheap permutation annealing on the *current* global best
        if best_seq is not None and rng.random() < 0.08:
            tmp = _permutation_annealing(best_seq, time.time() + 0.3)
            _consider(tmp)

    # ------------------------------------------------------------------
    # Phase 1 – continuous Adam refinement of elite candidates
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        sub_dead = min(deadline, time.time() + 15.0)   # up to ~15 s per elite

        refined = _adam_opt(seed_seq, sub_dead)

        # Phase 2 – targeted mass‑transfer annealing
        refined = _mass_transfer_annealing(refined, deadline)

        # Phase 3 – permutation annealing (final ordering tweak)
        refined = _permutation_annealing(refined, deadline)

        _consider(refined)

    # ------------------------------------------------------------------
    # Phase 4 – final global permutation annealing if time permits
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _permutation_annealing(best_seq, deadline)
        best_obj = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim near‑zero tail (cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional final verification (does not affect the returned value)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] SEARCH FINISHED – best obj = {best_obj:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
