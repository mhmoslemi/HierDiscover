def _conv_fft_two(x, y):
    n = len(x)+len(y)-1
    size = 1 << ((n).bit_length())
    fx = np.fft.rfft(x, n=size)

def subgradient_descent(n, deadline, rng, init_seq=None):
    target_sum = 1.0  # we will normalize sum to 1 for compute; after final we can rescale to any but evaluation unchanged.
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), 'constant')
    else:
        # random init: sample Dirichlet distribution to ensure sum=1
        a = rng.dirichlet(alpha=np.ones(n))
    # ensure positive
    a = np.maximum(a, 1e-12)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    conv = _conv_fft(a)
    max_b = conv.max()
    best_val = max_b

    start = time.time()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_b = conv.max()
        # compute gradient
        I = (conv >= max_b - 1e-12).astype(np.float64)
        grad = 2.0 * _conv_fft_two(a[::-1], I)[n-1 : 2*n-1]  # shape n

        # step size schedule
        # use diminishing step size: eta = step0 / sqrt(it+1)
        step0 = 0.5
        eta = step0 / np.sqrt(it + 1)

        # multiplicative update (Exponentiated Gradient)
        a = a * np.exp(-eta * grad)
        # avoid all zeros
        a = np.maximum(a, 1e-30)
        a = a / a.sum()  # project onto simplex (L1)
        # could also use the projection function.

        # Evaluate new max
        conv = _conv_fft(a)
        max_b = conv.max()
        if max_b < best_val - 1e-12:

            best_a = a.copy()

            a += rng.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        it += 1

    return best_a, best_val

    g_i = 2·∑_{j : i+j ∈ I} a_j ,
where I is the set of convolution indices attaining the current maximum.
The gradient is obtained efficiently by a single FFT‑based convolution
(a[::-1] * I).  The update step is
    a ← a·exp(−η·g)   and re‑normalised onto the simplex.
A modest amount of Gaussian jitter is added periodically to escape plateaus.
Several lengths are tried (the time budget is split between them) and the best
candidate is finally polished with the soft‑max gradient descent from the
reference solution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary real vectors using FFT."""
    n = x.size + y.size - 1
    size = _next_pow2(n)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (ℓ¹‑norm is assumed = 1)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Sub‑gradient / exponentiated‑gradient optimisation
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray,
                      conv: np.ndarray,
                      eps: float = 1e-12) -> np.ndarray:
    """
    Perform one exponentiated‑gradient step aimed at reducing the peak of
    the self‑convolution.  ``conv`` must be a*a (full length 2·n‑1).
    """
    n = a.size
    max_b = conv.max()
    # binary mask of indices that are (essentially) maximal
    I = (conv >= max_b - eps).astype(np.float64)

    # gradient of the max‑convolution:
    #   g_i = 2 * Σ_{j} a_j * I_{i+j}
    # which is a convolution of a[::-1] with I.
    grad_full = _conv_fft_two(a[::-1], I)           # length 3·n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]       # extract the n relevant entries
    return grad, max_b


def _exponential_update(a: np.ndarray,
                        grad: np.ndarray,
                        step: float) -> np.ndarray:
    """
    Multiplicative (exponentiated) mirror descent update on the simplex.
    """
    # prevent overflow/underflow
    a = a * np.exp(-step * grad)
    a = np.maximum(a, 1e-30)            # keep strictly positive for the log‑step
    a = a / a.sum()
    return a


def _sgd_descent(n: int,
                 deadline: float,
                 rng: np.random.Generator,
                 init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using exponentiated‑gradient steps.
    Returns the best sequence seen and its *max‑convolution* (i.e. the value
    2·n·max_b/(∑a)², because we keep ∑a = 1).
    """
    # initialise -----------------------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        # Dirichlet start gives a random point on the simplex
        a = rng.dirichlet(alpha=np.ones(n).astype(np.float64))

    a = np.maximum(a, 1e-12)
    a = _project_to_simplex(a, 1.0)

    # keep the best seen ---------------------------------------------------------
    conv = _conv_fft(a)
    _, cur_max = _subgradient_step(a, conv)
    best_a = a.copy()
    best_val = cur_max

    start = time.time()
    it = 0
    while time.time() < deadline:
        # compute current convolution and gradient
        conv = _conv_fft(a)
        grad, cur_max = _subgradient_step(a, conv)

        # step‑size schedule (diminishing)
        eta = 0.8 / np.sqrt(it + 1)

        # multiplicative update + simplex projection (already normalised)
        a = _exponential_update(a, grad, eta)

        # occasional Gaussian jitter to explore new basins
        if it % 500 == 0 and it > 0:
            scale = 0.02 * (1.0 - (time.time() - start) / (deadline - start))
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, 1.0)

        # record improvement
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} new best max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Soft‑max gradient descent (reference polishing)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()
    a_rev = a[::-1]
    pr = np.fft.rfft(p, n=size)
    ar = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(ar * pr, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = a.sum()                       # we keep the actual ℓ¹‑norm
    best_a = a.copy()
    best_val = _max_convolution(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD polish β={beta:.1f} max={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    ``evaluate_sequence``.  The strategy is:

    1. If a previous construction ``height_sequence_1`` exists, warm‑start the
       optimiser with it.
    2. Run the exponentiated‑gradient optimiser on several problem sizes
       (the time budget is split proportionally).  The optimiser works
       directly on the ℓ¹‑simplex and uses the exact sub‑gradient of the
       max‑convolution.
    3. Polishing: the best candidate is refined with the soft‑max gradient
       descent from the reference code.
    4. Trailing zeros are removed before returning.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    best_seq = None
    best_score = float("inf")                    # this will hold the *full* evaluation

    # ------------------------------------------------------------------
    # Helper to evaluate and keep the best (full objective, not just max)
    # ------------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray):
        nonlocal best_seq, best_score
        # evaluate via the given external routine (imported in the environment)
        val = evaluate_sequence(candidate.tolist())
        if val < best_score:
            best_score = val
            best_seq = candidate.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Warm‑start from the previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.size
        if time.time() < deadline - 5.0:
            run_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.20)
            cand, _ = _sgd_descent(n0, run_deadline, rng, init_seq=seq0)
            _maybe_update(cand)

    # ------------------------------------------------------------------
    # 2) Explore a handful of lengths with the exponentiated‑gradient method
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2500, 3000]   # modest spread
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        # give each size a slice of the remaining budget (≈20 % each)
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.20, 150.0)
        run_deadline = time.time() + slice_time
        cand, _ = _sgd_descent(n, run_deadline, rng)
        _maybe_update(cand)

    # ------------------------------------------------------------------
    # 3) Final polishing with the soft‑max gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        extra_deadline = time.time() + (deadline - time.time()) * 0.25
        polished, _ = _annealed_gradient_descent(best_seq, extra_deadline)
        _maybe_update(polished)

    # ------------------------------------------------------------------
    # 4) Strip trailing zeros (they do not affect the objective)
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] FINAL BEST ≈ {best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
