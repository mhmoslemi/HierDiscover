import time, numpy as np

def _max_convolution_fft(a):
    n = a.size
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return float(conv.max())

def _conv_fft(a):
    n = a.size
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

def _project_to_simplex(v, s=1.0):
    # Euclidean projection onto probability simplex.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1+n]   # middle part size n
    return grad, b

def _score(a):
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _arcsine_sequence(n):
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    raw = 1.0 / np.sqrt(x * (1.0 - x))
    a = raw / raw.sum()
    return a

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Construct a step sequence with a low benchmark score.
    The method uses the discrete arcsine distribution, which makes the
    self‑convolution almost flat (its continuous analogue yields a
    uniform distribution of the sum of two independent draws).  The
    initial sequence is then refined with a few iterations of projected
    gradient descent on a soft‑max approximation of the maximum
    convolution entry.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # a small safety margin

    # Choose n large enough to approach the continuous limit,
    # but not so large that FFT becomes a bottleneck.
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    # ------------------------------------------------------------------
    # 1) Initialise with arcsine distribution (sum normalised to 1)
    # ------------------------------------------------------------------
    a = _arcsine_sequence(n)
    a = _project_to_simplex(a, 1.0)   # ensure exact ℓ¹‑norm = 1

    best_a = a.copy()
    best_val = _score(best_a)
    print(f"[{time.time():.2f}] initial score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Short projected gradient descent on a soft‑max proxy of max(b)
    # ------------------------------------------------------------------
    beta = 1.0
    max_beta = 65536.0
    step_scale = 0.5   # base step size (relative to ℓ¹ norm)
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_scale / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, 1.0)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=step_scale * 0.003 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, 1.0)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    # ------------------------------------------------------------------
    # 3) Final clean‑up (optional tiny polishing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        # Perform a few extra iterations with a smaller step size.
        beta = max_beta
        for _ in range(200):
            g, _ = _gradient_softmax(best_a, beta)
            step = 0.05 * g / (np.linalg.norm(g) + 1e-12)
            best_a -= step
            best_a = _project_to_simplex(best_a, 1.0)

    # Trim trailing near‑zero entries for a compact output.
    final_arr = np.where(best_a > 1e-12)[0]
    if final_arr.size:
        best_a = best_a[: final_arr[-1] + 1]

    seq = best_a.tolist()
    # ------------------------------------------------------------------
    # 4) Report final evaluation using the provided benchmark.
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return seq

"""
A fresh search for a low‐benchmark step sequence.

The global optimum of the benchmark is obtained when the self‑convolution
a*a is as flat as possible.  A simple power‑law (a_i ∝ (i+1)^‑p) is a
good starting point – the previous state of the art used p≈0.55.
Here we explore a small family of parametrised shapes (power‑law and
geometric) and improve each candidate with a fast projected sub‑gradient
method that directly targets the maximum entry of the convolution.
The search stays within the time budget (default 1000 s) and finally
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·n‑1) using a real FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  Returns np.inf if sum is tiny."""
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – keeps ∑a = s and a_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial sequences
# ----------------------------------------------------------------------


def _power_law_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^‑p, normalised to unit ℓ¹‑norm."""
    i = np.arange(1, n + 1, dtype=np.float64)
    raw = i ** (-p)
    return raw / raw.sum()


def _geometric_seq(n: int, lam: float) -> np.ndarray:
    """
    Geometric decrease a_i ∝ exp(-lam·i), normalised.
    lam > 0 controls the steepness (larger lam → faster decay).
    """
    i = np.arange(n, dtype=np.float64)
    raw = np.exp(-lam * i)
    return raw / raw.sum()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(start: np.ndarray,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast sub‑gradient descent.
    The gradient of the current maximal convolution entry (k_max) is
        g_i = 2·a_{k_max‑i}   (if the index is in range, else 0).
    After each step we project back onto the probability simplex.
    Random jitter is injected occasionally to escape flat regions.
    """
    a = start.copy()
    a = _project_to_simplex(a, 1.0)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    step = 0.2          # initial additive step size
    decay = 0.995       # geometric decay of the step
    it = 0

    while time.time() < deadline:
        # 1) compute convolution and locate the current maximal index
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        # 2) sub‑gradient of the max entry w.r.t. a
        n = a.size
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        # derivative = 2·a_{k_max‑i}
        grad[idx] = 2.0 * a[k_max - idx]

        # 3) additive update + projection
        a -= step * grad
        a = _project_to_simplex(a, 1.0)

        # 4) occasional jitter to break symmetry
        if it % 400 == 0 and it > 0:
            jitter = rng.normal(scale=step * 0.5, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, 1.0)

        # 5) bookkeeping
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] sub‑gd it={it:5d} score={best_val:.6f}")

        # 6) step‑size decay
        step *= decay
        it += 1

    return best_a, best_val

# Main entry point required by the evaluation harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low benchmark

      1. Chooses a problem size `n` based on the allotted time.
      2. Generates a small pool of seed sequences (power‑law & geometric)

      3. Refines each seed with a fast projected sub‑gradient descent that
         directly minimises the maximal convolution entry.
      4. Returns the best sequence discovered.
    """

    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------

    # degrees of freedom (larger n → better theoretical lower bound).
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    # ------------------------------------------------------------------
    seeds = []

    for p in np.linspace(0.40, 0.70, 7):
        seeds.append(_power_law_seq(n, p))

    # geometric decays (λ ≈ 0 … 3/n, we express as λ = α / n)
    for α in [0.5, 1.0, 1.5, 2.0, 3.0]:
        lam = α / n
        seeds.append(_geometric_seq(n, lam))

    # keep the best seed found so far (helps later stages)
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    for idx, seed in enumerate(seeds):
