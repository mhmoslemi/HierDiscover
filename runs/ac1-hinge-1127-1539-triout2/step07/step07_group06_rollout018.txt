def propose_candidate(seed=42,budget_s=1000):
  rng = np.random.default_rng(seed)
  start = time.time()
  deadline = start + budget_s - 0.5

  best_seq = None
  best_score = float('inf')

  # we can test a few n values: start with smaller n for quick restarts, but larger n may give better final.
  n_candidates = [2000, 3000, 4000, 5000]
  
  for n in n_candidates:
    if time.time() >= deadline - 5:
        break
    # allocate time for this n
    remaining = deadline - time.time()
    # some portion of remaining time based on n (larger n less time)
    # allocate 0.5*remaining for this n if it's the last candidate.
    run_time = min(remaining * 0.4, 200.0)  # at most 200 sec per size.
    run_deadline = time.time() + run_time
    seq, score = _subgradient_optimize(n, run_deadline, rng)
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
        print(f"[{time.time():.1f}] n={n} new best score={best_score:.6f}")

  # final polishing? maybe a few more ADAM steps
  if best_seq is not None and time.time() < deadline - 5:
        n = best_seq.size
        extra_deadline = time.time() + (deadline - time.time()) * 0.3
        seq, score = _subgradient_optimize(n, extra_deadline, rng, init_seq=best_seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] final polish improve score={best_score:.6f}")

  # strip trailing zeros
  best_seq = np.array(best_seq)
  nz = np.where(best_seq>1e-12)[0]
  if nz.size>0:
      best_seq = best_seq[:nz[-1]+1]
  else:
      best_seq = np.empty(0)

  print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq)}")
  return best_seq.tolist()

def _subgradient_optimize(n, deadline, rng, init_seq=None):
    target = np.sqrt(2.0 * n)
    # init a
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size != n:
            # adjust size
            if a.size > n:
                a = a[:n]
            else:
                a = np.pad(a, (0,n-a.size), 'constant')
    else:
        # random dirichlet distribution
        a = rng.random(n)
        a = a / a.sum() * target
    a = _project_to_simplex(a, target)

    # ADAM parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2  # may adapt
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1

        # compute convolution and max
        conv = _conv_fft(a)
        cur_max = conv.max()
        # Determine max indices (maybe top 1)
        k_max = np.flatnonzero(conv >= cur_max - 1e-12)  # List of indices

        # Compute gradient g (subgradient) for max
        # For each k in k_max, g_k[i] = a[k-i] if valid else 0
        # Sum them
        g = np.zeros_like(a)
        for k in k_max:
            # compute start and end for i such that 0 <= k-i < n -> i in [k-n+1, k]
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # For i ranging i_start..i_end inclusive, we have g[i] += a[k - i]
            # Use slicing: g[i_start:i_end+1] += a[k-i_start : k-i_end-1 : -1] (reverse)
            # We can do direct loops but slice proper.
            # Let's compute slice length = i_end - i_start + 1
            length = i_end - i_start + 1
            # extract a segment reversed
            segment = a[k - i_start : k - i_end - 1 : -1] if length > 1 else a[k - i_start:k - i_end - 1:-1]
            # In Python slicing it's a[k-i_start : k-i_end-1 : -1]; but careful for inclusive endpoints.
            # Actually need segment = a[k - np.arange(i_start,i_end+1)] = a[ k - np.arange(i_start,i_end+1) ].
            # Simpler: use array indices.
            idx_range = np.arange(i_start, i_end+1)
            g[idx_range] += a[k - idx_range]

        # Multiply by 2 (derivative factor)
        g = 2.0 * g

        # ADAM update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # descent step (negative gradient)
        a_new = a - step

        # enforce non-negativity then project to simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target)

        # compute new objective
        conv_new = _conv_fft(a_new)
        new_max = conv_new.max()
        # Accept the step if improves or with probability (like simulated annealing)
        # Use simple acceptance: if new_max <= cur_max we accept; else maybe accept with small probability
        if new_max <= cur_max or rng.random() < 0.001:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] n={n} improvement max={best_max:.6f}")
        else:
            # optionally keep old a (skip)
            pass

        # occasional jitter
        if t % 200 == 0:
            scale = target * 0.001 * (1.0 - (time.time() - start) / (deadline - start))
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return best_a, best_max

"""
Candidate generation for the step‑function minimisation task.

The routine works in three phases within the allotted time budget:

1. **Fast random screening** – a few random power‑law sequences of several
   lengths are generated, scaled to the required ℓ¹‑norm and evaluated.
   The best among them is kept as a warm start.

2. **Incremental local search** – for a set of increasingly larger lengths
   an annealed mass‑transfer optimiser (the same engine that achieved the
   previous 1.5117 bound) is run for a substantial fraction of the remaining
   time.  The optimiser moves mass between indices while keeping the vector
   on the simplex {a ≥ 0, Σa = √(2·n)} and accepts moves that lower the
   current peak of the self‑convolution or that pass a Metropolis test.

3. **Projected‑gradient polishing** – the best sequence found so far is
   refined with an Adam‑style optimiser on a soft‑max approximation of the
   max‑convolution, again projecting onto the simplex after each step.

The whole procedure respects the 1000 s time limit and returns a list of
non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (assumes Σa = √(2·n))."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Fast objective value (max convolution) – sum is already √(2·n)."""
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Incremental mass‑transfer update (exact)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer an amount `delta` from `donor` to `receiver`.
    Updates both the coefficient vector `a` and the convolution array `conv`
    exactly (including the quadratic `delta**2` terms).
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    a_before = a.copy()

    # linear contribution:  -2·δ·a_before shifted by donor, +2·δ·a_before shifted by receiver
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic contribution from d*d
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # update the coefficients
    a[donor] -= delta
    a[receiver] += delta

    # protect against tiny negative values caused by rounding
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Local‑search core
# ----------------------------------------------------------------------


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Incremental local search for a given length `n`.
    Returns the best sequence found and its objective value (max convolution).
    """
    target = np.sqrt(2.0 * n)

    # ---- initialise ----------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        # schedules (warm → cold)
        max_frac = 0.5 * (0.005 / 0.5) ** tau          # max fraction of donor mass to move
        temp = 0.5 * (1e-5 / 0.5) ** tau               # temperature for Metropolis test

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass transfer (≈80 % of the moves)
        # --------------------------------------------------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]               # contribution of each index
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            # bias receiver towards small coefficients
            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            # Metropolis acceptance
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] n={n} new best max={best_max:.6f}")
            else:
                # revert the move
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing window (≈10 % of the moves)
        # --------------------------------------------------------------
        elif move_type < 0.95:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_window + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] n={n} smoothing improved max={best_max:.6f}")
                else:
                    # revert smoothing
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) Gaussian jitter (≈10 % of the moves)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] n={n} jitter improved max={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Projected gradient descent on a soft‑max approximation (reference code)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.

    The routine consists of three stages:
      * rapid random power‑law screening,
      * intensive incremental local‑search on several lengths,
      * final Adam‑style soft‑max gradient polishing.

    It respects the global time budget (default 1000 s) and returns the
    best sequence found as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0) quick random screening – obtain a decent warm start
    # ------------------------------------------------------------------
    for n in (1500, 2000, 2500):
        if time.time() >= deadline - 5.0:
            break
        target = np.sqrt(2.0 * n)
        for _ in range(5):
            p = rng.uniform(0.2, 1.2)
            a = _power_law_sequence(n, p, target)
            a = _project_to_simplex(a, target)
            cur = _max_convolution(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] screening n={n} p={p:.3f} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Incremental local‑search on increasingly larger sizes
    # ------------------------------------------------------------------
    # we allocate a larger slice of the remaining budget to the biggest
    # sizes – they give the algorithm more freedom to lower the peak.
    sizes = [3000, 3500, 4000, 4500, 5000]
    for n in sizes:
        if time.time() >= deadline - 5.0:
            break

        remaining = deadline - time.time()
        # give each size up to 0.3 of the remaining time but never more than 250 s
        time_for_this = min(remaining * 0.30, 250.0)

        run_deadline = time.time() + time_for_this

        # initialise with the current best (interpolated if lengths differ)
        init_seq = None
        if best_seq is not None:
            if best_seq.size == n:
                init_seq = best_seq
            else:
                # simple linear interpolation to the new length
                x_old = np.linspace(0, 1, best_seq.size)
                x_new = np.linspace(0, 1, n)
                init_seq = np.interp(x_new, x_old, best_seq)

        seq, score = _local_search(n, run_deadline, rng, init_seq=init_seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} → new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final Adam‑style soft‑max gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _annealed_gradient_descent(best_seq, extra_deadline)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 3) Strip trailing zeros – they do not affect the objective
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
