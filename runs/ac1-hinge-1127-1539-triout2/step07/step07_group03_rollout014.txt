def _build_conv_matrix(f):
    n = f.shape[0]
    row = []
    col = []
    data = []
    for j in range(n):
        start = j
        end = j + n
        rows_range = np.arange(start, end, dtype=int)
        row.extend(rows_range.tolist())
        col.extend([j] * n)
        data.extend([f[j]] * n)
    A = sparse.csr_matrix((data, (row, col)), shape=(2*n-1, n))
    return A

def _solve_lp_for_g(f, target_sum):
    n = f.size
    A = _build_conv_matrix(f)
    # Append column for -M
    M_col = -np.ones((2*n-1, 1))
    A_ub = sparse.hstack([A, M_col], format='csr')
    c = np.zeros(n+1)
    c[-1] = 1.0
    b_ub = np.zeros(2*n-1)
    # Equality for sum
    A_eq = sparse.csr_matrix(np.hstack([np.ones(n), np.zeros(1)])[np.newaxis, :])
    b_eq = np.array([target_sum])
    bounds = [(0, None)] * n + [(0, None)]
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                           A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x[:n]
    M_val = res.x[n]
    return g, M_val

"""
Propose a step‑function sequence with a low benchmark score.

Algorithm (≈1000 s budget)

1. **Small‑scale linear programming (if SciPy is available).**
   • Work on n_small≈500, solve a linear program that minimises the
     worst convolution value M for the current shape f:
       minimise   M
       s.t.       (f * g)[k] ≤ M   (k = 0 … 2·n_small‑2)
                 Σ g_i = √(2·n_small),  g_i ≥ 0
   • Iterate a few times, updating f ← ½ f + ½ g, keeping the best
     shape and up‑scaling it to the target size n.

2. **Initial candidates.**
   • The up‑scaled LP shape (if available).
   • The previous best global shape (height_sequence_1) if present.
   • Power‑law shapes a_i ∝ (i+1)^‑p for several exponents p.
   • Random jittered variants of the power‑law.

3. **Projected gradient descent with restarts.**
   • Uses a soft‑max approximation of the maximum convolution,
     Adam‑style momentum and an exponential schedule for the
     smoothing parameter β (1 → 2^16).  The iterate is always
     projected onto the simplex Σ a_i = √(2·n).

4. **Simulated annealing refinement.**
   • Mass‑transfer, local smoothing and index‑swap moves.
   • Temperature and allowed transfer fraction decay exponentially.

5. **Final short polish (projected gradient).**

All phases respect the global deadline.  The final sequence is trimmed
to the last non‑zero entry and returned as a plain list of floats.

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Return the benchmark value 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) of length n, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (rev(a) ⋆ p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Projected gradient descent (soft‑max) – the work‑horse optimizer
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style gradient descent on the soft‑max
    approximation of max(convolution).  β starts at 1 and grows
    exponentially to max_beta.  The iterate is always projected
    onto the simplex Σa = target_sum.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0

    while time.time() < deadline:
        # soft‑max gradient
        g, _ = _gradient_softmax(a, beta)

        # Adam update (un‑biased)
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # adaptive step size
        lr = step_base / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD‑Adam it={it:6d} β={beta:6.1f} score={best_val:.6f}")

        # increase β every few hundred steps
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Run several short gradient‑descent restarts with random jitter;
    keep the best result.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD‑restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 10.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.5,
                         max_frac_end: float = 0.01,
                         max_window: int = 100) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing (mass transfer / smoothing / swaps) to lower
    the maximum convolution value.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate current max‑convolution index
        k_max = int(np.argmax(conv))

        # contributions of each entry to this peak
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # -------- mass transfer ------------------------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # -------- smoothing ----------------------------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # -------- swap ---------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (math.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming phase (small scale) – optional
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """Construct the (2n‑1)×n matrix that maps g ↦ f*g."""
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for j in range(n):
        start = j
        end = j + n
        rows.extend(range(start, end))
        cols.extend([j] * n)
        data.extend([f[j]] * n)
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_for_g(f: np.ndarray, target_sum: float):
    """
    Linear program:
        minimise   M
        s.t.       f * g ≤ M   (all convolution entries)
                   Σ g_i = target_sum,   g_i ≥ 0,   M ≥ 0
    Returns (g, M) or (None, None) on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                     # (2n‑1)×n
    Mcol = -np.ones((2 * n - 1, 1))
    A_ub = sparse.hstack([A, Mcol], format='csr')
    c = np.zeros(n + 1)
    c[-1] = 1.0                                    # minimise M

    b_ub = np.zeros(2 * n - 1)

    # equality Σ g_i = target_sum
    A_eq = sparse.csr_matrix(np.hstack([np.ones(n), np.zeros(1)])[np.newaxis, :])
    b_eq = np.array([target_sum])

    bounds = [(0, None)] * n + [(0, None)]

    res = optimize.linprog(c,
                           A_ub=A_ub, b_ub=b_ub,
                           A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds,
                           method='highs')
    if not res.success:
        return None, None
    g = res.x[:n]
    M_val = res.x[n]
    return g, M_val


def _lp_phase(n_small: int, target_sum_small: float, deadline: float):
    """Iteratively improve a tiny shape by the LP described above."""
    f = _power_law_sequence(n_small, 0.55, target_sum_small)
    best_f = f.copy()
    best_score = _score(f)
    print(f"[{time.time():.2f}] LP start score={best_score:.6f}")
    while time.time() < deadline:
        g, M = _solve_lp_for_g(f, target_sum_small)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g = _project_to_simplex(g, target_sum_small)
        sc = _score(g)
        if sc < best_score:
            best_score = sc
            best_f = g.copy()
            print(f"[{time.time():.2f}] LP improved best score={best_score:.6f}")
        # blend for the next LP (helps convergence)
        f = 0.5 * f + 0.5 * g
    return best_f, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with a low benchmark score.
    Combines a tiny linear‑programming warm‑start, several initial
    candidates, projected gradient descent with restarts, simulated
    annealing and a final polish.
    """
    start = time.time()
    deadline = start + budget_s - 0.5                # safety margin

    # --------------------------------------------------------------
    # Problem size decision
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Phase 0 – tiny linear programme (if SciPy is present)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    if _SCIPY_AVAILABLE:
        n_small = min(600, max(200, n // 30))
        target_sum_small = math.sqrt(2.0 * n_small)
        lp_deadline = time.time() + min(120.0,
                                        (deadline - time.time()) * 0.15)
        try:
            f_small, sc_small = _lp_phase(n_small, target_sum_small, lp_deadline)
            # upscale
            cand = _resize_sequence(f_small, n)
            cand = _project_to_simplex(cand, target_sum)
            sc = _score(cand)
            best_seq = cand
            best_score = sc
            print(f"[{time.time():.2f}] LP up‑scaled initial score={sc:.6f}")
        except Exception as e:                     # pragma: no cover
            print("LP phase failed:", e)

    # --------------------------------------------------------------
    # Phase 1 – a small portfolio of deterministic starts
    # --------------------------------------------------------------
    # 1) use the previously discovered global best if available
    if "height_sequence_1" in globals():
        seq = _resize_sequence(np.array(globals()["height_sequence_1"],
                                       dtype=np.float64), n)
        seq = _project_to_simplex(seq, target_sum)
        sc = _score(seq)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.2f}] Global init improved score={sc:.6f}")

    # 2) power‑law shapes for several exponents
    for p in np.linspace(0.35, 0.70, 8):
        seq = _power_law_sequence(n, float(p), target_sum)
        sc = _score(seq)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.2f}] Power‑law p={p:.3f} improved score={sc:.6f}")

    # 3) random jittered variants of the 0.55 power‑law
    rng = np.random.default_rng()
    for _ in range(3):
        base = _power_law_sequence(n, 0.55, target_sum)
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq = np.maximum(base + jitter, 0.0)
        seq = _project_to_simplex(seq, target_sum)
        sc = _score(seq)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.2f}] Jittered init improved score={sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected gradient descent with restarts
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.55
    a_gd, sc_gd = _gradient_descent_with_restarts(best_seq,
                                                   target_sum,
                                                   gd_deadline)
    if sc_gd < best_score:
        best_seq, best_score = a_gd, sc_gd
        print(f"[{time.time():.2f}] GD phase improved best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_sa, sc_sa = _simulated_annealing(best_seq,
                                            deadline,
                                            start_temp=20.0,
                                            end_temp=1e-6,
                                            max_frac_start=0.5,
                                            max_frac_end=0.01,
                                            max_window=150)
        if sc_sa < best_score:
            best_seq, best_score = a_sa, sc_sa
            print(f"[{time.time():.2f}] SA phase improved best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – short final polish (if time allows)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, sc_fin = _annealed_gradient_descent_v2(best_seq,
                                                       deadline,
                                                       target_sum)
        if sc_fin < best_score:
            best_seq, best_score = a_fin, sc_fin
            print(f"[{time.time():.2f}] Final polish improved best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing zeros and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional check with the provided benchmark
    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:                               # pragma: no cover
        pass

    return final_seq
