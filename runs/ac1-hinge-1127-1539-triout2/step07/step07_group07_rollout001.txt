def max_convolution_fft(a):
    n=len(a)
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv.max()

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

The algorithm works in three stages:

1. **Broad parametric sweep** – a few families of decreasing sequences
   (power‑law, exponential, truncated linear) are generated for a relatively
   large length (default 20000).  Each candidate is normalised so that
   Σa = √(2 n) (the invariant for the scoring function).  The best few
   candidates are kept for refinement.

2. **Permutation + value transfer simulated annealing** – starting from each
   elite candidate we repeatedly apply either a random swap of two positions
   or a “transfer” of a random fraction of the mass from one position to
   another (the total ℓ¹‑norm stays constant).  The move is accepted with a
   Metropolis criterion and the temperature is slowly cooled.  This stage
   directly minimises the maximum entry of the linear convolution
   ``a*a`` (the true objective).

3. **Fine gradient refinement** – the best sequence from the SA stage is
   further optimised by projected Adam on a smooth log‑sum‑exp surrogate of
   the maximum convolution entry.  The surrogate temperature β is annealed
   up to 1e6, which drives the smooth objective to the true max.

All stages respect the global deadline (default 1000 s).  The final sequence
is trimmed of trailing near‑zero entries and returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simple families of decreasing sequences (already ℓ¹‑normalised)
# ----------------------------------------------------------------------


def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ¹‑norm = target."""
    scale = target / np.sum(v)
    return np.maximum(v * scale, 0.0)


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = (i + shift) ** (-p)
    return _scale_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = np.exp(-rate * (i + shift))
    return _scale_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * i)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _scale_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Simulated‑annealing on permutation + value‑transfer moves
# ----------------------------------------------------------------------


def _perm_transfer_sa(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated annealing that mixes:
      * swaps of two positions (preserves multiset)
      * transfer of a random fraction of mass from i to j (preserves sum)
    The objective is the max entry of the linear convolution.
    """
    a = a0.copy()
    n = a.size
    cur_val = _max_convolution_fft(a)
    best = a.copy()
    best_val = cur_val

    # temperature initialised as a fraction of the current score
    T = cur_val * 0.3 + 1e-12
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick move type
        if rng.random() < 0.65:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            new_val = _max_convolution_fft(a)
            dE = new_val - cur_val
            if dE <= 0 or rng.random() < np.exp(-dE / T):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = a.copy()
            else:
                # revert
                a[i], a[j] = a[j], a[i]
        else:                                        # transfer move
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 0.0:
                continue
            # random fraction up to 30 % of the donor mass
            frac = rng.random() * 0.30
            delta = a[i] * frac
            a[i] -= delta
            a[j] += delta
            new_val = _max_convolution_fft(a)
            dE = new_val - cur_val
            if dE <= 0 or rng.random() < np.exp(-dE / T):
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = a.copy()
            else:
                # revert the transfer
                a[i] += delta
                a[j] -= delta
                # tiny numerical drift may appear – re‑project if needed
                if a[i] < 0.0:
                    a[i] = 0.0
                if a[j] < 0.0:
                    a[j] = 0.0

        # anneal temperature
        T *= cooling
        # occasional temperature restart to keep the walk alive
        if T < 1e-8:
            T = best_val * 0.05 + 1e-12

    return best


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) surrogate and projected Adam optimiser
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ_k exp(β·b_k) where b = a*a (full linear conv).
    Returns (grad, full convolution).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)          # shape (2n‑1,)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1      # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation of the smoothed max‑convolution.
    Runs until the supplied Unix time deadline.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e6
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam biased moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate that keeps the ℓ¹‑norm stable
        lr = 0.5 * target_sum / ((beta + 1.0) * (np.sqrt(v_hat).mean() + 1e-8))

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the ℓ¹‑simplex Σa = √(2 n) , a ≥ 0
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal the surrogate temperature
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            pad = np.zeros(n - ext.size, dtype=float)
            ext = np.concatenate([ext, pad])
        elif ext.size > n:
            ext = ext[:n]
        ext = _scale_to_sum(ext, target_sum)
        _maybe_store(ext)

    # ------------------------------------------------------------------
    # 3. Refine elite candidates (SA + Adam)
    # ------------------------------------------------------------------
    elite.sort(key=lambda x: x[0])   # from best to worst

    for rank, (score0, seed_seq) in enumerate(elite):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # give 70 % of the remaining time to SA, 30 % to Adam
        sa_time = remaining * 0.70
        adam_time = remaining * 0.30

        # ---- simulated annealing on permutation / transfer moves ----
        refined = _perm_transfer_sa(seed_seq, time.time() + sa_time)

        # ---- fine‑grained Adam optimisation on the smoothed max ----
        refined = _adam_gradient_descent(refined, time.time() + adam_time)

        # evaluate final candidate
        cur_obj = _max_convolution_fft(refined)
        if cur_obj < best_obj:
            best_obj, best_seq = cur_obj, refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL best (refined #{rank}) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Final clean‑up: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (guaranteed feasible)
        best_seq = np.full(n, target_sum / n)
        best_obj = _max_convolution_fft(best_seq)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # sanity check – report the official evaluation value
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
