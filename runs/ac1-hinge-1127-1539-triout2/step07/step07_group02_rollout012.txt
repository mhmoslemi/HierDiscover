"""
Step‑function optimizer for the self‑convolution peak problem.

The code follows several stages:
1. Random analytic families (uniform, geometric, power‑law, beta‑like,
   sinusoidal, up‑sampled previous best) – cheap exploration.
2. Projected Adam on a soft‑max (log‑sum‑exp) approximation of the peak.
   Temperature (alpha) is raised up to 400 for a tight approximation.
3. Sub‑gradient descent on the true peak with occasional random swaps.
4. Active‑set linear‑programming refinement.
   Only the constraints that are nearly tight are kept, a small LP is
   solved (SciPy) to obtain a better direction, then a convex combination
   with the current iterate is attempted.
5. Simple hill‑climbing by swapping tiny masses.

The routine is time‑aware (budget_s seconds, 5 s safety margin) and
stores the best sequence globally as ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a * a (length = 2*n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
# w must be a probability vector (soft‑max weights).
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) probabilities over convolution entries.
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helpers (optional)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed peak (log‑sum‑exp)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the smooth max
    alpha = 0.1
    alpha_mul = 1.04
    max_alpha = 400.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        # occasional small jitter
        if rng.random() < 0.012:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max convolution entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.14
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient (valid sub‑gradient of max)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early exit if stuck
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Active‑set Linear‑Programming refinement
# ----------------------------------------------------------------------
def _active_set_lp_refine(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         enforce_sym: bool = False,
                         delta: float = 0.02,
                         max_iter: int = 15) -> tuple[np.ndarray, float]:
    """
    Keeps only constraints that are within (1‑delta) of the current peak,
    solves a small LP to maximise the total mass under those constraints,
    rescales the solution to the required total sum and takes a convex
    step towards it.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _full_convolution(a)
        M = conv.max()
        # indices where the bound is nearly active
        thresh = M * (1.0 - delta)
        active_idx = np.where(conv >= thresh)[0]
        # ensure we have a handful of constraints
        if active_idx.size < 5:
            active_idx = np.argsort(-conv)[:5]

        # build A_ub matrix (len(active_idx) x n)
        rows = []
        for k in active_idx:
            row = np.zeros(n)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            row[i_start:i_end + 1] = a[i_start:i_end + 1]
            rows.append(row)
        A = np.vstack(rows)
        b = np.full(A.shape[0], M)

        # maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)

        # solve the LP
        try:
            res = linprog(c,
                          A_ub=A,
                          b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={"presolve": True})
        except Exception:      # pragma: no cover
            break

        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # rescale so that the simplex sum equals S
        g_scaled = g * (S / sum_g)

        # try a convex combination, start with t=0.5 and halve if needed
        t = 0.5
        improved = False
        while t > 1e-4:
            cand = (1.0 - t) * a + t * g_scaled
            if enforce_sym:
                cand = _symmetrize(cand)
            cand = _project_simplex(cand, S)
            cand_score = np.max(_full_convolution(cand))
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                improved = True
                break
            t *= 0.5

        if not improved:
            # no improvement – stop early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the outer runner
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load previous best if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:  # pragma: no cover
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap analytic families
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.25 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1

        # length choice (mix of inherited and fresh)
        if base_seq is not None and rng.random() < 0.55:
            scale = rng.uniform(0.9, 1.1)
            n = max(800, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(1500, 15001))

        # pick a family
        fam = rng.integers(0, 6)
        if fam == 0 and base_seq is not None:
            a = _upsample_seq(base_seq, n)
        elif fam == 1:
            a = rng.random(n)                                 # uniform random
        elif fam == 2:
            rate = rng.uniform(0.94, 0.9999)                   # symmetric geometric
            left = rate ** np.arange(n)
            a = (left + left[::-1]) / 2.0
        elif fam == 3:
            p = rng.uniform(0.2, 4.5)                         # power‑law symmetric
            idx = np.arange(n, dtype=float)
            a = ((idx + 1) * (n - idx)) ** p
        elif fam == 4:
            p1 = rng.uniform(0.05, 5.5)                       # beta‑like
            p2 = rng.uniform(0.05, 5.5)
            x = np.linspace(0, 1, n)
            a = (x ** p1) * ((1 - x) ** p2)
        else:
            p = rng.uniform(0.4, 4.5)                         # sin‑power
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p

        a = np.clip(a, 0.0, None)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S

        # occasional symmetrisation
        if rng.random() < 0.45:
            a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on soft‑max of the peak
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        n = 5000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_overall_seq = a
        best_overall_score = np.max(_full_convolution(a))

    now = time.time()
    if now < deadline:
        t_adam = now + 0.55 * (deadline - now)
        a_adam, sc_adam = _optimize_adam(best_overall_seq, t_adam, rng,
                                         enforce_sym=False)
        if sc_adam < best_overall_score:
            best_overall_score = sc_adam
            best_overall_seq = a_adam.copy()
            print(f"[adam] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent with random swaps
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        t_sub = now + 0.30 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad(best_overall_seq, t_sub, rng,
                                          enforce_sym=False)
        if sc_sub < best_overall_score:
            best_overall_score = sc_sub
            best_overall_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Active‑set LP refinement (few iterations)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        t_lp = now + 0.10 * (deadline - now)
        a_lp, sc_lp = _active_set_lp_refine(best_overall_seq, t_lp, rng,
                                            enforce_sym=False,
                                            delta=0.018,
                                            max_iter=12)
        if sc_lp < best_overall_score:
            best_overall_score = sc_lp
            best_overall_seq = a_lp.copy()
            print(f"[lp] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final local swap hill‑climb
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_loc, sc_loc = _local_swap_search(best_overall_seq, deadline, rng,
                                           enforce_sym=False)
        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return result and store globally
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
