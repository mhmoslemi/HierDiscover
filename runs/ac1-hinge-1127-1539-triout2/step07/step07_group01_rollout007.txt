def _projected_subgradient_optimize(a0, deadline):
    n = len(a0); S = np.sqrt(2*n)
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution(a)
    t = 0
    while time.time() < deadline:
        t += 1
        b = _full_convolution_fft(a)
        max_b = b.max()
        # gradient
        mask = (b == max_b).astype(float)
        if mask.sum() > 1:
            mask = mask / mask.sum()
        # compute gradient
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        grad = 2.0 * conv[n-1:n-1+n]  # shape n

        # step size schedule
        step = 1.0 / np.sqrt(t) * 1.0  # maybe constant factor S? We'll tune.

        # project onto simplex of sum S and nonnegativity
        a = _project_simplex(a_new, S)

        # keep best
        if max_b < best_max:
            best_max = max_b
            best_a = a.copy()
    return best_a, best_max

Now we need to implement _project_simplex.

Standard algorithm:

```
def _project_simplex(v, s):
    # v: array, we want to find w >=0, sum w = s, minimizing ||w - v||^2
    # similar to algorithm by Duchi et al.
    n = len(v)
    if v.sum() == s and np.all(v >= 0): return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Performance O(n log n).

"""
Step‑function optimizer (max‑convolution minimisation).

The algorithm consists of three stages:

1. **Random exploration** – many candidate vectors are drawn from a
   collection of parametric families (uniform, Dirichlet, power‑law,
   exponential, linear, decreasing, sparse‑Gamma).  Each candidate is
   normalised to ℓ₁‑norm √(2 n) and its max‑convolution value is measured
   with an FFT.  The best few candidates are kept.

2. **Projected sub‑gradient refinement** – for each stored candidate the
   vector is (optionally) interpolated to a larger length and then
   improved by a projected sub‑gradient method on the simplex.
   At every iteration we compute the exact sub‑gradient of the
   max‑convolution (the gradient of the active convolution entry),
   take a diminishing step and project back onto the ℓ₁‑simplex.
   Multiple restarts and jitter are used to escape plateaus.

3. **Final evaluation** – the best vector found is converted to a plain
   Python list, stored in the global variable ``height_sequence_1`` and
   returned.  The external black‑box ``evaluate_sequence`` is called once
   at the end for a sanity check.

The whole routine respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of ``a``."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}.
    Returns a new array with the same shape as ``v``.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact sub‑gradient for max‑convolution
# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray):
    """
    Returns (g, max_b) where ``g`` is a sub‑gradient of
    f(a)=max_k (a*a)_k and ``max_b`` is the current peak value.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    mask = (b == max_b).astype(float)
    cnt = mask.sum()
    if cnt > 1:
        mask = mask / cnt            # average over all active indices

    a_rev = a[::-1]
    conv_len = mask.shape[0] + n - 1
    f_len = _next_pow2(conv_len)

    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]

    g = 2.0 * conv[n - 1 : n - 1 + n]    # slice to length ``n``
    return g, max_b


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise ``max(a*a)`` over the simplex using projected sub‑gradient descent.
    ``a0`` must already satisfy the ℓ₁‑constraint Σa = √(2 n).
    Returns the best vector found and its hard peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # gradient of the current max‑convolution
        g, peak = _exact_gradient(a)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # diminishing step size (tuned empirically)
        step = (S * 0.5) / np.sqrt(it)

        # sub‑gradient step
        a_tmp = a - step * g

        # project back onto the ℓ₁‑simplex
        a = _project_simplex(a_tmp, S)

        # occasional jitter to avoid stagnation
        if it % 800 == 0:
            a += rng.normal(scale=0.01 * S, size=n)
            a = _project_simplex(a, S)

    # final verification of the best peak
    best_peak = _max_convolution(best_a)
    return best_a, best_peak

# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(5e-4, 2e-2)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    return mix * lin + (1.0 - mix) * base


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_decreasing(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotonically decreasing exponential."""
    i = np.arange(n)
    lam = rng.uniform(0.001, 0.01)
    return np.exp(-lam * i)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length ``n``.
    With probability ~0.2 we warm‑start from the current global best,
    otherwise we sample from a random parametric family.
    The vector is normalised to ℓ₁‑norm √(2 n) and made non‑negative.
    """
    S = np.sqrt(2.0 * n)

    # Warm start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.full(n, 1.0)
        a = a * (S / a.sum())
        return a

    style = rng.choice([
        'powerlaw', 'exponential', 'linear',
        'gamma', 'uniform', 'dirichlet', 'decreasing'
    ])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:   # decreasing
        raw = _init_decreasing(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises

    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # tiny safety margin

    # Load a warm‑start if one already exists
    # ------------------------------------------------------------------
    base_seq = None

        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 60 % of the budget)

            best_n = n

        # maintain a short list of good candidates

        top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient refinement

    if best_seq is None:

        best_n = n
        best_score = _max_convolution(best_seq)

    # ensure the current best is also among the candidates
    top_candidates.append((best_score, best_seq.copy(), best_n))
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:8]

    # iterate over the (few) best seeds, scaling them to larger lengths
    for rank, (score, seq, n) in enumerate(top_candidates, start=1):
        if time.time() >= end_time - 30.0:
