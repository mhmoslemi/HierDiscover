def _project_simplex(v, S):
    # ensure vector v shape (n,)
    # algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v))+1
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Improved step‑function optimizer.

The optimizer consists of three stages:

1. **Broad random exploration** – many candidates are generated from a
   richer set of parametric families (including asymmetric linear / exponential
   shapes).  Each candidate is normalised so that its ℓ₁‑norm equals √(2 n) and
   its convolution peak is measured with an FFT.  The best candidate seen so
   far is kept globally.

2. **Refinement** – the current best vector is interpolated to several
   larger lengths, then each scaled version is optimised with two complementary
   methods:
        * Adam on the simplex (log‑softmax parametrisation) using a τ‑soft‑max
          approximation of the max‑convolution.
        * A plain sub‑gradient descent that works with the exact max
          (no smoothing).  Both runs respect the remaining time budget.

3. **Local perturbation** – a cheap “mass‑transfer’’ hill‑climber makes small
   random moves (transfer ε·a_i from one index to another) and accepts any
   move that reduces the peak.  This often squeezes a little more out of the
   solution found by the gradient phase.

Finally the candidate is evaluated with the provided ``evaluate_sequence``.
All operations obey the 1000 s time budget; if the budget runs out the last
best vector is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x)=s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("sum constraint must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Returns the best vector found and its *hard* peak (max convolution entry).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits (log‑softmax)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.35                     # a bit smaller than the reference

    # initialise temperature (soft‑max smoothing)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)       # start with a modest amount of smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (soft max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules (slower decay) -----
        tau *= 0.9999                # keep a little smoothing for longer
        lr *= 0.9999                 # gentle decay

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the *exact* max‑convolution.
    Works on the simplex by projecting after each step.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # Ensure feasibility
    a = np.maximum(a0, 0.0)
    a = a / a.sum() * S

    best_a = a.copy()
    best_peak = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        b = _full_convolution_fft(a)
        max_b = b.max()
        # pick one active index (the first maximal index)
        k = int(np.argmax(b))

        # sub‑gradient of max_b w.r.t a (2 * shifted copy)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # simple diminishing step size
        step = 0.4 / (np.sqrt(it) + 1.0)

        a_tmp = a - step * grad
        a = _project_simplex(a_tmp, S)

        # keep best
        peak = _max_convolution(a)
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # occasional random jitter to avoid stagnation
        if it % 800 == 0:
            noise = rng.normal(scale=0.02 * S, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum() * S

    return best_a, best_peak

# Parametric families for random initialisation

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric power‑law heavy on the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 4.0)          # exponent
    offset = rng.uniform(0.0, 20.0)    # shift of denominator
    mix = rng.uniform(0.0, 1.0)        # uniform background amount
    edge = (d + offset) ** (-p)
    raw = mix * edge + (1.0 - mix) * np.ones(n)
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.03)   # decay rate
    mix = rng.uniform(0.0, 1.0)
    raw = mix * np.exp(-lam * d) + (1.0 - mix) * np.ones(n)
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric triangular shape."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 6.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * lin + (1.0 - mix) * np.ones(n) * rng.uniform(0.0, 0.2)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Raw Dirichlet‑like start."""
    return rng.random(n)


def _init_asymmetric_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing from the left edge only (no symmetry)."""
    H = rng.uniform(0.5, 12.0)
    slope = rng.uniform(0.0, H / (n - 1))
    raw = np.maximum(0.0, H - slope * np.arange(n))
    return raw


def _init_asymmetric_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Different exponential decay rates on the two edges."""
    idx = np.arange(n)
    d_left = idx
    d_right = n - 1 - idx
    lam_left = rng.uniform(0.0005, 0.03)
    lam_right = rng.uniform(0.0005, 0.03)
    exp_part = np.where(idx <= n // 2,
                        np.exp(-lam_left * d_left),
                        np.exp(-lam_right * d_right))
    mix = rng.uniform(0.0, 1.0)
    raw = mix * exp_part + (1.0 - mix) * np.ones(n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best and otherwise samples
    from a random parametric family (more families than before).
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ~0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'asym_linear', 'asym_exponential'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'asym_linear':
        raw = _init_asymmetric_linear(n, rng)
    else:  # asym_exponential
        raw = _init_asymmetric_exponential(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a
