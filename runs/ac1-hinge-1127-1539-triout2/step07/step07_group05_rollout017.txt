def _subgradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0*n)
    t = 0
    # step schedule: start with step size proportional to target.
    step_factor = 0.1
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution and max idx
        # we can compute conv via FFT for speed
        size = 1 << ((2*n-1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft*a_fft, n=size)[:2*n-1]
        max_val = b.max()
        k = int(b.argmax())
        # compute gradient
        g = np.zeros_like(a)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # vectorized assignment:
        idxs = np.arange(i_start, i_end+1)
        g[idxs] = 2 * a[k - idxs]
        # step size
        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            break
        # step size maybe step_factor * target / norm_g
        lr = step_factor * target / norm_g
        a -= lr * g
        # projection
        a = _project_to_simplex(a, target)
        # optionally reduce step_factor gradually
        t += 1
        if t % 200 == 0:
            step_factor *= 0.95
    return a

def _subgradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0*n)
    lr_initial = 0.5 * target  # maybe large
    t = 0
    while time.time() < deadline:
        # compute convolution
        size = 1 << ((2*n-1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft*a_fft, n=size)[:2*n-1]
        kmax = np.argmax(b)
        # compute gradient subgradient
        g = np.zeros_like(a)
        i_start = max(0, kmax-(n-1))
        i_end = min(kmax, n-1)
        idx = np.arange(i_start, i_end+1)
        g[idx] = 2.0 * a[kmax - idx]
        # step size schedule
        t += 1
        lr = lr_initial / np.sqrt(t) / (np.linalg.norm(g)+1e-12)
        a -= lr * g
        # projection onto simplex
        a = _project_to_simplex(a, target)
        # optionally jitter at intervals
        if t % 200 == 0:
            # small random perturb
            jitter = np.random.normal(scale=target*1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
    return a

"""
Heuristic optimiser for the step‑function problem.

The optimiser proceeds in three stages:

1. **Random sampling** – generate many candidates from a variety of
   parametric families (exponential, power‑law, linear, quadratic,
   Dirichlet‑like).  Each candidate is projected onto the ℓ¹‑simplex
   Σx = √(2 n) (the normalisation used by the evaluator).

2. **Two‑phase projected descent** – every candidate is first refined
   using the smooth log‑sum‑exp surrogate (the same routine used in the
   reference implementation) and then a fast exact sub‑gradient step
   for the true max‑convolution objective.  Both phases keep the vector
   on the simplex by projection.

3. **Permutation annealing** – finally the ordering of the multiset is
   improved by a simulated‑annealing swap routine.

The code respects a user‑supplied time budget (default 1000 s).  The
best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# (Duchi et al., 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 8e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta (make surrogate sharper)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on f(a)=max_k (a*a)_k."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    t = 0
    # initial learning‑rate factor (will decay as 1/√t)
    lr_factor = 0.7 * target_sum

    while time.time() < deadline:
        # ----- compute convolution and index of the maximal entry -----
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        k = int(b.argmax())

        # ----- exact sub‑gradient for the selected lag k -----
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]

        # ----- decaying step size  (1/√t) -----
        t += 1
        norm = np.linalg.norm(g)
        if norm == 0.0:
            break
        lr = lr_factor / (np.sqrt(t) * norm)
        a -= lr * g

        # ----- projection onto the ℓ¹‑simplex -----
        a = _project_to_simplex(a, target_sum)

        # ----- occasional jitter to avoid stagnation -----
        if t % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    The ℓ¹‑norm stays unchanged, only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current best
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny reshuffle of a random subset
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Families of parametric sequences (all normalised to Σ = √(2 n))
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ¹‑sum equal to target."""
    tot = np.sum(v)
    if tot == 0.0:
        v = np.full_like(v, target / v.size)
    else:
        v = v * (target / tot)
    v = np.maximum(v, 0.0)
    return v


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _dirichlet_sequence(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    raw = rng.gamma(shape=alpha, scale=1.0, size=n)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exp_grid_sequence(n: int, rng: np.random.Generator, trials: int = 80) -> np.ndarray:
    """Search a small grid of exponential decay factors and return the best."""
    target = np.sqrt(2.0 * n)
    best_seq = None
    best_obj = np.inf
    # sample decay factors r = exp(-c / n) where c ∈ [0.5, 12]
    cs = np.linspace(0.5, 12.0, trials)
    for c in cs:
        r = np.exp(-c / n)
        raw = r ** np.arange(n, dtype=float)
        seq = raw * (target / raw.sum())
        obj = _max_convolution_fft(seq)
        if obj < best_obj:
            best_obj = obj
            best_seq = seq
    return best_seq


# ----------------------------------------------------------------------
# Utility to update the global best / elite list
# ----------------------------------------------------------------------
def _update_best(seq: np.ndarray,
                 best_seq: np.ndarray | None,
                 best_obj: float,
                 elite: list[tuple[float, np.ndarray]],
                 ) -> tuple[np.ndarray | None, float, list[tuple[float, np.ndarray]]]:
    obj = _max_convolution_fft(seq)
    if obj < best_obj:
        best_obj = obj
        best_seq = seq.copy()
        print(f"[{time.time():.1f}] New global best obj={best_obj:.6f} len={seq.size}")
    # maintain a small elite pool (sorted by objective)
    if len(elite) < 8 or obj < elite[-1][0]:
        elite.append((obj, seq.copy()))
        elite.sort(key=lambda x: x[0])
        elite = elite[:8]
    return best_seq, best_obj, elite


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective.  The routine respects the allotted
    time budget, explores several parametric families, refines promising
    candidates with a two‑phase projected descent (smooth + exact
    sub‑gradient) and finishes with a permutation‑annealing phase.
    The returned list is ready to be fed to `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None
    best_obj = np.inf
    elite: list[tuple[float, np.ndarray]] = []

    # --------------------------------------------------------------
    # 0. warm‑start from a previously discovered sequence if available
    # --------------------------------------------------------------
    init = globals().get("height_sequence_1", None)
    if init is not None:
        init_arr = np.asarray(init, dtype=float)
        n0 = init_arr.size
        init_arr = _normalize_to_sum(init_arr,
                                     np.sqrt(2.0 * n0))
        best_seq, best_obj, elite = _update_best(init_arr,
                                                best_seq,
                                                best_obj,
                                                elite)

    # --------------------------------------------------------------
    # 1. Random generation & quick two‑phase descent
    # --------------------------------------------------------------
    lengths = [3000, 4000, 5000, 6000, 8000, 10000, 12000, 15000]
    families = ['exp', 'power', 'linear', 'quadratic', 'dirichlet', 'exp_grid']

    # allocate ~70 % of the time budget for this exploratory phase
    gen_deadline = time.time() + 0.70 * (deadline - time.time())
    while time.time() < gen_deadline:
        n = int(rng.choice(lengths))
        target_sum = np.sqrt(2.0 * n)

        family = rng.choice(families)
        if family == 'exp':
            rate = rng.uniform(0.0002, 0.02)
            shift = rng.uniform(0.0, 5.0)
            base = _exponential_sequence(n, rate, shift)

        elif family == 'power':
            p = rng.uniform(0.30, 1.20)
            shift = rng.uniform(0.0, 5.0)
            base = _powerlaw_sequence(n, p, shift)

        elif family == 'linear':
            slope = rng.uniform(1e-5, 0.015)
            intercept = rng.uniform(0.5, 5.0)
            base = _linear_sequence(n, slope, intercept)

        elif family == 'quadratic':
            a0 = rng.uniform(0.5, 5.0)
            b = rng.uniform(0.0, 0.015)
            c = rng.uniform(0.0, 0.0001)
            base = _quadratic_sequence(n, a0, b, c)

        elif family == 'dirichlet':
            alpha = rng.uniform(0.2, 5.0)
            base = _dirichlet_sequence(n, alpha, rng)

        else:   # 'exp_grid' – a tiny grid search for the exponential decay
            base = _exp_grid_sequence(n, rng, trials=60)

        # ----- try a few deterministic orderings -----
        asc = np.sort(base)
        desc = asc[::-1]
        for cand in (desc, asc):
            best_seq, best_obj, elite = _update_best(cand,
                                                     best_seq,
                                                     best_obj,
                                                     elite)

        # ----- short refinement of the current best -----
        if best_seq is not None and time.time() < gen_deadline:
            # give ~6 s for a quick two‑phase optimisation of the current best
            sub_dead = min(gen_deadline,
                           time.time() + 6.0)
            refined = _gradient_descent_opt(best_seq, sub_dead)
            refined = _subgradient_descent_opt(refined, sub_dead)
            best_seq, best_obj, elite = _update_best(refined,
                                                     best_seq,
                                                     best_obj,
                                                     elite)

    # --------------------------------------------------------------
    # 2. Deeper refinement of elite candidates
    # --------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        # allocate ~15 s per elite (adjust if time is short)
        sub_dead = min(deadline, time.time() + 15.0)

        refined = _gradient_descent_opt(seed_seq, sub_dead)
        refined = _subgradient_descent_opt(refined, sub_dead)
        refined = _permutation_annealing(refined, deadline)

        best_seq, best_obj, elite = _update_best(refined,
                                                 best_seq,
                                                 best_obj,
                                                 elite)

    # --------------------------------------------------------------
    # 3. Final global permutation annealing (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _permutation_annealing(best_seq, deadline)
        best_obj = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # 4. Safety fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional verification (does not affect the return value)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
