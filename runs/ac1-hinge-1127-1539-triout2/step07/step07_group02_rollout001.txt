def _local_annealed_search(a0, deadline, rng, enforce_sym=False):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    cur_score = best_score
    start = time.time()
    while time.time() < deadline:
        # compute temperature based on progress
        progress = (time.time() - start) / (deadline - start)
        T = (1.0 - progress) * cur_score * 0.1  # start high, decay to near zero
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.05 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # accept with probability exp(-(cand_score - cur_score) / T)
            if T > 0 and rng.random() < np.exp(-(cand_score - cur_score) / T):
                accept = True
        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score

"""
Propose a non‑negative step function that minimises the self‑convolution peak.

Strategy
--------
1. **Broad random sampling** – generate many candidate shapes (uniform,
   sine‑power, beta‑like, power‑law, Gaussian, raised‑cosine) with a range of
   lengths (5k … 20k).  All candidates are normalised to Σa = sqrt(2·n) and
   optionally symmetrised.  The best candidate is kept.

2. **Multi‑start projected Adam** – starting from the current best (or a small
   perturbation of it) run a few short Adam optimisations that minimise a
   soft‑max approximation of the convolution maximum.  The Adam updates are
   projected onto the simplex {a_i ≥ 0, Σa = sqrt(2·n)} and symmetry is
   enforced at each step.  The smoothness parameter α is increased during the
   run so that the approximation becomes tighter.

3. **Simulated‑annealing hill‑climb** – with the best vector from step 2,
   perform many mass‑transfer moves (i → j) accepting improving moves and
   occasionally accepting worsening moves with a temperature that decays to
   zero.  After each move the vector is projected back onto the simplex and,
   if requested, symmetrised.

4. **Final greedy local swap** – a short pure hill‑climbing phase that only
   accepts improvements, to polish the result.

The whole routine respects a 1000 s budget (5 s safety margin) and returns the
best sequence it found.  The final sequence is stored globally as
``height_sequence_1`` for reuse in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights) of length 2*n‑1.
    """
    n = a.shape[0]
    # reverse the weights because convolution is cross‑correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # extract the n relevant entries (see derivation in the original code)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns the best vector encountered and its true (unsmoothed) peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.5
    alpha_mul = 1.03
    max_alpha = 500.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.08 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
                if enforce_sym:
                    a = _symmetrize(a)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search
# ----------------------------------------------------------------------
def _local_annealed_search(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator,
                           enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves with a decaying temperature.
    Accepts worsening moves probabilistically early on.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    cur_score = best_score

    start = time.time()
    while time.time() < deadline:
        # linear temperature schedule (relative progress)
        progress = (time.time() - start) / (deadline - start)
        T = (1.0 - progress) * cur_score * 0.12  # start fairly hot, end near zero

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        delta = 0.05 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_score = np.max(_full_convolution(cand))

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(cand_score - cur_score) / T):
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climbing (tiny swaps)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing that swaps small amounts of mass between random
    positions, keeping only improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score
