def _adam_optimize_symmetric(a0_full, deadline, rng):
    n = a0_full.shape[0]
    m = (n + 1)//2
    S = np.sqrt(2.0 * n)
    # extract half
    a0_half = a0_full[:m]
    # ensure at least epsilon
    eps = 1e-12
    soft = np.maximum(a0_half / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)
    # Adam states
    m_state = np.zeros_like(p)
    v_state = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4
    # precompute indices for aggregating
    full_idx = np.arange(n)
    half_idx = np.minimum(full_idx, n - 1 - full_idx)
    # best
    best_half = a0_half.copy()
    best_full = a0_full.copy()
    best_peak = _max_convolution(best_full)
    it = 0
    while time.time() < deadline:
        it += 1
        # logits -> softmax -> half a
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a_half = soft * S
        # full a symmetrical
        a_full = _symmetrize_half(a_half, n)
        # compute gradient and max via subgradient
        grad_full, peak = _subgrad_max_convolution(a_full)   # we need function
        # store best
        if peak < best_peak:
            best_peak = peak
            best_half = a_half.copy()
            best_full = a_full.copy()
        # aggregate gradient to half
        g_half = np.zeros(m, dtype=grad_full.dtype)
        np.add.at(g_half, half_idx, grad_full)
        # mu = np.dot(a_full, grad_full) / S
        mu = np.dot(a_full, grad_full) / S
        # gradient wrt logits
        grad_p = a_half * (g_half - mu)
        # optional clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)
        # Adam update
        m_state = beta1 * m_state + (1 - beta1) * grad_p
        v_state = beta2 * v_state + (1 - beta2) * (grad_p ** 2)
        m_hat = m_state / (1 - beta1 ** it)
        v_hat = v_state / (1 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # schedule
        lr *= 0.9998
        # jitter occasionally
        if it % 800 == 0:
            p += rng.normal(scale=0.01, size=m)
    return best_full, best_peak

"""Step‑function optimizer (symmetric, multi‑scale, Adam on the simplex).

The algorithm works in three phases:

1. **Random exploration** – generate many symmetric candidates from a few
   parametric families (power‑law, exponential, linear, Gamma, uniform, Dirichlet),
   normalise their ℓ¹‑norm to √(2 n) and keep the one with the smallest
   self‑convolution peak.

2. **Multi‑scale refinement** – take the current best sequence, linearly
   interpolate it to several larger lengths and run an Adam optimiser on the
   simplex.  The optimiser works on the *half* of the vector (the sequence is
   forced to be symmetric) which halves the dimensionality and speeds up the
   gradient computation.  The exact sub‑gradient of the max‑convolution is used
   (no temperature smoothing).

3. **Final evaluation** – evaluate the best sequence with the provided
   ``evaluate_sequence`` function, store it in the global variable
   ``height_sequence_1`` and return it.

The whole routine respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Peak of the self‑convolution."""
    return _full_convolution_fft(a).max()


def _subgrad_max_convolution(a: np.ndarray):
    """
    Returns (grad, max_b) where grad is a sub‑gradient of
    max_k (a*a)[k] with respect to a (exact, no smoothing).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)          # length 2n‑1
    max_b = b.max()
    # mask of active indices (tolerance to avoid numerical issues)
    mask = (b >= max_b - 1e-12).astype(float)

    # gradient = 2 * conv(mask, a_rev) restricted to length n
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad, max_b


# ----------------------------------------------------------------------
# Sequence utilities
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation while preserving shape."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _symmetrize_half(half: np.ndarray, n: int) -> np.ndarray:
    """From the left half (including centre if n odd) build the full symmetric vector."""
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, half[-2::-1]])


# ----------------------------------------------------------------------
# Parametric families defined on the *half* vector
# ----------------------------------------------------------------------
def _init_powerlaw_half(m: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(m)
    d = idx + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(m)
    return mix * edge + (1.0 - mix) * base


def _init_exponential_half(m: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(m)
    d = idx + 1.0
    lam = rng.uniform(0.0005, 0.02)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(m)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear_half(m: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(m)
    d = idx.astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (m + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base_val = rng.uniform(0.0, 0.2)
    base = np.full(m, base_val)
    return mix * lin + (1.0 - mix) * base


def _init_gamma_half(m: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=m)


def _init_uniform_half(m: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(m, 1.0)


def _init_dirichlet_half(m: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(m)


# ----------------------------------------------------------------------
# Symmetric initialisation
# ----------------------------------------------------------------------
def _initial_vector_symmetric(n: int,
                             rng: np.random.Generator,
                             warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible symmetric vector of length n with ℓ¹‑norm √(2 n).
    With probability ~0.2 a warm start (scaled previous best) is used,
    otherwise a random parametric family is sampled.
    """
    S = np.sqrt(2.0 * n)

    # ---- warm start ----------------------------------------------------
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0:
            a = a * (S / total)
        else:
            a = np.full(n, S / n)
        return a

    # ---- random families -----------------------------------------------
    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet'])
    m = (n + 1) // 2  # length of the half vector
    if style == 'powerlaw':
        raw_half = _init_powerlaw_half(m, rng)
    elif style == 'exponential':
        raw_half = _init_exponential_half(m, rng)
    elif style == 'linear':
        raw_half = _init_linear_half(m, rng)
    elif style == 'gamma':
        raw_half = _init_gamma_half(m, rng)
    elif style == 'uniform':
        raw_half = _init_uniform_half(m, rng)
    else:
        raw_half = _init_dirichlet_half(m, rng)

    full = _symmetrize_half(raw_half, n)
    total = full.sum()
    if total == 0.0:
        full = np.full(n, 1.0)
        total = n
    full = full * (S / total)
    return full


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (working on the half vector)
# ----------------------------------------------------------------------
def _adam_optimize_symmetric(a0_full: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for a symmetric sequence.
    Returns the best full sequence seen and its hard max‑convolution peak.
    """
    n = a0_full.shape[0]
    m = (n + 1) // 2
    S = np.sqrt(2.0 * n)

    # Initialise logits from a feasible half‑vector
    a0_half = a0_full[:m]
    eps = 1e-12
    soft = np.maximum(a0_half / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m_state = np.zeros_like(p)
    v_state = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4

    # Mapping from full indices to half indices (for gradient aggregation)
    full_idx = np.arange(n)
    half_idx = np.minimum(full_idx, n - 1 - full_idx)

    # Best solution tracking
    best_half = a0_half.copy()
    best_full = a0_full.copy()
    best_peak = _max_convolution(best_full)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- recover half‑vector from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a_half = soft * S

        # ----- build the full symmetric vector -----
        a_full = _symmetrize_half(a_half, n)

        # ----- gradient of the hard max (exact sub‑gradient) -----
        grad_full, peak = _subgrad_max_convolution(a_full)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_half = a_half.copy()
            best_full = a_full.copy()

        # ----- aggregate gradient onto the half vector -----
        g_half = np.zeros(m, dtype=grad_full.dtype)
        np.add.at(g_half, half_idx, grad_full)

        # ----- chain rule back to logits (log‑softmax) -----
        mu = np.dot(a_full, grad_full) / S      # ⟨a,∂b/∂a⟩ / S
        grad_p = a_half * (g_half - mu)          # gradient w.r.t. logits

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m_state = beta1 * m_state + (1.0 - beta1) * grad_p
        v_state = beta2 * v_state + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m_state / (1.0 - beta1 ** it)
        v_hat = v_state / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        lr *= 0.9998

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.01, size=m)

    return best_full, best_peak


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the time budget and stores the final construction in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    safety = 5.0
    end_time = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # Warm‑start from an existing global construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 20 % of the total budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + (budget_s * 0.20) - safety
    while time.time() < explore_deadline:
        attempt += 1
        # pick a length – we favour moderate sizes but also try very large ones
        if rng.random() < 0.6:
            n = int(rng.integers(800, 3000))
        else:
            n = int(rng.integers(3000, 20000))

        cand = _initial_vector_symmetric(n, rng, base_seq)
        peak = _max_convolution(cand)

        if peak < best_score:
            best_score = peak
            best_seq = cand.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback if random phase failed to produce anything
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)
        print("[fallback] using uniform start")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale refinement (≈ 75 % of the remaining budget)
    # ------------------------------------------------------------------
    # Build a list of target lengths, slowly increasing up to ~20 k
    lengths = [best_n]
    cur = best_n
    while cur < 18000:
        cur = int(cur * 1.7)          # geometric growth
        cur = min(cur, 20000)
        lengths.append(cur)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        if time.time() >= end_time - 10.0:
            break
        remaining = end_time - time.time()
        # allocate a slice (roughly 30 % of what remains, but at least 30 s)
        slice_time = max(30.0, remaining * 0.30)

        # interpolate current best to the new length and re‑normalise
        scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        scaled = np.clip(scaled, 0.0, None)
        total = scaled.sum()
        if total > 0.0:
            scaled = scaled * (S / total)
        else:
            scaled = np.full(target_n, S / target_n)

        # perform a few independent Adam runs (restarts) on this length
        restarts = 3
        time_per_restart = slice_time / restarts
        for r in range(restarts):
            if time.time() >= end_time - 5.0:
                break
            # tiny random jitter to diversify restarts
            jitter = rng.normal(scale=0.005, size=target_n)
            a0 = np.maximum(scaled + jitter, 0.0)
            a0_sum = a0.sum()
            if a0_sum > 0.0:
                a0 = a0 * (S / a0_sum)
            else:
                a0 = np.full(target_n, S / target_n)

            a_opt, peak = _adam_optimize_symmetric(a0,
                                                   time.time() + time_per_restart,
                                                   rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_opt.copy()
                best_n = target_n
                print(f"[refine] n={target_n:5d} restart={r} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check via the external evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
