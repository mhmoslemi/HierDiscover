    a = a0.copy()
    n = a.size
    target_sum = sqrt(2*n)
    # parameters
    step = target_sum * 0.5 # initial step maybe
    t = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        S = np.flatnonzero(b == max_b)
        # Build s vector of length 2n-1 with 1/|S| weighting
        s = np.zeros(2*n-1, dtype=float)
        s[S] = 1.0/len(S)   # weighting average to avoid blow up
        # compute gradient
        # s_rev
        s_rev = s[::-1]
        g = np.convolve(a, s_rev)[(2*n-2):(2*n-2+n)]
        g = 2.0 * g
        # step size schedule
        lr = step / np.sqrt(t+1)
        a -= lr * g
        # project to simplex
        a = _project_to_simplex(a, target_sum)
        t+=1
        # optionally stop early if no improvement
        if t%1000==0:
            # compute new max
            new_max = _max_convolution_fft(a)

"""
Propose a candidate step‑function that (approximately) minimises the
``evaluate_sequence`` objective, i.e. the maximal entry of the self‑convolution
when the ℓ¹‑norm is fixed to √(2 n).

Algorithm
---------

1.  If a previous construction ``height_sequence_1`` is present, use it as a
    seed.  Otherwise start from a random simplex draw.
2.  Run a multi‑resolution schedule.  For each length *n* we:
    *  up‑sample the current best sequence (or initialise randomly);
    *  perform **sub‑gradient descent** on the true max‑convolution
       (the sub‑gradient is 2·∑_{k∈S} a shifted, where *S* are the indices
       attaining the maximal convolution entry);
    *  optionally refine with the smooth‑max (soft‑max) gradient descent
       used in the previous baseline;
    *  finish the stage with a short hill‑climb that updates the full
       convolution in O(n) per move.
3.  After the schedule, spend all remaining time on a final sub‑gradient
    descent + hill‑climb.
4.  Trim trailing near‑zeros and return the best sequence as a plain list.

All heavy work uses FFT‑based convolutions (O(n log n)).  The time budget
(≈1000 s) is honoured by dynamic deadline checks.  Progress and intermediate
scores are printed.

The implementation is self‑contained and only relies on ``numpy`` and the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a using an FFT."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) gradient – kept from the baseline
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _softmax_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth maximum (soft‑max).
    Uses momentum, β‑annealing and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 2.0
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.2
    rho = 0.90
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max (convex but non‑smooth)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the objective max_k (a*a)[k].
    The sub‑gradient at a is 2·∑_{k∈S} a shifted,
    where S = {k | (a*a)[k] = max}.
    """
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    step0 = target * 0.6
    best_val = _max_convolution_fft(a)
    best_a = a.copy()
    no_improve = 0
    it = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        if max_b < best_val - 1e-12:
            best_val = max_b
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early stop if stuck
        if no_improve >= 300:
            break

        # indices attaining the maximum
        S = np.flatnonzero(b == max_b)

        # compute sub‑gradient g_i = 2·∑_{k∈S} a_{k-i}
        g = np.zeros_like(a)
        for k in S:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # diminishing step size
        lr = step0 / np.sqrt(it + 1)
        a -= lr * g

        # project back onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target)

        it += 1

    return best_a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by projection
    onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Hill‑climbing with exact convolution updates (same idea as baseline)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9
    cooling = 0.9995

    while time.time() < deadline:
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination bias towards small entries
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

        score_hc = b_hc.max()
        if score_hc < best_score:

            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={score_hc:.6f}")

        # short hill‑climb at the very end
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 2.0)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
