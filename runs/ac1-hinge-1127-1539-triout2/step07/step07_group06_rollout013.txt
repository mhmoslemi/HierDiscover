"""
Solution for the “minimise max‑self‑convolution” challenge.

The programme works in three phases:

1. **Fast simulated‑annealing (SA)** – a massive number of cheap
   “mass‑transfer” moves (O(n) incremental update) and occasional
   swaps / jitter / short smoothing moves.  The SA schedule is deliberately
   slow to keep a non‑zero temperature until the last seconds, which helps
   to escape shallow local minima.

2. **Soft‑max gradient descent** – a short Adam‑style optimisation of the
   smooth approximation  log‑sum‑exp of the convolution, starting from the
   best SA candidate.

3. **Linear‑programming polishing** – a handful of LP steps that maximise
   the ℓ¹‑norm of a partner vector `g` under the current peak constraint.
   The partner is mixed back into the current iterate with a shrinking
   step size, guaranteeing a monotone decrease of the peak.

All steps respect the simplex constraint Σa = √(2·n) and the non‑negativity
requirement.  The routine stops automatically after the allotted budget
(~1000 s) and returns the best sequence encountered.
"""

import time
import numpy as np

# Scipy is optional – used only for the LP polishing stage.
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None  # type: ignore

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n with ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (assumes Σa = √(2·n))."""
    return float(_conv_fft(a).max())


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer `delta` from `donor` to `receiver`.
    Updates `a` and `conv` **exactly** (including the quadratic terms).
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    # keep a copy of a before the move – needed for the linear part
    a_before = a.copy()

    # linear piece: -2·δ·a_before shifted by donor, +2·δ·a_before shifted by receiver
    conv[donor : donor + n] -= 2.0 * delta * a_before
    conv[receiver : receiver + n] += 2.0 * delta * a_before

    # quadratic contributions (only three positions are affected)
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # update coefficients
    a[donor] -= delta
    a[receiver] += delta

    # small negative numbers may appear because of rounding
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)

def _sa_optimize(n: int,
                 deadline: float,
                 rng: np.random.Generator,
                 init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Fast SA optimizer for length `n`.  Returns the best sequence and its
    (max‑convolution) score.
    """
    target = np.sqrt(2.0 * n)

    # ---- initialise ----------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.40, 0.80)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total = deadline - start

    # ------------------------------------------------------------------
    while time.time() < deadline:
        now = time.time()
        tau = (now - start) / total      # normalised time [0,1]

        # temperature schedule: start ~0.5, end ~1e‑5 (geometric decay)
        temp = 0.5 * (1e-5 / 0.5) ** tau
        # maximal fraction of donor mass that may be moved (decays slowly)
        max_frac = 0.5 * (0.005 / 0.5) ** tau

        move_choice = rng.random()

        # ==============================================================
        # 1) mass‑transfer move (≈70 % of the moves)
        # ==============================================================

        if move_choice < 0.70:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            # contribution of each index to the peak entry
            contrib = a[idx] * a[k_max - idx]
            if contrib.sum() > 0.0:
                donor = rng.choice(idx, p=contrib / contrib.sum())
            else:
                donor = rng.integers(n)

            # receiver is drawn preferentially from small coefficients
            inv_weights = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv_weights / inv_weights.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            # Metropolis acceptance
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{now:.1f}] SA new best (transfer) max={best_max:.6f}")
            else:
                # revert
                _apply_transfer(a, conv, receiver, donor, delta)

        # ==============================================================
        # 2) swap move (≈15 %)
        # ==============================================================

        elif move_choice < 0.85:
            # bias the donor to large coefficients, the receiver to small ones
            prob_large = a / (a.sum() + 1e-12)
            prob_small = (1.0 / (a + 1e-12))
            prob_small /= prob_small.sum()

            i = rng.choice(np.arange(n), p=prob_large)
            j = rng.choice(np.arange(n), p=prob_small)
            if i == j:
                continue

            a[i], a[j] = a[j], a[i]          # swap
            conv = _conv_fft(a)
            new_max = conv.max()

            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] SA new best (swap)   max={best_max:.6f}")
            else:
                # revert swap
                a[i], a[j] = a[j], a[i]
                conv = _conv_fft(a)

        # ==============================================================
        # 3) short smoothing window (≈10 %)
        # ==============================================================

        elif move_choice < 0.95:
            max_win = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_win + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] SA smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # ==============================================================
        # 4) Gaussian jitter (≈5 %)
        # ==============================================================

        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved max={best_max:.6f}")

    return best_a, best_max

def _softmax_grad(a: np.ndarray, beta: float):
    """
    Gradient of  (1/β)·log∑exp(β·conv)  w.r.t. a.
    Returns gradient and the current convolution vector.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    p = exps / exps.sum()               # soft‑max probabilities

    # convolution of a (reversed) with p
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # back‑propagation through the FFT
    return grad, conv


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   beta_start: float = 1.0,
                   beta_end: float = 4096.0) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = target * 0.10

    start = time.time()
    it = 0
    beta = beta_start

    best_a = a.copy()
    best_score = _max_convolution(a)

    while time.time() < deadline:
        grad, _ = _softmax_grad(a, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = lr_base / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # simplex projection
        a = _project_to_simplex(a, target)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = np.random.default_rng().normal(scale=target * 0.001 / (beta + 1.0),
                                                     size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        cur = _max_convolution(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam improved β={beta:.1f} score={best_score:.6f}")

        # beta schedule – geometric increase
        if it % 300 == 0 and beta < beta_end:
            beta = min(beta * 1.4, beta_end)

        it += 1

    return best_a, best_score

# Linear‑programming polishing (optional, requires scipy)
# ----------------------------------------------------------------------


def _lp_polish(a: np.ndarray,
               target_sum: float,
               max_conv: float,
               step: float,
               rng: np.random.Generator) -> np.ndarray:
    """
    One LP‑based polishing step.

    Solve  max Σ b   s.t. b ≥ 0  and  a * b ≤ max_conv,
    scale the solution to the target ℓ¹‑norm and mix it back:
        a ← (1‑step)·a + step·g .
    Returns the updated ``a`` (already projected onto the simplex).
    """
    if linprog is None:          # scipy unavailable – give up gracefully
        return a

    n = a.size
    # Build the Toeplitz‑like matrix A (shape (2n‑1, n))
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = a[i]

    # Linear program – maximize Σb  (i.e. minimize –Σb)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, max_conv, dtype=np.float64)
    bounds = [(0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})

    if res.status != 0 or not res.success:
        # LP failed – return the unchanged vector
        return a

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return a

    # Scale up to the required ℓ¹‑norm
    scale = target_sum / sum_b
    g = b * scale

    # Mix with current a
    a_new = (1.0 - step) * a + step * g
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The routine combines a long simulated‑annealing run, a short Adam
    optimisation on a soft‑max surrogate and a few LP polishing steps.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # slight safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Simulated‑annealing on a few problem sizes
    # ------------------------------------------------------------------
    sizes = [1500, 1800, 2100, 2400]   # a modest spread of lengths
    for n in sizes:
        if time.time() >= deadline - 5.0:
            break
        # allocate a slice of the remaining budget (≈30 % per size)
        remaining = deadline - time.time()
        time_for_this = min(remaining * 0.30, 250.0)

        run_deadline = time.time() + time_for_this
        seq, score = _sa_optimize(n, run_deadline, rng)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] SA size={n} → new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam‑style optimisation on the best SA candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra = time.time() + (deadline - time.time()) * 0.25
        seq_adam, score_adam = _adam_optimize(best_seq, extra)
        if score_adam < best_score:
            best_score = score_adam
            best_seq = seq_adam.copy()
            print(f"[{time.time():.1f}] Adam polishing → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional LP polishing (a few aggressive steps)
    # ------------------------------------------------------------------
    if best_seq is not None and linprog is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = np.sqrt(2.0 * n)
        step_sizes = [0.5, 0.35, 0.2, 0.1]      # diminishing mixing weights
        for step in step_sizes:
            if time.time() >= deadline - 5.0:
                break
            cur_max = _max_convolution(best_seq)
            new_seq = _lp_polish(best_seq, target, cur_max, step, rng)
            new_max = _max_convolution(new_seq)
            if new_max < best_score - 1e-6:
                best_score = new_max
                best_seq = new_seq.copy()
                print(f"[{time.time():.1f}] LP polish step={step:.2f} → score={best_score:.6f}")
            else:
                # no improvement – stop early
                break

    # ------------------------------------------------------------------
    # 4) Trim trailing near‑zero entries (they do not influence the score)
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
