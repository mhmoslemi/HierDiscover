def _adam_gradient_descent(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 2.0
    beta_max = 1e7
    beta_factor = 1.5
    while time.time() < deadline:
        t += 1
        # compute gradient of smooth max
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = (0.01 * target_sum) / (np.sqrt(v_hat) + eps)  # step scaling
        a -= step * m_hat

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter (maybe not needed)
        if t % 300 == 0:
            jitter = rng.normal(scale=target_sum*1e-5, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # beta annealing
        if t % 500 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

assistantfinal**Solution Explanation**

The goal is to minimise  

\[
\text{objective}= \frac{2\,n\; \max\limits_{k}(a*a)_k}{(\sum_i a_i)^2},
\]

where `a` is a non‑negative step‑function of length `n`.  
Because the objective is homogeneous, we can **fix** the ℓ¹‑norm  

\[
\sum_i a_i=\sqrt{2n}
\]

and simply minimise the maximal entry of the self‑convolution  
\(M(a)=\max_k (a*a)_k\).  

The programme below follows three ideas that together beat the previous
implementation

| Phase | Idea |
|------|------|
| **0 – seeding / random‑search** | start from the supplied construction, up‑sample it to a larger size and try a few random perturbations (fast, cheap evaluations). |
| **1 – Adam optimisation on the exact max** | minimise the exact max‑convolution (no log‑sum‑exp smoothing).  
  The sub‑gradient of `M(a)` is known analytically and can be obtained in *O(n)* after a single FFT.  Adam (first‑ and second‑moment adaptation) together with a projection onto the ℓ¹‑simplex provides a stable, fast optimiser. |
| **2 – refined hill‑climbing** | after the continuous optimisation a classic simulated‑annealing hill‑climber updates the convolution in *O(n)* per move, works directly on the true max and removes the remaining “roughness”. |

The whole budget (default 1000 s) is split automatically between the three phases.
All heavy work (convolutions) is performed with FFTs, the code works for
vectors up to a few × 10⁴ elements within the time limit.

The implementation is completely self‑contained – only `numpy` and the
already available `evaluate_sequence` are used.

```python
"""
A fast optimiser for the self‑convolution bound.

* Phase 0 – seed from the global variable `height_sequence_1`,
  up‑sample and probe a few random perturbations.
* Phase 1 – Adam optimiser directly on the exact maximum of the
  convolution (sub‑gradient is 2·a reversed at the active shift).
  After each step the vector is projected onto the ℓ¹‑simplex
  Σa = √(2 n).
* Phase 2 – a simulated‑annealing hill‑climber that updates the full
  convolution in O(n) per move.
The whole routine respects a wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return the maximum entry of a*a using an FFT (fast)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Random initialisers (used only in the very first phase)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    # very small Gaussian jitter, then projection onto the simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation onto a new length, then projection onto the correct simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Phase 1 – Adam optimiser on the exact maximal convolution entry
# ----------------------------------------------------------------------


def _max_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute the sub‑gradient of M(a)=max_k (a*a)_k and return (g,M).

    The gradient is 2·a_rev shifted to the active index.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    b = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

    M = b.max()
    idx = int(np.argmax(b))

    # active indices range
    lo = max(0, idx - (n - 1))
    hi = min(n - 1, idx)

    g = np.zeros_like(a)
    i_range = np.arange(lo, hi + 1, dtype=int)
    j_range = idx - i_range                     # the partner indices
    g[i_range] = 2.0 * a[j_range]               # sub‑gradient formula

    return g, M


def _adam_max_descent(a0: np.ndarray, deadline: float, rng) -> np.ndarray:
    """
    Adam optimisation of the exact max‑convolution.
    The iterate is projected onto the ℓ¹‑simplex after each step.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    it = 0
    while time.time() < deadline:
        it += 1

        # sub‑gradient of the true max
        g, _ = _max_subgradient(a)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # step size – loosely scaled with the target ℓ¹‑norm
        step = (0.12 * target_sum) / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # keep on the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape flat regions
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-5, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a source index to a destination index
    while maintaining the full self‑convolution in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9          # initial temperature
    cooling = 0.9996                    # slower cooling for fine search

    while time.time() < deadline:
        # choose one of the currently maximal convolution positions
        max_idxs = np.flatnonzero(b == cur_score)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index – weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination – bias to light indices but keep some randomness
        if rng.random() < 0.35:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution – vectorised O(n) slice updates
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        if 2 * i < b_new.size:
            b_new[2 * i] += delta * delta
        if 2 * j < b_new.size:
            b_new[2 * j] += delta * delta
        if i + j < b_new.size:
            b_new[i + j] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

    # Phase 2 – final hill‑climbing
    # ------------------------------------------------------------------
    rem = deadline - time.time()
    if rem > 10.0:
        hill_deadline = time.time() + rem * 0.70
        b_full = _full_convolution(best_seq)
        best_seq, b_full = _hill_climbing(best_seq, b_full, hill_deadline)
        best_score = b_full.max()
        print(f"[{time.time():.1f}] final hill‑climb => score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:                                   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```
