"""
Proposes a step‑function (non‑negative sequence) that minimises
evaluate_sequence().  The search combines:
* random Dirichlet / power‑law initialisations,
* Adam optimisation on a soft‑max approximation of the peak of
  the self‑convolution,
* a short incremental local‑search phase for final polishing.
The routine respects the 1000 s budget, gradually explores several
lengths and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) using zero‑padded FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (full length)."""
    return float(_conv_fft(a).max())

# ----------------------------------------------------------------------
# Gradient of a soft‑max approximation of the peak
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where conv is the full convolution a*a and
    grad = d/da  (1/beta) log sum_k exp(beta * conv_k).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # forward convolution
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]   # a*a

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max weights, sum to 1

    # gradient: 2 * (a_rev * p)  (restricted to n components)
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b

# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the peak of a*a under the simplex constraint.
    Returns the best sequence encountered and its peak value.
    """
    target_sum = np.sqrt(2.0 * n)

    # ------- initialise -------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        # random mixture: 50 % power‑law, 50 % Dirichlet
        if rng.random() < 0.5:
            p = rng.uniform(0.3, 0.8)
            a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        else:
            a = rng.exponential(scale=1.0, size=n)   # Dirichlet‑like
    a = _project_to_simplex(a, target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # schedule for the soft‑max temperature
    beta = 1.0
    max_beta = 4096.0

    best_a = a.copy()
    best_peak = _max_convolution(a)

    iteration = 0
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # gradient & current convolution
        grad, conv = _gradient_softmax(a, beta)

        # Adam update (learning rate scaled with target_sum)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # base step size – chosen empirically
        step = target_sum * 0.12 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # keep feasibility
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow minima
        if iteration % 250 == 0:
            scale = target_sum * 0.001 / (beta + 1.0)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase soft‑max sharpness
        if iteration % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        current_peak = conv.max()
        if current_peak < best_peak:
            best_peak = current_peak
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={iteration} β={beta:.1f} new best peak={best_peak:.6f}")

        # early‑exit if we already beat the target
        if best_peak < 1.503:
            # keep a short margin before returning – but we can break now
            break

    return best_a, best_peak

# ----------------------------------------------------------------------
# Incremental local search (identical to the reference routine)
# ----------------------------------------------------------------------
def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """Transfer δ from donor to receiver and update the full convolution exactly."""
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    a_before = a.copy()

    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    a[donor] -= delta
    a[receiver] += delta

    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


def _local_search(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """Very fast incremental stochastic hill‑climbing."""
    target = np.sqrt(2.0 * n)

    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))

    a = _project_to_simplex(a, target)

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        max_frac = 0.5 * (0.005 / 0.5) ** tau
        temp = 0.5 * (1e-5 / 0.5) ** tau

        move_type = rng.random()

        # 1) mass transfer ------------------------------------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mass‑transfer improved max={best_max:.6f}")
            else:
                _apply_transfer(a, conv, receiver, donor, delta)

        # 2) short smoothing ----------------------------------------------
        elif move_type < 0.95:
            max_win = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_win + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # 3) Gaussian jitter -----------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved max={best_max:.6f}")

    return best_a, best_max

            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best peak≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mass‑transfer improved max={best_max:.6f}")
            else:
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing (≈10 %)
        # --------------------------------------------------------------
        elif move < 0.95:
            max_win = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_win + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = new_max <= cur_max or rng.random() < np.exp(
                    -(new_max - cur_max) / max(temp, 1e-12)
                )
                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) Gaussian jitter (≈10 %)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved max={best_max:.6f}")

    return best_a, best_max

# Main entry point required by the framework

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Searches for a step‑function with a small `evaluate_sequence` score.
    Several Adam‑optimised runs on different lengths are performed,
    followed by a short local‑search polishing phase.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")                  # stores the peak of the convolution

    # --------------------------------------------------------------
    # Warm‑start from a previously found construction (if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.size
        if time.time() < deadline - 5.0:
            run_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.15)
            cand, sc = _adam_optimize(n0, run_deadline, rng, init_seq=seq0)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] warm‑start size={n0} → score={best_score:.6f}")

    # --------------------------------------------------------------
    # Explore a variety of lengths
    # --------------------------------------------------------------
    lengths = [1200, 1600, 2000, 2400, 2800, 3200, 3600, 4000]
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        # give each length a modest slice of the remaining budget
        time_for_this = min(remaining * 0.15, 120.0)
        run_deadline = time.time() + time_for_this

        cand, sc = _adam_optimize(n, run_deadline, rng)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.1f}] size={n} → new global best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing with the cheap incremental local search
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.25)
        cand, sc = _local_search(n, extra_deadline, rng, init_seq=best_seq)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.1f}] polishing improved to score={best_score:.6f}")

    # --------------------------------------------------------------
    # Strip trailing zeros (they do not affect the objective)
    # --------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best peak≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
