Now solving LP:

def solve_lp_max_sum(f, M, S):
    n = len(f)
    A_ub = build_A_ub(f)
    b_ub = np.full(2*n-1, M)
    # objective: maximize sum(b) => minimize -sum(b).
    c = -np.ones(n)
    # bounds (0,S)
    bounds = [(0, S) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback to original vector
        return None
    b = res.x
    return b

"""
Optimisation of a non‑negative step‑function to minimise the peak of its
self‑convolution (the evaluation function of the competition).

Strategy
--------
1. **Initial sampling** – generate many analytic families (uniform,
   power‑law, beta‑like, sinusoidal, Gaussian) for a moderate length.
   Keep the best one (lowest self‑convolution peak).

2. **Linear‑program refinement** – for a smaller “base” length (≈2000)
   iteratively solve a linear programme:
        maximise  Σ b_j
        s.t.      (f * b)_k ≤ M   (M = max(f*f))
                  b_j ≥ 0,  b_j ≤ S
   where `f` is the current sequence, `M` its current peak and `S =
   √(2·n)` the required ℓ₁‑mass. The optimiser mixes the current `f`
   with the normalised LP solution (`g`) using a small convex step.
   A few such updates dramatically flatten the convolution.

3. **Upsampling** – linearly interpolate the refined base sequence to a
   larger target length (≈12000) and renormalise the ℓ₁‑mass.

4. **Projected Adam on a smooth‑max** – minimise a log‑sum‑exp
   approximation of the peak.  Temperature is increased gradually up
   to 400, and the Adam parameters are tuned for fast convergence.
   The iterate is always projected onto the simplex {x≥0, Σx=S}.

5. **Sub‑gradient descent + tiny random‑swap hill‑climb** – final
   polishing on the true max‑convolution.

All steps are time‑aware: the total budget (default 1000 s) is split
between the phases, and the best sequence found so far is always kept.
The final sequence is stored globally as `height_sequence_1` for future
runs.

The code relies only on `numpy`, `scipy` (linprog & sparse) and the
provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Linear‑program utilities (LP refinement)
# ----------------------------------------------------------------------
def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """Build sparse matrix A_ub such that (f*b)[k] = (A_ub @ b)[k]."""
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # For each output index k we insert all pairs (i,j) with i+j=k
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        rows.extend([k] * i_vals.size)
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_max_sum(f: np.ndarray, M: float, S: float) -> np.ndarray | None:
    """
    Solve:
        maximise Σ b_j
        s.t.   (f*b)_k ≤ M   for all k
               0 ≤ b_j ≤ S
    Returns the optimal b (or None if the LP fails).
    """
    n = f.shape[0]
    A_ub = _build_A_ub(f)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)          # minimise -Σb ⇒ maximise Σb
    bounds = [(0.0, S) for _ in range(n)]
    # ``highs`` is the modern, fast interior‑point solver in SciPy
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


def _lp_refine(f: np.ndarray,
               iters: int,
               t_init: float,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a small number of LP‑based updates on ``f``.
    Returns the best sequence seen and its max‑convolution value.
    """
    n = f.shape[0]
    S = math.sqrt(2.0 * n)
    best_f = f.copy()
    best_score = np.max(_full_convolution(f))

    for it in range(iters):
        if time.time() > deadline:
            break
        M = np.max(_full_convolution(f))
        b = _solve_lp_max_sum(f, M, S)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 0:
            break
        scale = S / sum_b
        g = b * scale
        # enforce exact symmetry (helps the convolution peak)
        g = _symmetrize(g)
        # convex combine – step size gradually decays
        t = max(0.02, t_init * (0.85 ** it))
        f = (1.0 - t) * f + t * g
        f = _project_simplex(f, S)
        f = _symmetrize(f)

        cur = np.max(_full_convolution(f))
        if cur < best_score:
            best_score = cur
            best_f = f.copy()

        # tiny random jitter to escape possible plateaus
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and f[i] > 0:
                delta = 0.05 * f[i] * rng.random()
                cand = f.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand = _symmetrize(cand)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur:
                    f = cand
                    cur = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_f = cand.copy()
    return best_f, best_score

# Optimisers (Adam, sub‑gradient, local swap)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a soft‑max approximation of the peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.2
    alpha_mul = 1.07
    max_alpha = 400.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99996
    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,

    """
    Sub‑gradient descent on the true maximum convolution entry.
    """

    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i

        a = _project_simplex(a, S)

            best_a = a.copy()

                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)

                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()
        # early exit if no progress for a while
        if it % 8000 == 0 and best_score == cur_max:
            break
    return best_a, best_score


def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that swaps small masses between random positions.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a small safety margin so that we always have a chance to return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve previous best if it exists
    # ------------------------------------------------------------------
    base_seq_global = None
    if 'height_sequence_1' in globals():
        try:
            base_seq_global = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq_global = None

    # ------------------------------------------------------------------
    # Phase 1 – cheap analytic families (short length, many samples)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    attempts = 0

    phase1_deadline = start_time + 0.25 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1

        # pick length for the cheap phase – moderate to keep LP cheap later
        n = int(rng.integers(1800, 3500))

        # pick a family
        fam = rng.integers(0, 7)
        if fam == 0 and base_seq_global is not None:
            # down‑sample a known good sequence
            a = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, base_seq_global.size),
                          base_seq_global)
        elif fam == 1:
            a = rng.random(n)                         # uniform random
        elif fam == 2:
            # symmetric geometric decay from both ends
            rate = rng.uniform(0.94, 0.9995)
            left = rate ** np.arange(n)
            a = (left + left[::-1]) * 0.5
        elif fam == 3:
            # power‑law symmetric
            p = rng.uniform(0.2, 4.5)
            idx = np.arange(n, dtype=float)
            a = ((idx + 1) * (n - idx)) ** p
        elif fam == 4:
            # beta‑like skewed shape
            p1 = rng.uniform(0.1, 5.0)
            p2 = rng.uniform(0.1, 5.0)
            x = np.linspace(0, 1, n)
            a = (x ** p1) * ((1 - x) ** p2)
        elif fam == 5:
            # sinusoidal power
            p = rng.uniform(0.5, 4.0)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        else:
            # discrete Gaussian
            sigma = rng.uniform(0.12, 0.28) * n
            centre = (n - 1) / 2.0
            a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)

        a = np.clip(a, 0.0, None)

        # normalise to the required ℓ₁‑mass
        S = math.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # optional symmetrisation (helps later)
        if rng.random() < 0.45:
            a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[phase1] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # If nothing was found (unlikely) fall back to a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2500
        a = rng.random(n)
        a = a / a.sum() * math.sqrt(2.0 * n)
        best_seq = a
        best_score = np.max(_full_convolution(a))

    # ------------------------------------------------------------------
    # Phase 2 – LP based refinement on a modest base size (≈2000)
    # ------------------------------------------------------------------
    # Down‑sample (or up‑sample) the current best to a fixed base size
    base_n = 2000
    if best_seq.shape[0] != base_n:
        base_seq = np.interp(np.linspace(0, 1, base_n),
                             np.linspace(0, 1, best_seq.shape[0]),
                             best_seq)
    else:
        base_seq = best_seq.copy()

    # Ensure non‑negativity & normalise again (tiny numerical drift possible)
    base_seq = np.maximum(base_seq, 0.0)
    base_seq = _project_simplex(base_seq, math.sqrt(2.0 * base_n))

    # Run a few LP updates – time‑aware
    now = time.time()
    lp_deadline = now + 0.20 * (deadline - now)   # ~20 % of remaining budget
    base_seq, base_score = _lp_refine(base_seq,
                                      iters=12,
                                      t_init=0.22,
                                      deadline=lp_deadline,
                                      rng=rng)
    if base_score < best_score:
        best_score = base_score
        best_seq = base_seq.copy()
        print(f"[lp‑refine] score={base_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Upsample to the final target length
    # ------------------------------------------------------------------
    target_n = max(8000, int(best_seq.shape[0] * 1.8))
    target_n = min(target_n, 15000)   # stay comfortably below the memory ceiling
    seq = _upsample_seq(best_seq, target_n)
    seq = _project_simplex(seq, math.sqrt(2.0 * target_n))
    seq = _symmetrize(seq)

    # ------------------------------------------------------------------
    # Phase 4 – Projected Adam on the smooth‑max approximation
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        adam_deadline = now + 0.55 * (deadline - now)   # majority of remaining time
        seq, adam_score = _optimize_adam(seq, adam_deadline, rng, enforce_sym=True)
        if adam_score < best_score:
            best_score = adam_score
            best_seq = seq.copy()
            print(f"[adam] improved to {adam_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Sub‑gradient descent (true max)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        sub_deadline = now + 0.30 * (deadline - now)
        seq, sub_score = _optimize_subgrad(seq, sub_deadline, rng, enforce_sym=True)
        if sub_score < best_score:
            best_score = sub_score
            best_seq = seq.copy()
            print(f"[subgrad] improved to {sub_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – Tiny random‑swap hill‑climb (final polishing)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        seq, local_score = _local_swap_search(seq, deadline, rng, enforce_sym=True)
        if local_score < best_score:
            best_score = local_score
            best_seq = seq.copy()
            print(f"[local] improved to {local_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (also expose it globally for future runs)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
