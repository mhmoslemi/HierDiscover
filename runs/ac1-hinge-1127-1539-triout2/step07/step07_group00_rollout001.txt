"""
Propose a candidate step‑function that (approximately) minimises
``evaluate_sequence``.  

The search consists of three stages:

1. **Random‑family exploration** – generate many candidate vectors of
   different lengths from a handful of analytic families (exponential,
   power‑law, mixed, Dirichlet, piecewise) and from a few simple
   permutations of the values (reverse, alternating‑ends, random).  Each
   candidate is projected onto the ℓ¹‑simplex Σa = √(2 n) and the
   objective 2·n·max(a*a)/(Σa)² is evaluated with an FFT‑based convolution.

2. **Local refinement** – the current best vector is improved by projected
   gradient descent on a smooth max (soft‑max) of the convolution, with
   momentum, β‑annealing and jitter, followed by a fast exact‑convolution
   hill‑climber that moves small amounts of mass between indices while
   keeping the full self‑convolution up‑to‑date in O(n) per move.

3. **Final polishing** – a last round of gradient descent and hill‑climbing
   is performed on the remaining time budget.

All helper routines are top‑level; the main entry point is
``propose_candidate`` which respects the 1000 s wall‑clock limit and
returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT (fast)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).

    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Momentum, β‑annealing and occasional jitter are used.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 2.0
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.2
    rho = 0.90
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0005 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of
    mass from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # slower cooling for finer search

    while time.time() < deadline:
        # pick one of the currently worst convolution positions
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: bias towards low‑mass locations
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (O(n) slice updates)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

# Candidate generators
# ----------------------------------------------------------------------


def _random_family(n: int, rng) -> np.ndarray:
    """
    Produce a random vector on the ℓ¹‑simplex Σa = √(2 n) from a
    mixture of analytic families.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.8)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.02) + 1e-4
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.5)
        lam = rng.exponential(scale=0.02) + 1e-4
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                   # piecewise exponential (two segments)
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02) + 1e-4
        lam2 = rng.exponential(scale=0.02) + 1e-4
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n) + rng.normal(scale=target * 0.001, size=n)

    # project onto the required ℓ¹‑simplex
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _permute_spread(a: np.ndarray, rng) -> np.ndarray:
    """
    Return a permutation of ``a`` that spreads large values.
    Four simple strategies are tried at random:
        0 – keep original order
        1 – reverse
        2 – alternating ends (largest, smallest, 2nd‑largest, 2nd‑smallest,…)
        3 – completely random
    """
    n = a.size
    typ = rng.integers(0, 4)
    if typ == 0:
        return a
    if typ == 1:
        return a[::-1]
    if typ == 2:
        # descending values placed alternately at the left and right ends
        vals = np.sort(a)[::-1]
        out = np.empty_like(a)
        left, right = 0, n - 1
        for i, v in enumerate(vals):
            if i % 2 == 0:
                out[left] = v
                left += 1
            else:
                out[right] = v
                right -= 1
        return out
    # typ == 3
    return rng.permuted(a)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0      # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Helper to update the global best
    # ------------------------------------------------------------------
    def _update_best(seq: np.ndarray, source: str = ""):
        nonlocal best_seq, best_score
        n = seq.size
        target = np.sqrt(2.0 * n)
        # enforce exact ℓ¹‑norm (tiny numerical drift is possible)
        seq = _project_to_simplex(seq, target)
        maxb = _max_convolution_fft(seq)
        score = 2.0 * n * maxb / (target ** 2)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New best ({source}): n={n}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Seed from previously discovered construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            _update_best(seq0, "seed")
        except Exception as e:
            print("seed error:", e)

    # ------------------------------------------------------------------
    # 1) Random‑family exploration (major part of the budget)
    # ------------------------------------------------------------------
    lengths = [1000, 2000, 4000, 8000, 12000]   # diverse sizes
    while time.time() < deadline - 5.0:
        n = int(rng.choice(lengths))
        a = _random_family(n, rng)
        a = _permute_spread(a, rng)
        _update_best(a, "random_family")

        # Occasionally launch a very cheap refinement of the current best
        if best_seq is not None and time.time() < deadline - 25.0:
            # 10 s gradient descent from the best vector (may improve it)
            gd_dead = time.time() + 10.0
            a_gd = _gradient_descent_opt(best_seq, gd_dead)

    # ------------------------------------------------------------------
    # 2) Local refinement of the best candidate found so far
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        # Gradient descent on the remaining budget (except a little margin)
        gd_dead = time.time() + min(30.0, deadline - time.time() - 10.0)
        a_opt = _gradient_descent_opt(best_seq, gd_dead)
        _update_best(a_opt, "final_gd")

        # Hill‑climbing using the exact convolution
        if time.time() < deadline - 5.0:
            b_full = _full_convolution(a_opt)
            a_hc, b_hc = _hill_climbing(a_opt, b_full, deadline - 2.0)

    # 3) Final clean‑up & output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    # drop trailing near‑zeros for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print("Error calling evaluate_sequence:", exc)

    return best_seq.tolist()
