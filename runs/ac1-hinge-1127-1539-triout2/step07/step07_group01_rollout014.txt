"""
Improved step‑function optimizer.

Overview
--------
1. **Diversified random search** – generate candidates from many
   parametric families (power‑law, exponential, linear, sparse‑Gamma,
   uniform, Dirichlet, and a new Beta‑distribution family that mimics
   the optimal continuous ½‑power shape).  Each candidate is normalised
   so that its ℓ₁‑norm equals √(2 n) and its self‑convolution peak is
   measured with an FFT.  The best candidate is kept globally.

2. **Multi‑scale Adam refinement** – the current best vector is
   interpolated to a sequence of larger lengths.  For every target
   length we run several short Adam optimisations on the simplex (via
   a log‑softmax parametrisation) that minimise the exact ‑soft‑max
   approximation of the peak convolution.  A few restarts and periodic
   jitter help escape plateaus.

3. **Final evaluation** – the best vector found is returned and stored
   in the global variable ``height_sequence_1`` for possible warm‑starts
   in future runs.

The whole routine respects the 1000 s time budget and prints progress
information useful for debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the τ‑soft‑max of the convolution peak
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with
        grad = ∂ τ‑soft‑max(b) / ∂a   (shape (n,))
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the exact max‑convolution.
    Returns the best vector seen and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.5                             # a tad larger than the reference

    # initialise smoothing temperature
    _, b0 = _softmax_gradient(a0, 0.0)
    tau = max(b0, 0.5)

    best_a = a0.copy()
    best_peak = b0
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient w.r.t. a (soft‑max version)
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (chain rule)
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.9992            # decrease smoothing a little faster
        lr *= 0.9997             # very slow decay

        # periodic jitter to escape flat regions
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.1, 2.5)          # exponent (allowing values <1)
    offset = rng.uniform(0.0, 12.0)    # denominator shift
    mix = rng.uniform(0.0, 1.0)        # uniform background weight
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0008, 0.018)   # decay rate
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.6, 5.5)                     # height at the very edge
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))  # controls width
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.25))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.05, 0.85)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like start (raw random numbers)."""
    return rng.random(n)


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Beta‑distribution family (α,β∈(0,1)) – heavy at the edges.
    The density ∝ x^{α-1}(1−x)^{β-1} on (0,1).
    """
    alpha = rng.uniform(0.07, 0.92)
    beta = rng.uniform(0.07, 0.92)
    # mid‑point sampling to avoid exact 0/1
    i = np.arange(n) + 0.5
    x = i / n
    raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # numerical safety
    raw = np.nan_to_num(raw, nan=0.0, posinf=0.0, neginf=0.0)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best and otherwise
    samples from a random parametric family (including the new beta‑family).
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ~0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    family = rng.choice(['powerlaw', 'exponential', 'linear',
                         'gamma', 'uniform', 'dirichlet', 'beta'])
    if family == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif family == 'exponential':
        raw = _init_exponential(n, rng)
    elif family == 'linear':
        raw = _init_linear(n, rng)
    elif family == 'gamma':
        raw = _init_gamma(n, rng)
    elif family == 'uniform':
        raw = _init_uniform(n, rng)
    elif family == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:   # beta
        raw = _init_beta(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a
