def subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    step = target_sum  # initial step size maybe target_sum
    decay = 0.999
    iteration = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        n = a.shape[0]
        # compute convolution using FFT
        size = 1 << ((2*n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        maxb = b.max()
        # find active indices
        # Use tolerance relative to maxb
        eps = 1e-12
        active = np.where(b >= maxb - eps)[0]
        # compute subgradient
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j = k - i_range
            g[i_range] += 2 * a[j]
        # gradient step
        # adjust step magnitude
        if np.linalg.norm(g) > 0:
            # step size proportional to 1 / sqrt(iteration)
            lr = step / (np.sqrt(iteration + 1) + 1)
            a = a - lr * g
        # projection
        a = _project_to_simplex(a, target_sum)
        # jitter
        if iteration % 200 == 0:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        step *= decay
        iteration += 1

    return a

def _continuous_annealing(a0, deadline):
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_score = _max_convolution_fft(best)
    T = best_score * 0.005  # small initial temperature maybe.
    decay = 0.999
    rng = np.random.default_rng()
    while time.time() < deadline:
        i,j = rng.integers(0,n, size=2)
        if i==j: continue
        # choose delta
        max_delta = min(a[i], 0.1 * a[i])  # up to 10% of a[i]
        if max_delta <= 0:
            continue
        delta = rng.uniform(0, max_delta)
        # move delta from i to j
        a[i] -= delta
        a[j] += delta
        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score < best_score or rng.random() < np.exp(-dE / T):
            if cur_score < best_score:
                best_score, best = cur_score, a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
        T *= decay
    return best

"""
Improved step‑function search.

Algorithm
---------
1. Randomly generate a bunch of candidate probability vectors
   (power‑law, exponential, linear, quadratic, Dirichlet and mixtures).
   Each candidate is scaled so that Σ a_i = √(2·n) (the evaluator’s ℓ¹‑norm).

2. For every generated multiset try a few deterministic orderings:
   * descending, ascending, alternating, random shuffle and a
     “half‑reversed” pattern that pairs large with small values.
   The best few (elite) candidates are kept.

3. **Sub‑gradient descent on the true max‑convolution**:
   The objective is  max_k (a*a)[k]; its sub‑gradient is the sum of the
   gradients of the active (maximal) entries.
   An Euclidean projected sub‑gradient method (with a simple step‑size
   schedule and occasional jitter) drives the vector toward a
   Chebyshev‑center of the feasible polytope.

4. **Permutation annealing** – improves the ordering of the refined
   multiset by random swaps (simulated annealing).

5. **Mass‑transfer annealing** – a second simulated‑annealing phase
   that moves a small amount of mass from one position to another while
   keeping the ℓ¹‑norm fixed.  This can further lower the peak of the
   self‑convolution.

6. The global best (across all lengths explored) is returned as a plain
   Python list.  All steps respect the supplied time budget (default
   1000 s).  If a warm‑start vector ``height_sequence_1`` is defined it
   is used as an additional seed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x≥0   (Duchi‑et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (return a vector already ℓ¹‑normalised)
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ¹‑sum = target."""
    total = v.sum()
    if total <= 0:
        # fallback to a uniform vector
        return np.full_like(v, target / v.size)
    a = v * (target / total)
    return np.maximum(a, 0.0)


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _dirichlet_sequence(n: int, alpha: float) -> np.ndarray:
    rng = np.random.default_rng()
    raw = rng.gamma(alpha, 1.0, size=n)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _mixed_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of a power‑law and an exponential component."""
    w = rng.uniform(0.2, 0.8)
    # power‑law part
    p = rng.uniform(0.30, 0.90)
    sh_p = rng.uniform(0.0, 5.0)
    pow_seq = _powerlaw_sequence(n, p, sh_p)
    # exponential part
    rate = rng.uniform(0.0001, 0.02)
    sh_e = rng.uniform(0.0, 5.0)
    exp_seq = _exponential_sequence(n, rate, sh_e)
    mix = w * pow_seq + (1.0 - w) * exp_seq
    return _project_to_simplex(mix, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (convex but non‑smooth)
# ----------------------------------------------------------------------
def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on max_k (a*a)[k]."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # step‑size schedule
    step = target_sum * 0.5
    decay = 0.9995
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # full convolution via FFT
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        maxb = b.max()
        # indices where the max is attained (tolerance for floating noise)
        active = np.where(b >= maxb - 1e-12)[0]

        # compute sub‑gradient: sum over active shifts
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j = k - i_range
            g[i_range] += 2.0 * a[j]

        norm = np.linalg.norm(g)
        if norm > 0.0:
            lr = step / (np.sqrt(it + 1) + 1.0)
            a -= lr * g

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        step *= decay
        it += 1

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improves ordering by random swaps (annealed acceptance)."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # start temperature proportional to current best value
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # tiny random block shuffle now and then
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur = _max_convolution_fft(a)
                if cur < best_score:
                    best_score = cur
                    best = a.copy()

    return best

# Mass‑transfer simulated annealing (continuous moves that keep ℓ¹ fixed)
# ----------------------------------------------------------------------
def _mass_transfer_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Moves a small amount of mass between positions while preserving sum."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.01 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer at most 10 % of the donor mass
        max_transfer = 0.1 * a[i]
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)

        a[i] -= delta
        a[j] += delta

        cur = _max_convolution_fft(a)
        dE = cur - best_score
        if cur < best_score or (T > 0.0 and rng.random() < np.exp(-dE / T)):
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Ordering heuristics (given a multiset)
# ----------------------------------------------------------------------
def _generate_orderings(base: np.ndarray) -> list[np.ndarray]:
    """Return a list of candidate orderings for the same multiset."""
    n = base.size
    asc = np.sort(base)            # smallest → largest
    desc = asc[::-1]               # largest → smallest

    # alternating small–large (small first)
    alt = np.empty_like(asc)
    lo, hi = 0, n - 1
    for pos in range(n):
        if pos % 2 == 0:
            alt[pos] = asc[lo]
            lo += 1
        else:
            alt[pos] = asc[hi]
            hi -= 1

    # half‑reversed: first half descending, second half ascending
    half = n // 2
    rev_half = np.concatenate([asc[:half][::-1], asc[half:]])

    # a single random shuffle
    shuffle = asc.copy()
    rng = np.random.default_rng()
    rng.shuffle(shuffle)

    return [desc, asc, alt, rev_half, shuffle]

            c = rng.uniform(0.0, 5e-5)
            base = _quadratic_sequence(n, a0, b, c)
        elif family == 'dirichlet':
            alpha = rng.uniform(0.1, 5.0)
            base = _dirichlet_sequence(n, alpha)
        else:   # mixed
            base = _mixed_sequence(n, rng)

        # try several deterministic orderings
        for cand in _generate_orderings(base):
            _consider(cand)

        # occasional cheap permutation annealing on the current best
        if best_seq is not None and rng.random() < 0.07:
            tmp = _permutation_annealing(best_seq, time.time() + 0.25)
            _consider(tmp)

    # --------------------------------------------------------------
    # 2. Refinement of elite candidates
    # --------------------------------------------------------------
    # each elite gets a sub‑gradient phase followed by permutation & mass‑transfer
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        # sub‑gradient phase (≈15 s or whatever remains)
        sub_dead = min(deadline, time.time() + 15.0)
        refined = _subgradient_descent_opt(seed_seq, sub_dead)

        # permutation annealing
        refined = _permutation_annealing(refined, deadline)

        # final mass‑transfer annealing (small budget)
        refined = _mass_transfer_annealing(refined, deadline)

        _consider(refined)

    # --------------------------------------------------------------
    # 3. Final global permutation annealing (if time left)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _permutation_annealing(best_seq, deadline)

    # --------------------------------------------------------------
    # 4. Fallback / post‑processing
    # --------------------------------------------------------------
    if best_seq is None:
        # very simple uniform fallback
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim negligible tail entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional verification with the official evaluator
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective ≈{best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
