"""
Propose a step‑function that (approximately) minimizes

    2 * n * max(conv(a,a)) / (sum(a)**2)

The ratio is invariant to scaling, so we may (and do) enforce the
ℓ¹‑simplex Σa = √(2 n).  The objective then equals the maximal entry of
the self‑convolution.  The search consists of three stages:

1. **Broad random sampling** – several parametric families (power‑law,
   exponential, beta‑shaped, mixed) are sampled for many lengths up to
   30 000.  Every generated multiset is sorted (decreasing) – this tends
   to pair large values with small ones after reversal and therefore
   yields a low convolution peak.  The best 12 candidates are kept as
   *elite*.

2. **Refinement** – each elite candidate is improved by a projected
   Adam‑style gradient descent on a smooth log‑sum‑exp surrogate of the
   max‑convolution.  After every gradient step the vector is projected
   onto the ℓ¹‑simplex and re‑sorted in decreasing order (monotone cone
   projection).  The surrogate’s temperature β is annealed to make the
   optimisation focus on the true maximum.

3. **Permutation annealing** – the ordering of the multiset is
   subsequently refined by simulated‑annealing swaps (the objective is
   evaluated with a fast FFT based convolution).  The current global
   best is also subjected to a final annealing run.

All work respects the supplied time budget (default 1000 s) and all
numbers are forced to be non‑negative.  The routine returns a plain
Python list of floats that can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(conv) and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns (b, g) where
        b = a*a   (full convolution, length 2·n‑1)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a_rev ⋆ p)   with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shift = np.exp(beta * (b - max_b))
    p = exp_shift / np.sum(exp_shift)

    # gradient g = 2·(a_rev ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    A_rev_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1:n - 1 + n]
    return b, g


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed max‑convolution surrogate
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑style projected optimisation of the smooth surrogate.
    After each step the vector is projected onto the ℓ¹‑simplex and
    sorted in decreasing order to stay in the monotone cone.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    # temperature for the smooth max
    beta_temp = 1.0
    max_beta = 1e5

    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of the smooth surrogate
        _, g = _smooth_max_and_grad(a, beta_temp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size that respects the ℓ¹‑budget
        step = 0.5 * target_sum / (np.sqrt(v_hat) + eps)

        a -= step * m_hat

        # keep feasible
        a = _project_to_simplex(a, target_sum)

        # enforce monotone decreasing order (projection onto the cone)
        a = np.sort(a)[::-1]

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(
                scale=target_sum * 5e-4 / (beta_temp + 1.0), size=n
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)
            a = np.sort(a)[::-1]

        # anneal the temperature
        if it % 200 == 0 and beta_temp < max_beta:
            beta_temp = min(beta_temp * 1.07, max_beta)

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improves the ordering of a fixed multiset by random swaps."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current best
    T = best_score * 0.02 + 1e-12
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        if cur_score <= best_score:
            accept = True
        else:
            dE = cur_score - best_score
            accept = rng.random() < np.exp(-dE / T)

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # tiny global jitter every now and then
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.015
            if np.any(mask):
                sub = a[mask]
                rng.shuffle(sub)
                a[mask] = sub
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Families of raw (un‑normalised) shapes
# ----------------------------------------------------------------------
def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)


def _beta_sequence(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution shape on a discrete grid."""
    t = np.linspace(0.0, 1.0, n, dtype=float)
    raw = (t ** a) * ((1.0 - t) ** b)
    # avoid the exact zero at the ends (they would be dropped after
    # projection) – add a tiny epsilon before normalising.
    raw += 1e-12
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)


def _mixed_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Convex mixture of a power‑law and an exponential."""
    w = rng.uniform(0.2, 0.8)

    p = rng.uniform(0.30, 0.90)
    shift_p = rng.uniform(0.0, 5.0)
    pow_seq = _powerlaw_sequence(n, p, shift_p)

    rate = rng.uniform(0.0001, 0.02)
    shift_e = rng.uniform(0.0, 5.0)
    exp_seq = _exponential_sequence(n, rate, shift_e)

    mix = w * pow_seq + (1.0 - w) * exp_seq
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(mix, target)


# ----------------------------------------------------------------------
# Candidate handling utilities
# ----------------------------------------------------------------------
def _consider(seq: np.ndarray, best_dict: dict, elite: list, now: float):
    """Update global best and elite list."""
    obj = _max_convolution_fft(seq)
    if obj < best_dict["value"]:
        best_dict["value"] = obj
        best_dict["seq"] = seq.copy()
        print(f"[{now:.1f}] New global best obj={obj:.6f} len={seq.size}")

    # maintain elite (sorted by objective)
    if len(elite) < 12 or obj < elite[-1][0]:
        elite.append((obj, seq.copy()))
        elite.sort(key=lambda x: x[0])
        if len(elite) > 12:
            elite.pop()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` score.  The routine obeys the time budget,
    explores several parametric families, refines promising candidates
    with a projected Adam optimiser (monotone projection) and finishes
    with permutation‑annealing.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best = {"value": np.inf, "seq": None}
    elite = []       # list of (obj, array)

    # ------------------------------------------------------------------
    # 0. Optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        n0 = init.size
        init = _project_to_simplex(init, np.sqrt(2.0 * n0))
        _consider(init, best, elite, time.time())
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Broad random sampling (≈ 60 % of the budget)
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000, 10000, 12000, 15000, 20000, 25000, 30000]
    families = ["power", "exp", "beta", "mixed"]
    gen_deadline = time.time() + 0.60 * (deadline - time.time())

    while time.time() < gen_deadline:
        n = int(rng.choice(lengths))
        family = rng.choice(families)

        if family == "power":
            p = rng.uniform(0.30, 0.90)
            shift = rng.uniform(0.0, 5.0)
            base = _powerlaw_sequence(n, p, shift)

        elif family == "exp":
            rate = rng.uniform(0.0001, 0.02)
            shift = rng.uniform(0.0, 5.0)
            base = _exponential_sequence(n, rate, shift)

        elif family == "beta":
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            base = _beta_sequence(n, a, b)

        else:   # mixed
            base = _mixed_sequence(n, rng)

        # ------------------------------------------------------------------
        # Ordering heuristics – we start from a monotone decreasing order,
        # then also try a random shuffle and an alternating pattern.
        # ------------------------------------------------------------------
        desc = np.sort(base)[::-1]          # decreasing
        asc = desc[::-1]                    # increasing

        # alternating small–large (small first)
        alt = np.empty_like(desc)
        lo, hi = 0, n - 1
        for i in range(n):
            if i % 2 == 0:
                alt[i] = desc[lo]
                lo += 1
            else:
                alt[i] = desc[hi]
                hi -= 1

        # a single random shuffle
        shuf = desc.copy()
        rng.shuffle(shuf)

        for cand in (desc, asc, alt, shuf):
            _consider(cand, best, elite, time.time())

        # occasional cheap annealing on the current global best
        if best["seq"] is not None and rng.random() < 0.05:
            tmp = _permutation_annealing(best["seq"], time.time() + 0.25)
            _consider(tmp, best, elite, time.time())

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size

        # gradient phase – give each elite ≈ 20 s (or whatever is left)
        gd_deadline = min(deadline, time.time() + 20.0)
        refined = _adam_descent(seed_seq, gd_deadline)

        # permutation annealing – use whatever time remains globally
        refined = _permutation_annealing(refined, deadline)
        _consider(refined, best, elite, time.time())

    # ------------------------------------------------------------------
    # 3. Final global permutation annealing (if any time left)
    # ------------------------------------------------------------------
    if best["seq"] is not None and time.time() < deadline:
        best_seq = _permutation_annealing(best["seq"], deadline)
        best["seq"] = best_seq
        best["value"] = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 4. Safety fallback (should never trigger)
    # ------------------------------------------------------------------
    if best["seq"] is None:
        n = 8000
        best["seq"] = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing near‑zeros (purely cosmetic)
    nz = np.where(best["seq"] > 1e-12)[0]
    if nz.size:
        best["seq"] = best["seq"][: nz[-1] + 1]

    # Optional verification (the harness will also call evaluate_sequence)
    try:
        val = evaluate_sequence(best["seq"].tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best['value']:.6f} "
          f"length = {best['seq'].size}")

    return best["seq"].tolist()
