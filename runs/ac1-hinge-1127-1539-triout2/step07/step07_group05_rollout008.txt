"""
Propose a non‑negative step function with a small maximal self‑convolution.
The search consists of three stages:

1.  **Diverse random families** – power‑law, exponential, linear,
    quadratic and a simple mixture.  Each generated vector is projected onto
    the ℓ¹‑simplex Σ a_i = √(2 n) and a few deterministic orderings are
    tried (descending, ascending, alternating, random shuffle).

2.  **Two‑phase gradient refinement**
        a)  *Smooth* optimisation of the log‑sum‑exp surrogate of the max
            (the routine ``_gradient_descent_opt`` from the previous
            implementation).  This quickly moves the vector into a good
            basin.
        b)  *Exact* sub‑gradient descent on the true max – for the current
            maximal shift ``k`` the sub‑gradient is
            ``g_i = 2·a_{k‑i}`` (zero when the index is out of range).
            After each step the vector is projected back onto the simplex.
            This phase can lower the objective a little further.

    Both phases are stopped by the global time‑budget.

3.  **Permutation annealing** – with the multiset fixed, random swaps are
    performed and accepted with a Metropolis criterion.  This improves the
    ordering of the entries.  The best sequence seen so far is kept.

The algorithm runs for the full 1000 s budget, keeps an elite list of the
best few candidates and finishes with a final annealing of the current
global best.  All operations respect the security checks of
``evaluate_sequence`` – the returned list contains only non‑negative
floats and the ℓ¹‑norm is normalised as required.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from typing import List, Tuple

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _max_convolution_and_idx(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, index) of a*a using FFT (full convolution)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    val = float(conv[idx])
    return val, idx


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float) -> Tuple[np.ndarray, np.ndarray]:
    """
    Returns (g, b) where
        b = a*a  (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    b = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = rfft(a_rev, n=size2)
    Pr_fft = rfft(p, n=size2)
    conv_rev = irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max
# ----------------------------------------------------------------------
def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 2e5
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.6 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta (make surrogate sharper)
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max (piecewise‑linear)
# ----------------------------------------------------------------------
def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    Keeps the vector on the ℓ¹‑simplex Σa = √(2 n).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # initial step size – roughly inverse of typical gradient norm
    step = target_sum * 0.2
    decay = 0.9995
    rng = np.random.default_rng()

    best = a.copy()
    best_obj = _max_convolution_fft(a)

    while time.time() < deadline:
        cur_val, idx = _max_convolution_and_idx(a)

        # compute sub‑gradient for the active shift idx
        js = np.arange(n)
        mask = (idx - js >= 0) & (idx - js < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx - js[mask]]

        # normalise step to keep updates of comparable magnitude
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            lr = step / (gnorm + 1e-12)
        else:
            lr = 0.0

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        new_obj = _max_convolution_fft(a_new)
        # keep the best ever seen
        if new_obj < best_obj:
            best_obj = new_obj
            best = a_new.copy()

        # accept the new point if it improves, otherwise keep old (with tiny chance)
        if new_obj <= cur_val or rng.random() < 0.01:
            a = a_new

        # decay step size
        step *= decay

        # occasional random jitter to avoid stagnation
        if rng.random() < 0.008:
            jitter = rng.normal(scale=target_sum * 3e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    The L¹‑norm stays unchanged, only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current best
    T = best_score * 0.025 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score or (T > 0 and rng.random() < np.exp(-dE / T)):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny global jitter
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.015
            if np.any(mask):
                sh = a[mask].copy()
                rng.shuffle(sh)
                a[mask] = sh
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Helpers for generating families and normalisation
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))
