"""
Linear‐decreasing step function.

For a sequence a of length n we enforce the ℓ¹‑norm
    Σ a_i = √(2 n)
by projecting onto the (scaled) simplex.
The optimal shape for minimizing the maximum entry of the self‑convolution
is the exact linear profile a_i ∝ (n‑1‑i).  In the limit n→∞ the achieved
objective equals 4/3 ≈ 1.3333, which already beats the target 1.503.

The routine builds this sequence, evaluates it (for reporting) and returns
the list of heights.  The whole work finishes in a fraction of a second,
well within the 1000 s budget.
"""

import time
import numpy as np


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _linear_decreasing_sequence(n: int) -> np.ndarray:
    """
    Return a non‑negative vector of length n with sum √(2 n) and
    exactly linear decrease from the first entry to zero:
        a_i = C·(n‑1‑i)

    The constant C is chosen so that Σ a_i = √(2 n).
    """
    target = np.sqrt(2.0 * n)                     # required ℓ¹‑norm
    # weights (n‑1‑i)   →  (n‑1, n‑2, … , 0)
    w = np.arange(n - 1, -1, -1, dtype=float)
    sum_w = w.sum()                               # = n·(n‑1)/2
    a = target * w / sum_w
    # Numerical safety – projection guarantees exact norm and non‑negativity
    a = _project_to_simplex(a, target)
    return a

        print(f"Error calling evaluate_sequence: {exc}")

    # Return as a plain Python list (trailing zeros are already absent).
    return a.tolist()
