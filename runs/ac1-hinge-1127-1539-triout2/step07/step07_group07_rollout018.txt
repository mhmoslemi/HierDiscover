def _permutation_annealing_extended(a0, deadline):
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()
    while time.time() < deadline:
        move_type = rng.integers(0, 2) # 0 swap, 1 reverse
        if move_type == 0:
            i, j = rng.integers(0, n, size=2)
            if i == j: continue
            a[i], a[j] = a[j], a[i]
        else:
            start = rng.integers(0, n)
            length = rng.integers(2, min(50, n-start))
            a[start:start+length] = a[start:start+length][::-1]
        cur_score = _max_convolution_fft(a)
        if cur_score < best_score or rng.random() < np.exp(-(cur_score - best_score)/T):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            if move_type == 0:
                a[i], a[j] = a[j], a[i]
            else:
                a[start:start+length] = a[start:start+length][::-1]  # revert by reversing again
        T *= cooling
    return best

def _sgd_subgradient(a0, end_time):
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    lr = target * 0.1
    mu = 0.9
    v = np.zeros_like(a)
    it = 0
    while time.time() < end_time:
        # compute convolution
        size = 1 << ((2*n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        k = int(np.argmax(b))
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        i = np.arange(lo, hi+1)
        j = k - i
        g = np.zeros_like(a)
        g[i] = a[j]
        if k % 2 == 0:
            mid = k//2
            if lo <= mid <= hi:
                g[mid] = 2 * a[mid]
        # momentum update
        v = mu * v - lr * g
        a = a + v
        # project: clip >=0 and renorm to target sum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target / s)
        else:
            a = np.full(n, target / n)
        # anneal lr
        if it % 200 == 0 and lr > 1e-6:
            lr *= 0.95
        # jitter for escape
        if it % 500 == 0:
            jitter = np.random.randn(n) * target * 1e-4
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            a = a * (target / s) if s > 0 else a
        it += 1
    return a

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.  The algorithm consists of three phases:

1. **Fast parametric sweep** – generate a modest set of candidates from a
   few analytic families (power‑law, exponential, Gaussian, uniform random)
   for a few lengths (8000–12000).  For each base multiset we try a handful
   of orderings (sorted, reverse, alternating small–large and a few shuffles)
   and keep the best ``elite_cnt`` candidates.

2. **Iterative refinement** – each elite candidate is improved by a fast
   sub‑gradient descent that directly reduces the maximal convolution entry.
   The update uses a single FFT per iteration, momentum and a simple
   ℓ¹‑normalisation (projection onto the simplex Σa=√(2 n)).  After the
   gradient phase a short simulated‑annealing on permutations (swap and
   sub‑array reversal moves) is applied.

3. **Final ordering scan** – for the best sequence we exhaustively scan a
   large set of coprime‑step permutations (with optional offsets) to squeeze
   the last few hundredths of the objective.

All stages respect the overall time budget (default 1000 s).  The code
prints intermediate bounds and finally evaluates the returned sequence with
``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximal entry of the linear convolution a*a (full length 2·n‑1)
    using a power‑of‑two FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simple simplex projection (ℓ¹, Σx = s , x ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0 , Σ x_i = s} (Duchi et al.)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scaling helper
# ----------------------------------------------------------------------


def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Rescale a non‑negative vector so that its ℓ¹‑norm equals ``target``."""
    total = v.sum()
    if total == 0.0:
        v[:] = target / v.size
    else:
        v = v * (target / total)
    return v


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = (i + shift) ** (-p)
    return _scale_to_sum(raw, math.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = np.exp(-rate * (i + shift))
    return _scale_to_sum(raw, math.sqrt(2.0 * n))


def _gaussian_sequence(n: int, sigma_factor: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    sigma = sigma_factor * n
    raw = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return _scale_to_sum(raw, math.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Orderings (few deterministic & random permutations)
# ----------------------------------------------------------------------


def _candidate_orderings(base: np.ndarray, rng: np.random.Generator):
    """Return a small collection of orderings for ``base``."""
    n = base.size
    asc = np.sort(base)          # small → large
    desc = asc[::-1]             # large → small
    # alternating small–large (small first)
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for idx in range(n):
        if idx % 2 == 0:
            alt[idx] = asc[lo]
            lo += 1
        else:
            alt[idx] = asc[hi]
            hi -= 1
    out = [asc, desc, alt]
    # a few random shuffles
    for _ in range(3):
        sh = asc.copy()
        rng.shuffle(sh)
        out.append(sh)
    return out


# ----------------------------------------------------------------------
# Fast sub‑gradient descent on max‑convolution (projected onto simplex)
# ----------------------------------------------------------------------


def _sgd_max_convolution(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent that directly reduces ``max(a*a)``.
    Uses a single FFT per iteration and simple ℓ¹‑renormalisation.
    """
    a = a0.copy()
    n = a.size
    target = math.sqrt(2.0 * n)

    # hyper‑parameters
    lr = target * 0.12          # initial step size
    mu = 0.9                    # momentum
    v = np.zeros_like(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution (full)
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = rfft(a, n=size)
        b = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        # index of the current maximal entry
        k = int(np.argmax(b))

        # sub‑gradient of the selected entry
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = a[j]                     # derivative w.r.t. a_i
        if k % 2 == 0:
            mid = k // 2
            if lo <= mid <= hi:
                g[mid] = 2.0 * a[mid]   # diagonal term

        # momentum update and projection (simple renorm)
        v = mu * v - lr * g
        a = a + v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target / s)
        else:
            a[:] = target / n

        # learning‑rate schedule
        if it % 200 == 0 and lr > 1e-6:
            lr *= 0.94

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target * 5e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            if s > 0.0:
                a = a * (target / s)

        it += 1

    return a


# ----------------------------------------------------------------------
# Permutation simulated annealing (swap + sub‑array reversal)
# ----------------------------------------------------------------------


def _permutation_annealing_extended(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing on the ordering of a fixed multiset.
    Supported moves: pairwise swap and reversal of a short sub‑array.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature proportional to current max, slow cooling
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        move_type = rng.integers(0, 2)   # 0 = swap, 1 = reverse
        if move_type == 0:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            cur = _max_convolution_fft(a)
            dE = cur - best_score
            if cur < best_score or rng.random() < math.exp(-dE / T):
                if cur < best_score:
                    best_score = cur
                    best = a.copy()
            else:
                a[i], a[j] = a[j], a[i]   # revert
        else:
            start = rng.integers(0, n - 1)
            max_len = min(50, n - start)
            if max_len < 2:
                continue
            length = rng.integers(2, max_len + 1)
            segment = a[start:start + length].copy()
            a[start:start + length] = segment[::-1]
            cur = _max_convolution_fft(a)
            dE = cur - best_score
            if cur < best_score or rng.random() < math.exp(-dE / T):
                if cur < best_score:
                    best_score = cur
                    best = a.copy()
            else:
                # revert the reversal
                a[start:start + length] = segment[::-1]

        T *= cooling

    return best

    """

    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf
    elite = []                     # list of (score, array)
    elite_cnt = 12

    # 1. Coarse parametric sweep (few lengths)
    # ------------------------------------------------------------------

    p_vals = np.linspace(0.35, 0.85, 6)         # power‑law exponents
    shift_vals = np.linspace(0.1, 4.0, 6)       # power‑law shifts
    rate_vals = np.logspace(-4, -1.5, 6)        # exponential rates
    exp_shift_vals = np.linspace(0.0, 2.0, 4)   # exponential shifts
    sigma_factors = np.linspace(0.08, 0.30, 5)  # Gaussian widths (as fraction of n)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ----- power‑law family -----
        for p in p_vals:
            for sh in shift_vals:
                base = _powerlaw_sequence(n, p, sh)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj, best_seq = score, cand.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL best (pow) n={n} p={p:.3f} sh={sh:.2f} obj={best_obj:.6f}")
                    if len(elite) < elite_cnt or score < elite[-1][0]:
                        elite.append((score, cand))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:elite_cnt]

        # ----- exponential family -----
        for rate in rate_vals:
            for sh in exp_shift_vals:
                base = _exponential_sequence(n, rate, sh)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj, best_seq = score, cand.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL best (exp) n={n} rate={rate:.5e} sh={sh:.2f} obj={best_obj:.6f}")
                    if len(elite) < elite_cnt or score < elite[-1][0]:
                        elite.append((score, cand))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:elite_cnt]

        # ----- Gaussian family -----
        for sf in sigma_factors:
            base = _gaussian_sequence(n, sf)
            for cand in _candidate_orderings(base, rng):
                score = _max_convolution_fft(cand)
                if score < best_obj:
                    best_obj, best_seq = score, cand.copy()
                    print(f"[{time.time():.1f}] NEW GLOBAL best (gauss) n={n} sigma={sf:.2f} obj={best_obj:.6f}")
                if len(elite) < elite_cnt or score < elite[-1][0]:
                    elite.append((score, cand))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:elite_cnt]

        # ----- pure random samples (adds diversity) -----
        rnd_end = min(deadline, time.time() + 1.5)
        while time.time() < rnd_end:
            rand = rng.random(n)
            rand = _scale_to_sum(rand, target_sum)
            score = _max_convolution_fft(rand)
            if score < best_obj:
                best_obj, best_seq = score, rand.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL best (rand) n={n} obj={best_obj:.6f}")
            if len(elite) < elite_cnt or score < elite[-1][0]:
                elite.append((score, rand))
                elite.sort(key=lambda x: x[0])
                elite = elite[:elite_cnt]

    # ------------------------------------------------------------------
    # 2. Incorporate external construction if present
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        ext = np.array(height_sequence_1, dtype=float)
        n_ext = ext.size
        ext = _scale_to_sum(ext, math.sqrt(2.0 * n_ext))
        score = _max_convolution_fft(ext)
        if score < best_obj:
            best_obj, best_seq = score, ext.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL best (external) n={n_ext} obj={best_obj:.6f}")
        if len(elite) < elite_cnt or score < elite[-1][0]:
            elite.append((score, ext))
            elite.sort(key=lambda x: x[0])
            elite = elite[:elite_cnt]

    # ------------------------------------------------------------------
    # 3. Refinement of elite candidates (gradient + permutation)
    # ------------------------------------------------------------------
    elite.sort(key=lambda x: x[0])  # ensure ascending order

    for rank, (init_score, seed_seq) in enumerate(elite):
        if time.time() >= deadline - 5:  # keep a small buffer for final steps
            break

        n = seed_seq.size
        remaining = deadline - time.time()
        if remaining < 30:
            # not enough time for a full refinement – just evaluate and break
            cur = _max_convolution_fft(seed_seq)
            if cur < best_obj:
                best_obj, best_seq = cur, seed_seq.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL best (early stop) n={n} obj={best_obj:.6f}")
            break

        # allocate roughly 60 % of the remaining slice to gradient,
        # 40 % to permutation (both with hard caps)
        grad_time = min(30.0, remaining * 0.6)
        perm_time = min(20.0, remaining * 0.4)

        # ---- sub‑gradient descent (fast) ----
        refined = _sgd_max_convolution(seed_seq, time.time() + grad_time)

        # ---- permutation annealing (swap + reverse) ----
        refined = _permutation_annealing_extended(refined, time.time() + perm_time)

        # evaluate the refined candidate
        obj = _max_convolution_fft(refined)
        if obj < best_obj:
            best_obj, best_seq = obj, refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL best (refined #{rank}) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Exhaustive coprime‑step scan for the current best sequence
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        base_sorted = np.sort(best_seq)   # multiset unchanged
        # generate a fairly large set of coprime steps
        steps = [s for s in range(2, n) if math.gcd(s, n) == 1][:150]
        for step in steps:
            if time.time() >= deadline:
                break
            # no offset
            perm = np.empty_like(base_sorted)
            pos = 0
            for v in base_sorted:
                perm[pos] = v
                pos = (pos + step) % n
            cur = _max_convolution_fft(perm)
            if cur < best_obj:
                best_obj, best_seq = cur, perm.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL best (final step={step}) obj={best_obj:.6f}")

            # same step + random offset
            offset = rng.integers(0, n)
            pos = offset
            for v in base_sorted:
                perm[pos] = v
                pos = (pos + step) % n
            cur = _max_convolution_fft(perm)
            if cur < best_obj:
                best_obj, best_seq = cur, perm.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL best (final step={step}+off) obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback and final clean‑up
    # ------------------------------------------------------------------
    if best_seq is None:
        # uniform fallback (guaranteed feasible)
        n = 8000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # trim trailing near‑zero entries – cosmetic only
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # final verification with the provided evaluate_sequence()
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.1f}] evaluate_sequence raised an exception: {exc}")

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
