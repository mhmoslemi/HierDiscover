def gradient_descent_admm(a, deadline):
    a = a.copy()
    n = a.size
    target_sum = np.sqrt(2*n)
    beta = 0.1
    beta_max = 1e6
    beta_factor = 1.1
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        grad, b = _gradient_softmax(a, beta)
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        # learning rate scaling: maybe target_sum * 0.02
        lr = 0.1 * target_sum / (np.sqrt(v_hat)+eps)
        a = a - lr * m_hat
        a = np.maximum(a,0.0)
        a = _project_to_simplex(a, target_sum)
        # update beta
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
    return a
```

import time, numpy as np

def _next_pow2(x):
    return 1<<(x-1).bit_length()

def _max_convolution_fft(a):
    n = a.size
    size = _next_pow2(2*n-1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return float(conv.max())

def _full_convolution(a):
    n = a.size
    size = _next_pow2(2*n-1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

def _project_to_simplex(v, s):
    # Duchi algorithm (as above)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_softmax(a, beta):
    n = a.size
    size1 = _next_pow2(2*n-1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[:2*n-1]
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)
    # convolution a (rev) * p
    a_rev = a[::-1]
    conv_len = n + (2*n-1) - 1
    size2 = _next_pow2(conv_len)
    A_rev_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, b

def _adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.size
    s = np.sqrt(2.0*n)
    beta = 0.5
    beta_max = 1e6
    beta_factor = 1.15
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    # learning rate scalar
    base_lr = 0.5  # tune
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        lr = base_lr / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat
        # ensure nonnegative
        a = np.maximum(a, 0.0)
        # project to simplex
        a = _project_to_simplex(a, s)
        # anneal beta
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
    return a

        # find minimal contribution

        if i == j:

        b_new = b.copy()
        # linear changes: for all shifts k where i contributes: b[k] -= 2*delta*a[k-i]; i contributes to indices i .. i+n-1
        start_i = i

        # add from j

        k2j = 2*j
        kij = i + j

            b_new[k2j] += delta*delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta*delta
        new_max = b_new.max()
        # Metropolis acceptance
        if new_max < cur_max:
            accept = True
        else:
            # probability exp(-(new - cur)/T)
            accept = rng.random() < np.exp(-(new_max - cur_max) / T)
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_a = a.copy()
                # optional print for debugging
                #print(f"[{time.time():.1f}] hill climb new best: {best_score:.6f}")
        # cool temperature
        T *= cooling
    # return best found
    return best_a, b

"""
Propose a step‑function minimizing ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Projected Adam optimisation** – minimise a smooth log‑sum‑exp
   surrogate of the worst autocorrelation entry on the ℓ₁‑simplex
   (Σa = √(2 n)).  The inverse temperature β is annealed up to a
   large value, so the surrogate becomes the true maximum.

2. **Mass‑transfer hill‐climbing** – a simulated‑annealing style
   local search that moves a small amount of mass from an index
   contributing heavily to the current worst convolution entry towards
   an index that adds little or no contribution there.  The full
   self‑convolution is kept up‑to‑date in O(n) per move.

3. **Final polishing** – a short Adam run with the remaining time.

The routine adapts the sequence length to the supplied wall‑clock
budget (up to 60 000 entries) and respects all constraints enforced by
``evaluate_sequence``.  It returns the best sequence found before the
deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p),   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    A_rev_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected optimisation
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    After each step the vector is projected onto the ℓ₁‑simplex
    {a_i ≥ 0, Σa_i = √(2 n)}.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 1e6
    beta_factor = 1.15

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    base_lr = 0.5

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # per‑coordinate learning‑rate (AdaGrad‑style scaling)
        lr_vec = base_lr / (np.sqrt(v_hat) + eps)
        a -= lr_vec * m_hat

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing hill climbing that transfers mass from a source
    index (chosen proportionally to its contribution to the current
    worst convolution entry) to a destination index that adds as little
    as possible to that entry.  The full convolution is updated in O(n)
    per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_b = b.copy()
    best_score = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.12 + 1e-9          # initial temperature
    cooling = 0.9994

    while time.time() < deadline:
        # worst convolution index (random tie‑break)
        max_idxs = np.flatnonzero(b == cur_max)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source candidates that actually contribute to M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - i_range
        contrib = a[i_range] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(i_range, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] <= 1e-14:
            continue

        # destination that minimises the added contribution to entry M
        arr = np.zeros(n, dtype=float)           # contribution of each j to M
        valid_lo = max(0, M - (n - 1))
        valid_hi = min(n - 1, M)
        j_vals = np.arange(valid_lo, valid_hi + 1, dtype=int)
        arr[j_vals] = a[M - j_vals]              # a[M‑j] where defined
        min_contrib = arr.min()
        cand = np.flatnonzero(np.isclose(arr, min_contrib, atol=1e-12))
        if cand.size == 0:
            j = rng.integers(0, n)
        else:
            j = rng.choice(cand)

        if i == j:
            continue

        max_delta = a[i] * 0.85
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- linear update of the convolution --------------------------------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # ---- quadratic correction terms ---------------------------------------
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- Metropolis acceptance --------------------------------------------
        if new_max < cur_max:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_max - cur_max) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_a = a.copy()
                best_b = b.copy()

        T *= cooling

    return best_a, best_b
