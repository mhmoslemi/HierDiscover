def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5
    # Acquire base sequence
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        base = _scale_to_sum(base, np.sqrt(2.0 * n))
    else:
        n = 8000
        # use powerlaw with tuned params
        base = _powerlaw_sequence(n, 0.5, 0.5)
    # initial evaluate
    best_seq = base.copy()
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.1f}] Starting n={n}, initial score={best_score:.6f}")
    # allocate times
    time_left = deadline - time.time()
    if time_left <= 0:
        return best_seq.tolist()
    # Gradient part
    grad_time = time_left * 0.7
    grad_deadline = time.time() + grad_time
    print(f"[{time.time():.1f}] Gradient phase for {grad_time:.1f}s")
    refined = _adam_gradient_descent(best_seq, grad_deadline)
    refined_score = _max_convolution_fft(refined)
    print(f"[{time.time():.1f}] After gradient score={refined_score:.6f}")
    if refined_score < best_score:
        best_seq, best_score = refined, refined_score
    # Permutation phase
    time_left = deadline - time.time()
    if time_left > 0:
        perm_deadline = time.time() + time_left
        print(f"[{time.time():.1f}] Permutation phase for {time_left:.1f}s")
        permuted = _permutation_annealing(best_seq, perm_deadline)
        perm_score = _max_convolution_fft(permuted)
        print(f"[{time.time():.1f}] After permutation score={perm_score:.6f}")
        if perm_score < best_score:
            best_seq, best_score = permuted, perm_score
    # Final exhaustive coprime steps many tries
    time_left = deadline - time.time()
    if time_left > 0:
        print(f"[{time.time():.1f}] Final fine scan ({time_left:.1f}s left)")
        base_sorted = np.sort(best_seq)
        n = base_sorted.size
        steps_tested = 0
        while time.time() < deadline and steps_tested < 1000:
            step = rng.integers(2, n)
            if np.gcd(step, n) != 1:
                continue
            # no offset
            pos = 0
            perm = np.empty_like(base_sorted)
            for v in base_sorted:
                perm[pos] = v
                pos = (pos + step) % n
            cur = _max_convolution_fft(perm)
            if cur < best_score:
                best_score, best_seq = cur, perm.copy()
                print(f"[{time.time():.1f}] New best (step {step}) score={best_score:.6f}")
            # random offset
            offset = rng.integers(0, n)
            pos = offset
            for v in base_sorted:
                perm[pos] = v
                pos = (pos + step) % n
            cur = _max_convolution_fft(perm)
            if cur < best_score:
                best_score, best_seq = cur, perm.copy()
                print(f"[{time.time():.1f}] New best (step {step}+off) score={best_score:.6f}")
            steps_tested += 1
    # Trim near-zero entries
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[:nz[-1]+1]
    # Evaluate
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {ev:.6f}")
    except Exception as e:
        print("Exception during evaluation:", e)
    print(f"[{time.time():.1f}] Finished. Best objective = {best_score:.6f}, length = {best_seq.size}")
    return best_seq.tolist()

"""
propose_candidate
-----------------
A focused optimisation that starts from the best known construction (or a
simple power‑law if no external solution is available), normalises the
ℓ¹‑norm to √(2 n) and then performs a long projected‑Adam stage on a
smooth log‑sum‑exp surrogate of the maximal auto‑convolution entry.
The bulk of the remaining time is spent in an aggressive simulated‑annealing
permutation search (swap moves, occasional global reshuffles) and a final
exhaustive scan of coprime‑step permutations (with random offsets).  The
whole routine respects a 1000 s wall‑clock budget and prints intermediate
scores for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s , x ≥ 0
# (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ_k exp(β·b_k) where b = a*a (full conv).

    Returns
        g  – gradient w.r.t a (same shape)
        b  – full convolution (length 2·n‑1)
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)               # shape (2n‑1,)

    # gradient: 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation of the smoothed max‑convolution.
    Runs until ``deadline`` (unix time).  Returns the final vector.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0          # temperature of the soft‑max
    max_beta = 1e6
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update (biased moments)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate adapted to the current temperature
        lr = 0.5 * target_sum / ((beta + 1.0) * (np.sqrt(v_hat).mean() + 1e-8))

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex Σa = √(2 n)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal the temperature
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Permutation annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing on the ordering of a fixed multiset.
    Returns the best ordering found before ``deadline``.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # start temperature proportional to current peak
    T = best_score * 0.05 + 1e-9
    cooling = 0.9998
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        a[i], a[j] = a[j], a[i]
        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny global reshuffle (2 % of entries)
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                tmp = a[mask].copy()
                rng.shuffle(tmp)
                a[mask] = tmp
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Simple parametric families (only power‑law needed for the initial sweep)
# ----------------------------------------------------------------------


def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale non‑negative vector v so that Σv = target."""
    scale = target / np.sum(v)
    a = v * scale
    return np.maximum(a, 0.0)


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i+shift)^{-p}, normalised to Σa = √(2 n)."""
    i = np.arange(n, dtype=float)
    raw = (i + shift) ** (-p)
    return _scale_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a non‑negative height sequence that (approximately) minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Global deadline (tiny safety margin)
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 0.2

    # --------------------------------------------------------------
    # 1. Initialise a good starting multiset
    # --------------------------------------------------------------
    candidates = []

    # external construction, if available
    if 'height_sequence_1' in globals():
        ext = np.array(height_sequence_1, dtype=float)
        n_ext = ext.size
        ext = _scale_to_sum(ext, np.sqrt(2.0 * n_ext))
        candidates.append(ext)

    # a few hand‑chosen power‑law shapes (very cheap to evaluate)
    n_default = 8000
    for p in (0.45, 0.5, 0.55):
        for shift in (0.2, 0.4, 0.6, 1.0, 2.0):
            cand = _powerlaw_sequence(n_default, p, shift)
            candidates.append(cand)

    # pick the best multiset (ignoring ordering for the moment)
    best_seq = None
    best_score = np.inf
    for base in candidates:
        score = _max_convolution_fft(base)   # descending order (default)
        if score < best_score:
            best_score = score
            best_seq = base.copy()
        # also test the reverse ordering (ascending)
        rev = base[::-1]
        score_rev = _max_convolution_fft(rev)
        if score_rev < best_score:
            best_score = score_rev
            best_seq = rev.copy()

    print(f"[{time.time():.1f}] Initial multiset selected, n={best_seq.size}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Gradient refinement (projected Adam on the smoothed max)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left <= 0:
        return best_seq.tolist()

    grad_time = time_left * 0.65               # allocate ~65 % for gradient
    grad_deadline = time.time() + grad_time
    print(f"[{time.time():.1f}] Gradient phase – {grad_time:.1f}s")
    refined = _adam_gradient_descent(best_seq, grad_deadline)
    refined_score = _max_convolution_fft(refined)
    print(f"[{time.time():.1f}] After gradient, score={refined_score:.6f}")
    if refined_score < best_score:
        best_seq, best_score = refined, refined_score

    # --------------------------------------------------------------
    # 3. Permutation annealing (heavy simulated‑annealing)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        perm_deadline = time.time() + time_left
        print(f"[{time.time():.1f}] Permutation phase – {time_left:.1f}s")
        permuted = _permutation_annealing(best_seq, perm_deadline)
        perm_score = _max_convolution_fft(permuted)
        print(f"[{time.time():.1f}] After permutation, score={perm_score:.6f}")
        if perm_score < best_score:
            best_seq, best_score = permuted, perm_score

    # --------------------------------------------------------------
    # 4. Final exhaustive scan of coprime‑step permutations
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        print(f"[{time.time():.1f}] Final coprime‑step scan (≈{time_left:.1f}s left)")
        base_sorted = np.sort(best_seq)      # preserve multiset
        n = base_sorted.size
        steps_tried = 0
        # try up to a few hundred coprime steps (or until time runs out)
        while time.time() < deadline and steps_tried < 500:
            step = rng.integers(2, n)
            if np.gcd(step, n) != 1:
                continue
            # no offset
            perm = np.empty_like(base_sorted)
            pos = 0
            for v in base_sorted:
                perm[pos] = v
                pos = (pos + step) % n
            cur = _max_convolution_fft(perm)
            if cur < best_score:
                best_score, best_seq = cur, perm.copy()
                print(f"[{time.time():.1f}] New best (step {step}) -> {best_score:.6f}")
            # same step with a random offset
            offset = rng.integers(0, n)
            perm_off = np.empty_like(base_sorted)
            pos = offset
            for v in base_sorted:
                perm_off[pos] = v
                pos = (pos + step) % n
            cur = _max_convolution_fft(perm_off)
            if cur < best_score:
                best_score, best_seq = cur, perm_off.copy()
                print(f"[{time.time():.1f}] New best (step {step}+off) -> {best_score:.6f}")
            steps_tried += 1

    # --------------------------------------------------------------
    # 5. Minor clean‑up and final reporting
    # --------------------------------------------------------------
    # Trim leading zeros (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Final evaluation with the official function (should equal best_score)
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {ev:.6f}")
    except Exception as e:
        print("Evaluation error:", e)

    print(
        f"[{time.time():.1f}] Finished – best objective = {best_score:.6f}, length = {best_seq.size}"
    )
    return best_seq.tolist()
