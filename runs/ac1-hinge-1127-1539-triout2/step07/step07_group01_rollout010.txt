def _alternating_lp(a, max_iter=20, rng=None):
    """
    Alternating linear‑programming refinement.
    Starting from vector a (already ℓ₁‑normalised), repeatedly:
      * solve maximise sum(b) subject to b>=0 and conv(a,b) ≤ M,
        where M = max(conv(a,a));
      * rescale b to the required ℓ₁‑norm,
      * update a ← (1‑t)*a + t*b with a small step size.
    Returns the final a and its max‑convolution value.
    """
    import cvxpy as cp
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    t = 0.2
    for it in range(max_iter):
        M = _max_convolution(a)
        # Build constraint matrix
        # A shape (2n-1, n)
        rows = 2 * n - 1
        A = np.zeros((rows, n))
        for i in range(n):
            A[i:i + n, i] = a[i]
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        if b_var.value is None:
            break
        b_opt = np.maximum(b_var.value, 0.0)
        sum_b = b_opt.sum()
        if sum_b == 0:
            break
        b_opt = b_opt * (S / sum_b)
        a = (1 - t) * a + t * b_opt
        # optionally add random jitter for diversity
        if rng is not None and it % 5 == 0:
            a += rng.normal(scale=0.01, size=n)
            a = _project_to_simplex(a, S)
    M_final = _max_convolution(a)
    return a, M_final

"""
Hierarchical step‑function optimizer.

The search proceeds in three stages:

1. **Fast random exploration** – many candidates are drawn from a handful of
   parametric families (power‑law, exponential, triangular, Gamma, uniform,
   Dirichlet).  The best candidate (measured by the internal max–convolution)
   is kept.

2. **Linear‑program alternating refinement** (optional) – a very small version
   of the current best sequence is improved by repeatedly solving a linear
   program that maximises the ℓ₁‑norm of a partner function while keeping the
   convolution bound unchanged.  The result is up‑scaled back to the original
   length.

3. **Progressive projected‑gradient descent** – the (possibly refined) best
   sequence is up‑scaled to larger lengths and optimised with a projected
   gradient method on the simplex.  The gradient of the exact max‑convolution
   is computed via an FFT‑based soft‑max (temperature τ → 0).  The step size
   and τ are annealed; occasional jitter helps escape plateaus.

All stages stop before the overall time budget expires (default 1000 s).
The final sequence is stored in the global variable ``height_sequence_1`` and
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum value of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Sub‑gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with
        grad = ∂ τ‑soft‑max(b) / ∂a .
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, Σ x_i = S }.
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fallback to uniform distribution
        theta = (np.sum(v) - S) / n
    else:
        rho = ind[cond][-1] - 1          # zero‑based index
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _pgd_optimize(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the simplex.
    Minimises the exact max‑convolution (τ‑soft‑max with τ → 0).

    Returns the best sequence found and its hard‑max peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    a = a0.copy()
    # initial temperature – a fraction of the current peak
    tau = max(_max_convolution(a) * 0.5, 0.5)

    # learning rate – start moderate, then decay
    lr = 0.5
    best_a = a.copy()
    best_peak = _max_convolution(a)

    it = 0
    while time.time() < deadline:
        it += 1

        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # step size clipping to avoid too large jumps
        step = lr * g
        step_norm = np.linalg.norm(step)
        max_step = S * 0.4
        if step_norm > max_step and step_norm > 0:
            step = step * (max_step / step_norm)

        a_new = a - step
        a = _project_onto_simplex(a_new, S)

        # schedules
        tau *= 0.9985
        lr *= 0.9997

        # occasional jitter to shake out of plateaus
        if it % 800 == 0:
            a += rng.normal(scale=0.008, size=n)
            a = _project_onto_simplex(a, S)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Optional alternating linear‑program refinement (small scale)
# ----------------------------------------------------------------------
def _alternating_lp(a: np.ndarray,
                    max_iter: int = 20,
                    rng: np.random.Generator | None = None) -> tuple[np.ndarray, float]:
    """
    Alternating LP refinement described in the problem statement.
    Works on a fairly small vector (few hundred entries) because the LP
    size grows quadratically with n.
    Returns the refined vector and its max‑convolution value.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        raise RuntimeError("cvxpy not available") from e

    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    t = 0.18  # mixing factor

    for it in range(max_iter):
        M = _max_convolution(a)

        # construct convolution matrix A (size (2n-1) × n)
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=float)
        for i in range(n):
            A[i:i + n, i] = a[i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)

        if b_var.value is None:
            break
        b_opt = np.maximum(b_var.value, 0.0)
        sum_b = b_opt.sum()
        if sum_b == 0.0:
            break

        # rescale to required ℓ₁‑norm
        b_opt = b_opt * (S / sum_b)

        # convex combination
        a = (1 - t) * a + t * b_opt

        if rng is not None and (it + 1) % 5 == 0:
            a += rng.normal(scale=0.005, size=n)
            a = _project_onto_simplex(a, S)

    final_M = _max_convolution(a)
    return a, final_M

# Parametric families for random initialisation

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)          # exponent
    offset = rng.uniform(0.0, 15.0)    # denominator shift
    mix = rng.uniform(0.0, 1.0)        # uniform mixing weight
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)   # decay rate
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n) * rng.uniform(0.0, 0.2)
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Raw Dirichlet‑like start."""
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best and otherwise samples
    from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a
