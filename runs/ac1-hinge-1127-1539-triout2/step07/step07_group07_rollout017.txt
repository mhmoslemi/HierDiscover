def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution via FFT
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_b = b.max()
        # find indices
        # Use tolerance of 0 (exact) or small epsilon?
        # Compute mask by equality (floating rounding) may cause few indices.
        mask = np.isclose(b, max_b, atol=1e-12, rtol=0)
        idxs = np.nonzero(mask)[0]
        # compute gradient
        g = np.zeros_like(a)
        # add |M|*a
        g += len(idxs) * a
        rev = a[::-1]
        for k in idxs:
            if k < n:
                # slice length k+1
                g[:k+1] += a[:k+1][::-1]
            else:
                lo = k - n + 1
                seg_len = n - lo
                # rev[:seg_len] corresponds to a[n-1 .. lo]
                g[lo:] += rev[:seg_len]
        # step size schedule
        # Determine step length: maybe use step = step0 / sqrt(it+1)
        step0 = target_sum * 0.5
        lr = step0 / np.sqrt(it+1)
        # Apply
        a -= lr * g
        # project
        a = _project_to_simplex(a, target_sum)
        # jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (np.sqrt(it+1)+1), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
        it += 1
    return a

"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  The algorithm is based on a fast sub‑gradient
descent on the *max* of the linear convolution a*a (the quantity that
determines the objective after the L¹‑normalisation Σa = √(2 n)).  It
repeatedly:
  1. builds a random or V‑shaped seed of a given length,
  2. runs a projected sub‑gradient descent that moves mass away from
     the currently maximal convolution entry,
  3. keeps the best candidate found,
  4. finishes with a short simulated‑annealing swap phase to polish the
     ordering.

All steps respect the supplied time budget (default 1000 s).  The
implementation uses only NumPy and pure Python, works on a single CPU
and returns a Python list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑two length."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return the maximal entry of a*a (full convolution)."""
    return float(_conv_full_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0
# (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on max (a*a)  (projected onto the simplex)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent that directly minimises the peak
    entry of the convolution a*a.  The update direction is the
    sub‑gradient of max(b) where b = a*a.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    it = 0
    rng = np.random.default_rng()
    # tiny constant to avoid division by zero
    eps = 1e-12

    while time.time() < deadline:
        # full convolution and maximal entry
        b = _conv_full_fft(a)
        max_b = b.max()
        # indices attaining the maximum (within a tiny tolerance)
        max_mask = np.isclose(b, max_b, atol=1e-12, rtol=0.0)
        max_idx = np.nonzero(max_mask)[0]

        # ---- sub‑gradient of max_b ---------------------------------
        g = np.zeros_like(a)
        # contribution |M| * a
        g += len(max_idx) * a

        rev = a[::-1]
        for k in max_idx:
            if k < n:
                # j = 0 … k   →   a_{k-j} = a[:k+1][::-1]
                g[: k + 1] += a[: k + 1][::-1]
            else:
                # k ≥ n, valid j are lo … n‑1 where lo = k‑n+1
                lo = k - n + 1
                seg_len = n - lo
                # a_{k-j} for j∈[lo,n‑1] equals rev[:seg_len]
                g[lo:] += rev[:seg_len]

        # ---- step‑size ------------------------------------------------
        # Use a diminishing step proportional to 1/√it and normalised
        # by the gradient norm to obtain a roughly constant L¹‑move.
        g_norm = np.linalg.norm(g) + eps
        base_lr = 0.02 * target_sum                     # base magnitude
        lr = base_lr / g_norm / np.sqrt(it + 1.0 + eps)  # diminishing

        # gradient step and projection
        a -= lr * g
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat plateaus
        if it % 200 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (it + 1), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Light weight simulated‑annealing on the ordering (swap moves)
# ----------------------------------------------------------------------


def _swap_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Very cheap simulated‑annealing that only swaps entries.
    The L¹‑norm is invariant; the best ordering found before the
    deadline is returned.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9   # initial temperature
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a swap
        a[i], a[j] = a[j], a[i]
        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Parametric families for initial seeds
# ----------------------------------------------------------------------


def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ¹‑norm = target."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _vshape_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    Symmetric V‑shape: a_i = (min(i, n‑1‑i) + shift)^{‑p}.
    Large values are placed at the two ends, small values in the centre.
    """
    i = np.arange(n, dtype=float)
    d = np.minimum(i, n - 1 - i)
    raw = (d + shift) ** (-p)
    return _scale_to_sum(raw, np.sqrt(2.0 * n))


def _random_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random positive vector, ℓ¹‑normalised."""
    raw = rng.random(n)
    return _scale_to_sum(raw, np.sqrt(2.0 * n))


def _linear_falling_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    """Truncated linear decreasing from the left side."""
    i = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * i)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _scale_to_sum(raw, np.sqrt(2.0 * n))

# Main entry point
