"""
Propose a step‑function that minimises the maximum entry of its self‑convolution.

The algorithm works in three phases for a sequence of increasing lengths:

1. **Initialisation** – random draws from several families (power‑law,
   exponential, Dirichlet, sine, Gaussian, piecewise) and linear up‑sampling
   of the best sequence found so far.

2. **Projected (soft‑max) gradient descent** – minimise a smooth log‑sum‑exp
   surrogate of the peak of the convolution.  The inverse temperature β is
   annealed to a very large value so that the surrogate becomes the true
   maximum.  The step size is kept independent of β, and after each step the
   vector is projected onto the ℓ¹‑simplex {aᵢ≥0, Σaᵢ=√(2n)}.

3. **Deterministic hill‑climbing** – repeatedly moves a fraction of mass
   from the index that contributes most to the current peak (obtained from
   the gradient of the max‑indicator) to one of the smallest entries.
   The full convolution is updated in O(n) per move, and a move is kept only
   if it strictly reduces the peak.

The two phases are interleaved for each length, the remaining time budget
is spent on a final high‑β gradient descent.  All vectors stay non‑negative
and are finally trimmed of near‑zero tails before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto {x_i ≥ 0, Σx_i = s}.
    Duchi et al., 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient utilities (soft‑max surrogate of max)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, b) where b = a*a (full convolution) and
    grad = 2·(a ⋆ p) with p_i ∝ exp(β·b_i) (log‑sum‑exp gradient).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_from_distribution(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient 2·(a ⋆ p) for an arbitrary non‑negative distribution p of length 2n‑1.
    """
    n = a.size
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad

# Projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _grad_desc_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent that drives max(a*a) down.
    Uses a log‑sum‑exp surrogate with annealed β.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 5e5
    beta_factor = 1.3

    # step size coefficient (tuned empirically)
    step_coeff = 0.45 * target_sum

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        norm = np.linalg.norm(grad)
        if norm > 0.0:
            lr = step_coeff / norm
            a -= lr * grad

        a = _project_to_simplex(a, target_sum)

        # increase β every few iterations
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine half‑wave
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

    b = b0.copy()

    best_a = a.copy()

                if new_max < best_score:
                    best_score = new_max
                    best_a = a.copy()
                improved = True
                break

        if not improved:
            # No improving destination found – stop early.
            break

    return best_a, b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seeding from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – progressively larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000, 60000]

    for n in schedule:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase 1 – gradient descent (major share of the stage budget)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 6.0:
            break
        gd_deadline = time.time() + remaining * 0.55
        a_gd = _grad_desc_max(a_init, gd_deadline)
        score_gd = _max_convolution_fft(a_gd)
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement n={n} score={score_gd:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – deterministic hill climbing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 6.0:
            break
        hc_deadline = time.time() + remaining * 0.40
        a_cur = best_seq.copy() if best_seq.size == n else _upsample_sequence(best_seq, n)
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)
        a_hc, b_hc = _hill_climb_targeted(a_cur, b_cur, hc_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement n={n} score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Optional short refinement by gradient descent
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            refine_deadline = time.time() + min(remaining * 0.05, 20.0)
            a_ref = _grad_desc_max(best_seq.copy(), refine_deadline)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Post‑HC refinement n={n} score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        a_final = _grad_desc_max(best_seq.copy(), deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
