def _exp_gradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    beta = 1.0
    beta_max = 1e6
    beta_factor = 2.0
    it = 0
    # momentum not needed for EG
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)  # gradient on smoothed max
        # compute step size: we want to diminish a_i with large gradient
        # Choose step = eta * g, where eta chosen as 0.5 / (np.max(g)+1e-12)
        # Could adapt; for now constant base
        eta = 0.5 / (beta + 1.0)  # scaling with beta
        # update multiplicatively
        a = a * np.exp(-eta * g)
        # renormalize to target sum
        total = a.sum()
        if total > 0:
            a = a * (target_sum / total)
        else:
            # reinitialize uniform
            a = np.full_like(a, target_sum / n)
        # increase beta gradually
        if it % 100 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
        it += 1
    return a

"""
Propose a step‑function (non‑negative heights) whose self‑convolution
maximum is as small as possible.  The algorithm works in three stages:

1.  **Exponentiated‑gradient descent** on a smooth approximation of the
    maximal convolution entry.  The gradient of the soft‑max
    (log‑sum‑exp) of the convolution is computed by FFT, and the
    probabilities are used to form a multiplicative update
    a←a·exp(‑η·∇).  After each update the vector is re‑scaled to the
    ℓ¹‑simplex Σa=√(2 n).  β (the soft‑max temperature) is increased
    geometrically, so the optimisation converges to the true max.

2.  **Refined hill‑climbing** that moves a random amount of mass from a
    source index to a destination index while keeping the full
    convolution up‑to‑date in O(n) per move.  The source is chosen
    proportionally to its contribution to the currently worst
    convolution entry, the destination is biased towards indices with
    little mass, and a Metropolis step with a slowly cooling temperature
    accepts occasional worsening moves.

3.  **Iterative length schedule** – we repeatedly up‑sample the current
    best vector to larger lengths (2000 → 4000 → 8000 → 16000) and repeat
    the two optimisation phases, always keeping the globally best
    sequence.  The remaining time is spent on a final polishing run.

All operations respect the 1000 s wall‑clock budget; the code prints
progress and the final objective value returned by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean – used only for occasional jitter)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth maximum (log‑sum‑exp) of the convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full linear convolution).

    Returns (g, b) where
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)

    # ----- forward convolution (b) -----
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities -----
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # ----- gradient (2·a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the smooth max
# ----------------------------------------------------------------------


def _exp_gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Multiplicative (exponentiated) projected gradient descent on the
    smoothed maximum of the self‑convolution.  The vector is kept on the
    ℓ¹‑simplex Σa=√(2 n) after every update.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    beta = 2.0                # start temperature
    beta_max = 1e5           # final temperature (sharp max)
    beta_factor = 1.5
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # normalise the step size; we shrink the largest component at most
        # by a factor of ≈exp(-0.5)
        maxg = np.max(g)
        eta = 0.5 / (maxg + 1e-12)

        # multiplicative update (keeps a ≥ 0)
        a = a * np.exp(-eta * g)

        # re‑scale to the simplex (preserves the sum exactly)
        s = a.sum()
        if s > 0.0:
            a = a * (target / s)
        else:                     # numerical safety – fall back to uniform
            a = np.full_like(a, target / n)

        # increase the soft‑max temperature periodically
        if it % 100 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are
    sampled to provide diverse seeds.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing (simulated annealing) with exact O(n) updates
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # geometric cooling

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # pick a convolution index among the largest values (to focus effort)
        # ------------------------------------------------------------------
        threshold = cur_score * (0.9 + 0.1 * rng.random())   # a little random wiggle
        candidates = np.flatnonzero(b >= threshold)
        if candidates.size == 0:
            M = rng.integers(0, b.size)
        else:
            M = rng.choice(candidates)

        # ------------------------------------------------------------------
        # source index i – weighted by its contribution to entry M
        # ------------------------------------------------------------------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:                           # fall back to uniform selection
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # ------------------------------------------------------------------
        # destination index j – bias towards positions with little mass
        # ------------------------------------------------------------------
        if rng.random() < 0.5:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # ------------------------------------------------------------------
        # amount of mass to move (up to 80 % of source)
        # ------------------------------------------------------------------
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # apply the move (in‑place update of b)
        # ------------------------------------------------------------------
        b[i:i + n] -= 2.0 * delta * a
        b[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b.size:
            b[k2i] += delta * delta
        if k2j < b.size:
            b[k2j] += delta * delta
        if kij < b.size:
            b[kij] -= 2.0 * delta * delta

        new_score = b.max()

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            a[i] -= delta
            a[j] += delta
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")
        else:
            # revert the convolution update (a unchanged)
            b[i:i + n] += 2.0 * delta * a
            b[j:j + n] -= 2.0 * delta * a
            if k2i < b.size:
                b[k2i] -= delta * delta
            if k2j < b.size:
                b[k2j] -= delta * delta
            if kij < b.size:
                b[kij] += 2.0 * delta * delta

        T *= cooling

    return best_a, b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1.  Seed from any pre‑computed construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = cand.size
            cand = _project_to_simplex(cand, np.sqrt(2.0 * n0))
            val = evaluate_sequence(cand.tolist())
            best_seq = cand.copy()
            best_val = val
            print(f"[{time.time():.1f}] seeded from height_sequence_1 (n={n0}) val={val:.6f}")
        except Exception as e:
            print(f"seeding error: {e}")

    # ------------------------------------------------------------------
    # 2.  Length schedule – gradually increase the dimension
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        # --------------------------------------------------------------
        # Phase A – exponentiated gradient descent (several restarts)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 10.0:
            break
        gd_budget = min(0.35 * remaining, 320.0)   # allocate a decent slice
        gd_deadline = time.time() + gd_budget
        restart_cnt = 4
        local_best = None
        local_best_val = np.inf

        for r in range(restart_cnt):
            if time.time() >= gd_deadline:
                break
            # split the remaining GD budget fairly among the restarts
            per_restart = min(gd_deadline - time.time(),
                             (gd_budget / (restart_cnt - r)))
            restart_deadline = time.time() + per_restart

            a_start = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_opt = _exp_gradient_descent_opt(a_start, restart_deadline)

            # quick objective (max of convolution) – sum is already normalised
            val = _max_convolution_fft(a_opt)

            if val < local_best_val:
                local_best_val = val
                local_best = a_opt.copy()
                print(f"[{time.time():.1f}] GD restart {r+1}/{restart_cnt} n={n} val={val:.6f}")

        # keep the best from the GD phase
        if local_best is not None and local_best_val < best_val:
            best_val = local_best_val
            best_seq = local_best.copy()
            print(f"[{time.time():.1f}] GD improvement for n={n} val={local_best_val:.6f}")

        # --------------------------------------------------------------
        # Phase B – hill climbing with exact convolution updates
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 5.0:
            break
        hill_budget = min(0.25 * remaining, 260.0)
        hill_deadline = time.time() + hill_budget

        # ensure we work with the current length
        if best_seq is None or best_seq.size != n:
            a_cur = _upsample_sequence(best_seq, n) if best_seq is not None else a_init
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target)
        b_full = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_full, hill_deadline)
        val_hc = b_hc.max()                     # sum already normalised
        if val_hc < best_val:
            best_val = val_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} val={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final polishing on the best sequence (as much time as left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = np.sqrt(2.0 * n)

        # a) final exponentiated gradient descent
        gd_deadline = time.time() + min(120.0, deadline - time.time() - 80.0)
        a_final = _exp_gradient_descent_opt(best_seq.copy(), gd_deadline)

        # b) final hill climbing
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final,
                                          deadline - time.time() - 2.0)

        final_val = b_final.max()
        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: val={final_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Trim trailing near‑zero entries for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5.  Official evaluation
    # ------------------------------------------------------------------
    try:
        obj = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {obj:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error during final evaluation: {exc}")

    return best_seq.tolist()
