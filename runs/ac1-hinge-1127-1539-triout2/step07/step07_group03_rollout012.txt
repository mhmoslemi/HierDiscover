"""
Propose a step‑function (non‑negative sequence) with a low benchmark score.

The algorithm follows four stages:

1. **Shape search** – try simple parametric families (power‑law and exponential)
   and keep the best‐scoring sequence.  The score is the exact benchmark
   objective (no `evaluate_sequence` call needed).

2. **Projected gradient descent with restarts** – a soft‑max approximation of
   the maximum convolution entry is optimised with Adam‑like momentum.
   The temperature (β) is increased aggressively up to 1 e⁶.

3. **Simulated annealing** – mass‑transfer / smoothing / swap moves are
   performed while a temperature schedule slowly cools to a very low value.
   The move set is identical to the previous state‑of‑the‑art code but the
   schedule is a little more aggressive.

4. **Sub‑gradient refinement** – once the SA phase ends a pure
   sub‑gradient step on the (non‑smooth) max‑convolution is applied with a
   decreasing step size, followed by a final short projected‑gradient pass.

The whole routine respects the 1000 s time budget, chooses a problem size
(`n`) depending on the allocated time and always returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, length 2·n‑1)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective & projection utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Exact benchmark objective 2·n·max(a*a) / (∑a)²."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, r: float, target_sum: float) -> np.ndarray:
    """Create a geometric sequence r**i (0<r<1) and normalise to target_sum."""
    a = r ** np.arange(n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient descent with a soft‑max approximation of max()
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)

    # full convolution a*a
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution

    # gradient = 2 * (a_rev) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation.
    Uses a large β‑schedule (up to 1 e⁶) and Adam‑like momentum.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1_000_000.0          # more aggressive than the reference code
    step_base = target_sum * 0.8   # slightly larger steps
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9                    # stronger momentum
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:  # jitter to shake out of plateaus
            jitter = rng.normal(scale=target_sum * 0.003 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Run several short GD restarts with random jitter; keep the best result.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)   # each restart ≤30 s
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                                time.time() + run_budget,
                                                target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,

        # index of the current maximum convolution entry
        k_max = int(np.argmax(conv))

        # sensitivities (derivative of the max entry w.r.t. each a_i)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # ------------------- mass transfer -------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # new convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pure sub‑gradient refinement of max(convolution)
# ----------------------------------------------------------------------


def _max_subgradient_refine(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the *exact* max‑convolution objective.
    Decreases step size like 1/√t.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k = int(conv.argmax())
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # decreasing step size
        lr = target_sum * 0.4 / np.sqrt(it + 1.0)
        a -= lr * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with a low benchmark score.
    The routine adapts the problem size to the time budget,
    explores a few parametric families, runs aggressive projected
    gradient descent, performs simulated annealing and finishes
    with a pure sub‑gradient refinement.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose a problem size based on how much time we have.
    # Larger n gives a better theoretical lower bound but costs a bit
    # more per iteration.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1) Quick parametric shape search
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # power‑law family
    for p in np.linspace(0.30, 1.20, 21):
        seq = _power_law_sequence(n, p, target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] power‑law p={p:.3f} → score={val:.6f}")

    # exponential family
    for r in np.linspace(0.85, 0.985, 21):
        seq = _exponential_sequence(n, r, target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] exponential r={r:.3f} → score={val:.6f}")

    # reuse a previously discovered construction if available
    if "height_sequence_1" in globals():
        cand = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[{time.time():.2f}] global init → score={val:.6f}")

    print(f"[{time.time():.2f}] After shape search best score={best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Gradient descent with restarts (≈30 % of remaining budget)
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                   target_sum,
                                                   gd_deadline)
    if val_gd < best_val:
        best_val, best_seq = val_gd, a_gd.copy()
    print(f"[{time.time():.2f}] After GD best score={best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated annealing on the rest of the time
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, deadline,
                                            start_temp=12.0,
                                            end_temp=1e-6,
                                            max_frac_start=0.55,
                                            max_frac_end=0.015,
                                            max_window=200)
        if val_sa < best_val:
            best_val, best_seq = val_sa, a_sa.copy()
        print(f"[{time.time():.2f}] After SA best score={best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Pure sub‑gradient refinement (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() + 0.5 < deadline:
        a_ref, val_ref = _max_subgradient_refine(best_seq,
                                                 target_sum,
                                                 deadline)
        if val_ref < best_val:
            best_val, best_seq = val_ref, a_ref.copy()
        print(f"[{time.time():.2f}] Sub‑gradient refined best={best_val:.6f}")

    # --------------------------------------------------------------
    # 5) Final short projected‑gradient polish (a few seconds left)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent_v2(best_seq,
                                                       deadline,
                                                       target_sum)
        if val_fin < best_val:
            best_val, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved score={best_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return as a plain list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Optional final check with the provided benchmark
    try:
        val_check = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final benchmark evaluation = {val_check:.6f}")
    except Exception:
        pass

    return final_seq
