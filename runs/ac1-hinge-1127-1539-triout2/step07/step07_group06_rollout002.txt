def _lp_iterative_step(f, target_sum, deadline, rng):
   best_f = f.copy()
   best_max = _max_convolution(f)
   while time.time() < deadline:
       M = best_max
       # Solve LP for g
       g0 = _solve_lp_for_g(best_f, M)
       if g0 is None:
           break
       sum_g0 = np.sum(g0)
       if sum_g0 < 1e-12:
           break
       # scale
       scale = target_sum / sum_g0
       g = g0 * scale
       # try mixing factors
       improved = False
       for t in [0.2, 0.1, 0.05, 0.02, 0.01]:
           f_new = (1 - t) * best_f + t * g
           # optionally project to simplex to correct rounding
           f_new = _project_to_simplex(f_new, target_sum)
           max_new = _max_convolution(f_new)
           if max_new < best_max - 1e-8:
               best_f = f_new
               best_max = max_new
               improved = True
               print(f"[{time.time():.1f}] LP step improved max={best_max:.6f}")
               break
       if not improved:
           # no improvement with given t; break loop to avoid endless
           break
   return best_f, best_max

"""
search_candidate.py

Hybrid optimizer for the step‑function problem.
-------------------------------------------------
The objective (after L¹‑normalisation Σa = √(2·n)) is simply
    max_k (a * a)_k .
We combine three ideas:

1. **Linear‑programming direction** – for the current sequence f we
   solve the LP
        maximise   Σ g_j
        s.t.      (f * g)_k ≤ M   (for all 0≤k<2n‑1)
                  g_j ≥ 0 ,
   where M = max(f*f).  The optimal g concentrates mass where
   f contributes little to the peak.  After scaling g to the required
   ℓ¹‑norm we blend f←(1‑t)f+t g (small t) – this provably does not
   increase the peak and usually reduces it.

2. **Local stochastic refinement** – after several LP steps we run
   the incremental “mass‑transfer” local search (the code that gave
   1.5117 before).  Starting from a much better point the local
   search can push the bound further.

3. **Projected gradient polishing** – finally a short projected‑gradient
   descent on a soft‑max approximation of the peak is applied.

All three phases respect the time budget (default 1000 s).  The code
returns a list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)**(-p) (i=0..n‑1) renormalised to have ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Linear‑programming direction
# ----------------------------------------------------------------------


def _conv_matrix_fixed_f(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix T such that T @ g = f * g.
    For any column i the rows i … i+n‑1 contain the vector f.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    T = np.empty((rows, n), dtype=np.float64)
    T[:] = 0.0
    for i in range(n):
        T[i:i + n, i] = f
    return T


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   s.t.   (f * g)_k ≤ M  , g ≥ 0.
    Returns the optimal g (not yet normalised) or None on failure.
    """
    T = _conv_matrix_fixed_f(f)
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = [T @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        # ECOS is fast for medium‑sized LPs; we use a tight tolerance.
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-9, reltol=1e-9)
    except Exception as e:  # pragma: no cover – safeguard
        print("[LP] solver error:", e)
        return None

    if g.value is None:
        return None
    g0 = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    return g0


def _lp_iterate(f0: np.ndarray,
                target_sum: float,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑driven improvement steps.
    Returns the best sequence found and its max‑convolution value.
    """
    f = f0.copy()
    best_f = f.copy()
    best_max = _max_convolution(f)

    while time.time() < deadline:
        M = best_max
        g0 = _solve_lp_for_g(best_f, M)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            break

        # Scale to the required ℓ¹‑norm.
        scale = target_sum / sum_g0
        g = g0 * scale

        improved = False
        # try a handful of mixing coefficients, largest first
        for t in (0.25, 0.15, 0.10, 0.07, 0.04, 0.02):
            f_new = (1.0 - t) * best_f + t * g
            f_new = _project_to_simplex(f_new, target_sum)
            new_max = _max_convolution(f_new)
            if new_max < best_max - 1e-8:
                best_f = f_new
                best_max = new_max
                improved = True
                print(f"[{time.time():.1f}] LP step t={t:.3f} improved max={best_max:.6f}")
                break
        if not improved:
            # No mixing helped – stop the LP phase.
            break
    return best_f, best_max


# ----------------------------------------------------------------------
# Incremental local‑search core (same as the reference implementation)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer an amount `delta` from `donor` to `receiver`.
    Exact O(n) update of both the coefficient vector `a` and its
    self‑convolution `conv`.
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]
    a_before = a.copy()

    # linear part
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic part
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    a[donor] -= delta
    a[receiver] += delta
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Fast incremental local search (mass‑transfer + smoothing + jitter).
    Returns best sequence and its max‑convolution.
    """
    target = np.sqrt(2.0 * n)

    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        max_frac = 0.5 * (0.005 / 0.5) ** tau
        temp = 0.5 * (1e-5 / 0.5) ** tau
        move_type = rng.random()

        # ------------------ mass transfer (≈80 %) ------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]
            total_w = w.sum()
            donor = rng.choice(idx, p=w / total_w) if total_w > 0 else rng.integers(n)

            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] local new best max={best_max:.6f}")
            else:
                _apply_transfer(a, conv, receiver, donor, delta)

        # ------------------ smoothing (≈10 %) ------------------
        elif move_type < 0.95:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_window + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # ------------------ Gaussian jitter (≈10 %) ------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved max={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Projected gradient descent on soft‑max (reference polish)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search: LP‑driven direction → stochastic local search →
    soft‑max gradient polish.  Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # keep a small safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1) Warm‑start from a previously found construction (if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.size
        target0 = np.sqrt(2.0 * n0)
        seq0 = _project_to_simplex(seq0, target0)
        if time.time() < deadline - 5.0:
            sub_dead = time.time() + min(30.0, (deadline - time.time()) * 0.15)
            seq0_lp, val0 = _lp_iterate(seq0, target0, sub_dead, rng)
            if val0 < best_val:
                best_val = val0
                best_seq = seq0_lp.copy()
                print(f"[{time.time():.1f}] warm‑start LP improved max={best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Main optimisation loop over a few problem sizes
    # --------------------------------------------------------------
    # Size choices – larger gives more freedom but incurs a heavier LP.
    size_list = [1200, 1500, 1800]
    for n in size_list:
        if time.time() >= deadline - 5.0:
            break

        target = np.sqrt(2.0 * n)
        # initialise with a power‑law shape
        p = rng.uniform(0.4, 0.8)
        a0 = _power_law_sequence(n, p, target)
        a0 = _project_to_simplex(a0, target)

        # ----- LP‑driven direction -------------------------------------------------
        remaining = deadline - time.time()
        lp_time = min(remaining * 0.30, 150.0)   # allocate up to 30 % of remaining time
        lp_dead = time.time() + lp_time
        a_lp, val_lp = _lp_iterate(a0, target, lp_dead, rng)

        # ----- Local stochastic refinement ----------------------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            loc_time = min(remaining * 0.35, 250.0)
            loc_dead = time.time() + loc_time
            a_loc, val_loc = _local_search(n, loc_dead, rng, init_seq=a_lp)
            if val_loc < val_lp:
                a_lp, val_lp = a_loc, val_loc

        # ----- Gradient‑descent polish ---------------------------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            gd_time = min(remaining * 0.25, 150.0)
            gd_dead = time.time() + gd_time
            a_gd, val_gd = _annealed_gradient_descent(a_lp, gd_dead)
            if val_gd < val_lp:
                a_lp, val_lp = a_gd, val_gd

        # keep the best overall result
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[{time.time():.1f}] new overall best n={n} max={best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Final tidy‑up (strip trailing zeros)
    # --------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best max≈{best_val:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
