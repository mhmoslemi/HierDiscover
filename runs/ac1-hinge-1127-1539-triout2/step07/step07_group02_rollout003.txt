def random_flat_spectrum_seq(n, rng):
    N = 2 * n
    # magnitude sqrt(1/2) perhaps?
    # random phases uniform 0,2π
    # For real irfft, we need half-spectrum of length N//2+1 (real for k=0 and N/2).
    # We'll set magnitude=1 for all frequencies.
    phases = rng.random(N//2+1) * 2*np.pi
    # amplitude 1 for all frequencies (except maybe DC)
    coeffs = np.exp(1j*phases)
    # Ensure DC (k=0) and Nyquist (k=N/2 if N even) are real: set imaginary part zero.
    coeffs[0] = np.abs(coeffs[0])  # real positive
    if N % 2 == 0:
        coeffs[-1] = np.abs(coeffs[-1])  # Nyquist term real
    seq_full = np.fft.irfft(coeffs, n=N)
    # take first n entries
    a = seq_full[:n]
    # shift to nonnegative
    a = a - a.min()
    # normalise sum to sqrt(2*n)
    S = np.sqrt(2.0*n)
    a = a / a.sum() * S
    return a

"""
Optimisation of a non‑negative step function for minimum self‑convolution peak.

The algorithm proceeds in three stages:

1. **Broad random sampling** – many analytic families (uniform,
   geometric, power‑law, beta‑like, sinusoidal, inverse‑power, flat‑spectrum)
   are sampled for several lengths (2 k … 30 k).  Each candidate is
   normalised to Σa = √(2·n) and its convolution peak is recorded.
   The 5 best candidates are kept.

2. **Projected Adam on a soft‑max approximation** – the peak of the
   convolution is approximated by a log‑sum‑exp with a temperature α.
   α is annealed from 0.1 up to 2000, making the objective increasingly
   tight.  Adam with simplex projection (non‑negativity + fixed L1‑norm)
   minimises this smooth objective.  The best result of each run updates
   the global best.

3. **Simulated‑annealing hill‑climb** – on the current best vector a small
   random mass‑swap is performed.  Moves that improve the true peak are
   always accepted; worsening moves are accepted with a temperature‑
   dependent probability.  The temperature decays exponentially.
   This stage refines the solution to a local optimum.

All steps respect the 1000 s wall‑clock budget.  The final sequence is
stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights) of length 2*n‑1.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Random families (used in the exploratory phase)
# ----------------------------------------------------------------------
def _random_flat_spectrum_seq(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a non‑negative sequence with a flat magnitude spectrum."""
    N = 2 * n                     # length of the FFT (even)
    # random phases, keep DC and Nyquist real
    phases = rng.random(N // 2 + 1) * 2.0 * np.pi
    # unit magnitude for all frequencies
    coeffs = np.exp(1j * phases)
    coeffs[0] = np.abs(coeffs[0])        # DC component real
    coeffs[-1] = np.abs(coeffs[-1])      # Nyquist component real (when N even)
    seq_full = np.fft.irfft(coeffs, n=N)
    a = seq_full[:n]
    a = a - a.min()                      # shift to non‑negative
    S = np.sqrt(2.0 * n)
    a = a / a.sum() * S
    return a


def _make_candidate(n: int, fam: int, rng: np.random.Generator,
                    base_seq: np.ndarray | None) -> np.ndarray:
    """Create a raw (unnormalised) candidate vector for a given family."""
    if fam == 0 and base_seq is not None:          # up‑sample previous best
        a = _upsample_seq(base_seq, n)
    elif fam == 1:                                 # uniform random
        a = rng.random(n)
    elif fam == 2:                                 # symmetric geometric decay
        rate = rng.uniform(0.95, 0.9999)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) * 0.5
    elif fam == 3:                                 # power‑law symmetric
        p = rng.uniform(0.2, 4.0)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:                                 # beta‑like
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 5:                                 # sinusoid power
        p = rng.uniform(0.5, 4.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 6:                                 # inverse‑power (U‑shaped)
        p = rng.uniform(0.2, 3.0)
        eps = 1e-12
        idx = np.arange(n, dtype=float) + 0.5
        a = 1.0 / ((idx * (n + 0.5 - idx)) ** p + eps)
    else:                                          # flat‑spectrum random
        a = _random_flat_spectrum_seq(n, rng)
    a = np.clip(a, 0.0, None)      # safety
    return a


# ----------------------------------------------------------------------
# Optimiser – Projected Adam on log‑sum‑exp of the convolution
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray, deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False,
                   max_alpha: float = 2000.0,
                   alpha_mul: float = 1.05,
                   init_lr: float = 0.6,
                   lr_decay: float = 0.99997) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # temperature schedule for the soft‑max
    alpha = 0.1

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= init_lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # tiny jitter to escape flat regions
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        init_lr *= lr_decay

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local refinement
# ----------------------------------------------------------------------
def _local_anneal(a0: np.ndarray, deadline: float,
                  rng: np.random.Generator,
                  enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_score = np.max(_full_convolution(a))
    best_a = a.copy()
    best_score = cur_score

    # initial temperature proportional to the current excess over the lower bound 1
    T = max(0.05, (cur_score - 1.0) * 0.5)

    # geometric cooling
    decay = 0.9999

    while time.time() < deadline:
        # propose a tiny mass move
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = 0.04 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)

        cand_score = np.max(_full_convolution(cand))

        # accept if better or with Metropolis probability
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        T *= decay
        # occasional temperature reset to avoid freezing too early
        if rng.random() < 0.001:
            T = max(T, 0.02)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) whose
    self‑convolution peak is as low as possible.  The routine respects a
    wall‑clock budget (default 1000 s) and stores the best sequence in the
    global variable ``height_sequence_1`` for reuse.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final conversion & printing
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve previous best if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap random families (≈10% of time budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.10 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1

        # pick a length – favouring the size of the previous best but also exploring
        if base_seq is not None and rng.random() < 0.45:
            scale = rng.uniform(0.8, 1.2)
            n = max(2000, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(2500, 25001))

        fam = rng.integers(0, 8)      # eight different families (see above)
        a_raw = _make_candidate(n, fam, rng, base_seq)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        s = a_raw.sum()
        if s == 0.0:
            a_raw = np.full(n, 1.0)
            s = a_raw.sum()
        a = a_raw / s * S

        # optional forced symmetry (helps the landscape)
        if rng.random() < 0.35:
            a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = a.copy()
            print(f"[phase1] attempt={attempts} n={n} fam={fam} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Ensure we have at least one candidate
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        n = 8000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_overall_seq = a
        best_overall_score = np.max(_full_convolution(a))

    # ------------------------------------------------------------------
    # Keep the top few candidates for intensive refinement
    # ------------------------------------------------------------------
    # we will keep 4 best candidates (the current best plus three variants)
    top_candidates = [(best_overall_seq, best_overall_score)]

    # create a few more variations by small random perturbations of the best
    for _ in range(3):
        a = best_overall_seq.copy()
        n = a.shape[0]
        S = np.sqrt(2.0 * n)
        for _ in range(8):
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
        a = _project_simplex(a, S)
        a = _symmetrize(a) if rng.random() < 0.5 else a
        sc = np.max(_full_convolution(a))
        top_candidates.append((a, sc))

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈60% of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # allocate most of the remaining budget to Adam
    adam_deadline = now + 0.60 * remaining
    for idx, (seq, _) in enumerate(top_candidates):
        if time.time() >= adam_deadline:
            break
        # each run gets an equal share of the Adam budget
        sub_deadline = time.time() + (adam_deadline - time.time()) / (len(top_candidates) - idx)
        a_opt, sc_opt = _optimize_adam(seq, sub_deadline, rng,
                                       enforce_sym=True,
                                       max_alpha=2000.0,
                                       alpha_mul=1.06,
                                       init_lr=0.7,
                                       lr_decay=0.99996)
        if sc_opt < best_overall_score:
            best_overall_score = sc_opt
            best_overall_seq = a_opt.copy()
            print(f"[adam] candidate={idx} improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing local refinement (≈30% of remaining)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_loc, sc_loc = _local_anneal(best_overall_seq, deadline, rng, enforce_sym=True)
        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[anneal] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final conversion & reporting
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
