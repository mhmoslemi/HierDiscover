"""
Propose a step‑function (non‑negative vector) that minimizes the
self‑convolution peak.
The search consists of three main stages:

1. **Exploratory sampling** – generate many analytic families (uniform,
   geometric, power‑law with both positive and negative exponents,
   beta‑like, sinusoidal, arcsine‑like, exponential, piece‑wise constant,
   random spikes).  Each candidate is normalised so that Σa = √(2·n) and
   the best few are kept.

2. **Aggressive Adam optimisation** – minimise a soft‑max approximation
   of the convolution peak with a very slowly increasing temperature
   (α up to 2000).  The optimiser works on the symmetric version of the
   sequence (a[i]=a[n‑1‑i]) which halves the dimensionality and forces the
   peak to be spread evenly.  The Euclidean projection onto the simplex
   (∑a = √(2·n), a≥0) is applied after every update.

3. **Refinement** – a sub‑gradient descent on the true maximum followed
   by a cheap random‑swap hill‑climb.  The whole pipeline is time‑aware
   (budget ≈ 1000 s) and stores the best sequence globally in
   ``height_sequence_1`` for reuse in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max objective (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    # reverse weights because of convolution definition
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # indices that correspond to the derivative of each a_i
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over the entries of conv."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation utilities
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] == v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Candidate families for the exploratory phase
# ----------------------------------------------------------------------
def _generate_candidate(rng: np.random.Generator,
                        base_seq: np.ndarray | None,
                        target_n: int) -> np.ndarray:
    """
    Produce one random candidate of length ``target_n`` using a mixture
    of analytic families.
    """
    n = target_n
    # Choose family
    fam = rng.integers(0, 9)
    if fam == 0 and base_seq is not None:
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                     # uniform random
    elif fam == 2:
        # geometric decay from both ends
        rate = rng.uniform(0.9, 0.9999)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) / 2.0
    elif fam == 3:
        # power‑law (positive exponent)
        p = rng.uniform(0.2, 4.0)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        # power‑law with possibly negative exponent (heavy tails)
        p = rng.uniform(-4.0, -0.2)  # negative exponent
        idx = np.arange(n, dtype=float) + 0.5
        a = ((idx) * (n + 0.5 - idx)) ** p
    elif fam == 5:
        # beta‑like shape
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 6:
        # sinusoidal power
        p = rng.uniform(0.5, 4.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    elif fam == 7:
        # arcsine‑like (inverse‑sqrt) shape
        eps = 1e-12
        i = np.arange(n, dtype=float) + 0.5
        a = 1.0 / np.sqrt(i * (n + 0.5 - i) + eps)
    elif fam == 8:
        # exponential decay from the left, mirrored
        lam = rng.uniform(0.0003, 0.005)
        left = np.exp(-lam * np.arange(n))
        a = (left + left[::-1]) / 2.0
    else:  # fam == 9 (random spikes)
        a = np.full(n, 0.0)
        k = rng.integers(3, 10)                # number of spikes
        for _ in range(k):
            pos = rng.integers(0, n)
            height = rng.random() * 5.0 + 0.5
            a[pos] = height
    # Ensure non‑negativity
    a = np.clip(a, 0.0, None)

    # Normalise to Σa = √(2·n)
    S = np.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0)
        s = a.sum()
    a = a / s * S

    # Occasionally enforce symmetry (helps later stages)
    if rng.random() < 0.4:
        a = _symmetrize(a)

    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max objective
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the smooth‑max (log‑sum‑exp) of the
    self‑convolution.  The temperature α starts small (0.1) and is
    multiplied each iteration up to ``max_alpha`` (default 2000).
    Returns the best vector encountered and its true convolution peak.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature schedule
    alpha = 0.1
    alpha_mul = 1.06
    max_alpha = 2000.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights and gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional small random jitter to escape plateaus
        if rng.random() < 0.007:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        # modest decay of learning rate
        lr *= 0.99997

    # final evaluation with true max
    final_score = np.max(_full_convolution(best_a))
    return best_a, final_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the genuine max‑convolution entry.
    The step size decays as η₀ / √t.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.22
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient (outer‑product slice)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random redistribution (simulated annealing flavour)
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early stop if we stopped improving for a while
        if it % 12000 == 0 and best_score == cur_max:
            break

    return best_a, best_score

# Tiny random‑swap hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climb that moves a tiny fraction of mass between two
    randomly chosen positions."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.018 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score

            top_heap.append((cur_score, a.copy()))
            top_heap.sort(key=lambda x: x[0])
        elif cur_score < top_heap[-1][0]:
            top_heap[-1] = (cur_score, a.copy())
            top_heap.sort(key=lambda x: x[0])

        if attempts % 2000 == 0:
            print(f"[phase0] attempts={attempts} best={top_heap[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive Adam optimisation (majority of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # allocate ~60 % of remaining time to Adam, split among top candidates
    adam_time_total = now + 0.60 * remaining
    adam_deadline_per = None
    if len(top_heap) == 0:
        # fallback – generate a simple uniform candidate
        n_fallback = 8000
        a_fallback = np.full(n_fallback, 1.0)
        a_fallback = a_fallback / a_fallback.sum() * np.sqrt(2.0 * n_fallback)
        top_heap = [(np.max(_full_convolution(a_fallback)), a_fallback)]

    # Distribute Adam budget evenly over the stored candidates
    per_candidate_time = (adam_time_total - now) / len(top_heap)
    best_overall_a = None
    best_overall_score = np.inf
    for idx, (_, seq) in enumerate(top_heap):
        cand_deadline = now + per_candidate_time
        a_adam, sc_adam = _optimize_adam(seq, cand_deadline, rng,
                                         enforce_sym=True)
        now = time.time()
        print(f"[adam #{idx+1}] score={sc_adam:.6f}")
        if sc_adam < best_overall_score:
            best_overall_score = sc_adam
            best_overall_a = a_adam.copy()
        # break early if we have exhausted the global deadline
        if now >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent (≈30 % of what is left)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        subgrad_deadline = now + 0.30 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad(best_overall_a, subgrad_deadline,
                                          rng, enforce_sym=True)
        if sc_sub < best_overall_score:
            best_overall_score = sc_sub
            best_overall_a = a_sub.copy()
        print(f"[subgrad] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final tiny random‑swap hill‑climb (≈10 % of the rest)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_loc, sc_loc = _local_swap_search(best_overall_a, deadline, rng,
                                           enforce_sym=True)
        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_a = a_loc.copy()
        print(f"[local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (also expose it globally for future runs)
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
