def _gradient_exact_max(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    k = np.argmax(conv)
    # compute grad
    grad = np.zeros_like(a)
    # term1: where 0 <= k - i < n => i in [k-(n-1), k]
    lo1 = max(0, k - (n - 1))
    hi1 = min(k, n - 1)
    i1 = np.arange(lo1, hi1+1)
    grad[i1] += a[k - i1]
    # term2: where 0 <= i - k < n => i in [k, k + n - 1] intersect [0,n-1]
    lo2 = max(k, 0)
    hi2 = min(n - 1, k + n - 1)
    i2 = np.arange(lo2, hi2+1)
    grad[i2] += a[i2 - k]
    return grad, conv

#!/usr/bin/env python
"""
Propose a low‐benchmark step function.

The algorithm works in three phases:

1. **Initialisation & short projected‐gradient runs**  
   Several seeds are generated (power‑law, mirrored power‑law and a few
   random Dirichlet vectors).  For each seed a few seconds of
   projected‑gradient descent on a soft‑max approximation of the
   convolution's maximum are performed.  The best result is retained.

2. **Guided simulated annealing**  
   Starting from the best candidate a long annealing phase moves mass
   from indices that contribute heavily to the current maximal
   convolution entry to indices that contribute little.  Moves also
   include local smoothing and random swaps.  The temperature and the
   allowed transfer fraction decay exponentially.

3. **Final polishing (exact‑max gradient descent)**  
   The final candidate is refined by a short projected‑gradient descent
   that uses the exact sub‑gradient of the max‑convolution (i.e. a single
   dominant offset).  This removes any remaining bias introduced by the
   soft‑max approximation.

All updates keep the ℓ¹‑norm fixed (∑aᵢ = √(2·n)) by Euclidean projection
onto the simplex, so the evaluation function reduces to the maximal
convolution value.  The routine respects the 1000 s budget and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT of given size."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * a.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray, size_fft: int) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    conv = _conv_fft(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
# Scoring
# ----------------------------------------------------------------------


def _score(a: np.ndarray, size_fft: int) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For a normalized vector (∑a = √(2·n)) the denominator cancels and the
    score equals the maximal convolution entry.
    """
    n = a.size
    max_b = _max_convolution_fft(a, size_fft)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Initial sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _sym_power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Symmetric power‑law (mirror around the centre)."""
    half = (n + 1) // 2
    a_half = (np.arange(1, half + 1, dtype=np.float64) ** (-p))
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, a_half[-2::-1]])
    a *= target_sum / a.sum()
    return a


def _random_simplex_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Random vector from a Dirichlet distribution, projected onto the simplex."""
    # small alpha encourages sparsity, which can be useful
    alpha = np.full(n, 0.3, dtype=np.float64)
    a = rng.dirichlet(alpha)
    a *= target_sum
    return a


# ----------------------------------------------------------------------
# Gradient based optimisation (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float, size_fft: int):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, max_b) where grad has the same shape as a.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, n=size_fft)
    b = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]

    max_b = np.max(b)
    # numerical stability
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    pa = np.fft.rfft(p, n=size_fft)
    ar = np.fft.rfft(a_rev, n=size_fft)
    conv_rev = np.fft.irfft(ar * pa, n=size_fft)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, max_b


def _annealed_gradient_descent(
    a0: np.ndarray,
    target_sum: float,
    size_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation.
    Uses a β schedule that rapidly grows to a large value (≈1e5).
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a, size_fft)

    beta = 1.0
    max_beta = 1e5
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta, size_fft)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a, size_fft)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β slowly
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray,
    target_sum: float,
    size_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run several short restarts of the soft‑max projected GD.
    The best result among the restarts is returned.
    """
    best_a = init.copy()
    best_val = _score(best_a, size_fft)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # each restart gets at most 25 s or 30 % of the remaining budget
        run_budget = min(25.0, remaining * 0.30)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start, target_sum, size_fft,
                                            time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max gradient descent (sub‑gradient)
# ----------------------------------------------------------------------


def _gradient_exact_max(a: np.ndarray, size_fft: int):
    """
    Sub‑gradient of max(convolution) at the current maximal offset.
    Returns (grad, max_b, conv) where conv is the full convolution.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)

    # term a_{k‑i}
    lo1 = max(0, k - (n - 1))
    hi1 = min(k, n - 1)
    if lo1 <= hi1:
        idx1 = np.arange(lo1, hi1 + 1)
        grad[idx1] += a[k - idx1]

    # term a_{i‑k}
    lo2 = max(k, 0)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        grad[idx2] += a[idx2 - k]

    max_b = conv[k]
    return grad, max_b, conv


def _exact_gradient_descent(
    a0: np.ndarray,
    target_sum: float,
    size_fft: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Simple momentum is used and occasional jitter helps to escape plateaus.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a, size_fft)

    step_base = target_sum * 0.4
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9
    it = 0

    while time.time() < deadline:
        grad, max_b, _ = _gradient_exact_max(a, size_fft)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a, size_fft)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExactGD it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided simulated annealing
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    size_fft: int,
    deadline: float,
    start_temp: float = 8.0,
    end_temp: float = 1e-6,
    max_frac_start: float = 0.5,
    max_frac_end: float = 0.005,
    max_window: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that preferentially moves mass away from the
    current maximal convolution entry.
    """
    a = a0.copy()
    n = a.size
    cur_conv = _conv_fft(a, size_fft)
    cur_score = _score(a, size_fft)

    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    total = max(deadline - start_time, 1e-9)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate current maximal offset
        k_max = int(np.argmax(cur_conv))

        # contribution weight for each index (a_i * a_{k_max-i})
        weight = np.zeros(n, dtype=np.float64)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        weight[idx] = a[idx] * a[k_max - idx]

        move_type = None
        # ------------------- mass transfer -------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # choose a receiver among low‑weight entries
            low_thr = np.percentile(weight, 12)
            low_idxs = np.where(weight <= low_thr)[0]
            if low_idxs.size:
                receiver = rng.choice(low_idxs)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.88:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                seg = a[start:end].copy()
                avg = seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution (FFT is cheap enough)
        conv_new = _conv_fft(a, size_fft)
        new_score = _score(a, size_fft)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            cur_conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to avoid stagnation
        if it % 7000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.0008, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            cur_conv = _conv_fft(a, size_fft)
            cur_score = _score(a, size_fft)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    benchmark score.  The routine respects a total wall‑clock budget
    given by *budget_s* (seconds) and returns the best sequence found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more freedom)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)
    size_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Prepare a handful of starting points
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # power‑law
    candidates.append(_power_law_sequence(n, p=0.62, target_sum=target_sum))

    # symmetric power‑law
    candidates.append(_sym_power_law_sequence(n, p=0.60, target_sum=target_sum))

    # a few random Dirichlet vectors
    for _ in range(3):
        candidates.append(_random_simplex_sequence(n, target_sum, rng))

    # ------------------------------------------------------------------
    # Phase 1 – short projected gradient descent runs on each seed
    # ------------------------------------------------------------------
    gd_time_total = (deadline - time.time()) * 0.35
    per_seed = gd_time_total / max(len(candidates), 1)
    best_a = None
    best_val = np.inf

    for idx, seq in enumerate(candidates):
        sub_deadline = min(time.time() + per_seed, deadline - 2.0)
        if sub_deadline <= time.time():
            break
        a_opt, val_opt = _gradient_descent_with_restarts(seq, target_sum,
                                                         size_fft,
                                                         sub_deadline)
        print(f"[{time.time():.2f}] Seed {idx} GD best={val_opt:.6f}")
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()

    # ------------------------------------------------------------------
    # Phase 2 – guided simulated annealing
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – create a simple power‑law if everything failed
        best_a = _power_law_sequence(n, p=0.6, target_sum=target_sum)
        best_val = _score(best_a, size_fft)

    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_a, size_fft, deadline,
                                            start_temp=12.0,
                                            end_temp=1e-6,
                                            max_frac_start=0.5,
                                            max_frac_end=0.008,
                                            max_window=180)
        if val_sa < best_val:
            best_a = a_sa
            best_val = val_sa
        print(f"[{time.time():.2f}] SA finished best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing with exact‑max gradient descent
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_final, val_final = _exact_gradient_descent(best_a, target_sum,
                                                     size_fft,
                                                     deadline)
        if val_final < best_val:
            best_a = a_final
            best_val = val_final
            print(f"[{time.time():.2f}] Final exact GD improved score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_a, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final sanity check using the provided benchmark
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
