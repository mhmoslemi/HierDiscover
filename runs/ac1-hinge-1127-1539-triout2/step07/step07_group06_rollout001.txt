def _apply_transfer(a, conv, donor, receiver, delta):
    if delta <=0 ...:
        return
    n = a.shape[0]
    a_before = a.copy()
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before
    # quadratic contributions
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta
    a[donor] -= delta
    a[receiver] += delta
    # fix negative
    if a[donor] < 0.0: a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)

"""
Directed peak‑reduction optimizer for the step‑function heights.

The algorithm works on a fixed length `n` (chosen from several candidates)
and repeatedly performs *targeted* mass‑transfer moves that are guaranteed
to lower the current maximum entry of the self‑convolution `a*a`.
Each move transfers a fraction of mass from an index that contributes
most to the current peak to an index that does not affect (or weakly affects)
that peak.  The move is applied exactly with an O(n) incremental update.
If no improving move can be found we fall back to a small Gaussian jitter.

After the time budget for a given length is exhausted the best sequence
found is polished with the previously supplied soft‑max gradient descent.
The whole routine respects the 1000 s limit and returns a list ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (assumes Σa = √(2·n))."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Fast objective value (max convolution) – sum already √(2·n)."""
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Incremental update for a mass transfer
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer an amount `delta` from `donor` to `receiver`.
    Updates both the coefficient vector `a` and the convolution array `conv`
    exactly (including the quadratic `delta**2` terms).
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    # keep a copy of the coefficients before the move
    a_before = a.copy()

    # linear part of the convolution update
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic contributions
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # update the coefficients
    a[donor] -= delta
    a[receiver] += delta

    # clean possible tiny negatives caused by rounding
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Directed peak‑reduction local search
# ----------------------------------------------------------------------


def _directed_search(n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Perform a fast, deterministic peak‑reduction search for a given length `n`.
    Returns the best sequence found (as a NumPy array) and its objective value
    (the maximum entry of the self‑convolution, which equals the evaluation
    score after the required L1‑normalisation).
    """
    target = np.sqrt(2.0 * n)

    # ----- initialise ----------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.45, 0.85)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)

    conv = _conv_fft(a)          # length 2*n‑1
    cur_max = conv.max()
    best_a = a.copy()
    best_val = cur_max

    start = time.time()
    moves = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # current peak
        k_max = int(conv.argmax())
        cur_max = float(conv[k_max])

        # second largest entry (used for a safe delta estimate)
        temp = conv.copy()
        temp[k_max] = -np.inf
        second_max = float(temp.max())

        # indices that actually influence the current peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        w = a[idx] * a[k_max - idx]

        # donors – take a few best contributors
        top_k = min(5, idx.size)
        donor_candidates = idx[np.argpartition(-w, top_k - 1)[:top_k]]

        # receivers –prefer indices that do NOT affect the peak
        outside = np.where((np.arange(n) < lo) | (np.arange(n) > hi))[0]
        if outside.size > 0:
            receiver = int(outside[np.argmin(a[outside])])
        else:
            # all indices affect the peak – pick the one with smallest partner
            partner_vals = a[k_max - idx]  # valid because idx inside window
            receiver = int(idx[np.argmin(partner_vals)])

        moved = False

        # try donors in order of decreasing contribution
        for donor in donor_candidates:
            # contribution of donor to the peak
            a_k_minus_donor = a[k_max - donor] if 0 <= k_max - donor < n else 0.0
            a_k_minus_receiver = a[k_max - receiver] if 0 <= k_max - receiver < n else 0.0
            diff = a_k_minus_donor - a_k_minus_receiver
            if diff <= 0.0:
                continue          # moving mass from this donor would not help

            # compute a delta that would (approximately) bring the peak down
            # to the second‑largest value – but never exceed half the donor mass
            if cur_max - second_max > 0:
                delta_needed = (cur_max - second_max) / (2.0 * diff)
            else:
                delta_needed = a[donor] * 0.5
            delta = min(a[donor] * 0.5, delta_needed)
            delta = max(delta, 0.0)
            if delta <= 0.0:
                continue

            # backup state
            a_backup = a.copy()
            conv_backup = conv.copy()

            # apply the transfer and test
            _apply_transfer(a, conv, int(donor), int(receiver), float(delta))
            new_max = float(conv.max())

            if new_max < cur_max:
                cur_max = new_max
                moved = True
                break
            else:
                # try a smaller step (halving) before giving up on this donor
                a[:] = a_backup
                conv[:] = conv_backup
                delta_small = delta * 0.5
                if delta_small > 0:
                    _apply_transfer(a, conv, int(donor), int(receiver), float(delta_small))
                    new_max2 = float(conv.max())
                    if new_max2 < cur_max:
                        cur_max = new_max2
                        moved = True
                        break
                # revert again
                a[:] = a_backup
                conv[:] = conv_backup

        # if no improving transfer was found, use a tiny random jitter
        if not moved:
            # scaling factor slowly decays as we approach the deadline
            remaining = max(deadline - time.time(), 0.0)
            total = max(deadline - start, 1e-12)
            scale = target * 0.001 * (remaining / total)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = float(conv.max())

        # record the best solution seen so far
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.1f}] new best (n={n}) max={best_val:.6f}")

        moves += 1

        # every thousand moves re‑project to the simplex to curb numerical drift
        if moves % 1000 == 0:
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = float(conv.max())

        # optional early exit if we already beat the target bound
        if best_val <= 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Soft‑max gradient descent (polishing step – same as reference)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    The routine runs a deterministic, directed peak‑reduction local search
    on a handful of sequence lengths, keeps the best candidate and finishes
    with a short soft‑max gradient‑descent polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.size
        # give the warm‑start a modest slice of the time budget
        if time.time() < deadline - 5.0:
            ws_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.20)
            seq, sc = _directed_search(n0, ws_deadline, rng, init_seq=seq0)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] warm‑start n={n0} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Directed search on several lengths (larger n gives more freedom)
    # ------------------------------------------------------------------
    # lengths are chosen to fit comfortably into the remaining budget
    lengths = [1800, 2400, 3000, 3600, 4200]  # a progression up to ~4 k
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        # allocate at most 25 % of the remaining time to a single length
        slice_time = min(remaining * 0.25, 180.0)
        run_deadline = time.time() + slice_time
        seq, sc = _directed_search(n, run_deadline, rng)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] length={n} → new best score={best_score:.6f}")

        # early exit if we have already reached the target bound
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2) Polishing with the soft‑max gradient descent (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _annealed_gradient_descent(best_seq, extra_deadline)
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] after polishing score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is not None:

        if nz.size:

        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
