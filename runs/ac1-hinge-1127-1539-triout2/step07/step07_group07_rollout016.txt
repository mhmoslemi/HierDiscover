        # pick i index with a_i > epsilon
        i = rng.integers(0, n)
        if a[i] <= 1e-12:

        # choose j
        j = rng.integers(0, n)

            continue

        if eps <= 0:
            continue

        a_new[i] -= eps
        a_new[j] += eps
        # ensure nonneg (should be)
        # evaluate
        new_obj = _max_convolution_fft(a_new)
        dE = new_obj - cur_obj
        accept = False
        if dE <= 0:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True
        if accept:
            a = a_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj, best_seq = new_obj, a.copy()
        # cooling
        T *= 0.9999
        it += 1
        if it % 1000 == 0:

                cur_obj = _max_convolution_fft(a)
                if cur_obj < best_obj:
                    best_obj, best_seq = cur_obj, a.copy()

def _subgradient_max(a):
    n = a.shape[0]
    size = 1 << ((3*n-2).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    # full convolution a*a

def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)
    best_seq = a.copy()
    best_val, _ = _subgradient_max(a)
    it = 0
    rng = np.random.default_rng()
    step0 = 1.0
    while time.time() < deadline:
        val, grad = _subgradient_max(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
        # step size
        norm = np.linalg.norm(grad)
        if norm == 0:
            # random jitter
            jitter = rng.normal(scale=target_sum*1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            continue

        lr = step0 / math.sqrt(it + 1) * (target_sum / (norm + 1e-12))
        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)
        a = a_new
        it += 1
        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum*5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
    return best_seq, best_val

def _permutation_annealing(a0, deadline):
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)
    T = best_score * 0.05 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]
        cur = _max_convolution_fft(a)
        dE = cur - best_score
        if cur < best_score:
            best_score, best = cur, a.copy()
        else:
            if rng.random() < math.exp(-dE / T):
                # accept worse move: keep a
                pass
            else:
                # revert swap
                a[i], a[j] = a[j], a[i]
        T *= cooling
        # occasional block reversal
        if rng.random() < 0.001:
            start = rng.integers(0, n)
            length = rng.integers(2, min(100, n-start))
            a[start:start+length] = a[start:start+length][::-1]
    return best

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.  The objective after the normalisation performed in
``evaluate_sequence`` is simply  max_k (a*a)_k .  The algorithm proceeds in
the following stages, each respecting the global deadline:

1. **Fast parametric sweep** – generate candidate multisets from
   power‑law, exponential and truncated‑linear families (lengths 6000,
   8000, 10000).  For each family we test a few simple orderings
   (original, reversed and an alternating small–large pattern) and keep
   the best one.

2. **Projected sub‑gradient descent** on the exact L∞‑norm of the
   autocorrelation.  The sub‑gradient of ``max(a*a)`` is
   ``g = 2·(a ⋆ I_max)`` where ``I_max`` is the indicator of the active
   convolution positions.  After each step the vector is projected onto
   the ℓ¹‑simplex {x ≥ 0, Σx = √(2 n)}.

3. **Permutation annealing** – a simulated‑annealing loop that swaps
   two positions of the current multiset.  Swaps are accepted with a
   Metropolis rule based on the current maximum convolution value.

4. **Value‑transfer annealing** – a second annealing phase that moves a
   small amount of mass from one index to another while preserving the
   ℓ¹‑norm.  This refines the multiset values themselves.

The three refinement phases are allocated dynamically according to the
remaining wall‑clock time, guaranteeing that a feasible sequence is
returned before the deadline.  Progress and intermediate objective
values are printed for monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0 , Σ x_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximal entry of the linear convolution a*a (full length
    2·n‑1) using an FFT of length a power‑of‑two.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """
    Compute the exact sub‑gradient of   max_k (a*a)_k .
    Returns
        max_val – the current maximal convolution entry
        grad    – a vector of shape a.shape containing the sub‑gradient
    """
    n = a.shape[0]
    # FFT length that can accommodate a (len n) and mask (len 2n‑1)
    size = 1 << ((3 * n - 2).bit_length())
    a_fft = np.fft.rfft(a, n=size)

    # full convolution a*a (length 2n‑1)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    max_val = conv_full.max()

    # indicator of active constraints (all positions attaining the max)
    mask = (conv_full >= max_val - 1e-12).astype(float)

    mask_fft = np.fft.rfft(mask, n=size)
    # convolution a ⋆ mask (length 3n‑2)
    grad_full = np.fft.irfft(a_fft * mask_fft, n=size)[: 3 * n - 2]

    # central n entries correspond to the gradient w.r.t. a
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    return max_val, grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Perform projected sub‑gradient descent on max(a*a).  Returns the best
    sequence seen and its objective value before the deadline.
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    best_seq = a.copy()
    best_val, _ = _subgradient_max(a)

    it = 0
    rng = np.random.default_rng()
    # initial step‑size factor
    step0 = 1.0

    while time.time() < deadline:
        cur_val, grad = _subgradient_max(a)

        # keep the best seen so far
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            # jitter to escape flat regions
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            continue

        # diminishing step size
        lr = step0 / math.sqrt(it + 1) * (target_sum / (norm + 1e-12))

        a_new = a - lr * grad
        a = _project_to_simplex(a_new, target_sum)

        # occasional random jitter
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation annealing (swap moves)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float):
    """
    Simulated‑annealing on the ordering of a fixed multiset.
    Only swaps are performed; the ℓ¹‑norm is invariant.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature proportional to current max
    T = best_score * 0.05 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose swap
        a[i], a[j] = a[j], a[i]
        cur_score = _max_convolution_fft(a)

        if cur_score < best_score:
            best_score, best = cur_score, a.copy()
            # keep the improvement
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur_score - best_score) / T):
                # accept worsening move
                pass
            else:
                # revert
                a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional small block reversal to increase mixing
        if rng.random() < 0.001:
            start = rng.integers(0, n)
            length = rng.integers(2, min(80, n - start))
            a[start:start + length] = a[start:start + length][::-1]

    return best


# ----------------------------------------------------------------------
# Value‑transfer annealing (move mass between entries)
# ----------------------------------------------------------------------


def _value_transfer_annealing(a0: np.ndarray, deadline: float):
    """
    Simulated‑annealing where a small amount of mass is transferred from one
    index to another while the ℓ¹‑norm stays fixed.
    """
    a = a0.copy()
    n = a.size
    cur_obj = _max_convolution_fft(a)
    best_seq = a.copy()
    best_obj = cur_obj

    T = cur_obj * 0.08 + 1e-9      # initial temperature
    cooling = 0.9999
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick a donor with non‑zero mass
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a fraction of the donor's mass
        eps = a[i] * rng.uniform(0.0, 0.15)
        if eps <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        new_obj = _max_convolution_fft(a_new)
        delta = new_obj - cur_obj

        if delta <= 0.0 or rng.random() < math.exp(-delta / T):
            a = a_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj, best_seq = new_obj, a.copy()

        T *= cooling

        # periodic jitter to avoid stagnation
        if rng.random() < 0.005:
            jitter = rng.normal(scale=cur_obj * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, math.sqrt(2.0 * n))
            cur_obj = _max_convolution_fft(a)
            if cur_obj < best_obj:
                best_obj, best_seq = cur_obj, a.copy()

    return best_seq, best_obj

# Parameterised families (power‑law, exponential, linear)
# ----------------------------------------------------------------------


def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ¹‑norm equals target."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = (i + shift) ** (-p)
    return _scale_to_sum(raw, math.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = np.exp(-rate * (i + shift))
    return _scale_to_sum(raw, math.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * i)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _scale_to_sum(raw, math.sqrt(2.0 * n))


def _alternating(arr: np.ndarray) -> np.ndarray:
    """Small‑large alternating ordering (small first)."""
    n = arr.size
    lo, hi = 0, n - 1
    out = np.empty_like(arr)
    for k in range(n):
        if k % 2 == 0:
            out[k] = arr[lo]
            lo += 1
        else:
            out[k] = arr[hi]
            hi -= 1
    return out
