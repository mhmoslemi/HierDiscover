"""
Propose a step‑function (non‑negative vector) whose self‑convolution peak
is as small as possible.

The algorithm consists of four stages that respect a 1000 s budget:

1. **Exploratory sampling** – generate many candidate families (uniform,
   geometric, power‑law, beta‑like, sinusoidal, up‑sampled previous best)
   with lengths in the 8 k–12 k range, normalise them to Σa = √(2·n) and keep
   the best one.

2. **Soft‑max Adam optimisation** – minimise a smooth log‑sum‑exp
   approximation of the convolution peak.  The temperature (“α”) is increased
   up to 1000, the learning‑rate schedule is slower than the original code,
   and a tiny amount of jitter is added to escape flat regions.  The vector
   is projected onto the simplex after each update.

3. **Sub‑gradient descent on the true maximum** – after the Adam phase we
   switch to the exact max‑convolution sub‑gradient (the classic 2·a[k−i]
   direction).  A diminishing step‑size η = 0.2 / √t is used and a short
   simulated‑annealing style random‑swap is performed occasionally.

4. **Fine‑grained random‑swap hill‑climb** – a cheap local search that swaps
   tiny masses between random positions while keeping the simplex constraint.

If time remains after the first full run the whole cycle is repeated with a
new random seed; the best sequence found overall is returned and stored in
the global variable ``height_sequence_1`` for reuse by later runs.

All operations are vectorised (FFT‑based convolutions) and the code avoids
any closures, uses only NumPy and the standard library, and respects the
time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x ≥ 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # Sort v into descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    # Length of linear convolution of a (size n) with w_rev (size 2n-1)
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # The gradient entries correspond to the central n entries of the full conv
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.4          # a bit larger than before
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.05
    max_alpha = 1000.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights and gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Clip extremely large steps – helps stability
        max_step = 0.05 * S
        norm_step = np.linalg.norm(step, ord=np.inf)
        if norm_step > max_step:
            step = step * (max_step / norm_step)

        a -= step

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to avoid stagnation
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # update schedules
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99998   # very slow decay – we want to keep momentum late

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum convolution entry, with occasional
    random mass‑swap (simulated annealing style) to escape shallow minima.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.22  # initial step size factor
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the active index (full 2·a_{k−i} contribution)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        # each pair contributes twice except the middle when i=j
        g[idx_i] = 2.0 * a[idx_j]
        # correct the double‑count on the diagonal (i=j) – remove extra factor
        diag = (idx_i == idx_j)
        g[diag] = a[idx_i[diag]]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random mass‑swap once in a while (simulated‑annealing)
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                # accept if better, otherwise accept with small probability
                if cand_max < cur_max or rng.random() < 0.15:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early exit when progress stalls
        if it % 10000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Tiny random‑swap hill‑climbing (no schedule) – runs until the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.025 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score

def _generate_candidate(rng: np.random.Generator,
                        base_seq: np.ndarray | None,
                        attempts: int) -> tuple[np.ndarray, int]:
    """
    Returns a new candidate vector and its length.
    Families: upsampled previous best, uniform, geometric, power‑law,
    beta‑like, sinusoid‑powered.
    """
    # Choose length – we favour the size of the previous best if we have one
    if base_seq is not None and rng.random() < 0.55:
        scale = rng.uniform(0.9, 1.1)
        n = max(8000, int(base_seq.shape[0] * scale))
    else:
        n = int(rng.integers(8000, 12001))

    fam = rng.integers(0, 6)
    if fam == 0 and base_seq is not None:
        a = _upsample_seq(base_seq, n)
    elif fam == 1:
        a = rng.random(n)                     # uniform random
    elif fam == 2:
        rate = rng.uniform(0.95, 0.9999)       # geometric from both ends
        left = rate ** np.arange(n)
        a = (left + left[::-1]) / 2.0
    elif fam == 3:
        # power‑law symmetric: (i·(n‑i))^p
        p = rng.uniform(0.2, 4.0)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 4:
        # skewed beta‑like shape
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    else:
        # sinusoidal power
        p = rng.uniform(0.5, 4.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p

    a = np.clip(a, 0.0, None)

    # normalise to Σa = √(2·n)
    S = np.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0)
        s = a.sum()
    a = a / s * S

    # optional symmetrisation (helps many runs)
    if rng.random() < 0.45:
        a = _symmetrize(a)

    return a, n


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the Python interpreter
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Retrieve previous best if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 0 – cheap random families (≈10 % of budget)
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.10 * budget_s
    while time.time() < phase0_deadline:
        attempts += 1
        a, n = _generate_candidate(rng, base_seq, attempts)
        cur_score = np.max(_full_convolution(a))
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = a.copy()
            print(f"[init] attempt={attempts:5d} n={n:5d} score={cur_score:.6f}")

    # If nothing survived, fall back to a simple uniform vector
    if best_overall_seq is None:
        n = 9000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_overall_seq = a
        best_overall_score = np.max(_full_convolution(a))
        print("[fallback] using uniform seed")

    # ------------------------------------------------------------------
    # Main optimisation loop – we may repeat the whole pipeline several
    # times if time permits.
    # ------------------------------------------------------------------
    run_id = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 30.0:                     # not enough time for a full run
            break
        run_id += 1
        print(f"[run] start #{run_id}, remaining={remaining:.1f}s")

        # ---- Adam (≈55 % of the remaining time) ----
        t_adam_end = time.time() + 0.55 * remaining
        a_adam, sc_adam = _optimize_adam(best_overall_seq,
                                         t_adam_end,
                                         rng,
                                         enforce_sym=True)
        if sc_adam < best_overall_score:
            best_overall_score = sc_adam
            best_overall_seq = a_adam.copy()
            print(f"[adam] run#{run_id} improved to {best_overall_score:.6f}")

        # ---- Sub‑gradient descent (≈35 % of the rest) ----
        now = time.time()
        if now >= deadline:
            break
        t_sub_end = now + 0.35 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad(best_overall_seq,
                                          t_sub_end,
                                          rng,
                                          enforce_sym=True)
        if sc_sub < best_overall_score:
            best_overall_score = sc_sub
            best_overall_seq = a_sub.copy()
            print(f"[subgrad] run#{run_id} improved to {best_overall_score:.6f}")

        # ---- Final local swap (≈10 % of the rest) ----
        now = time.time()
        if now < deadline:
            a_loc, sc_loc = _local_swap_search(best_overall_seq,
                                               deadline,
                                               rng,
                                               enforce_sym=True)
            if sc_loc < best_overall_score:
                best_overall_score = sc_loc
                best_overall_seq = a_loc.copy()
                print(f"[local] run#{run_id} improved to {best_overall_score:.6f}")

        # ------------------------------------------------------------------
        # If we still have a few seconds left, generate a fresh seed and
        # continue the outer loop; the best sequence is carried over.
        # ------------------------------------------------------------------
        # (Nothing else needed – the while‑loop will start the next run.)

    # ------------------------------------------------------------------
    # Return the result and expose it globally for possible reuse
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
