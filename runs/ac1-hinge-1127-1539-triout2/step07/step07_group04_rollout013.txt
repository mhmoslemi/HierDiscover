"""
Search for a non‑negative step‑function minimising ``evaluate_sequence``.

Algorithm
---------
1.  Initial candidates:
    * up‑sampled best solution from the previous stage,
    * a deterministic “inverse‑sqrt’’ shape (a_i ∝ 1/√(i+1)) – the analytic
      constant‑autocorrelation ansatz,
    * several random families (power‑law, exponential, Dirichlet, …).
2.  For each length in an increasing schedule we perform a two‑phase local
    optimisation:
    * **Projected gradient descent** on a smooth log‑sum‑exp surrogate of the
      maximal convolution entry.  The inverse temperature β is annealed up to
      1e5, momentum is used and periodic jitter helps to escape plateaus.
    * **Mass‑transfer hill climbing** (simulated‑annealing style) that keeps
      the full self‑convolution up‑to‑date in O(n) per move.  The source
      index is sampled proportionally to its contribution to the current
      worst convolution entry, the destination is biased towards small‑mass
      indices.
    The two phases share the remaining wall‑clock budget for the stage.
3.  After the schedule a final high‑β gradient descent polishing step is run.
All vectors are projected onto the ℓ₁‑simplex Σa = √(2 n) after each move,
ensuring the official evaluator sees a valid step function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p), p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max (log‑sum‑exp surrogate).
    Uses momentum, annealed inverse temperature β and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # temperature schedule
    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.3

    # step‑size and momentum
    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter for escaping plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

# Random initialisers (various families)

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled (power‑law, exponential, Dirichlet,
    sine, Gaussian, mixed) to diversify the search space.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine wave (positive half‑wave)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    elif family == 6:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # uniform tiny perturbation
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _inverse_sqrt_initial_sequence(n: int) -> np.ndarray:
    """
    Deterministic shape a_i ∝ 1/√(i+1).  After normalisation Σa = √(2 n)
    this vector is the classical constant‑autocorrelation ansatz.
    """
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(idx)
    target = np.sqrt(2.0 * n)
    a = a * (target / a.sum())
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that transfers mass from a
    source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    The source is chosen proportionally to its contribution to the current
    worst convolution entry; the destination is biased towards low‑mass
    indices.  Temperature cools slowly, allowing many uphill moves early
    on.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9  # initial temperature
    cooling = 0.9994

    while time.time() < deadline:
        # ---- pick one of the current worst convolution indices ----
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # ---- source index weighted by its contribution to entry M ----
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        tot = contrib.sum()
        if tot > 0.0:
            probs = contrib / tot
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # ---- destination index (biased to low‑mass positions) ----
        if rng.random() < 0.4:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # ---- amount of mass to move -----------------------------------------
        max_delta = a[i] * 0.85
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- update convolution (linear part) --------------------------------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # ---- quadratic correction terms ---------------------------------------
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---- Metropolis acceptance --------------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        # ---- temperature decay -------------------------------------------------
        T *= cooling

    return best_a, b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seeding from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – increasingly larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000, 60000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Build a pool of initial candidates for this length
        # --------------------------------------------------------------
        candidates = []

        # 1) upsample previous best if we have one
        if best_seq is not None:
            candidates.append(_upsample_sequence(best_seq, n))

        # 2) deterministic inverse‑sqrt shape
        candidates.append(_inverse_sqrt_initial_sequence(n))

        # 3) a few random starts
        for _ in range(2):
            candidates.append(_random_initial_sequence(n, rng))

        # ensure all are projected onto the simplex
        candidates = [_project_to_simplex(c, target_sum) for c in candidates]

        # --------------------------------------------------------------
        # Phase A – gradient descent on several starts
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 6.0:
            break
        # allocate up to 30 % of remaining budget for this stage's GD
        gd_budget = min(remaining * 0.30, 250.0)

        local_best_score = np.inf

        # split GD budget roughly equally among the candidates
        per_cand = gd_budget / max(1, len(candidates))

            if time.time() >= deadline:

            a_gd = _gradient_descent_opt(c, gd_deadline)
            score = _max_convolution_fft(a_gd)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_gd.copy()

        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time():.1f}] GD improvement for n={n} score={local_best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – hill climbing with exact convolution updates
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 6.0:
            break
        hill_budget = min(remaining * 0.45, 300.0)   # allocate up to ~300 s
        hill_deadline = time.time() + hill_budget

        # Work with the current best of the appropriate length
        if best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = best_seq.copy()
        a_current = _project_to_simplex(a_current, target_sum)
        b_full = _full_convolution(a_current)

        a_hc, b_hc = _hill_climbing(a_current, b_full, hill_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Optional post‑hill‑climb gradient refinement
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 4.0:
            refine_deadline = time.time() + min(remaining * 0.20, 60.0)
            a_ref = _gradient_descent_opt(best_seq.copy(), refine_deadline)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Post‑HC GD refinement n={n} score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        a_final = _gradient_descent_opt(a, deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros (nice output)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
