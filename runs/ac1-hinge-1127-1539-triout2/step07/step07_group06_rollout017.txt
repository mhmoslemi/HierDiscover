def _conv_fft(a):
    n = a.size
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv
```

"""
Improved construction for the step‑function problem.

Idea
----
* The objective after normalising the ℓ¹‑norm to √(2n) is simply
    max_k (a * a)[k].
* The average value of the convolution is
        (∑a_i)² / (2n‑1) = 2n/(2n‑1) ≈ 1.
  So the theoretical lower bound of the objective is 1.
* A distribution that puts more mass near the ends of the interval
  (where the convolution has fewer overlapping terms) and less mass
  in the centre flattens the convolution.  The continuous “arcsine”
  density f(x)∝1/√{x(1‑x)} has exactly this property.
* Discretising this density gives a family of symmetric sequences

        a_i(p) = C(p)·[(i+½)(n‑i‑½)]^{-p} ,   p∈[0,1] ,

  where p=0 is uniform, p=½ is the discrete arcsine shape, and larger
  p concentrate more mass at the borders.
* For a fixed length `n` we can search a single scalar `p` (golden‑section)
  to minimise the peak of the convolution.  The search is cheap
  because the evaluation uses an FFT‑based convolution.
* The best sequence from the 1‑D search is then polished by the
  projected‑gradient method from the reference solution (soft‑max
  approximation of the max‑convolution).  This quickly drives the
  objective very close to the lower bound.
* The whole routine respects the 1000 s budget, tries a few different
  lengths and returns the best candidate found.

The code below is self‑contained, uses only NumPy and the supplied
`evaluate_sequence` function, and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (∑a is assumed to be √(2n))."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Fast objective value (≡ max‑convolution when the ℓ¹‑norm is normalised)."""
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Family of candidate sequences (parametrised by a single exponent p)
# ----------------------------------------------------------------------


def _make_family_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """
    Returns a symmetric sequence a_i ∝ [(i+½)(n‑i‑½)]^{-p}
    scaled so that Σa_i = target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    raw = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        # fallback to uniform – should never happen for p≥0
        raw = np.ones_like(raw)
        raw_sum = raw.sum()
    a = raw * (target_sum / raw_sum)
    return a


# ----------------------------------------------------------------------
# Gradient‑descent polishing (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # gradient w.r.t. a (only the central n entries of conv_rev are needed)
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0.0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.2f} score={best_val:.6f}")

        # schedule for β
        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    The algorithm:

    1.  Try several problem sizes.
    2.  For each size perform a 1‑dimensional golden‑section search
        over the exponent p of the family   a_i ∝ [(i+½)(n‑i‑½)]^{-p}.
    3.  Keep the sequence with the smallest peak.
    4.  Polish the best candidate with the annealed gradient‑descent
        method from the reference solution.
    5.  Trim trailing zeros and return the list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # 1) Scan a few lengths (larger n gives a lower theoretical bound)
    # ------------------------------------------------------------------
    sizes = [2000, 4000, 6000, 8000, 10000, 12000]
    for n in sizes:
        if time.time() > deadline - 5.0:
            break
        target = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # 1‑D golden‑section search on the exponent p ∈ [0, 1]
        # ------------------------------------------------------------------
        lo, hi = 0.0, 1.0
        phi = (np.sqrt(5) - 1) / 2          # ≈0.618
        p1 = hi - phi * (hi - lo)
        p2 = lo + phi * (hi - lo)

        seq1 = _make_family_sequence(n, p1, target)
        obj1 = _max_convolution(seq1)
        seq2 = _make_family_sequence(n, p2, target)
        obj2 = _max_convolution(seq2)

        for _ in range(30):                # enough iterations for convergence
            if time.time() > deadline - 5.0:
                break
            if obj1 < obj2:
                hi = p2
                p2 = p1
                obj2 = obj1
                p1 = hi - phi * (hi - lo)
                seq1 = _make_family_sequence(n, p1, target)
                obj1 = _max_convolution(seq1)
            else:
                lo = p1
                p1 = p2
                obj1 = obj2
                p2 = lo + phi * (hi - lo)
                seq2 = _make_family_sequence(n, p2, target)
                obj2 = _max_convolution(seq2)

        # select the better of the two final candidates
        if obj1 < obj2:
            cur_seq, cur_obj = seq1, obj1
            cur_p = (p1 + lo + hi) / 3.0
        else:
            cur_seq, cur_obj = seq2, obj2
            cur_p = (p2 + lo + hi) / 3.0

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = cur_seq.copy()
            print(f"[{time.time():.1f}] size={n}  p≈{cur_p:.4f}  obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2) Polishing with gradient descent (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra_deadline = time.time() + (deadline - time.time()) * 0.6
        best_seq, best_obj = _annealed_gradient_descent(best_seq, extra_deadline)

    # ------------------------------------------------------------------
    # 3) Remove trailing zeros (they have no effect on the objective)
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best ≈{best_obj:.6f}, length={len(best_seq)}")
    # The returned list is ready for `evaluate_sequence`.
    return best_seq.tolist()
