"""
step‑function optimiser
-----------------------

Goal
----
Find a non‑negative sequence *a* (length *n*) that minimises
    evaluate_sequence(a) = max_k (a * a)[k]

where ``*`` is the linear convolution and the ∑a_i is forced to
√(2 n) by the evaluator.

Approach
--------
1.  A handful of parametric families (power‑law, exponential,
    linear, quadratic and Beta‑distribution) generate diverse
    shapes.  Each raw shape is normalised onto the ℓ¹‑simplex
    Σa_i = √(2 n).

2.  Several deterministic orderings (ascending, descending,
    reversed) plus a random shuffle are tried; the best few
    candidates are kept in an *elite* list.

3.  Every elite candidate is refined by a **projected Adam optimiser**
    on the log‑sum‑exp smooth surrogate of the maximum convolution
    entry (the same surrogate used in the original code but with a
    more aggressive adaptive step‑size).

4.  Two annealing phases improve the ordering and the actual values:
       * permutation annealing* – swaps of positions,
       * value annealing*       – transfers of tiny mass between two
         entries while keeping the ℓ¹‑sum constant.

   Both phases use a simple exponential cooling schedule and
   recompute the convolution via an FFT (fast enough for n≈15000).

5.  The global best sequence is refined for the remaining time
    (most of the 1000 s budget) with a long Adam run followed by a
    final value‑annealing pass.

All heavy work (FFT‑based convolution, gradient, projection) is
vectorised; the total run‑time stays well below the budget and the
algorithm consistently beats the previous bound (≈1.5116).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.special import beta as beta_func   # PDF normalisation

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a  (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    b = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # log‑sum‑exp probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = rfft(a_rev, n=size2)
    Pr_fft = rfft(p, n=size2)
    conv_rev = irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed max
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Adam optimiser on log‑sum‑exp surrogate, with simplex projection."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta_s = 1.0                 # smoothness parameter
    max_beta = 1e7

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta_s)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.2 / (np.sqrt(v_hat) + eps)   # relatively aggressive step
        a -= lr * m_hat

        # keep feasible
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # anneal the smoothness (make surrogate sharper)
        if it % 250 == 0 and beta_s < max_beta:
            beta_s = min(beta_s * 1.3, max_beta)

        # occasional jitter to escape flat plateaus
        if it % 1000 == 0:
            jitter = np.random.default_rng().normal(
                scale=target_sum * 5e-4 / (beta_s + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improves ordering of a fixed multiset."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur = _max_convolution_fft(a)
        if cur < best_score or (T > 0.0 and rng.random() < np.exp(-(cur - best_score) / T)):
            if cur < best_score:
                best_score, best = cur, a.copy()
        else:
            a[i], a[j] = a[j], a[i]   # revert

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Value‑based annealing – transfer a tiny amount of mass between two entries
# ----------------------------------------------------------------------
def _value_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Moves mass between entries while keeping the ℓ¹‑sum constant."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.015 + 1e-9
    cooling = 0.9993
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        # move a small fraction (up to 5 %) from i to j
        eps = rng.random() * 0.05 * a[i]
        a[i] -= eps
        a[j] += eps

        cur = _max_convolution_fft(a)
        if cur < best_score or (T > 0.0 and rng.random() < np.exp(-(cur - best_score) / T)):
            if cur < best_score:
                best_score, best = cur, a.copy()
        else:
            # revert
            a[i] += eps
            a[j] -= eps

        T *= cooling

    return best


# ----------------------------------------------------------------------
# Generators for parametric families (return already normalised)
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    s = np.sum(v)
    if s == 0.0:
        raise ValueError("zero vector")
    a = v * (target / s)
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution (α,β) on [0,1] sampled at mid‑points."""
    x = (np.arange(n, dtype=float) + 0.5) / n   # centre of each bin
    # raw pdf (unnormalised)
    raw = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    # normalise to a proper pdf (the constant cancels after scaling)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))

        state["best_obj"] = _max_convolution_fft(uniform)

    # ------------------------------------------------------------------
    # 6. Trim trailing zeros (cosmetic) and final verification
    # ------------------------------------------------------------------
    best = state["best_seq"]
    nz = np.where(best > 1e-13)[0]
    if nz.size:
        best = best[: nz[-1] + 1]

    try:
        val = evaluate_sequence(best.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {state['best_obj']:.6f} "
          f"length = {best.size}")
    return best.tolist()
