"""
Propose a step‑function minimising ``evaluate_sequence``.

The construction is completely analytic and needs no heavy optimisation
loops – it is based on a discretised exponential decay whose shape is
close to the theoretical optimum of the max‑self‑convolution problem.
The decay rate is tuned by a short projected‑gradient descent that works
directly on the smooth log‑sum‑exp surrogate of the maximum
convolution entry.  The whole routine respects the 1000 s wall‑clock
budget and returns the best vector found (as a plain Python list).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (fast for large n)."""
    return float(_full_convolution(a).max())


def _project_to_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σ x_i = s}.
    """
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _grad_softmax_conv(a: np.ndarray, beta: float):
    """
    Gradient of  (1/β)·log Σ exp(β·c_k)  where c = a*a.
    Returns ``grad`` (shape a) and the raw convolution ``c``.
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    c = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over the convolution entries
    max_c = np.max(c)
    exps = np.exp(beta * (c - max_c))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    size2 = _next_pow2(3 * n - 2)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[: 3 * n - 2]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, c


def _projected_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    beta_max: float = 5e5,
    start_beta: float = 0.5,
):
    """
    Projected gradient descent on the smooth max surrogate.
    Uses a simple momentum scheme and an exponential annealing of β.
    """
    a = a0.copy()
    n = a.size
    beta = start_beta
    step = 0.5 / np.sqrt(n)          # a modest base step size
    momentum = np.zeros_like(a)
    rho = 0.85                       # momentum decay

    while time.time() < deadline:
        grad, _ = _grad_softmax_conv(a, beta)

        # move opposite to the gradient, keep the iterate non‑negative
        momentum = rho * momentum + (1.0 - rho) * grad
        a -= step * momentum
        a = _project_to_simplex(a, s=1.0)

        # anneal the temperature
        if np.random.rand() < 0.02 and beta < beta_max:
            beta = min(beta * 1.30, beta_max)

    return a

# Sequence generators
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Exponential decay a_i ∝ exp(-λ i) with λ tuned roughly to 1/n,
    then projected onto the unit‑simplex.
    """
    lam = 1.0 / n
    a = np.exp(-lam * np.arange(n, dtype=float))
    a = _project_to_simplex(a, s=1.0)
    return a


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Small random perturbation of the exponential start – sufficient to
    escape pathological symmetries while keeping the good shape.
    """
    a = _exponential_sequence(n, rng)
    noise = rng.normal(scale=1e-4, size=n)
    a += noise
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, s=1.0)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a larger grid (keeps the shape)."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new, s=1.0)
    return a_new

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seeding from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = globals()["height_sequence_1"]
        if isinstance(hs, (list, tuple, np.ndarray)):
            tmp = np.asarray(hs, dtype=float)
            if tmp.size > 0:
                tmp = _project_to_simplex(tmp, s=1.0)
                best_seq = tmp
                best_score = _max_convolution_fft(tmp)
                print(f"[{time.time():.1f}] seeded from height_sequence_1, n={tmp.size}, eval={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – increasing with re‑optimisation at each stage
    # ------------------------------------------------------------------
    schedule = [4000, 8000, 16000, 32000, 64000]

    for n in schedule:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise at this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _exponential_sequence(n, rng)

        # --------------------------------------------------------------
        # Phase 1 – projected gradient descent on log‑sum‑exp surrogate
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        gd_deadline = time.time() + 0.45 * remaining
        a_opt = _projected_gradient_descent(a, gd_deadline)
        score = _max_convolution_fft(a_opt)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] GD improvement n={n}, score={score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – simple hill‑climbing to fine‑tune large coefficients
        # --------------------------------------------------------------
        # (move a tiny amount of mass from the most “active” index to a
        #  random low‑mass index; keep the convolution up‑to‑date in O(n))
        cur = a_opt.copy()
        b_full = _full_convolution(cur)
        cur_score = b_full.max()
        best_local = cur.copy()
        best_local_score = cur_score

        hc_deadline = time.time() + 0.35 * remaining
        T = cur_score * 0.1 + 1e-12          # simulated‑annealing temperature
        cool = 0.9996

        while time.time() < hc_deadline:
            # pick the worst convolution index
            bad_idx = np.argmax(b_full)
            i_low = max(0, bad_idx - (n - 1))
            i_high = min(n - 1, bad_idx)
            idx = np.arange(i_low, i_high + 1, dtype=int)
            jdx = bad_idx - idx
            contrib = cur[idx] * cur[jdx]
            total = contrib.sum()
            if total > 0:
                probs = contrib / total
                i = rng.choice(idx, p=probs)
            else:
                i = rng.integers(0, n)

            if cur[i] < 1e-12:
                continue

            # destination – bias towards the smallest entries
            if rng.random() < 0.4:
                inv = 1.0 / (cur + 1e-12)
                j = rng.choice(n, p=inv / inv.sum())
            else:
                j = rng.integers(0, n)

            if i == j:
                continue

            # move a fraction of mass from i to j
            delta = rng.random() * 0.7 * cur[i]
            if delta <= 0.0:
                continue

            # update convolution manually (O(n))
            b_new = b_full.copy()
            b_new[i:i + n] -= 2.0 * delta * cur
            b_new[j:j + n] += 2.0 * delta * cur

            # quadratic correction (i,i) and (j,j)
            b_new[2 * i] += delta * delta
            b_new[2 * j] += delta * delta
            b_new[i + j] -= 2.0 * delta * delta

            new_score = b_new.max()
            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                cur[i] -= delta
                cur[j] += delta
                b_full = b_new
                cur_score = new_score
                if cur_score < best_local_score - 1e-12:
                    best_local_score = cur_score
                    best_local = cur.copy()

            T *= cool

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time():.1f}] HC improvement n={n}, score={best_local_score:.6f}")

        # --------------------------------------------------------------
        # Final short refinement (high β) on the best vector for this size
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 2.0:
            fin_deadline = time.time() + min(remaining * 0.15, 10.0)
            refined = _projected_gradient_descent(best_seq.copy(), fin_deadline, beta_max=1e6)
            ref_score = _max_convolution_fft(refined)
            if ref_score < best_score:
                best_score = ref_score
                best_seq = refined.copy()
                print(f"[{time.time():.1f}] final refine n={n}, score={ref_score:.6f}")

    # ------------------------------------------------------------------
    # If time permits, a final polish on the overall best (any size)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        extra = time.time() + min(0.1 * (deadline - time.time()), 15.0)
        final = _projected_gradient_descent(best_seq.copy(), extra, beta_max=2e6)
        final_score = _max_convolution_fft(final)
        if final_score < best_score:
            best_score = final_score
            best_seq = final.copy()
            print(f"[{time.time():.1f}] final polish score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for nicer output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a modest exponential (should never be reached)
        n = 5000
        best_seq = _exponential_sequence(n, rng)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (for debugging purposes)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"evaluation error: {exc}")

    return best_seq.tolist()
