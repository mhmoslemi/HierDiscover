def _subgradient_descent(a, deadline):
    a = a.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    iteration = 0
    while time.time() < deadline:
        # compute conv
        size = _next_pow2(2*n-1)
        fa = np.fft.rfft(a, n=size)
        b = np.fft.irfft(fa*fa, n=size)[:2*n-1]
        max_val = b.max()
        M = np.flatnonzero(b == max_val)[0]  # can pick first

        # compute gradient g
        indices = np.arange(n)
        j = M - indices
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step size diminishing
        step = 0.5 * target / (np.sqrt(iteration + 1) + 1)
        a -= step * g
        a = _project_to_simplex(a, target)
        iteration += 1
    return a

import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _project_to_simplex(v, s):
    # Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution_fft(a):
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _full_convolution(a):
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _compute_score(a):
    n = a.size
    if n == 0:
        return np.inf
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)

def _random_initial_sequence(n, rng):
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine wave (positive half‑wave)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

def _upsample_sequence(a, new_n):
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _gradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.4

    step_base = target * 0.1

    rho = 0.85

            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        if it % 200 == 0:

            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

            beta = min(beta * beta_factor, beta_max)

        it += 1

def _hill_climbing(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9
    cooling = 0.9996

    while time.time() < deadline:
        # identify a worst convolution index
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: the one with largest contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        if contrib.size == 0:
            continue
        i = idx[np.argmax(contrib)]

        # destination index: biased towards small entries
        if rng.random() < 0.5:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to a fraction of the source)
        max_delta = a[i] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (linear part)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # accept if improvement or by Metropolis rule
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

def _subgradient_descent(a0, deadline):

    n = a.size
    target = np.sqrt(2.0 * n)
    it = 0
    while time.time() < deadline:
        size = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]
        max_val = conv.max()
        M = np.flatnonzero(conv == max_val)[0]

        idx = np.arange(n)
        j = M - idx
        mask = (j >= 0) & (j < n)

        g[mask] = 2.0 * a[j[mask]]

        step = target / (np.sqrt(it + 1) + 1.0) * 0.5
        a -= step * g
        a = _project_to_simplex(a, target)

        it += 1
    return a

# ----------------------------------------------------------------------
# Basic utilities (FFT size, simplex projection, convolution helpers)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi‑et‑al 2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT version)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _compute_score(a: np.ndarray) -> float:

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)

        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a larger grid, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)

    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    return _project_to_simplex(a_new, target)

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of (1/β)·log Σ exp(β·(a*a)) w.r.t. a."""
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    total_time = deadline - time.time()
    early_budget = min(0.2 * total_time, 200.0)          # at most 200 s for phase 1
    n_restarts = max(1, int(early_budget // 20.0))      # roughly 20 s per restart
    per_restart = early_budget / max(1, n_restarts)

    for r in range(n_restarts):
        if time.time() >= deadline:
            break
        if best_seq is not None and best_seq.size == n_early:
            a_start = best_seq.copy()
        elif best_seq is not None:
            a_start = _upsample_sequence(best_seq, n_early)
        else:
            a_start = _random_initial_sequence(n_early, rng)

        a_start = _project_to_simplex(a_start, target_early)

        gd_deadline = time.time() + per_restart
        a_gd = _gradient_descent_opt(a_start, gd_deadline)
        score = _compute_score(a_gd)

        if score < best_score:
            best_score = score
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] Early GD improvement n={n_early} score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upsample to a larger problem size
    # ------------------------------------------------------------------
    n_final = 20000
    target_final = np.sqrt(2.0 * n_final)
    if best_seq is None:
        a_cur = _random_initial_sequence(n_final, rng)
    else:
        a_cur = _upsample_sequence(best_seq, n_final)
    a_cur = _project_to_simplex(a_cur, target_final)

    b_full = _full_convolution(a_cur)

    # ------------------------------------------------------------------
    # Phase 3 – intensive hill climbing (majority of the remaining budget)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0          # keep a few seconds for polishing
    a_hc, b_hc = _hill_climbing(a_cur, b_full, hill_deadline)
    score_hc = b_hc.max()
    if score_hc < best_score:
        best_score = score_hc
        best_seq = a_hc.copy()
        print(f"[{time.time():.1f}] Hill‑climb final improvement n={n_final} score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final polishing with a sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        polishing_deadline = deadline
        a_polished = _subgradient_descent(best_seq.copy(), polishing_deadline)
        final_score = _compute_score(a_polished)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_polished.copy()
            print(f"[{time.time():.1f}] Polishing improvement score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy output
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Official evaluation (only for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
