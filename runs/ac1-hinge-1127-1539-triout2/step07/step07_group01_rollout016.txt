def _correlate_fft(d, a):
    n = len(a)
    a_rev = a[::-1]
    conv_len = len(d) + n - 1
    f_len = _next_pow2(conv_len)

    conv = np.fft.irfft(D * A_rev, n=f_len)[:conv_len]

def _adam_optimize_flat(a0, deadline, rng):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    # initialize p from a0 (nonzero)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p); v = np.zeros_like(p)
    beta1 = 0.9; beta2 = 0.999; eps_adam=1e-8; lr=0.5

    best_a = a0.copy()
    # compute initial loss to track improvement?
    b = _full_convolution_fft(best_a)
    avg = (best_a.sum())**2/(2*n-1)
    loss = np.sum((b - avg)**2)
    best_loss = loss
    best_peak = b.max()

    it = 0
    while time.time() < deadline:
        it += 1
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # compute loss and gradient
        b = _full_convolution_fft(a)
        avg = (S**2)/(2*n - 1)   # sum a = S constant
        d = b - avg
        # gradient wrt a
        grad_a = 2.0 * _correlate_fft(d, a)

        # compute loss
        loss = np.sum(d**2)

        # update best
        if loss < best_loss:
            best_loss = loss
            best_a = a.copy()
            best_peak = b.max()

        # gradient wrt logits
        mu = np.dot(a, grad_a) / S
        grad_p = a * (grad_a - mu)

        # optional clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 10.0 * S:
            grad_p = grad_p * (10.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_p
        v = beta2 * v + (1 - beta2) * (grad_p**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        lr *= 0.9997

        if it % 500 == 0:
            p += rng.normal(scale=0.005, size=n)

    final_peak = _max_convolution(best_a)
    return best_a, final_peak

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0
    # best
    best_seq = None
    best_peak = np.inf

    # Setup lengths for exploration: maybe [2000,4000,6000,8000,12000,20000]
    # We'll sample n each iteration within range.

    # Phase exploration
    explore_time = end_time - 300.0  # keep 300 sec for refinements
    attempt = 0
    while time.time() < explore_time:
        attempt += 1
        # Choose length randomly
        if rng.random() < 0.7:
            n = int(rng.integers(2000, 8000))
        else:
            n = int(rng.integers(8000, 20000))
        a = _random_candidate(n, rng, base_seq=None)
        # Compute peak
        peak = _max_convolution(a)
        if peak < best_peak:
            best_peak = peak
            best_seq = a
            best_n = n
            print(f"[explore] attempt={attempt} n={n}  new best peak={best_peak:.6f}")

    # If no candidate yet, fallback
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_peak = _max_convolution(best_seq)
        print("[fallback] using uniform default")

    # Phase refinement: scaling and flat optimizer
    # define scaling targets: current n, 1.5x, 2x, up to 20000 maybe.
    targets = [best_n]
    if best_n < 20000:
        targets.append(min(20000, best_n * 2))
    if best_n < 15000:
        targets.append(int(best_n * 1.5))
    targets = sorted(set(targets))

    for target_n in targets:
        remaining = end_time - time.time()
        if remaining < 30:
            break
        # allocate slice
        slice_time = min(remaining * 0.4, 180.0)  # allocate up to 3 minutes per target
        # scale
        a_scaled = _scale_sequence(best_seq, target_n)
        # normalize to sqrt(2n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        total = a_scaled.sum()
        if total == 0:
            a_scaled = np.full(target_n, 1.0)
            total = target_n
        a_scaled = a_scaled * (S / total)
        # run optimizer
        a_opt, peak = _adam_optimize_flat(a_scaled, time.time() + slice_time, rng)
        if peak < best_peak:
            best_peak = peak
            best_seq = a_opt
            best_n = target_n
            print(f"[refine] target_n={target_n} improved peak={best_peak:.6f}")

    # final evaluation
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f}  internal peak = {best_peak:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq

def _local_opt(seq, iterations, rng):
    a = seq.copy()
    n = len(a)
    S = a.sum()
    for i in range(iterations):
        # pick two indices i, j randomly
        i1 = rng.integers(0, n)

"""
Step‑function optimizer (flat‑convolution version).

The objective used by `evaluate_sequence` (with the usual ℓ₁‑normalisation
∑a_i = √(2 n)) is simply the peak of the self‑convolution,
    score = max_k (a * a)_k .
If a is normalised this way, the denominator in the original formula cancels
and the score equals the peak value.

The algorithm works in three stages:

1. **Random exploration** – many feasible candidates are drawn from a few
   parametric families (exponential, gamma, power‑law, Dirichlet‑like,
   mixed spikes).  All candidates are normalised so that their ℓ₁‑norm equals
   √(2 n).  The best peak found so far is stored globally.

2. **Flatten‑convolution Adam refinement** – starting from the best candidate
   (and from its linearly interpolated versions of larger lengths) we minimise
   the squared deviation of the full convolution from its average value:
        L(a) = Σ_k ( (a*a)_k – μ )² ,   μ = (∑a)² / (2 n‑1) .
   The exact gradient of *L* is obtained analytically via FFTs and the update
   is performed on the simplex using a log‑softmax (mirrored‐Adam) parametrisation.
   This drives the convolution towards a flat shape, which provably pushes the
   maximum close to its theoretical lower bound 2 n/(2 n‑1) ≈ 1.

3. **Final sanity check** – the best sequence is evaluated with the provided
   ``evaluate_sequence`` function, stored in the global variable
   ``height_sequence_1`` and returned.

The whole routine respects the supplied time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)_k."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _correlate_fft(d: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation of d (length 2n‑1) with a (length n) – returns an array of length n.
    Uses FFTs and extracts the central slice conv[n‑1 : n‑1+n].
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = d.shape[0] + n - 1
    f_len = _next_pow2(conv_len)
    D = np.fft.rfft(d, n=f_len)
    A_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(D * A_rev, n=f_len)[:conv_len]
    return conv[n - 1:n - 1 + n]


# ----------------------------------------------------------------------
# Random initialisation families
# ----------------------------------------------------------------------
def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.exponential(scale=1.0, size=n)


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.9)
    return rng.gamma(shape, 1.0, size=n)


def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    return (d + offset) ** (-p)


def _init_dirichlet_like(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.05, 0.5)
    raw = rng.gamma(alpha, 1.0, size=n)   # Dirichlet after normalisation
    return raw


def _init_mixed_spikes(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse high spikes on a small exponential background."""
    background = rng.exponential(scale=0.5, size=n)
    spikes = np.zeros(n)
    k = max(1, n // 1000)
    idx = rng.choice(n, size=k, replace=False)
    spikes[idx] = rng.uniform(5.0, 20.0, size=k)
    return background + spikes


def _initial_candidate(n: int,
                      rng: np.random.Generator,
                      warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible candidate of length n (ℓ₁‑norm = √(2 n)).
    Occasionally warm‑starts from a global best sequence.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with modest probability
    if warm_seq is not None and rng.random() < 0.15:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        return a * (S / total)

    style = rng.choice(
        ["exponential", "gamma", "powerlaw", "dirichlet", "mixed"]
    )
    if style == "exponential":
        raw = _init_exponential(n, rng)
    elif style == "gamma":
        raw = _init_gamma(n, rng)
    elif style == "powerlaw":
        raw = _init_powerlaw(n, rng)
    elif style == "dirichlet":
        raw = _init_dirichlet_like(n, rng)
    else:  # mixed spikes
        raw = _init_mixed_spikes(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Adam optimiser minimising the *flattening* loss
# ----------------------------------------------------------------------
def _adam_optimize_flat(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax) that minimises
        L(a) = Σ_k ( (a*a)_k – μ )² ,
    with μ = (∑a)²/(2 n‑1) (constant because ∑a = √(2 n)).
    Returns the best sequence seen and the corresponding peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # fixed ℓ₁‑norm
    eps = 1e-12

    # initialise logits from the feasible a0
    soft0 = np.maximum(a0 / S, eps)
    soft0 /= soft0.sum()
    p = np.log(soft0)                    # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.6

    # constant µ for the current length
    mu_const = (S * S) / (2.0 * n - 1.0)

    best_a = a0.copy()
    best_peak = _max_convolution(best_a)
    best_loss = np.inf

    it = 0
    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- compute loss and its gradient w.r.t. a -----
        b = _full_convolution_fft(a)          # length 2n‑1
        d = b - mu_const                       # deviation from the average
        loss = np.sum(d * d)

        # gradient of loss w.r.t. a (exact, via correlation)
        grad_a = 2.0 * _correlate_fft(d, a)

        # keep the best (by peak, not by loss)
        peak = b.max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()
            best_loss = loss

        # ----- chain rule: gradient w.r.t. logits -----
        mu = np.dot(a, grad_a) / S               # ⟨a,∂L/∂a⟩ / S
        grad_p = a * (grad_a - mu)

        # optional clipping for stability
        max_abs = np.max(np.abs(grad_p))
        if max_abs > 10.0 * S:
            grad_p = grad_p * (10.0 * S / max_abs)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        lr *= 0.9998

        # occasional jitter to escape plateaus
        if it % 700 == 0:
            p += rng.normal(scale=0.008, size=n)

    final_peak = _max_convolution(best_a)
    return best_a, final_peak


# ----------------------------------------------------------------------
# (fallback) Adam optimiser that directly minimises the peak via a τ‑softmax
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = mask.shape[0] + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize_max(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that directly minimises the (soft) peak of the
    self‑convolution.  Used as a short fine‑tuning after the flattening phase.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    soft = np.maximum(a0 / S, 1e-12)
    soft /= soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.4

    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # stability clipping
        max_abs = np.max(np.abs(grad_p))
        if max_abs > 5.0 * S:
            grad_p = grad_p * (5.0 * S / max_abs)

        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        tau *= 0.9995
        lr *= 0.9999

        if it % 600 == 0:
            p += rng.normal(scale=0.01, size=n)

    final_peak = _max_convolution(best_a)
    return best_a, final_peak

            a_scaled = np.full(target_n, 1.0)
            total = target_n
        a_scaled = a_scaled * (S / total)

        # ----- flatten‑convolution optimisation -----
        a_opt, peak = _adam_optimize_flat(
            a_scaled,
            time.time() + slice_time,
            rng
        )
        # short fine‑tuning with the direct‑peak optimiser (optional)
        fine_time = time.time() + min(30.0, slice_time * 0.15)
        a_opt, peak = _adam_optimize_max(a_opt, fine_time, rng)

        if peak < best_peak:
            best_peak = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n:5d}  improved peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal peak = {best_peak:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
