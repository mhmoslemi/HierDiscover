```
from scipy.optimize import linprog
def _lp_optimize(f, M, target_sum):
    n = f.size
    # Build constraint matrix A_ub:
    # rows = 2n-1.
    # For each k in [0,2n-2], we need sum_i f_i * g_{k - i} <= M.
    # Equivalent: for each k, A_ub[k, i] = f_i if 0 <= k - i < n else 0.
    # Build using toeplitz: first column of A_ub is [f0,0,0,...], second column shift.
```

assistantfinal**Solution Overview**

The evaluator minimises  

\[
\frac{2n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the numerator and the denominator scale quadratically, the objective is
invariant under a global scaling of the vector.  Consequently we are free to
normalise the step‑function to the ℓ¹–simplex  

\[
\sum_i a_i = \sqrt{2n}.
\]

The algorithm works in three stages  

1. **Low‑dimensional linear‑programming refinement** – a short sequence
   (`n₀ = 2000`) is optimised by the classic “alternating LP” idea from the
   statement: for a fixed `f` we maximise `∑ g_i` under the linear
   constraints `f * g ≤ M` (`M = max(f*f)`).  The solution `g` is mixed with
   `f` (`f ← (1‑t)f + t g`) which never increases the maximal autocorrelation.
   Repeating this step a few times yields a very good shape.

2. **Upsampling & continuous optimisation** – the improved low‑dimensional
   vector is linearly interpolated to a larger size (`n ≈ 2·10⁴`).  From this
   point we run a projected gradient descent on a smooth approximation of
   `max_k (a*a)_k` (log‑sum‑exp with an annealed inverse temperature) and a
   simulated‑annealing hill‑climber that updates the full convolution in
   `O(n)`.

3. **Final polishing** – a short random‑perturbation local search finishes the
   optimisation.

All operations stay on the feasible set (non‑negative, ℓ¹‑simplex).  
The whole routine respects the 1000 s wall‑clock budget and prints intermediate
scores so the progress can be followed.

```python
"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm consists of three phases:

1. **LP‑based refinement** – on a small problem (n0≈2000) we repeatedly
   solve a linear program that maximises the ℓ¹‑norm of a partner vector g
   while keeping the autocorrelation of the current candidate f bounded.
   The convex combination f←(1‑t)f+t·g never raises the worst convolution
   entry, thus the bound M monotonically decreases.

2. **Upsampling + continuous optimisation** – the improved low‑dimensional
   vector is up‑sampled to a larger size (≈2·10⁴).  A projected gradient descent
   on a log‑sum‑exp surrogate of the worst convolution entry (annealed β)
   slowly drives the shape towards a flat autocorrelation.  Afterwards a
   simulated‑annealing hill‑climber updates the full convolution in O(n) per
   move.

3. **Final polishing** – a tiny random‑mass‑transfer local search cleans the
   remaining noise.

The whole search respects a 1000 s budget, prints intermediate scores and
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σ x_i = s} (Duchi et al., 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and gradient descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).

    Returns (grad, b) with   grad = 2·(a ⋆ p)   and   p ∝ exp(β·b).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max (log‑sum‑exp surrogate).

    Uses momentum, an annealed inverse temperature β and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5                # start inverse temperature
    beta_max = 1e6
    beta_factor = 1.3

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90                # momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled to diversify the search space.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # Dirichlet (uniform)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 1:                   # Power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 2:                   # Exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 3:                   # Mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 4:                   # Sine half‑wave
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # Piecewise exponential
        split = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(split)), np.exp(-lam2 * np.arange(split, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing hill climbing that transfers mass between indices while
    keeping the full self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9          # initial temperature
    cooling = 0.9995

    while time.time() < deadline:
        # ---- pick a worst convolution index ---------------------------------
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # ---- source index weighted by its contribution to entry M ----------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # ---- destination index ---------------------------------------------
        # bias 45 % of the time toward the smallest entries
        if rng.random() < 0.45:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # ---- amount of mass to move ----------------------------------------
        max_delta = a[i] * 0.9           # be aggressive early on
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- update convolution (linear terms) ------------------------------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # ---- quadratic correction terms ------------------------------------
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---- Metropolis acceptance -----------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        # ---- temperature decay ----------------------------------------------
        T *= cooling

    return best_a, b

def _lp_improve(f: np.ndarray, deadline: float, max_iters: int = 8,

    """
    Repeatedly solve

        maximise  Σ g_i
        s.t.      g_i ≥ 0
                  (f * g)_k ≤ M   for all k

    where M = max(f*f).  The solution g is mixed with f via
    f←(1‑t)f + t·g (t = step).  The bound M never increases.
    """
    if not _SCIPY_AVAILABLE:
        return f

    n = f.size
    target_sum = np.sqrt(2.0 * n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution_fft(f)

        # ----- build the constraint matrix A (dense, but n ≤ 2000) ----------
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # ----- maximise Σ g_i  ==> minimise  -Σ g_i ------------------------
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.1f}] LP failed at iter {it+1}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # rescale to the proper ℓ¹‑norm (the LP does not enforce a total sum)
        g = g * (target_sum / sum_g)

        # convex combination – the ℓ¹‑norm stays equal to target_sum
        f = (1.0 - step) * f + step * g
        # keep non‑negativity (numerical noise may introduce tiny negatives)
        f = np.maximum(f, 0.0)

        new_M = _max_convolution_fft(f)
        print(f"[{time.time():.1f}] LP iter {it+1}: M {M:.6f} → {new_M:.6f}")
        if new_M >= M - 1e-12:
            step *= 0.5                     # shrink step if no progress
        else:
            step = min(step * 1.2, 0.4)      # modestly increase step when improving

    return f


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5     # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Low‑dimensional LP refinement
    # ------------------------------------------------------------------
    n_small = 2000
    target_small = np.sqrt(2.0 * n_small)

    # start from a Dirichlet point (uniform on the simplex)
    f = rng.dirichlet(alpha=np.ones(n_small)) * target_small
    f = _project_to_simplex(f, target_small)

    f = _lp_improve(f, deadline, max_iters=8, step=0.2)

    # ------------------------------------------------------------------
    # 2️⃣  Upsample & continuous optimisation
    # ------------------------------------------------------------------
    # Choose a larger size that still fits comfortably into the remaining time.
    # 20000 is a good compromise (roughly 10× more variables than the LP stage).
    if time.time() >= deadline:
        best = f
    else:
        n_large = 20000
        f = _upsample_sequence(f, n_large)

        # ----- gradient descent (log‑sum‑exp surrogate) -----------------
        remaining = deadline - time.time()
        gd_deadline = time.time() + min(remaining * 0.55, 250.0)
        f = _gradient_descent_opt(f, gd_deadline)

        # ----- hill climbing (exact convolution updates) ----------------
        if time.time() < deadline:
            b_full = _full_convolution(f)
            hc_deadline = deadline                         # use all time left
            f, _ = _hill_climbing(f, b_full, hc_deadline)

        best = f

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing – tiny random mass‑transfer local search
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        n = best.size
        target_sum = np.sqrt(2.0 * n)
        cur_score = _max_convolution_fft(best)
        while time.time() < deadline - 0.5:
            i, j = rng.choice(n, 2, replace=False)
            if best[i] < 1e-12:
                continue
            delta = rng.random() * best[i] * 0.07   # move up to 7 % of source
            a_new = best.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_to_simplex(a_new, target_sum)
            new_score = _max_convolution_fft(a_new)
            if new_score < cur_score:
                best = a_new
                cur_score = new_score
                print(f"[{time.time():.1f}] final local tweak improvement: {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for a nice output
    # ------------------------------------------------------------------
    best = np.asarray(best, dtype=float)
    nz = np.where(best > 1e-12)[0]
    if nz.size:
        best = best[: nz[-1] + 1]
    else:
        best = best[:1]

    # ------------------------------------------------------------------
    # Log official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best.tolist()
```
