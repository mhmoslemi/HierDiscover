"""
Propose a step‐function (non‑negative vector) with a small self‑convolution peak.

Algorithm
---------
1. **Initial candidates** – generate many analytic families (uniform,
   geometric, power‑law, beta‑like, sinusoidal) of various lengths, enforce
   symmetry, normalise to Σa = √(2·n) and keep the best one.
2. **Active‑set LP improvement** – for the current best vector `f`,
   compute its self‑convolution `M = max(f*f)`.  Take the indices where the
   convolution is within a small relative tolerance of `M` (at most a few
   dozen) and build the corresponding linear constraints
   `f * b ≤ M`.  Solve the LP   maximise Σb  subject to those constraints
   and `b ≥ 0`.  If the LP yields a larger ℓ₁‑norm, scale `b` back to the
   target sum `S = √(2·n)` and replace `f` by a convex combination
   `(1‑t) f + t b`.  A tiny line‑search over `t∈[0,1]` is performed.
   This step is repeated while time permits.
3. **Fine‑grained Adam optimisation** – minimise a soft‑max
   approximation of the convolution peak using projected Adam.
4. **Local random‑swap hill‑climbing** – cheap swaps of tiny masses
   between random positions to polish the final result.

All steps respect the 1000 s budget, keep the vector non‑negative and
store the best sequence globally as ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Projected Adam optimiser (soft‑max of convolution)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising a soft‑max approximation of the max‑conv."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.04
    max_alpha = 200.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.012:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple random‑swap hill‑climbing
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Hill‑climbing that swaps tiny masses between random positions."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Active‑set linear‑programming improvement
# ----------------------------------------------------------------------
def _active_set_lp(f: np.ndarray,
                   M: float,
                   active_ks: np.ndarray,
                   target_sum: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, bool]:
    """
    Solve  max Σb  s.t.  f * b ≤ M (for selected ks)  and  b ≥ 0.
    Returns the scaled vector g (with sum = target_sum) and a flag
    indicating whether an improvement (sum(b) > target_sum) was found.
    """
    n = f.shape[0]
    rows = len(active_ks)

    # Build sparse constraint matrix (rows × n)
    A_lil = sp.lil_matrix((rows, n), dtype=float)
    for r, k in enumerate(active_ks):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        cols = np.arange(j_start, j_end + 1, dtype=int)
        vals = f[k - cols]                     # f_i where i = k‑j
        A_lil[r, cols] = vals
    A_ub = A_lil.tocsr()
    b_ub = np.full(rows, M, dtype=float)

    # maximise Σb  => minimise -Σb
    c = -np.ones(n, dtype=float)

    # Solve via HiGHS (fast interior‑point)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return f, False

    b = res.x
    sum_b = b.sum()
    if sum_b <= target_sum * (1.0 + 1e-9):
        # No genuine improvement
        return f, False

    alpha = target_sum / sum_b
    g = b * alpha
    return g, True


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin

    # --------------------------------------------------------------
    # Phase 0 – generate a pool of cheap analytic candidates
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    attempts = 0

    # Try to re‑use the previous best if available
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    phase0_deadline = start_time + 0.25 * budget_s
    while time.time() < phase0_deadline:
        attempts += 1

        # length choice
        if base_seq is not None and rng.random() < 0.55:
            scale = rng.uniform(0.9, 1.1)
            n = max(6000, int(base_seq.shape[0] * scale))
            a = _upsample_seq(base_seq, n)
        else:
            n = int(rng.integers(6000, 12001))

        fam = rng.integers(0, 6)
        if fam == 0 and base_seq is not None:
            a = _upsample_seq(base_seq, n)
        elif fam == 1:
            a = rng.random(n)                                 # uniform random
        elif fam == 2:
            rate = rng.uniform(0.95, 0.9999)                   # geometric from both ends
            left = rate ** np.arange(n)
            a = (left + left[::-1]) * 0.5
        elif fam == 3:
            p = rng.uniform(0.2, 4.0)                         # symmetric power‑law
            idx = np.arange(n, dtype=float)
            a = ((idx + 1) * (n - idx)) ** p
        elif fam == 4:
            p1 = rng.uniform(0.1, 5.0)
            p2 = rng.uniform(0.1, 5.0)
            x = np.linspace(0, 1, n)
            a = (x ** p1) * ((1 - x) ** p2)
        else:
            p = rng.uniform(0.5, 4.0)                         # sinusoidal power
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p

        a = np.clip(a, 0.0, None)

        # Occasionally enforce mirror symmetry
        if rng.random() < 0.4:
            a = _symmetrize(a)

        # Normalise to Σa = √(2·n) (any scaling works for the objective)
        S_target = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S_target

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

    # Fallback in the unlikely case no candidate was generated
    if best_seq is None:
        n = 8000
        best_seq = rng.random(n)
        best_seq = best_seq / best_seq.sum() * np.sqrt(2.0 * n)
        best_score = np.max(_full_convolution(best_seq))

    # ------------------------------------------------------------------
    # Phase 1 – active‑set LP refinement (majority of remaining budget)
    # ------------------------------------------------------------------
    S_target = np.sqrt(2.0 * best_seq.shape[0])

    # Ensure strictly positive entries to keep LP bounded
    tiny = 1e-12
    best_seq = np.maximum(best_seq, tiny)
    best_seq = best_seq / best_seq.sum() * S_target

    lp_iter = 0
    while time.time() < deadline - 200.0:   # leave ~200 s for later phases
        lp_iter += 1
        conv = _full_convolution(best_seq)
        M = conv.max()

        # tolerance for the active set (relative)
        rel_tol = 0.001
        thresh = M * (1.0 - rel_tol)
        cand_ks = np.where(conv >= thresh)[0]
        max_active = 50
        if cand_ks.size > max_active:
            # keep the indices with the largest values
            top_idx = np.argpartition(conv[cand_ks], -max_active)[-max_active:]
            cand_ks = cand_ks[top_idx]

        # Solve the LP on this active set
        g, improved = _active_set_lp(best_seq, M, cand_ks, S_target, rng)
        if not improved:
            # No progress – break out of LP loop
            print(f"[lp] iteration={lp_iter} no improvement, stop")
            break

        # Small line‑search over convex combinations of f and g
        best_local_seq = best_seq
        best_local_score = best_score
        for t in np.linspace(0.0, 1.0, 11):
            cand = (1.0 - t) * best_seq + t * g
            cand_score = np.max(_full_convolution(cand))
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_seq = cand.copy()
        if best_local_score < best_score - 1e-8:
            print(f"[lp] iteration={lp_iter} improved {best_score:.6f} → {best_local_score:.6f}")
            best_seq = best_local_seq
            best_score = best_local_score
        else:
            # Even if the line‑search failed, keep g if it is better alone
            g_score = np.max(_full_convolution(g))
            if g_score < best_score - 1e-8:
                print(f"[lp] iteration={lp_iter} g alone improved {best_score:.6f} → {g_score:.6f}")
                best_seq = g
                best_score = g_score
            else:
                # No improvement – stop the LP stage
                print(f"[lp] iteration={lp_iter} no improvement after line‑search")
                break

        # Keep the vector strictly positive for the next LP
        best_seq = np.maximum(best_seq, tiny)
        best_seq = best_seq / best_seq.sum() * S_target

    # ------------------------------------------------------------------
    # Phase 2 – short Adam refinement on the current best sequence
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline - 60.0:
        remaining = deadline - now
        adam_deadline = now + 0.4 * remaining
        a_adam, sc_adam = _optimize_adam(best_seq, adam_deadline, rng, enforce_sym=False)
        if sc_adam < best_score:
            print(f"[adam] improved to {sc_adam:.6f}")
            best_seq = a_adam
            best_score = sc_adam

    # ------------------------------------------------------------------
    # Phase 3 – final random‑swap local search
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_loc, sc_loc = _local_swap_search(best_seq, deadline, rng, enforce_sym=False)
        if sc_loc < best_score:
            print(f"[local] improved to {sc_loc:.6f}")
            best_seq = a_loc
            best_score = sc_loc

    # --------------------------------------------------------------
    # Return the result and expose it globally
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
