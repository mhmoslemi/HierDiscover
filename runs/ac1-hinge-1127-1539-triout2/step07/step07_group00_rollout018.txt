def linear_program_step(f, M):
   n = len(f)
   # Build constraint matrix A shape (2n-1,n)
   rows = 2*n-1
   # Use sparse representation
   data, row_inds, col_inds = [], [], []
   for k in range(rows):
       i_min = max(0, k-(n-1))
       i_max = min(k, n-1)
       for i in range(i_min, i_max+1):
           j = k - i
           # constraint: f[i]*g[j] appears
           row_inds.append(k)
           col_inds.append(j)
           data.append(f[i])
   # Build sparse matrix
   A = sparse.csr_matrix((data, (row_inds, col_inds)), shape=(rows, n))
   # LP: maximize sum(g)
   c = -np.ones(n)  # Since linprog minimizes c^T x, we set c = -1 to maximize sum
   # bounds: g_j >=0
   bounds = [(0, None) for _ in range(n)]
   # constraints: A @ g <= M
   from scipy.optimize import linprog
   res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
   if not res.success: raise
   g = res.x
   return g

"""
Propose a candidate step‑function that (approximately) minimizes the
``evaluate_sequence`` objective.

The algorithm consists of three phases:

1. **Linear‑programming refinement** – Starting from a random
   non‑negative simplex vector we repeatedly solve a linear program

        max  Σ g_i
        s.t. conv(f, g) ≤ M
             g_i ≥ 0

   where *f* is the current vector and *M* = max(conv(f, f)).
   The solution tries to pack as much ℓ¹‑mass as possible under the
   same convolution bound.  After scaling the solution to the required
   ℓ¹‑norm we mix it with *f* (mixing factor gradually decays).  This
   step is cheap (a dense LP of size roughly (2 n)×n) and quickly
   drives the maximum convolution value down.

2. **Projected smooth‑max gradient descent** – From the best vector
   obtained in phase 1 we run a few short runs of the soft‑max
   gradient descent already used in the reference implementation, to
   polish the shape.

3. **Exact hill‑climbing** – Finally we perform a simulated‑annealing
   style hill‑climb that updates the full convolution in O(n) per
   move, using the same update rule as the reference code.

All phases respect the wall‑clock budget (default 1000 s).  The
function returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT based)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth maximum.
    Uses momentum, β‑annealing and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 2.0
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.2
    rho = 0.90
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0005 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a

# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # slower cooling for finer search

    while time.time() < deadline:
        # pick a currently worst convolution position
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: 30 % bias towards indices with small mass
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Linear‑programming refinement (convex‑quadratic step)
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, M: float, target_sum: float, mix: float) -> np.ndarray:
    """
    One LP refinement step:
        max Σ g_i  s.t. conv(f, g) ≤ M,  g ≥ 0.
    The optimal g is then scaled to the required ℓ¹‑norm and mixed
    with f using the factor ``mix``.
    Returns the new vector (already projected onto the simplex).
    """
    n = f.shape[0]

    # Build the dense Toeplitz matrix for conv(f, g)
    # First column = [f, 0, ..., 0]  (length 2n‑1)
    # First row    = [f0, 0, ..., 0] (length n)
    col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=float)])
    try:
        # SciPy may not be available – the import is performed lazily
        from scipy.linalg import toeplitz
        from scipy.optimize import linprog
    except Exception as exc:
        # If SciPy is missing we simply return the input vector
        print(f"LP step skipped (SciPy not available): {exc}")
        return f

    A = toeplitz(col, row)                      # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # Linear programme: minimise   -Σ g_i   (i.e. maximise Σ g_i)
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n

    # Use HiGHS which is fast for medium‑sized dense LPs
    res = linprog(
        c=c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method='highs',
        options={'presolve': True}
    )

    if not res.success:
        # If LP fails we fall back to the current vector
        print(f"LP not successful: {res.message}")
        return f

    g = res.x
    sum_g = np.sum(g)
    if sum_g <= 0.0:
        return f

    # Scale to required ℓ¹‑norm and mix
    g_scaled = g * (target_sum / sum_g)
    new_f = (1.0 - mix) * f + mix * g_scaled
    new_f = _project_to_simplex(new_f, target_sum)
    return new_f


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – we work with n = 2000 (the length that gave the
    # best result in the reference run).  The LP matrix for this size
    # is about 64 MiB, easily fitting into memory.
    # ------------------------------------------------------------------
    n = 2000
    target_sum = np.sqrt(2.0 * n)

    # initialise a random point on the simplex (Dirichlet)
    a = rng.dirichlet(alpha=np.ones(n)) * target_sum
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_M = _max_convolution_fft(best_a)
    print(f"[{time.time():.1f}] initial max = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP based refinement (lots of cheap iterations)
    # ------------------------------------------------------------------
    mix = 0.3                # initial mixing factor
    mix_decay = 0.995        # slowly decay towards 0.05
    min_mix = 0.05

    while time.time() < deadline - 20.0:   # keep ~20 s for the final polish
        cur_M = _max_convolution_fft(a)
        a = _lp_refine(a, cur_M, target_sum, mix)

        # record improvement
        new_M = _max_convolution_fft(a)
        if new_M + 1e-10 < best_M:
            best_M = new_M
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP improvement: max = {best_M:.6f}")

        # adapt mixing factor
        mix = max(min_mix, mix * mix_decay)

        # occasional jitter to escape flat regions
        if rng.random() < 0.05:
            jitter = rng.normal(scale=target_sum * 0.0003, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Phase 2 – gradient‑descent polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        a = _gradient_descent_opt(best_a, deadline - 5.0)
        best_a = a
        best_M = _max_convolution_fft(best_a)
        print(f"[{time.time():.1f}] after gradient descent: max = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing (exact convolution updates)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        b_full = _full_convolution(best_a)
        a_hc, b_hc = _hill_climbing(best_a, b_full, deadline - 1.0)
        best_M = b_hc.max()
        best_a = a_hc
        print(f"[{time.time():.1f}] final hill‑climb: max = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and final evaluation
    # ------------------------------------------------------------------
    best_a = np.asarray(best_a, dtype=float)
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]
    else:
        best_a = best_a[:1]

    # official evaluation (objective = max of self‑convolution for a
    # normalised ℓ¹‑norm)
    try:
        val = evaluate_sequence(best_a.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_a.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_a.tolist()
