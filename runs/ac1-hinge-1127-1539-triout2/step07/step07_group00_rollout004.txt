"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence`` which, after the L¹‑normalisation, equals the
maximum entry of the self‑convolution a*a.
The optimiser combines:

*  a cheap grid‑search over two simple families (exponential decay and
   power‑law) that already give good candidates,
*  several random simplex draws,
*  a projected gradient descent on a smooth approximation of the max
   (log‑sum‑exp) with aggressive β‑annealing, momentum and occasional
   jitter,
*  a simulated‑annealing hill‑climber that updates the full convolution
   in O(n) per move,
*  a multi‑stage schedule that gradually increases the dimension,
   up‑sampling the current best solution before re‑optimising.

The routine respects a wall‑clock budget (default 1000 s) and returns the
best sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth maximum.
    Uses momentum, aggressive β‑annealing and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # hyper‑parameters (tuned for speed vs quality)
    beta = 2.0
    beta_max = 2 ** 20          # ≈ 1e6
    beta_factor = 1.7

    step_base = target_sum * 0.22
    rho = 0.92
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0006 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing – climb quickly to a hard max
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the L¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.12, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by projection
    onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9          # initial temperature
    cooling = 0.9994                    # slower cooling → finer search

    while time.time() < deadline:
        # pick a worst convolution position (could be many)
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # indices that contribute to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination: favour low‑mass locations
        if rng.random() < 0.35:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move
        max_delta = a[i] * 0.78
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optional seed from an earlier run
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] seeded from height_sequence_1 (n={n0}) score={cur:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # --------------------------------------------------------------
    # Length schedule – gradually increase dimension
    # --------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----------------------------------------------------------
        # Build a small pool of candidates for this length
        # ----------------------------------------------------------
        candidates = []

        # up‑sample the current best (if any)
        if best_seq is not None:
            if best_seq.size != n:
                cand = _upsample_sequence(best_seq, n)
                candidates.append(cand)
            else:
                candidates.append(best_seq.copy())

        # deterministic grid‑search families
        try:
            cand_exp, _ = _grid_search_exponential(n, target_sum)
            candidates.append(cand_exp)
        except Exception as e:
            print(f"Exp grid error (n={n}): {e}")

        try:
            cand_pow, _ = _grid_search_powerlaw(n, target_sum)
            candidates.append(cand_pow)
        except Exception as e:
            print(f"Power grid error (n={n}): {e}")

        # a few random simplex draws
        for _ in range(4):
            a_rand = _random_initial_sequence(n, rng)
            candidates.append(a_rand)

        # keep only unique candidates (by copy)
        local_best_a = None
        local_best_score = np.inf

        for cand in candidates:
            if cand is None:
                continue
            s = _max_convolution_fft(cand)
            if s < local_best_score:
                local_best_score = s
                local_best_a = cand.copy()

        # ----------------------------------------------------------
        # Gradient‑descent refinement
        # ----------------------------------------------------------
        if time.time() < deadline - 5.0:
            # allocate roughly 30 % of the remaining time to gradient work
            remaining = deadline - time.time()
            gd_deadline = time.time() + min(remaining * 0.32, 120.0)
            a_opt = _gradient_descent_opt(local_best_a, gd_deadline)
            score_opt = _max_convolution_fft(a_opt)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] GD improvement for n={n} score={score_opt:.6f}")

        # ----------------------------------------------------------
        # Hill‑climbing refinement
        # ----------------------------------------------------------
        if time.time() < deadline - 5.0:
            a_current = best_seq.copy() if best_seq.size == n else _upsample_sequence(best_seq, n)
            a_current = _project_to_simplex(a_current, target_sum)
            b_full = _full_convolution(a_current)
            a_hc, b_hc = _hill_climbing(a_current, b_full, deadline)
            score_hc = b_hc.max()
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

    # --------------------------------------------------------------
    # Final polishing with any remaining budget
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final longer gradient descent
        a_final = _gradient_descent_opt(best_seq.copy(), deadline - 4.0)

        # final short hill‑climb
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros and report official value
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # official evaluation (objective = max of the self‑convolution)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
