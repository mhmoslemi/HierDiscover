    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2*n)

    iter = 0
    last_time = time.time()
    # momentum?
    momentum = np.zeros_like(a)
    rho = 0.9
    base_lr = target_sum * 0.2

    while time.time() < deadline and iter < max_iter:
        # compute convolution and max location
        b = _full_convolution(a)  # maybe reuse as needed for max and gradient
        max_idx = int(np.argmax(b))
        max_val = b[max_idx]

        # gradient subgradient
        g = np.zeros_like(a)
        i_low = max(0, max_idx - (n - 1))
        i_high = min(n - 1, max_idx)
        # vectorized:
        i = np.arange(i_low, i_high + 1)
        j = max_idx - i
        g[i] = 2.0 * a[j]

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g

        # step size adapt
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0:
            lr = base_lr / (norm_m * (1 + iter * 1e-4))  # decreasing over time
            a -= lr * momentum

        # enforce nonnegative, project to simplex
        a = np.maximum(a, 0.0)

            a = _project_to_simplex(a, target_sum)

        iter += 1

    return a

"""
Propose a step‑function that (approximately) minimizes ``evaluate_sequence``.

Algorithm
---------
1.  Use the previously found construction (global ``height_sequence_1``) as a seed
    if it exists.
2.  Run a schedule of increasing lengths.  For each length *n*:
    *  initialise a vector (upsampled best or random simplex draw);
    *  run a fast *sub‑gradient* descent that directly works on the
       active maximum of the self‑convolution (the exact ℓ∞‑subgradient);
       after each step the vector is projected onto the ℓ¹‑simplex
       {x≥0, Σx = √(2 n)} and symmetrised (a = (a+rev(a))/2);
    *  follow‑up with the stochastic hill‑climbing routine from the
       reference implementation (exact O(n) convolution updates);
    *  keep the best vector seen.
3.  After the schedule, spend the remaining budget on a final refinement:
    a longer sub‑gradient descent + a short hill‑climb.
4.  Trim trailing near‑zeros and return the sequence as a plain list.

The sub‑gradient step works with the true ℓ∞‑objective (the maximum entry of
the self‑convolution) instead of the smooth log‑sum‑exp used before.
This provides a more accurate descent direction and usually yields a lower
maximum value for the same amount of work.  Symmetrising the vector after
each iteration reduces the number of independent variables and helps the
convolution peak to drop faster.

All helper functions are defined at module level, no closures or lambdas are
used, and the routine respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT) – always work with zero‑padding to the next power
# of two for speed.
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a via FFT."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Random initial sequences on the ℓ¹‑simplex
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random non‑negative vector with Σ = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true ℓ∞‑objective (the maximum convolution entry)
# ----------------------------------------------------------------------
def _subgradient_descent_opt(a0: np.ndarray,
                             deadline: float,
                             max_iters: int = 10_000_000) -> np.ndarray:
    """Projected sub‑gradient descent that always follows the active max entry."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    rho = 0.9                 # momentum coefficient
    base_lr = target_sum * 0.25
    it = 0

    while time.time() < deadline and it < max_iters:
        # 1) compute full convolution and locate the maximal entry
        conv = _full_convolution(a)
        max_idx = int(np.argmax(conv))

        # 2) build the sub‑gradient of the active max entry:
        #    g_i = 2 * a_{max_idx-i} for indices where the term exists
        g = np.zeros_like(a)
        i_low = max(0, max_idx - (n - 1))
        i_high = min(n - 1, max_idx)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = max_idx - i_range
        g[i_range] = 2.0 * a[j_range]

        # 3) momentum update
        momentum = rho * momentum + (1.0 - rho) * g

        # 4) adaptive step size
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = base_lr / (norm_m * (1.0 + it * 1e-3))
            a -= lr * momentum

        # 5) enforce non‑negativity and the ℓ¹‑simplex constraint
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # 6) symmetrise (helps to keep the convolution peak low)
        a = (a + a[::-1]) * 0.5

        # 7) occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            a = (a + a[::-1]) * 0.5

        it += 1

    return a


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (same as the reference code)
# ----------------------------------------------------------------------
def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves mass from a source
    index to a destination index while keeping the full self‑convolution
    up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # slower cooling for finer search

    while time.time() < deadline:
        # pick one of the worst convolution positions
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: 30 % bias towards indices with small mass
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

        a_opt = _subgradient_descent_opt(a_init, gd_deadline)

        score = _max_convolution_fft(a_opt)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] sub‑grad improvement for n={n} score={score:.6f}")

        # --------------------------------------------------------------
        # Phase B – hill‑climbing (exact convolution updates)
        # --------------------------------------------------------------
        if time.time() >= deadline:
            break
        hc_deadline = time.time() + hc_time

        # ensure best_seq has the correct length
        if best_seq.size != n:
            a_cur = _upsample_sequence(best_seq, n)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_full = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_full, hc_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] hill‑climb improvement for n={n} score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final sub‑gradient descent
        a_final = _project_to_simplex(best_seq.copy(), target_sum)
        a_final = _subgradient_descent_opt(a_final, deadline)

        # short hill‑climb at the very end
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 2.0)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5️⃣ Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
