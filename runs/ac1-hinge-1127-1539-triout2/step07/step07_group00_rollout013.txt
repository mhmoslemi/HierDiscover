"""
propose_candidate
=================

Search for a non‑negative step‑function (list of heights) that minimises the
``evaluate_sequence`` objective

    2·n·max(a*a) / (sum(a)**2)

The algorithm follows a multi‑scale stochastic optimisation scheme:

1.  A schedule of increasing lengths (n = 2 k) is processed.
2.  For each length a pool of candidates is built:
    *  an up‑sampled version of the current best,
    *  a large number of random vectors generated from diverse families
       (power‑law, exponential, mixed, Dirichlet, piecewise‑exponential,
       monotone random, sinusoid, logistic).
    *  every candidate is projected onto the ℓ¹‑simplex
       Σa = √(2 n) (objective is homogeneous, the scaling is irrelevant
       but the projection keeps the numbers in a numerically stable range).
3.  The pool is evaluated cheaply by the (max‑convolution) objective.
    The best few (default 5) are refined by projected gradient descent on a
    smoothed max (log‑sum‑exp) using a momentum update and an aggressive
    β‑annealing schedule.
    After each descent step the vector is forced to be monotone decreasing
    (sorting) and re‑projected – this exploits the rearrangement inequality
    which is known to lower the centre autocorrelation.
4.  For moderate lengths (n ≤ 20 000) a fast O(n) simulated‑annealing
    hill‑climber moves mass from a source index to a destination index while
    keeping the full convolution up‑to‑date.
5.  After the schedule the remaining time is spent on a final polishing run:
    a longer gradient descent followed by a short hill‑climb.
6.  Trailing near‑zeros are trimmed and the sequence is returned as a plain
    Python list.

All helper functions are top‑level, there are no closures or lambdas and the
code respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------


def _objective_seq(a: np.ndarray) -> float:
    """
    Compute the raw objective 2·n·max(a*a) / (sum(a)**2).
    The function is cheap because it uses an FFT.
    """
    n = a.size
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    max_b = _max_convolution_fft(a)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ_k exp(β·b_k) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)   (⋆ = cross‑correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth maximum.
    Uses momentum, aggressive β‑annealing, and occasional jitter.
    The vector is forced to be monotone decreasing after each projection.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # hyper‑parameters
    beta = 4.0
    beta_max = 1e5
    beta_factor = 2.0

    step_base = target_sum * 0.25
    rho = 0.90
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # enforce monotone decreasing (rearrangement inequality) and simplex
        a = np.sort(a)[::-1]                     # descending order
        a = _project_to_simplex(a, target_sum)

        # occasional jitter (helps escape flat regions)
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = np.sort(a)[::-1]
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, mixed, Dirichlet,
    piecewise‑exponential, monotone random, sinusoid, logistic) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.2, 2.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.1, 1.5)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential (two zones)
        b = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    elif family == 5:                   # monotone random (sorted uniform)
        a = rng.random(n)
        a = np.sort(a)[::-1]            # descending

    elif family == 6:                   # half‑sine wave (positive)
        a = np.sin(np.pi * (idx + 0.5) / n)   # values in (0,1]
        a = np.maximum(a, 0.0)

    else:                               # logistic decay (smooth step)
        scale = rng.uniform(0.02, 0.12)
        centre = rng.uniform(0.3, 0.7) * n
        a = 1.0 / (1.0 + np.exp((idx - centre) / scale))

    a += rng.normal(scale=target * 0.0005, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by projection
    onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (fast O(n) per move)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # slow cooling for fine search

    while time.time() < deadline:
        # pick one of the positions attaining the current maximum
        max_idxs = np.flatnonzero(b == cur_score)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: bias towards low‑mass positions
        if rng.random() < 0.35:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule: geometrically increasing, stop when time runs out
    # ------------------------------------------------------------------
    schedule = []
    n = 2000
    while n <= 64000:
        schedule.append(n)
        n *= 2

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        print(f"[{time.time():.1f}] start length n={n}")

        # --------------------------------------------------------------
        # Build a pool of candidates
        # --------------------------------------------------------------
        candidates = []

        # upsample current best, if any, as a natural seed
        if best_seq is not None:
            candidates.append(_upsample_sequence(best_seq, n))

        # number of random draws – adapt to remaining time
        remaining = deadline - time.time()
        max_random = 35
        # each random draw is cheap; we cap the pool to avoid overshooting budget
        pool_size = min(max_random, int(remaining // 0.05))
        for _ in range(pool_size):
            candidates.append(_random_initial_sequence(n, rng))

        # --------------------------------------------------------------
        # Quick evaluation of the pool
        # --------------------------------------------------------------
        evaluated = []
        for cand in candidates:
            obj = _objective_seq(cand)
            evaluated.append((obj, cand))
        evaluated.sort(key=lambda x: x[0])

        # keep the best from the pool
        local_best_obj, local_best_seq = evaluated[0]
        if local_best_obj < best_obj:
            best_obj = local_best_obj
            best_seq = local_best_seq.copy()
            print(f"[{time.time():.1f}] pool best n={n} obj={best_obj:.6f}")

        # --------------------------------------------------------------
        # Gradient‑descent refinement of the top‑k candidates
        # --------------------------------------------------------------
        topk = min(5, len(evaluated))
        for rank in range(topk):
            if time.time() >= deadline - 5.0:
                break
            cand = evaluated[rank][1]
            # allocate a modest budget for each refinement
            remaining = deadline - time.time()
            gd_time = min(remaining * 0.18, 28.0)   # cap to 28 s per run
            gd_deadline = time.time() + gd_time
            a_opt = _gradient_descent_opt(cand, gd_deadline)
            obj_opt = _objective_seq(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] GD refinement rank={rank} n={n} obj={best_obj:.6f}")

        # --------------------------------------------------------------
        # Hill‑climbing (only for moderate lengths)
        # --------------------------------------------------------------
        if n <= 20000:
            remaining = deadline - time.time()
            if remaining > 12.0:
                hill_time = min(remaining * 0.20, 24.0)
                hill_deadline = time.time() + hill_time
                a_cur = best_seq if best_seq.size == n else _upsample_sequence(best_seq, n)
                a_cur = _project_to_simplex(a_cur, target_sum)
                b_full = _full_convolution(a_cur)
                a_hc, b_hc = _hill_climbing(a_cur, b_full, hill_deadline)
                obj_hc = _objective_seq(a_hc)
                if obj_hc < best_obj:
                    best_obj = obj_hc
                    best_seq = a_hc.copy()
                    print(f"[{time.time():.1f}] hill‑climb n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        a = _gradient_descent_opt(a, deadline - 2.0)

        # brief hill‑climb if length permits
        if n <= 20000 and time.time() < deadline - 1.0:
            b = _full_convolution(a)
            a, _ = _hill_climbing(a, b, deadline - 1.0)

        best_obj = _objective_seq(a)
        best_seq = a.copy()
        print(f"[{time.time():.1f}] final polishing obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a very short uniform vector (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Official evaluation (for the log printed at the end)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
