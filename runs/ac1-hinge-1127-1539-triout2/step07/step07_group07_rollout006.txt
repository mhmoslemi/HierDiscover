"""
Propose a candidate step‑function (list of non‑negative heights) that
approximately minimises the objective used in the competition:

    evaluate_sequence(seq) = max_k (seq * seq)[k]          # after L¹‑normalisation

The algorithm consists of three stages:

1. Initialisation  
   • If a previous construction is available in the global variable
     `height_sequence_1`, it is taken as the starting point (scaled to the
     required ℓ¹‑norm).  Otherwise a random vector of length 8000 is used.

2. Projected Adam optimisation on a smooth log‑sum‑exp surrogate of the
   maximal convolution entry.  The surrogate temperature `beta` is annealed to
   a very large value (≈1e6) so that the gradient approaches the true
   sub‑gradient of the max‑convolution.  After every step the vector is
   projected onto the simplex Σa_i = √(2 n) (non‑negative ℓ¹‑projection).

3. Simulated‑annealing on the ordering of the multiset.  Swaps are proposed
   and accepted with a Metropolis criterion; the temperature starts at a
   sizable fraction of the current objective and cools slowly.  A few
   occasional global shuffles of a tiny random subset accelerate escape from
   local minima.

The two optimisation phases share the global time budget (default 1000 s);
the first 78 % of the remaining time is spent on the Adam optimiser, the
rest on permutation annealing.  Small Gaussian jitter steps are inserted
periodically to help the search escape flat regions.

The final sequence is trimmed of trailing near‑zero entries and returned
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximal entry of the linear convolution a*a (full length 2·n‑1)
    using an FFT of length a power‑of‑two.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s , x ≥ 0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0 , Σ x_i = s}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Compute the gradient of  (1/β)·log Σ_k exp(β·b_k)  where b = a*a (full conv).

    Returns
        g  – gradient vector (same shape as a)
        b  – the full convolution (numpy array of length 2·n‑1)
    """
    n = a.shape[0]

    # ---- full convolution a*a ---------------------------------------
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ---- soft‑max probabilities --------------------------------------
    max_b = np.max(b)
    # shift for numerical stability
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)  # shape (2n‑1,)

    # ---- gradient g = 2·(a ⋆ p) ------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max (projected onto the simplex)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimisation of the smoothed max‑convolution.
    The algorithm runs until ``deadline`` (unix time) is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0                # start temperature
    max_beta = 1e6
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update (biased moments)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # simple learning‑rate schedule (scaled to keep the L¹‑norm stable)
        lr = 0.5 * target_sum / ((beta + 1.0) * (np.sqrt(v_hat).mean() + 1e-8))

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex Σa = √(2 n), a ≥ 0
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 5e-4 / (beta + 1.0), size=n
            )
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal temperature
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Permutation annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float,
                           init_temp_factor: float = 0.5,
                           cooling: float = 0.9999) -> np.ndarray:
    """
    Simulated‑annealing on the ordering of a fixed multiset.
    Swaps are proposed; acceptance follows the Metropolis rule.
    Parameters
    ----------
    a0 : np.ndarray
        Starting sequence (values already ℓ¹‑normalised).
    deadline : float
        Wall‑clock time at which the routine must stop.
    init_temp_factor : float
        Fraction of the current best objective used as initial temperature.
    cooling : float
        Multiplicative cooling factor applied after each iteration.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature
    T = best_score * init_temp_factor + 1e-9
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        a[i], a[j] = a[j], a[i]
        cur_score = _max_convolution_fft(a)

        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny global shuffle to increase mixing
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                tmp = a[mask].copy()
                rng.shuffle(tmp)
                a[mask] = tmp
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Simple ℓ¹‑scaling helper
# ----------------------------------------------------------------------


def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale non‑negative vector v so that Σv = target."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise from previous construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        seq = np.array(globals()['height_sequence_1'], dtype=float)
        n = seq.size
        target = np.sqrt(2.0 * n)
        seq = _scale_to_sum(seq, target)
        print(f"[{time.time():.1f}] Using external seed (n={n})")
    else:
        n = 8000
        target = np.sqrt(2.0 * n)
        seq = rng.random(n)
        seq = _scale_to_sum(seq, target)
        print(f"[{time.time():.1f}] No external seed – using random (n={n})")

    best_seq = seq.copy()
    best_obj = _max_convolution_fft(best_seq)
    print(f"[{time.time():.1f}] Initial objective = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2. Gradient descent (≈78 % of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    grad_deadline = time.time() + remaining * 0.78
    seq = _adam_gradient_descent(seq, grad_deadline)

    cur_obj = _max_convolution_fft(seq)
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_seq = seq.copy()
        print(f"[{time.time():.1f}] After gradient   obj = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Permutation annealing (use the rest of the time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq = _permutation_annealing(best_seq, deadline,
                                     init_temp_factor=0.5,
                                     cooling=0.9999)
        cur_obj = _max_convolution_fft(seq)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] After SA         obj = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Optional small jitter‑refinement loops while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # small random perturbation
        jitter = rng.normal(scale=target * 5e-4, size=best_seq.size)
        pert = np.maximum(best_seq + jitter, 0.0)
        pert = _project_to_simplex(pert, target)

        # short gradient run
        sub_deadline = time.time() + min(20.0, deadline - time.time())
        pert = _adam_gradient_descent(pert, sub_deadline)

        # short permutation run
        sub_deadline = time.time() + min(20.0, deadline - time.time())
        pert = _permutation_annealing(pert, sub_deadline,
                                      init_temp_factor=0.3,
                                      cooling=0.9995)

        cur_obj = _max_convolution_fft(pert)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = pert.copy()
            print(f"[{time.time():.1f}] Jitter loop improvement → obj = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5. Final tidy‑up and reporting
    # ------------------------------------------------------------------
    # trim trailing negligible entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception as e:
        print(f"[{time.time():.1f}] Evaluation error: {e}")

    print(f"[{time.time():.1f}] Finished – best objective = {best_obj:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
