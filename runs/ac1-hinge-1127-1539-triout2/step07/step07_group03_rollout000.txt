def _adam_optimize(a0, target_sum, deadline):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    bet1 = 0.9
    bet2 = 0.999
    eps = 1e-8
    lr = target_sum * 0.5  # base learning rate
    beta = 1.0
    max_beta = 1e5
    step = 0
    best_a = a.copy()
    best_score = _score(a)
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        step += 1
        m = bet1 * m + (1 - bet1) * g
        v = bet2 * v + (1 - bet2) * (g * g)
        m_hat = m / (1 - bet1 ** step)
        v_hat = v / (1 - bet2 ** step)
        step_size = lr / (np.sqrt(v_hat) + eps)
        a -= step_size * m_hat
        a = _project_to_simplex(a, target_sum)
        # jitter every certain steps
        if step % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        # increase beta
        if step % 500 == 0 and beta < max_beta:
            beta = min(beta * 2, max_beta)
        val = _score(a)
        if val < best_score:
            best_score = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={step} beta={beta:.1f} score={best_score:.6f}")
    return best_a, best_score

def _targeted_sa(a0, deadline, start_temp=5.0, end_temp=1e-4):
    a = a0.copy()
    n = a.size
    cur_conv = _conv_fft(a)
    cur_max = cur_conv.max()
    cur_score = cur_max  # since sum fixed.
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = 0.5 * (0.01 / 0.5) ** tau  # decays from 0.5 to 0.01
        # compute current max index
        k_max = int(np.argmax(cur_conv))
        # compute donor weights
        w = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        w[idx] = a[idx] * a[k_max - idx]
        total_w = w.sum()
        if total_w > 0:
            donor = rng.choice(n, p=w / total_w)
        else:
            donor = rng.integers(n)
        # choose receiver randomly among low-weight indices
        low_thr = np.percentile(w, 30)  # choose lower 30%
        low_idx = np.where(w <= low_thr)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)
        if donor == receiver:
            continue
        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        # evaluate new convolution
        new_conv = _conv_fft(a)
        new_max = new_conv.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_conv = new_conv
            cur_max = new_max
            cur_score = cur_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Target SA improve best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
        # occasional global jitter
        if rng.random() < 0.001:
            jig = rng.normal(scale=a.sum()*0.0005, size=n)
            a = np.maximum(a + jig, 0)
            a = _project_to_simplex(a, np.sqrt(2*n))
            cur_conv = _conv_fft(a)
            cur_max = cur_conv.max()
    return best_a, best_score

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  With the ℓ¹‑norm fixed
    to √(2·n) this reduces to the maximal convolution entry."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the soft‑max surrogate of max‑convolution.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.copy(), target_sum)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = target_sum * 0.5    # base learning‑rate scale

    beta = 1.0
    max_beta = 1e5
    it = 0
    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr_base / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # increase soft‑max sharpness
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


def _targeted_sa(a0: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 start_temp: float = 5.0,
                 end_temp: float = 1e-4,
                 max_frac_start: float = 0.5,
                 max_frac_end: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves mass directly from indices that
    contribute most to the current peak of the convolution to low‑sensitivity
    indices.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.size
    cur_conv = _conv_fft(a)
    cur_max = float(cur_conv.max())
    best_a = a.copy()
    best_score = cur_max

    start_time = time.time()
    total = max(deadline - start_time, 1e-9)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total

        # geometric cooling
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # index of the current maximal convolution entry
        k_max = int(np.argmax(cur_conv))

        # sensitivities: how much each a_i contributes to that peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = np.zeros(n, dtype=np.float64)
        sens[idx] = a[idx] * a[k_max - idx]

        total_sens = sens.sum()
        if total_sens > 0:
            donor = rng.choice(n, p=sens / total_sens)
        else:
            donor = rng.integers(n)

        # receivers are taken from low‑sensitivity region
        low_thr = np.percentile(sens, 30)
        low_candidates = np.where(sens <= low_thr)[0]
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        delta = a[donor] * rng.random() * max_frac
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        new_conv = _conv_fft(a)
        new_max = float(new_conv.max())

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_conv = new_conv
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Target SA new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta

        # occasional global jitter to prevent dead‑ends
        if rng.random() < 0.001:
            jig = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jig, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_conv = _conv_fft(a)
            cur_max = float(cur_conv.max())

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of heights with a low benchmark score.
    The routine combines an Adam‑style projected gradient descent
    with a targeted simulated‑annealing phase.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # Choose problem size – larger n tends to give lower max‑convolution.
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        init = _project_to_simplex(init, target_sum)
    else:
        # a slightly tuned exponent, 0.55 has worked well historically
        init = _power_law_sequence(n, 0.55, target_sum)

    # ------------------------------------------------------------------
    # Phase 1 – Adam projected gradient descent (≈ 60 % of budget)
    # ------------------------------------------------------------------
    gd_deadline = time.time() + 0.60 * (deadline - time.time())
    a_gd, val_gd = _adam_optimize(init, target_sum, gd_deadline)
    print(f"[{time.time():.2f}] Adam phase best score={val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Targeted simulated annealing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _targeted_sa(a_gd, target_sum, deadline,
                                    start_temp=8.0,
                                    end_temp=1e-5,
                                    max_frac_start=0.5,
                                    max_frac_end=0.01)
        if val_sa < val_gd:
            best_seq = a_sa
            best_score = val_sa
        else:
            best_seq = a_gd
            best_score = val_gd
        print(f"[{time.time():.2f}] SA phase best score={best_score:.6f}")
    else:
        best_seq = a_gd
        best_score = val_gd

    # ------------------------------------------------------------------
    # Phase 3 – final polishing (a short Adam run if time permits)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_final, val_final = _adam_optimize(best_seq, target_sum,
                                            deadline)
        if val_final < best_score:
            best_seq = a_final
            best_score = val_final
            print(f"[{time.time():.2f}] Final polish improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)

    # drop trailing zeros (they never affect the score)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (for logging)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
