"""
Greedy‑mass‑transfer + occasional Adam refinement.

The search works in three stages:

1.  **Initialisation** – obtain a starting vector either from the previous
    best (`height_sequence_1`) or from a random symmetric analytic family.
    The vector is projected onto the simplex
    Σa = √(2·n) (the normalization that makes the evaluation equal to the
    peak of the self‑convolution).

2.  **Greedy reduction of the peak** – repeatedly locate the index k where
    (a*a)[k] is maximal.  From that index choose the pair (i, k‑i) that
    contributes the most to the peak and move a small amount of mass from
    i (and its symmetric counterpart) to an index j whose partner
    a[k‑j] is (almost) zero.  The move is accepted only if it strictly
    reduces the peak, guaranteeing a monotone descent.  Symmetry is
    enforced after each move, and a projection onto the simplex restores the
    exact mass constraint.

    This cheap operation (one full FFT per iteration) can be performed
    thousands of times within the time budget and typically drives the
    peak well below the previous best.

3.  **Optional Adam polishing** – with the remaining time a projected
    Adam optimiser (soft‑max approximation of the max‑convolution) refines
    the solution a little further.

The whole routine respects the 1000 s budget, prints progress messages,
stores the best sequence globally as ``height_sequence_1`` and finally
returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Soft‑max helpers for the Adam phase
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (soft‑max surrogate of the peak)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    alpha = 0.1
    alpha_mul = 1.04
    max_alpha = 200.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.012:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy mass‑transfer optimiser (strictly decreasing the true peak)
# ----------------------------------------------------------------------
def _greedy_optimize(a0: np.ndarray,
                     end_time: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from the most harmful pair
    (i, k‑i) to a location that does not influence the current peak.
    The method never increases the peak (except with a tiny random chance)
    and therefore converges to a local optimum.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_seq = a.copy()
    best_peak = np.max(_full_convolution(a))

    # early‑exit guard
    if best_peak < 1e-12:
        return best_seq, best_peak

    while time.time() < end_time:
        conv = _full_convolution(a)
        cur_peak = conv.max()
        k_max = int(np.argmax(conv))

        # contribution of each pair (i, k_max-i)
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        i_idx = np.arange(i_low, i_high + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        i_star = i_idx[np.argmax(contrib)]

        # pick a destination whose partner a[k_max - j] is minimal (zero if out of range)
        idx_all = np.arange(n)
        partner = np.zeros(n)
        mask = (k_max - idx_all >= 0) & (k_max - idx_all < n)
        partner[mask] = a[k_max - idx_all[mask]]
        j_target = int(np.argmin(partner))
        if j_target == i_star:                     # avoid the trivial move
            sorted_idx = np.argsort(partner)
            for cand in sorted_idx:
                if cand != i_star:
                    j_target = int(cand)
                    break

        # symmetric partners
        i_sym = n - 1 - i_star
        j_sym = n - 1 - j_target

        # maximal transferable mass (keep non‑negativity)
        if i_star == i_sym:
            max_delta = a[i_star] * 0.5
        else:
            max_delta = min(a[i_star], a[i_sym]) * 0.5
        if max_delta <= 1e-12:
            # nothing to move, break out of inner loop
            continue
        delta = max_delta * rng.random()   # random fraction of the allowed mass

        # construct candidate
        cand = a.copy()
        cand[i_star] -= delta
        if i_sym != i_star:
            cand[i_sym] -= delta
        cand[j_target] += delta
        if j_sym != j_target:
            cand[j_sym] += delta

        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand, S)          # restores exact sum S
        if enforce_sym:
            cand = _symmetrize(cand)

        new_peak = np.max(_full_convolution(cand))

        # accept only if we improve (tiny probability to accept a worsening move)
        if new_peak < cur_peak - 1e-12:
            a = cand
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best_seq = cand.copy()
                print(f"[greedy] improved to {best_peak:.6f} (n={n})")
        else:
            # occasional uphill move (annealing style) – very low probability
            if rng.random() < 0.001:
                a = cand
                cur_peak = new_peak

        # occasional re‑symmetrisation (keeps drift under control)
        if rng.random() < 0.01:
            a = _symmetrize(a)

    return best_seq, best_peak


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function with a minimal self‑convolution peak.
    The function respects the 1000 s budget, stores the best found sequence in the
    global variable ``height_sequence_1`` and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    best_seq = None
    best_peak = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Helper: create a symmetric random initial vector of length n
    # ------------------------------------------------------------------
    def make_initial(n: int) -> np.ndarray:
        fam = rng.integers(0, 6)
        if fam == 0:
            a = rng.random(n)                     # uniform random
        elif fam == 1:
            # power‑law symmetric: (i·(n‑i))^p
            p = rng.uniform(0.2, 4.0)
            idx = np.arange(n, dtype=float)
            a = ((idx + 1) * (n - idx)) ** p
        elif fam == 2:
            # beta‑like shape
            p1 = rng.uniform(0.1, 5.0)
            p2 = rng.uniform(0.1, 5.0)
            x = np.linspace(0, 1, n)
            a = (x ** p1) * ((1 - x) ** p2)
        elif fam == 3:
            # sinusoidal power
            p = rng.uniform(0.5, 4.0)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        else:
            # two spikes at the ends
            a = np.zeros(n)
            a[0] = a[-1] = 1.0
        a = np.clip(a, 0.0, None)
        a = _symmetrize(a)
        S = np.sqrt(2.0 * n)
        a = a / a.sum() * S
        return a

    # ------------------------------------------------------------------
    # 1) Start from a previously known good sequence if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float).copy()
            n_prev = prev.shape[0]
            prev = _symmetrize(prev)
            S_prev = np.sqrt(2.0 * n_prev)
            prev = _project_simplex(prev, S_prev)
            # allocate the bulk of the remaining time to greedy optimisation
            remaining = deadline - time.time()
            if remaining > 10.0:
                t_end = time.time() + 0.70 * remaining
                seq, peak = _greedy_optimize(prev, t_end, rng)
                if peak < best_peak:
                    best_peak = peak
                    best_seq = seq.copy()
                    print(f"[seeded] peak={best_peak:.6f} (n={n_prev})")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) Perform a few short greedy runs from random starts (diversification)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1
        # pick a length in a sensible range
        n = int(rng.integers(6000, 12001))
        a0 = make_initial(n)

        # allocate a modest slice of time to this run (≈10 s)
        slice_time = min(12.0, deadline - time.time())
        if slice_time <= 1.0:
            break
        t_end = time.time() + slice_time
        seq, peak = _greedy_optimize(a0, t_end, rng)
        if peak < best_peak:
            best_peak = peak
            best_seq = seq.copy()
            print(f"[restart] attempt={attempts} n={n} peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing with Adam (soft‑max surrogate) if time permits
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        t_end = deadline
        seq_adam, peak_adam = _optimize_adam(best_seq, t_end, rng, enforce_sym=False)
        if peak_adam < best_peak:
            best_peak = peak_adam
            best_seq = seq_adam.copy()
            print(f"[adam‑final] improved to {best_peak:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: a simple uniform vector
        n = 8000
        a = np.full(n, 1.0)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_seq = a

    final_list = best_seq.tolist()
    external_score = evaluate_sequence(final_list)
    print(f"[final] external={external_score:.6f} internal={best_peak:.6f} "
          f"length={len(final_list)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_list, dtype=float)
    return final_list
