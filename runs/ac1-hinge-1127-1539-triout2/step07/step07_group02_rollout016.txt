"""
Propose a low‑peak self‑convolution step function.

Algorithm
---------
1. **Exploratory families (15 % of the time)**
   – Sample a variety of *monotone decreasing* families (power, exponential,
     piece‑wise linear, mixed‑exponential, cumulative random).
   – Lengths are drawn from 8 000 … 20 000 (large n reduces the theoretical
     lower bound ≈ S²/(2n‑1) ≈ 1).  Each candidate is normalised to
     Σa = √(2·n) and its peak convolution is recorded.

2. **Monotone projected Adam (≈ 60 %)**
   – Smooth‐max approximation of the max‑convolution is minimised.
   – After every Adam step the vector is projected onto the *monotone
     simplex* (non‑negative, sorted decreasing, fixed ℓ₁‑norm).  This keeps
     the advantageous ordering that pairs large and small entries.
   – A fast temperature schedule (α from 0.5 up to 300) tightens the soft‑max.

3. **Monotone sub‑gradient descent (≈ 20 %)**
   – True max‑convolution is tackled with a sub‑gradient (the active shift)
     and projection onto the monotone simplex.
   – Occasional random mass‑swaps are accepted greedily (simulated‑annealing
     style) to escape plateaus.

4. **Final local hill‑climb (≈ 5 %)**
   – Tiny swaps between random positions followed by monotone projection.
   – Accept only improvements; stop when time expires.

The best sequence found is stored globally as ``height_sequence_1`` for reuse.

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex & monotone projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _project_monotone_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project onto the set {x ≥ 0, x₀ ≥ x₁ ≥ … ≥ x_{n‑1}, Σx = s}.
    The implementation simply sorts descending and rescales to the target sum.
    This is a heuristic but works well for our continuous optimisation.
    """
    v = np.maximum(v, 0.0)
    v = np.sort(v)[::-1]               # enforce monotonicity
    total = v.sum()
    if total > 0:
        v *= s / total                  # renormalise to the required ℓ₁‑norm
    else:                               # degenerate case – fall back to uniform
        v = np.full_like(v, s / v.shape[0])
    return v

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) via real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    ``w`` must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # indices picking the needed terms
    grad = 2.0 * full[idx]
    return grad


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries (temperature α)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w

# Monotone Adam optimiser (soft‑max)
# ----------------------------------------------------------------------
def _optimize_adam_monotone(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth max‑convolution, with monotone‑simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule (soft‑max)
    alpha = 0.5
    alpha_mul = 1.20
    max_alpha = 300.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # monotone‑simplex projection
        a = _project_monotone_simplex(a, S)

        # occasional jitter to explore flat regions
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_monotone_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99996

    return best_a, best_score

    remaining = deadline - now
    adam_deadline = now + 0.60 * remaining
    a_adam, sc_adam = _optimize_adam_monotone(best_overall_seq, adam_deadline, rng)
    if sc_adam < best_overall_score:
        best_overall_score = sc_adam
        best_overall_seq = a_adam.copy()
        print(f"[adam] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – monotone sub‑gradient descent (≈20 % of the rest)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        sub_deadline = now + 0.20 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad_monotone(best_overall_seq, sub_deadline, rng)
        if sc_sub < best_overall_score:
            best_overall_score = sc_sub
            best_overall_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final tiny swap hill‑climb (≈5 % of the rest)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_loc, sc_loc = _local_swap_search_monotone(best_overall_seq, deadline, rng)
        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (store globally for next runs)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
