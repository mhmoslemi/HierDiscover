"""
Implements a multi‑stage optimizer for the step‑function bound.
Stage 1: Broad random search with several parametric families,
          keeping the top‑K candidates.
Stage 2: Long ADAM refinement on the simplex for each top candidate,
          using a soft‑max of the convolution peak.
Stage 3: One LP‑based “direction” step (max‑convolution ≤ current peak)
          and a final short ADAM fine‑tune.
The best sequence found is stored in the global variable
`height_sequence_1` and returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv

def _max_convolution(a: np.ndarray) -> float:
    return _full_convolution_fft(a).max()

def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw

def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw

def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    raw = mix * lin + (1.0 - mix) * base
    return raw

def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw

def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)

def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw

def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    # cosine on [0, pi]; scaled to [0,1]
    c = 0.5 * (1.0 + np.cos(np.pi * (idx - (n - 1) / 2) / (n - 1)))
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    raw = mix * c + (1.0 - mix) * base
    return raw

def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Create a feasible vector of length n, optionally warm‑starting."""
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:
        raw = _init_cosine(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

# Adam optimiser on the simplex (soft‑max version)
# --------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Return (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂a."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.4,
                   temp_decay: float = 0.9995,
                   lr_decay: float = 0.9998) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that works on the simplex via a log‑softmax
    parametrisation. Returns the best feasible vector seen and its
    hard peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature from current peak
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1
        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient w.r.t. a (soft‑max of peak)
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedule updates
        tau *= temp_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if it % 600 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak

# --------------------------------------------------------------
# Linear‑programming improvement step
# --------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                rng: np.random.Generator,
                time_limit: float) -> tuple[np.ndarray, float]:
    """
    Solve  max Σ g_j  s.t. g≥0 and (a * g)_i ≤ M for every i,
    where M = max(a * a). The resulting g is scaled to the required
    ℓ₁‑norm and mixed back with the original a (t=0.2).  The function
    returns the mixed vector and its new peak.
    """
    n = a.shape[0]
    M = _max_convolution(a)

    # Build sparse constraint matrix A (shape (2n‑1, n))
    rows = []
    cols = []
    data = []
    # For each column (g_j) we add the kernel a shifted by j
    for j in range(n):
        # rows i = j … j+n‑1
        row_start = j
        row_end = j + n  # exclusive
        rows.extend(range(row_start, row_end))
        cols.extend([j] * n)
        data.extend(a.tolist())
    A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Upper‑bound constraints: A @ g ≤ M
    ub = np.full(2 * n - 1, M)

    # Objective: maximise Σ g_j  -> minimise -Σ g_j
    c = -np.ones(n)

    # Bounds for variables: g_j ≥ 0
    bounds = [(0, None)] * n

    # Use the 'highs' method (supports sparse matrices)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs',
                  options={'time_limit': max(0.0, time_limit - 0.1)})

    if not res.success:
        # If LP fails, return original vector unchanged
        return a, M

    g0 = res.x
    sum_g = g0.sum()
    if sum_g == 0.0:
        # fallback: keep original
        return a, M

    # Rescale to meet ℓ₁‑norm = √(2n)
    S = np.sqrt(2.0 * n)
    g = g0 * (S / sum_g)

    # Mix a and g – a small step towards g
    t = 0.2
    a_new = (1 - t) * a + t * g
    # Re‑normalise (numerical drift)
    a_new = a_new * (S / a_new.sum())

    new_peak = _max_convolution(a_new)
    return a_new, new_peak

def propose_candidate(seed: int = 42,

    if base_seq is not None:

        except Exception:
            base_seq = None

    best_n = None

            best_seq = a0.copy()
            best_n = n

        # Maintain a small pool of the best candidates for deeper work

            best_n = len(best_seq)

    # ------------------------------------------------------------------

                                      time.time() + slice_time,

                                      lr_decay=0.9998)

"""
Multi‑stage optimizer for the step‑function bound.

Stage 1 – Broad random sampling from several parametric families,
           keeping a small pool of the best candidates.
Stage 2 – Long ADAM refinement on the simplex (log‑softmax
           parametrisation) for each pool element.  The ADAM uses a
           τ‑soft‑max approximation of the convolution peak and a
           slowly decreasing learning‑rate and temperature.
Stage 3 – One optional linear‑programming “direction” step that
           maximises the ℓ₁‑mass under the current convolution bound,
           mixes the result back into the current sequence and refines
           again.  Finally a modest up‑scale and a last ADAM run are
           performed.

The best sequence found is stored in the global variable
`height_sequence_1` and returned as a plain list.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    return mix * edge + (1.0 - mix) * base


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    return mix * exp_part + (1.0 - mix) * base


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    return mix * lin + (1.0 - mix) * base


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    return rng.gamma(shape, 1.0, size=n)


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    # cosine that peaks in the centre, shifted to [0,1]
    c = 0.5 * (1.0 + np.cos(np.pi * (idx - (n - 1) / 2) / (n - 1)))
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    return mix * c + (1.0 - mix) * base


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """Create a feasible vector of length n, possibly warm‑starting."""
    S = np.sqrt(2.0 * n)

    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'cosine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:
        raw = _init_cosine(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    return raw * (S / total)

# ----------------------------------------------------------------------
# ADAM optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Gradient of the τ‑soft‑max approximation of max(b) where b = a*a."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.4,
                   temp_decay: float = 0.9995,
                   lr_decay: float = 0.9998) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that works on the simplex via a log‑softmax
    parametrisation. Returns the best feasible vector seen and its
    hard convolution peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from a feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initialise temperature from current peak
    tau = max(_full_convolution_fft(a0).max(), 0.5)

    best_a = a0.copy()
    best_peak = _max_convolution(a0)
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient w.r.t. a (soft‑max of the peak)
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clipping for numerical stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= temp_decay
        lr *= lr_decay

        # occasional jitter
        if it % 600 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak

# ----------------------------------------------------------------------
# Linear‑programming improvement step (optional, cheap for n ≤ 5000)
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                rng: np.random.Generator,
                time_limit: float) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_j   s.t.   g ≥ 0  and  (a * g)_i ≤ M  ∀i,
    where M = max(a * a).  The obtained g is scaled to the required
    ℓ₁‑norm and mixed back into a with a small step (t=0.2).  The
    function returns the mixed vector and its new peak.
    """
    n = a.shape[0]
    if n > 5000:            # keep memory requirements reasonable
        return a, _max_convolution(a)

    M = _max_convolution(a)

    # Build sparse matrix A such that (A @ g) = a * g  (full linear conv)
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    idx = 0
    for j in range(n):
        rows[idx:idx + n] = np.arange(j, j + n)
        cols[idx:idx + n] = j
        data[idx:idx + n] = a
        idx += n

    A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Upper‑bound constraints: A @ g ≤ M
    ub = np.full(2 * n - 1, M)

    # Objective: maximise Σ g_j  → minimise −Σ g_j
    c = -np.ones(n)

    # Bounds: g_j ≥ 0
    bounds = [(0, None)] * n

    # Solve with HiGHS (sparse support).  The 'time_limit' option is
    # honoured by recent SciPy versions; if it is ignored it simply
    # proceeds until convergence.
    options = {}
    if time_limit > 0:
        options["time_limit"] = max(0.0, time_limit - 0.1)

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options=options)

    if not res.success:
        return a, M

    g0 = res.x
    s = g0.sum()
    if s == 0.0:
        return a, M

    S = np.sqrt(2.0 * n)
    g = g0 * (S / s)

    # Mix a little towards g
    t = 0.2
    a_new = (1 - t) * a + t * g
    a_new = a_new * (S / a_new.sum())

    return a_new, _max_convolution(a_new)
