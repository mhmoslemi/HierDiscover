import cvxpy as cp

def lp_step(f, M):

"""
Improved search for a non‑negative step function whose self‑convolution peak
is as small as possible.

Algorithm
---------
1. *Exploratory sampling* (≈30 % of the time budget):
   – Randomly pick a length n (4000 … 12000) and generate a candidate from a
     rich collection of analytic families (uniform, power‑law, sinusoidal,
     beta‑like, Gaussian, Laplace, semi‑circular, inverse‑quadratic).
   – Enforce exact mirror symmetry, normalise the sum to √(2 n) and keep the
     best candidate seen so far.

2. *Smooth Adam optimisation* (≈70 % of the remaining time):
   – Minimise a log‑sum‑exp (soft‑max) surrogate of the convolution maximum.
   – Gradient = 2·(a ∗ w_rev) + λ·(log a+1)  (entropy regulariser encourages a
     spread‑out shape).  Project onto the simplex after each update and keep
     symmetry.
   – Learning‑rate and soft‑max temperature are annealed; entropy weight λ
     decays slowly.

3. *Sub‑gradient descent* (≈20 % of the rest):
   – Directly optimises the true Chebyshev norm using the sub‑gradient
     (the slice of a that contributes to the current maximal convolution entry).
   – Includes a small random mass‑swap noise term.

4. *Fine‑grained hill climbing* (≈10 % of the rest):
   – Repeated tiny mass‑swaps between random positions, always respecting the
     simplex and symmetry.

All phases are time‑aware; the best sequence found at any moment is stored in
the global variable ``height_sequence_1`` for reuse in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Analytic families for fast initialisation
# ----------------------------------------------------------------------
def _gaussian_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.15, 0.45) * n
    centre = (n - 1) / 2.0
    idx = np.arange(n)
    return np.exp(-0.5 * ((idx - centre) / sigma) ** 2)


def _laplace_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    scale = rng.uniform(0.1, 0.35) * n
    centre = (n - 1) / 2.0
    idx = np.arange(n)
    return np.exp(-np.abs(idx - centre) / scale)


def _semicircle_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    radius = rng.uniform(0.35, 0.48) * n
    centre = (n - 1) / 2.0
    d = np.arange(n) - centre
    mask = np.abs(d) <= radius
    a = np.zeros(n)
    a[mask] = np.sqrt(1.0 - (d[mask] / radius) ** 2)
    return a


def _inverse_quad_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    c = rng.uniform(0.1, 5.0)
    centre = (n - 1) / 2.0
    d = np.arange(n) - centre
    # scale the denominator to the interval length for numerical stability
    scale = (n / 2.0) ** 2
    return 1.0 / (c + (d ** 2) / scale)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam_v2(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a smooth approximation of the convolution peak.
    An entropy regulariser (λ·H) is added to promote spread‑out shapes.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.5
    alpha_mul = 1.02
    max_alpha = 300.0

    # entropy weight schedule
    lam = 0.22                     # start weight
    lam_decay = 0.9995

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # smooth max gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # entropy gradient (λ·∇H = λ·(log a + 1))
        ent_grad = np.log(np.maximum(a, 1e-12)) + 1.0
        grad = grad + lam * ent_grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.012:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99996
        lam = max(lam * lam_decay, 1e-5)

    return best_a, best_score


def _optimize_subgrad_v2(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true convolution maximum, with a modest
    momentum term and random mass‑swap noise.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the Chebyshev norm
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random mass‑swap perturbation (simulated‑annealing style)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early exit if no improvement for a while
        if it % 10000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _local_swap_search_v2(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Tiny random mass‑swap hill‑climbing with optional symmetry enforcement.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.025 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score

            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap random families (≈30 % of total budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1

        # pick a length – favour up‑sampling the previous best but also explore
        if base_seq is not None and rng.random() < 0.55:
            scale = rng.uniform(0.85, 1.15)
            n = max(2000, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(4000, 12001))

        # 0‑based family selector – eight families (including the new ones)
        fam = rng.integers(0, 8)

        if fam == 0:                                 # uniform random
            a = rng.random(n)
        elif fam == 1:                               # power‑law (i·(n‑i))^p
            p = rng.uniform(0.2, 4.5)
            idx = np.arange(n, dtype=float)
            a = ((idx + 1) * (n - idx)) ** p
        elif fam == 2:                               # sinusoidal power
            p = rng.uniform(0.5, 4.5)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        elif fam == 3:                               # beta‑like
            p1 = rng.uniform(0.1, 5.0)
            p2 = rng.uniform(0.1, 5.0)
            x = np.linspace(0, 1, n)
            a = (x ** p1) * ((1 - x) ** p2)
        elif fam == 4:                               # Gaussian
            a = _gaussian_shape(n, rng)
        elif fam == 5:                               # Laplace
            a = _laplace_shape(n, rng)
        elif fam == 6:                               # Semi‑circular
            a = _semicircle_shape(n, rng)
        elif fam == 7:                               # Inverse‑quadratic
            a = _inverse_quad_shape(n, rng)
        else:
            a = rng.random(n)   # fallback (should never happen)

        a = np.clip(a, 0.0, None)
        a = _symmetrize(a)                     # enforce exact mirror symmetry
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive refinement (Adam → Sub‑gradient → Local)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – a modest uniform vector
        n = 6000
        a = np.full(n, 1.0)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_overall_seq = a
        best_overall_score = np.max(_full_convolution(a))

    # 2a – Adam optimisation (≈70 % of the remaining time)
    now = time.time()
    remaining = deadline - now
    adam_deadline = now + 0.70 * remaining
    a_adam, sc_adam = _optimize_adam_v2(best_overall_seq, adam_deadline, rng,
                                        enforce_sym=True)
    if sc_adam < best_overall_score:
        best_overall_score = sc_adam
        best_overall_seq = a_adam.copy()
        print(f"[adam] improved to {best_overall_score:.6f}")

    # 2b – Sub‑gradient descent (≈20 % of the rest)
    now = time.time()
    if now < deadline:
        subgrad_deadline = now + 0.20 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad_v2(best_overall_seq, subgrad_deadline,
                                             rng, enforce_sym=True)
        if sc_sub < best_overall_score:
            best_overall_score = sc_sub
            best_overall_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_overall_score:.6f}")

    # 2c – Local random‑swap hill‑climbing (≈10 % of the rest)
    now = time.time()
    if now < deadline:
        a_loc, sc_loc = _local_swap_search_v2(best_overall_seq, deadline, rng,
                                              enforce_sym=True)
        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return result (also expose globally)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
