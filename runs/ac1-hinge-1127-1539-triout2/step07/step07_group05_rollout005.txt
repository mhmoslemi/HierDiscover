def subgradient_descent(a0, deadline, tol=1e-12):
    a = a0.copy()
    n = a.size
    total_sum = np.sqrt(2.0*n)
    # precompute slice indices for gradient extraction
    slice_start = n-1
    slice_end = slice_start + n
    it = 0
    while time.time() < deadline:
        b = max_convolution(a)
        max_b = b.max()
        # find active indices
        active = np.where(b >= max_b - 1e-12)[0]
        # assign weight 1/len(active) each
        w = np.zeros_like(b)
        if active.size > 0:
            w[active] = 1.0 / active.size
        # gradient = 2 * (rev_a conv w)[slice_start:slice_end]
        a_rev = a[::-1]
        size2 = 1 << ((len(w) + n - 1).bit_length())
        # compute convolution via FFT
        Ar = np.fft.rfft(a_rev, n=size2)
        Wr = np.fft.rfft(w, n=size2)
        conv = np.fft.irfft(Ar*Wr, n=size2)[: len(w)+n-1]  # length 3n-2
        grad = 2.0 * conv[slice_start:slice_end]
        # step size decreasing: lr = step0 / sqrt(it+1)
        step0 = total_sum * 0.5 / (np.linalg.norm(grad)+1e-12)

        # project onto simplex
        a = project_to_simplex(a, total_sum)
        # optional jitter at intervals
        if it % 200 == 0:

            a += jitter
            a = np.maximum(a, 0)
            a = project_to_simplex(a, total_sum)
        it += 1
    return a

assistantfinal**Solution Overview**

The objective of `evaluate_sequence` is simply the maximal entry of the self‑convolution
`b = a * a` (the ℓ¹‑norm is normalised to `√(2 n)`).  
We therefore minimise  

```
max(bk)   with   a ≥ 0,   Σ ai = √(2 n) .
```

The problem is convex: each `bk(a)=aᵀQk a` is a convex quadratic and `maxk bk` is a
convex function.  A practical way to approach the optimum is a two‑stage
continuous optimisation:

1. **Smooth surrogate** – minimise a log‑sum‑exp approximation of the max.
   This yields a smooth gradient that can be optimised efficiently with an
   Adam‑style optimiser and a projection onto the ℓ¹‑simplex after each step.

2. **Exact max‑subgradient** – after the surrogate has sharpened we switch to the
   true sub‑gradient of `maxk bk`.  The sub‑gradient for the active constraints
   can be computed by a single convolution, so each iteration still costs only
   one FFT.

Both stages keep the iterates feasible (non‑negative, right ℓ¹‑norm).  
After the continuous optimisation we apply a short simulated‑annealing phase
that only permutes the entries – this can improve the ordering without
changing the multiset of values.

The search starts from a diverse set of parametric families (power‑law,
exponential, linear, quadratic, β‑distribution and sine‑powers).  The best few
candidates are refined with the two‑stage optimiser and finally with the
permutation annealing.  The whole routine respects the 1000 s budget and
returns the best sequence found.

   random Dirichlet draws, normalising each to Σ a_i = √(2 n).
2. keep the best ∼10 candidates (elite list).
3. for every elite candidate
      – smooth optimisation (log‑sum‑exp surrogate) with Adam + simplex projection
      – switch to exact sub‑gradient descent of max(b) (active set) + projection
      – permutation annealing (swap moves) to improve ordering.
4. with the remaining time run a long refinement on the current global best.
5. return the best sequence (as plain Python list).  All helper functions are
   top‑level, no closures, only NumPy and time are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())     # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_value(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0   (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of the max and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Return (grad, b) where
       b = a*a (full convolution)
       grad = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # log‑sum‑exp in a numerically stable way
    max_b = np.max(b)
    exp_shift = np.exp(beta * (b - max_b))
    p = exp_shift / np.sum(exp_shift)               # probabilities, sum to 1

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1                  # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * p_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser for the smooth surrogate
# ----------------------------------------------------------------------
def _adam_smooth_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Adam optimisation of the log‑sum‑exp surrogate, projection after each step."""
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    beta = 1.0                # start with a mild smoothing
    max_beta = 1e5
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam updates
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size proportional to the remaining ℓ¹‑mass
        norm = np.linalg.norm(m_hat)
        if norm > 0:
            lr = 0.5 * target / (norm * (beta + 1.0) * math.sqrt(it))
            a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep the iterate feasible
        a = _project_to_simplex(a, target)

        # increase smoothing (make surrogate sharper)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        # occasional jitter to avoid plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent for the true max(b) (active‐set)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on max_k (a*a)_k."""
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    it = 0
    rng = np.random.default_rng()
    # pre‑computed slice indices for the gradient extraction
    start = n - 1
    end = start + n

    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        # active set (indices whose value is within 1e‑12 of the max)
        act = np.where(b >= max_b - 1e-12)[0]
        if act.size == 0:       # should never happen
            break
        w = np.zeros_like(b)
        w[act] = 1.0 / act.size

        # gradient = 2·(a_rev ⋆ w) restricted to the original length
        a_rev = a[::-1]
        conv_len = len(w) + n - 1            # = 3·n‑2
        size2 = 1 << (conv_len.bit_length())
        ar_fft = np.fft.rfft(a_rev, n=size2)
        w_fft = np.fft.rfft(w, n=size2)
        conv = np.fft.irfft(ar_fft * w_fft, n=size2)[:conv_len]
        grad = 2.0 * conv[start:end]

        # simple diminishing step size
        it += 1
        step = target / ((np.linalg.norm(grad) + 1e-12) * math.sqrt(it))
        a -= step * grad

        # projection onto the simplex
        a = _project_to_simplex(a, target)

        # tiny random jitter every few hundred iterations
        if it % 300 == 0:
            jitter = rng.normal(scale=target * 3e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Permutation based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improve the ordering of a fixed multiset by random swaps."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_value(best)

    # initial temperature (roughly 2 % of the current best)
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # swap
        cur = _max_convolution_value(a)
        if cur < best_score:
            best_score = cur
            best = a.copy()
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / T):
                best_score = cur
                best = a.copy()
            else:
                a[i], a[j] = a[j], a[i]  # revert

        T *= cooling

        # occasional tiny reshuffle of a random subset (helps escape)
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                sh = a[mask].copy()
                rng.shuffle(sh)
                a[mask] = sh

    return best


# ----------------------------------------------------------------------
# Parametric families (all return a normalised to Σ = √(2 n))
# ----------------------------------------------------------------------
def _normalize(v: np.ndarray, target: float) -> np.ndarray:
    """Scale non‑negative vector to the required ℓ¹‑sum."""
    if v.sum() == 0:
        return np.full_like(v, target / v.size)
    scale = target / v.sum()
    return np.maximum(v * scale, 0.0)


def _powerlaw_seq(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize(raw, math.sqrt(2.0 * n))


def _exp_seq(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize(raw, math.sqrt(2.0 * n))


def _linear_seq(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    return _normalize(raw, math.sqrt(2.0 * n))


def _quadratic_seq(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    return _normalize(raw, math.sqrt(2.0 * n))


def _beta_seq(n: int, alpha: float, beta_: float) -> np.ndarray:
    """Beta‑type shape: (i+1)^{α‑1}·(n-i)^{β‑1}  (α,β>0)."""
    i = np.arange(1, n + 1, dtype=float)
    raw = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta_ - 1.0))
    return _normalize(raw, math.sqrt(2.0 * n))


def _sine_seq(n: int, power: float) -> np.ndarray:
    """sin^{p} shape on (0,π)."""
    i = np.arange(n, dtype=float) + 0.5
    raw = np.sin(np.pi * i / n) ** power
    return _normalize(raw, math.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Helper to keep the elite list up‑to‑date
# ----------------------------------------------------------------------
def _consider(seq: np.ndarray, elite: list, best):
    """Update global best and elite list; returns possibly new best."""
    obj = _max_convolution_value(seq)
    if obj < best[0]:
        best[0] = obj
        best[1] = seq.copy()
        print(f"[{time.time():.1f}] New global best = {obj:.6f}  len={seq.size}")

    if len(elite) < 12 or obj < elite[-1][0]:
        elite.append((obj, seq.copy()))
        elite.sort(key=lambda x: x[0])
        elite[:] = elite[:12]        # keep only the 12 best
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a step‑function (non‑negative list) that minimises
    evaluate_sequence.  The implementation respects the time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5        # small safety margin
    best = [float("inf"), None]                   # [objective, array]
    elite = []                                     # list of (obj, array)

    # --------------------------------------------------------------
    # 0. optional warm‑start from a previously discovered construction
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        n0 = init.size
        init = _normalize(init, math.sqrt(2.0 * n0))
        best = _consider(init, elite, best)
    except NameError:
        pass

    # --------------------------------------------------------------
    # 1. Random parametric generation (70 % of the budget)
    # --------------------------------------------------------------
    lengths = [3000, 4000, 5000, 6000, 8000, 10000, 12000, 15000, 18000]
    families = ['power', 'exp', 'linear', 'quadratic', 'beta', 'sine', 'mixed']

    gen_deadline = time.time() + 0.70 * (deadline - time.time())

    while time.time() < gen_deadline:
        n = int(rng.choice(lengths))
        target_sum = math.sqrt(2.0 * n)

        fam = rng.choice(families)
        if fam == 'power':
            p = rng.uniform(0.3, 1.0)
            shift = rng.uniform(0.0, 5.0)
            base = _powerlaw_seq(n, p, shift)
        elif fam == 'exp':
            rate = rng.uniform(0.0001, 0.02)
            shift = rng.uniform(0.0, 5.0)
            base = _exp_seq(n, rate, shift)
        elif fam == 'linear':
            slope = rng.uniform(0.00001, 0.012)
            intercept = rng.uniform(0.5, 4.0)
            base = _linear_seq(n, slope, intercept)
        elif fam == 'quadratic':
            a0 = rng.uniform(0.5, 5.0)
            b = rng.uniform(0.0, 0.012)
            c = rng.uniform(0.0, 6e-5)
            base = _quadratic_seq(n, a0, b, c)
        elif fam == 'beta':
            alpha = rng.uniform(0.2, 5.0)
            beta_ = rng.uniform(0.2, 5.0)
            base = _beta_seq(n, alpha, beta_)
        elif fam == 'sine':
            power = rng.uniform(0.5, 5.0)
            base = _sine_seq(n, power)
        else:                     # mixed: convex combo of power‑law & exponential
            w = rng.uniform(0.2, 0.8)
            p = rng.uniform(0.3, 1.0)
            shift_p = rng.uniform(0.0, 5.0)
            pow_seq = _powerlaw_seq(n, p, shift_p)
            rate = rng.uniform(0.0001, 0.02)
            shift_e = rng.uniform(0.0, 5.0)
            exp_seq = _exp_seq(n, rate, shift_e)
            mix = w * pow_seq + (1.0 - w) * exp_seq
            base = _project_to_simplex(mix, target_sum)

        # ---------------------------------------------------------
        # a few deterministic orderings (they often change the max)
        # ---------------------------------------------------------
        asc = np.sort(base)            # small → large
        desc = asc[::-1]               # large → small
        alt = np.empty_like(asc)       # alternating small‑large
        lo, hi = 0, n - 1
        for i in range(n):
            if i % 2 == 0:
                alt[i] = asc[lo]
                lo += 1
            else:
                alt[i] = asc[hi]
                hi -= 1
        shuf = asc.copy()
        rng.shuffle(shuf)

        for cand in (desc, asc, alt, shuf):
            best = _consider(cand, elite, best)

        # occasional cheap permutation annealing on the current best
        if rng.random() < 0.07 and best[1] is not None:
            tmp = _permutation_annealing(best[1], time.time() + 0.25)
            best = _consider(tmp, elite, best)

    # --------------------------------------------------------------
    # 2. Refinement of elite candidates (30 % of the budget)
    # --------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break

        n = seed_seq.size
        # ---- stage 1: smooth Adam optimiser ----
        sub_dead = min(deadline, time.time() + 12.0)   # ~12 s for smooth phase
        refined = _adam_smooth_opt(seed_seq, sub_dead)

        # ---- stage 2: exact sub‑gradient descent ----
        sub_dead2 = min(deadline, time.time() + 8.0)    # ~8 s for sub‑gradient
        refined = _subgradient_descent(refined, sub_dead2)

        # ---- stage 3: permutation annealing ----
        refined = _permutation_annealing(refined, deadline)

        best = _consider(refined, elite, best)

    # --------------------------------------------------------------
    # 3. Final long refinement on the global best (if time left)
    # --------------------------------------------------------------
    if best[1] is not None and time.time() < deadline:
        # use almost the whole remaining budget for a final smooth+sub‑gradient run
        remaining = deadline - time.time()
        # split remaining time roughly 2/3 smooth, 1/3 sub‑gradient
        t_smooth = remaining * 0.65
        t_sub = remaining - t_smooth - 2.0          # keep ~2 s for a tiny final anneal
        a_mid = _adam_smooth_opt(best[1], time.time() + t_smooth)
        a_mid = _subgradient_descent(a_mid, time.time() + t_sub)
        a_mid = _permutation_annealing(a_mid, deadline)
        best = _consider(a_mid, elite, best)

    # --------------------------------------------------------------
    # 4. Fallback and post‑processing
    # --------------------------------------------------------------
    if best[1] is None:
        # uniform fallback (modest length)
        n = 4000
        best[1] = np.full(n, math.sqrt(2.0 * n) / n)

    # remove trailing near‑zeros (only cosmetic)
    nz = np.where(best[1] > 1e-12)[0]
    if nz.size:
        best[1] = best[1][: nz[-1] + 1]

    # sanity check with the provided evaluator (optional)
    try:
        val = evaluate_sequence(best[1].tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best[0]:.6f}  length = {best[1].size}")
    return best[1].tolist()
