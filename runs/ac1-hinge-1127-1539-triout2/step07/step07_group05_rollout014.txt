def _subgradient_descent_opt(a0, deadline, target_sum=None):
   if target_sum None: compute sqrt(2*n)
   a = a0.copy()
   n = a.shape[0]
   S = target_sum if target_sum else np.sqrt(2.0 * n)

   iteration = 0
   rng = np.random.default_rng()
   while time.time() < deadline:
       # compute convolution
       # use FFT for full conv to get max
       n = a.size
       size = 1 << ((2 * n - 1).bit_length())
       a_fft = np.fft.rfft(a, n=size)
       b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
       max_idx = int(np.argmax(b))
       # compute subgradient
       g = np.zeros_like(a)
       i_low = max(0, max_idx - (n - 1))
       i_high = min(n - 1, max_idx)
       idx_i = np.arange(i_low, i_high+1)
       idx_j = max_idx - idx_i
       g[idx_i] += a[idx_j]
       g[idx_j] += a[idx_i]  # this counts twice if i=j, which is correct
       # step size schedule
       # use step = a0_scale / sqrt(iter+1)
       step = S / (np.sqrt(iteration+1) * (np.linalg.norm(g) + 1e-12))
       a -= step * g
       # projection onto simplex
       a = _project_to_simplex(a, S)
       # occasional jitter
       if iteration % 500 == 0:
           sigma = S * 1e-3 / (np.sqrt(iteration+1) + 1)
           jitter = rng.normal(scale=sigma, size=n)
           a += jitter
           a = np.maximum(a, 0.0)
           a = _project_to_simplex(a, S)
       iteration += 1
   return a

def _subgradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    iteration = 0
    while time.time() < deadline:
        # compute convolution
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        k = int(np.argmax(b))
        # subgradient
        g = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k - idx_i
        g[idx_i] += a[idx_j]
        g[idx_j] += a[idx_i]

        # step size schedule
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        step = 0.5 * target_sum / (norm_g * np.sqrt(iteration + 1))
        a -= step * g

        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if iteration % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-3 / (np.sqrt(iteration + 1) + 1), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        iteration += 1
    return a

"""
Propose a non‑negative step‑function (a list of heights) that minimises
``evaluate_sequence`` – equivalently the maximum entry of the self‑convolution.
The search is a multi‑stage stochastic optimisation:

1. Random generation of many candidate sequences from a rich family of
   parametric shapes (power‑law, exponential, linear, quadratic,
   logistic, beta, sparse Dirichlet and spike‑mixtures).  For each raw
   shape the ℓ¹‑simplex Σa = √(2 n) is enforced by Euclidean projection.
   Several deterministic orderings (descending, ascending, alternating)
   and a random shuffle are tried.

2. The best few candidates (the *elite* list) are refined.
   • A projected gradient descent on the smooth log‑sum‑exp surrogate of
     ``max(a*a)`` (``_gradient_descent_opt``).
   • A projected sub‑gradient descent that directly optimises the true
     maximum (``_subgradient_descent_opt``).
   • A permutation‑based simulated annealing that improves the ordering
     of the multiset without changing the values (``_permutation_annealing``).

3. A final global permutation annealing run is performed on the overall
   best sequence.

All steps respect the given time budget (default 1000 s).  The routine
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution (length 2 n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Normalisation helper (enforce Σa = √(2 n))
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have sum == target."""
    if v.sum() == 0.0:
        # a tiny uniform filler to avoid division by zero
        v = np.full_like(v, 1.0)
    scale = target / v.sum()
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _logistic_sequence(n: int, mu: float, sigma: float) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    idx = np.arange(n, dtype=float)
    raw = 1.0 / (1.0 + np.exp((idx - mu) / sigma))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shaped (on the discrete grid)."""
    t = np.arange(n, dtype=float) / (n - 1.0)
    raw = (t ** (alpha - 1.0)) * ((1.0 - t) ** (beta - 1.0))
    raw = np.nan_to_num(raw)  # safety for extreme parameters
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _sparse_dirichlet_sequence(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Heavy‑tailed Dirichlet (alpha<1) gives many tiny entries."""
    raw = rng.gamma(alpha, 1.0, size=n)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _spike_sequence(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k random spikes (the rest exactly zero)."""
    raw = np.zeros(n, dtype=float)
    pos = rng.choice(n, size=min(k, n), replace=False)
    raw[pos] = rng.random(k) + 0.1  # avoid exactly zero mass
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a  (full convolution)
        g = ∇_a ( (1/β)·log Σ exp(β·b) )
          = 2·(a ⋆ p) , p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient = 2 * (a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected (Adam‑style) gradient descent on the smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam‑style step
        it += 1
        lr = 0.02 / (np.sqrt(it) + 1.0)          # diminishing step
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a -= step

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase β (sharpen the surrogate)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max(b)
# ----------------------------------------------------------------------
def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Simple sub‑gradient descent on max(b) with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # full convolution and index of the maximal entry
        b = _full_convolution(a)
        k = int(np.argmax(b))

        # sub‑gradient of max(b) w.r.t. a
        g = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k - idx_i
        g[idx_i] += a[idx_j]
        g[idx_j] += a[idx_i]   # double counts the diagonal (correct)

        norm = np.linalg.norm(g)
        if norm == 0.0:
            break

        # step size schedule ∝ 1/√it
        step = 0.6 * target_sum / (norm * np.sqrt(it + 1.0))
        a -= step * g

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 500 == 0:
            jitter = rng.normal(
                scale=target_sum * 8e-4 / (np.sqrt(it + 1) + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    The L¹‑norm stays unchanged, only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9994
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        if cur_score <= best_score:
            best_score = cur_score
            best = a.copy()
        else:
            # Metropolis acceptance
            if rng.random() < np.exp((best_score - cur_score) / T):
                best_score = cur_score
                best = a.copy()
            else:
                a[i], a[j] = a[j], a[i]   # revert

        T *= cooling

        # occasional tiny random shuffle of a small subset
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                sub = a[mask]
                rng.shuffle(sub)
                a[mask] = sub
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Helper to keep the elite list and the global best
# ----------------------------------------------------------------------
def _update_elite(seq: np.ndarray, elite: list, best_info: dict):
    """Update global best and elite list (sorted by objective)."""
    obj = _max_convolution_fft(seq)
    if obj < best_info["obj"]:
        best_info["obj"] = obj
        best_info["seq"] = seq.copy()
        print(f"[{time.time():.1f}] New global best obj={obj:.6f} len={seq.size}")

    if len(elite) < 12 or obj < elite[-1][0]:
        elite.append((obj, seq.copy()))
        elite.sort(key=lambda x: x[0])
        elite[:] = elite[:12]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the target
    ``evaluate_sequence`` objective (maximal entry of its self‑convolution).
    The routine respects the allotted time budget, explores many parametric
    families, refines promising candidates with projected gradient,
    sub‑gradient and permutation annealing phases, and finally returns the
    best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    best_info = {"seq": None, "obj": np.inf}
    elite = []

    # ------------------------------------------------------------------
    # 0. Warm start from a previously found construction if available
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        n0 = init.size
        init = _project_to_simplex(init, np.sqrt(2.0 * n0))
        _update_elite(init, elite, best_info)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Random parametric generation
    # ------------------------------------------------------------------
    lengths = [3000, 4000, 5000, 6000, 8000, 10000, 12000, 15000]
    families = [
        "power", "exp", "linear", "quadratic",
        "logistic", "beta", "sparse_dirichlet", "spike"
    ]

    # allocate ~65 % of the total budget to random generation
    gen_deadline = time.time() + 0.65 * (deadline - time.time())

    while time.time() < gen_deadline:
        n = int(rng.choice(lengths))
        target_sum = np.sqrt(2.0 * n)

        family = rng.choice(families)

        if family == "power":
            p = rng.uniform(0.30, 0.95)
            shift = rng.uniform(0.0, 5.0)
            base = _powerlaw_sequence(n, p, shift)

        elif family == "exp":
            rate = rng.uniform(5e-4, 0.02)
            shift = rng.uniform(0.0, 5.0)
            base = _exponential_sequence(n, rate, shift)

        elif family == "linear":
            slope = rng.uniform(1e-5, 0.012)
            intercept = rng.uniform(0.4, 5.0)
            base = _linear_sequence(n, slope, intercept)

        elif family == "quadratic":
            a0 = rng.uniform(0.5, 6.0)
            b = rng.uniform(0.0, 0.015)
            c = rng.uniform(0.0, 8e-5)
            base = _quadratic_sequence(n, a0, b, c)

        elif family == "logistic":
            mu = rng.uniform(0.2, n * 0.9)
            sigma = rng.uniform(0.5, max(1.0, n * 0.05))
            base = _logistic_sequence(n, mu, sigma)

        elif family == "beta":
            alpha = rng.uniform(0.3, 3.0)
            beta = rng.uniform(0.3, 3.0)
            base = _beta_sequence(n, alpha, beta)

        elif family == "sparse_dirichlet":
            alpha = rng.uniform(0.05, 0.5)   # <1 gives heavy tail
            base = _sparse_dirichlet_sequence(n, alpha, rng)

        else:   # spike
            k = rng.integers(2, max(3, n // 30))
            base = _spike_sequence(n, k, rng)

        # ------------------------------------------------------------------
        # Ordering heuristics (desc, asc, alternating, random)
        # ------------------------------------------------------------------
        asc = np.sort(base)
        desc = asc[::-1]

        # alternating small–large
        alt = np.empty_like(asc)
        lo, hi = 0, n - 1
        for i in range(n):
            if i % 2 == 0:
                alt[i] = asc[lo]
                lo += 1
            else:
                alt[i] = asc[hi]
                hi -= 1

        rand = asc.copy()
        rng.shuffle(rand)

        for cand in (desc, asc, alt, rand):
            _update_elite(cand, elite, best_info)

        # occasional cheap permutation annealing on the current global best
        if best_info["seq"] is not None and rng.random() < 0.07:
            tmp = _permutation_annealing(best_info["seq"], time.time() + 0.25)
            _update_elite(tmp, elite, best_info)

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size

        # a) gradient descent on smooth max
        sub_dead = min(deadline, time.time() + 12.0)
        refined = _gradient_descent_opt(seed_seq, sub_dead)

        # b) sub‑gradient descent on the true max
        sub_dead2 = min(deadline, time.time() + 10.0)
        refined = _subgradient_descent_opt(refined, sub_dead2)

        # c) permutation annealing (uses whatever time remains)
        refined = _permutation_annealing(refined, deadline)

        _update_elite(refined, elite, best_info)

    # ------------------------------------------------------------------
    # 3. Final global permutation annealing (if any time left)
    # ------------------------------------------------------------------
    if best_info["seq"] is not None and time.time() < deadline:
        best_info["seq"] = _permutation_annealing(best_info["seq"], deadline)
        best_info["obj"] = _max_convolution_fft(best_info["seq"])

    # ------------------------------------------------------------------
    # 4. Fallback in the extremely unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_info["seq"] is None:
        n = 4000
        best_info["seq"] = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_info["seq"] > 1e-12)[0]
    if nz.size:
        best_info["seq"] = best_info["seq"][: nz[-1] + 1]

    # Verify once with the official evaluator (optional)
    try:
        val = evaluate_sequence(best_info["seq"].tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_info['obj']:.6f} "
          f"length = {best_info['seq'].size}")

    return best_info["seq"].tolist()
