from scipy.optimize import linprog
import scipy.sparse as sp

def linear_step(f, M):
    n = f.size
    # Build convolution matrix
    # Use sparse: for each j, f_j influences rows i=j..j+n-1 (since g_i includes f_j * b_{i-j})
    rows = []
    cols = []
    data = []
    for j in range(n):
        # rows range from j to j + n - 1 (inclusive)
        row_start = j
        row_end = j + n  # exclusive
        rows.extend(range(row_start, row_end))
        cols.extend([j] * n)
        data.extend([f[j]] * n)
    # Build CSR
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Constraints: A @ b <= M
    # Additional constraint: sum(b) is objective, maximize sum => minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        b = res.x
        return b
    else:
        return None
```

def _softmax_optimize(a_init, beta_schedule, deadline):
    n = a_init.size
    target = np.sqrt(2.0 * n)
    # initial logits
    eps = 1e-12
    z = np.log(a_init + eps)
    best_a = a_init.copy()
    best_val = _max_convolution(best_a)
    for beta in beta_schedule:
        if time.time() >= deadline - 1.0:
            break
        # closure capturing beta
        def fun(z_vec):
            # compute objective and gradient for current beta
            a_raw = np.exp(z_vec - np.max(z_vec))
            a = a_raw / a_raw.sum() * target
            # compute conv, etc.
            n_local = a.shape[0]
            size = 1 << ((2 * n_local - 1).bit_length())
            a_fft = np.fft.rfft(a, n=size)
            b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n_local - 1]
            max_b = b.max()
            # compute p
            exp_beta = np.exp(beta * (b - max_b))
            sum_exp = exp_beta.sum()
            p = exp_beta / sum_exp
            # gradient via convolution
            a_rev = a[::-1]
            p_fft = np.fft.rfft(p, n=size)
            a_rev_fft = np.fft.rfft(a_rev, n=size)
            conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n_local - 1]
            g_a = 2.0 * conv_rev[n_local - 1 : n_local - 1 + n_local]

            s = a / target
            avg = np.dot(s, g_a)
            grad_z = target * s * (g_a - avg)
            # smooth max objective
            obj = max_b + np.log(sum_exp) / beta
            return obj, grad_z.astype(np.float64)

        # Use L-BFGS-B
        options = {'maxiter': 200, 'ftol': 1e-12}
        try:
            res = scipy.optimize.minimize(fun, z, method='L-BFGS-B', jac=True, options=options)

            print("L-BFGS failure", e)
            break

"""
Stepping‑stone optimizer for the step‑function max‑convolution problem.

Approach
--------
1.  Initialise a non‑negative sequence of length *n* (power‑law, then projected
    onto the ℓ¹‑simplex Σa = √(2·n)).
2.  Optimise a smooth approximation of the maximum of the self‑convolution
    using a soft‑max (log‑sum‑exp) surrogate.
    * a is re‑parameterised as  a = s·√(2·n)  with  s_i = exp(z_i)/∑exp(z).
    * For a fixed temperature β the objective is
          Fβ(z) = max(b) + (1/β)·log∑exp(β·(b‑max(b))) ,
      where b = a*a (full convolution).  ∇Fβ can be obtained analytically:
          ∇aFβ = 2·conv_rev(p)  (p – soft‑max weights)
          ∇zFβ = √(2·n)·s·(∇aFβ – ⟨s,∇aFβ⟩) .
    * An L‑BFGS‑B optimisation (SciPy) is run for a schedule of increasing
      β = [1, 4, 16, 64, 256, 1024]; each stage starts from the result of the
      previous one.
3.  Several independent restarts (different lengths, seeds) are executed
    within the time budget; the best sequence (smallest true max‑convolution)
    is kept.
4.  A short incremental local‑search (mass‑transfer moves) is applied as a
    final polish.
5.  Trailing zeros are stripped and the list of floats is returned.

The whole routine respects the 1000 s budget and never returns an empty
sequence.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helpers for generating initial sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n, renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (Σa = √(2·n) assumed)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Incremental local‑search (used only for a final polish)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """Exact O(n) update of a and its convolution after moving mass."""
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]
    a_before = a.copy()

    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    a[donor] -= delta
    a[receiver] += delta
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """Very fast hill‑climbing optimiser (mass‑transfer, smoothing, jitter)."""
    target = np.sqrt(2.0 * n)

    # ---- initialise ----------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        max_frac = 0.5 * (0.005 / 0.5) ** tau          # max donor fraction
        temp = 0.5 * (1e-5 / 0.5) ** tau               # Metropolis temperature

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass transfer (≈80 %)
        # --------------------------------------------------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            inv = 1.0 / (a + 1e-12)                      # bias receiver to small coeffs
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
            else:
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing window (≈10 %)
        # --------------------------------------------------------------
        elif move_type < 0.95:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_window + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = new_max <= cur_max or (rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)))
                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) Gaussian jitter (≈10 %)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a, best_max


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) optimiser using L‑BFGS‑B
# ----------------------------------------------------------------------


def _softmax_optimize(a_init: np.ndarray,
                      beta_schedule,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a smooth approximation of max(a*a) by L‑BFGS‑B.
    The sequence is kept on the simplex via a soft‑max parametrisation.
    """
    n = a_init.size
    target = np.sqrt(2.0 * n)
    eps = 1e-12
    # initialise logits
    z = np.log(a_init + eps)

    best_a = a_init.copy()
    best_val = _max_convolution(best_a)

    for beta in beta_schedule:
        if time.time() >= deadline - 0.5:
            break

        # --------------------------------------------------------------
        # objective + gradient for a given beta (return (f, g))
        # --------------------------------------------------------------
        def fun(z_vec):
            # soft‑max to obtain a
            z_shift = z_vec - np.max(z_vec)        # prevent overflow
            exp_z = np.exp(z_shift)
            a = exp_z / exp_z.sum() * target

            # convolution a*a (full)
            size = 1 << ((2 * n - 1).bit_length())
            a_fft = np.fft.rfft(a, n=size)
            b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

            max_b = b.max()
            # soft‑max weights (probability distribution over the entries)
            exp_beta = np.exp(beta * (b - max_b))
            sum_exp = exp_beta.sum()
            p = exp_beta / sum_exp

            # gradient wrt a  (2 * convolution of reversed a with p)
            a_rev = a[::-1]
            p_fft = np.fft.rfft(p, n=size)
            a_rev_fft = np.fft.rfft(a_rev, n=size)
            conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
            g_a = 2.0 * conv_rev[n - 1: n - 1 + n]

            # gradient wrt logits z
            s = a / target
            avg = np.dot(s, g_a)
            grad_z = target * s * (g_a - avg)

            # smooth max value
            f = max_b + np.log(sum_exp) / beta
            return f, grad_z.astype(np.float64)

        # L‑BFGS‑B – a modest number of iterations per beta stage
        opts = {"maxiter": 200, "ftol": 1e-12}
        try:
            res = scipy.optimize.minimize(fun,
                                          z,
                                          method="L-BFGS-B",
                                          jac=True,
                                          options=opts)
        except Exception as exc:
            # any failure – abort the remaining schedule
            print("L‑BFGS error:", exc)
            break

        z = res.x
        # evaluate true max‑convolution of the result
        z_shift = z - np.max(z)
        exp_z = np.exp(z_shift)
        a = exp_z / exp_z.sum() * target
        cur_val = _max_convolution(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] beta={beta:.0f}  new best max={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the provided evaluation function.
    The routine tries several lengths, runs a soft‑max L‑BFGS optimiser
    (with an increasing temperature schedule) and finishes with a brief
    mass‑transfer local search for polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) try a few problem sizes (larger n gives more freedom)
    # ------------------------------------------------------------------
    sizes = [1500, 2000, 2500, 3000]
    beta_sched = [1, 4, 16, 64, 256, 1024]          # temperature increase

    for n in sizes:
        if time.time() >= deadline - 5.0:
            break

        # we will dedicate a slice of the remaining budget to this size
        remaining = deadline - time.time()
        # keep at least ~5 s for the final polish
        alloc = min(remaining * 0.30, 200.0)

        run_deadline = time.time() + alloc

        # a few random restarts for robustness
        for restart in range(3):
            if time.time() >= run_deadline:
                break

            # ---- initialise candidate ---------------------------------
            p = rng.uniform(0.3, 1.0)                       # power‑law exponent
            target = np.sqrt(2.0 * n)
            a0 = _power_law_sequence(n, p, target)
            # small jitter to break symmetry
            a0 += rng.normal(scale=target * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target)

            # ---- run the soft‑max optimiser -------------------------
            a_opt, score = _softmax_optimize(a0, beta_sched, run_deadline)

            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] size={n} restart={restart} → best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) final polishing with cheap incremental local search (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        polish_deadline = time.time() + min(12.0, deadline - time.time() - 0.2)
        seq, score = _local_search(best_seq.size, polish_deadline, rng, init_seq=best_seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] post‑polish improved max={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Strip trailing zeros (they have no effect on the objective)
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
