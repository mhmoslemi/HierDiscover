"""
Propose a step‑function (non‑negative sequence) that minimizes
`evaluate_sequence`.  The code combines three complementary optimisation
phases:

1. **Initialisation** – a power‑law sequence is projected onto the
   ℓ¹‑simplex {a ≥ 0, Σa = √(2·n)}.  If a previous construction
   (`height_sequence_1`) is available it is used as a warm start.

2. **Alternating linear‑programming refinement** – for a fixed length *n*
   we repeatedly:
        * compute M = max(a * a)  (self‑convolution),
        * solve a linear programme   max Σg_j   s.t.   a * g ≤ M,  g ≥ 0,
        * rescale the LP solution to the same ℓ¹‑norm as *a*,
        * blend the new vector into *a* with a decreasing step size.
   Each LP is tiny (variables = n, constraints = 2·n‑1) and is solved
   with SciPy’s HiGHS driver using a sparse convolution matrix.

3. **Soft‑max gradient descent polish** – finally a projected gradient
   descent on a smooth approximation of the max‑convolution is applied.
   The temperature β is annealed up to 4096 and a small amount of jitter
   is injected periodically.

The algorithm is time‑aware – every phase receives a slice of the
remaining budget and stops early if the deadline is reached.  The
returned list of floats is already normalised (Σa = √(2·n)) and contains
no trailing zeros.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
try:

except Exception:   # pragma: no cover
    _SCIPY_OK = False

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = v.astype(np.float64, copy=False)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a."""
    return float(np.convolve(a, a).max())

# Linear‑programming sub‑routine
# ----------------------------------------------------------------------


def _solve_lp_for_g(f: np.ndarray, M: float,
                    rows: np.ndarray, cols: np.ndarray) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   f * g ≤ M   (element‑wise),  g ≥ 0.
    `rows` and `cols` encode the fixed sparsity pattern of the convolution
    matrix; only the data array (repeating f) changes each iteration.
    Returns the optimal vector `g` or None on failure.
    """
    n = f.shape[0]
    data = np.repeat(f, n)                    # length n*n
    A = _sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsc()
    c = -np.ones(n)                           # maximise Σg  → minimise –Σg
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _alternating_lp_optimize(n: int,
                             target_sum: float,
                             deadline: float,
                             rng: np.random.Generator,
                             init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Alternating linear‑programming refinement for a fixed length `n`.
    Returns the best sequence found and its max‑convolution value.
    """
    # ---- initialise ----------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target_sum)

    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _max_convolution(a)

    # pre‑compute the constant index pattern of the convolution matrix
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)

    start = time.time()
    total = deadline - start
    iteration = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total

        # current maximal convolution entry
        M = _max_convolution(a)

        # ----- solve LP -------------------------------------------------
        g0 = _solve_lp_for_g(a, M, rows, cols)
        if g0 is None:
            # LP failed – abort the LP phase
            break

        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            break

        # scale to the required ℓ¹‑norm
        g = g0 * (target_sum / sum_g0)

        # ----- blending -------------------------------------------------
        # step size decays from 0.5 to ~0.05
        t = max(0.05, 0.5 * (0.2 ** tau))
        new_a = (1.0 - t) * a + t * g
        new_val = _max_convolution(new_a)

        # accept if the max does not increase
        if new_val <= M + 1e-12:
            a = new_a
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] LP improve (n={n}) max={best_val:.6f}")
        else:
            # try a smaller step; if it helps, keep it
            t_small = t * 0.3
            new_a2 = (1.0 - t_small) * a + t_small * g
            new_val2 = _max_convolution(new_a2)
            if new_val2 < M:
                a = new_a2
                if new_val2 < best_val:
                    best_val = new_val2
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] LP small‑t improve (n={n}) max={best_val:.6f}")

        # occasional jitter to escape plateaus
        if rng.random() < 0.04:
            scale = target_sum * 0.001 * (1.0 - tau)
            a = a + rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        iteration += 1
        if iteration >= 80:               # generous hard cap per length
            break

    return best_a, best_val

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} max={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val
