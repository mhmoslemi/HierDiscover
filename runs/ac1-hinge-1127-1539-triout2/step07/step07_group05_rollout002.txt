"""
Propose a step‑function that minimises ``evaluate_sequence``.
-----------------------------------------------------------

The evaluator returns the maximal entry of the self‑convolution of the
sequence (the L¹‑norm is normalised to √(2 n) inside the routine, so the
objective is just max{(a∗a)_k}).  The algorithm works in three phases
inside the given time budget (≈1000 s).

1. **Diverse random initialisation**  
   * Several families (power‑law, exponential, linear, quadratic,
     mixtures, random Dirichlet) are sampled for many lengths (≈4‑15 k).  
   * For each raw vector we project onto the ℓ¹‑simplex Σa = √(2n)
     (guaranteeing a fair comparison) and try a few deterministic
     orderings (sorted, reversed, alternating) together with a random
     shuffle.  The best few candidates are kept in an *elite* list.

2. **Shape optimisation** – *projected gradient descent*  
   The smoothed maximum (log‑sum‑exp) of the convolution is minimised.
   Its gradient can be computed efficiently with FFTs.  A momentum
   scheme, a schedule for the smoothing parameter β and small Gaussian
   jitter are used.  Several restarts are performed for every elite
   candidate.

3. **Ordering optimisation** – *simulated‑annealing permutation*  
   With the multiset of values fixed, random swaps and random sub‑array
   reversals are proposed.  An exponential cooling schedule accepts
   downhill moves always and uphill moves with probability exp(‑Δ/T).

Finally the best sequence found is returned (as a plain list of floats).
All helper functions are defined at module level – there are no
closures – and the total run‑time is kept inside the supplied budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a  (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max
# ----------------------------------------------------------------------
def _gradient_descent_opt(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """Annealed projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5          # start with a fairly smooth surrogate
    max_beta = 1e6
    momentum = np.zeros_like(a)
    rho = 0.9          # momentum decay
    it = 0

    while time.time() < deadline:
        # gradient of the smooth surrogate
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # step size that keeps us roughly on the simplex surface
            lr = 0.6 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 6e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta (make surrogate sharper)
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap & reverse moves)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps and
    occasional sub‑array reversals.  The L¹‑norm stays unchanged.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current best
    T = best_score * 0.025 + 1e-9
    cooling = 0.9996
    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.85:                     # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
        else:                                        # reverse a random block
            i = rng.integers(0, n - 1)
            j = rng.integers(i + 1, n)
            a[i:j] = a[i:j][::-1]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if dE <= 0.0 or rng.random() < np.exp(-dE / T):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert move
            if rng.random() < 0.85:
                a[i], a[j] = a[j], a[i]          # swap revert
            else:
                a[i:j] = a[i:j][::-1]            # reverse revert

        T *= cooling

        # tiny global shuffle now and then (helps escape deep valleys)
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.015
            if np.any(mask):
                part = a[mask]
                rng.shuffle(part)
                a[mask] = part
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best

def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _random_dirichlet_sequence(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    raw = rng.gamma(shape=alpha, scale=1.0, size=n)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _alternating_sequence(base: np.ndarray) -> np.ndarray:
    """Return an alternating small‑large ordering of ``base``."""
    asc = np.sort(base)
    n = asc.size
    alt = np.empty_like(asc)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    return alt

            base = _quadratic_sequence(n, a0, b, c)

        elif family == 'mixed':
            w = rng.uniform(0.15, 0.85)

            p = rng.uniform(0.30, 0.90)
            shift_p = rng.uniform(0.0, 6.0)
            pow_seq = _powerlaw_sequence(n, p, shift_p)

            rate = rng.uniform(0.00005, 0.025)
            shift_e = rng.uniform(0.0, 6.0)
            exp_seq = _exponential_sequence(n, rate, shift_e)

            mix = w * pow_seq + (1.0 - w) * exp_seq
            base = _project_to_simplex(mix, target_sum)

        else:   # dirichlet
            alpha = rng.uniform(0.1, 2.0)
            base = _random_dirichlet_sequence(n, alpha, rng)

        # ------------------------------------------------------------------
        # Ordering heuristics
        # ------------------------------------------------------------------
        asc = np.sort(base)          # smallest → largest
        desc = asc[::-1]             # largest → smallest
        alt = _alternating_sequence(base)
        rnd = asc.copy()
        rng.shuffle(rnd)

        for cand in (desc, asc, alt, rnd):
            _consider(cand)

        # occasional cheap permutation annealing on the current global best
        if best_seq is not None and rng.random() < 0.06:
            tmp = _permutation_annealing(best_seq, time.time() + 0.30, rng)
            _consider(tmp)

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (gradient + permutation SA)
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        # give each elite candidate a few gradient‑descent restarts
        for _ in range(3):
            if time.time() >= deadline:
                break
            # jitter the seed a little before optimisation
            jitter = rng.normal(scale=0.001 * np.sqrt(2.0 * n), size=n)
            a0 = np.maximum(seed_seq + jitter, 0.0)
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n))

            sub_dead = min(deadline, time.time() + 14.0)
            refined = _gradient_descent_opt(a0, sub_dead, rng)

            # finish with permutation annealing (uses remaining time)
            refined = _permutation_annealing(refined, deadline, rng)
            _consider(refined)

    # ------------------------------------------------------------------
    # 3. Final global permutation annealing (if time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _permutation_annealing(best_seq, deadline, rng)
        best_obj = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 4. Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = _project_to_simplex(best_seq, np.sqrt(2.0 * n))

    # Trim trailing near‑zero entries – purely cosmetic
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional verification using the provided evaluator
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
