"""
Propose a candidate step function (list of non‑negative heights) that
minimises the `evaluate_sequence` objective.

Approach
--------
1. **Coarse parametric sweep** – generate a modest number of candidates
   from a few simple families (power‑law, exponential, truncated linear)
   for several moderate lengths (≈ 2000‑4000).  For each family we keep the
   best few candidates (the “elite”).

2. **Projected sub‑gradient descent** – each elite candidate is refined
   by a projected Adam optimiser that uses the true sub‑gradient of the
   maximum convolution entry (no soft‑max surrogate).  This converges
   quickly to a locally optimal shape while preserving the ℓ¹‑norm.

3. **Permutation annealing** – the ordering of the multiset is then
   optimised by simulated‑annealing swaps (FFT based evaluation of the
   maximal convolution entry).

4. **Final local shuffles** – a short random‑shuffle stage explores a
   few additional orderings of the current best sequence.

The routine respects a 1000 s budget (default).  Intermediate best
objective values are printed for monitoring.  The returned list is
already normalised to Σa = √(2 n), so `evaluate_sequence` returns the
maximal convolution entry.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s , x ≥ 0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0 , Σ x_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_max_conv(a: np.ndarray):
    """
    Compute sub‑gradient of   max_k (a ★ a)_k   w.r.t. a
    and also return the current maximal convolution value.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    b = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k = int(np.argmax(b))
    max_b = float(b[k])

    # sub‑gradient: g_i = 2 * a_{k-i}  if 0 ≤ k-i < n, else 0
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1)
    g[i] = 2.0 * a[k - i]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the true sub‑gradient (projected onto simplex)
# ----------------------------------------------------------------------


def _adam_subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser that minimises max‑convolution entry.
    Runs until the supplied unix time `deadline` is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    rng = np.random.default_rng()
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        g, _ = _subgradient_max_conv(a)

        # Adam biased moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # simple learning‑rate schedule (scaled to keep ℓ¹ norm stable)
        lr = 0.5 * target_sum / (np.sqrt(v_hat).mean() + 1e-8)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex Σa = √(2 n), a ≥ 0
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    # final projection (guard against drift)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Permutation annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing on the ordering of a fixed multiset.
    Only swaps are performed; the ℓ¹‑norm is invariant.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature proportional to the current max
    T = best_score * 0.02 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        a[i], a[j] = a[j], a[i]
        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < math.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert the move
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny global shuffle to increase diversity
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                tmp = a[mask].copy()
                rng.shuffle(tmp)
                a[mask] = tmp
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Simple parametric families (power‑law, exponential, truncated linear)
# ----------------------------------------------------------------------


def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale non‑negative vector v so that Σv = target."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = (i + shift) ** (-p)
    return _scale_to_sum(raw, math.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = np.exp(-rate * (i + shift))
    return _scale_to_sum(raw, math.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * i)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _scale_to_sum(raw, math.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence``.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5  # safety margin
    best_seq = None
    best_obj = np.inf
    elite = []          # list of tuples (score, array)
    elite_cnt = 12

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep (moderate lengths)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000]      # keep lengths modest for speed
    p_grid = np.linspace(0.30, 0.80, 7)
    shift_grid = np.linspace(0.0, 5.0, 7)
    rate_grid = np.linspace(0.0002, 0.0080, 7)
    exp_shift_grid = np.linspace(0.0, 3.0, 5)
    slope_grid = np.linspace(0.0, 0.008, 6)
    intc_grid = np.linspace(0.2, 5.0, 6)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ---- power‑law -------------------------------------------------
        for p in p_grid:
            for sh in shift_grid:
                base = _powerlaw_sequence(n, p, sh)
                for cand in (base, base[::-1]):  # asc / desc
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj, best_seq = score, cand.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL best (pow) n={n} p={p:.3f} sh={sh:.2f} obj={score:.6f}")
                    if len(elite) < elite_cnt or score < elite[-1][0]:
                        elite.append((score, cand))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:elite_cnt]

        # ---- exponential -----------------------------------------------
        for rate in rate_grid:
            for sh in exp_shift_grid:
                base = _exponential_sequence(n, rate, sh)
                for cand in (base, base[::-1]):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj, best_seq = score, cand.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL best (exp) n={n} rate={rate:.5f} sh={sh:.2f} obj={score:.6f}")
                    if len(elite) < elite_cnt or score < elite[-1][0]:
                        elite.append((score, cand))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:elite_cnt]

        # ---- linear (truncated) ----------------------------------------
        for slope in slope_grid:
            for ic in intc_grid:
                base = _linear_sequence(n, slope, ic)
                for cand in (base, base[::-1]):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj, best_seq = score, cand.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL best (lin) n={n} sl={slope:.5f} ic={ic:.2f} obj={score:.6f}")
                    if len(elite) < elite_cnt or score < elite[-1][0]:
                        elite.append((score, cand))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:elite_cnt]

        # ---- random samples --------------------------------------------
        for _ in range(15):
            rand = rng.random(n)
            rand = _scale_to_sum(rand, target_sum)
            score = _max_convolution_fft(rand)
            if score < best_obj:
                best_obj, best_seq = score, rand.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL best (rnd) n={n} obj={score:.6f}")
            if len(elite) < elite_cnt or score < elite[-1][0]:
                elite.append((score, rand))
                elite.sort(key=lambda x: x[0])
                elite = elite[:elite_cnt]

    # ------------------------------------------------------------------
    # 2. Incorporate optional external construction (if provided)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        ext = np.array(height_sequence_1, dtype=float)
        n_ext = ext.size
        ext = _scale_to_sum(ext, math.sqrt(2.0 * n_ext))
        score = _max_convolution_fft(ext)
        if score < best_obj:
            best_obj, best_seq = score, ext.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL best (external) n={n_ext} obj={score:.6f}")
        if len(elite) < elite_cnt or score < elite[-1][0]:
            elite.append((score, ext))
            elite.sort(key=lambda x: x[0])
            elite = elite[:elite_cnt]

    # ------------------------------------------------------------------
    # 3. Refine elite candidates (sub‑gradient Adam + permutation SA)
    # ------------------------------------------------------------------
    elite.sort(key=lambda x: x[0])   # strongest first

    for rank, (score0, seed_seq) in enumerate(elite):
        if time.time() >= deadline:
            break
        n = seed_seq.size
        remaining = deadline - time.time()
        # allocate most of the remaining time to shape optimisation
        shape_time = remaining * 0.75
        perm_time = remaining * 0.25

        # ---- shape optimisation via Adam on true sub‑gradient ----------
        refined = _adam_subgradient_descent(seed_seq, time.time() + shape_time)

        # ---- permutation annealing ------------------------------------
        refined = _permutation_annealing(refined, time.time() + perm_time)

        # evaluate candidate
        obj = _max_convolution_fft(refined)
        if obj < best_obj:
            best_obj, best_seq = obj, refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL best (refined #{rank}) n={n} obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Final short random‑shuffle polishing on the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        while time.time() < deadline:
            perm = rng.permutation(n)
            cand = best_seq[perm]
            cur = _max_convolution_fft(cand)
            if cur < best_obj:
                best_obj, best_seq = cur, cand.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL best (final shuffle) obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5. Final handling and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (guaranteed feasible)
        n = 2500
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim trailing negligible entries (purely cosmetic)
    best_seq = np.array(best_seq)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Report the final evaluate_sequence value (should equal best_obj)
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
        best_obj = eval_val
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
