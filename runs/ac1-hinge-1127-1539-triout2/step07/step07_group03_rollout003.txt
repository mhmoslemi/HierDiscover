def _power_exp_sequence(n, p, q, target_sum):
    i = np.arange(n, dtype=np.float64)
    a = (i+1) ** (-p) * np.exp(-q * i)
    sum_a = a.sum()
    if sum_a == 0:
        return np.zeros_like(a)
    a = a * (target_sum / sum_a)
    return a
```

def _subgradient_descent_multi(a0, target_sum, deadline):
    best_seq = a0.copy()
    best_val = _score(best_seq)
    rng = np.random.default_rng()
    while time.time() < deadline - 5.0:
        # allocate small run time maybe 20 secs
        remaining = deadline - time.time()
        run_time = min(20.0, remaining * 0.5)
        jitter = rng.normal(scale=target_sum * 0.01, size=a0.shape)
        start = best_seq + jitter
        start = np.maximum(start, 0)
        start = _project_to_simplex(start, target_sum)
        a_res, val_res = _subgradient_descent(start, target_sum, time.time() + run_time)

            best_val = val_res
            best_seq = a_res.copy()
            print(f"[{time.time():.2f}] Subgrad restart improved to {best_val:.6f}")
    return best_seq, best_val

    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    step_scale = target_sum * 0.5
    min_step = target_sum * 1e-6
    step = step_scale
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        n = a.shape[0]
        grad = np.zeros_like(a)

        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[max_idx - idx]  # vectorized, safe even if max_idx-idx out-of-range ?

        # adaptive step based on gradient norm
        norm = np.linalg.norm(grad)
        if norm > 0:

        else:
            step = step_scale

        # update

        val_new = _score(a_new)

        # accept if improves else possibly reject

            best_val = val_new
            best = a_new.copy()

        if val_new <= _score(a):
            a = a_new
        else:

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Coarse grid search** over a family of shapes
   a_i = (i+1)^(-p) * exp(-q*i)   (p,q are small grids)
   for several problem sizes n (≈ 8000‑15000).  The best shape
   (lowest benchmark value) seeds the local optimisation.

2. **Projected sub‑gradient descent** on the true max‑convolution
   objective.  The sub‑gradient at the current maximum entry k_max is
   g_i = 2·a_{k_max‑i} (zero where the index is out of range).  After
   each step the vector is projected onto the simplex
   {x ≥ 0, Σx = √(2n)}.  Several short random restarts are performed.

3. **Simulated annealing** with temperature‑controlled mass‑transfer,
   local smoothing and swaps.  The move‑selection uses the sensitivity
   (the same sub‑gradient) to choose donors and receivers.

4. **Final polish** – a short run of the soft‑max projected gradient
   descent from the original code (very effective for fine tuning).

The routine respects the 1000 s budget, prints progress and finally
returns the best sequence (trailing zeros removed).  The provided
`evaluate_sequence` function is called once at the end for reporting.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_exp_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """
    Shape a_i = (i+1)^(-p) * exp(-q*i)  normalised to the required ℓ¹‑norm.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i)
    total = a.sum()
    if total == 0:
        return np.zeros_like(a)
    a *= target_sum / total
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to change the length of a sequence."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent (projected) on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simple projected sub‑gradient descent on max(a*a).
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    step_scale = target_sum * 0.5
    min_step = target_sum * 1e-6
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # current convolution and maximum index
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        n = a.shape[0]
        # sub‑gradient of the maximum entry
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k_max - idx]

        # adaptive step size (diminishing)
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = step_scale / (norm * (it + 1) ** 0.5)
        else:
            step = step_scale

        # projected update
        a_new = _project_to_simplex(a - step * grad, target_sum)
        val_new = _score(a_new)

        # keep the best overall
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        # simple acceptance rule
        if val_new <= _score(a):
            a = a_new
        else:
            # if not improving, keep the old a and shrink step_scale a little
            step_scale *= 0.9

        # occasional random jitter to shake out of plateaus
        if it % 100 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        # stop if step becomes tiny
        if step < min_step:
            break

        it += 1

    return best_a, best_val


def _subgradient_descent_multi(a0: np.ndarray,
                               target_sum: float,
                               deadline: float):
    """
    Run several short sub‑gradient runs with random restarts.
    Keeps the best vector seen.
    """
    best_a = a0.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # allocate a modest budget for each restart
        run_budget = min(15.0, remaining * 0.4)
        if run_budget <= 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _subgradient_descent(start, target_sum, time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Restart improved best={best_val:.6f}")

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 10.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.5,
                         max_frac_end: float = 0.01,
                         max_window: int = 100) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Moves: mass transfer (biased by the sub‑gradient), local smoothing,
    random swaps.  Temperature and allowed transfer fraction decay
    exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate the current maximum convolution entry
        k_max = int(np.argmax(conv))

        # sub‑gradient (sensitivities) for the max entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens  # higher weight → more likely donor
        move_type = None

        # ------------------- mass transfer (biased) -------------------
        if rng.random() < 0.65:
            tot_w = weight.sum()
            if tot_w > 0:
                donor = rng.choice(n, p=weight / tot_w)
            else:
                donor = rng.integers(n)

            # pick a receiver among low‑weight positions
            low_thr = np.percentile(weight, 15)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ----------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert rejected move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of max(convolution).
    Uses a larger step base, stronger momentum, and a higher β ceiling.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence with a low benchmark score.
    The method combines a fast shape grid search, projected sub‑gradient
    descent with restarts, simulated annealing and a final soft‑max GD polish.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem sizes according to the time budget
    # ------------------------------------------------------------------
    if budget_s >= 900:
        candidate_ns = [15000, 12000, 10000]
    elif budget_s >= 600:
        candidate_ns = [12000, 9000, 7000]
    else:
        candidate_ns = [8000, 6000, 5000]

    # ------------------------------------------------------------------
    # Grid search over the (p,q) family
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.30, 1.00, 15)      # exponent
    q_vals = np.linspace(0.0, 0.0015, 7)     # exponential cutoff

    best_seq = None
    best_score = np.inf

    for n in candidate_ns:
        target_sum = np.sqrt(2.0 * n)

        # optionally seed from a previously found construction
        if "height_sequence_1" in globals():
            base = np.array(globals()["height_sequence_1"], dtype=np.float64)
            base = _resize_sequence(base, n)
            base = _project_to_simplex(base, target_sum)
            seq0 = base
        else:
            seq0 = None

        # scan the (p,q) grid
        for p in p_vals:
            for q in q_vals:
                seq = _power_exp_sequence(n, p, q, target_sum)
                # optionally blend with the old construction (tiny mixture)
                if seq0 is not None and np.random.rand() < 0.02:
                    seq = 0.98 * seq + 0.02 * seq0
                    seq = _project_to_simplex(seq, target_sum)

                sc = _score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
        print(f"[{time.time():.2f}] Grid n={n} best score so far={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent with restarts
    # ------------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("Failed to initialise a candidate sequence.")

    sg_deadline = time.time() + (deadline - time.time()) * 0.35
    a_sg, val_sg = _subgradient_descent_multi(best_seq,
                                              np.sqrt(2.0 * best_seq.size),
                                              sg_deadline)
    if val_sg < best_score:
        best_seq = a_sg
        best_score = val_sg
        print(f"[{time.time():.2f}] Sub‑gradient phase best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing on the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, deadline,
                                            start_temp=12.0,
                                            end_temp=1e-6,
                                            max_frac_start=0.45,
                                            max_frac_end=0.008,
                                            max_window=120)
        if val_sa < best_score:
            best_seq = a_sa
            best_score = val_sa
            print(f"[{time.time():.2f}] SA phase best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short polishing GD (if a little time remains)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_final, val_final = _annealed_gradient_descent_v2(best_seq,
                                                            deadline,
                                                            np.sqrt(2.0 * best_seq.size))
        if val_final < best_score:
            best_seq = a_final
            best_score = val_final
            print(f"[{time.time():.2f}] Final polish improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # One final evaluation using the provided benchmark (optional)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
