def _lp_update(a, M, target_sum, t=0.5):
    n = a.size
    # Build Toeplitz matrix
    from scipy.sparse import diags
    diags_data = [np.full(n, a[i]) for i in range(n)]
    offsets = -np.arange(n)
    A = diags(diags_data, offsets, shape=(2*n-1, n), format='csc')
    b = np.full(2*n-1, M)

    # linear program: maximize sum(g), equivalent to minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return a unchanged
        return a
    g = res.x
    # project onto simplex with sum target_sum
    g = _project_to_simplex(g, target_sum)
    # convex combination
    a_new = (1 - t) * a + t * g
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------

1. **Coarse optimisation on a small problem (n≈2000)**  
   – Initialise a few random candidates.  
   – For each candidate run a few iterations of an *alternating LP*:
       maximise the ℓ₁‑mass of a vector `g` while keeping the convolution
       `a*g` below the current worst autocorrelation value `M`.
       The LP is tiny (≈2000 variables, ≤4000 constraints) and solved
       with ``scipy.optimize.linprog``.
   – After each LP step blend the new vector `g` with the current one
       (`a ← (1‑t)a + t g`) and project back onto the ℓ₁‑simplex
       (sum = √(2 n)).  Keep the best small‑scale vector.

2. **Upsample to a larger size (n≈20000)**  
   Linear interpolation followed by projection onto the appropriate simplex
   preserves the total mass while giving a richer search space.

3. **Exact hill‑climbing on the large problem**  
   A simulated‑annealing style mass‑transfer algorithm (the same as in the
   reference code) updates the full self‑convolution in *O(n)* per move.
   It runs for the bulk of the remaining time and returns the best
   vector found.

4. **Fine polishing** – a short projected gradient descent on the smooth
   log‑sum‑exp surrogate of the maximum convolution entry, with an
   aggressively annealed inverse temperature (β up to 1e5).

5. **Truncate trailing zeros** and return the sequence as a plain list.

The combination of an LP‑based global reshaping step and a long
hill‑climbing phase on a larger dimension typically beats the previous
best bound (≈1.511) within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p), p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth log‑sum‑exp surrogate.
    Uses momentum, annealed inverse temperature β and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # temperature schedule
    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.3

    # step‑size and momentum
    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter for escaping plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

# Random initialisers (various families)

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled (power‑law, exponential, Dirichlet,
    sine, Gaussian, piecewise) to diversify the search space.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine wave (positive half‑wave)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that transfers mass from a
    source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9  # initial temperature
    cooling = 0.9994

    while time.time() < deadline:
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index weighted by contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index
        if rng.random() < 0.4:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        max_delta = a[i] * 0.85
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (linear terms)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Linear‑programming (LP) step
# ----------------------------------------------------------------------


def _lp_update(a: np.ndarray, M: float, target_sum: float, t: float = 0.3) -> np.ndarray:
    """
    One LP‑based refinement of ``a``:
    maximise Σ g_j subject to g ≥ 0 and (a * g)_k ≤ M.
    The solution ``g`` is projected onto the ℓ₁‑simplex and blended
    with ``a`` using a convex combination weight ``t``.
    """
    n = a.size
    try:
        import scipy.sparse as sp
        from scipy.optimize import linprog
    except Exception:                      # pragma: no cover
        # If SciPy is unavailable, fall back to a no‑op.
        return a

    # Build the Toeplitz constraint matrix: (a * g)_k = Σ_i a_i g_{k-i}
    # Each diagonal offset = -i (i = 0..n‑1) has constant value a[i].
    diags_data = [np.full(n, a[i]) for i in range(n)]
    offsets = -np.arange(n)
    A = sp.diags(diags_data, offsets, shape=(2 * n - 1, n), format='csc')
    b = np.full(2 * n - 1, M, dtype=float)

    # Linear program: maximise Σ g_j  <=> minimise -Σ g_j
    c = -np.ones(n, dtype=float)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})

    if not res.success:                     # pragma: no cover
        return a

    g = res.x
    g = _project_to_simplex(g, target_sum)

    a_new = (1.0 - t) * a + t * g
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new

    if time.time() < deadline - 1:
        a_ref = _gradient_descent_opt(best_seq.copy(), deadline)
        val_ref = _max_convolution_fft(a_ref)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.1f}] Final gradient polishing -> score {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (for diagnostics)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
