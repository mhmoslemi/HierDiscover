def _adam_optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0*n)
    best_a = a.copy()
    best_val = _max_convolution(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999
    eps=1e-8
    t=0
    beta = 1.0
    max_beta=4096.0
    step_scale = target * 0.1  # base step

    while time.time() < deadline:
        t+=1
        # gradient
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = step_scale / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat
        # projection
        a = _project_to_simplex(a, target)
        # evaluate
        val = _max_convolution(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam n={n} beta={beta:.1f} val={val:.6f}")
        # increase beta periodically
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)
        # occasional jitter
        if t % 200 == 0:
            scale = target * 0.001 / (beta+1.0)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target)

        # early exit if no improvement for long? maybe not needed.

    return best_a, best_val

def _adam_optimize(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _max_convolution(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 1.0
    max_beta = 4096.0
    step_scale = target * 0.12
    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = step_scale / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat
        a = _project_to_simplex(a, target)

        # evaluate
        val = _max_convolution(a)
        if val < best_val:
            best_val = val; best_a = a.copy()
            print(f"[{time.time():.1f}] Adam n={n} beta={beta:.1f} val={val:.6f}")

        # increase beta every 200 steps
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)
        # jitter occasionally
        if t % 500 == 0:
            scale = target * 0.001 / (beta + 1.0)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return best_a, best_val

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence.  The algorithm is a hybrid of
  * fast incremental local search (mass transfer, smoothing,
    random jitter and occasional swaps),
  * a multi‑scale exploration in several lengths,
  * a final Adam‑style optimiser on a soft‑max approximation
    of the peak of the self‑convolution,
  * a tiny amount of post‑processing (trimming trailing zeros).

All helper routines are defined at module level – no closures or lambdas –
so they can be pickled or inspected if needed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (Σa is assumed = √(2·n))."""
    return float(_conv_fft(a).max())


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Incremental mass‑transfer update (exact)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer amount `delta` from `donor` to `receiver`.
    Updates the vector `a` and the full convolution `conv` exactly.
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    # keep a copy of the coefficient vector for the linear part
    a_before = a.copy()

    # linear contribution of the transfer
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic terms (δ²)
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # update the coefficients
    a[donor] -= delta
    a[receiver] += delta

    # protect against tiny negative values caused by rounding
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by the Adam‑style final polish)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution over the 2n‑1 entries

    a_rev = a[::-1]

    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # gradient w.r.t. a (see derivation in the reference code)
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b

# Local search (mass transfer + occasional swaps / smoothing / jitter)
# ----------------------------------------------------------------------


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Fast incremental local search for a given length `n`.
    Returns the best sequence seen and its peak of the self‑convolution.
    """
    target = np.sqrt(2.0 * n)

    # ---- initialise ----------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.4, 0.9)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time                      # in [0,1]

        # cooling schedules (very gentle)
        max_frac = 0.5 * (0.005 / 0.5) ** tau           # max fraction of donor mass to move
        temp = 0.5 * (1e-5 / 0.5) ** tau                # Metropolis temperature

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass transfer (≈70 % of the moves)
        # --------------------------------------------------------------
        if move_type < 0.70:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]                # contribution of each index to the peak
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            # receiver is favoured to be small (inverse‑mass sampling)
            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            # Metropolis acceptance
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] new best max={best_max:.6f}")
            else:
                # revert
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing window (≈15 % of the moves)
        # --------------------------------------------------------------
        elif move_type < 0.85:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end_win = min(start_win + length, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) swap two entries (≈5 % of the moves)
        # --------------------------------------------------------------
        elif move_type < 0.90:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]          # simple swap, sum unchanged
            conv = _conv_fft(a)
            new_max = conv.max()
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] swap improved max={best_max:.6f}")
            else:
                # revert swap
                a[i], a[j] = a[j], a[i]
                conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 4) Gaussian jitter (≈10 % of the moves)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved max={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Adam optimiser on a soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Adam‑style optimiser on the smooth soft‑max approximation of the peak.
    Returns the best vector found and its (exact) peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 1.0
    max_beta = 4096.0
    step_scale = target * 0.12

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = step_scale / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat
        a = _project_to_simplex(a, target)

        val = _max_convolution(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam n={n} β={beta:.1f} val={best_val:.6f}")

        # anneal β
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        # occasional tiny jitter to escape plateaux
        if t % 500 == 0:
            scale = target * 0.001 / (beta + 1.0)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Strategy:
      1. Warm‑start from a previously discovered sequence if available.
      2. Run several independent local‑search runs on a range of
         problem sizes (1500‑4000).  Each run gets a small time slice
         and explores a fresh random power‑law start.
      3. Polish the best candidate with the Adam optimiser.
      4. Trim trailing zeros and expose the best vector as
         `height_sequence_1` for future runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # leave a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Warm‑start from a previously known construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.size
        if time.time() < deadline - 5.0:
            run_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.15)
            seq, score = _local_search(n0, run_deadline, rng, init_seq=seq0)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] warm‑start n={n0} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Exploration on several lengths
    # ------------------------------------------------------------------
    # a few lengths that are known to give a good trade‑off between
    # expressive power and runtime.
    lengths = [1500, 2000, 2500, 3000, 3500, 4000]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # allocate a slice of the remaining time to this length
        remaining = deadline - time.time()
        slice_time = remaining * 0.18
        # each local‑search run will consume roughly 25 s; adjust if needed
        n_runs = max(1, int(slice_time // 25.0))

        for run in range(n_runs):
            if time.time() > deadline - 5.0:
                break

            # random power‑law start
            p = rng.uniform(0.3, 1.2)
            target = np.sqrt(2.0 * n)
            a0 = _power_law_sequence(n, p, target)
            # a small random jitter before entering the search
            a0 += rng.normal(scale=target * 0.01, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target)

            run_deadline = time.time() + min(22.0, (deadline - time.time()) * 0.13)
            seq, score = _local_search(n, run_deadline, rng, init_seq=a0)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] size={n} run={run} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final Adam‑type polishing (if there is still time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra_deadline = time.time() + (deadline - time.time()) * 0.30
        seq, val = _adam_optimize(best_seq, extra_deadline, rng)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] Adam polish improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Strip trailing zeros – they never affect the objective
    # ------------------------------------------------------------------
    if best_seq is not None:
        arr = np.array(best_seq, dtype=float)
        nz = np.where(arr > 1e-12)[0]
        if nz.size:
            arr = arr[: nz[-1] + 1]
        else:
            arr = np.array([], dtype=float)
        best_seq = arr.tolist()
        # make the sequence available for the next invocation
        globals()["height_sequence_1"] = best_seq

    print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq) if best_seq else 0}")
    return best_seq
