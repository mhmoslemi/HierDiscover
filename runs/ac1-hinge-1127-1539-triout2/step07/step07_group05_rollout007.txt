    it = 0

        b = _max_convolution_fft(a) # Actually we also need full b for gradient indices.

        mask = (b_full >= max_b - 1e-12)  # bool array

        # Use FFT for correlation: grad = 2 * iff FT(a) * conj(FT(mask))

        grad = 2.0 * grad_full[n-1:n-1 + n]

        it += 1
    return a

def _mass_transfer_annealing(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_b = b.copy()
    best_max = b.max()
    T = best_max * 0.02 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # pick donor index with a_i > epsilon
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # choose delta
        max_transfer = a[i] * rng.uniform(0.1, 0.5)
        delta = max_transfer
        # apply transfer
        # compute new b copy
        b_new = b.copy()
        # shift contributions
        b_new[i:i+n] -= 2 * delta * a
        b_new[j:j+n] += 2 * delta * a
        # delta^2 terms
        b_new[2*i] += delta*delta
        b_new[2*j] += delta*delta
        b_new[i + j] -= 2 * delta * delta
        new_max = b_new.max()
        dE = new_max - best_max
        if new_max < best_max or rng.random() < np.exp(-dE / T):
            # accept move
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                best_b = b.copy()
        # else reject: no changes
        T *= cooling
        it += 1
        # occasional global jitter/reshuffle
        if it % 1000 == 0:
            # small random reseed of a few entries
            idx = rng.choice(n, size=int(0.01*n), replace=False)

            vals = a[idx]

            a_fft = np.fft.rfft(a, n=size)
            b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    return best_a

"""
Optimiser for the “max‑self‑convolution” problem.

The objective after normalising Σ a = √(2 n) is simply
    max_k (a * a)[k] .
The algorithm proceeds in three stages.

1.  Random parametric families (power‑law, exponential, linear,
    quadratic, mixture) generate a diverse elite set of candidates.
2.  Each elite candidate is refined by *exact* sub‑gradient descent
    on the true max‑function (no smooth surrogate).  The sub‑gradient
    is obtained by a single FFT‑based correlation with the mask of
    maximal convolution entries.
3.  A dedicated *mass‑transfer simulated annealing* moves a small
    amount of mass from one index to another while updating the
    convolution incrementally (O(n) per move).  This fine‑tunes the
    ordering and the values and usually pushes the objective a few
    thousandths lower.

All steps respect a global time budget (default 1000 s).  The best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the initial random search)
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on  f(a)=max_k (a*a)[k] .
    Uses the exact sub‑gradient (correlation with the mask of maximal
    entries) and simple momentum.  The ℓ¹‑sum is kept fixed by projection.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0

    while time.time() < deadline:
        # full convolution
        b_full = _full_convolution_fft(a)
        max_b = b_full.max()
        # mask of maximal positions (allow tiny tolerance)
        mask = (b_full >= max_b - 1e-12).astype(np.float64)

        # correlation a ⋆ mask  -> gradient (central slice)
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        mask_fft = np.fft.rfft(mask, n=size)
        corr_full = np.fft.irfft(a_fft * np.conj(mask_fft), n=size)[: 2 * n - 1]
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm + 1e-12)
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Mass‑transfer simulated annealing (continuous values)
# ----------------------------------------------------------------------
def _mass_transfer_annealing(a0: np.ndarray, b0: np.ndarray,
                            deadline: float) -> np.ndarray:
    """
    Improves a fixed multiset by repeatedly moving a small amount of mass
    from one index to another.  The convolution is updated incrementally,
    making each move O(n) rather than O(n log n).  A simple exponential
    cooling schedule is used.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_b = b.copy()
    best_max = b.max()

    # initial temperature – a few percent of the current objective
    T = best_max * 0.02 + 1e-12
    cooling = 0.9995
    rng = np.random.default_rng()

    # re‑normalise every few thousand steps to avoid drift
    recompute_every = 4000
    steps = 0

    while time.time() < deadline:
        # Choose donor (i) and receiver (j)
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer amount: a fraction of the donor’s mass
        delta = a[i] * rng.uniform(0.05, 0.30)

        # ----- incremental update of the convolution -----
        # shift contributions
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic delta^2 terms
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        dE = new_max - best_max

        accept = False
        if new_max <= best_max:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                best_b = b.copy()
        # else: reject – keep a,b unchanged

        T *= cooling
        steps += 1

        # periodic full recomputation to suppress numerical drift
        if steps % recompute_every == 0:
            size = 1 << ((2 * n - 1).bit_length())
            a_fft = np.fft.rfft(a, n=size)
            b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        # very occasional tiny global jitter
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                size = 1 << ((2 * n - 1).bit_length())
                a_fft = np.fft.rfft(a, n=size)
                b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
                if b.max() < best_max:
                    best_max = b.max()
                    best_a = a.copy()
                    best_b = b.copy()

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    maximal entry of its self‑convolution.  The routine respects the
    overall time budget, explores several parametric families,
    refines a set of elite candidates with exact sub‑gradient descent,
    and finishes with a dedicated mass‑transfer annealing phase.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf
    elite = []       # list of (obj, array)

    # ------------------------------------------------------------------
    # Helper to keep the current best and elite list up‑to‑date
    # ------------------------------------------------------------------
    def _consider(seq: np.ndarray):
        nonlocal best_seq, best_obj, elite
        obj = _max_convolution_fft(seq)
        if obj < best_obj:
            best_obj = obj
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New global best obj={best_obj:.6f} len={seq.size}")
        if len(elite) < 12 or obj < elite[-1][0]:
            elite.append((obj, seq.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:12]

    # ------------------------------------------------------------------
    # Warm start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        n0 = init.size
        init = _project_to_simplex(init, np.sqrt(2.0 * n0))
        _consider(init)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Random parametric search (covers many families & orderings)
    # ------------------------------------------------------------------
    lengths = [2500, 3000, 3500, 4000, 5000, 6000, 8000, 10000, 12000]
    families = ['power', 'exp', 'linear', 'quadratic', 'mixed']

    # Rough budget split: 65 % for random generation, 35 % for refinement
    gen_deadline = time.time() + 0.65 * (deadline - time.time())

    while time.time() < gen_deadline:
        n = int(rng.choice(lengths))
        target_sum = np.sqrt(2.0 * n)
        family = rng.choice(families)

        if family == 'power':
            p = rng.uniform(0.30, 0.90)
            shift = rng.uniform(0.0, 5.0)
            base = _powerlaw_sequence(n, p, shift)

        elif family == 'exp':
            rate = rng.uniform(0.0001, 0.02)
            shift = rng.uniform(0.0, 5.0)
            base = _exponential_sequence(n, rate, shift)

        elif family == 'linear':
            slope = rng.uniform(0.00001, 0.01)
            intercept = rng.uniform(0.5, 4.0)
            base = _linear_sequence(n, slope, intercept)

        elif family == 'quadratic':
            a0 = rng.uniform(0.5, 5.0)
            b = rng.uniform(0.0, 0.01)
            c = rng.uniform(0.0, 0.00005)
            base = _quadratic_sequence(n, a0, b, c)

        else:   # mixed
            w = rng.uniform(0.2, 0.8)

            p = rng.uniform(0.30, 0.90)
            shift_p = rng.uniform(0.0, 5.0)
            pow_seq = _powerlaw_sequence(n, p, shift_p)

            rate = rng.uniform(0.0001, 0.02)
            shift_e = rng.uniform(0.0, 5.0)
            exp_seq = _exponential_sequence(n, rate, shift_e)

            mix = w * pow_seq + (1.0 - w) * exp_seq
            base = _project_to_simplex(mix, target_sum)

        # ----- deterministic orderings (they often have a strong impact) -----
        asc = np.sort(base)                # increasing
        desc = asc[::-1]                   # decreasing
        orderings = [desc, asc]

        # alternating small–large (small first)
        alt = np.empty_like(asc)
        lo, hi = 0, n - 1
        for i in range(n):
            if i % 2 == 0:
                alt[i] = asc[lo]
                lo += 1
            else:
                alt[i] = asc[hi]
                hi -= 1
        orderings.append(alt)

        # a single random shuffle
        shuf = asc.copy()
        rng.shuffle(shuf)
        orderings.append(shuf)

        for cand in orderings:
            _consider(cand)

        # occasional cheap permutation annealing on the current global best
        if rng.random() < 0.07 and best_seq is not None:
            tmp = _permutation_annealing(best_seq, time.time() + 0.25)
            _consider(tmp)

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (sub‑gradient descent + mass‑transfer)
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break

        n = seed_seq.size
        # sub‑deadline for the gradient phase (≈ 15 s or remaining time)
        sub_dead = min(deadline, time.time() + 15.0)

        refined = _subgradient_descent_opt(seed_seq, sub_dead)

        # recompute full convolution for the annealing phase
        b_full = _full_convolution_fft(refined)

        # remaining time for mass‑transfer annealing (global deadline)
        refined = _mass_transfer_annealing(refined, b_full, deadline)

        _consider(refined)

    # ------------------------------------------------------------------
    # 3. Final global mass‑transfer annealing on the best sequence
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        b_full = _full_convolution_fft(best_seq)
        best_seq = _mass_transfer_annealing(best_seq, b_full, deadline)
        best_obj = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 4. Fallback in case nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing near‑zero entries (cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Optional verification with the official evaluator
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
