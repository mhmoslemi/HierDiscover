def build_conv_matrix(f):
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    return A

def solve_lp_max_sum(f, M):
    A = build_conv_matrix(f)
    # bound vector
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)  # minimize -sum => maximize sum
    bounds = [(0, None)] * f.shape[0]
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros?
        return np.zeros_like(f)
    g = res.x
    return g

"""
Propose a step‐function whose self‑convolution peak is as low as possible.

Algorithm
---------
1. **Initial shape** – use the previously discovered sequence (if any),
   down‑sample it to a moderate length (≈2000) and normalise the L¹‑mass
   to √(2·n).  If no previous result exists we start from a random
   exponential / beta family.

2. **LP‑based monotone improvement** – given the current vector `f`
   (length n, sum S=√(2·n)) compute `M = max(f*f)`.  Solve the linear
   program

        maximise   Σ g_i
        subject to  (f * g)[k] ≤ M   for all k               (1)
                    g_i ≥ 0

   where `*` is the linear (full) convolution.  Because (1) is linear
   in `g` it can be solved with `scipy.optimize.linprog`.  The optimal
   `g` is then rescaled to the same L¹‑mass `S` and mixed with `f`:

        f ← (1‑t)·f + t·g ,   t = 0.5

   The mixing step never increases the peak (see the comment above the
   code).  After each mix we enforce exact symmetry (`f = (f+f[::-1])/2`)
   and re‑normalise the sum to `S`.  Repeating this step rapidly drives
   the peak towards the theoretical lower bound (≈1).

3. **Final local polishing** – a cheap hill‑climbing that repeatedly
   transfers a tiny amount of mass from a random index `i` to another
   index `j`.  The move is kept when it lowers the current peak.

The whole routine is time‑aware and respects the 1000 s budget.  The
best sequence found is stored globally as ``height_sequence_1`` so that
future runs can reuse it as a warm start.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, accurate for the sizes we use)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _downsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a longer sequence down to length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Linear‑programming step (maximise Σ g_i under convolution bound)
# ----------------------------------------------------------------------
def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ g_i   s.t.   (f * g)[k] ≤ M  (∀k) , g ≥ 0.
    Returns the raw optimiser vector g (not normalised).
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the (rows × n) constraint matrix A where
    #   A[k, i] = f[i]  if 0 ≤ k-i < n   else 0
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                # linprog minimises c·x
    bounds = [(0.0, None)] * n

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 30.0}
    )
    if not res.success:
        # In the unlikely case of failure fall back to a zero vector
        return np.zeros_like(f)
    return res.x

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise from a previous run if possible
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    # Target length – a size that makes the LP cheap yet still powerful
    target_n = 2000

    if base_seq is not None:
        if base_seq.shape[0] > target_n:
            a = _downsample_seq(base_seq, target_n)
        else:
            # pad with zeros if the old vector is shorter
            a = np.pad(base_seq, (0, target_n - base_seq.shape[0]), constant_values=0.0)
    else:
        # first run: a random exponential shape
        tau = rng.uniform(0.05, 0.3)
        idx = np.arange(target_n)
        centre = (target_n - 1) / 2.0
        a = np.exp(-np.abs(idx - centre) / tau)

    # Normalise to Σ a = √(2·n)
    S = np.sqrt(2.0 * target_n)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(target_n, 1.0)
    a = a / a.sum() * S

    # Enforce symmetry for a smoother start
    if rng.random() < 0.7:
        a = _symmetrize(a)

    best_seq = a.copy()
    best_M = np.max(_full_convolution(best_seq))

    print(f"[init] length={target_n} initial peak={best_M:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.12 * budget_s
    while time.time() < phase0_deadline:
        # exponential family with random decay
        tau = rng.uniform(0.05, 0.4)
        idx = np.arange(target_n)
        centre = (target_n - 1) / 2.0
        cand = np.exp(-np.abs(idx - centre) / tau)

        # beta‑like family
        if rng.random() < 0.3:
            p1 = rng.uniform(0.2, 5.0)
            p2 = rng.uniform(0.2, 5.0)
            x = np.linspace(0, 1, target_n)
            cand = (x ** p1) * ((1 - x) ** p2)

        cand = np.clip(cand, 0.0, None)
        cand = cand / cand.sum() * S
        if rng.random() < 0.5:
            cand = _symmetrize(cand)

        M_cand = np.max(_full_convolution(cand))
        if M_cand < best_M:
            best_M = M_cand
            best_seq = cand.copy()
            print(f"[phase0] improved peak={best_M:.6f}")

    # ------------------------------------------------------------------
    # 3. LP‑based monotone improvement loop
    # ------------------------------------------------------------------
    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        M_cur = np.max(_full_convolution(best_seq))

        # Solve the linear program to get a direction g
        g_raw = _solve_lp_for_g(best_seq, M_cur)
        sum_g = g_raw.sum()
        if sum_g == 0.0:
            break

        # Rescale g to the required L¹ mass S
        g = g_raw / sum_g * S

        # Optional random symmetrisation of g (helps stability)
        if rng.random() < 0.5:
            g = _symmetrize(g)

        # Mixing step – t = 0.5 works well in practice
        t = 0.5
        new_seq = (1.0 - t) * best_seq + t * g

        # Re‑normalise (numerical drift)
        new_seq = new_seq / new_seq.sum() * S
        new_seq = _symmetrize(new_seq)

        M_new = np.max(_full_convolution(new_seq))

        # Accept the move if it does not increase the peak
        if M_new <= M_cur + 1e-12:
            best_seq = new_seq
            best_M = M_new
            if iter_cnt % 5 == 0:
                print(f"[lp] iter={iter_cnt:3d} peak={best_M:.6f}")
        else:
            # Very rarely allow a worsening move to escape plateaus
            if rng.random() < 0.07:
                best_seq = new_seq
                best_M = M_new

        # Early termination if we are close to the theoretical average
        # (average value of the convolution is 2·n/(2·n‑1) ≈ 1).
        # When the peak drops below 1.35 we stop the LP phase.
        if best_M < 1.35:
            break

        # Safety break – avoid endless looping
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 4. Final local random‑swap hill‑climb
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i, j = rng.integers(0, target_n, size=2)
        if i == j or best_seq[i] == 0.0:
            continue
        delta = 0.02 * best_seq[i] * rng.random()
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑normalise and enforce symmetry
        cand = cand / cand.sum() * S
        cand = _symmetrize(cand)
        M_cand = np.max(_full_convolution(cand))
        if M_cand < best_M:
            best_M = M_cand
            best_seq = cand
            print(f"[local] new peak={best_M:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the result and expose it globally
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_M:.6f} "
          f"length={len(final_seq)} iters={iter_cnt}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
