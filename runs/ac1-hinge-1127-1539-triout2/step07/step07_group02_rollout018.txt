    while time.time() < deadline:

        # linear temp decay

        # choose i with a[i] > 0
        i = rng.choice(np.where(a > 0)[0])
        # choose j random in [0,n)
        j = rng.integers(0, n)
        if i == j: continue
        delta = a[i] * rng.random() * move_factor
        if delta <= 0: continue
        # apply diff using a (old)
        conv[i:i+n] -= 2 * delta * a
        conv[j:j+n] += 2 * delta * a
        a[i] -= delta
        a[j] += delta
        # compute new max
        new_max = conv.max()
        # acceptance
        if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-12)):

            a[i] += delta
            a[j] -= delta
            conv[i:i+n] += 2 * delta * a

        iteration += 1
        # periodic recompute
        if iteration % recalc_interval == 0:
            conv = np.convolve(a, a)
            cur_max = conv.max()
    return best_a, best_max

"""
Propose a step‑function (non‑negative vector) with minimal self‑convolution peak.

Strategy
--------
1. **Exploratory sampling** – generate many analytic families (uniform,
   geometric “U‑shape”, power‑law, beta‑like, sinusoidal).  Each candidate is
   normalised so that Σa = √(2·n).  The best candidate is kept as a starter.

2. **Projected Adam** – minimise a smooth log‑sum‑exp approximation of the
   max‑convolution while staying on the simplex.  The temperature (α) is
   increased up to 500 to tighten the bound.

3. **Sub‑gradient descent** – directly reduces the true maximum entry of the
   convolution using the (2·a_{k‑i}) gradient of the peak position.

4. **Fast simulated‑annealing local search** – moves a tiny amount of mass
   from a randomly chosen index *i* to *j* (Δ = ε·a[i]).  The convolution is
   updated **in‑place** using the identity

        conv ← conv – 2·Δ·a (shifted by i) + 2·Δ·a (shifted by j)

   which costs O(n) per move and avoids a full FFT recomputation.  A
   temperature schedule permits occasional uphill moves, and the best state
   is remembered.  After a few thousand steps the convolution is recomputed
   exactly to wipe out accumulated round‑off.

The whole pipeline is time‑aware (budget 1000 s) and returns the best
sequence found.  The final score printed by ``evaluate_sequence`` equals the
maximum entry of the self‑convolution (because the vector is always normalised
to Σa = √(2·n)).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _normalise(a: np.ndarray) -> np.ndarray:
    """Scale a so that Σa = √(2·len(a))."""
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a)
        s = a.sum()
    return a / s * S


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (full length = 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution)."""
    return np.max(_full_convolution_fft(a))


def _generate_family(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose one analytic family and return a raw (non‑negative) vector."""
    fam = rng.integers(0, 6)
    if fam == 0:                       # geometric U‑shape (both ends)
        rate = rng.uniform(0.94, 0.9995)
        left = rate ** np.arange(n)
        a = left + left[::-1]
    elif fam == 1:                     # uniform random
        a = rng.random(n)
    elif fam == 2:                     # power‑law symmetric
        p = rng.uniform(0.3, 3.5)
        idx = np.arange(n, dtype=float)
        a = ((idx + 1) * (n - idx)) ** p
    elif fam == 3:                     # beta‑like (peaked in centre)
        p1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.1, 5.0)
        x = np.linspace(0, 1, n)
        a = (x ** p1) * ((1 - x) ** p2)
    elif fam == 4:                     # sinusoidal power
        p = rng.uniform(0.5, 4.0)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    else:                               # mixed: linear interpolation between two extremes
        a = np.linspace(0, 1, n) ** rng.uniform(0.5, 2.0)
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Optimisers – Adam (smooth max) and sub‑gradient (true max)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max of the convolution."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.1                 # start temperature for log‑sum‑exp
    alpha_mul = 1.06
    max_alpha = 500.0

    best_a = a.copy()
    best_score = np.max(_full_convolution_fft(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        m_scaled = np.max(scaled)
        exp_shift = np.exp(scaled - m_scaled)
        w = exp_shift / exp_shift.sum()               # probability vector

        # gradient of (1/α)·log‑sum‑exp(α·conv) w.r.t. a
        # Conv gradient = 2·(w convolved with a) (see derivation in analysis)
        # Use FFT for speed
        Fa = np.fft.rfft(a, n=conv.shape[0] + a.shape[0] - 1)
        Fw = np.fft.rfft(w[::-1], n=conv.shape[0] + a.shape[0] - 1)
        full = np.fft.irfft(Fa * Fw)[len(w) - 1:len(w) - 1 + n]  # size n
        grad = 2.0 * full

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a, S)

        # schedule
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

        # occasional jitter to escape plateaus
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true maximum convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution_fft(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # gradient of (a*a)[k_max] w.r.t. a
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g = np.zeros_like(a)
        # contribution 2·a[j] (except when i==j, still 2·a[i])
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution_fft(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random swap for exploration
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast simulated‑annealing local search (in‑place convolution updates)
# ----------------------------------------------------------------------
def _local_anneal(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny‑mass swaps with in‑place convolution update.
    Accepts uphill moves according to a simple temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _full_convolution_fft(a)          # exact full convolution (len 2n‑1)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    # temperature schedule (start ≈ 5 % of current peak, decay to ≈0)
    T0 = max(1e-6, cur_max * 0.05)
    start = time.time()
    total = deadline - start
    move_factor = 0.07          # maximal fraction of a[i] moved in one step
    recalc_interval = max(2000, n // 4)   # periodic full recompute

    it = 0
    while time.time() < deadline:
        # linear temperature decay
        elapsed = time.time() - start
        T = max(1e-12, T0 * (1.0 - elapsed / total))

        # pick a source index with positive mass
        src_candidates = np.where(a > 0)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(0, n)               # destination (may equal i – rejected later)
        if i == j:
            continue

        # choose a small delta proportional to the current source mass
        delta = a[i] * rng.random() * move_factor
        if delta <= 0:
            continue

        # store old values for possible revert
        ai_old = a[i]
        aj_old = a[j]

        # ---- apply in‑place convolution update using the *old* a ----
        conv[i:i + n] -= 2.0 * delta * a          # shift by i
        conv[j:j + n] += 2.0 * delta * a          # shift by j

        # ---- update the vector itself ----
        a[i] = ai_old - delta
        a[j] = aj_old + delta

        new_max = conv.max()

        # acceptance test (Metropolis)
        if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        else:
            # ---- revert ----
            a[i] = ai_old
            a[j] = aj_old
            conv[i:i + n] += 2.0 * delta * a      # a now restored ⇒ original a
            conv[j:j + n] -= 2.0 * delta * a

        it += 1
        # periodic exact recomputation to avoid drift
        if it % recalc_interval == 0:
            conv = _full_convolution_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function whose self‑convolution peak is minimal.
    Returns the best sequence (list of floats) before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Load previous best if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
            base_seq = _normalise(base_seq)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # 2.  Phase‑1 : cheap random families (≈15 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.15 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1
        # decide a length – either upscale previous best or sample afresh
        if base_seq is not None and rng.random() < 0.45:
            scale = rng.uniform(0.9, 1.15)
            n = max(800, int(base_seq.shape[0] * scale))
            a = np.interp(np.linspace(0.0, 1.0, n),
                          np.linspace(0.0, 1.0, base_seq.shape[0]),
                          base_seq)
        else:
            n = int(rng.integers(2000, 12001))

        a = _generate_family(n, rng)
        a = _normalise(a)

        # optional random symmetrisation (may help but not required)
        if rng.random() < 0.33:
            a = 0.5 * (a + a[::-1])

        cur_score = np.max(_full_convolution_fft(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Phase‑2 : Adam optimisation (≈35 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if best_seq is None:
        # fallback to a plain uniform vector of moderate size
        n = 5000
        a = rng.random(n)
        best_seq = _normalise(a)
        best_score = np.max(_full_convolution_fft(best_seq))

    remain = deadline - now
    adam_deadline = now + 0.35 * remain
    seq_adam, score_adam = _optimize_adam(best_seq, adam_deadline, rng)
    if score_adam < best_score:
        best_score = score_adam
        best_seq = seq_adam.copy()
        print(f"[adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Phase‑3 : Sub‑gradient descent (≈25 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        sub_deadline = now + 0.25 * (deadline - now)
        seq_sub, score_sub = _optimize_subgrad(best_seq, sub_deadline, rng)
        if score_sub < best_score:
            best_score = score_sub
            best_seq = seq_sub.copy()
            print(f"[subgrad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Phase‑4 : Fast local annealing (use up the rest)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        seq_loc, score_loc = _local_anneal(best_seq, deadline, rng)
        if score_loc < best_score:
            best_score = score_loc
            best_seq = seq_loc.copy()
            print(f"[anneal] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Return result (also expose globally for reuse)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)   # external validation (max‑convolution)
    print(f"[final] external={external:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
