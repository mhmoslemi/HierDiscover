  amount of mass from a source index that contributes heavily to the
  current worst convolution entry to a destination index with low mass.
  For each (source,destination) pair several candidate transfer amounts
  are evaluated analytically (exact O(n) update of the convolution) and
  the best one is taken.  An annealed Metropolis rule allows occasional
  uphill moves, improving the ability to escape local minima.

* **Stage‑wise length schedule** – the search is performed for a sequence
  of increasing lengths (2000 → 5000 → 10000 → 20000).  The best vector
  from a stage is up‑sampled (linear interpolation) to initialise the
  next stage.

* **Final polishing** – whatever time remains is spent on a final
  high‑β gradient descent.

All steps respect the ℓ¹‑simplex constraint, keep the entries non‑negative,
and are timed out gracefully: the function returns the best sequence found
before the global deadline (≈ 1000 s).  The output is a plain Python list of
floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p), p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # compute soft‑max probabilities over b in a numerically stable way
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed‑max surrogate.
    Uses momentum, annealed temperature β and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # temperature schedule
    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.3

    # step‑size and momentum
    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers (various families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled to diversify the search space.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine wave (positive half‑wave)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length,
    followed by a projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Improved mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing hill climbing that transfers mass between two
    indices while keeping the full convolution up‑to‑date.
    For each chosen (source, destination) pair several candidate
    transfer amounts are evaluated; the best one (in terms of the new
    maximal convolution entry) is taken.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9  # initial temperature
    cooling = 0.9995

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) pick a worst convolution index M
        # --------------------------------------------------------------
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # --------------------------------------------------------------
        # 2) pick a source index i weighted by its contribution to entry M
        # --------------------------------------------------------------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # --------------------------------------------------------------
        # 3) pick a destination index j (biased towards low‑mass positions)
        # --------------------------------------------------------------
        if rng.random() < 0.5:
            # 30 % of the time choose among the lowest‑mass indices
            n_low = max(1, n // 20)
            low_ids = np.argpartition(a, n_low)[:n_low]
            j = int(rng.choice(low_ids))
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # --------------------------------------------------------------
        # 4) evaluate a few candidate deltas; keep the best
        # --------------------------------------------------------------
        max_delta = a[i] * 0.95   # keep a[i] non‑negative after the move
        if max_delta <= 0.0:
            continue

        # generate a few candidate deltas (including a very small one)
        cand_deltas = [max_delta * rng.random() for _ in range(3)]
        cand_deltas.append(max_delta * 0.01)

        best_delta = 0.0
        best_new_score = cur_score
        best_b_temp = None

        for delta in cand_deltas:
            # linear part of the update
            b_temp = b.copy()
            b_temp[i:i + n] -= 2.0 * delta * a
            b_temp[j:j + n] += 2.0 * delta * a

            # quadratic correction terms
            k2i = 2 * i
            k2j = 2 * j
            kij = i + j
            if k2i < b_temp.size:
                b_temp[k2i] += delta * delta
            if k2j < b_temp.size:
                b_temp[k2j] += delta * delta
            if kij < b_temp.size:
                b_temp[kij] -= 2.0 * delta * delta

            new_score = b_temp.max()
            if new_score < best_new_score:
                best_new_score = new_score
                best_delta = delta
                best_b_temp = b_temp

        # --------------------------------------------------------------
        # 5) accept the move (Metropolis rule)
        # --------------------------------------------------------------
        if best_delta > 0.0:
            accept = False
            if best_new_score < cur_score:
                accept = True
            else:
                prob = np.exp(-(best_new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

            if accept:
                a[i] -= best_delta
                a[j] += best_delta
                b = best_b_temp
                cur_score = best_new_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] hill‑climb improvement: {best_score:.6f}")

        # --------------------------------------------------------------
        # 6) temperature decay
        # --------------------------------------------------------------
        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the wall‑clock budget
    and returns the best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        early = globals()["height_sequence_1"]
        if isinstance(early, (list, tuple, np.ndarray)):
            a0 = np.asarray(early, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – increasing dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – gradient‑descent restarts
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 8.0:
            break
        gd_budget = min(remaining * 0.30, 250.0)          # allocate ≤30 % of remaining time
        n_restarts = 6 if n <= 10000 else 4
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            now = time.time()
            per_restart = min((deadline - now) * 0.33, 80.0)
            gd_deadline = now + per_restart

            a_start = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_gd = _gradient_descent_opt(a_start, gd_deadline)
            score = _max_convolution_fft(a_gd)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_gd.copy()

        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time():.1f}] GD improvement for n={n} score={local_best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – improved hill climbing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 8.0:
            break
        hill_budget = min(remaining * 0.35, 300.0)   # allocate up to ~300 s
        hill_deadline = time.time() + hill_budget

        # make sure we are working with the current length
        if best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = best_seq.copy()
        a_current = _project_to_simplex(a_current, target_sum)
        b_full = _full_convolution(a_current)

        a_hc, b_hc = _hill_climbing_improved(a_current, b_full, hill_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Phase C – short refinement gradient descent
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            gd2_deadline = time.time() + min(remaining * 0.15, 60.0)
            a_refine = _gradient_descent_opt(best_seq.copy(), gd2_deadline)
            score_ref = _max_convolution_fft(a_refine)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_refine.copy()
                print(f"[{time.time():.1f}] Post‑HC GD refinement n={n} score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        final_deadline = deadline
        a_final = _gradient_descent_opt(a, final_deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: trim trailing near‑zeros for a tidy output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation for debugging
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
