def _lp_update(a, M, target_sum, t=0.2, deadline=None):
    n = a.size
    rows = 2*n - 1
    # build conv matrix A
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # fill
        # f index = k - j
        idx = np.arange(start, end+1)
        A[k, idx] = a[k - idx]
    # solve LP
    from scipy.optimize import linprog
    c = -np.ones(n)  # maximize sum
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M, dtype=np.float64)
    # optionally use callback to enforce deadline? We'll just rely on time.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a    # no update
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 < 1e-12:
        return a
    g = g0 * (target_sum / sum_g0)
    a_new = (1 - t) * a + t * g
    # ensure nonnegative and sum
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------

1.  Initialise a base sequence.
    * If a previously found construction ``height_sequence_1`` exists,
      it is used and projected onto the ℓ¹‑simplex
      Σa = √(2 n).  Otherwise a random Dirichlet draw is taken.
    * The length is kept at the size of the seed (≈2000).  Longer
      lengths were experimentally slower for the exact LP updates.

2.  **Iterative linear‑programming refinement**  
    For a few iterations (while time permits) we solve the LP

        maximise   Σ g_j
        subject to  A·g ≤ M ,   g ≥ 0

    where ``A`` is the convolution matrix of the current sequence
    (so ``A·g = f*g``) and ``M`` = max(f*f).  After solving we
    rescale ``g`` to have the required ℓ¹‑norm and mix it with the
    current vector:   f←(1‑t)·f + t·g .  This is the “alternating
    LP” technique from the reference description and is very effective
    at decreasing the maximal self‑convolution.

3.  **Targeted hill‑climbing**  
    With the remaining budget a specialised simulated‑annealing style
    hill‑climber moves mass from an index that contributes most to the
    current worst convolution entry to a low‑mass destination.
    Each candidate move is evaluated exactly (O(n) update of the full
    convolution) and accepted if it strictly reduces the current
    maximum.  The temperature schedule is deliberately gentle to
    allow many improving steps.

4.  **Final polishing** – a short soft‑max gradient descent
    (projected onto the simplex) followed by a brief hill‑climb.

All steps respect the 1000 s wall‑clock budget.  The best vector found
is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based for max, full for updates)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (using FFT)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Random initialisers (used if no seed is available)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    else:                               # near‑uniform with jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step” from the reference)
# ----------------------------------------------------------------------


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = f * g
    (full linear convolution).  Each column i contains the vector f
    starting at row i.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _lp_refine(f: np.ndarray, M: float, target_sum: float,
               mix: float = 0.3) -> np.ndarray:
    """
    One LP refinement step:
        maximise   Σ g_j
        subject to (f * g)_k ≤ M  (k=0..2n‑2),  g ≥ 0
    The solution g is rescaled to the required ℓ¹‑norm and mixed with f.
    """
    n = f.size
    A = _build_convolution_matrix(f)          # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises c·x ⇒ maximise Σ g  ↔  minimise -Σ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – keep the original vector
        return f.copy()

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 1e-12:
        return f.copy()

    # rescale to the exact ℓ¹‑norm and mix
    g = g0 * (target_sum / sum_g0)
    f_new = (1.0 - mix) * f + mix * g
    # tiny numerical drift – re‑project onto the simplex
    f_new = _project_to_simplex(f_new, target_sum)
    return f_new


# ----------------------------------------------------------------------
# Targeted hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing that always moves mass from
    a source index that contributes most to the current worst convolution
    entry to a low‑mass destination.  The full convolution is kept
    up‑to‑date in O(n) per move.
    Returns the best vector and its full convolution.
    """
    a = a0.copy()
    b = _full_convolution(a)
    n = a.size
    cur_score = b.max()
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    T = cur_score * 0.5 + 1e-9          # initial temperature (generous)
    cooling = 0.9997                    # very slow cooling – many refinements

    while time.time() < deadline:
        # --- locate the worst convolution entry --------------------------------
        k_max = int(np.argmax(b))
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = k_max - idx
        contrib = a[idx] * a[jdx]
        if contrib.size == 0:
            break
        i = idx[np.argmax(contrib)]               # source index

        if a[i] < 1e-14:
            # nothing to move – pick a random source
            i = rng.integers(0, n)

        # --- pick a destination – biased to small entries --------------------
        inv_a = 1.0 / (a + 1e-12)
        prob_j = inv_a / inv_a.sum()
        # draw a few candidates and keep the best one
        cand_js = rng.choice(n, size=7, replace=False, p=prob_j)
        best_candidate = None
        best_new_score = cur_score
        best_delta = 0.0

        # try moving a fraction of the source mass to each candidate
        for j in cand_js:
            if j == i:
                continue
            for frac in (0.5, 0.3, 0.1):
                delta = a[i] * frac
                if delta <= 0.0:
                    continue
                # ----- compute new convolution (O(n) update) -----------------
                b_new = b.copy()
                a_before = a.copy()
                # update slices
                b_new[i:i + n] -= 2.0 * delta * a_before
                b_new[j:j + n] += 2.0 * delta * a_before
                k2i = 2 * i
                k2j = 2 * j
                kij = i + j
                if k2i < b_new.size:
                    b_new[k2i] += delta * delta
                if k2j < b_new.size:
                    b_new[k2j] += delta * delta
                if kij < b_new.size:
                    b_new[kij] -= 2.0 * delta * delta
                new_score = b_new.max()
                # keep the best improving move
                if new_score < best_new_score - 1e-12:
                    best_new_score = new_score
                    best_candidate = j
                    best_delta = delta

        # --- apply the best move if it improves -------------------------------
        if best_candidate is not None:
            j = best_candidate
            delta = best_delta
            # apply to the *original* a
            a[i] -= delta
            a[j] += delta
            # update b using the same formula as above (now with original a)
            b[i:i + n] -= 2.0 * delta * a_before
            b[j:j + n] += 2.0 * delta * a_before
            if 2 * i < b.size:
                b[2 * i] += delta * delta
            if 2 * j < b.size:
                b[2 * j] += delta * delta
            if i + j < b.size:
                b[i + j] -= 2.0 * delta * delta
            cur_score = best_new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                # occasional logging
                print(f"[{time.time():.1f}] hill‑climb improved max={best_score:.6f}")

        # Metropolis acceptance for non‑improving moves (exploration)
        else:
            # propose a tiny random jitter move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            delta = a[i] * rng.random() * 0.05
            if delta <= 0.0:
                continue
            b_new = b.copy()
            a_before = a.copy()
            b_new[i:i + n] -= 2.0 * delta * a_before
            b_new[j:j + n] += 2.0 * delta * a_before
            k2i = 2 * i
            k2j = 2 * j
            kij = i + j
            if k2i < b_new.size:
                b_new[k2i] += delta * delta
            if k2j < b_new.size:
                b_new[k2j] += delta * delta
            if kij < b_new.size:
                b_new[kij] -= 2.0 * delta * delta
            new_score = b_new.max()
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                # accept
                a[i] -= delta
                a[j] += delta
                b = b_new
                cur_score = new_score

        T *= cooling

    return best_a, _full_convolution(best_a)


# ----------------------------------------------------------------------
# Soft‑max gradient descent (used for final polishing)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed max (log‑sum‑exp) of the convolution.
    Returns (g, b) where b = a*a and g = 2·(a ⋆ p) with p the softmax
    probabilities over the convolution entries.
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected (simple) gradient descent on the smoothed max.
    Uses momentum, β‑annealing and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # hyper‑parameters
    beta = 2.0
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.2
    rho = 0.90
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0005 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Seed vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=float)

        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n})")
    else:

        a = _random_initial_sequence(n, rng)

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------

    lp_iter = 0

        M = _max_convolution_fft(a)
        # mix factor decreases a little each iteration (large steps early)
        mix = 0.5 if lp_iter == 0 else 0.3 if lp_iter == 1 else 0.15
        a_new = _lp_refine(a, M, target_sum, mix=mix)
        M_new = _max_convolution_fft(a_new)
        if M_new < M - 1e-7:
            a = a_new
            print(f"[{time.time():.1f}] LP iteration {lp_iter} improved max={M_new:.6f}")
        else:
            # no progress – break early
            print(f"[{time.time():.1f}] LP iteration {lp_iter} made no improvement")
            break
        lp_iter += 1

    # ------------------------------------------------------------------
    # 3️⃣  Aggressive hill‑climbing with the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a, b = _hill_climb_targeted(a, deadline - 1.0)
        print(f"[{time.time():.1f}] Hill‑climb finished, max={b.max():.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing (gradient descent + short hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a = _gradient_descent_opt(a, deadline - 1.5)
        a, b = _hill_climb_targeted(a, deadline - 0.5)

    # ------------------------------------------------------------------
    # 5️⃣  Trim trailing near‑zeros and evaluate
    # ------------------------------------------------------------------
    nz = np.where(a > 1e-12)[0]

    else:

        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={a.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return a.tolist()
