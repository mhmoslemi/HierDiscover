"""
Improved search for a non‑negative step function that minimises the
self‑convolution peak (the same objective that ``evaluate_sequence`` returns).

Strategy
--------
1.  Use any previously found sequence (``height_sequence_1``) as a seed.
2.  A short projected gradient descent on a smoothed max‑objective (log‑sum‑exp)
    quickly brings the sequence close to a local optimum.  The smoothing
    parameter β is slowly increased up to 10⁶.
3.  A specialised hill‑climber finishes the job.
    *  It always moves mass from an index that contributes heavily to the
       currently worst convolution entry.
    *  For the destination it tries a small set of promising candidates
       (low‑mass indices, neighbours, random spots) and keeps the one that
       yields the smallest new peak.
    *  A simple simulated‑annealing scheme with a slowly cooling temperature
       allows occasional uphill moves.
    *  Convolution updates are performed in O(n) using the analytic update
       formula, so millions of moves can be tried within the budget.
4.  The whole procedure is run for a schedule of lengths
    [2000, 4000, 8000]; larger lengths give a finer discretisation and typically
    allow a lower maximal entry.
5.  The best sequence found so far is refined once more with gradient descent
    and a final hill‑climb.

The code respects the 1000 s wall‑clock limit, never writes to disk and relies
only on NumPy and the already‑imported ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (using FFT)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Uses momentum, β‑annealing up to 1e6 and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # hyper‑parameters
    beta = 2.0
    beta_max = 1_000_000.0
    beta_factor = 1.5

    step_base = target_sum * 0.2
    rho = 0.90
    momentum = np.zeros_like(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softnorm(a, beta)  # typo placeholder – will be fixed below

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0005 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a


def _gradient_softnorm(a: np.ndarray, beta: float):
    """
    Wrapper that returns the gradient of the smoothed max and the full convolution.
    """
    return _gradient_softmax(a, beta)

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by projection
    onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _apply_move(a: np.ndarray, b: np.ndarray, i: int, j: int, delta: float):
    """
    Return a copy of ``b`` after moving ``delta`` mass from index i to j.
    The convolution update follows the exact analytic formula (O(n) work).
    """
    n = a.size
    b_new = b.copy()

    # linear part of the update
    b_new[i:i + n] -= 2.0 * delta * a
    b_new[j:j + n] += 2.0 * delta * a

    # quadratic correction terms
    k2i = 2 * i
    if k2i < b_new.size:
        b_new[k2i] += delta * delta
    k2j = 2 * j
    if k2j < b_new.size:
        b_new[k2j] += delta * delta
    kij = i + j
    if kij < b_new.size:
        b_new[kij] -= 2.0 * delta * delta

    return b_new


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing.
    At each step we pick a source index heavily involved in the current worst
    convolution entry and test a handful of candidate destination indices.
    The move giving the lowest new peak is performed (or accepted probabilistically).
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()
    best_b = b.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9997                    # slower cooling → finer search

    while time.time() < deadline:
        # --- 1. pick a worst convolution index ---------------------------------
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # --- 2. choose a source index weighted by its contribution to M -------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # --- 3. build a small set of promising destination indices ------------
        dests = []

        # a) indices with very small current mass
        inv_a = 1.0 / (a + 1e-12)
        probs_low = inv_a / inv_a.sum()
        for _ in range(3):
            dests.append(rng.choice(n, p=probs_low))

        # b) pure random indices
        for _ in range(2):
            dests.append(rng.integers(0, n))

        # c) neighbours of the source (helpful for local smoothing)
        dests.append((i + 1) % n)
        dests.append((i - 1) % n)

        # ensure we have at least one different index
        dests = [j for j in dests if j != i]
        if not dests:
            continue

        # --- 4. amount of mass to move -----------------------------------------
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # --- 5. evaluate all candidates, keep the best -------------------------
        best_local_score = cur_score
        best_local_j = None
        best_local_b = None

        for j in dests:
            b_new = _apply_move(a, b, i, j, delta)
            new_score = b_new.max()
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_j = j
                best_local_b = b_new

        # --- 6. apply the best improvement if any ------------------------------
        moved = False
        if best_local_j is not None and best_local_score < cur_score:
            # deterministic improvement
            a[i] -= delta
            a[best_local_j] += delta
            b = best_local_b
            cur_score = best_local_score
            moved = True
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_b = b.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        # --- 7. otherwise accept probabilistically (classic Metropolis) ----------
        if not moved:
            # pick the first destination as a fallback
            j = dests[0]
            b_new = _apply_move(a, b, i, j, delta)
            new_score = b_new.max()
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                a[i] -= delta
                a[j] += delta
                b = b_new
                cur_score = new_score

        # --- 8. temperature cooling ---------------------------------------------
        T *= cooling

    return best_a, best_b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises the
    ``evaluate_sequence`` objective.  The routine respects a wall‑clock budget
    (default 1000 s) and returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seed from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            earlier = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = earlier.size
            earlier = _project_to_simplex(earlier, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(earlier)
            best_seq = earlier.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule – larger n give finer discretisation
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000]

    for n in schedule:
        # stop if we already ran out of time
        if time.time() >= deadline - 5.0:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # initialise a candidate of length n
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – short projected gradient descent (smooth max)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        gd_budget = min(remaining * 0.15, 120.0)     # keep a chunk for later phases
        gd_deadline = time.time() + gd_budget
        a_gd = _gradient_descent_opt(a_init, gd_deadline)

        # evaluate after gradient step
        b_gd = _full_convolution(a_gd)
        cur_score = b_gd.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement for n={n} score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – intensive hill climbing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        hill_deadline = time.time() + remaining * 0.7
        a_hc, b_hc = _hill_climbing_improved(a_gd, b_gd, hill_deadline)
        cur_score = b_hc.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={cur_score:.6f}")

        # early exit if we already beat the target by a comfortable margin
        if best_score <= 1.503 + 1e-4:
            break

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final gradient descent (smooth max) – use the remaining time
        a_final = _project_to_simplex(best_seq.copy(), target_sum)
        a_final = _gradient_descent_opt(a_final, deadline)

        # final short hill climb
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing_improved(a_final, b_final, deadline - 2.0)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
