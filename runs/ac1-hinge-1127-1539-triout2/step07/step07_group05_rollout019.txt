def _adam_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0*n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 1.0
    max_beta = 1e5
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = (target / np.sqrt(np.sum(m_hat**2) + eps)) * (m_hat/(np.sqrt(v_hat)+eps))
        # move opposite direction (to minimize)
        a -= step
        a = _project_to_simplex(a, target)
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta*1.05, max_beta)
        # occasional jitter
        if t % 250 == 0:
            jitter = rng.normal(scale=target * 5e-4 / (beta+1.0), size=n)
            a += jitter
            a = np.maximum(a,0.0)
            a = _project_to_simplex(a, target)
    return a
```

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective (the maximal entry of the self‑convolution).
The routine uses three stages:

1. **Systematic sweep of a 1‑parameter family** – power‑law sequences
   a_i ∝ (i+δ)^‑p (normalised to Σa_i = √(2 n)).  A fine grid over
   the exponent p∈[0.40,0.60] and the shift δ∈[0,10] is explored for a
   few convenient lengths (8000 and 12000).  The best candidate from this
   exhaustive search is kept.

2. **Local refinement** – the best few candidates are refined by
   projected Adam‑gradient descent on a smooth log‑sum‑exp surrogate of
   the max‑convolution, followed by a short permutation‑annealing phase.

3. **Global annealing** – the current global best is finally polished by
   a longer permutation‑annealing run.

All helper functions are defined at module level (no closures).  The
search respects the allotted time budget and returns the best sequence
as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x≥0  (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a   (linear convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smoothed max
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected Adam optimisation of the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size tuned to keep the ℓ¹‑norm roughly correct
        lr = 0.5 * target / (np.linalg.norm(m_hat) + eps)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target)

        # anneal the smoothness
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target * 8e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improves the ordering of a fixed multiset by random swaps."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score or (T > 0.0 and rng.random() < np.exp(-dE / T)):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny random reshuffle of a few entries
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Families of sequences
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift)^{-p}  (descending order kept)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _mixed_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Convex mixture of a power‑law and an exponential tail."""
    w = rng.uniform(0.2, 0.8)

    p = rng.uniform(0.30, 0.90)
    shift_p = rng.uniform(0.0, 5.0)
    pow_seq = _powerlaw_sequence(n, p, shift_p)

    rate = rng.uniform(0.0001, 0.02)
    shift_e = rng.uniform(0.0, 5.0)
    exp_seq = _exponential_sequence(n, rate, shift_e)

    mix = w * pow_seq + (1.0 - w) * exp_seq
    return _project_to_simplex(mix, np.sqrt(2.0 * n))

# Main entry point
# ----------------------------------------------------------------------

        if len(elite) < 12 or obj < elite[-1][0]:

    try:

    # ------------------------------------------------------------------
