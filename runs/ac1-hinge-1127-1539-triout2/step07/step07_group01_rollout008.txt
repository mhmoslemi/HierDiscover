def _anneal_optimize(a, deadline, rng):
    best_a = a.copy()
    best_peak = _max_convolution(a)
    cur_a = a.copy()
    cur_peak = best_peak
    # Temperature schedule linear: start_T = best_peak * 0.2, end_T = 1e-4
    start_T = best_peak * 0.2 + 0.1
    end_T = 1e-4
    total_time = deadline - time.time()
    while time.time() < deadline:
        remaining = deadline - time.time()
        fraction = (total_time - remaining) / total_time
        T = start_T * (1 - fraction) + end_T * fraction
        n = len(cur_a)
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        # ensure i != j
        if i == j:
            continue
        # amount to move
        eps = cur_a[i] * rng.uniform(0.001, 0.01)
        if eps <= 0:
            continue
        cur_a[i] -= eps
        cur_a[j] += eps
        # compute new peak
        new_peak = _max_convolution(cur_a)
        # acceptance
        if new_peak < cur_peak or rng.random() < np.exp(-(new_peak - cur_peak) / max(T, 1e-12)):
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = cur_a.copy()
        else:
            cur_a[i] += eps
            cur_a[j] -= eps
    return best_a, best_peak

Improved step‑function optimizer.

The optimizer works in three stages:

1️⃣  Random exploration – many candidate vectors are drawn from a richer
    collection of parametric families (power‑law, exponential, linear,
    sparse‑gamma, uniform, Dirichlet and a new “sparse random” family).

3️⃣  Final sanity check – the best sequence found is evaluated with the
    supplied ``evaluate_sequence`` function, stored in the global variable
    ``height_sequence_1`` and returned.

The routine respects the 1000 s budget (the last 5 s are kept as a safety

    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        # smooth soft‑max over the full convolution vector
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    # extract the part that corresponds to the original length
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _project_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, sum(x) = S}.
    Based on the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full(n, S / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ≈0).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.6                       # a little more aggressive than the reference

    # initialise temperature (soft‑max smoothing)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 2.0)       # start with a modest amount of smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (soft‑max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak (τ → 0)
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9997                # shrink smoothing more slowly
        lr *= 0.9999                 # very slow decay

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a, best_peak


def _anneal_optimize(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex.
    Moves a tiny amount of mass between two random positions.
    Accepts worsening moves with probability exp(‑Δ / T).
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _max_convolution(cur_a)

    best_a = cur_a.copy()
    best_peak = cur_peak

    # temperature schedule (linear in log‑time)
    start_T = max(0.3 * cur_peak, 0.05)
    end_T = 1e-5
    total_time = deadline - time.time()

    while time.time() < deadline:
        elapsed = deadline - time.time()
        frac = 1.0 - (elapsed / total_time)   # 0 → 1
        T = start_T * (1.0 - frac) + end_T * frac

        # propose a move: take ε from i and give to j
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if i == j:
            continue
        if cur_a[i] <= 0.0:
            continue
        eps = cur_a[i] * rng.uniform(0.001, 0.01)   # 0.1‑1 % of source
        cur_a[i] -= eps
        cur_a[j] += eps

        new_peak = _max_convolution(cur_a)

        # Metropolis acceptance
        accept = False
        if new_peak < cur_peak:
            accept = True
        else:
            prob = np.exp(-(new_peak - cur_peak) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = cur_a.copy()
        else:
            # revert
            cur_a[i] += eps
            cur_a[j] -= eps

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)          # exponent
    offset = rng.uniform(0.0, 15.0)    # shift of the denominator
    mix = rng.uniform(0.0, 1.0)        # uniform background weight
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)   # decay rate
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)                     # height at the edge
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))  # controls width of the plateau
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw


def _init_sparse_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random spikes – the number of spikes is 1‑10 % of n."""
    density = rng.uniform(0.01, 0.10)               # fraction of non‑zero entries
    k = max(1, int(density * n))
    idx = rng.choice(n, size=k, replace=False)
    values = rng.random(k) + 0.1                    # avoid exact zeros
    raw = np.zeros(n)
    raw[idx] = values
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ~0.2 we warm‑start from a known good sequence,
    otherwise we sample from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'sparse'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:  # 'sparse'
        raw = _init_sparse_random(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found and stores it in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 20 % of the total budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + 0.20 * budget_s
    while time.time() < explore_deadline:
        attempt += 1
        # length selection – a broad distribution, favouring moderate sizes
        if rng.random() < 0.7:
            n = int(rng.integers(1500, 5000))
        else:
            n = int(rng.integers(5000, 22000))

        a0 = _initial_vector(n, rng, base_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – progressive scaling + Adam + simulated‑annealing
    # ------------------------------------------------------------------
    if best_seq is None:
        # (unlikely) fall back to a uniform construction
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # Define a schedule of target lengths (geometrically increasing)
    lengths = [best_n]
    cur = best_n
    while cur < 18000:
        cur = int(min(18000, cur * 1.8))
        lengths.append(cur)
    lengths = sorted(set(lengths))

    for target_n in lengths:
        remaining = end_time - time.time()
        if remaining < 30.0:
            break

        # allocate a slice of the remaining budget for this length
        time_slice = min(remaining * 0.35, 180.0)

        # interpolate & re‑normalise
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # Adam optimisation
        adam_deadline = time.time() + time_slice * 0.70
        a_opt, peak_adam = _adam_optimize(a_scaled, adam_deadline, rng)

        # simulated annealing (use the rest of the slice)
        anneal_deadline = time.time() + time_slice * 0.30
        a_opt2, peak_anneal = _anneal_optimize(a_opt, anneal_deadline, rng)

        # decide which result to keep
        final_a = a_opt2 if peak_anneal < peak_adam else a_opt
        final_peak = min(peak_adam, peak_anneal)

        if final_peak < best_score:
            best_score = final_peak
            best_seq = final_a.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
