def max_conv_and_gradient(a):
    n = len(a)
    # compute conv via FFT (fast)
    conv = _conv_fft(a) # length 2n-1
    k = np.argmax(conv)
    max_val = conv[k]
    # compute gradient
    grad = np.zeros_like(a)
    # contribution from a[k-i]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    # indices i_start..i_end inclusive
    idx_i = np.arange(i_start, i_end+1)
    grad[idx_i] += a[k - idx_i]

    # contribution from a[i-k]
    j_start = max(k, 0)
    j_end = min(k + n - 1, n - 1)
    idx_j = np.arange(j_start, j_end+1)
    grad[idx_j] += a[idx_j - k]

    return max_val, grad, k, conv

def _subgradient_descent(initial, target_sum, deadline):
    a = initial.copy()
    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    gamma = 0.9
    base_step = target_sum * 0.2  # step size scaling

    while time.time() < deadline:
        conv = _conv_fft(a)
        k = int(np.argmax(conv))
        max_val = conv[k]
        # compute gradient via vectorized method
        n = a.size
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1)
        grad[idx_i] += a[k - idx_i]

        j_start = max(k, 0)
        j_end = min(k + n - 1, n - 1)
        idx_j = np.arange(j_start, j_end + 1)
        grad[idx_j] += a[idx_j - k]

        # step size schedule
        step = base_step / np.sqrt(it + 1)
        momentum = gamma * momentum + grad
        direction = momentum / (np.linalg.norm(momentum) + 1e-12)
        a -= step * direction
        a = _project_to_simplex(a, target_sum)

        # evaluate score
        cur_score = max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time():.2f}] SGD it={it:6d} k={k} score={best_score:.6f}")
        else:
            no_improve += 1

        # random restart if stagnation
        if no_improve >= 2000:
            # jitter best and restart
            jitter = rng.normal(scale=target_sum * 0.005, size=a.shape)
            a = best_a + jitter
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a, target_sum)
            momentum = np.zeros_like(a)
            no_improve = 0
            print(f"[{time.time():.2f}] Restart after stagnation it={it}")

            a = np.clip(a, 0, None)

        it += 1
    return best_a, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimises the
benchmark value

    2·n·max(conv) / (sum(a)^2) ,

where conv = a * a is the linear convolution of a with itself.
The algorithm works in three phases and respects the 1000 s budget.

Phase‑1 – initial candidates:
    * a resized previous best (if the global variable `height_sequence_1`
      exists)
    * several power‑law shapes
    * a few random normalised vectors
The best of those is kept.

Phase‑2 – projected sub‑gradient descent on the exact
          maximum–convolution term.
          Momentum, an adaptive step size (∝ 1/√t) and occasional jitter
          are used; if no improvement happens for a while a restart from
          the current best with random jitter is performed.
          The ℓ¹‑norm is kept fixed (∑a_i = √(2·n)) by Euclidean projection
          onto the simplex.

Phase‑3 – simulated annealing refinement.
          Moves are mass transfer, short‑window smoothing and swaps.
          Temperature and the allowed transfer fraction decay
          exponentially.  A final short sub‑gradient polishing pass is
          performed if time permits.

All updates keep the entries non‑negative.  Progress and intermediate
scores are printed.  The returned list is trimmed of trailing
near‑zero elements.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For our normalised sequences sum(a) = √(2·n) so this is just max(conv).
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution term
# ----------------------------------------------------------------------


def _subgradient_descent(initial: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution.
    Uses a diminishing step size (∝ 1/√t) and momentum‑free updates.
    Stagnation triggers a random jitter restart.
    """
    a = _project_to_simplex(initial.copy(), target_sum)
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    it = 0
    no_improve = 0
    base_step = target_sum * 0.1          # controls the initial step magnitude

    while time.time() < deadline:
        # ----- compute convolution and gradient of the maximal entry -----
        conv = _conv_fft(a)
        k = int(np.argmax(conv))
        max_val = conv[k]

        n = a.size
        grad = np.zeros_like(a)

        # first contribution: a[k - i] for i where 0 ≤ k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1)
        grad[idx_i] += a[k - idx_i]

        # second contribution: a[i - k] for i where 0 ≤ i-k < n
        j_start = max(k, 0)
        j_end = min(k + n - 1, n - 1)
        idx_j = np.arange(j_start, j_end + 1)
        grad[idx_j] += a[idx_j - k]

        # ----- gradient step -----
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = base_step / (norm * np.sqrt(it + 1))
            a -= step * grad

        # ---- projection back onto the simplex ----
        a = _project_to_simplex(a, target_sum)

        # ---- bookkeeping ----
        cur_score = max_val                     # because sum is fixed
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time():.2f}] SGD it={it:6d} k={k} score={best_score:.6f}")
        else:
            no_improve += 1

        # ---- stagnation handling ----
        if no_improve >= 2000:
            jitter = rng.normal(scale=target_sum * 0.005, size=n)
            a = best_a + jitter
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a, target_sum)
            no_improve = 0
            print(f"[{time.time():.2f}] Restart after stagnation it={it}")

        # ---- occasional small jitter to avoid tiny cycles ----
        if it % 800 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing refinement (similar to the previous baseline)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 10.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.5,
                         max_frac_end: float = 0.01,
                         max_window: int = 100) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Moves: mass transfer, local smoothing, random swap.
    Temperature and allowed transfer fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # most influential index of the current max convolution entry
        k_max = int(np.argmax(conv))

        # sensitivity of each position (derivative of the max entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # ------------------- mass transfer -------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            donor = rng.choice(n, p=weight / total_w) if total_w > 0 else rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.clip(a, 0.0, None)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

    """
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the allotted time
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1) Build a pool of plausible initial sequences
    # ------------------------------------------------------------------
    candidates = []

    # (a) use a previous best if it is present
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq = _resize_sequence(seq, n)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

    # (b) power‑law families
    for p in [0.30, 0.45, 0.55, 0.65, 0.75, 0.85]:
        candidates.append(_power_law_sequence(n, p, target_sum))

    # (c) a few random normalised vectors
    rng = np.random.default_rng()
    for _ in range(3):
        seq = np.abs(rng.normal(scale=1.0, size=n))
        seq = seq / seq.sum() * target_sum
        candidates.append(seq)

    # ------------------------------------------------------------------
    # Pick the best of the initial pool
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for c in candidates:
        sc = _score(c)
        if sc < best_score:
            best_score = sc
            best_a = c.copy()
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Projected sub‑gradient descent (≈35 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    phase1_dead = now + remaining * 0.35
    a_sgd, sc_sgd = _subgradient_descent(best_a, target_sum, phase1_dead)
    if sc_sgd < best_score:
        best_score = sc_sgd
        best_a = a_sgd.copy()
    print(f"[{time.time():.2f}] Sub‑gradient phase best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Simulated annealing refinement (≈50 % of the rest)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 10.0:
        phase2_dead = now + remaining * 0.50
        a_sa, sc_sa = _simulated_annealing(
            best_a,
            phase2_dead,
            start_temp=20.0,
            end_temp=1e-6,
            max_frac_start=0.6,
            max_frac_end=0.02,
            max_window=200,
        )
        if sc_sa < best_score:
            best_score = sc_sa
            best_a = a_sa.copy()
        print(f"[{time.time():.2f}] SA phase best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final short polishing sub‑gradient descent (use whatever time left)
    # ------------------------------------------------------------------
    now = time.time()
    if deadline - now > 5.0:
        a_final, sc_final = _subgradient_descent(best_a, target_sum, deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
        print(f"[{time.time():.2f}] Final polishing best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Clean up, trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_a, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final check with the provided benchmark
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
