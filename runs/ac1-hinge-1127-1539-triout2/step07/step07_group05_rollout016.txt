def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    # optionally track previous max changes
    while time.time() < deadline:
        # compute convolution and current max
        b = np.convolve(a, a)
        max_b = b.max()
        # define active mask
        eps = max_b * 1e-12
        p = (b >= max_b - eps).astype(float)
        # compute gradient using FFT: gradient = 2 * conv(a_rev, p)[n-1:n-1+n]
        size = 1 << ((n + (2*n-1) - 1).bit_length())
        # Actually we need size for convolution of a_rev (size n) and p (size 2n-1)
        # conv_len = n + (2n-1) - 1 = 3n - 2
        conv_len = 3*n - 2
        size2 = 1 << ((conv_len).bit_length())
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size2)
        Pr_fft = np.fft.rfft(p, n=size2)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
        g = 2.0 * conv_rev[n-1:n-1+n]  # length n
        # step size
        norm = np.linalg.norm(g)
        if norm > 0:
            # Choose step decreasing with iteration count: lr = alpha / (iteration+1)
            # use constant factor to keep updates moderate.
            # we'll compute lr = 0.5 * target_sum / (norm + 1e-12)
            lr = 0.5 * target_sum / (norm + 1e-12)
            a -= lr * g
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        else:
            break
    return a

"""
Hybrid search for a non‑negative step function that minimises
evaluate_sequence (i.e. the maximal entry of the self‑convolution).

The algorithm consists of three stages:

1. **Broad parametric sweep** – many families (exponential, power‑law,
   truncated power, logistic, plateau+exponential, mixed) are sampled
   with random parameters for several lengths (2k … 12k).  Each candidate
   is normalised to Σa = sqrt(2·n) so that the evaluation reduces to
   max( a*a ).  The best few (elite) candidates are kept.

2. **Exact sub‑gradient optimisation** – for every elite vector we run a
   short projected sub‑gradient descent on the true max‑convolution
   objective (no smoothing).  The sub‑gradient is obtained by convolving
   the current vector with the 0/1 mask of active maximal constraints.
   After each step the vector is projected onto the ℓ¹‑simplex
   {x ≥ 0, Σx = sqrt(2·n)}.

3. **Permutation annealing** – the ordering of the entries can still
   change the convolution peak.  A simple simulated‑annealing swap loop
   improves the ordering for the refined vectors and finally for the
   global best.

All work respects the 1000 s wall‑clock budget.  The function returns
the best sequence found as a plain ``list[float]``.  If a warm‑start
variable ``height_sequence_1`` is present it is used as an initial
candidate.
"""

import time
import numpy as np
import scipy.sparse as sp  # only for the optional LP warm‑start (not required)

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families for generating candidate shapes
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _truncated_powerlaw_sequence(
    n: int, p: float, shift: float, cutoff_frac: float
) -> np.ndarray:
    """Power law that is zero after a given fraction of the length."""
    cutoff = int(cutoff_frac * n)
    raw = np.zeros(n, dtype=float)
    idx = np.arange(cutoff, dtype=float)
    raw[:cutoff] = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _logistic_sequence(n: int, steep: float, midpoint: float) -> np.ndarray:
    """Monotonically decreasing logistic shape."""
    idx = np.arange(n, dtype=float)
    raw = 1.0 / (1.0 + np.exp(steep * (idx - midpoint)))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _plateau_exponential_sequence(
    n: int, plateau_frac: float, rate: float
) -> np.ndarray:
    """Constant plateau (first part) followed by an exponential tail."""
    w = max(1, int(round(plateau_frac * n)))
    tail_len = n - w
    raw = np.empty(n, dtype=float)
    raw[:w] = 1.0
    if tail_len > 0:
        raw[w:] = np.exp(-rate * np.arange(tail_len, dtype=float))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _mixed_exp_power_sequence(
    n: int, p: float, rate: float, w: float
) -> np.ndarray:
    """Mixture w·pow + (1‑w)·exp, both normalised before the mixture."""
    pow_seq = _powerlaw_sequence(n, p, shift=0.0)
    exp_seq = _exponential_sequence(n, rate, shift=0.0)
    mix = w * pow_seq + (1.0 - w) * exp_seq
    return _project_to_simplex(mix, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (non‑smoothed)
# ----------------------------------------------------------------------
def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on max(a*a)."""
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    it = 0
    while time.time() < deadline:
        b = np.convolve(a, a)
        max_b = b.max()
        # active set (mask) – entries attaining the maximal value
        eps = max_b * 1e-12
        p = (b >= max_b - eps).astype(float)

        # gradient = 2 * (a_rev ⋆ p) restricted to length n
        conv_len = 3 * n - 2                     # n + (2n‑1) – 1
        size = 1 << (conv_len.bit_length())
        a_rev = a[::-1]
        Ar = np.fft.rfft(a_rev, n=size)
        Pr = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar * Pr, n=size)[:conv_len]
        g = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Diminishing step size (simple 1/sqrt schedule)
        norm = np.linalg.norm(g)
        if norm == 0.0:
            break
        lr = 0.6 * target / (norm * np.sqrt(it + 1))
        a -= lr * g

        # projection back onto the simplex
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

        it += 1

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improves ordering of a fixed multiset via stochastic swaps."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # start temperature at a few percent of the current score
    T = max(best_score * 0.02, 1e-9)
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]                # propose swap

        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny random shuffle to avoid stagnation
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if mask.any():
                shuffled = a[mask].copy()
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur = _max_convolution_fft(a)
                if cur < best_score:
                    best_score = cur
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the target
    ``evaluate_sequence`` objective.  The routine respects a wall‑clock
    budget, explores many parametric families, refines promising candidates
    with exact sub‑gradient descent and a permutation‑annealing phase,
    and finally returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # a thin safety margin

    best_seq = None
    best_obj = np.inf
    elite = []       # sorted list of (obj, array) – smallest obj first

    # ------------------------------------------------------------------
    # Helper to keep track of the global best and elite list
    # ------------------------------------------------------------------
    def _consider(seq: np.ndarray):
        nonlocal best_seq, best_obj, elite
        obj = _max_convolution_fft(seq)          # = evaluation after normalisation
        if obj < best_obj:
            best_obj = obj
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New global best obj={best_obj:.6f} len={seq.size}")
        # keep up to 12 elites
        if len(elite) < 12 or obj < elite[-1][0]:
            elite.append((obj, seq.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > 12:
                elite.pop()

    # ------------------------------------------------------------------
    # 0. Warm start from the previously discovered construction (if any)
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        n0 = init.size
        init = _project_to_simplex(init, np.sqrt(2.0 * n0))
        _consider(init)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Broad parametric random search (≈70 % of the time budget)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 8000, 10000, 12000]
    families = [
        "exp",
        "power",
        "trunc_power",
        "logistic",
        "plateau_exp",
        "mixed",
    ]

    gen_deadline = time.time() + 0.70 * (deadline - time.time())

    while time.time() < gen_deadline:
        n = int(rng.choice(lengths))
        target_sum = np.sqrt(2.0 * n)

        family = rng.choice(families)
        if family == "exp":
            rate = rng.uniform(5e-5, 0.02)          # log‑uniform preference
            shift = rng.uniform(0.0, 5.0)
            cand = _exponential_sequence(n, rate, shift)

        elif family == "power":
            p = rng.uniform(0.05, 2.0)              # extended range
            shift = rng.uniform(0.0, 10.0)
            cand = _powerlaw_sequence(n, p, shift)

        elif family == "trunc_power":
            p = rng.uniform(0.2, 2.0)
            shift = rng.uniform(0.0, 5.0)
            cutoff = rng.uniform(0.2, 0.9)          # fraction of n
            cand = _truncated_powerlaw_sequence(n, p, shift, cutoff)

        elif family == "logistic":
            steep = rng.uniform(0.001, 0.12)
            midpoint = rng.uniform(0.3, 0.7) * n
            cand = _logistic_sequence(n, steep, midpoint)

        elif family == "plateau_exp":
            plateau_frac = rng.uniform(0.2, 0.9)
            rate = rng.uniform(0.001, 0.02)
            cand = _plateau_exponential_sequence(n, plateau_frac, rate)

        else:  # mixed exponential / power‑law
            w = rng.uniform(0.15, 0.85)
            p = rng.uniform(0.2, 1.5)
            rate = rng.uniform(5e-5, 0.02)
            cand = _mixed_exp_power_sequence(n, p, rate, w)

        # try a few deterministic orderings (most families are already monotone)
        _consider(cand)                                   # as‑generated
        _consider(cand[::-1])                             # reversed order
        # random shuffle as a cheap perturbation
        shuf = cand.copy()
        rng.shuffle(shuf)
        _consider(shuf)

        # occasional cheap permutation annealing on the current global best
        if best_seq is not None and rng.random() < 0.07:
            tmp = _permutation_annealing(best_seq, time.time() + 0.25)
            _consider(tmp)

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (sub‑gradient + permutation SA)
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:

        n = seed_seq.size
        # time budget for this candidate – at most 20 s or what remains

        refined = _subgradient_descent_max(seed_seq, sub_dead)

        # finish with permutation annealing (up to the global deadline)
        refined = _permutation_annealing(refined, deadline)

        _consider(refined)

    # ------------------------------------------------------------------
    # 3. Final global permutation annealing (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _permutation_annealing(best_seq, deadline)

    # ------------------------------------------------------------------
    # 4. Safety fall‑back (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim trailing near‑zero entries – purely cosmetic
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:

    # Optional final verification (the returned list will be evaluated by the
    # external harness, so we only print for our own information).
    try:

        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
