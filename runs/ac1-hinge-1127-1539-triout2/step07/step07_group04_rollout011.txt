"""
Propose a step‑function minimizing ``evaluate_sequence``.

Main ideas
-----------
* **Multi‑stage length schedule** – start with a modest size and
  gradually increase up to 40 000.  The best vector found for the
  previous size is up‑sampled (linear interpolation) and reused.
* **Projected gradient descent** – minimise a smooth log‑sum‑exp
  surrogate of the worst convolution entry.  The inverse temperature β
  is annealed up to 5·10⁵, a momentum term speeds up convergence and
  occasional Gaussian jitter helps to escape plateaus.
* **Deterministic mass‑transfer hill‑climbing** – repeatedly pick the
  index that contributes most to the current worst convolution entry
  and move a fraction of its mass to the currently smallest entry.
  The full convolution is kept up‑to‑date in O(n) per move, a
  simulated‑annealing acceptance rule permits occasional uphill steps.
* **Time‑aware budget split** – for each length the remaining wall‑clock
  time is split dynamically (≈15 % GD, ≈55 % hill‑climb, ≈20 % polishing).
  The final remaining budget is spent on a high‑β gradient sweep.

All vectors are projected onto the ℓ¹‑simplex
  {aᵢ≥0, Σaᵢ = √(2 n)} after each move, so the official evaluator sees a
valid step function.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p), p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 5e5
    beta_factor = 1.2

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

# Random initialisers (various families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine half‑wave
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Deterministic mass‑transfer hill climbing
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing.
    At each step we move mass from the index that contributes most to the
    current worst convolution entry towards the smallest entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.08 + 1e-9          # start temperature (small)
    cooling = 0.9995

    while time.time() < deadline:
        # ---- current worst entry ------------------------------------------------
        M = int(np.argmax(b))

        # ---- source index: biggest contribution to entry M ----------------------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx_range = np.arange(i_low, i_high + 1)
        jdx = M - idx_range
        contrib = a[idx_range] * a[jdx]
        if contrib.size == 0:
            break
        i_src = idx_range[int(np.argmax(contrib))]

        if a[i_src] < 1e-14:
            # nothing to move – break to avoid endless loop
            break

        # ---- destination: smallest entry (avoid picking src) --------------------
        j_dst = int(np.argmin(a))
        if j_dst == i_src:
            # pick second smallest
            order = np.argsort(a)
            for cand in order:
                if cand != i_src:
                    j_dst = cand
                    break

        # ---- amount of mass to transfer ----------------------------------------
        max_delta = a[i_src] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- update convolution (linear part) -----------------------------------
        b_new = b.copy()
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[j_dst:j_dst + n] += 2.0 * delta * a

        # ---- quadratic correction terms -----------------------------------------
        k2i = 2 * i_src
        k2j = 2 * j_dst
        kij = i_src + j_dst
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---- Metropolis acceptance -----------------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i_src] -= delta
            a[j_dst] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b
