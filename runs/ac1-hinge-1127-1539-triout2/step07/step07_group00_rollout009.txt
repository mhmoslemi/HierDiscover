def _gradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0*n)
    beta = 2.0
    beta_max = 1e5
    beta_factor = 1.5

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        lr = 0.1 * target_sum / (np.sqrt(v_hat) + eps).mean()  # heuristic
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)
        # jitter
        if it % 200 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum*0.001/(beta+1), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        # beta annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
        it += 1
    return a

def _hill_climbing(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_score = b.max()
    best_a = a.copy()
    rng = np.random.default_rng()
    T = best_score * 0.1 + 1e-9
    cooling = 0.9998
    eps = 1e-12

    while time.time() < deadline:
        M = np.argmax(b)  # index of max
        # source indices i that contribute to M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # Destination selection: aim to minimize a[M - j]
        # compute counterpart masses
        # for each j, counterpart = a[M - j] if in range else 0
        counterpart = np.zeros(n)
        low_j = max(0, M - (n - 1))
        high_j = min(n - 1, M)
        j_range = np.arange(low_j, high_j + 1)
        counterpart[j_range] = a[M - j_range]
        # compute weight inversely proportional to a[j] and counterpart
        weight = 1.0 / (a + eps)
        weight[j_range] *= 1.0 / (counterpart[j_range] + eps)
        # optionally bias to low a
        weight = weight / weight.sum()
        j = rng.choice(n, p=weight)

        if i == j:
            continue

        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # Update b in place and keep copy for revert if needed
        b_new = b.copy()
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[j:j+n] += 2.0 * delta * a
        k2i = 2*i
        k2j = 2*j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()
        # acceptance
        if new_score < best_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - best_score) / T)
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            best_score = new_score
            if new_score < best_score - 1e-12:
                best_a = a.copy()
                # print optional
                # print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")
        T *= cooling
    return best_a, b

"""
Propose a candidate step‑function that (approximately) minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1.  Use an existing construction (if any) as a seed.
2.  Run a length schedule – each length is optimised independently.
    For a given length:
       a) initialise by up‑sampling the current best or by a random simplex draw;
       b) perform projected gradient descent on a smooth approximation of the
          maximal self‑convolution entry (soft‑max); β (smoothness) is
          annealed to a very large value;
       c) improve the result with a simulated‑annealing hill‑climber that
          updates the full convolution in O(n) per move.  The destination
          of a mass transfer is chosen by a heuristic that directly targets
          the current worst convolution position.
    The GD and hill‑climbing phases are iterated while time permits.
3.  A final polishing stage runs a long GD pass followed by a short
    hill‑climb on the remaining budget.
4.  Trailing near‑zeros are trimmed and the sequence is returned.

All updates respect the ℓ¹‑simplex Σa = √(2 n) (the normalisation that makes
the objective equal to the maximal entry of the self‑convolution).
The routine respects the 1000 s wall‑clock budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a * a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities (log‑sum‑exp trick)
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Projected gradient descent on the smooth maximum.
    Uses momentum, β‑annealing and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # hyper‑parameters
    beta = 2.0
    beta_max = 1e6
    beta_factor = 1.5

    step_base = target_sum * 0.2
    rho = 0.90
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing (make the soft‑max sharper)
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a

# Random initialisers

def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)),
             np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length,
    followed by projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Hill‑climbing with exact convolution updates (improved destination heuristic)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    The destination is chosen to directly diminish the current maximal
    convolution entry.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9998                    # slower cooling for finer search
    eps = 1e-12

    while time.time() < deadline:
        M = np.argmax(b)                # index of the current worst entry

        # --- pick source index i (weighted by its contribution to M) ---
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # --- destination j: aim at positions that have a small counterpart
        #     a[M‑j] (or none) so that moving mass to j does not increase
        #     the maximal entry.
        counterpart = np.zeros(n)
        low_j = max(0, M - (n - 1))
        high_j = min(n - 1, M)
        j_range = np.arange(low_j, high_j + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # weight inversely by current mass and by counterpart mass
        weight = 1.0 / (a + eps)
        weight[j_range] *= 1.0 / (counterpart[j_range] + eps)

        # normalise and sample
        weight_sum = weight.sum()
        if weight_sum <= 0.0:
            j = rng.integers(0, n)
        else:
            probs_j = weight / weight_sum
            j = rng.choice(n, p=probs_j)

        if i == j:
            continue

        # --- amount of mass to move (up to 80 % of source) ---
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # --- update convolution (vectorised O(n) slices) ---
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b
