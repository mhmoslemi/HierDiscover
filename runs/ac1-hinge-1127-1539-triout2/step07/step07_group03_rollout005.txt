"""
Improved search for a non‑negative step sequence that minimizes
    2·n·max(a∗a) / (∑a)²
The algorithm consists of three stages:

1.  Multiple cheap initialisations (power‑law shapes with a few
    exponents and their reversed versions).
2.  A long Adam optimiser on a soft‑max approximation of the max‑convolution
    followed by a sub‑gradient descent on the exact max entry.
    Both stages keep the ℓ¹‑norm fixed by projecting onto the simplex.
3.  If time permits a simulated‑annealing refinement and a final short
    Adam polishing pass.

The optimiser works directly on the normalised distribution
    b_i = a_i / √(2·n)  (so that ∑b_i = 1),
and the objective reduces to `max(b∗b)`.  The code keeps the sequence
positive, projects onto the simplex after every update and logs progress.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value: 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^{-p} of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient helpers (soft‑max and exact max)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    b = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (rev(a) ⋆ p)
    a_rev = a[::-1]
    Ar_fft = rfft(a_rev, n=size)
    Pr_fft = rfft(p, n=size)
    conv_rev = irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_max_exact(a: np.ndarray):
    """
    Sub‑gradient of the exact max entry of a*a.
    Returns (grad, conv, k_max) where grad has same shape as a.
    """
    n = a.size
    conv = _conv_fft(a)
    k_max = int(np.argmax(conv))
    # gradient: 2 * a[k_max - i] for all i where the index is valid
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad[idx] = 2.0 * a[k_max - idx]
    return grad, conv, k_max


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_softmax_optimize(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max‑convolution.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    beta = 1.0
    max_beta = 1e6
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (decays slowly)
        lr = target_sum * 0.2 / (np.sqrt(t) + 1.0)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasionally increase the soft‑max temperature
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        # small jitter to escape plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the exact max entry.
    The step size decays polynomially.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, conv, k_max = _gradient_max_exact(a)

        # step size schedule – a small power‑law decay
        step = target_sum * 0.3 / ((it + 1) ** 0.6)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake out of shallow minima
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SGD it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


def _optimize_sequence(init_seq: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Combined Adam + sub‑gradient optimisation.
    60 % of the remaining budget → Adam (soft‑max),
    the rest → sub‑gradient on the true max.
    Returns the best sequence found and its score.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 5.0:
        # not enough time – just return the initial candidate
        return init_seq, _score(init_seq)

    adam_deadline = now + remaining * 0.6
    a_adam, val_adam = _adam_softmax_optimize(init_seq, target_sum, adam_deadline)

    # sub‑gradient phase on whatever time is left
    if time.time() < deadline:
        a_sub, val_sub = _subgradient_optimize(a_adam, target_sum, deadline)
        if val_sub < val_adam:
            return a_sub, val_sub
    return a_adam, val_adam


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (kept from the previous state‑of‑the‑art)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 10.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.5,
                         max_frac_end: float = 0.01,
                         max_window: int = 100) -> tuple[np.ndarray, float]:
    """
    Classic SA with three move types:
      • mass transfer (biased by current sensitivities),
      • local smoothing,
      • random swap.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # location of current maximum convolution entry
        k_max = int(np.argmax(conv))

        # sensitivities for a (derivative of that entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # ------------------- mass transfer -------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            donor = rng.choice(n, p=weight / total_w) if total_w > 0 else rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                a[start:end] = old_seg.mean()
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # new convolution & score
        fa = rfft(a, n=size_fft)
        conv_new = irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis decision
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low benchmark score.
    The algorithm explores several power‑law initialisations, runs an Adam optimiser
    on a soft‑max approximation, refines with exact sub‑gradient descent, and finally
    applies a simulated‑annealing polish.  The best sequence found before the
    deadline is returned.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # problem size – larger sizes give a richer search space
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1.  try a handful of power‑law seeds (including mirrored versions)
    # ------------------------------------------------------------------
    exponents = [0.45, 0.50, 0.55, 0.60, 0.65]
    best_seq = None
    best_val = np.inf

    for p in exponents:
        # forward power law
        init = _power_law_sequence(n, p, target_sum)
        seq, val = _optimize_sequence(init, target_sum, deadline)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New best from p={p:.2f}: {best_val:.6f}")

        # mirrored version (large mass at the other end)
        init_rev = init[::-1]
        seq_rev, val_rev = _optimize_sequence(init_rev, target_sum, deadline)
        if val_rev < best_val:
            best_val = val_rev
            best_seq = seq_rev.copy()
            print(f"[{time.time():.2f}] New best from reversed p={p:.2f}: {best_val:.6f}")

        # stop early if we are about to run out of time
        if time.time() > deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # 2.  simulated‑annealing refinement (uses remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, deadline,
                                            start_temp=5.0,
                                            end_temp=1e-6,
                                            max_frac_start=0.4,
                                            max_frac_end=0.01,
                                            max_window=80)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa
            print(f"[{time.time():.2f}] SA improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  final short Adam polish if any time remains
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() + 1.0 < deadline:
        a_final, val_final = _adam_softmax_optimize(best_seq,
                                                    target_sum,
                                                    deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[{time.time():.2f}] Final Adam polish improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  tidy up output
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final official evaluation (will be ignored if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
