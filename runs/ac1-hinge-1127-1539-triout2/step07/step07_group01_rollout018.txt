def _init_central_triangle(n, rng):
    idx = np.arange(n)
    center = (n - 1) / 2
    max_h = rng.uniform(0.5, 5.0)
    # linear taper from center to edges: height = max_h * (1 - abs(idx-center)/ (center+1e-9))
    tri = np.maximum(0, max_h * (1 - np.abs(idx-center) / (center+1e-9)))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n) * rng.uniform(0.0, 0.2)
    raw = mix * tri + (1 - mix) * base
    return raw

def _init_gaussian(n, rng):
    idx = np.arange(n)
    center = (n-1)/2
    sigma = rng.uniform(0.1, n/3)
    amp = rng.uniform(0.5, 5.0)
    raw = amp * np.exp(- ((idx-center)**2) / (2*sigma**2))
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n) * rng.uniform(0.0, 0.2)
    return mix * raw + (1-mix) * base

def _init_cosine(n,rng):
    idx=np.arange(n)
    # produce half cosine shape peaked in middle
    raw = 0.5*(1+np.cos(np.pi*(idx - (n-1)/2)/(n-1))) # range 0 to 1
    # scale amplitude
    amp = rng.uniform(0.5,5.0)
    raw = raw*amp
    mix = rng.uniform(0.0,1.0)
    base = np.ones(n) * rng.uniform(0.0, 0.2)
    return mix*raw + (1-mix)*base

def propose_candidate(seed=42,budget_s=1000,**kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5

    best_seq = None
    best_score = np.inf

    # deterministic grid search for monotonic powerlaw shapes
    lengths = [2000, 4000, 8000, 12000, 16000, 20000]
    for n in lengths:
        S = np.sqrt(2.0 * n)
        # power exponents
        for p in range(1, 11):
            idx = np.arange(n)[::-1] + 1  # n, n-1,...,1
            raw = idx.astype(np.float64) ** p
            a = raw * (S / raw.sum())
            peak = _max_convolution(a)
            score = 2.0 * n * peak / (a.sum()**2)  # = peak
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[grid] n={n} p={p} score={score:.6f}")

        # exponential family
        for lam in np.logspace(-4, -1, num=6):
            raw = np.exp(-lam * np.arange(n))
            a = raw * (S / raw.sum())
            peak = _max_convolution(a)
            score = 2.0 * n * peak / (a.sum()**2)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[grid-exp] n={n} lam={lam:.6g} score={score:.6f}")

    # Random exploration for remaining time if any
    while time.time() < end_time:
        n = int(rng.integers(2500, 20000))
        style = rng.choice(['monotonic_powerlaw', 'monotonic_exponential', 'powerlaw', 'exponential', 'linear', 'gamma','uniform'])
        if style == 'monotonic_powerlaw':
            p = rng.uniform(0.5, 10.0)
            idx = np.arange(n)[::-1] + 1
            raw = idx.astype(np.float64) ** p
        elif style == 'monotonic_exponential':
            lam = rng.uniform(1e-4, 2e-2)
            raw = np.exp(-lam * np.arange(n))
        elif style == 'powerlaw':
            raw = _init_powerlaw(n, rng)
        elif style == 'exponential':
            raw = _init_exponential(n, rng)
        elif style == 'linear':
            raw = _init_linear(n, rng)
        elif style == 'gamma':
            raw = _init_gamma(n, rng)
        else:
            raw = _init_uniform(n, rng)

        raw = np.clip(raw, 0.0, None)
        S = np.sqrt(2.0 * n)
        total = raw.sum()
        if total == 0:
            a = np.full(n, S / n)
        else:
            a = raw * (S / total)
        peak = _max_convolution(a)
        score = 2.0 * n * peak / (a.sum()**2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[rand] n={n} style={style} score={score:.6f}")

    # Optionally refine best using Adam for a short remaining time
    if best_seq is not None and time.time() < end_time - 10:
        # allocate a few seconds
        refined_seq, refined_peak = _adam_optimize(best_seq,
                                                   time.time() + min(60.0, end_time - time.time() - 5),
                                                   rng)
        refined_score = 2.0 * len(refined_seq) * refined_peak / (refined_seq.sum()**2)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[refine] improved score={best_score:.6f}")

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal score = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq

"""
A fast optimizer for the step‑function attack.

The algorithm consists of three lightweight phases that fit easily
into the 1000 s budget.

1. **Deterministic grid** – try monotone decreasing power‑law
   shapes a_i ∝ (n‑i)^p for several lengths (2 k … 20 k) and several
   integer exponents p∈[1,10].  The shape is scaled so that
   Σa_i = √(2 n), then the peak of the self‑convolution is measured
   with an FFT.  The best candidate is kept.

2. **Exponential family** – the same grid is repeated for
   exponential decays a_i ∝ exp(‑λ i) with λ sampled on a log‑scale.

3. **Random exploration & optional Adam refinement** – while time
   permits a few random candidates from the previously used families
   are generated; the current best is optionally polished with a few
   seconds of Adam optimisation on the simplex (log‑softmax
   parametrisation) using the exact max‑convolution gradient.

The best sequence is stored in the global variable ``height_sequence_1``
and returned as a plain ``list[float]``.  The internal score equals the
quantity that ``evaluate_sequence`` returns when the ℓ₁‑norm is fixed
to √(2 n), so the external score is printed for sanity.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Gradient utilities – Adam on the simplex (log‑softmax)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ≈0).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4

    # start with a modest amount of smoothing
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (soft max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9995                # shrink smoothing quickly
        lr *= 0.9998                 # very slow decay

        # occasional jitter to escape plateaus
        if it % 600 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parametric families for random initialisation (kept for the
# optional random phase – they are the same as in the reference code)
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n) * rng.uniform(0.0, 0.2)
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the time budget and stores the final construction in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – deterministic grid (monotone power‑law & exponential)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000, 20000]
    for n in lengths:
        S = np.sqrt(2.0 * n)

        # ---- monotone power‑law a_i ∝ (n‑i)^p ----
        for p in range(1, 11):               # integer exponents 1 … 10
            idx = np.arange(n)[::-1] + 1      # n, n‑1, … , 1
            raw = idx.astype(np.float64) ** p
            a = raw * (S / raw.sum())
            peak = _max_convolution(a)
            score = 2.0 * n * peak / (a.sum() ** 2)   # = peak because sum = S
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[grid] n={n:5d} p={p:2d}  score={score:.6f}")

        # ---- monotone exponential a_i ∝ exp(-λ i) ----
        for lam in np.logspace(-4, -1, num=6):   # λ from 1e‑4 to 1e‑1
            raw = np.exp(-lam * np.arange(n))
            a = raw * (S / raw.sum())
            peak = _max_convolution(a)
            score = 2.0 * n * peak / (a.sum() ** 2)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[grid‑exp] n={n:5d} λ={lam:.2e}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random exploration (uses all families, incl. monotone)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 150.0   # keep ~150 s for final refinement
    while time.time() < explore_deadline:
        n = int(rng.integers(2500, 20000))
        style = rng.choice(['monotonic_powerlaw',
                            'monotonic_exponential',
                            'powerlaw',
                            'exponential',
                            'linear',
                            'gamma',
                            'uniform',
                            'dirichlet'])
        if style == 'monotonic_powerlaw':
            p = rng.uniform(0.5, 12.0)
            idx = np.arange(n)[::-1] + 1
            raw = idx.astype(np.float64) ** p
        elif style == 'monotonic_exponential':
            lam = rng.uniform(1e-4, 2e-2)
            raw = np.exp(-lam * np.arange(n))
        elif style == 'powerlaw':
            raw = _init_powerlaw(n, rng)
        elif style == 'exponential':
            raw = _init_exponential(n, rng)
        elif style == 'linear':
            raw = _init_linear(n, rng)
        elif style == 'gamma':
            raw = _init_gamma(n, rng)
        elif style == 'dirichlet':
            raw = _init_dirichlet(n, rng)
        else:  # uniform
            raw = _init_uniform(n, rng)

        raw = np.clip(raw, 0.0, None)
        S = np.sqrt(2.0 * n)
        total = raw.sum()
        if total == 0.0:
            a = np.full(n, S / n)
        else:
            a = raw * (S / total)

        peak = _max_convolution(a)
        score = 2.0 * n * peak / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[rand] n={n:5d} style={style:20s}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – brief Adam refinement of the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 30:
        # give the optimizer at most 60 s (or less if remaining time is smaller)
        remaining = end_time - time.time()
        opt_time = min(60.0, remaining)
        refined_seq, refined_peak = _adam_optimize(best_seq,
                                                   time.time() + opt_time,
                                                   rng)
        refined_score = 2.0 * len(refined_seq) * refined_peak / (refined_seq.sum() ** 2)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[refine] Adam improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal best = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
