"""
Local-search optimizer for the step‑function benchmark.

The algorithm keeps the ℓ¹‑norm fixed (∑a_i = √(2·n)) and repeatedly
moves a small amount of mass from the part of the sequence that
contributes most to the current maximal autocorrelation entry to a
position that contributes little.  The convolution is updated exactly
by a cheap O(n) slice operation, so millions of moves can be tried
within the time budget.  A modest simulated‑annealing acceptance rule
allows occasional uphill moves, and a full FFT recomputation is done
periodically to avoid numerical drift.  The final sequence is trimmed
to the last non‑zero entry and returned.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return a (i+1)^(-p) sequence of length n, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _move_mass(a: np.ndarray,
               conv: np.ndarray,
               donor: int,
               receiver: int,
               delta: float,
               a_before: np.ndarray) -> None:
    """
    Apply an exact update of the convolution when `delta` mass moves
    from ``donor`` to ``receiver``.  ``a_before`` must be the array *before*
    the move (a copy of ``a``).  The function updates ``conv`` in‑place
    and mutates ``a`` accordingly.
    """
    n = a.size
    # Update the convolution using the slice representation.
    conv[donor:donor + n] -= 2.0 * delta * a_before
    conv[receiver:receiver + n] += 2.0 * delta * a_before

    # Quadratic correction for the two self‑terms and the cross‑term.
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # Apply the mass transfer to the sequence itself.
    a[donor] -= delta
    a[receiver] += delta

    entries that heavily participate in the current maximal autocorrelation
    to entries that contribute little.  Convolution updates are performed
    exactly in O(n) using slice arithmetic, which permits millions of
    trial moves inside the time limit.
    """

    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # Choose a problem size that fits the time budget.
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    a = _project_to_simplex(a, target_sum)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    conv = _conv_fft(a)
    cur_M = float(conv.max())
    best_a = a.copy()
    best_M = cur_M

    rng = np.random.default_rng()
    it = 0
    total_time = deadline - start_time

    # Temperature schedule (exponential decay).
    start_temp = 0.05
    end_temp = 1e-6

    # Main optimisation loop.
    while time.time() < deadline:
        # ----------------------------------------------------------
        # 1. Identify the index where the autocorrelation attains its
        #    current maximum and the contribution of each entry.
        # ----------------------------------------------------------
        cur_M = float(conv.max())
        k_max = int(conv.argmax())

        idx = k_max - np.arange(n)                # k_max - i
        valid = (idx >= 0) & (idx < n)
        contrib = np.zeros(n, dtype=a.dtype)
        contrib[valid] = a[valid] * a[idx[valid]]

        donor = int(np.argmax(contrib))

        # Choose a receiver among the least contributing indices.
        low_thr = np.percentile(contrib, 15)
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size == 0:
            receiver = int(rng.integers(n))
        else:
            receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            # Nothing to move – skip this iteration.
            it += 1
            continue

        # ----------------------------------------------------------
        # 2. Pick a transfer amount.
        # ----------------------------------------------------------
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        # Fraction allowed: linearly decay from 0.5 to 0.01.
        max_frac = 0.5 * (1.0 - tau) + 0.01
        delta = rng.random() * a[donor] * max_frac
        if delta <= 0.0:
            it += 1
            continue

        # ----------------------------------------------------------
        # 3. Perform an exact update of conv and a (store a copy for rollback).
        # ----------------------------------------------------------
        a_before = a.copy()
        _move_mass(a, conv, donor, receiver, delta, a_before)

        new_M = float(conv.max())

        # ----------------------------------------------------------
        # 4. Acceptance rule (simulated annealing).
        # ----------------------------------------------------------
        accept = False
        if new_M <= cur_M:
            accept = True
        else:
            # temperature decays exponentially
            temp = start_temp * (end_temp / start_temp) ** tau
            prob = np.exp(-(new_M - cur_M) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_M = new_M
            if new_M < best_M:
                best_M = new_M
                best_a = a.copy()
                print(f"[{time.time():.2f}] it={it:7d} new best max={best_M:.6f}")
        else:
            # revert the move (use the same a_before to undo the slice update)
            # Undo convolution update.
            conv[donor:donor + n] += 2.0 * delta * a_before
            conv[receiver:receiver + n] -= 2.0 * delta * a_before
            conv[2 * donor] -= delta * delta
            conv[2 * receiver] -= delta * delta
            conv[donor + receiver] += 2.0 * delta * delta
            # Restore a.
            a = a_before

        it += 1

        # ----------------------------------------------------------
        # 5. Occasionally recompute convolution from scratch to avoid drift.
        # ----------------------------------------------------------
        if it % 20000 == 0:
            conv = _conv_fft(a)

    # ------------------------------------------------------------------
    # Final clean‑up: project onto the simplex one last time and trim.
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, target_sum)
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]

    final_seq = best_a.tolist()

    # ------------------------------------------------------------------
    # Optional reporting using the supplied benchmark.
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq

            conv[donor:donor + n] += 2.0 * delta * a_before
            conv[receiver:receiver + n] -= 2.0 * delta * a_before
            conv[2 * donor] -= delta * delta
            conv[2 * receiver] -= delta * delta
            conv[donor + receiver] += 2.0 * delta * delta
            a = a_before

        it += 1

        # ----------------------------------------------------------
        # 5. Periodically recompute the full convolution to avoid drift.
        # ----------------------------------------------------------
        if it % 20000 == 0:
            conv = _conv_fft(a)

    # ------------------------------------------------------------------
    # Final clean‑up: project onto the simplex one last time and trim.
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, target_sum)
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]

    final_seq = best_a.tolist()

    # ------------------------------------------------------------------
    # Optional reporting using the provided benchmark function.
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
