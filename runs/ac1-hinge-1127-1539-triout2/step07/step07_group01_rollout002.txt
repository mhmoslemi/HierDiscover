"""
Hybrid random‑search + Adam refinement for minimizing the peak of a
self‑convolution under an ℓ₁‑normalisation constraint.

The algorithm consists of three phases:

1. **Random exploration** – many candidate vectors are drawn from a
   diverse set of parametric families (power‑law, exponential, linear,
   gamma, uniform, Dirichlet, and a new Beta‑distribution family).  Each
   candidate is normalised so that Σa = √(2 n) and its max‑convolution
   peak is measured with an FFT.  The best candidate seen so far is kept.

2. **Scaling & long‑run Adam optimisation** – the current best vector is
   interpolated to several larger lengths (up to ≈ 20000 entries) and
   each scaled vector is refined with an Adam optimiser that works on
   the probability simplex via a log‑softmax parametrisation.  The
   optimiser minimises a τ‑soft‑max approximation of the peak; τ is
   decayed aggressively (≈ 0.998 per iteration) so that the optimiser
   concentrates on the true max near the end.

3. **Final evaluation** – the best sequence is evaluated with the
   provided ``evaluate_sequence`` function, stored in the global variable
   ``height_sequence_1`` and returned.

The routine respects a 1000 s budget and prints progress information.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence to length ``new_n`` by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# τ‑soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Approximate gradient of max(b) where b = a*a using a τ‑soft‑max.
    Returns (grad, max_b) where grad = ∂/∂a τ‑soft‑max(b).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   tau_decay: float = 0.998) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the exact max‑convolution (τ‑soft‑max).
    Returns the best sequence encountered and its hard (non‑smoothed) peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # starting temperature (more smoothing → smoother landscape)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax) and enforce ℓ₁‑norm S
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of the τ‑soft‑max and the hard peak value
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule: ∂L/∂p = a * (g - ⟨a,g⟩/S)
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional gradient clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau = max(tau * tau_decay, 1e-12)
        lr *= 0.9999  # very slow decay

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    raw = mix * edge + (1.0 - mix) * np.ones(n)
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    raw = mix * exp_part + (1.0 - mix) * np.ones(n)
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n) * rng.uniform(0.0, 0.2)
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Beta‑distribution family on [0,1] (U‑shaped for α,β<1, flat for =1,
    centre‑heavy for >1).  Optionally mixed with a uniform background.
    """
    alpha = rng.uniform(0.1, 5.0)
    beta = rng.uniform(0.1, 5.0)
    mix = rng.uniform(0.0, 1.0)
    x = (np.arange(n) + 0.5) / n
    beta_shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    raw = mix * beta_shape + (1.0 - mix) * np.ones(n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best and otherwise samples
    from a random parametric family (including the new Beta family).
    """
    S = np.sqrt(2.0 * n)

    # warm‑start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'beta'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:   # beta
        raw = _init_beta(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

        slice_time = max(30.0, remaining_time - min_needed)
        slice_time = min(slice_time, 250.0)  # cap to keep things responsive

        target_n = pending.pop(0)
        # Resize current best to the target length
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # Run Adam optimisation on the scaled vector
        deadline = time.time() + slice_time
        a_opt, peak = _adam_optimize(a_scaled, deadline, rng,
                                     lr0=0.5, tau_decay=0.998)

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
