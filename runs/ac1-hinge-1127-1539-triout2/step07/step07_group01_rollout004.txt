def build_convolution_matrix(f):
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # for j from i_min to i_max:
        # coefficient for g_j is f_{k-j}
        # so set A[k, i_min:i_max+1] = f[k - np.arange(i_min,i_max+1)]
        idx = np.arange(i_min, i_max+1)
        A[k, idx] = f[k - idx]
    return A

"""
Adversarial step‑function optimizer (enhanced).

The optimizer works in three phases:

1. **Random exploration** – many candidate vectors are drawn from a
   collection of parametric families (power‑law, exponential, linear,
   gamma, uniform, Dirichlet).  Each candidate is normalised so that its
   ℓ₁‑norm equals √(2 n).  The best candidates (by the exact internal
   objective 2·n·max(a*a)/‖a‖₁²) are remembered.

2. **Refinement** – for the current best candidate we apply two
   complementary refinements:
      * a long‑run Adam optimiser on the simplex (log‑softmax
        parametrisation) that directly minimises the true peak of the
        autocorrelation,
      * a linear‑programming step that, keeping the autocorrelation
        peak unchanged, maximises the ℓ₁‑mass of the vector (the
        classic “LP‑step” used in the reference algorithm).  The result
        is mixed into the current iterate and a short Adam run cleans
        the final shape.  The Adam optimiser works on a *symmetric*
        vector (the sequence is mirrored after each update), which
        strongly reduces the convolution peak.

   The refinement is performed on several increasing lengths
   (up‑sampling the current best by linear interpolation) to exploit
   the improvement that larger dimensions usually bring.

3. **Final check** – the best internal candidate is evaluated with the
   provided black‑box ``evaluate_sequence`` function, stored in the
   global variable ``height_sequence_1`` and returned.

The whole routine respects the supplied time budget (default 1000 s)
and prints a few progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the (soft) max of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂/∂a (τ‑soft‑max(b)).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Adam optimiser that minimises the exact peak of the self‑convolution.
    The iterate is always projected onto the simplex and forced to be
    symmetric (a ↔ a[::-1]).
    Returns the best vector found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # Initialise logits from a0 (already feasible)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits (real numbers)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.35                      # a bit more aggressive

    # Temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- enforce symmetry and renormalise -----
        a = 0.5 * (a + a[::-1])
        a = a * (S / a.sum())

        # ----- convolution & gradient (hard max) -----
        g, peak = _softmax_gradient(a, tau)

        # remember best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9995                # reduce smoothing
        lr *= 0.9998                 # very slow decay

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Linear‑programming step (maximise ℓ₁‑mass under a convolution bound)
# ----------------------------------------------------------------------
def _lp_optimize(f: np.ndarray, M: float, S: float, time_limit: float = 10.0):
    """
    Solve
        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f * g)_k ≤ M  for all k,
    where * is linear convolution.
    The solution is returned normalised to ℓ₁‑norm = S.
    For moderate dimensions (n ≲ 5000) cvxpy solves the problem quickly.
    Returns None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = len(f)
    # Skip the LP for very large n – it becomes slow and memory intensive.
    if n > 5000:
        return None

    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, value=f, constant=True)
    conv = cp.conv(f_param, g)           # length 2n‑1
    constraints = [conv <= M, g <= 1000.0]   # honour the 1000‑cap used by the evaluator
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is usually the fastest for LP‑type problems; fall back to SCS.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-6,
                   reltol=1e-6, feastol=1e-8, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=20000,
                       verbose=False)
        except Exception:
            return None

    if g.value is None:
        return None

    g_opt = np.array(g.value).flatten()
    sum_g = g_opt.sum()
    if sum_g <= 0.0:
        return None
    # Rescale to the required ℓ₁‑norm S (keeps the peak ≤ M after mixing)
    g_scaled = g_opt * (S / sum_g)
    g_scaled = np.clip(g_scaled, 0.0, 1000.0)
    return g_scaled


# ----------------------------------------------------------------------
# Internal score (the quantity the optimiser actually minimises)
# ----------------------------------------------------------------------
def _internal_score(a: np.ndarray) -> float:
    n = len(a)
    max_b = _max_convolution(a)
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n)


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With modest probability we warm‑start from the global best;
    otherwise we sample a random parametric family.
    The vector is projected onto the ℓ₁‑sphere of radius √(2 n).
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    # Enforce symmetry right away (helps the later optimiser)
    a = 0.5 * (a + a[::-1])
    a = a * (S / a.sum())
    return a


# ----------------------------------------------------------------------
# Combination of Adam and LP refinement for a single candidate
# ----------------------------------------------------------------------
def _refine_candidate(a_start: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a long Adam optimisation, then an LP step that maximises the ℓ₁‑mass
    while keeping the convolution peak unchanged, mix the result and finish
    with a short Adam run.  Returns the best vector found and its peak.
    """
    n = len(a_start)
    S = np.sqrt(2.0 * n)

    # --- Adam phase (majority of the allotted time) ---
    adam_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.6)
    a_adam, peak_adam = _adam_optimize(a_start, adam_deadline, rng)

    # --- LP improvement -------------------------------------------------
    M = _max_convolution(a_adam)
    g_opt = _lp_optimize(a_adam, M, S)
    if g_opt is not None:
        # small mixing with the LP solution (t≈0.2)
        t = 0.2
        a_mix = (1.0 - t) * a_adam + t * g_opt
        a_mix = np.maximum(a_mix, 0.0)
        a_mix = a_mix * (S / a_mix.sum())
        # symmetrise the mixture (keeps the search space small)
        a_mix = 0.5 * (a_mix + a_mix[::-1])
        a_mix = a_mix * (S / a_mix.sum())

        # --- short Adam fine‑tuning ------------------------------------
        short_deadline = deadline
        if short_deadline - time.time() > 5.0:
            a_final, peak_final = _adam_optimize(a_mix, short_deadline, rng)
            if peak_final < peak_adam:
                return a_final, peak_final

    return a_adam, peak_adam


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and stores
    the final construction in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 20 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + max(30.0, (end_time - start_time) * 0.2)
    while time.time() < explore_deadline:
        attempts += 1
        # choose a length – bias toward medium sizes, occasionally go larger
        if rng.random() < 0.7:
            n = int(rng.integers(1500, 5000))
        else:
            n = int(rng.integers(5000, 18000))

        a0 = _initial_vector(n, rng, warm_seq)
        score = _internal_score(a0)

        if score < best_score:
            best_score = score
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempts:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling + Adam + LP refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: simple uniform vector
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _internal_score(best_seq)

    # Decide on a cascade of target lengths (up‑sampling the current best)
    target_lengths = [best_n]
    if best_n < 8000:
        target_lengths.append(min(12000, best_n * 2))
    if best_n < 15000:
        target_lengths.append(int(best_n * 1.5))
    if best_n < 25000:
        target_lengths.append(20000)
    target_lengths = sorted(set(target_lengths))

    for target_n in target_lengths:
        if time.time() >= end_time - 30.0:
            break

        # Allocate a slice of the remaining budget for this length
        remaining = end_time - time.time()
        slice_time = min(remaining * 0.45, 250.0)

        # Scale the current best to the new length
        a_scaled = _scale_sequence(best_seq, target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        S_target = np.sqrt(2.0 * target_n)
        a_scaled = a_scaled * (S_target / a_scaled.sum())
        a_scaled = 0.5 * (a_scaled + a_scaled[::-1])
        a_scaled = a_scaled * (S_target / a_scaled.sum())

        # Run the refinement (Adam → LP → short Adam)
        refined_seq, refined_peak = _refine_candidate(a_scaled,
                                                      time.time() + slice_time,
                                                      rng)

        refined_score = _internal_score(refined_seq)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}   internal score = {best_score:.6f}   length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
