"""
Propose a step‑function minimizing ``evaluate_sequence``.

The algorithm consists of three stages per length:

* **Initialisation** – random simplex points from several families, or
  up‑sample the best known construction.
* **Projected gradient descent** – minimise a smooth log‑sum‑exp
  surrogate of the worst self‑convolution entry.  The temperature β is
  annealed quickly to ~1e5 so the surrogate becomes the true maximum.
* **Targeted hill‑climbing** – repeatedly pick the current worst shift,
  compute its exact gradient (how each coefficient contributes to that
  entry) and move a small amount of mass from a high‑gradient index to a
  low‑gradient one.  The full self‑convolution is kept up‑to‑date in O(n)
  per move and moves are accepted with a slow simulated‑annealing
  schedule.  This is more directed than the previous random‑dest
  strategy and typically reduces the maximum further.

A length schedule (2 k → 4 k → 8 k → 16 k → 32 k) is used; the budget is
split dynamically between the gradient and hill‑climbing phases.
Finally any remaining time is spent on a high‑β polishing gradient
descent.  All vectors are projected onto the ℓ₁‑simplex
{aᵢ≥0, Σaᵢ=√(2 n)} after each gradient step, guaranteeing that the
official evaluator sees a valid step function.

The routine respects the supplied wall‑clock budget (default 1000 s)
and returns the best sequence it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p), p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max (log‑sum‑exp surrogate).
    Uses momentum, annealed inverse temperature β and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # temperature schedule
    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.3

    # step‑size and momentum
    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter for escaping plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers (various families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled (power‑law, exponential, Dirichlet,
    sine, Gaussian, piecewise) to diversify the search space.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine wave (positive half‑wave)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing driven by the exact worst‑shift gradient
# ----------------------------------------------------------------------


def _hill_climb_improved(a: np.ndarray,
                         b_full: np.ndarray,
                         deadline: float):
    """
    Simulated‑annealing style hill climbing that always attacks the
    currently worst convolution entry.  The source and destination indices
    are chosen as argmax/argmin of the exact gradient of that entry,
    which yields a much more directed move than a random destination.
    """
    a = a.copy()
    b = b_full.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9           # initial temperature
    cooling = 0.9996                     # slower cooling than before

    while time.time() < deadline:
        # --- locate a worst convolution index (may be several) ---
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        s = rng.choice(max_idxs)                   # index in [0,2n-2]

        # corresponding shift distance d (negative allowed)
        d = s - (n - 1)                             # d ∈ [-(n-1), +(n-1)]

        # gradient of b_s = Σ a_i a_{i+d}
        # g_j = a_{j+d} (if in range) + a_{j-d} (if in range)
        g = np.zeros(n, dtype=float)
        if d >= 0:
            # contribution from a_{j+d}
            g[:n - d] += a[d:]
            # contribution from a_{j-d}
            g[d:] += a[:n - d]
        else:
            d2 = -d
            # now d is negative: a_{j+d} means a_{j-d2}
            g[d2:] += a[:n - d2]
            g[:n - d2] += a[d2:]

        # choose source (high gradient) and destination (low gradient)
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        if i == j:
            # fallback: random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

        if a[i] < 1e-14:
            continue

        # amount of mass to move (up to half of source)
        delta = rng.random() * a[i] * 0.5
        if delta <= 0.0:
            continue

        # --- update the full convolution exactly (O(n)) ---
        b_new = b.copy()
        # linear impact of the change at i and at j
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms (self‑pair and cross‑pair)
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # --- Metropolis acceptance ---
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_improved improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    (as a plain Python list) before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5     # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optional seeding from a previously stored construction
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # --------------------------------------------------------------
    # Length schedule – gradually increase dimension
    # --------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----------------------------------------------------------
        # Initialise a candidate for this length
        # ----------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # ----------------------------------------------------------
        # Phase A – short gradient‑descent restarts
        # ----------------------------------------------------------
        remain = deadline - time.time()
        if remain <= 6.0:
            break
        gd_budget = min(remain * 0.30, 120.0)   # at most 2 min for this stage
        gd_deadline = time.time() + gd_budget
        n_restarts = 4 if n <= 10000 else 2
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            a_start = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_gd = _gradient_descent_opt(a_start, gd_deadline)
            score = _max_convolution_fft(a_gd)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_gd.copy()

        if local_best_a is not None and local_best_score < best_score:
            best_seq = local_best_a.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] GD improvement n={n} score={local_best_score:.6f}")

        # ----------------------------------------------------------
        # Phase B – improved hill climbing
        # ----------------------------------------------------------
        remain = deadline - time.time()
        if remain <= 6.0:
            break
        hill_budget = min(remain * 0.40, 250.0)   # allocate a sizeable chunk
        hill_deadline = time.time() + hill_budget

        # make sure we work with the current length
        if best_seq.size != n:
            a_cur = _upsample_sequence(best_seq, n)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_full = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climb_improved(a_cur, b_full, hill_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_seq = a_hc.copy()
            best_score = score_hc
            print(f"[{time.time():.1f}] Hill‑climb improvement n={n} score={score_hc:.6f}")

        # ----------------------------------------------------------
        # Optional post‑HC gradient refinement
        # ----------------------------------------------------------
        remain = deadline - time.time()
        if remain > 5.0:
            refine_deadline = time.time() + min(remain * 0.20, 60.0)
            a_ref = _gradient_descent_opt(best_seq.copy(), refine_deadline)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score:
                best_seq = a_ref.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] post‑HC GD refinement n={n} score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any leftover budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        final_deadline = deadline
        a_final = _gradient_descent_opt(a, final_deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_seq = a_final.copy()
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for a tidy output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
