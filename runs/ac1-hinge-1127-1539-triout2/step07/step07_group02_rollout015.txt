"""
Propose a step‑function (non‑negative vector) with a minimal self‑convolution
peak.  The search proceeds in three stages:

1.  **Diverse random sampling** – many analytic families (uniform, geometric,
    power‑law, beta, sinusoidal, arcsine, exponential, piecewise‑linear) are
    generated for a variety of lengths (2 k‑20 k).  Each candidate is
    normalised to Σa = √(2·n) and the best by the raw convolution peak is kept.

2.  **Projected Adam optimisation** – the best candidate is refined by
    projected Adam minimising a smooth (log‑sum‑exp) approximation of the
    convolution maximum.  Symmetry is enforced at every step, the learning
    rate and soft‑max temperature follow a slow decay schedule and occasional
    jitter is added to escape flat regions.

3.  **Sub‑gradient descent + random‑swap hill‑climbing** – a true
    sub‑gradient descent on the actual maximum is run, followed by a cheap
    random‑swap local search.

The routine respects a 1000 s time budget, stores the final vector in the
global variable ``height_sequence_1`` for possible re‑use, and returns the
sequence as a plain ``list`` of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    The vector w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) helper
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation / up‑sampling helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Random families – return a non‑negative vector (not yet normalised)
# ----------------------------------------------------------------------
def _family_uniform(rng: np.random.Generator, n: int) -> np.ndarray:
    return rng.random(n)


def _family_geometric(rng: np.random.Generator, n: int) -> np.ndarray:
    rate = rng.uniform(0.80, 0.995)               # stronger decay than before
    left = rate ** np.arange(n)
    a = (left + left[::-1]) / 2.0
    return a


def _family_powerlaw(rng: np.random.Generator, n: int) -> np.ndarray:
    p = rng.uniform(-2.0, 4.0)                    # allow edge‑heavy shapes
    idx = np.arange(n, dtype=float)
    a = ((idx + 1) * (n - idx)) ** p
    return a


def _family_beta(rng: np.random.Generator, n: int) -> np.ndarray:
    p1 = rng.uniform(0.01, 10.0)
    p2 = rng.uniform(0.01, 10.0)
    x = np.linspace(0.0, 1.0, n)
    a = (x ** p1) * ((1.0 - x) ** p2)
    return a


def _family_sinusoidal(rng: np.random.Generator, n: int) -> np.ndarray:
    p = rng.uniform(0.5, 10.0)
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    return a


def _family_arcsine(rng: np.random.Generator, n: int) -> np.ndarray:
    # f(x) ∝ 1 / sqrt(x(1‑x)).  Add a tiny epsilon to avoid division by zero.
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x) + eps)
    return a


def _family_exponential(rng: np.random.Generator, n: int) -> np.ndarray:
    tau = rng.uniform(0.1 * n, 0.5 * n)
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-np.abs(idx - centre) / tau)
    return a


def _family_piecewise_linear(rng: np.random.Generator, n: int) -> np.ndarray:
    # Random ramp up to a peak then ramp down.
    peak_pos = rng.integers(1, n - 1)
    left = np.linspace(0.0, 1.0, peak_pos, endpoint=False)
    right = np.linspace(1.0, 0.0, n - peak_pos)
    a = np.concatenate([left, right])
    scale = rng.uniform(0.5, 2.0)
    return a * scale


# ----------------------------------------------------------------------
# Adam optimiser (smooth max)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the smooth‑max approximation of the peak."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.05
    alpha_mul = 1.05
    max_alpha = 500.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce exact symmetry each iteration
        a = _symmetrize(a)

        # projection onto simplex with fixed ℓ₁‑norm
        a = _project_simplex(a, S)

        # occasional jitter to escape plateaus
        if rng.random() < 0.025:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
                a = _symmetrize(a)

        # update schedules
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true peak
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the genuine max‑convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient (single active term) – valid for a convex max
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it + 1.0)
        a -= eta * g

        # enforce symmetry and simplex constraints
        a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # small random mass‑swap (simulated‑annealing flavour)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climbing
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing that swaps tiny masses between random positions."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.025 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score

        a_up = a_up / a_up.sum() * S_up
        a_up = _symmetrize(a_up)

        second_adam_deadline = now + 0.20 * (deadline - now)
        a_adam2, sc_adam2 = _optimize_adam(a_up, second_adam_deadline, rng)
        if sc_adam2 < best_overall_score:
            best_overall_score = sc_adam2
            best_overall_seq = a_adam2.copy()
            print(f"[adam‑2] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent (≈15 % of remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        subgrad_deadline = now + 0.15 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad(best_overall_seq, subgrad_deadline, rng)
        if sc_sub < best_overall_score:
            best_overall_score = sc_sub
            best_overall_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final tiny random‑swap hill‑climb (remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_loc, sc_loc = _local_swap_search(best_overall_seq, deadline, rng)
        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & reporting
    # ------------------------------------------------------------------
    n_final = best_overall_seq.shape[0]
    S_final = np.sqrt(2.0 * n_final)
    best_overall_seq = _project_simplex(best_overall_seq, S_final)  # enforce exact sum
    final_seq = best_overall_seq.tolist()

    # The external evaluation uses the same normalisation, but we call it
    # for the official score display.
    external_score = evaluate_sequence(final_seq)

    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={n_final} attempts={attempts}")

    # expose for possible reuse in a subsequent run
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
