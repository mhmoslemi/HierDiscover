def _adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0*n)

    # Adam hyperparameters
    alpha = target * 0.5   # base learning rate (maybe adjust)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 2.0
    beta_max = 65536.0
    beta_factor = 1.2

    t=0
    best_a = a.copy()
    best_score = np.inf
    while time.time() < deadline:
        # compute gradient
        g, b = _gradient_softmax(a, beta)
        # Adam updates
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = alpha / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        # projection
        a = _project_to_simplex(a, target)
        # update best
        cur_val = b.max()
        if cur_val < best_score:
            best_score = cur_val
            best_a = a.copy()
        # beta schedule
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)
    return best_a

assistantfinal**Solution Overview**

The goal is to minimise  

```
evaluate_sequence(a) = 2·n·max(a★a) / (∑a)²                (★ = linear convolution)
```

for a non‑negative vector `a`.  
`evaluate_sequence` first projects every component onto `[0,1000]` and
rescales the ℓ¹‑norm to `√(2 n)`.  Consequently the objective is simply the
maximum entry of the self‑convolution when the sum is fixed, i.e.

```
objective = max_k (a★a)_k            with   ∑ a_i = √(2 n) .
```

The algorithm below follows three well‑known ideas:

1. **Projected Adam optimisation** – minimise a smooth surrogate of the
   maximum, namely the soft‑max  
   `ϕβ(a)= (1/β)·log Σ exp(β·(a★a)_k)`.  
   Its gradient can be written in closed form and the Adam optimiser
   (adaptive step size, bias correction) is used.  After each step the
   vector is projected onto the simplex `Σ a_i = √(2 n), a_i ≥ 0`.

2. **Simulated‑annealing hill climbing** – after the Adam phase we keep the
   exact convolution and repeatedly move a small amount of mass from a
   “source” index to a “destination” index while updating the convolution
   in `O(n)` time.  A Metropolis rule with a slowly decreasing temperature
   accepts uphill moves rarely, allowing the algorithm to escape local
   minima.

3. **Multi‑scale schedule** – we start from the best sequence found so far
   (the global variable `height_sequence_1` if it exists) and up‑sample it
   to a larger length (`n = 20000`).  The large dimension gives the optimiser
   many degrees of freedom, while the up‑sampling guarantees a good initial
   shape.

The whole search respects the 1000 s wall‑clock budget: the majority of
time is spent on Adam optimisation (≈ 70 % of the remaining budget), the
rest on hill climbing and a final short polishing step.

```python
"""
Propose a candidate step‑function minimising ``evaluate_sequence``.

Algorithm
---------
1.  Choose a target length (20000).  If a previous construction
    ``height_sequence_1`` exists, up‑sample it, otherwise start from a
    random point on the ℓ¹‑simplex Σa = √(2 n).
2.  Alternating optimisation:
    *  Adam optimiser on the smooth max (soft‑max) of the self‑convolution.
       The gradient is computed via FFT, the iterate is projected onto the
       simplex after each update and β is annealed.
    *  Simulated‑annealing hill‑climbing that moves a small amount of mass
       between two indices while updating the full convolution in O(n).
3.  Repeat steps 2 until the wall‑clock budget is exhausted, then perform
    a short final polishing.
4.  Trim trailing near‑zeros and return the sequence as a plain list.

All heavy work is performed with NumPy/FFT; the routine never exceeds the
time limit and typically reaches an objective < 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    Duchi‑et‑al algorithm (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a★a (real FFT, returns length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]
    return conv


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth maximum ϕβ(a) = (1/β)·log Σ exp(β·b_k),
    where b = a★a (full convolution).

    Returns (g, b) where
        g = 2·(a ★ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # --- b = a★a (full) -----------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # --- soft‑max probabilities -----------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # shift for numeric stability
    p = exps / np.sum(exps)

    # --- gradient g = 2·(a ★ p) ----------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3·n‑2
    size2 = _next_pow2(conv_len)
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑style optimisation of the smooth maximum.
    Returns the best vector found before the deadline (projected onto the simplex).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.2                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature (β) schedule
    beta = 2.0
    beta_max = 65536.0
    beta_factor = 1.15

    t = 0
    best_a = a.copy()
    best_score = np.inf

    while time.time() < deadline:
        g, b = _gradient_softmax(a, beta)

        # Adam update (bias‑corrected)
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # keep feasibility
        a = _project_to_simplex(a, target_sum)

        cur_score = b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress message
            print(f"[{time.time():.1f}] Adam improvement: max(b)={cur_score:.6f}")

        # β‑annealing
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return best_a


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing hill climbing.
    Moves a random amount of mass from a source index to a destination index,
    updates the full convolution in O(n) and accepts moves with a Metropolis rule.
    Returns the best vector and its full convolution.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # cooling factor per move

    while time.time() < deadline:
        # pick a position attaining the current maximum
        max_idxs = np.flatnonzero(b == cur_score)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # choose a source index proportional to its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination – 30 % bias towards indices with little mass
        if rng.random() < 0.3:
            inv = 1.0 / (a + 1e-12)
            probs_j = inv / inv.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to transfer
        delta = rng.random() * a[i] * 0.8
        if delta <= 0.0:
            continue

        # --- update convolution in O(n) --------------------------------
        b_new = b.copy()
        # contribution of the moved mass with all other entries
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # squares and double‑count correction
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Random point on the ℓ¹‑simplex Σa = √(2 n).
    Several families (Dirichlet, power‑law, exponential) are mixed.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if family == 0:                # power‑law
        p = rng.uniform(0.2, 1.0)
        a = (idx + 1.0) ** (-p)
    elif family == 1:              # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)
    elif family == 2:              # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))
    elif family == 3:              # mixed power‑law × exponential
        p = rng.uniform(0.2, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    else:                          # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects a 1000 s wall‑clock budget and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # small safety margin

    # ------------------------------------------------------------------
    # Choose target length (large enough to give the optimizer many degrees
    # of freedom, still fast for FFT‑based gradients)
    # ------------------------------------------------------------------
    n_target = 20000
    target_sum = np.sqrt(2.0 * n_target)

    # ------------------------------------------------------------------
    # Initialise – use previous construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            a = _upsample_sequence(a0, n_target)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1, upsampled to n={n_target}")
        except Exception as e:
            print(f"Seeding error ({e}), falling back to random init")
            a = _random_initial_sequence(n_target, rng)
    else:
        a = _random_initial_sequence(n_target, rng)
        print(f"[{time.time():.1f}] Random initial sequence, n={n_target}")

    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_score = np.inf

    # ------------------------------------------------------------------
    # Main alternating loop: Adam optimisation -> hill climbing
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        now = time.time()
        remaining = deadline - now
        # Allocate ≈70 % of the remaining time to Adam optimisation
        grad_deadline = now + remaining * 0.70
        a_opt = _adam_optimize(a, grad_deadline)

        # Evaluate after gradient phase
        b_conv = _full_convolution(a_opt)
        cur_score = b_conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] New best after Adam: max(b)={cur_score:.6f}")

        # Allocate ≈25 % of the remaining time to hill climbing
        now = time.time()
        remaining = deadline - now
        if remaining <= 5.0:
            break
        hill_deadline = now + remaining * 0.25
        a_hc, b_hc = _hill_climbing(a_opt, b_conv, hill_deadline)
        cur_score_hc = b_hc.max()
        if cur_score_hc < best_score:
            best_score = cur_score_hc
            best_a = a_hc.copy()
            print(f"[{time.time():.1f}] New best after hill‑climb: max(b)={cur_score_hc:.6f}")

        # Continue from the current best
        a = best_a.copy()

    # ------------------------------------------------------------------
    # Final polishing (short Adam + hill climbing) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_final = _adam_optimize(best_a, deadline - 1.0)
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 0.5)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_a = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: max(b)={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for readability
    # ------------------------------------------------------------------
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]
    else:
        best_a = best_a[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_a.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_a.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_a.tolist()
```
