def _subgradient_max(a):
    n = a.size
    conv = _conv_fft(a)
    max_b = conv.max()
    idx = np.where(conv >= max_b - 1e-12)[0]  # active positions
    grad = np.zeros_like(a)
    for s in idx:
        lo = max(0, s - (n - 1))
        hi = min(s, n - 1)
        # indices i from lo to hi inclusive. as int
        # For each i, j = s-i
        # grad[i] += 2*a[j]
        grad[lo:hi+1] += 2*a[s - np.arange(lo, hi+1)]
    return grad, max_b

def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    best = a.copy()
    best_val = _score(a)
    it=0
    step0 = target_sum * 0.2   # base step size
    while time.time() < deadline:
        grad, maxb = _subgradient_max(a)  # compute gradient of max conv
        # gradient points where a increase leads to increase max. So we move opposite direction.
        # step size decaying
        lr = step0 / (np.sqrt(it+1) * np.linalg.norm(grad) + 1e-12)
        a -= lr * grad
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")
        it += 1
    return best, best_val

"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm consists of four stages:

1. **Smart initialisation** – try several power‑law exponents, keep the
   best (and a few jittered variants).
2. **Projected gradient descent with a soft‑max approximation** – many
   restarts, a rapidly increasing temperature (β) and momentum, all
   updates projected onto the simplex ∑a = √(2 n).
3. **Simulated‑annealing refinement** – moves are mass transfers,
   local smoothing and swaps.  Donor/receiver selection is biased by a
   sensitivity vector derived from the (few) largest convolution entries.
4. **Sub‑gradient descent on the true ℓ∞‑norm** – the exact gradient of
   the maximal convolution entry is computed and a diminishing step size
   is used.  A final short projected‑gradient pass cleans up the result.

All stages respect the non‑negativity and ℓ¹‑normalisation constraints.
The routine adapts the problem size (n≈5 k‑12 k) to the available time‑budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using FFT (length 2·n‑1).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^{-p} sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _initial_candidate(n: int, target_sum: float) -> np.ndarray:
    """
    Scan a few power‑law exponents, keep the best, and jitter it a few times.
    """
    rng = np.random.default_rng()
    best_seq = None
    best_val = np.inf
    for p in np.linspace(0.30, 0.90, 9):
        seq = _power_law_sequence(n, p, target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq
    # a few random perturbations of the best seed
    for _ in range(6):
        cand = best_seq + rng.normal(scale=target_sum * 0.01, size=n)
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_val:
            best_val, best_seq = val, cand
    return best_seq


# ----------------------------------------------------------------------
# Gradient descent (soft‑max) utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    b = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    grad_rev = irfft(rfft(a_rev, n=size) * rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = 2.0 * grad_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of max(convolution).
    More aggressive parameters than the reference implementation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1_048_576.0            # larger than before for a tighter soft‑max
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.99                      # stronger momentum
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to encourage exploration
        if it % 150 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                    target_sum: float,
                                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Several short GD restarts with random jitter; keep the best result.
    """
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(35.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start, time.time() + run_budget,
                                            target_sum)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# Simulated annealing (improved)

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 15.0,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.6,
                         max_frac_end: float = 0.02,
                         top_k: int = 5,
                         max_window: int = 120) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using simulated annealing.
    Moves:
        * mass transfer (biased by a sensitivity vector built from the top k peaks)
        * local averaging (smooth)
        * random swap
    Temperature and allowed transfer fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # ------------------------------------------------------------------
        # Build a sensitivity vector from the top‑k convolution entries
        # ------------------------------------------------------------------
        # indices of the k largest values (may contain duplicates if the max is flat)
        idx_top = np.argpartition(-conv, top_k)[:top_k]
        sens = np.zeros(n)                     # ∂max/∂a (linearised)
        for s in idx_top:
            lo = max(0, s - (n - 1))
            hi = min(s, n - 1)
            # for every i in [lo, hi] we add a_{s-i}
            sens[lo:hi + 1] += a[s - np.arange(lo, hi + 1)]

        weight = a * sens                     # entries that contribute most to the peak
        move_type = None

        # ------------------- mass transfer -------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # pick a low‑weight receiver (bottom 15 % of weight)
            low_thr = np.percentile(weight, 15)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.85:               # smoother more often than swap
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution & score
        conv_new = irfft(rfft(a, n=size_fft) * rfft(a, n=size_fft), n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis criterion
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to avoid stagnation
        if it % 7000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _subgradient_max(a: np.ndarray):
    """
    Compute the (sub)gradient of the true max(convolution) w.r.t. a.
    Returns (grad, max_val) where grad has the same shape as a.
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    # indices that are (almost) maximal
    idx = np.where(conv >= max_val - 1e-12)[0]

    grad = np.zeros_like(a)
    for s in idx:
        lo = max(0, s - (n - 1))
        hi = min(s, n - 1)
        # contribution 2*a_{s-i}
        grad[lo:hi + 1] += 2.0 * a[s - np.arange(lo, hi + 1)]
    return grad, max_val


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple diminishing‑step sub‑gradient method on the true max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    step0 = target_sum * 0.5

    while time.time() < deadline:
        grad, _ = _subgradient_max(a)
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        lr = step0 / ((it + 1) ** 0.5 * norm)
        a -= lr * grad
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with a low benchmark score.
    The routine adapts the problem size to the allocated time, runs
    several optimisation phases, and returns the best candidate found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – larger n gives more freedom but costs more time.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 0 – smart initialisation (power‑law scan + jitter)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init = init[:n]                     # truncate / pad if necessary
        if init.size < n:
            # simple linear extrapolation if the global is shorter
            init = np.interp(np.linspace(0, 1, n),
                             np.linspace(0, 1, init.size), init)
        init = _project_to_simplex(init, target_sum)
    else:
        init = _initial_candidate(n, target_sum)

    # ------------------------------------------------------------------
    # Phase 1 – projected gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.25
    a_gd, val_gd = _gradient_descent_with_restarts(init, target_sum, gd_deadline)
    best_seq = a_gd
    best_score = val_gd
    print(f"[{time.time():.2f}] After GD phase best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, deadline,
                                            start_temp=15.0,
                                            end_temp=1e-6,
                                            max_frac_start=0.6,
                                            max_frac_end=0.02,
                                            top_k=5,
                                            max_window=120)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
        print(f"[{time.time():.2f}] SA finished best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent on the exact ℓ∞‑norm
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
            print(f"[{time.time():.2f}] Sub‑gradient phase improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short polishing projected gradient descent (optional)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _annealed_gradient_descent(best_seq,
                                                    deadline,
                                                    target_sum)
        if val_fin < best_score:
            best_seq, best_score = a_fin, val_fin
            print(f"[{time.time():.2f}] Final polish improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean up – drop trailing zeros and return plain Python list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:

    final_seq = arr.tolist()

    # Optional final verification using the provided benchmark function
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
