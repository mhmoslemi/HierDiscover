Define `_flatten_spectrum(a, rng)`:

```
def _flatten_spectrum(a, rng):
    n = len(a)
    # compute FFT size power of two
    N = 1 << (n-1).bit_length()
    A = np.fft.rfft(a, n=N)
    mag = np.abs(A)
    ang = np.angle(A)
    mean_power = np.mean(mag**2)
    new_mag = np.sqrt(mean_power) * np.ones_like(mag)
    A_new = new_mag * np.exp(1j * ang)
    a_new = np.fft.irfft(A_new, n=N)[:n]
    # Clip negative
    a_new = np.clip(a_new, 0, None)
    return a_new

def _flatten_spectrum_step(a, mix=0.2):

"""
Proposes a step‑function (non‑negative vector) with a low self‑convolution peak.

Algorithm
---------
1. **Initial sampling** – generate many analytic families (including
   very‑heavy‑end beta‑shapes) for a variety of lengths, normalise the sum
   to √(2·n) and keep the best.

2. **Iterative refinement** – starting from the best candidate:
   * several *Adam* runs on the smooth‑max approximation of the peak,
   * occasional *spectral flattening* (forcing the FFT magnitude towards a
     constant) to spread power evenly,
   * *targeted mass moves* that reduce the current maximal convolution entry,
   * a final cheap random‑swap hill‑climb.

3. **Upsampling** – after a successful improvement the sequence is linearly
   up‑sampled (doubling its length, up to a limit) and the whole refinement
   loop repeats.  Larger lengths give more degrees of freedom and usually
   lower peaks.

The routine is time‑aware (budget ≈ 1000 s) and stores the best sequence in
the global variable ``height_sequence_1`` for reuse in later runs.

The returned value is a plain ``list[float]`` (the heights).  The external
score printed at the end equals ``evaluate_sequence`` which, for a normalised
vector, is exactly the maximal convolution entry.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # indices that map back to the original convolution gradient
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) probabilities over the convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helper (optional)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] == v[-1-i]."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Spectral flattening – bring the FFT magnitude towards a constant
# ----------------------------------------------------------------------
def _flatten_spectrum_step(a: np.ndarray, mix: float = 0.2) -> np.ndarray:
    """One step of spectral flattening (mix determines aggressiveness)."""
    n = a.shape[0]
    N = 1 << (n - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=N)
    mag = np.abs(A)
    ang = np.angle(A)
    mean_power = np.mean(mag ** 2)
    target_mag = np.sqrt(mean_power)
    new_mag = (1.0 - mix) * mag + mix * target_mag
    A_new = new_mag * np.exp(1j * ang)
    a_new = np.fft.irfft(A_new, n=N)[:n]
    return a_new


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max of the convolution
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.7
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.06
    max_alpha = 300.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape plateaux
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.07 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max (single‑term sub‑gradient)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.12
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # build sub‑gradient (all pairs that contribute to k_max)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]          # contribution from (i,j)

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (help escape shallow minima)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.025 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑swap hill‑climbing
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that swaps tiny masses between random positions.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.018 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted mass moves that explicitly reduce the current maximal entry
# ----------------------------------------------------------------------
def _targeted_move(a: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    Repeatedly move a small amount of mass away from indices that
    contribute to the current maximal convolution entry.
    """
    n = a.shape[0]
    while time.time() < deadline:
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # pick a contributor i (uniformly among those that actually have mass)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        if i_low > i_high:
            break
        i = rng.integers(i_low, i_high + 1)
        if a[i] <= 1e-12:
            continue
        # pick a destination far from the contributing region
        # (indices that do NOT belong to [i_low, i_high])
        possible = np.arange(n)
        mask = (possible < i_low) | (possible > i_high)
        if not mask.any():
            break
        j = rng.choice(possible[mask])
        eps = 0.04 * a[i] * rng.random()
        a[i] -= eps
        a[j] += eps
        # keep on the simplex (the move already preserves sum, only need non‑negativity)
        a = np.maximum(a, 0.0)
    return a

    deadline = start_time + budget_s - 5.0      # safety margin

                best_score = np.max(_full_convolution(best_seq))
                print(f"[reuse] loaded previous seq, length={arr.size}, score={best_score:.6f}")
        except Exception:
            best_seq = None

    # Phase 1 – cheap random families (≈20 % of time)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    attempts = 0
    while time.time() < phase1_deadline:
        attempts += 1

        # length choice – a mix of moderate and large sizes
        if best_seq is not None and rng.random() < 0.45:
            n = max(1500, int(best_seq.shape[0] * rng.uniform(0.9, 1.1)))
        else:
            n = int(rng.integers(2500, 12001))

        # pick a analytic family
        fam = rng.integers(0, 9)
        if fam == 0 and best_seq is not None:
            a = _upsample_seq(best_seq, n)
        elif fam == 1:
            a = rng.random(n)                               # uniform random
        elif fam == 2:
            # geometric tail on both sides
            rate = rng.uniform(0.93, 0.9995)
            left = rate ** np.arange(n)
            a = (left + left[::-1]) * 0.5
        elif fam == 3:
            # power‑law symmetric
            p = rng.uniform(0.1, 5.0)
            idx = np.arange(n, dtype=float)
            a = ((idx + 1) * (n - idx)) ** p
        elif fam == 4:
            # beta‑like (allow very sharp ends)
            p1 = rng.uniform(0.02, 0.8)
            p2 = rng.uniform(0.02, 0.8)
            x = np.linspace(0, 1, n)
            a = (x ** p1) * ((1 - x) ** p2)
        elif fam == 5:
            # sinusoidal power
            p = rng.uniform(0.3, 4.5)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        elif fam == 6:
            # quadratic hump (concave)
            a = 1.0 - ((np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)) ** 2
            a = np.clip(a, 0, None)
        elif fam == 7:
            # exponential decay from both ends
            decay = rng.uniform(0.001, 0.02)
            a = np.exp(-decay * np.abs(np.arange(n) - (n - 1) / 2))
        else:
            # combination of two separated bumps
            split = rng.integers(n // 4, 3 * n // 4)

            a = np.zeros(n)
            left = max(0, split - width // 2)
