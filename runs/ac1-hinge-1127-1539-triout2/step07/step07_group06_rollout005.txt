def softmax_max_and_grad(a, beta):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    # Autocorrelation
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = np.max(b)
    # compute softmax weights
    # subtract max_b for numeric stability
    diff = beta * (b - max_b)
    # maybe subtract max of diff to avoid overflow (but diff has max =0)
    exp_diff = np.exp(diff)
    p = exp_diff / exp_diff.sum()
    # Now gradient w.r.t a: g = 2 * (reverse of a convolved with p) at appropriate indices.
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2*n-1]
    # g_i = 2 * conv_rev[n-1 + i]  (since conv_rev length 2n-1)
    g = 2.0 * conv_rev[n-1 : n-1 + n]
    # objective: max_b + (1/beta) * log sum exp(beta*(b - max_b))
    softmax_val = max_b + (1.0 / beta) * np.log(exp_diff.sum())
    return softmax_val, g, b
```

def _local_refine(a, deadline, rng):
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    # Ensure sum normalisation (a may sum to target)
    a = a.copy()

    s = a.sum()
    if s != 0:
        a *= target / s
    # perform some random mass transfers
    cur_max = _max_convolution(a)
    best_a = a.copy()
    best_val = cur_max
    while time.time() < deadline:
        # pick donor and receiver
        donor = rng.integers(n)
        receiver = rng.integers(n)
        if donor == receiver:
            continue
        delta = a[donor] * rng.random() * 0.1   # move up to 10%
        if delta <= 0:
            continue
        a2 = a.copy()
        a2[donor] -= delta
        a2[receiver] += delta
        # enforce nonnegativity
        if a2[donor] < 0:
            continue
        # re-normalize sum to target
        a2 *= target / a2.sum()
        new_max = _max_convolution(a2)
        if new_max < cur_max:
            a = a2
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a2.copy()
                print(f"[{time.time():.1f}] local refine improved max={best_val:.6f}")
    return best_a

def _softmax_max_and_grad(a, beta):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = b.max()
    # compute softmax weights
    # diff = beta * (b - max_b)   # all <=0
    diff = beta * (b - max_b)
    exp_diff = np.exp(diff)  # values in (0,1]
    sum_exp = exp_diff.sum()
    p = exp_diff / sum_exp
    # compute gradient w.r.t a using convolution with p
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2*n-1]
    g = 2.0 * conv_rev[n-1 : n-1 + n]
    # objective value
    val = max_b + (1.0 / beta) * np.log(sum_exp)
    return val, g
```

"""
Step‑function height optimizer.

The algorithm searches for a non‑negative sequence a = (a_0,…,a_{n‑1})
with ℓ¹‑norm Σ a_i = √(2·n) that minimises the peak of the
self‑convolution a∗a, which is exactly the value returned by
`evaluate_sequence`.  A smooth approximation of the peak
(max‑convolution) is optimised with L‑BFGS‑B on the soft‑max
formulation.  The coefficients are parametrised by a soft‑max over an
unconstrained vector y, guaranteeing positivity and the ℓ¹‑constraint
by construction.  The temperature β of the soft‑max is increased
geometrically (β = 1,2,4,…,4096) so that the optimisation first sees a
smooth landscape and finally focuses on the true maximum.
For several problem sizes (2000–4000) the optimiser is run within the
overall time budget (≈1000 s).  The best sequence found is returned.
A short random‑jitter polishing is performed at the end.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper 1 – convolution (full linear) via FFT
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of size power‑of‑2."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Helper 2 – soft‑max approximation of the max entry and its gradient
# ----------------------------------------------------------------------
def _softmax_max_and_grad(a: np.ndarray, beta: float):
    """
    Returns a smooth approximation of max(a*a) together with the gradient
    w.r.t. the coefficient vector a.

    The approximation is
        fβ(a) = max_b + (1/β)·log Σ exp(β·(b_i−max_b)),
    where b = a*a.
    The gradient is
        ∇_a fβ(a) = 2·conv_rev(p)[n‑1:n‑1+n],
    with p_i = exp(β·(b_i−max_b))/Σ exp(·) and conv_rev denoting the
    convolution of the reversed a with p.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    # autocorrelation b = a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # numerically safe soft‑max weights (all ≤ 1)
    diff = beta * (b - max_b)
    exp_diff = np.exp(diff)
    sum_exp = exp_diff.sum()
    p = exp_diff / sum_exp

    # gradient w.r.t. a (see notes above)
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # smooth value of the max
    value = max_b + (1.0 / beta) * np.log(sum_exp)
    return value, grad_a


# ----------------------------------------------------------------------
# Helper 3 – objective (smooth max) and its gradient w.r.t. y
# ----------------------------------------------------------------------
def _obj_and_grad(y: np.ndarray, beta: float, target: float):
    """
    Soft‑max objective and gradient w.r.t. the unconstrained vector y.
    The coefficients are a_i = target·softmax(y)_i,
    guaranteeing a_i ≥ 0 and Σ a_i = target.
    """
    # stable soft‑max
    y_shift = y - np.max(y)
    exp_y = np.exp(y_shift)
    sum_exp = np.sum(exp_y)
    p = exp_y / sum_exp                      # probabilities, Σ p = 1
    a = target * p                            # coefficients, Σ a = target

    # smooth max and its gradient w.r.t. a
    value, grad_a = _softmax_max_and_grad(a, beta)

    # chain rule: ∂a_i/∂y_j = target·p_i·(δ_{ij}−p_j)
    avg_grad = np.dot(p, grad_a)              # ⟨p , grad_a⟩
    grad_y = target * p * (grad_a - avg_grad)

    return value, grad_y


# ----------------------------------------------------------------------
# Core optimiser for a single length n
# ----------------------------------------------------------------------
def _optimize_length(n: int, deadline: float, rng: np.random.Generator):
    """
    Optimises a sequence of length n until `deadline` is reached.
    Returns the best coefficient array and its exact max‑convolution value.
    """
    target = np.sqrt(2.0 * n)          # required ℓ¹‑norm
    # start from a uniform distribution (y = 0)
    y = np.zeros(n, dtype=np.float64)

    best_a = None
    best_val = np.inf
    beta = 1.0
    max_beta = 4096.0

    # main continuation loop over the soft‑max temperature
    while time.time() < deadline and beta <= max_beta:
        # remaining time budget for this β
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # L‑BFGS‑B options – a few dozen iterations are enough per β
        maxiter = max(30, int(remaining * 0.1))
        try:
            res = scipy.optimize.minimize(
                _obj_and_grad,
                y,
                args=(beta, target),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter, "disp": False},
            )
        except Exception as exc:          # safety net if SciPy misbehaves
            print(f"scipy.optimize failed at β={beta}: {exc}")
            break

        y = res.x

        # construct a from the final y of this stage
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        p = exp_y / exp_y.sum()
        a = target * p

        cur_max = _max_convolution(a)
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n:4d} β={beta:6.1f}  max={best_val:.6f}")

        beta = min(beta * 2.0, max_beta)

    # fallback – if nothing was stored, use the last iterate
    if best_a is None:
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        p = exp_y / exp_y.sum()
        best_a = target * p
        best_val = _max_convolution(best_a)

    # ------------------------------------------------------------------
    # Very short discrete refinement (random jitter + re‑normalisation)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        refine_deadline = deadline
        a = best_a.copy()
        while time.time() < refine_deadline:
            scale = target * 0.001
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            # re‑normalise to the exact ℓ¹‑norm (soft‑max style)
            s = a.sum()
            if s > 0:
                a *= target / s
            else:
                a[:] = target / n
            cur = _max_convolution(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter refine improved max={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget and returns a list of
    non‑negative floats (the heights).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5                     # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optionally warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.size
        if time.time() < deadline - 5.0:
            run_deadline = time.time() + min(25.0, (deadline - time.time()) * 0.15)
            seq, val = _optimize_length(n0, run_deadline, rng)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] warm‑start n={n0} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try several lengths (larger n often gives a better bound)
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000]
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        # allocate at most 30 % of the remaining budget to this size
        slice_time = min(remaining * 0.30, 120.0)
        run_deadline = time.time() + slice_time
        seq, val = _optimize_length(n, run_deadline, rng)

        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] new global best n={n} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (very cheap random jitter) if time permits
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.size
        target = np.sqrt(2.0 * n)
        a = best_seq.copy()
        while time.time() < deadline - 0.3:
            scale = target * 0.0005
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a *= target / s
            else:
                a[:] = target / n
            cur = _max_convolution(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] final jitter improved max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing negligible entries (they do not affect the objective)
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
