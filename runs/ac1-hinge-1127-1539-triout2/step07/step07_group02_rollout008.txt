"""
Propose a step‑function (non‑negative vector) whose self‑convolution peak is
as low as possible.  The search runs in four timed phases:

1. **Broad sampling** – draws many analytic families (uniform, beta‑like,
   power‑law, geometric, sinusoidal) for random lengths and keeps the best
   candidate (symmetrised and normalised to Σa = √(2·n), which makes the
   evaluation equal to the convolution peak).

2. **Projected Adam** – minimises a smooth soft‑max approximation of the
   convolution peak.  The temperature α is increased up to 2000, the learning
   rate decays slowly and the iterate is projected onto the simplex after
   each step.  Symmetry is enforced throughout.

3. **Refined sub‑gradient descent** – uses a high‑α soft‑max to obtain a
   (near‑optimal) sub‑gradient of the true maximum, with a diminishing step
   size and occasional random mass swaps.

4. **Tiny‑swap hill‑climb** – final local search that repeatedly moves a tiny
   amount of mass between two positions.

The best sequence found is stored in the global ``height_sequence_1`` for
future runs and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    ``w`` must be a probability vector (soft‑max weights) of length 2*n‑1.
    """
    n = a.shape[0]
    # reverse w for convolution (c_k = Σ_i a_i a_{k-i})
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # the gradient entries correspond to indices (2n‑2‑i)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights for a smooth max
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.04
    max_alpha = 2000.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.012:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
                if enforce_sym:
                    a = _symmetrize(a)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

    return best_a, best_score


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Refined sub‑gradient descent on the true maximum (via a high‑α soft‑max).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        # smooth‑max with a very high temperature to obtain a near‑exact sub‑gradient
        w = _smooth_weights(conv, alpha=500.0)
        grad = _conv_with_weights(a, w)

        # diminishing step size
        eta = 0.25 / np.sqrt(it)
        a -= eta * grad

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta

                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early exit test – if no progress for a while, break
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing that repeatedly moves a tiny amount of mass
    between two random positions.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin of a few seconds so that the runner does not overrun
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve previous best if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_val = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – cheap random families (≈25 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    while time.time() < phase1_end:
        attempts += 1

        # pick a length – favour moderate sizes but keep some variability
        n = int(rng.integers(1800, 12001))

        # sample a family
        fam = rng.integers(0, 6)
        if fam == 0 and base_seq is not None:
            # linearly up‑sample a known good sequence
            x_old = np.linspace(0.0, 1.0, base_seq.size)
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, base_seq)
        elif fam == 1:
            a = rng.random(n)                               # pure uniform noise
        elif fam == 2:
            # geometric decay from both ends
            rate = rng.uniform(0.95, 0.9999)
            left = rate ** np.arange(n)
            a = (left + left[::-1]) * 0.5
        elif fam == 3:
            # symmetric power‑law (i·(n‑i))**p
            p = rng.uniform(0.2, 4.0)
            idx = np.arange(n, dtype=float)
            a = ((idx + 1) * (n - idx)) ** p
        elif fam == 4:
            # beta‑like shape
            p1 = rng.uniform(0.1, 5.0)
            p2 = rng.uniform(0.1, 5.0)
            x = np.linspace(0.0, 1.0, n)
            a = (x ** p1) * ((1 - x) ** p2)
        else:
            # sinusoidal power
            p = rng.uniform(0.5, 4.0)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p

        a = np.clip(a, 0.0, None)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # occasionally enforce exact symmetry
        if rng.random() < 0.6:
            a = _symmetrize(a)

        cur_val = np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[phase1] attempt={attempts} n={n} val={cur_val:.6f}")

    # fallback in the (unlikely) case we have no candidate yet
    if best_seq is None:
        n = 5000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_seq = a
        best_val = np.max(_full_convolution(a))

    # ------------------------------------------------------------------
    # Phase 2 – projected Adam (≈55 % of the remaining budget)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    adam_deadline = now + 0.55 * remaining
    a_adam, val_adam = _adam_optimize(best_seq, adam_deadline, rng,
                                      enforce_sym=True)
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – refined sub‑gradient descent (≈30 % of the rest)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        sub_deadline = now + 0.30 * (deadline - now)
        a_sub, val_sub = _subgrad_optimize(best_seq, sub_deadline, rng,
                                           enforce_sym=True)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – tiny‑swap hill‑climbing (≈remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_loc, val_loc = _local_swap_search(best_seq, deadline, rng,
                                            enforce_sym=True)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final result, external evaluation and global export
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_val:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
