def _hill_climbing_improved(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    cur_score = best_score

    rng = np.random.default_rng()
    T0 = cur_score * 0.15 + 1e-9
    T = T0
    cooling = 0.9998

    while time.time() < deadline:
        # find indices of max entries
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        M = rng.choice(max_idxs)

        # compute contributions to this entry M (as before)
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        # choose source proportional to contrib
        if total_contrib > 0:
            i = rng.choice(idx, p=contrib / total_contrib)
        else:
            i = rng.integers(n)

        # source must have mass
        if a[i] <= 1e-12:
            continue

        # choose destination biased towards low a
        if rng.random() < 0.6:
            # probability inversely proportional to a
            inv_a = 1.0 / (a + 1e-12)
            probs = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs)
        else:
            j = rng.integers(n)
        if i == j:
            continue

        # delta adapt: fraction of source mass scaled with temperature
        max_delta = a[i] * min(0.9, (T / T0))  # early big moves
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # update convolution
        # linear terms
        b_new = b.copy()
        # subtract contribution of source in all positions it appears
        b_new[i:i + n] -= 2.0 * delta * a
        # add contribution of dest
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction for squares and cross term
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # accept
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb_improved improvement: {best_score:.6f}")
        # cooling
        T *= cooling
    return best_a, b

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm outline
-----------------
* **Stage‑wise length schedule** – start with modest lengths and grow up to
  80 000, re‑using the best solution from the previous stage (linear up‑sampling).

* **Diverse random initialisers** – power‑law, exponential, Dirichlet,
  sine, Gaussian, triangular, raised‑cosine, sinc‑square and piecewise
  exponential families.

* **Projected Adam optimisation** – minimise a smooth log‑sum‑exp
  surrogate of the worst autocorrelation entry on the ℓ₁‑simplex
  Σa = √(2 n).  The inverse temperature β is annealed up to 10⁶,
  while Adam’s adaptive step‑size accelerates convergence.

* **Aggressive hill‑climbing** – simulated‑annealing style mass transfer
  that updates the full self‑convolution in O(n) per move.
  Source indices are chosen proportionally to their contribution to a
  currently worst convolution entry; destination indices are biased
  towards low‑mass entries.  Temperature decays very slowly to allow
  many uphill moves.

* **Final polishing** – a last high‑β Adam run on the best vector.

All vectors are projected onto the simplex after every move, so the
official evaluator sees a valid non‑negative step‑function.  The routine
obeys the wall‑clock budget (default 1000 s) and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi et al. 2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).

    Returns (grad, b) with
        grad = 2·(a ⋆ p),   p_k ∝ exp(β·b_k)
    where ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]

    # ----- compute full convolution b = a*a -----
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities over b -----
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # ----- gradient = 2·(a ⋆ p) -----
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected Adam optimisation of the soft‑max surrogate."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # temperature for the soft‑max
    beta = 0.5
    beta_max = 1e6
    beta_factor = 1.5

    # base learning‑rate (scaled with problem size)
    lr_base = target_sum * 0.06

    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Parameter update
        step = lr_base * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # Projection onto the ℓ₁‑simplex
        a = _project_to_simplex(a, target_sum)

        # Anneal β (sharpens the surrogate)
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # Occasionally add a tiny jitter to escape plateaus
        if t % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 10)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # positive half‑wave sine squared
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    elif family == 6:                   # triangular shape
        a = np.minimum(idx + 1.0, n - idx)

    elif family == 7:                   # raised cosine (full period)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))

    elif family == 8:                   # squared sinc (normalized)
        # np.sinc(x) = sin(πx)/(πx)
        a = np.sinc(idx / (n - 1)) ** 2

    else:                               # piecewise exponential (fallback)
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    # Small Gaussian perturbation for extra diversity
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact O(n) updates)
# ----------------------------------------------------------------------


def _hill_climbing_improved(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing hill climbing that moves a small amount of mass
    from a source index (chosen proportional to its contribution to a worst
    convolution entry) to a destination index (biased towards low mass).
    The full self‑convolution is updated in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T0 = cur_score * 0.15 + 1e-9
    T = T0
    cooling = 0.9998    # very slow cooling – more uphill moves

    while time.time() < deadline:
        # pick one of the worst entries uniformly
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            i = rng.choice(idx, p=contrib / total_contrib)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index – prefer low‑mass locations
        if rng.random() < 0.6:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass – larger while temperature is high
        max_delta = a[i] * min(0.9, T / T0)
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update the convolution (linear part) -----
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # ----- quadratic correction terms (squares and cross) -----
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ----- Metropolis acceptance -----
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb_improved improvement: {best_score:.6f}")

        # temperature decay
        T *= cooling

    return best_a, b

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seeding from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – increasingly larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000, 80000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – Adam gradient descent (few restarts)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 15.0:
            break
        gd_total_budget = min(remaining * 0.30, 260.0)   # allocate up to ~260 s
        n_restarts = 4 if n <= 10000 else 2
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            now = time.time()
            per_restart = min((deadline - now) * 0.33, gd_total_budget / n_restarts)
            gd_deadline = now + per_restart

            a_start = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_gd = _adam_opt(a_start, gd_deadline)
            score = _max_convolution_fft(a_gd)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_gd.copy()

        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time():.1f}] Adam GD improvement (n={n}) score={local_best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – aggressive hill climbing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 15.0:
            break
        hill_budget = min(remaining * 0.35, 300.0)   # allocate up to ~300 s
        hill_deadline = time.time() + hill_budget

        if best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = best_seq.copy()
        a_current = _project_to_simplex(a_current, target_sum)
        b_full = _full_convolution(a_current)

        a_hc, b_hc = _hill_climbing_improved(a_current, b_full, hill_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Optional fine‑grained Adam refinement
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 6.0:
            gd2_deadline = time.time() + min(remaining * 0.15, 80.0)
            a_refine = _adam_opt(best_seq.copy(), gd2_deadline)
            score_ref = _max_convolution_fft(a_refine)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_refine.copy()
                print(f"[{time.time():.1f}] Post‑HC Adam refinement (n={n}) score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target)
        a_final = _adam_opt(a, deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros (nice output)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log official evaluation (for debugging/visibility)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
