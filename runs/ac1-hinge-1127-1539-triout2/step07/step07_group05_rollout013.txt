"""
Propose a non‑negative step‑function (list of heights) that minimises the
``evaluate_sequence`` objective.

Algorithm (≈ 1000 s budget)
----------------------------
1. **Random parametric families** – power law, exponential, linear,
   quadratic and a mixed power‑law / exponential.  Each candidate is
   projected onto the ℓ¹‑simplex Σa = √(2 n).

2. **Fast local optimisation** – a projected gradient descent on a
   smoothly‑approximated max‑norm (log‑sum‑exp) is run for a generous
   number of iterations, keeping the ℓ¹‑sum fixed.

3. **Linear‑program refinement** – given a current sequence *a*,
   compute M = max(a*a).  The LP  

        maximise   Σ g_j
        subject to (a * g)_k ≤ M   for all k,
                   g_j ≥ 0

   is solved (SciPy “highs” interior‑point).  The optimal *g* has a larger
   ℓ¹‑norm; after scaling it back to the target sum the self‑convolution
   bound strictly decreases.  This step is repeated until no further
   improvement is observed (or time runs out).

4. **Permutation annealing** – the multiset of values is kept fixed while
   random swaps are performed with a temperature schedule; the best
   ordering found is kept.

5. The best sequence over the whole run (different lengths, random seeds,
   and the two refinement stages) is returned.

All helper functions are defined at top level; no closures or lambdas are
used.  The code respects the time budget and prints progress messages.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed surrogate of max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where
        conv = a*a (full convolution)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·conv_k) )
             = 2·(a ⋆ p) ,  p_k ∝ exp(β·conv_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # log‑sum‑exp probabilities (stable)
    max_c = np.max(conv)
    exp_shifted = np.exp(beta * (conv - max_c))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient = 2 * (a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    A_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    rev = np.fft.irfft(A_fft * P_fft, n=size2)[:conv_len]
    grad = 2.0 * rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max objective
# ----------------------------------------------------------------------
def gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5           # start with a gentle smoothing
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9            # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.7 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        a = project_to_simplex(a, target_sum)

        # occasional jitter to shake out of plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 6e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = project_to_simplex(a, target_sum)

        # increase smoothing sharpness
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.12, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    Only the permutation changes; the ℓ¹‑sum stays identical.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur = max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # tiny random shuffle very rarely
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                tmp = a[mask].copy()
                rng.shuffle(tmp)
                a[mask] = tmp
                cur = max_convolution_fft(a)
                if cur < best_score:
                    best_score = cur
                    best = a.copy()
    return best


# ----------------------------------------------------------------------
# Build the (2n‑1) × n convolution matrix for a fixed left factor a
# ----------------------------------------------------------------------
def build_conv_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Returns sparse matrix A where (A @ g) = a * g (full linear convolution).
    Shape: (2n‑1, n)
    """
    n = a.size
    rows = []
    cols = []
    data = []

    # iterate over output index k
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # j indexes the variable g_j; coefficient = a_{k-j}
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=float)
    return A


# ----------------------------------------------------------------------
# Linear‑program refinement step (one iteration)
# ----------------------------------------------------------------------
def lp_refine_step(a: np.ndarray, target_sum: float, deadline: float):
    """
    One LP‑based refinement.
    Returns a new sequence (possibly None if no improvement or time out).
    """
    n = a.size
    M = max_convolution_fft(a)
    A = build_conv_matrix(a)

    # RHS of constraints
    b = np.full(2 * n - 1, M)

    # maximise Σg  → minimise -Σg
    c = -np.ones(n)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method='highs',
        options={'time_limit': max(0.0, deadline - time.time())}
    )
    if not res.success:
        return None

    g = res.x
    sum_g = g.sum()
    if sum_g <= target_sum * (1.0 + 1e-9):
        return None

    scale = target_sum / sum_g
    new_a = g * scale
    return new_a


def lp_refine(a: np.ndarray, deadline: float, max_iters: int = 8) -> np.ndarray:
    """
    Repeatedly apply the LP refinement until no further improvement or
    time runs out.
    """
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    best = a.copy()
    best_obj = max_convolution_fft(best)

    it = 0
    while it < max_iters and time.time() < deadline:
        cand = lp_refine_step(best, target_sum, deadline)
        if cand is None:
            break
        cand_obj = max_convolution_fft(cand)
        if cand_obj < best_obj:
            best, best_obj = cand, cand_obj
            # small permutation anneal after each successful LP step
            sub_dead = min(deadline, time.time() + 3.0)
            best = permutation_annealing(best, sub_dead)
            best_obj = max_convolution_fft(best)
        else:
            # if LP does not improve, stop
            break
        it += 1
    return best


# ----------------------------------------------------------------------
# Parametric families (output already ℓ¹‑normalised)
# ----------------------------------------------------------------------
def normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale vector to have ℓ¹‑sum = target (preserving non‑negativity)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, target / v.size)
    a = v * (target / s)
    a = np.maximum(a, 0.0)
    return a


def powerlaw_seq(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return normalize_to_sum(raw, np.sqrt(2.0 * n))


def exponential_seq(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return normalize_to_sum(raw, np.sqrt(2.0 * n))


def linear_seq(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return normalize_to_sum(raw, np.sqrt(2.0 * n))


def quadratic_seq(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return normalize_to_sum(raw, np.sqrt(2.0 * n))

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 37, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimizing ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_obj = np.inf
    elite = []          # (obj, array)

    def consider(seq: np.ndarray):
        """Update global best and elite list."""
        nonlocal best_seq, best_obj, elite
        obj = max_convolution_fft(seq)
        if obj < best_obj:
            best_obj = obj
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST obj={best_obj:.6f} len={seq.size}")
        if len(elite) < 14 or obj < elite[-1][0]:
            elite.append((obj, seq.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:14]

    # ------------------------------------------------------------------
    # 0. Warm start from a supplied construction, if present
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        n0 = init.size
        init = normalize_to_sum(init, np.sqrt(2.0 * n0))
        consider(init)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Random parametric generation (70 % of time budget)
    # ------------------------------------------------------------------
    lengths = [1800, 2200, 2500, 2800, 3200, 3500, 3800, 4200, 4600, 5000]
    families = ['power', 'exp', 'linear', 'quadratic', 'mixed']

    gen_deadline = time.time() + 0.70 * (deadline - time.time())

    while time.time() < gen_deadline:
        n = int(rng.choice(lengths))
        target = np.sqrt(2.0 * n)

        fam = rng.choice(families)
        if fam == 'power':
            p = rng.uniform(0.3, 0.9)
            shift = rng.uniform(0.0, 4.0)
            base = powerlaw_seq(n, p, shift)

        elif fam == 'exp':
            rate = rng.uniform(0.0002, 0.015)
            shift = rng.uniform(0.0, 6.0)
            base = exponential_seq(n, rate, shift)

        elif fam == 'linear':
            slope = rng.uniform(1e-5, 0.012)
            intercept = rng.uniform(0.8, 5.0)
            base = linear_seq(n, slope, intercept)

        elif fam == 'quadratic':
            a0 = rng.uniform(0.7, 6.0)
            b = rng.uniform(0.0, 0.012)
            c = rng.uniform(0.0, 8e-5)
            base = quadratic_seq(n, a0, b, c)

        else:  # mixed
            w = rng.uniform(0.2, 0.8)

            p = rng.uniform(0.3, 0.9)
            shift_p = rng.uniform(0.0, 4.0)
            pow_seq = powerlaw_seq(n, p, shift_p)

            rate = rng.uniform(0.0002, 0.015)
            shift_e = rng.uniform(0.0, 6.0)
            exp_seq = exponential_seq(n, rate, shift_e)

            mix = w * pow_seq + (1.0 - w) * exp_seq
            base = project_to_simplex(mix, target)

        # ----- ordering heuristics -----
        asc = np.sort(base)                 # increasing
        desc = asc[::-1]                    # decreasing
        # alternating small‑large (small first)
        alt = np.empty_like(asc)
        lo, hi = 0, n - 1
        for i in range(n):
            if i % 2 == 0:
                alt[i] = asc[lo]
                lo += 1
            else:
                alt[i] = asc[hi]
                hi -= 1
        # quick random shuffle
        shuf = asc.copy()
        rng.shuffle(shuf)

        for cand in (desc, asc, alt, shuf):
            consider(cand)

        # occasional cheap permutation anneal on current global best
        if best_seq is not None and rng.random() < 0.06:
            tmp = permutation_annealing(best_seq, time.time() + 0.22)
            consider(tmp)

    # ------------------------------------------------------------------
    # 2. Gradient descent refinement of elite candidates
    # ------------------------------------------------------------------
    for _, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        sub_dead = min(deadline, time.time() + 16.0)

        refined = gradient_descent_opt(seed_seq, sub_dead)

        # finalize with permutation annealing
        refined = permutation_annealing(refined, deadline)
        consider(refined)

    # ------------------------------------------------------------------
    # 3. Linear‑program refinement (high‑impact step, limited to one length)
    # ------------------------------------------------------------------
    # pick the currently best length that is not too large (≈3000)
    if best_seq is not None:
        n_best = best_seq.size
        if n_best > 3200:
            # down‑sample by averaging adjacent pairs to keep complexity manageable
            factor = n_best // 3000 + 1
            new_n = n_best // factor
            reshaped = best_seq[: new_n * factor].reshape(new_n, factor).mean(axis=1)
            best_seq = normalize_to_sum(np.array(reshaped), np.sqrt(2.0 * new_n))
            n_best = new_n
            consider(best_seq)

        # now run the LP refinement
        lp_dead = time.time() + min(30.0, deadline - time.time() - 3.0)
        refined_lp = lp_refine(best_seq, lp_dead, max_iters=7)
        consider(refined_lp)

    # ------------------------------------------------------------------
    # 4. Final permutation annealing on the overall best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = permutation_annealing(best_seq, deadline)
        best_obj = max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 5. Fallback if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # Trim insignificant trailing entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Final verification (optional)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] FINISHED. Best objective ≈ {best_obj:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
