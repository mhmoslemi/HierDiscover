"""
Propose a step‑function (non‑negative vector) with minimal self‑convolution peak.

Algorithm
---------
1. **Exploratory sampling** – generate many analytic families and random
   permutations for a variety of lengths.  Keep the best raw candidate.

2. **Projected Adam** – minimise a soft‑max (log‑sum‑exp) approximation of the
   convolution peak.  The temperature (α) is increased up to 800, the learning
   rate decays, and the iterate is projected onto the simplex
   {x ≥ 0, Σx = √(2·n)} after each step.

3. **Sub‑gradient descent** – direct descent on the true peak (the max of the
   convolution).  The gradient is the exact sub‑gradient of the active
   diagonal, i.e. 2·a_{k‑i} for i ≠ k‑i and a_i for i = k‑i.

4. **Permutation annealing** – the multiset of heights is kept fixed while we
   search for a favourable ordering.  Swaps are accepted with a Metropolis
   criterion; the temperature is exponentially cooled.

5. **Fine local swaps** – a cheap hill‑climber that repeatedly moves a tiny
   amount of mass between two positions.

All phases respect the global time budget (default 1000 s).  The best sequence
found is stored in the global variable ``height_sequence_1`` for reuse.
"""

# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights over convolution entries
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    m = np.max(scaled)
    e = np.exp(scaled - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective w.r.t. a
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a (1/α)·log‑sum‑exp(α·(a*a))  =  (a * w_rev + w_rev * a)
    where w_rev is the reversed weight vector.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # picks the needed n entries
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # hyper‑parameters
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.2               # start temperature for soft‑max
    alpha_mul = 1.05
    max_alpha = 800.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = 0.5 * (a + a[::-1])

        a = _project_simplex(a, S)

        # occasional jitter to escape plateaus
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient: 2·a_{k‑i} except on the diagonal where coefficient is 1
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        coeff = np.where(idx_i == idx_j, 1.0, 2.0)
        g = np.zeros_like(a)
        g[idx_i] = coeff * a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = 0.5 * (a + a[::-1])
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.035 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early exit if stuck
        if it % 6000 == 0 and best_score == cur_max:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climbing on the exact max‑convolution
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.025 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = 0.5 * (cand + cand[::-1])
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous best if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 0 – cheap random families / permutations
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.20 * budget_s
    while time.time() < phase0_deadline:
        attempts += 1

        # pick a length (allow a wide range)
        if base_seq is not None and rng.random() < 0.45:
            # up‑sample the previous best
            scale = rng.uniform(0.85, 1.15)
            n = max(600, int(base_seq.shape[0] * scale))
            a = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, base_seq.shape[0]),
                          base_seq)
        else:
            n = int(rng.integers(1800, 12001))

        fam = rng.integers(0, 9)
        if fam == 0:
            a = rng.random(n)                                 # pure uniform random
        elif fam == 1:
            # geometric decay
            r = rng.uniform(0.94, 0.998)
            a = r ** np.arange(n, dtype=float)
        elif fam == 2:
            # power‑law (increasing)
            p = rng.uniform(0.2, 4.0)
            a = (np.arange(1, n + 1, dtype=float) ** p)
        elif fam == 3:
            # power‑law (decreasing)
            p = rng.uniform(0.2, 4.0)
            a = (np.arange(n, 0, -1, dtype=float) ** p)
        elif fam == 4:
            # beta‑like skew
            a = (rng.random(n) ** rng.uniform(0.1, 5.0)) * \
                ((1 - rng.random(n)) ** rng.uniform(0.1, 5.0))
        elif fam == 5:
            # sinusoidal power
            p = rng.uniform(0.5, 4.0)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        elif fam == 6:
            # alternating high‑low pattern built from a random base
            base = rng.random(n)
            base.sort()
            high = base[int(n/2):][::-1]
            low = base[:int(n/2)][::-1]
            a = np.empty(n)
            a[0::2] = high[:((n+1)//2)]
            a[1::2] = low[:(n//2)]
        elif fam == 7:
            # random permutation of a smooth beta shape
            p1 = rng.uniform(0.2, 3.0)
            p2 = rng.uniform(0.2, 3.0)
            x = np.linspace(0, 1, n)
            a = (x ** p1) * ((1 - x) ** p2)
            rng.shuffle(a)
        else:
            # start from a previously known candidate (if any) with random jitter
            if base_seq is not None:
                a = np.interp(np.linspace(0, 1, n),
                              np.linspace(0, 1, base_seq.shape[0]),
                              base_seq)
                # tiny random perturbation
                a += rng.normal(scale=0.01 * a.mean(), size=n)
                a = np.clip(a, 0, None)
            else:
                a = rng.random(n)

        a = np.clip(a, 0.0, None)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, S / n)
        else:
            a = a / total * S

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] attempt={attempts:5d} n={n:5d} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (soft‑max)
    # ------------------------------------------------------------------
    if best_seq is not None:
        now = time.time()
        remaining = deadline - now
        t_adam = now + 0.50 * remaining   # ~ half of remaining time
        a_adam, sc_adam = _optimize_adam(best_seq, t_adam, rng, enforce_sym=False)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
            print(f"[adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Sub‑gradient descent on genuine max
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        t_sub = now + 0.30 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad(best_seq, t_sub, rng, enforce_sym=False)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Permutation annealing (re‑ordering only)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_perm = best_seq.copy()
        cur_score = np.max(_full_convolution(a_perm))
        best_perm = a_perm.copy()
        best_perm_score = cur_score

        T0 = max(cur_score * 0.12, 1e-3)
        T = T0
        cooling = 0.9994
        it = 0

        while time.time() < deadline:
            it += 1
            i, j = rng.integers(0, a_perm.size, size=2)
            if i == j:
                continue
            # swap two entries
            a_new = a_perm.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]

            new_score = np.max(_full_convolution(a_new))
            delta = new_score - cur_score
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                a_perm = a_new
                cur_score = new_score
                if new_score < best_perm_score:
                    best_perm_score = new_score
                    best_perm = a_new.copy()
                    print(f"[perm] iter={it:7d} score={best_perm_score:.6f}")
            T *= cooling
            if it % 8000 == 0:
                T = max(T, T0 * 0.05)

        if best_perm_score < best_score:
            best_score = best_perm_score
            best_seq = best_perm.copy()
            print(f"[perm‑best] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final tiny random‑swap hill‑climbing
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_local, sc_local = _local_swap_search(best_seq, deadline, rng,
                                               enforce_sym=False)
        if sc_local < best_score:
            best_score = sc_local
            best_seq = a_local.copy()
            print(f"[local] final improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result (also expose it globally)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
