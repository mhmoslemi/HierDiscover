"""
A fast simulated‑annealing + sub‑gradient search for the step‑function problem.

The objective (for a sequence a of length n normalised so that
∑a_i = √(2·n)) is simply max_k (a*a)_k, i.e. the largest entry of the
self‑convolution.  This implementation

1. builds a fairly large initial candidate (power‑law shape, optionally
   resized from a previously found solution);
2. runs a brief projected‑gradient descent on a soft‑max approximation
   of the maximum (to quickly get a good basin);
3. performs an extensive SA loop where three specialised moves are used:
   – a *targeted* mass‑transfer that pushes mass away from the indices
     contributing most to the current maximal convolution entry;
   – a *mirror‑averaging* move that equalises a_i and a_{n‑1‑i}, which
     directly reduces the centre of the convolution;
   – the classic “smooth block”, “swap” and “jitter” moves.
   Acceptance follows the Metropolis rule with an exponential cooling
   schedule; the best sequence ever seen is kept.
4. a final short polishing gradient descent (if time permits).

All updates keep the ℓ¹‑norm fixed by Euclidean projection onto the
simplex {x≥0, Σx_i = √(2·n)}.  The routine returns the best list of
floats before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) computed via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (single pass)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  With the normalisation
    sum(a)=√(2·n) this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:          # should never happen – safeguard
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) sequence of length n, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


# ----------------------------------------------------------------------
# Gradient descent on soft‑max of max‑convolution (used as warm‑up)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, conv) where conv is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)                     # soft‑max weights

    # grad = 2 * (a reversed) ⋆ p   (same derivation as in the reference code)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]       # keep the central n entries
    return grad, conv


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max of the max‑convolution.
    The schedule for β and the momentum parameters are tuned for fast
    convergence on the given time budget.
    """
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 32768.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The method combines a short projected‑gradient warm‑up with a
    long simulated‑annealing phase that uses specialised moves
    (targeted mass transfer, mirror averaging, smoothing, swap, jitter).
    The best sequence found before the time budget expires is returned.

    deadline = start + budget_s - 0.5          # safety margin for the caller

    # Choose problem size (larger n gives more freedom for the same budget)
    # ------------------------------------------------------------------

    elif budget_s >= 400:
        n = 15000
    else:

    # ------------------------------------------------------------------

        init = _project_to_simplex(init, target_sum)

    rng.shuffle(init)
    a = _project_to_simplex(init, target_sum)

    best_score = _score(a)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

        if cur_score < best_score:

            best_seq = a.copy()
        print(f"[{time.time():.2f}] After GD warm‑up best score = {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # SA parameters (exponential cooling)
    T_start = 8.0
    T_end = 1e-5
    max_frac_start = 0.40
    max_frac_end = 0.01

    it = 0

    # current convolution & max

    cur_max = conv.max()
    cur_score = cur_max          # because of the normalisation the score == max

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time
        T = T_start * (T_end / T_start) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # recompute convolution only when needed (after an accepted move)
        conv = _conv_fft(a)
        cur_max = conv.max()
        cur_score = cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] SA it={it:6d} new best = {best_score:.6f}")

        k_max = int(conv.argmax())

        # --------------------------------------------------------------
        # Choose a move
        # --------------------------------------------------------------
        move_type = None
        a_old = a.copy()          # keep a backup for possible rejection

        choice = rng.random()

        # ----- targeted mass transfer (most effective) -----
        if choice < 0.48:
            # focus on the indices that directly contribute to the current maximum
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            partner = k_max - idx
            prod = a[idx] * a[partner]                        # contribution of each pair
            w = prod
            if w.sum() > 0.0:
                donor = rng.choice(idx, p=w / w.sum())
                # receiver: pick an index where the contribution is small
                low_mask = prod <= np.percentile(prod, 12)
                if low_mask.any():
                    receiver_candidates = idx[low_mask]
                else:
                    receiver_candidates = np.setdiff1d(np.arange(n), [donor])
                receiver = rng.choice(receiver_candidates)
                if donor != receiver:
                    delta = a[donor] * rng.random() * max_frac
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ----- mirror averaging (reduces centre of convolution) -----
        elif choice < 0.72:
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            if i != j:
                avg = (a[i] + a[j]) * 0.5
                a[i] = avg
                a[j] = avg
                move_type = "mirror"

            start_idx = rng.integers(0, n)
            length = rng.integers(2, 101)          # block length up to 100

                seg = a[start_idx:end_idx]
                avg = seg.mean()
