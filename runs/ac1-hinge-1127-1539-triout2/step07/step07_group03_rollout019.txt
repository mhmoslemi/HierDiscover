"""
Propose a sequence of non‑negative heights that minimizes
the benchmark from evaluate_sequence().
The algorithm proceeds in three phases:

1. **Parametric grid search** –  a simple decreasing power‑law
   a_i = (i + 1 + offset)^{-p} is generated for many (p,offset)
   pairs, each candidate is normalised to the required ℓ¹‑norm
   (∑ a_i = √(2 n)) and scored by the exact max‑convolution
   (FFT).  The best candidate from this cheap search is kept.

2. **Fine‑grained sub‑gradient refinement** –  starting from the
   best candidate we minimise the exact max‑convolution value.
   The active index k* = argmax (a∗a) gives a sub‑gradient
   g_i = 2·a_{k*‑i} (when the index is valid).  A projected
   sub‑gradient descent with a decreasing step‑size is run
   until the time budget expires.

3. **Cleaning & return** –  trailing zeros are stripped,
   the final sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (FFT‑based convolution and objective)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parameterised family (shifted power law) and cheap grid search
# ----------------------------------------------------------------------


def _make_powerlaw_seq(n: int, p: float, offset: float, target_sum: float) -> np.ndarray:
    """
    Create a decreasing sequence a_i = (i+1+offset)^{-p} and scale it
    so that Σ a_i = target_sum.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-p)
    # safety – avoid NaNs if offset is negative and leads to zero denominator
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return a
    a *= target_sum / total
    return a


def _grid_search(n: int, target_sum: float, deadline: float):
    """
    Very cheap exhaustive search over the (p,offset) plane.
    Returns the best sequence found together with its score.
    """
    best_seq = None
    best_val = np.inf
    # search ranges – chosen empirically
    p_vals = np.linspace(0.48, 0.66, 37)          # step ≈ 0.005
    offset_vals = np.linspace(0.0, 30.0, 61)     # step 0.5
    for p in p_vals:
        for off in offset_vals:
            a = _make_powerlaw_seq(n, p, off, target_sum)
            val = _score(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{time.time():.2f}] grid p={p:.3f} off={off:.1f}  score={val:.6f}")
            if time.time() > deadline - 1.0:
                return best_seq, best_val
        if time.time() > deadline - 1.0:
            break
    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_refine(a0: np.ndarray,
                        target_sum: float,
                        deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    The sub‑gradient at the active index k* is
        g_i = 2·a_{k*‑i}  (if the index exists, else 0).
    A diminishing step size is used; the iterate is always kept
    on the simplex.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    iteration = 0
    while time.time() < deadline:
        # full convolution – we need the argmax as well
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # compute sub‑gradient of the active entry
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        j_range = k - i_range
        grad = np.zeros_like(a)
        grad[i_range] = a[j_range]
        grad = 2.0 * grad                     # factor 2 (except the centre term – negligible)

        # simple diminishing step size
        step = 0.5 / (1.0 + 0.0005 * iteration)

        # gradient step, projection back onto the simplex
        a -= step * grad
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] sub‑grad it={iteration:5d}  new best={best_val:.6f}")

        iteration += 1
        # safety – stop after a few thousand iterations
        if iteration > 20000:
            break

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (height sequence) with a low
    benchmark score.  The method first performs a fast grid search over
    a 2‑parameter family (shifted power‑law), then refines the best
    candidate with a few thousand projected sub‑gradient steps.
    The implementation respects the 1000 s time budget.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # a small safety margin

    # problem size – larger n gives a better asymptotic bound,
    # but we keep it modest to stay well inside the time budget.
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 1 – cheap exploration of the (p,offset) family
    # ------------------------------------------------------------------
    a_grid, val_grid = _grid_search(n, target_sum, deadline)
    print(f"[{time.time():.2f}] Grid search best score = {val_grid:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _subgradient_refine(a_grid, target_sum, deadline)
        final_score = _score(a_refined)
        print(f"[{time.time():.2f}] Refinement final score = {final_score:.6f}")
        best_seq = a_refined
    else:
        best_seq = a_grid

    # ------------------------------------------------------------------
    # Clean up – drop trailing near‑zero entries and return a list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Optional final evaluation using the provided benchmark (ignore errors)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq

# ... helper functions: _max_convolution_fft, _conv_fft, _score, _project_to_simplex

def _make_piecewise_powerlaw(n, p1, p2, frac, target_sum):
    """
    Piecewise power‑law:
       a_i = (i+1)^{-p1}            for i < m
       a_i = (i+1)^{-p2}            for i ≥ m
    where m = int(frac * n).
    The sequence is then scaled to have ℓ¹‑norm = target_sum.
    """
    m = max(1, int(frac * n))
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(idx)
    a[:m] = idx[:m] ** (-p1)
    a[m:] = idx[m:] ** (-p2)
    a_sum = a.sum()
    if a_sum == 0.0:
        return a
    a *= target_sum / a_sum
    return a

def _grid_search_piecewise(n, target_sum, deadline):
    best_a = None
    best_val = np.inf
    # parameter ranges – chosen after quick experimentation
    p1_vals = np.linspace(0.35, 0.55, 5)         # 0.35 … 0.55
    p2_vals = np.linspace(0.80, 1.40, 7)         # 0.80 … 1.40
    frac_vals = np.linspace(0.10, 0.50, 9)       # 0.10 … 0.50

        for p2 in p2_vals:

                a = _make_piecewise_powerlaw(n, p1, p2, f, target_sum)

"""
Search for a non‑negative step function (height sequence) with a low
benchmark score.  The method consists of two stages:

1. **Fast grid search** over a three‑parameter family of piecewise
   power‑law sequences
       a_i = (i+1)^{-p1} for i < f·n
       a_i = (i+1)^{-p2} for i ≥ f·n
   The parameters p1, p2 and the split fraction f are explored on a
   coarse grid; each candidate is normalised to the required ℓ¹‑norm
   (∑a_i = √(2 n)) and scored with the exact max‑convolution (FFT).

2. **Projected sub‑gradient refinement** –  starting from the best
   candidate we minimise the exact maximum convolution value.
   The active index k* = argmax (a∗a) yields the sub‑gradient
       g_i = 2·a_{k*‑i}   (if the index is in range, otherwise 0).
   A simple diminishing step size together with projection onto the
   simplex keeps the iterate feasible.

The whole procedure respects a 1000 s time budget and returns the
best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (FFT based convolution and objective)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Benchmark objective: 2·n·max(a*a) / (sum(a)²).
    With the ℓ¹‑norm fixed to √(2·n) the value reduces to max(a*a).
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Piecewise power‑law family
# ----------------------------------------------------------------------


def _make_piecewise_powerlaw(n: int,

                             p2: float,
                             frac: float,

    """
    Produce a decreasing sequence consisting of two power‑law parts:

        a_i = (i+1)^{-p1}   for i < m
        a_i = (i+1)^{-p2}   for i ≥ m

    where m = int(frac * n).  The result is scaled so that
    Σ a_i = target_sum.
    """
    m = max(1, int(frac * n))
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(idx)
    a[:m] = idx[:m] ** (-p1)
    a[m:] = idx[m:] ** (-p2)
    total = a.sum()
    if total == 0.0:
        return a
    a *= target_sum / total
    return a


def _grid_search_piecewise(n: int,
                           target_sum: float,
                           deadline: float):
    """
    Exhaustive scan over a modest grid of (p1, p2, frac) triples.
    Returns the best sequence found together with its score.
    """
    best_a = None
    best_val = np.inf

    # Parameter ranges – chosen after quick manual experimentation.
    p1_vals = np.linspace(0.35, 0.55, 5)          # 0.35 … 0.55
    p2_vals = np.linspace(0.80, 1.40, 7)          # 0.80 … 1.40
    frac_vals = np.linspace(0.10, 0.50, 9)        # 0.10 … 0.50

    for p1 in p1_vals:
        for p2 in p2_vals:
            for f in frac_vals:
                a = _make_piecewise_powerlaw(n, p1, p2, f, target_sum)
                val = _score(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] grid p1={p1:.2f} p2={p2:.2f} f={f:.2f} → {val:.6f}")
                if time.time() > deadline - 1.0:
                    return best_a, best_val
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient refinement (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_refine(a0: np.ndarray,
                        target_sum: float,
                        deadline: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on max(a*a).
    The active index k* = argmax (a*a) gives a sub‑gradient
        g_i = 2·a_{k*‑i}   (if 0 ≤ k*‑i < n, else 0).
    A diminishing step size is employed; each iterate is projected
    back onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        j_range = k - i_range
        grad = np.zeros_like(a)
        grad[i_range] = a[j_range]
        grad = 2.0 * grad                     # factor 2 (the diagonal term is negligible)

        # diminishing step size
        step = 0.8 / (1.0 + 0.001 * it)

        a -= step * grad
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] sub‑grad it={it:5d} → {best_val:.6f}")

        it += 1
        if it > 20000:      # safety guard
            break

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with a low benchmark score.
    The routine adapts the vector length to the allotted time,
    performs a cheap grid search over a piecewise power‑law family,
    and finally refines the best candidate with projected sub‑gradient
    descent.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # small safety margin

    # Choose the problem size based on the available budget.
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 1 – cheap grid search
    # ------------------------------------------------------------------
    a_grid, val_grid = _grid_search_piecewise(n, target_sum, deadline)
    print(f"[{time.time():.2f}] Grid search best score = {val_grid:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _subgradient_refine(a_grid, target_sum, deadline)
        final_score = _score(a_refined)
        print(f"[{time.time():.2f}] Refinement final score = {final_score:.6f}")
        best_seq = a_refined
    else:
        best_seq = a_grid

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zero entries and return a Python list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Optional final check with the provided benchmark (ignore errors)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
