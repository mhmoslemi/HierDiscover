"""
Propose a non‑negative step‑function (list of heights) that minimises
the maximal entry of its self‑convolution (equivalently the
``evaluate_sequence`` value).  The search combines:
  * random sampling of several parametric families (power‑law, exponential,
    Beta, logistic, linear, quadratic);
  * deterministic orderings, especially an “ends” ordering that spreads
    the largest values to opposite ends of the array;
  * fast FFT‑based evaluation of the max‑convolution;
  * elite management;
  * projected gradient descent on a smooth log‑sum‑exp surrogate;
  * permutation‑annealing (swap moves) to improve the ordering.
The routine respects a 1000 s time budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return max entry of the linear convolution a*a (full length 2·n‑1).
    Uses an FFT of the next power‑of‑two length for speed.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s , x≥0   (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, b) where
        b = a*a  (full convolution)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
              = 2·(a ⋆ p) ,  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max
# ----------------------------------------------------------------------
def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e7
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # step size inversely proportional to gradient magnitude and beta
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 8e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta (make surrogate sharper)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a

# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    The ℓ¹‑norm stays unchanged, only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to current best
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # very occasional tiny shuffle to keep diversity
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask].copy()
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best

# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------
def _ends_order(desc: np.ndarray) -> np.ndarray:
    """
    Spread the largest values to opposite ends:
    indices 0, n‑1, 1, n‑2, …
    Input must be sorted in descending order.
    """
    n = desc.size
    out = np.empty_like(desc)
    left, right = 0, n - 1
    idx = 0
    while left <= right:
        out[left] = desc[idx]
        idx += 1
        left += 1
        if left <= right:
            out[right] = desc[idx]
            idx += 1
            right -= 1
    return out


def _alternating_small_large(asc: np.ndarray) -> np.ndarray:
    """
    Small → large → small → large … (starting with the smallest).
    """
    n = asc.size
    out = np.empty_like(asc)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            out[i] = asc[lo]
            lo += 1
        else:
            out[i] = asc[hi]
            hi -= 1
    return out

def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _beta_sequence(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta‑distribution shape on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    t = (i + 0.5) / n                     # centre of each bin in (0,1)
    # use logs to avoid overflow when exponents are < 0
    log_raw = (alpha - 1.0) * np.log(t) + (beta_param - 1.0) * np.log1p(-t)
    raw = np.exp(log_raw)
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)


def _logistic_sequence(n: int, mu: float, sigma: float) -> np.ndarray:
    """Logistic (sigmoid) shape."""
    i = np.arange(n, dtype=float)
    raw = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)
