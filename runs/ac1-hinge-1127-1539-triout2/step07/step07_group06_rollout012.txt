"""
Step‑function optimizer.

The evaluator returns 2·n·max(a*a) / (sum(a)**2).  Since the objective
is homogeneous we can fix Σa = √(2·n) and minimize just the peak of the
self‑convolution.  The algorithm works in three stages:

1.  Fast family‑based search.
    Several parametric families (V‑shape, sum of power‑laws, product of
    distances, symmetric exponentials) are generated for a handful of
    lengths.  Each candidate is scaled to the required ℓ¹‑norm and its
    peak convolution (the objective) is evaluated.  The best few
    candidates are kept.

2.  Incremental local search.
    Starting from the best candidate a fast “mass‑transfer” local search
    (the routine from the reference solution) is run for the remaining
    time budget.  The move set consists of moving a fraction of mass
    from a donor that strongly contributes to the current peak to a
    receiver that contributes little.

3.  Soft‑max gradient polish.
    Finally a projected gradient descent on a smooth approximation of the
    max‑convolution (soft‑max with annealing) is performed to polish the
    solution.

All steps respect the simplex constraint Σa = √(2·n) and non‑negativity.
The function returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _scale_to_target(a: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ¹‑norm equals *target*."""
    total = a.sum()
    if total == 0:
        return np.full_like(a, target / a.size)
    return a * (target / total)


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (assumes Σa = √(2·n))."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Incremental mass‑transfer local search (exact updates)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer an amount `delta` from `donor` to `receiver`.
    Updates both the coefficient vector `a` and the convolution array `conv`
    exactly (including the quadratic `delta**2` terms).
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    # keep a copy for the linear part of the update
    a_before = a.copy()

    # linear contribution:  -2·δ·a_before shifted by donor, +2·δ·a_before shifted by receiver
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic contribution from d*d
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # update the coefficients
    a[donor] -= delta
    a[receiver] += delta

    # numerical safety
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Incremental local search for a fixed length ``n``.
    Returns the best sequence found and its peak convolution value.
    """
    target = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        # random power‑law start
        p = rng.uniform(0.4, 0.9)
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a = _scale_to_target(a, target)

    a = _project_to_simplex(a, target)

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        # annealing schedules
        max_frac = 0.5 * (0.005 / 0.5) ** tau      # up to 0.5 at start, 0.005 at end
        temp = 0.5 * (1e-5 / 0.5) ** tau           # high temperature → low

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass‑transfer move (≈80 % of iterations)
        # --------------------------------------------------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            # contribution of each admissible donor to the peak
            w = a[idx] * a[k_max - idx]
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            # receiver biased towards small “partner” values
            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] new best max={best_max:.6f}")
            else:
                # revert
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing move (≈10 % of iterations)
        # --------------------------------------------------------------
        elif move_type < 0.95:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_window + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) Gaussian jitter (≈10 % of iterations)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved max={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Soft‑max gradient descent (reference implementation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β starts at 1 and is multiplied by 1.5 periodically
    up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric shape families
# ----------------------------------------------------------------------


def shape_vee(n: int, p: float) -> np.ndarray:
    """V‑shaped profile a[i] = (|i‑center|/center) ** p (zero at centre)."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    d = np.abs(idx - center) / center
    return d ** p


def shape_power_sum(n: int, p: float) -> np.ndarray:
    """Sum of two symmetric power‑law tails."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p) + (n - i + 1) ** (-p)


def shape_product_power(n: int, p: float) -> np.ndarray:
    """Product of distances to the two ends raised to a negative power."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return ((i) * (n - i + 1)) ** (-p)


def shape_exponential(n: int, p: float) -> np.ndarray:
    """Symmetric sum of exponentials."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-p * i) + np.exp(-p * (n - 1 - i))
