def _subgradient_descent_opt(a0, N0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * N0)   # same as sqrt2n
    step_base = target_sum * 0.5
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute max conv and index
        b = _full_convolution(a)
        max_b = b.max()
        # index of max (choose any)
        k = int(np.flatnonzero(b == max_b)[0])
        # compute gradient g
        g = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]
        # step size
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (it + 1) ** 0.5)   # decreasing
            a -= lr * g
        # project to simplex
        a = _project_to_simplex(a, target_sum)
        # jitter occasionally
        if it % 100 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        it += 1
    return a

Propose a step‑function minimizing the evaluation objective.
The search consists of many restarts of a simple sub‑gradient descent
on the exact max entry of the self‑convolution (the quantity that
appears in ``evaluate_sequence``), followed by a long simulated‑annealing
hill‑climbing phase that updates the full convolution in O(n) per move.
The time budget (default 1000 s) is split between the two phases and a
final polishing pass.  The implementation uses only NumPy and the
provided ``evaluate_sequence`` function.

import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σ x = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Draw a random non‑negative vector on the ℓ¹‑simplex Σ a = √(2 n)."""
    target = np.sqrt(2.0 * n)
    # pick a Dirichlet sample (uniform on the simplex) and add a small jitter
    a = rng.dirichlet(alpha=np.ones(n)) * target
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent on max (a*a)."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # exact maximum of the self‑convolution and its index
        b = _full_convolution(a)
        max_b = b.max()
        k = int(np.flatnonzero(b == max_b)[0])   # any argmax is fine

        # sub‑gradient of the maximal entry (2 a_{k-i})
        g = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]

        norm_g = np.linalg.norm(g)
        if norm_g > 0.0:
            lr = step_base / (norm_g * (it + 1) ** 0.5)
            a -= lr * g

        # project back onto the simplex (keeps the ℓ¹‑norm fixed)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing hill‑climbing that moves a random amount of mass
    while keeping the full self‑convolution up‑to‑date in O(n) per move.
    Accepts a move if the max entry of the convolution does not increase,
    otherwise with a Metropolis probability.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_max = cur_max
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_max * 0.1 + 1e-9
    cooling = 0.9995

    while time.time() < deadline:
        # pick a worst convolution position
        max_idxs = np.flatnonzero(b == cur_max)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total = contrib.sum()
        if total > 0.0:
            probs = contrib / total
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: bias towards small entries
        if rng.random() < 0.3:
            inv = 1.0 / (a + 1e-12)
            probs_j = inv / inv.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # Metropolis acceptance
        if new_max < cur_max:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_max - cur_max) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_max:.8f}")

        T *= cooling

    return best_a, b

def _evaluate_score(a: np.ndarray, N0: int) -> float:
    """Return the value that evaluate_sequence would produce (lower is better)."""
    max_b = _max_convolution_fft(a)
    nz = np.flatnonzero(a > 1e-12)
    if nz.size:
        n_eff = nz[-1] + 1
    else:
        n_eff = 1
    return n_eff * max_b / N0

    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Fixed problem size – 2000 works well in practice
    # ------------------------------------------------------------------
    n = 2000
    N0 = n
    target_sum = np.sqrt(2.0 * N0)

    # ------------------------------------------------------------------
    # Seed from any previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            if a0.size != n:
                a0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, a0.size), a0)
            a0 = _project_to_simplex(a0, target_sum)
            best_seq = a0.copy()
            best_val = _evaluate_score(best_seq, N0)
            print(f"[{time.time():.1f}] seeded from height_sequence_1, score={best_val:.6f}")
        except Exception as e:
            print(f"seed error: {e}")
            best_seq = _random_initial_sequence(n, rng)
            best_val = np.inf
    else:
        best_seq = _random_initial_sequence(n, rng)
        best_val = np.inf

    # ------------------------------------------------------------------
    # Main optimisation loop – many restarts of sub‑gradient descent
    # ------------------------------------------------------------------
    while time.time() < deadline - 60.0:        # reserve ~1 min for final polishing
        now = time.time()
        remaining = deadline - now - 30.0    # keep 30 s for a quick hill‑climb
        if remaining <= 5.0:
            break

        # --- sub‑gradient descent run ---------------------------------
        run_time = min(remaining * 0.5, 120.0)   # at most two minutes per run
        a_start = _random_initial_sequence(n, rng)
        a_opt = _subgradient_descent_opt(a_start, now + run_time)
        val = _evaluate_score(a_opt, N0)
        if val < best_val:
            best_val = val
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] sub‑grad improvement: {best_val:.6f}")

        # --- short hill‑climbing refinement ---------------------------
        now = time.time()
        remaining = deadline - now - 10.0    # keep 10 s for final polishing
        if remaining <= 5.0:
            break
        hc_time = min(remaining * 0.5, 40.0)  # up to 40 s
        a_cur = best_seq.copy()
        b_cur = _full_convolution(a_cur)
        a_hc, b_hc = _hill_climbing(a_cur, b_cur, now + hc_time)
        val_hc = _evaluate_score(a_hc, N0)
        if val_hc < best_val:
            best_val = val_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] hill‑climb refinement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing: a longer sub‑gradient run + short hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        now = time.time()
        a_final = _project_to_simplex(best_seq.copy(), target_sum)
        a_final = _subgradient_descent_opt(a_final, deadline - 10.0)
        b_final = _full_convolution(a_final)
        a_final, _ = _hill_climbing(a_final, b_final, deadline - 2.0)
        best_val = _evaluate_score(a_final, N0)
        best_seq = a_final.copy()
        print(f"[{time.time():.1f}] final polishing: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    nz = np.flatnonzero(best_seq > 1e-12)
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (optional – helpful for debugging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
