def _adam_softmax_descent(a0, deadline):
    a = a0.astype(np.float64).copy()
    n = a.size
    target = np.sqrt(2.0*n)
    beta = 1.0
    beta_max = 8192.0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_base = target * 0.1
    it = 0
    best_a = a.copy()
    best_val = _score(a)
    while time.time() < deadline:
        # schedule: gradually increase beta
        tau = (it+1) / 10000.0
        if beta < beta_max and it % 500 == 0:
            beta = min(beta * 1.5, beta_max)
        # compute gradient
        grad, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        # step size adapt: scale inversely with sqrt(v_hat) maybe
        lr = step_base / (np.sqrt(v_hat) + eps)
        # apply update
        a = a - lr * m_hat
        # enforce nonnegativity and sum
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)
        # optional symmetrization periodically
        if it % 2000 == 0:
            a = (a + a[::-1]) / 2.0
            a = _project_to_simplex(a, target)
        # jitter occasionally
        if it % 500 == 0:
            scale = target * 0.001 / (beta + 1.0)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
        # compute score (max convolution)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam β={beta:.1f} best={best_val:.6f}")
        it += 1
    return best_a, best_val

def _local_search(n, deadline, rng, init_seq=None):
    target = np.sqrt(2.0*n)
    # init a
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target)
    a = _project_to_simplex(a, target)
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    move_counter = 0
    start = time.time()
    total_time = deadline - start
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time
        # schedule for max fraction and temperature
        max_frac = 0.4 * (0.02 / 0.4) ** tau   # initial max fraction ~0.4 decays to 0.02
        temp = 0.4 * (1e-5 / 0.4) ** tau
        move_type = rng.random()
        if move_type < 0.70:
            # mass transfer (most frequent)
            k_max = int(np.argmax(conv))
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi+1)
            w = a[idx] * a[k_max - idx]
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w/total_w)
            else:
                donor = rng.integers(n)
            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv/inv.sum())
            if receiver == donor:
                continue
            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue
            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] new best max={best_max:.6f}")
            else:
                # revert
                _apply_transfer(a, conv, receiver, donor, delta)
        elif move_type < 0.85:
            # smoothing window
            max_window = min(200, n // 5)
            start_win = rng.integers(0, n)
            length = rng.integers(2, max_window+1)
            end_win = min(start_win + length, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()
                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True
                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)
        elif move_type < 0.95:
            # swap move
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                a[i], a[j] = a[j], a[i]

                new_max = conv.max()
                accept = False
                if new_max <= cur_max:
                    accept = True

                        accept = True

                        best_a = a.copy()
                        print(f"[{time.time():.1f}] swap improved max={best_max:.6f}")

                    a[i], a[j] = a[j], a[i]

    a = a0.copy()

    target = np.sqrt(2.0 * n)

    beta_max = 8192.0

    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_scale = target * 0.2
    best_a = a.copy()

    last_beta_increase = time.time()
    while time.time() < deadline:

"""
Propose a step‑function (non‑negative heights) that minimizes
evaluate_sequence(seq) = 2·n·max(convolution(seq,seq)) / (sum(seq)²).

The search consists of three stages:

1.  **Warm‑start** – if a previous construction (`height_sequence_1`)
    is available it is refined with a short local search.

2.  **Multi‑scale local search** – for several problem sizes we run a
    simulated‑annealing style optimizer that repeatedly transfers mass
    between indices, applies smoothing, swaps values and adds jitter.
    The convolution is updated incrementally for cheap moves; a symmetry
    projection (a ← (a+rev(a))/2) is applied periodically.

3.  **Adam‑softmax polishing** – the best candidate found is refined by
    projected Adam optimisation on a soft‑max approximation of the
    maximal convolution entry.  The temperature β is annealed up to a
    large value, the iterate stays on the simplex Σa = √(2·n) and is
    symmetrised every few hundred steps.

All intermediate vectors are projected onto the simplex
{a ≥ 0, Σa = √(2·n)} so the objective value is exactly the peak of the
self‑convolution.  The routine respects the 1000 s time budget and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Basic sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and renorm to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (assumes Σa = √(2·n))."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Fast objective value (the maximal convolution entry)."""
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Incremental mass‑transfer update (exact)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer an amount `delta` from `donor` to `receiver`.
    Updates the coefficient vector `a` and the convolution array `conv`
    exactly (including quadratic terms).
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    # keep a copy of the current coefficients for the linear part
    a_before = a.copy()

    # linear terms   –2·δ·a_before shifted by donor, +2·δ·a_before shifted by receiver
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic contributions
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # update coefficients
    a[donor] -= delta
    a[receiver] += delta

    # numerical safety
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Gradient of a soft‑max approximation of max(convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where grad is the gradient of
    softmax_beta(max_i (a*a)_i) w.r.t. a, and conv is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # convolution via FFT
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # length 2n‑1

    # gradient = 2 * (a reversed * p)   (only the central n entries matter)
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]   # central n entries
    return g, b


# ----------------------------------------------------------------------
# Simulated‑annealing style local search
# ----------------------------------------------------------------------


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Incremental local optimisation for a fixed length `n`.
    Returns the best sequence and its maximal convolution value.
    """
    target = np.sqrt(2.0 * n)

    # ----- initialise ----------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        # a random power‑law start
        p = rng.uniform(0.3, 0.9)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start
    move_counter = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        # schedules (warm → cold)
        max_frac = 0.4 * (0.02 / 0.4) ** tau          # max fraction of donor mass to move
        temp = 0.4 * (1e-5 / 0.4) ** tau               # temperature for Metropolis test

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass transfer (≈70 % of the moves)
        # --------------------------------------------------------------
        if move_type < 0.70:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]               # contribution of each index
            total_w = w.sum()
            donor = rng.choice(idx, p=w / total_w) if total_w > 0 else rng.integers(n)

            # bias receiver towards small coefficients
            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            # Metropolis acceptance
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] new best max={best_max:.6f}")
            else:
                # revert the move
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing window (≈10 %)
        # --------------------------------------------------------------
        elif move_type < 0.85:
            max_window = min(200, n // 5)
            start_win = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end_win = min(start_win + length, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()
                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True
                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) swap two entries (≈5 %)
        # --------------------------------------------------------------
        elif move_type < 0.95:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                a[i], a[j] = a[j], a[i]
                conv = _conv_fft(a)
                new_max = conv.max()
                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True
                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] swap improved max={best_max:.6f}")
                else:
                    # revert swap
                    a[i], a[j] = a[j], a[i]
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 4) Gaussian jitter (≈15 %)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved max={best_max:.6f}")

        move_counter += 1
        # occasional symmetry projection (helps flatten the convolution)
        if move_counter % 2000 == 0:
            a = (a + a[::-1]) / 2.0
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()

    return best_a, best_max


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_softmax_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on a soft‑max approximation of the max
    convolution.  The temperature β grows up to 8192.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    beta = 1.0
    beta_max = 8192.0

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_scale = target * 0.2

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        # increase temperature periodically
        if it % 500 == 0 and beta < beta_max:
            beta = min(beta * 1.5, beta_max)

        # gradient of the soft‑max surrogate
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        lr = step_scale / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

        # occasional symmetrisation
        if it % 1000 == 0:
            a = (a + a[::-1]) / 2.0
            a = _project_to_simplex(a, target)

        # occasional small jitter to escape plateaus
        if it % 300 == 0:
            scale = target * 0.001 / (beta + 1.0)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam β={beta:.1f} new best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the outer harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The routine runs a series of local searches on several sizes and
    finishes with a short Adam‑based soft‑max polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.size
        # a quick refinement (≈30 s)
        if time.time() < deadline - 5.0:
            run_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.20)
            seq, score = _local_search(n0, run_deadline, rng, init_seq=seq0)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] warm‑start size={n0} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Explore several problem sizes with independent searches
    # ------------------------------------------------------------------
    sizes = [500, 1000, 1500, 2000, 2500, 3000, 4000, 6000]
    for n in sizes:
        if time.time() >= deadline - 5.0:
            break

        remaining = deadline - time.time()
        # allocate at most 150 s per size, but keep a fraction of the remaining budget
        time_for_this = min(remaining * 0.18, 150.0)

        run_deadline = time.time() + time_for_this

        # random initialisation: mix power‑law and Dirichlet draws
        if rng.random() < 0.4:
            p = rng.uniform(0.2, 1.0)
            init = _power_law_sequence(n, p, np.sqrt(2.0 * n))
        else:
            alpha = rng.uniform(0.1, 4.0)
            init = rng.dirichlet(np.full(n, alpha)) * np.sqrt(2.0 * n)

        seq, score = _local_search(n, run_deadline, rng, init_seq=init)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} → new global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final Adam‑softmax polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra_deadline = time.time() + min(60.0, deadline - time.time())
        best_seq, best_score = _adam_softmax_descent(best_seq, extra_deadline)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 4) Trim trailing zeros – they do not affect the objective
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
