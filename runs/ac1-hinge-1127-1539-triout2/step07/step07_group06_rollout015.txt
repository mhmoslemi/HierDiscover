"""
Hybrid linear‑programming + local‑search optimizer for the step‑function
bound.  The objective after normalising Σa_i = √(2·n) is simply
max_k (a*a)[k].  The algorithm proceeds as follows:

1. Initialise a non‑negative vector a of length n and project it onto the
   simplex {a ≥ 0, Σa = √(2·n)} (power‑law + jitter).
2. Repeatedly improve a by solving a linear programme:
      maximise   Σ g_j
      s.t.       (a * g)[k] ≤ M   for all k = 0,…,2n‑2
                 g_j ≥ 0
   where M = max(a*a).  The feasible set is convex and contains a.
   The optimiser pushes mass towards indices that are weakly constrained
   (typically the tails).  After the LP we scale g to have the correct
   ℓ¹‑norm and replace a by the best convex combination (1‑t)a + t g.
   The loop stops when no improvement is found or the time budget expires.
3. Run the fast incremental local‑search (mass‑transfer + smoothing +
   jitter) from the best vector obtained above.
4. Polish the result with a projected gradient descent on a soft‑max
   approximation of the max‑convolution.
5. Strip trailing zeros and return the sequence as a ``list[float]``.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (assumes Σa = √(2·n))."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Fast objective value – after normalisation this equals max convolution."""
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Incremental mass‑transfer update (exact)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer an amount `delta` from `donor` to `receiver`.
    Updates both the coefficient vector `a` and the convolution array `conv`
    exactly (including the quadratic `delta**2` terms).
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    a_before = a.copy()

    # linear contribution:  -2·δ·a_before shifted by donor,
    #                     +2·δ·a_before shifted by receiver
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic contribution
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # update coefficients
    a[donor] -= delta
    a[receiver] += delta

    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Linear‑programming helper (build the Toeplitz convolution matrix)
# ----------------------------------------------------------------------


def _build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Construct the sparse matrix T of shape (2·n‑1, n) that satisfies
    T @ g = conv(f, g).
    Row k, column j contains f[k‑j] if 0 ≤ k‑j < n, else 0.
    """
    n = f.size
    # each diagonal offset i (0 ≤ i < n) gets the constant value f[i]
    diags = [np.full(n, f_i, dtype=np.float64) for f_i in f]
    offsets = np.arange(n, dtype=int)
    T = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    return T


# ----------------------------------------------------------------------
# Local‑search core (unchanged from the reference implementation)
# ----------------------------------------------------------------------


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Incremental local search for a given length `n`.
    Returns the best sequence and its max‑convolution value.
    """
    target = np.sqrt(2.0 * n)

    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        max_frac = 0.5 * (0.005 / 0.5) ** tau          # max fraction of donor mass
        temp = 0.5 * (1e-5 / 0.5) ** tau               # Metropolis temperature

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass transfer (≈80 % of the moves)
        # --------------------------------------------------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]               # contribution of each index
            total_w = w.sum()
            donor = rng.choice(idx, p=w / total_w) if total_w > 0 else rng.integers(n)

            # bias receiver towards small coefficients
            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            # Metropolis acceptance
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] new best max={best_max:.6f}")
            else:
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing window (≈10 % of the moves)
        # --------------------------------------------------------------
        elif move_type < 0.95:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_window + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) Gaussian jitter (≈10 % of the moves)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved max={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Projected gradient descent on a soft‑max approximation (reference code)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function bound.
    Uses a few LP iterations to push mass to the tails, then an
    aggressive incremental local search and a final gradient‑descent polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Main optimisation loop (allow a few independent restarts)
    # ------------------------------------------------------------------
    attempts = 3
    for attempt in range(attempts):
        if time.time() >= deadline - 5.0:
            break

        # problem size – a compromise between expressivity and LP cost
        n = 2000
        target_sum = np.sqrt(2.0 * n)

        # ---- initialise ------------------------------------------------
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target_sum)
        # small random jitter to diversify restarts
        a += rng.normal(scale=target_sum * 0.02, size=n)
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # current peak of the self‑convolution
        M = _max_convolution(a)

        # --------------------------------------------------------------
        # 2) LP‑based mass relocation (few iterations)
        # --------------------------------------------------------------
        max_lp_iters = 5
        for lp_it in range(max_lp_iters):
            if time.time() >= deadline - 5.0:
                break

            # build Toeplitz convolution matrix for the current a
            T = _build_constraint_matrix(a)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)   # maximise Σg_j  <=> minimise -Σg_j

            remaining = deadline - time.time()
            lp_time_limit = max(0.5, min(30.0, remaining * 0.5))

            res = linprog(c,
                          A_ub=T,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': lp_time_limit})

            if not res.success or res.x is None:
                break

            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 0:
                break

            # Scale to the correct ℓ¹‑norm
            g = g0 * (target_sum / sum_g0)

            # Try a few convex‑combination weights and keep the best
            best_local_f = a
            best_local_M = M
            for t in (0.2, 0.5, 0.8, 1.0):
                candidate = (1.0 - t) * a + t * g
                cand_M = _max_convolution(candidate)
                if cand_M < best_local_M - 1e-9:
                    best_local_M = cand_M
                    best_local_f = candidate.copy()

            # accept if we made progress
            if best_local_M < M - 1e-9:
                a = best_local_f
                M = best_local_M
            else:
                break   # LP can no longer improve the peak

        # --------------------------------------------------------------
        # 3) Fine‑grained local search from the LP‑enhanced start
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            seq, score = _local_search(n, deadline, rng, init_seq=a)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] attempt {attempt} new best max={best_score:.6f}")

        # --------------------------------------------------------------
        # 4) Final gradient‑descent polish (if time permits)
        # --------------------------------------------------------------
        if best_seq is not None and time.time() < deadline - 5.0:
            extra_deadline = time.time() + (deadline - time.time()) * 0.25
            best_seq, _ = _annealed_gradient_descent(best_seq, extra_deadline)
            best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # 5) Strip trailing zeros and return the candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        best_seq = np.array(best_seq, dtype=np.float64)
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=np.float64)

    print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
