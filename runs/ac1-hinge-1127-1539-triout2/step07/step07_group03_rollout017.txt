    A = np.zeros((len(active_set), n), dtype=np.float64)
    for idx, k in enumerate(active_set):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # for j in lo..hi: g_j coefficient = f_{k - j}
        # so j ranges, we fill A[idx, lo:hi+1] = f[k - lo: k - hi :-1]? Let's compute reversed.
        # Actually for j from lo to hi, f_{k - j} = f[ k - np.arange(lo, hi+1) ] = reversed slice of f.

        A[idx, lo:hi+1] = f[k - j_range]
    b = np.full(len(active_set), M)   # Upper bound: <= M
    # objective: maximize sum(g) = - minimize -sum(g)
    c = -np.ones(n)   # linprog does min c^T x
    bounds = [(0, None)] * n
    # Solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        return g

"""
A hybrid optimizer for the step‑function benchmark.

The search consists of four stages:

1. **Adam‑style projected gradient descent** on the soft‑max
   approximation of the max‑convolution.  This quickly drives the
   population toward a low‑peak shape while keeping the ℓ¹ norm fixed.

2. **Active‑set linear‑programming refinement**.
   For the current sequence `a` we compute the maximal convolution value
   `M = max(a*a)`.  Using only the `K` largest convolution indices we
   solve a small LP
        maximise   Σ g_i
        s.t.      (a * g)_k ≤ M   (k in the active set)
                  0 ≤ g_i ≤ Σa          (ℓ¹‑bound)
   The solution `g` is scaled to the target ℓ¹‑norm and mixed with `a`
   (`a ← (1‑t)·a + t·g`).  Repeating this step monotonically reduces the
   peak.

3. **Simulated annealing** with mass‑transfer, smoothing and swaps,
   starting from the best candidate of the previous phases.

4. **Final polishing** – a short Adam run, if time permits.

All updates keep the sum of the coefficients equal to `√(2·n)` by
projecting onto the simplex.  The implementation uses only NumPy and
SciPy (HiGHS LP solver) and respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalized to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (beta‑smooth approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where ``grad`` has the same shape as ``a``.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """Projected Adam optimisation on the soft‑max surrogate."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # hyper‑parameters
    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step_base / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Active‑set linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_refinement(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   active_k: int = 10,
                   mix_t: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a tiny LP:
        maximise Σ g_i
        s.t. (a * g)_k ≤ M   for the `active_k` largest conv‑entries,
             0 ≤ g_i ≤ target_sum.
    The solution is scaled to the target ℓ¹‑norm and mixed with the
    current vector (convex combination with weight ``mix_t``).
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()

        # pick the K largest indices (might contain duplicates)
        if active_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -active_k)[-active_k:]

        K = len(idxs)
        # Build A @ g ≤ M
        A = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(idxs):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # indices of g that appear in this constraint
            j = np.arange(lo, hi + 1, dtype=np.int64)
            # coefficient = a_{k-j}
            A[row, lo:hi + 1] = a[k - j]

        b = np.full(K, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  → minimise -Σ g_i
        bounds = [(0.0, target_sum) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        except Exception:  # pragma: no cover
            break

        if not res.success:
            # LP could become unbounded or infeasible for the tiny active set
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        # Scale to the exact ℓ¹‑norm required by the benchmark
        g = g * (target_sum / sum_g)

        # Convex mixing
        a = (1.0 - mix_t) * a + mix_t * g
        a = _project_to_simplex(a, target_sum)   # numerical clean‑up

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP mix_t={mix_t:.2f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (kept from the reference implementation)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 10.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.5,
                         max_frac_end: float = 0.01,
                         max_window: int = 100) -> tuple[np.ndarray, float]:
    """Heavy‑weight annealing – mass transfers, smoothing, swaps."""
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # location of the current maximum
        k_max = int(np.argmax(conv))

        # sensitivity of each coordinate to the peak (linearised)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # ----------- mass transfer -----------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            donor = rng.choice(n, p=weight / total_w) if total_w > 0 else rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0.0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ----------- local smoothing ---------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ----------- swap -------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # new convolution & score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis test
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0.0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a low‑score height sequence within the given time budget."""
    start = time.time()
    deadline = start + budget_s - 0.5      # tiny safety margin

    # problem size – larger n gives more freedom but needs more time
    if budget_s >= 900:
        n = 16000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        init = _project_to_simplex(init, target_sum)
    else:
        init = _power_law_sequence(n, 0.50, target_sum)

    # ------------------------------------------------------------------
    # Phase 1 – Adam gradient descent (soft‑max)
    # ------------------------------------------------------------------
    gd_deadline = time.time() + 0.35 * (deadline - time.time())
    a_gd, val_gd = _adam_gradient_descent(init, target_sum, gd_deadline)
    print(f"[{time.time():.2f}] Phase 1 (Adam) best score={val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – linear‑programming refinement
    # ------------------------------------------------------------------
    lp_deadline = time.time() + 0.15 * (deadline - time.time())
    a_lp, val_lp = _lp_refinement(a_gd, target_sum, lp_deadline,
                                  active_k=12, mix_t=0.1)
    print(f"[{time.time():.2f}] Phase 2 (LP) best score={val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing on the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(a_lp, deadline,
                                            start_temp=10.0,
                                            end_temp=1e-5,
                                            max_frac_start=0.5,
                                            max_frac_end=0.01,
                                            max_window=100)
        # pick the best of the three
        candidates = [(a_gd, val_gd), (a_lp, val_lp), (a_sa, val_sa)]
        best_seq, best_score = min(candidates, key=lambda x: x[1])
        print(f"[{time.time():.2f}] Phase 3 (SA) best score={best_score:.6f}")
    else:
        best_seq, best_score = a_lp, val_lp

    # ------------------------------------------------------------------
    # Phase 4 – final short Adam polish (if we still have a couple seconds)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_final, val_final = _adam_gradient_descent(best_seq,
                                                    target_sum,
                                                    deadline)
        if val_final < best_score:
            best_seq = a_final
            best_score = val_final
            print(f"[{time.time():.2f}] Final polish improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    # optional final benchmark evaluation
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
