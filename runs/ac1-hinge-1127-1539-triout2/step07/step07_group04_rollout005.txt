"""
Propose a step‑function minimizing ``evaluate_sequence``.

Approach
--------
* **Stage‑wise length schedule** – we start with modest lengths and
  increase up to 40 k, re‑using the best solution found so far.
* **Differential Evolution (DE)** – a small population evolves on the
  ℓ₁‑simplex using mutation/ crossover/ projection.  DE is very good at
  escaping local minima and quickly discovers diverse high‑quality
  candidates.
* **Projected gradient descent** – a smooth log‑sum‑exp surrogate of the
  worst autocorrelation entry is minimised with an Adam‑like update.
  The inverse temperature β is annealed to 10⁵, focussing on the true
  maximum in the final phase.
* **Exact mass‑transfer hill climbing** – after the DE/gradient phases
  we polish the current best vector with the O(n) update routine from the
  previous state‑of‑the‑art.  The move size is chosen analytically to
  reduce the current peak as much as possible.
* **Final polishing** – any remaining budget is spent on a high‑β
  gradient descent.

All vectors stay non‑negative and are projected onto the simplex
{aᵢ≥0, Σaᵢ=√(2 n)} after every move, so the official evaluator sees a
valid step function.  The routine respects a wall‑clock budget (default
1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (various families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled (power‑law, exponential, Dirichlet,
    sine, Gaussian, piecewise) to diversify the search space.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine wave (positive half‑wave)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its Adam‑style gradient descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p), p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam‑style gradient descent on the smooth max (log‑sum‑exp
    surrogate).  An exponential schedule for the inverse temperature β
    drives the optimisation towards the true L∞‑norm.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 0.5            # start temperature
    beta_max = 1e5
    beta_factor = 1.3

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = 0.5 * target_sum / (np.sqrt(v_hat) + eps)  # base step size
        a -= step * m_hat

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 150 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that transfers mass from a
    source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9
    cooling = 0.9994

    while time.time() < deadline:
        # ---- pick a worst convolution index ------------------------------------
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # ---- source index weighted by contribution to entry M ------------------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # ---- destination index ------------------------------------------------
        if rng.random() < 0.4:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # ---- amount of mass to move -----------------------------------------
        max_delta = a[i] * 0.85
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- update convolution (linear terms) --------------------------------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # ---- quadratic correction terms ---------------------------------------
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---- Metropolis acceptance --------------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Differential Evolution on the simplex
# ----------------------------------------------------------------------


def _differential_evolution_opt(a0: np.ndarray,
                               deadline: float,
                               pop_size: int = 12,
                               F: float = 0.5,
                               Cr: float = 0.9) -> np.ndarray:
    """
    Simple DE optimiser constrained to the ℓ₁‑simplex.
    Uses a mix of ``rand/1`` and ``best/1`` mutation strategies.
    """
    n = a0.size
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()

    # initialise population (include a0)
    pop = [a0.copy()] + [_random_initial_sequence(n, rng) for _ in range(pop_size - 1)]
    pop = [_project_to_simplex(p, target_sum) for p in pop]
    scores = [_max_convolution_fft(p) for p in pop]

    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            if time.time() >= deadline:
                break

            # choose three distinct individuals (excluding i)
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)

            if rng.random() < 0.5:               # rand/1
                mutant = pop[r1] + F * (pop[r2] - pop[r3])
            else:                                # best/1
                mutant = best + F * (pop[r1] - pop[r2])

            mutant = np.maximum(mutant, 0.0)
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover
            mask = rng.random(n) < Cr
            if not np.any(mask):
                mask[rng.integers(n)] = True
            trial = np.where(mask, mutant, pop[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _max_convolution_fft(trial)

            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best = trial.copy()
                    best_score = trial_score
                    print(f"[{time.time():.1f}] DE improvement: {best_score:.6f}")

    return best

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best
    sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seeding from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – increasingly larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase 1 – Differential Evolution
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 6.0:
            break
        de_budget = min(remaining * 0.30, 180.0)          # up to ~3 min
        de_deadline = time.time() + de_budget
        a_de = _differential_evolution_opt(a_init, de_deadline,
                                           pop_size=12, F=0.5, Cr=0.9)
        score_de = _max_convolution_fft(a_de)
        if score_de < best_score:
            best_seq = a_de.copy()
            best_score = score_de
            print(f"[{time.time():.1f}] DE stage improvement (n={n}) score={score_de:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Hill climbing (exact convolution updates)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 6.0:
            break
        hill_budget = min(remaining * 0.30, 200.0)
        hill_deadline = time.time() + hill_budget

        # ensure we work with the current length
        if best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = best_seq.copy()
        a_current = _project_to_simplex(a_current, target_sum)
        b_full = _full_convolution(a_current)

        a_hc, b_hc = _hill_climbing(a_current, b_full, hill_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_seq = a_hc.copy()
            best_score = score_hc
            print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Gradient descent refinement
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 6.0:
            gd_deadline = time.time() + min(remaining * 0.30, 120.0)
            a_gd = _gradient_descent_opt(best_seq.copy(), gd_deadline)
            score_gd = _max_convolution_fft(a_gd)
            if score_gd < best_score:
                best_seq = a_gd.copy()
                best_score = score_gd
                print(f"[{time.time():.1f}] Gradient refinement (n={n}) score={score_gd:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        a_final = _gradient_descent_opt(a, deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_seq = a_final.copy()
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros (nice output)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
