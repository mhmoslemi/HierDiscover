def lp_step(f, S, t, M=None):
    if M is None: M = max_convolution_fft(f)
    n = f.shape[0]
    # Build constraint matrix A (2n-1 x n)
    # We'll use sparse matrix from scipy.sparse
    row = []
    col = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            row.append(k)
            col.append(i)
            data.append(f[i])  # coefficient f[i]
    A = scipy.sparse.csr_matrix((data, (row, col)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    res = scipy.optimize.linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success: return f
    g_opt = res.x
    sum_g = g_opt.sum()
    if sum_g == 0: return f
    alpha = S / sum_g
    g = alpha * g_opt
    new_f = (1 - t) * f + t * g
    return new_f

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, scipy.sparse, scipy.optimize

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # initialize length schedule
    schedule = [500, 1000, 2000, 4000]

    best_seq = None
    best_score = np.inf

    # If height_sequence_1 exists, seed; else uniform.

    for n in schedule:
        if time.time() >= deadline: break
        # target sum
        S = np.sqrt(2 * n)

        # initialize f
        if best_seq is None:
            f = np.full(n, S / n)
        else:
            # upsample/downsample best_seq to new n
            f = _upsample_sequence(best_seq, n)

        # Ensure sum = S
        f = _project_to_simplex(f, S)   # use earlier function

        # iterative LP refinement
        max_iter = 10   # maybe adapt based on time
        t = 0.5         # mixing parameter

        for it in range(max_iter):
            if time.time() >= deadline: break
            # compute max conv
            M = _max_convolution_fft(f)
            # build matrix once per iteration (depends on f)
            A = _conv_matrix_sparse(f)   # shape (2n-1, n)
            # solve LP
            c = -np.ones(n)  # maximize sum => minimize negative sum
            b = np.full(2 * n - 1, M)
            # Solve with highs
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
            if not res.success:
                print(f"[{time.time():.1f}] LP failed: {res.message}")
                break
            g_opt = res.x
            sum_g = g_opt.sum()
            if sum_g <= 1e-12: break
            # scale to keep total sum S
            g_scaled = g_opt * (S / sum_g)
            # mixing
            f_new = (1 - t) * f + t * g_scaled
            # optional small jitter
            if it % 2 == 0:
                f_new = _project_to_simplex(f_new, S)   # keep normalization
            f = f_new

        # evaluate sequence after LP iterations
        seq = f.tolist()
        try:
            val = evaluate_sequence(seq)
        except Exception as exc:
            val = np.inf

"""
Propose a step‑function that (approximately) minimizes
``evaluate_sequence``.  The method combines three ideas:

1.  Start from the best known construction (the global variable
    ``height_sequence_1``) or from a uniform distribution.
2.  Repeatedly solve a *linear programme*:
        maximise Σ g_i  s.t.   g_i ≥ 0  and  (f * g)_k ≤ M   for every shift k,
    where ``f`` is the current candidate, ``M = max(f*f)``.
    The LP is feasible because all entries of ``f`` are kept strictly
    positive, and the solution ``g`` typically has a larger ℓ¹‑norm.
    After rescaling ``g`` to the required total mass the new candidate is
    a convex combination ``f ← (1‑t)·f + t·g`` (t≈0.5).  Each iteration
    strictly reduces the maximal convolution value.
3.  Finish with a short simulated‑annealing hill‑climb that updates the
    full convolution in O(n) per move.

The whole search respects the 1000 s wall‑clock budget, uses only NumPy,
SciPy (``linprog``) and the supplied ``evaluate_sequence`` function,
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (fast via FFT)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by
    projection onto the simplex {x ≥ 0, Σ x = √(2·new_n)}.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Construction of the dense convolution matrix (f * g).
# ----------------------------------------------------------------------


def _conv_matrix_dense(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2*n‑1, n)) such that
        A @ g = f * g   (full linear convolution).
    For column j the vector f is inserted starting at row j.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (same as in the previous
# baseline, kept here for the final polishing phase).
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of
    mass from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9
    cooling = 0.9995

    while time.time() < deadline:
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination bias toward small entries
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution efficiently
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    import scipy.optimize

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Seed: either the previously stored construction or a uniform start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            print(f"[{time.time():.1f}] Using height_sequence_1 as seed (len={a0.size})")
        except Exception as e:
            print(f"Seed extraction error: {e}")
            a0 = None
    else:
        a0 = None

    if a0 is None or a0.size == 0:
        n0 = 2000
        a0 = np.full(n0, np.sqrt(2.0 * n0) / n0, dtype=float)

    # ------------------------------------------------------------------
    # Schedule of lengths (start with the seed length, then optionally double)
    # ------------------------------------------------------------------
    schedule = [a0.size]
    if a0.size * 2 <= 3000:
        schedule.append(a0.size * 2)

    best_seq = None
    best_score = np.inf

    for n in schedule:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            f = _upsample_sequence(best_seq, n)
        else:
            f = _upsample_sequence(a0, n)

        # --------------------------------------------------------------
        # Ensure strict positivity (required for the LP) and normalise
        # --------------------------------------------------------------
        eps = 1e-9
        f = np.maximum(f, eps)
        f = f * (target_sum / f.sum())

        # --------------------------------------------------------------
        # Main LP‑based refinement loop
        # --------------------------------------------------------------
        it = 0
        max_lp_iters = 30
        t_mix = 0.5                # mixing factor between old f and new g

        while time.time() < deadline and it < max_lp_iters:
            it += 1

            # current maximal convolution entry
            M = _max_convolution_fft(f)

            # build dense convolution matrix A such that A @ g = f * g
            A = _conv_matrix_dense(f)          # shape (2n‑1, n)

            # linear programme: maximise Σ g_i  subject to A g ≤ M, g ≥ 0
            c = -np.ones(n)                     # linprog minimizes, so negate
            b_ub = np.full(2 * n - 1, M)

            try:
                res = scipy.optimize.linprog(
                    c,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=[(0, None)] * n,
                    method="highs",
                )
            except Exception as exc:
                print(f"[{time.time():.1f}] linprog exception: {exc}")
                break

            if not res.success:
                print(f"[{time.time():.1f}] LP failed (iter {it}): {res.message}")
                break

            g_opt = res.x
            sum_g = g_opt.sum()
            if sum_g <= 0:
                print(f"[{time.time():.1f}] LP returned zero sum, aborting.")
                break

            # scale g back to the required ℓ¹‑norm and mix
            g = g_opt * (target_sum / sum_g)
            f = (1.0 - t_mix) * f + t_mix * g

            # keep a small jitter to avoid stagnation
            if it % 5 == 0:
                jitter = rng.normal(scale=target_sum * 1e-6, size=n)
                f += jitter
                f = np.maximum(f, eps)
                f = f * (target_sum / f.sum())

            # monitor improvement
            cur_score = _max_convolution_fft(f)
            if cur_score < best_score - 1e-5:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[{time.time():.1f}] LP iter {it}, n={n}, score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Optional short hill‑climbing refinement for this length
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            a_hc, b_hc = _hill_climbing(f, _full_convolution(f), deadline - 2.0)
            score_hc = b_hc.max()
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] post‑LP hill‑climb n={n} score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (tiny random tweaks + hill‑climb) if we still have time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        # tiny random jitter before the final hill‑climb
        jitter = rng.normal(scale=target_sum * 5e-7, size=n)
        best_seq = np.maximum(best_seq + jitter, 0.0)
        best_seq = best_seq * (target_sum / best_seq.sum())
        a_final, b_final = _hill_climbing(best_seq, _full_convolution(best_seq), deadline - 1.0)
        final_val = b_final.max()
        if final_val < best_score:
            best_score = final_val
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] final hill‑climb improvement: {final_val:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for a tidy output
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the final evaluation (the official objective)
    # ------------------------------------------------------------------
    try:
        final_obj = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {final_obj:.6f} (length={best_seq.size})")
    except Exception as exc:
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
