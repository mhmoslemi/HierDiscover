def propose_candidate(seed=123, budget_s=1000):

   schedule = [2000, 5000, 10000, 20000, 40000]

       target = np.sqrt(2*n)
       
       if best_seq is not None and best_seq.size != n:
           a_init = _upsample_sequence(best_seq, n)
       else:
           a_init = _random_initial_sequence(n, rng)

       remaining = deadline - time.time()
       if remaining < 10: break

       n_restarts = 8 if n <= 10000 else 4
       
       local_best_a = None
       local_best_score = np.inf
       
       for r in range(n_restarts):
           if time.time() >= deadline: break
           # allocate per restart time
           per = min(gd_budget / max(1, n_restarts - r), 200.0)
           gd_deadline = time.time() + per
           
           if r == 0:
               start_a = a_init.copy()
           else:
               # choose deterministic initial families
               choice = rng.integers(0, 5)
               if choice == 0:

               elif choice == 2:
                   start_a = _random_initial_sequence(n, rng)

               else:
                   start_a = _random_initial_sequence(n, rng)
               start_a = _project_to_simplex(start_a, target)

           score = _max_convolution_fft(a_opt)
           if score < local_best_score:
               local_best_score = score
               local_best_a = a_opt.copy()
       
       if local_best_a is not None and local_best_score < best_score:
           best_score = local_best_score
           best_seq = local_best_a.copy()
           print(f"[{time.time():.1f}] GD best n={n} score={local_best_score:.6f}")
       
       # Phase hill climbing
       remaining = deadline - time.time()
       if remaining < 6: break
       hill_budget = remaining * 0.35

       # optional refinement gradient

   # final polishing

       n = best_seq.size
       target = np.sqrt(2.0 * n)
       a = _project_to_simplex(best_seq.copy(), target)
       final = _gradient_descent_opt(a, deadline)

       if final_score < best_score:

           best_seq = final.copy()

   # trim zeros

   nz = np.where(best_seq > 1e-12)[0]
   if nz.size:

       print(f"evaluate_sequence error: {e}")
   
   return best_seq.tolist()

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The evaluator normalises the vector to Σa = √(2n), therefore the
objective reduces to minimising the infinity‑norm of the linear
self‑convolution.
The algorithm mixes several ideas that proved useful for the previous
best construction:

* **Multi‑scale schedule** – start with modest lengths (2 k‑4 k) and
  progressively up‑sample to larger lengths (up to 40 k).  The best
  vector found at a stage is used as a warm start for the next stage.

* **Diversified initialisers** – random families (power‑law,
  exponential, Dirichlet) together with deterministic shapes
  (sine‑squared, triangular bump).  This provides a wide variety of
  starting points for the local optimiser.

* **Projected Adam optimiser on a log‑sum‑exp surrogate** – we smooth
  the max‑operator with a soft‑max (temperature β) and minimise the
  surrogate by Adam on the ℓ¹‑simplex (the projection onto the simplex
  is performed after each update).  β is annealed aggressively up to
  10⁶, so the optimiser finally concentrates on the true maximum.

* **Mass‑transfer hill climbing** – after the gradient phase we run a
  simulated‑annealing style hill‑climber that moves a small amount of
  mass from indices that heavily contribute to the current worst
  convolution entry towards indices of very low weight.  The full
  convolution is updated in O(n) per move, allowing many cheap moves.

* **Final polishing** – any remaining budget is spent on a last high‑β
  Adam run.

All vectors stay non‑negative and are projected onto the simplex
{aᵢ≥0, Σaᵢ = √(2·n)} after every move, guaranteeing that the official
evaluator sees a valid step function.

The routine respects the wall‑clock budget (default 1000 s) and returns
the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p),   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_descent(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Projected Adam optimiser on the log‑sum‑exp surrogate of the maximum
    autocorrelation entry.  The temperature β is annealed (×1.5) up to 1e6.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    beta = 0.5
    beta_max = 1e6
    beta_factor = 1.5

    # step‑size scale – proportional to the feasible ℓ₁‑radius
    base_lr = target_sum * 0.15

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat

        # projection onto the ℓ₁‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Deterministic and random initialisers
# ----------------------------------------------------------------------


def _sine_initial(n: int) -> np.ndarray:
    """Positive half‑wave sine‑squared shape."""
    idx = np.arange(n, dtype=float)
    a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12
    return a


def _triangular_initial(n: int) -> np.ndarray:
    """Triangular bump vanishing at the boundaries."""
    idx = np.arange(n, dtype=float)
    a = (idx + 1.0) * (n - idx) + 1e-12
    return a


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled to diversify the search space.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine (positive half‑wave)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )
    # tiny Gaussian noise + non‑negativity guard
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float,
                   rng: np.random.Generator):
    """
    Simulated‑annealing hill climbing that transfers mass from a source
    index (chosen proportionally to its contribution to the current
    worst convolution entry) to a destination index (chosen among the
    smallest entries).  The full convolution is kept up‑to‑date in O(n)
    per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    T = cur_score * 0.12 + 1e-9          # initial temperature
    cooling = 0.9994

    # limit number of iterations to keep the runtime bounded
    max_iters = 200_000
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # ---- pick a worst convolution index ------------------------------------
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # ---- source index weighted by contribution to entry M ------------------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # ---- destination index: pick a very small entry (deterministic) -----
        j = int(np.argmin(a))
        if j == i:
            # if the smallest entry coincides with the source, pick a random other
            j = (i + rng.integers(1, n)) % n

        # ---- amount of mass to move -----------------------------------------
        # move a moderately large fraction of the source mass
        max_delta = a[i] * rng.uniform(0.25, 0.65)
        delta = max_delta
        if delta <= 0.0:
            continue

        # ---- update convolution (linear terms) --------------------------------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # ---- quadratic correction terms ---------------------------------------
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ---- Metropolis acceptance --------------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        # ---- temperature decay -------------------------------------------------
        T *= cooling

    return best_a, b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # leave a small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seeding from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        try:
            a0 = np.asarray(earlier, dtype=float)
            if a0.ndim == 1 and a0.size > 0:
                n0 = a0.size
                a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
                cur = _max_convolution_fft(a0)
                best_seq = a0.copy()
                best_score = cur
                print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")
        except Exception as e:
            print(f"Failed to use height_sequence_1: {e}")

    # ------------------------------------------------------------------
    # Length schedule – increasingly larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – gradient descent with several restarts
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        # allocate roughly 30 % of the remaining budget for gradient work
        gd_budget = remaining * 0.30
        n_restarts = 8 if n <= 10000 else 4
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            # time slice for this restart (cap at 120 s)
            now = time.time()
            per_restart = min((deadline - now) * 0.33, 120.0)
            gd_deadline = now + per_restart

            if r == 0:
                start_a = a_init.copy()
            else:
                # deterministic seeds add diversity
                choice = rng.integers(0, 5)
                if choice == 0:
                    start_a = _sine_initial(n)
                elif choice == 1:
                    start_a = _triangular_initial(n)
                else:
                    start_a = _random_initial_sequence(n, rng)
                start_a = _project_to_simplex(start_a, target_sum)

            a_opt = _adam_descent(start_a, gd_deadline, rng)
            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time():.1f}] GD improvement for n={n} score={local_best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – hill climbing with exact convolution updates
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 6.0:
            break
        hill_budget = remaining * 0.35
        hill_deadline = time.time() + hill_budget

        # ensure we have the correct length vector
        if best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = best_seq.copy()
        a_current = _project_to_simplex(a_current, target_sum)
        b_full = _full_convolution(a_current)

        a_hc, b_hc = _hill_climbing(a_current, b_full, hill_deadline, rng)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Phase C – optional fine‑grained gradient refinement
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            refine_deadline = time.time() + min(remaining * 0.20, 80.0)
            a_ref = _adam_descent(best_seq.copy(), refine_deadline, rng)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Refinement improvement for n={n} score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        a_final = _adam_descent(a, deadline, rng)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros (nice output)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
