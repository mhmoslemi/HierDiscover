"""
Propose a step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
* **Progressive length schedule** – start with a modest length (2 k) and
  repeatedly up‑sample the best vector to a larger length.  Larger
  lengths give more degrees of freedom, which normally lowers the
  worst autocorrelation.
* **Projected gradient descent on a smooth max** – minimise a
  log‑sum‑exp surrogate of the worst entry of the self‑convolution.
  The surrogate becomes the true max for a large inverse temperature β.
  A simple momentum scheme and occasional jitter help escaping plateaus.
* **Heavy time allocation** – the budget (default 1000 s) is split so
  that the *later* (larger) stages receive the majority of the time,
  because they have the greatest impact on the final score.
* **Final polishing** – any remaining time is spent on a last
  high‑β gradient run on the best vector found.

All vectors are projected onto the ℓ₁‑simplex
{aᵢ≥0, Σaᵢ=√(2 n)} after every move, guaranteeing that the official
evaluator sees a valid step function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008).
    """
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p), p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # log‑sum‑exp probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the log‑sum‑exp surrogate.
    Uses momentum, an annealed inverse temperature β and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # temperature schedule
    beta = 0.5
    beta_max = 1e7          # stronger focus on the true max
    beta_factor = 1.25      # faster annealing than the base code

    # momentum parameters
    momentum = np.zeros_like(a)
    rho = 0.93               # a bit more inertia

    # step‑size base (relative to target ℓ₁ norm)
    step_base = target_sum * 0.07

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            # adaptive learning rate – larger β → smaller steps
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ₁‑simplex
        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations to shake off plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a

    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled to diversify the search space.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine wave (positive half‑wave)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    (as a plain Python list) found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previous construction is present, use it as a seed.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – start small, finish large.
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000, 60000]

    for idx_stage, n in enumerate(schedule):
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise vector for this stage (upsample if possible).
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # ------------------------------------------------------------------
        # Allocate time for this stage.
        #   The later (larger) stages receive the majority of the budget.
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break

        # split: 60 % of remaining time for non‑final stages, 90 % for the last stage
        fraction = 0.6 if idx_stage < len(schedule) - 1 else 0.9
        stage_deadline = time.time() + min(remaining * fraction, 300.0)

        # ------------------------------------------------------------------
        # Gradient descent optimisation.
        # ------------------------------------------------------------------
        a_opt = _gradient_descent_opt(a_init, stage_deadline)

        # ------------------------------------------------------------------
        # Evaluate current candidate.
        # ------------------------------------------------------------------
        cur_score = _max_convolution_fft(a_opt)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] Stage n={n} improvement: score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # (Optional) quick post‑stage refinement if a tiny slice of time is left.
        # ------------------------------------------------------------------
        if time.time() < deadline - 4.0:
            # a few extra gradient steps with a very high β (focus on true max)
            extra_deadline = time.time() + min( (deadline - time.time()) * 0.5, 30.0 )
            a_refine = _gradient_descent_opt(best_seq.copy(), extra_deadline)
            ref_score = _max_convolution_fft(a_refine)
            if ref_score < best_score:
                best_score = ref_score
                best_seq = a_refine.copy()
                print(f"[{time.time():.1f}] Post‑stage refinement n={n} score={ref_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        print(f"[{time.time():.1f}] Final polishing (n={n}) – remaining time {deadline - time.time():.1f}s")
        a_final = _gradient_descent_opt(best_seq.copy(), deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for a tidy output.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log official evaluation.
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
