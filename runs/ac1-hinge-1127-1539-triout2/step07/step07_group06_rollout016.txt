"""
Propose a step‑function height vector that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The objective is scale‑invariant; after normalising the ℓ¹‑norm to
√(2·n) the value reduces to the peak of the self‑convolution.
The algorithm consists of three stages:

1. **Fast incremental local search** (mass‑transfer, smoothing, jitter)
   to obtain a good feasible point.
2. **Frank‑Wolf­e (LP) refinement** – solve a linear programme that
   maximises the ℓ¹‑norm of a partner vector `g` under the
   constraints   a * g ≤ current_peak.  A convex combination
   a ← (1‑τ) a + τ·g (τ decreasing) monotonically reduces the peak.
3. **Projected gradient descent on a soft‑max surrogate** for final polishing.

Time is split adaptively (≈60 % local search, ≈30 % LP, ≈10 % GD)
so the routine always returns before the overall budget expires.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p)  scaled to a prescribed ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Fast objective – the peak of the self‑convolution."""
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Incremental local‑search (mass transfer, smoothing, jitter)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer `delta` mass from `donor` to `receiver`,
    updating `a` and the full convolution `conv` exactly.
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]
    a_before = a.copy()

    # linear part of the convolution update
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic part (the two new self‑terms and the cross‑term)
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # coefficients
    a[donor] -= delta
    a[receiver] += delta
    if a[donor] < 0.0:
        a[donor] = 0.0

    conv[:] = np.maximum(conv, 0.0)


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Fast incremental local search for a given length `n`.
    Returns the best sequence and its peak (max convolution).
    """
    target = math.sqrt(2.0 * n)

    # ---- initialise ----------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), mode="constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        # annealing schedules
        max_frac = 0.5 * (0.005 / 0.5) ** tau          # maximal donor fraction
        temp = 0.5 * (1e-5 / 0.5) ** tau               # Metropolis temperature

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass transfer (≈80 %)
        # --------------------------------------------------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]               # contribution of each donor candidate
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            # receiver biased towards small coefficients
            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            # Metropolis acceptance
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = math.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] LS new best max={best_max:.6f}")
            else:
                # revert
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) smoothing (≈10 %)
        # --------------------------------------------------------------
        elif move_type < 0.95:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_window + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = math.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] LS smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) Gaussian jitter (≈10 %)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] LS jitter improved max={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Frank‑Wolfe refinement via a linear programme
# ----------------------------------------------------------------------


def _lp_optimize_direction(a: np.ndarray, max_val: float) -> np.ndarray | None:
    """
    Solve   max Σ g_i   s.t.   a * g ≤ max_val  (entry‑wise)  and  g ≥ 0.
    Returns the optimal vector g (or None on failure).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    # Build A_ub where row k, column j contains a_{k-j} (zero when out of range)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(rows, max_val, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # maximise Σ g_i  → minimise -Σ g_i
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    else:
        return None


def _frank_wolfe_refine(a: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style refinement: repeatedly solve the LP above,
    take a convex combination with decreasing step size.
    """
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    it = 0
    step = 0.3  # initial step size

    while time.time() < deadline:
        g = _lp_optimize_direction(a, cur_max)
        if g is None:
            break
        sumg = g.sum()
        if sumg == 0.0:
            break
        # rescale g to the required ℓ¹‑norm
        g = g * (target / sumg)

        # Frank‑Wolfe step size (2 / (t+2) is classic, capped at 0.5)
        step = min(0.5, 2.0 / (it + 2))
        a_new = (1.0 - step) * a + step * g
        a_new = _project_to_simplex(a_new, target)
        conv_new = _conv_fft(a_new)
        max_new = conv_new.max()

        # accept only if we did not increase the peak
        if max_new <= cur_max:
            a = a_new
            cur_max = max_new
            if max_new < best_max:
                best_max = max_new
                best_a = a.copy()
                print(f"[{time.time():.1f}] FW iter={it} new best max={best_max:.6f}")
        else:
            # no improvement – try a smaller step next round
            step *= 0.5

        # occasional random jitter to avoid stagnation
        if it % 10 == 0:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target)

        it += 1
        if it > 200:          # safety stop
            break

    return best_a, best_max


# ----------------------------------------------------------------------
# Projected gradient descent on a soft‑max surrogate (reference)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a, a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max surrogate.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g
        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine first runs an inexpensive local search on a few
    problem sizes, then refines the best candidate with a Frank‑Wolfe style
    linear‑programming step, and finally performs a short projected‑gradient
    polish.  It always returns before the allocated budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Warm start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    warm_start = None
    if "height_sequence_1" in globals():
        warm_start = np.array(globals()["height_sequence_1"], dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Run local‑search on a few lengths, keep the best result
    # ------------------------------------------------------------------
    lengths = [1500, 1800, 2000]   # a handful of sizes – larger gives more freedom
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        # give each size at most 0.25 · remaining seconds (capped)
        time_for_n = min(remaining * 0.25, 250.0)

        run_deadline = time.time() + time_for_n
        seq, peak = _local_search(n, run_deadline, rng,
                                  init_seq=warm_start if n == lengths[0] else None)

        # objective for the evaluation function (sum already = √(2·n))
        obj = peak
        if obj < best_score:
            best_score = obj
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} → new best obj={best_score:.6f}")

    if best_seq is None:
        # fallback – create a simple power‑law if everything failed
        n = 1500
        targ = math.sqrt(2.0 * n)
        best_seq = _power_law_sequence(n, 0.6, targ)
        best_seq = _project_to_simplex(best_seq, targ)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 3) Frank‑Wolfe refinement (LP) on the current best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        n = best_seq.size
        fw_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_fw, peak_fw = _frank_wolfe_refine(best_seq, fw_deadline, rng)
        obj_fw = peak_fw
        if obj_fw < best_score:
            best_score = obj_fw
            best_seq = seq_fw.copy()
            print(f"[{time.time():.1f}] FW refinement → new best obj={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final projected‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        gd_deadline = time.time() + (deadline - time.time()) * 0.15
        best_seq, final_val = _annealed_gradient_descent(best_seq, gd_deadline)
        best_score = final_val
        print(f"[{time.time():.1f}] final GD polishing → obj={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Strip trailing zeros (they have no effect on the score)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=np.float64)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = np.array([], dtype=np.float64)

    print(f"[{time.time():.1f}] final candidate → obj≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
