"""
Improved search for a step‑function that minimises ``evaluate_sequence``.
The algorithm expands the family of candidate shapes (adds a beta‑shape
and a log‑normal shape), uses a longer gradient‑descent phase with a
more aggressive β‑schedule, and performs a more thorough permutation
annealing (swap moves plus occasional block reversals).  The total
budget (default 1000 s) is split 60 % random generation, 30 %
refinement of the elite set and 10 % final annealing.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    # Next power of two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x≥0 (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max (aggressive β schedule)
# ----------------------------------------------------------------------
def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the log‑sum‑exp surrogate.
    The β‑parameter is exponentially increased up to 1e8,
    and the momentum / learning‑rate schedule is tuned for stability.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sum(a)               # keep the current ℓ¹‑norm (scale‑invariant)

    beta = 1.0
    max_beta = 1e8
    # momentum variables
    momentum = np.zeros_like(a)
    rho = 0.9
    it = 0
    rng = np.random.default_rng()
    # heuristics for step size
    base_lr = 0.7 * target_sum

    while time.time() < deadline:
        # gradient of the smooth max
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = base_lr / (norm * (beta + 1.0))
            a -= lr * momentum

        # project back to the ℓ¹‑simplex (preserves scale)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 6e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing (sharpen the surrogate)
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.12, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap + block‑reverse moves)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset.
    In addition to pairwise swaps it occasionally reverses a random
    contiguous block, which enables larger‑scale rearrangements.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # temperature tied to current objective
    T = best_score * 0.025 + 1e-9
    cooling = 0.9993
    rng = np.random.default_rng()

    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.85:                # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
        else:                               # block reverse
            # pick block length (log‑uniform)
            length = int(np.exp(rng.uniform(np.log(2), np.log(n // 2 + 1))))
            start = rng.integers(0, n - length + 1)
            a[start:start + length] = a[start:start + length][::-1]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert the move (swap or block‑reverse)
            if move_type < 0.85:
                a[i], a[j] = a[j], a[i]
            else:
                # reverse back
                a[start:start + length] = a[start:start + length][::-1]

        T *= cooling

        # tiny random shuffle with very low probability – helps escape deep traps
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.015
            if np.any(mask):
                shuffled = a[mask].copy()
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Helper families for candidate generation
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float = 1.0) -> np.ndarray:
    """Scale vector to have ℓ¹‑sum *target* (preserving non‑negativity)."""
    total = v.sum()
    if total <= 0:
        # fallback: uniform tiny mass
        return np.full_like(v, target / v.size)
    a = v * (target / total)
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw)


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw)


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw)


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw)


def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Shape proportional to (i+1)^α * (n-i)^β (β‑parameter is named 'beta' here)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + 1) ** alpha * (n - idx) ** beta
    return _normalize_to_sum(raw)


def _lognormal_sequence(n: int, mu: float, sigma: float, shift: float) -> np.ndarray:
    """Discrete log‑normal shape."""
    idx = np.arange(n, dtype=float) + shift + 1.0
    raw = np.exp(-0.5 * ((np.log(idx) - mu) / sigma) ** 2)
    return _normalize_to_sum(raw)
