def _subgradient_step(a, target_sum, lr):
    # compute conv and max index
    n = a.size
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_idx = int(conv.argmax())
    # gradient
    g = np.zeros_like(a)
    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    i = np.arange(lo, hi+1)
    g[i] = 2 * a[max_idx - i]
    # update
    a_new = a - lr * g
    # project onto simplex
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new, conv.max()
```

def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    step_factor = 0.5
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute gradient for max entry or near max
        n = a.size
        size = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # optionally consider near max indices
        epsilon = max_val * 0.001
        idxs = np.where(conv >= max_val - epsilon)[0]
        # compute gradient as sum over those idxs
        # Use convolution of reversed a with weight vector
        w = np.zeros(2*n-1)
        w[idxs] = 1.0
        # compute grad = 2 * (rev_a) * w (conv)
        rev_a = a[::-1]
        rev_fft = np.fft.rfft(rev_a, n=size)
        w_fft = np.fft.rfft(w, n=size)
        grad_rev = np.fft.irfft(rev_fft * w_fft, n=size)[:2*n-1]   # length 2n-1
        grad = 2.0 * grad_rev[n-1:n-1+n]   # pick middle segment
        # step size
        norm = np.linalg.norm(grad)
        if norm > 0:
            lr = step_factor * target_sum / (norm + 1e-12)
            a_new = a - lr * grad
            a_new = _project_to_simplex(a_new, target_sum)
        else:
            # jitter
            jitter = rng.normal(scale=target_sum*0.001, size=a.shape)
            a_new = a + jitter
            a_new = _project_to_simplex(a_new, target_sum)
        a = a_new
        # maybe reduce step_factor over time
        it += 1
        if it % 100 == 0:
            step_factor *= 0.95   # decaying
        cur_score = _max_convolution_fft(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_score:.6f}")
    return best_a, best_score
```

"""
A multi‑phase optimizer for the “max of a*a” benchmark.

1.  Choose a problem size n based on the time budget (≈5 k–12 k).
2.  Generate a diverse set of candidate step‑functions (power‑law,
    exponential, linear ramp, sine, sine‑power and products).  Each
    candidate is normalised so that Σa = √(2 n).  Keep the one with the
    smallest max‑convolution value.
3.  Refine the best candidate with a cheap sub‑gradient descent that
    follows the exact gradient of the current maximal convolution entry.
    The update is projected back onto the simplex by simple rescaling.
4.  Run an aggressive simulated‑annealing phase (mass transfers,
    local smoothing and swaps) on the remaining time.
5.  If a little time is left, perform a short soft‑max projected‑gradient
    polish (the routine used in the previous state‑of‑the‑art).

The code respects the 1000 s time limit, never writes to disk and only
uses NumPy, time and the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Exact Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _simplex_rescale(v: np.ndarray, s: float) -> np.ndarray:
    """Fast feasible projection: clip negatives and renormalise to sum s."""
    v = np.maximum(v, 0.0)
    tot = v.sum()
    if tot > 0:
        v = v * (s / tot)
    else:
        v = np.full_like(v, s / v.size)
    return v


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


# ----------------------------------------------------------------------
# Gradient helpers (soft‑max version – retained from the previous code)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max approx. of max(conv)."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 1.0
    max_beta = 16384.0
    step_base = target_sum * 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.002 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = _simplex_rescale(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                                time.time() + run_budget,
                                                target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 10.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.5,
                         max_frac_end: float = 0.01,
                         max_window: int = 100) -> tuple[np.ndarray, float]:
    """Annealing on a, moving mass, smoothing windows and swapping entries."""
    a = a0.copy()
    n = a.size
    cur_score = _max_convolution_fft(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # location of current maximum
        k_max = int(np.argmax(conv))

        # sensitivities for a (derivative of the max entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # ------------------- mass transfer -------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                a[donor] -= delta
                a[receiver] += delta
                move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _max_convolution_fft(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _simplex_rescale(a, np.sqrt(2.0 * n))
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent: each step follows the gradient of the
    current maximal convolution entry, then rescales to the simplex.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    rng = np.random.default_rng()
    step_factor = 0.5
    it = 0

    while time.time() < deadline:
        # convolution via FFT (needed for max index)
        size = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
        max_val = float(conv.max())
        k = int(conv.argmax())

        # exact gradient of the maximal entry
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]

        norm = np.linalg.norm(grad)
        if norm > 0:
            lr = step_factor * target_sum / (norm + 1e-12)
            a_new = a - lr * grad
        else:
            # pure jitter when gradient vanishes
            a_new = a + rng.normal(scale=target_sum * 0.001, size=a.shape)

        a = _simplex_rescale(a_new, target_sum)

        it += 1
        if it % 200 == 0:
            step_factor *= 0.90  # gradual decay

        # keep the best encountered point
        if max_val < best_score:
            best_score = max_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    max‑convolution value.  The routine adapts its problem size to the
    time budget and combines shape exploration, sub‑gradient descent,
    simulated annealing and a final soft‑max GD polish.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ----- problem size -------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ----- 1) diverse shape search (≈15 % of budget) -------------------
    shape_deadline = time.time() + (deadline - time.time()) * 0.15

    best_seq = None
    best_score = np.inf

    # reuse previous best if available
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size != n:
            seq = _resize_sequence(seq, n)
        seq = _simplex_rescale(seq, target_sum)
        sc = _max_convolution_fft(seq)
        best_seq, best_score = seq, sc

    # parameter grids
    p_vals = np.arange(0.2, 1.6, 0.2)          # power‑law exponents
    lam_vals = np.linspace(0.0, 0.0015, 5)    # exponential rates
    gamma_vals = np.linspace(0.0, 2.0 / (n - 1), 5)  # linear ramps
    q_vals = np.arange(0.4, 2.2, 0.4)         # sine‑power exponents

    rng = np.random.default_rng()

    while time.time() < shape_deadline:
        shape_type = rng.integers(0, 6)
        if shape_type == 0:          # pure power‑law
            p = float(rng.choice(p_vals))
            seq = _power_law_sequence(n, p, target_sum)

        elif shape_type == 1:        # exponential decay
            lam = float(rng.choice(lam_vals))
            seq = np.exp(-lam * np.arange(n, dtype=np.float64))
            seq = seq * target_sum / seq.sum()

        elif shape_type == 2:        # linear drop to zero
            gamma = float(rng.choice(gamma_vals))
            seq = np.maximum(0.0, 1.0 - gamma * np.arange(n, dtype=np.float64))
            seq = seq * target_sum / seq.sum()

        elif shape_type == 3:        # pure sine shape
            base = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
            seq = base * target_sum / base.sum()

        elif shape_type == 4:        # sine raised to a power
            q = float(rng.choice(q_vals))
            base = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
            seq = np.power(base, q)
            seq = seq * target_sum / seq.sum()

        else:                        # product of power‑law & exponential
            p = float(rng.choice(p_vals))
            lam = float(rng.choice(lam_vals))
            seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p)) \
                  * np.exp(-lam * np.arange(n, dtype=np.float64))
            seq = seq * target_sum / seq.sum()

        sc = _max_convolution_fft(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Shape type={shape_type} score={best_score:.6f}")

    # ----- 2) sub‑gradient descent (≈50 % of remaining time) -----------
    if best_seq is None:
        best_seq = _power_law_sequence(n, 0.55, target_sum)

    subgrad_deadline = time.time() + (deadline - time.time()) * 0.5
    a_sgd, val_sgd = _subgradient_descent(best_seq, target_sum, subgrad_deadline)
    if val_sgd < best_score:
        best_score = val_sgd
        best_seq = a_sgd.copy()
        print(f"[{time.time():.2f}] Sub‑gradient best score={best_score:.6f}")

    # ----- 3) simulated annealing on the remaining time --------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq,
                                            deadline,
                                            start_temp=20.0,
                                            end_temp=1e-6,
                                            max_frac_start=0.8,
                                            max_frac_end=0.01,
                                            max_window=200)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] SA refined best score={best_score:.6f}")

    # ----- 4) final polishing with soft‑max GD (if time permits) -----
    if time.time() + 2.0 < deadline:
        a_final, val_final = _annealed_gradient_descent_v2(best_seq,
                                                            deadline,
                                                            target_sum)
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final polish improved score={best_score:.6f}")

    # ----- tidy up and return -------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional verification (won’t affect the returned value)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
