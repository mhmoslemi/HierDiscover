"""
Penalty‑based binary‑search optimizer for the self‑convolution ℓ∞‑norm.

The idea is to search for the smallest feasible bound M such that a
non‑negative vector a (∑a = √(2n)) satisfies   a∗a ≤ M  (entry‑wise).
For a given M we minimise the squared ReLU‑penalty
    P(a;M)=∑_k max(0, (a∗a)_k – M)^2
by projected (momentum) gradient descent.
If the penalty reaches ~0 the bound M is feasible; otherwise we raise M.
A simple binary search over M together with occasional random restarts
produces a sequence with a lower maximal convolution entry than the
previous gradient‑softmax / hill‑climb approach.
The algorithm works for several lengths (2000…32000) and finishes within
the 1000 s wall‑clock budget.  Finally a short hill‑climb is applied to
polish the result.

The returned list is ready to be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Projection onto the simplex Σ a_i = s, a_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns a NumPy array of length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Random initialisers (simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families (power‑law, exponential, Dirichlet, piecewise) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 5)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.2, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.2, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by projection
    onto the appropriate simplex (sum = √(2·new_n)).
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Penalty gradient (squared ReLU) for a fixed bound M
# ----------------------------------------------------------------------


def _gradient_penalty(a: np.ndarray, M: float):
    """
    Compute gradient of the squared‑ReLU penalty
        P(a) = Σ_k max(0, (a∗a)_k – M)^2
    and return (grad, penalty, full_convolution).
    """
    n = a.shape[0]

    # 1) full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # 2) ReLU residuals
    resid = conv - M
    resid = np.where(resid > 0, resid, 0.0)          # max(0, •)
    penalty = np.dot(resid, resid)                  # Σ resid²

    # 3) gradient = 4 * correlation(resid, a)
    a_rev = a[::-1]
    size2 = _next_pow2(resid.size + a_rev.size - 1)  # length 3n‑2 rounded up
    resid_fft = np.fft.rfft(resid, n=size2)
    a_rev_fft = np.fft.rfft(a_rev, n=size2)
    corr_full = np.fft.irfft(resid_fft * a_rev_fft, n=size2)
    # slice the part that corresponds to indices 0 … n‑1
    grad = 4.0 * corr_full[n - 1: n - 1 + n]

    return grad, penalty, conv


# ----------------------------------------------------------------------
# Projected (momentum) gradient descent for a fixed M
# ----------------------------------------------------------------------


def _penalty_opt(a0: np.ndarray, M: float, target_sum: float, deadline: float):
    """
    Try to find a feasible vector a (sum = target_sum, a ≥ 0) such that
    max(a∗a) ≤ M.  Returns (a, max_convolution, feasible_flag).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)
    rho = 0.90
    # base step roughly proportional to the ℓ¹‑norm
    base_lr = target_sum * 0.3
    rng = np.random.default_rng()

    best_a = a.copy()
    best_max = np.inf
    best_pen = np.inf

    it = 0
    while time.time() < deadline:
        grad, pen, conv = _gradient_penalty(a, M)

        # check feasibility (within numerical tolerance)
        if pen < 1e-12:
            return a, conv.max(), True

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0:
            lr = base_lr / (norm_m + 1e-12)
            a -= lr * momentum
        else:
            a -= base_lr * grad

        # projection onto simplex (keeps ℓ¹‑norm fixed)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # keep best (even if infeasible) – useful for final polishing
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_pen = pen

        it += 1

    # deadline reached → return best found (probably infeasible)
    return best_a, best_max, False


# ----------------------------------------------------------------------
# Greedy hill‑climbing (exact O(n) update) – unchanged from prior code
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of mass
    from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # slower cooling for finer search

    while time.time() < deadline:
        # pick one of the currently worst convolution positions
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index: weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index: 30 % bias towards indices with small mass
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution (vectorised O(n) slices)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf          # this stores the *max convolution* (since sum is fixed)

    # ------------------------------------------------------------------
    # If a previous construction exists, use it as a seed
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = np.sqrt(2.0 * n0)
            prev = _project_to_simplex(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"[{time.time():.1f}] seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule – gradually increase dimension
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000, 32000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = _random_initial_sequence(n, rng)

        # --------------------------------------------------------------
        # Binary search on the maximal convolution bound M
        # --------------------------------------------------------------
        # Upper bound – start from the uniform distribution (max ≈ 2)
        uniform = np.full(n, target_sum / n)
        high = _max_convolution_fft(uniform)
        low = 0.0

        # If we already have a feasible sequence, tighten the lower bound
        if best_seq is not None and best_seq.size == n:
            low = best_score

        # keep a copy of the best feasible sequence for this length
        best_feasible_seq = None
        best_feasible_M = None

        # binary search (max 12 rounds or until time runs out)
        for bisect_iter in range(12):
            if time.time() >= deadline - 5.0:
                break
            M_mid = (low + high) / 2.0

            feasible = False
            best_local_seq = None
            best_local_max = np.inf

            # a few random restarts for each M
            for restart in range(3):
                remaining = deadline - time.time()
                if remaining < 5.0:
                    break
                # start from current or fresh random vector
                a_start = a_current.copy() if restart == 0 else _random_initial_sequence(n, rng)

                # allocate a small slice of time for this optimisation run
                run_deadline = time.time() + min(8.0, remaining * 0.15)

                a_opt, cur_max, ok = _penalty_opt(a_start, M_mid, target_sum, run_deadline)

                if ok:
                    feasible = True
                    best_local_seq = a_opt.copy()
                    best_local_max = cur_max
                    break
                else:
                    if cur_max < best_local_max:
                        best_local_max = cur_max
                        best_local_seq = a_opt.copy()

            if feasible:
                high = M_mid
                best_feasible_seq = best_local_seq.copy()
                best_feasible_M = M_mid
                print(f"[{time.time():.1f}] n={n} feasible M={M_mid:.6f}")
            else:
                low = M_mid
                print(f"[{time.time():.1f}] n={n} infeasible M={M_mid:.6f}")

        # --------------------------------------------------------------
        # If we discovered a feasible sequence, keep it as the new best
        # --------------------------------------------------------------
        if best_feasible_seq is not None and best_feasible_M < best_score:
            best_seq = best_feasible_seq.copy()
            best_score = best_feasible_M
            print(f"[{time.time():.1f}] New best after binary search: M={best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – hill climbing refinement (uses exact O(n) updates)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            # Ensure we have a sequence of the current length
            if best_seq is None or best_seq.size != n:
                a_hc = _random_initial_sequence(n, rng)
            else:
                a_hc = best_seq.copy()
            a_hc = _project_to_simplex(a_hc, target_sum)
            b_full = _full_convolution(a_hc)

            hill_deadline = time.time() + min(remaining * 0.25, 180.0)
            a_hc, b_hc = _hill_climbing(a_hc, b_full, hill_deadline)
            score_hc = b_hc.max()
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)

        # final penalty‑based optimisation (tighten the bound a little more)
        final_deadline = deadline - 2.0
        a_final, cur_max, _ = _penalty_opt(best_seq.copy(), best_score, target_sum, final_deadline)

        # short hill climbing at the very end
        b_final = _full_convolution(a_final)
        a_final, b_final = _hill_climbing(a_final, b_final, deadline - 2.0)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
