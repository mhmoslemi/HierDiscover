def _simulated_annealing(seq, end_time, rng):
    best_seq = seq.copy()
    best_score = _max_convolution(seq)
    cur_seq = seq.copy()
    cur_score = best_score
    temperature = 0.1 * best_score
    while time.time() < end_time:
        # choose random i, j
        i = rng.integers(0, len(seq))
        j = rng.integers(0, len(seq))
        if i == j:
            continue
        delta = rng.uniform(0, cur_seq[i] * 0.2)
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure nonneg, may need to adjust if negative
        if new_seq[i] < 0:
            continue
        new_seq = new_seq * (np.sqrt(2*len(seq))/new_seq.sum())
        new_score = _max_convolution(new_seq)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score)/temperature):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
        # decay temperature
        temperature *= 0.99999
    return best_seq, best_score

"""
Adversarial step‑function optimizer – third generation.

The routine follows three stages that fit comfortably into the 1000 s
budget.

1️⃣  Random exploration – many candidates are drawn from a collection of
    parametric families (Dirichlet / Gamma, left‑hand power‑law,
    left‑hand exponential, uniform and a sparse‑bump mixture).  The
    vectors are normalised so that their ℓ₁‑norm equals √(2 n) (the
    objective becomes simply the peak of the self‑convolution).  The
    best *k* candidates are retained.

2️⃣  Multi‑scale refinement – each retained candidate is interpolated to
    a series of larger lengths (up to ≈ 30 000).  For every target length
    a projected‑gradient Adam optimiser works directly on the exact
    max‑convolution (a τ‑soft‑max with τ annealed to zero).  Hyper‑
    parameters (learning‑rate, temperature schedule, jitter) differ
    between runs to increase diversity.

3️⃣  Simple Simulated‑Annealing fine‑tuning – if time remains we perform
    a cheap mass‑transfer walk on the simplex that can escape shallow
    local minima.

The final construction is stored in the global variable
``height_sequence_1`` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of τ‑soft‑max (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.45,
                   tau_decay: float = 0.9994,
                   lr_decay: float = 0.9998,
                   jitter_every: int = 500) -> tuple[np.ndarray, float]:
    """
    Minimise the exact max‑convolution of ``a`` under the ℓ₁‑norm
    constraint.  Returns the best vector seen and its *hard* peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits from the feasible a0  (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # temperature for τ‑soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution & gradient (soft max) -------
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- gradient w.r.t. logits (chain rule) -------
        mu = np.dot(a, g) / S               # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)               # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- annealing schedules -------
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if jitter_every and it % jitter_every == 0:
            p += rng.normal(scale=0.008, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Random families for the exploration phase
# ----------------------------------------------------------------------
def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑draw → Dirichlet‑like vector."""
    alpha = rng.uniform(0.2, 5.0)
    raw = rng.gamma(alpha, 1.0, size=n)
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """All ones."""
    return np.full(n, 1.0)


def _init_left_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law, optionally mixed with a uniform background."""
    idx = np.arange(n)
    p = rng.uniform(0.3, 3.0)                # exponent
    offset = rng.uniform(0.0, 10.0)           # horizontal shift
    mix = rng.uniform(0.0, 1.0)               # how much background
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * (idx + offset) ** (-p)
    return raw


def _init_left_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing exponential, optionally mixed with a uniform background."""
    idx = np.arange(n)
    lam = rng.uniform(0.001, 0.02)           # decay rate
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * base + (1.0 - mix) * np.exp(-lam * idx)
    return raw


def _init_sparse_bumps(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random rectangular bumps on a tiny flat background."""
    raw = np.full(n, rng.uniform(0.0, 0.15))    # low background
    nbumps = rng.integers(1, max(2, n // 250))
    for _ in range(nbumps):
        center = rng.integers(0, n)
        width = rng.integers(3, max(5, n // 100))
        height = rng.uniform(0.8, 4.0)
        lo = max(0, center - width // 2)
        hi = min(n, lo + width)
        raw[lo:hi] = np.maximum(raw[lo:hi], height)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    With probability ~0.25 we warm‑start from the global best,
    otherwise we draw from one of the parametric families.
    """
    S = np.sqrt(2.0 * n)

    # Warm start occasionally
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['dirichlet', 'gamma', 'uniform',
                        'left_powerlaw', 'left_exponential', 'sparse'])
    if style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'left_powerlaw':
        raw = _init_left_powerlaw(n, rng)
    elif style == 'left_exponential':
        raw = _init_left_exponential(n, rng)
    else:  # sparse bumps
        raw = _init_sparse_bumps(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (very cheap, used only if time left)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer walk on the simplex with exponential acceptance."""
    n = len(seq)
    S = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _max_convolution(cur)
    best = cur.copy()
    best_score = cur_score
    temperature = 0.15 * cur_score        # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] == 0.0:
            continue
        # transfer a random fraction of the mass at i to j
        delta = rng.uniform(0.0, 0.3) * cur[i]
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # re‑normalise (keeps ℓ₁‑norm exactly S)
        new = new * (S / new.sum())
        new_score = _max_convolution(new)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        # geometric cooling
        temperature *= 0.99995
        if temperature < 1e-6:
            temperature = 1e-6

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  The algorithm consists of:

    1. Random exploration of many vectors from diverse families.
    2. Multi‑scale Adam refinement on the best few candidates.
    3. Optional simulated‑annealing fine‑tuning.

    The final construction is stored in the global variable
    ``height_sequence_1`` and returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if present
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 55 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 300.0          # reserve ~5 min for refinement
    best_seq = None
    best_score = np.inf
    top_k = 5
    top_candidates = []   # list of (score, seq)

    attempt = 0
    while time.time() < explore_deadline:
        attempt += 1
        # pick a length – logarithmic spread gives a good variety
        if rng.random() < 0.6:
            n = int(rng.integers(1800, 5000))
        else:
            n = int(rng.integers(5000, 25000))

        a0 = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

        # keep a small list of the best candidates for refinement
        if len(top_candidates) < top_k:
            top_candidates.append((peak, a0.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif peak < top_candidates[-1][0]:
            top_candidates[-1] = (peak, a0.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Ensure we have at least one candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n
        top_candidates = [(best_score, best_seq.copy())]

    # ------------------------------------------------------------------
    # Phase 2 – multi‑scale Adam refinement
    # ------------------------------------------------------------------
    # Length schedule – gradually grow the support up to a safe ceiling
    max_len = min(30000, best_n * 6)
    lengths = [best_n]
    if best_n * 1.4 < max_len:
        lengths.append(int(best_n * 1.4))
    if best_n * 2.0 < max_len:
        lengths.append(int(best_n * 2.0))
    if best_n * 3.0 < max_len:
        lengths.append(int(best_n * 3.0))
    lengths.append(int(max_len))
    lengths = sorted(set(lengths))

    # iterate over the shortlisted candidates (most promising first)
    for init_idx, (init_score, init_seq) in enumerate(top_candidates):
        cur_seq = init_seq.copy()
        cur_score = init_score
        cur_n = len(cur_seq)

        for target_n in lengths:
            remaining = end_time - time.time()
            if remaining < 35.0:
                break

            # interpolate to target length and normalise
            a_scaled = _scale_sequence(cur_seq, target_n)
            a_scaled = np.clip(a_scaled, 0.0, None)
            a_scaled = a_scaled * (np.sqrt(2.0 * target_n) / a_scaled.sum())

            # allocate a slice of remaining time (but never longer than 180 s)
            slice_budget = min(remaining * 0.45, 180.0)

            # run Adam – employ slightly different hyper‑parameters per run
            lr0 = rng.choice([0.42, 0.48, 0.55])
            tau_decay = rng.choice([0.9993, 0.9995, 0.9997])
            refined, peak = _adam_optimize(a_scaled,
                                           time.time() + slice_budget,
                                           rng,
                                           lr0=lr0,
                                           tau_decay=tau_decay,
                                           lr_decay=0.99985,
                                           jitter_every=500)

            if peak < best_score:
                best_score = peak
                best_seq = refined.copy()
                best_n = target_n
                print(f"[refine] init#{init_idx+1} len={target_n:5d}  improved score={best_score:.6f}")

            # use the refined vector as the seed for the next larger scale
            cur_seq = refined
            cur_n = target_n

        # stop early if budget is exhausted
        if time.time() >= end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing fine‑tuning if time permits
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 25.0:
        final_seq, final_peak = _anneal(best_seq,
                                        time.time() + remaining,
                                        rng)
        if final_peak < best_score:
            best_score = final_peak
            best_seq = final_seq.copy()
            print(f"[anneal] final annealed score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
