"""
adversarial step‑function optimizer – LP + Adam hybrid.

Algorithm outline
-----------------
1. **Random exploration** – generate many feasible candidates from a few
   parametric families, keep the one with the smallest self‑convolution peak.
2. **Linear‑programming descent** – for the current sequence `a` compute
   `M = max(a*a)`.  Solve the LP

        max   sum(g)
        s.t.  a * g ≤ M           (component‑wise)
              g ≥ 0

   The LP gives a feasible direction `g₀` that can increase the ℓ₁‑norm
   without breaking the current peak bound.  After scaling `g₀` to the
   required ℓ₁‑norm we perform a small convex combination
   `a ← (1‑t)·a + t·g` (line‑search on `t`).  Re‑repeat until no
   improvement is possible or time runs out.
3. **Scale‑up** – linearly interpolate the best sequence to a larger length
   and repeat the LP descent.  Larger `n` gives a richer search space.
4. **Adam fine‑tuning** – run the log‑softmax Adam optimiser (the same
   optimiser that the reference solution uses) for the remaining time.
5. **Final check** – evaluate the final sequence with the black‑box
   `evaluate_sequence`, store it in the global variable `height_sequence_1`
   and return it.

All numbers are forced to be non‑negative and the ℓ₁‑norm is kept equal to
`sqrt(2·n)`, which makes the external score exactly the maximum entry of the
self‑convolution.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum value of the full self‑convolution `a*a`."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# Random families for the initial exploration
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)       # exponent
    offset = rng.uniform(0.0, 15.0) # denominator shift
    mix = rng.uniform(0.0, 1.0)     # uniform background weight
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible vector of length n with ℓ₁‑norm = √(2n).
    """
    S = np.sqrt(2.0 * n)

    # Occasionally warm‑start from a known good sequence
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

# Linear‑programming direction (LP) utilities
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix M such that M @ g == a * g (full convolution).
    """
    n = len(a)
    m = 2 * n - 1
    M = np.zeros((m, n), dtype=np.float64)
    # each column i of M is a shifted copy of a
    for i in range(n):
        M[i:i + n, i] = a
    return M


def _lp_direction(a: np.ndarray, M_peak: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   sum(g)
        subject to M @ g <= M_peak
                   g >= 0
    Returns the (feasible) vector g.  If the solver fails we fall back
    to a uniform distribution.
    """
    n = len(a)
    A = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M_peak, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                 # minimise -sum(g)

    res = scipy.optimize.linprog(c,
                                 A_ub=A,
                                 b_ub=b_ub,
                                 bounds=[(0, None)] * n,
                                 method='highs',
                                 options={'presolve': True})
    if res.success:
        return res.x
    # fallback – uniform feasible point (always respects the bound)
    S = np.sqrt(2.0 * n)
    return np.full(n, S / n)


def _mix_step(a: np.ndarray, g_raw: np.ndarray, M_old: float) -> tuple[np.ndarray, float]:
    """
    Scale `g_raw` to the correct ℓ₁‑norm and try a short line‑search on the
    mixing parameter `t`.  Return a possibly improved pair (a_new, M_new).
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    sum_g = g_raw.sum()
    if sum_g <= 0.0:
        return a, M_old
    g = g_raw * (S / sum_g)            # now sum(g) = S

    # start with a moderately large step, shrink if necessary
    t = 0.5
    a_best = a
    M_best = M_old
    while t > 1e-4:
        a_new = (1.0 - t) * a + t * g
        M_new = _max_convolution(a_new)
        if M_new < M_best - 1e-9:
            return a_new, M_new
        t *= 0.6
    return a_best, M_best


def _lp_optimize(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly apply the LP‑direction step until no improvement is observed
    or the time budget is exhausted.
    """
    while time.time() < deadline:
        M = _max_convolution(a)
        # early stop if the bound is already very low
        if M < 1.40:
            break
        g_dir = _lp_direction(a, M)
        a_new, M_new = _mix_step(a, g_dir, M)
        if M_new < M - 1e-8:
            a = a_new
            print(f"[LP] improved peak to {M_new:.6f} (n={len(a)})")
        else:
            # no progress – stop the LP phase for this length
            break
    return a

# Adam optimiser (same as in the reference code, slightly tweaked)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ≈0).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.5                       # a little more aggressive than reference

    # initialise temperature (soft‑max smoothing)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1
        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient of (soft) max‑convolution
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.9995               # smoothness decay
        lr *= 0.9999                # very slow decay

        # occasional jitter
        if it % 500 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a 1000 s budget, stores the final sequence in the global
    variable ``height_sequence_1`` and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    warm_seq = None
    if 'height_sequence_1' in globals():
        try:
            warm_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            warm_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.30 * budget_s
    while time.time() < explore_deadline:
        attempts += 1
        # length choice: favour moderate lengths (1200‑2500) for speed
        n = int(rng.integers(1200, 2500))
        a = _initial_vector(n, rng, warm_seq)
        peak = _max_convolution(a)
        if peak < best_score:
            best_score = peak
            best_seq = a.copy()
            print(f"[explore] attempt={attempts:4d} n={n:4d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based descent on the current best length
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform sequence of a modest size
        n = 1500
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = _max_convolution(best_seq)

    best_seq = _lp_optimize(best_seq, deadline - 10.0)

    # ------------------------------------------------------------------
    # Phase 3 – progressive scaling to larger lengths with LP then Adam
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        cur_n = len(best_seq)
        # stop scaling once we reach ~8000 (larger would make LP expensive)
        if cur_n >= 8000:
            break
        # target length: increase by factor ~1.7 but cap at 8000
        target_n = min(8000, int(cur_n * 1.7))
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # a few LP iterations on the new length
        a_scaled = _lp_optimize(a_scaled, deadline - 10.0)

        # short Adam refinement to polish the scaled vector
        remaining = deadline - time.time()
        if remaining > 30.0:
            slice_deadline = time.time() + min(remaining * 0.4, 120.0)
            a_scaled, peak = _adam_optimize(a_scaled, slice_deadline, rng)
            print(f"[refine] scaled to n={target_n:4d} → peak={peak:.6f}")

        best_seq = a_scaled
        best_score = _max_convolution(best_seq)
        print(f"[scale] now n={target_n:4d}, peak={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final Adam fine‑tuning on the (largest) sequence
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        slice_deadline = time.time() + remaining * 0.9
        best_seq, final_peak = _adam_optimize(best_seq, slice_deadline, rng)
        best_score = final_peak
        print(f"[final‑adam] peak after fine‑tune = {final_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – external sanity check
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}   internal max = {best_score:.6f}   length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
