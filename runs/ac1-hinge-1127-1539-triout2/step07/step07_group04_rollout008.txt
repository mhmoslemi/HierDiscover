"""
Propose a step‑function minimizing ``evaluate_sequence``.

Core ideas
----------
* **Diverse random initialisers** – the sampler now includes a
  “U‑shaped’’ (concave) family that places most mass at the ends,
  which is beneficial because the central convolution entries then
  receive fewer large contributions.
* **Targeted mass‑transfer hill‑climbing** – instead of the fully
  stochastic moves of the previous version, each iteration deterministically
  picks the index that contributes most to the current worst convolution
  entry and moves mass to a location that does not affect that entry (or
  has the smallest partner).  This aggressively lowers the maximum
  autocorrelation value.
* **Projected gradient descent** – a smoothed log‑sum‑exp surrogate
  (as before) is applied both before and after hill‑climbing to smooth
  the landscape and to polish the final solution.
* **Length schedule with adaptive budgeting** – we start at modest
  lengths and progressively double the size, re‑using the best solution
  from the previous stage (linear up‑sampling).  The time budget is
  split dynamically, giving the larger problems more resources.
* **Final polishing** – any remaining time is spent on a high‑β
  gradient descent on the best vector found.

The algorithm respects the 1000 s wall‑clock limit, never returns a
sequence containing negative or out‑of‑range values, and projects onto the
ℓ₁‑simplex {a_i ≥ 0, Σ a_i = √(2 n)} after every move, ensuring that the
official evaluator sees a valid step function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p), p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max (log‑sum‑exp surrogate).
    Uses momentum, annealed inverse temperature β and occasional jitter.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.3

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers (more families, including U‑shaped)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    Several families are sampled (power‑law, exponential,
    double‑exponential, Dirichlet, sine, Gaussian, concave quadratic,
    piecewise) to diversify the search space.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 9)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine wave (positive half‑wave)
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian‑like bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    elif family == 6:                   # double‑exponential (U‑shaped)
        lam = rng.exponential(scale=0.02) + 0.001
        a = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))

    elif family == 7:                   # concave quadratic (edges heavy)
        a = (np.abs(idx - (n - 1) / 2) + 1.0) ** 2
        a = np.max(a) - a + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Targeted mass‑transfer hill climbing (deterministic + annealing)
# ----------------------------------------------------------------------


def _targeted_hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Deterministic hill climbing that moves mass from the index that
    contributes most to the current worst convolution entry towards a
    location that has little or no influence on that entry.
    A tiny uphill‑acceptance probability provides occasional escapes.
    The full convolution is updated in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    # occasional full re‑projection to avoid drift
    proj_counter = 0

    while time.time() < deadline:
        # ---- locate worst convolution index ----------
        M = int(np.argmax(b))

        # ---- source index: maximise contribution a[i] * a[M-i] ----------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        i = int(idx[np.argmax(contrib)])

        # ---- destination index: prefer zero‑contribution indices ----------
        # indices j with M-j out of range give zero contribution to b[M]
        zero_candidates = []
        # j < M-(n-1)  →  M-j > n-1  (out of range)
        if M - (n - 1) > 0:
            up = M - n
            if up >= 0:
                zero_candidates.append(np.arange(0, up + 1, dtype=int))
        # j > M  →  M-j < 0
        if M + 1 < n:
            zero_candidates.append(np.arange(M + 1, n, dtype=int))

        if zero_candidates:
            cand = np.concatenate(zero_candidates)
            # choose the smallest‑mass index among them
            j = int(cand[np.argmin(a[cand])])
        else:
            # all j affect b[M]; pick the one whose partner a[M-j] is minimal
            partner = np.full(n, np.inf)
            valid = (M - np.arange(n) >= 0) & (M - np.arange(n) < n)
            partner[valid] = a[M - np.arange(n)[valid]]
            j = int(np.argmin(partner))
            if j == i:                     # avoid self‑move
                partner[j] = np.inf
                j = int(np.argmin(partner))
                if j == i:
                    # fallback random
                    j = rng.integers(0, n)
                    while j == i:
                        j = rng.integers(0, n)

        if a[i] <= 1e-14 or i == j:
            continue

        # ---- amount of mass to move ----------
        max_delta = a[i] * 0.7                # up to 70 % of source mass
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- update convolution (linear part) ----------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # ---- quadratic correction ----------
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance test ----------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            # tiny uphill probability that decays with time
            temperature = max(1e-6, (deadline - time.time()) / (deadline - start_time))
            if rng.random() < np.exp(-(new_max - cur_max) / (temperature + 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] target_hc improvement: {best_max:.6f}")

        # ---- periodic re‑projection to keep numerical drift under control ----------
        proj_counter += 1
        if proj_counter % 1024 == 0:
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            b = _full_convolution(a)
            cur_max = b.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a, b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    (as a plain Python list) found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin
    global start_time
    start_time = time.time()

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – increasingly larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng)

        a = _project_to_simplex(a, target_sum)

        # --------------------------------------------------------------
        # Phase A – a short burst of projected gradient descent (several restarts)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        gd_budget = min(remaining * 0.25, 120.0)     # give roughly 25 % of remaining time
        n_restarts = 4 if n <= 10000 else 2
        best_a_gd = None
        best_gd_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            now = time.time()
            per_restart = min(gd_budget / (n_restarts - r), deadline - now)
            gd_deadline = now + per_restart

            a_start = a.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_gd = _gradient_descent_opt(a_start, gd_deadline)
            score = _max_convolution_fft(a_gd)
            if score < best_gd_score:
                best_gd_score = score
                best_a_gd = a_gd.copy()

        if best_a_gd is not None and best_gd_score < best_score:
            best_score = best_gd_score
            best_seq = best_a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement for n={n} score={best_gd_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – targeted hill climbing
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        hc_budget = min(remaining * 0.55, 300.0)   # majority of time to hill‑climb
        hc_deadline = time.time() + hc_budget

        # ensure length matches
        if best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = best_seq.copy()
        a_current = _project_to_simplex(a_current, target_sum)
        b_full = _full_convolution(a_current)

        a_hc, b_hc = _targeted_hill_climbing(a_current, b_full, hc_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Targeted HC improvement for n={n} score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Phase C – final refinement with gradient descent (high‑β)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            gd2_deadline = time.time() + min(remaining * 0.20, 60.0)
            a_refine = _gradient_descent_opt(best_seq.copy(), gd2_deadline)
            score_ref = _max_convolution_fft(a_refine)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_refine.copy()
                print(f"[{time.time():.1f}] Post‑HC GD refinement n={n} score={score_ref:.6f}")

        # early exit if we have already reached the target
        if best_score <= 1.503:
            print(f"[{time.time():.1f}] Target reached (score={best_score:.6f}), stopping early.")
            break

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        final_deadline = deadline
        a_final = _gradient_descent_opt(a, final_deadline)
        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros (nice output)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
