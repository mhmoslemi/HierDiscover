def project_to_simplex(v, s):
    """Project vector v onto simplex {x >=0, sum x = s}."""
    # Use standard algorithm.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Step‑function optimizer (improved).

The search runs in three phases:

1.  Random exploration – many candidates are drawn from a richer
    collection of parametric families (power‑law, exponential,
    triangular, gamma, uniform, sinusoid, random block).  Each vector
    is normalised to the required ℓ₁‑norm √(2 n) and its self‑convolution
    peak is measured with an FFT.  The best candidate is kept.

2.  Refinement – the best candidate is interpolated to a few larger
    lengths.  For each length two complementary optimisers are applied
    inside the remaining time budget:
        * a simple projected sub‑gradient method that works on the
          simplex (log‑softmax with τ≈0 gives the exact sub‑gradient of
          the max‑convolution);
        * an Adam optimiser (log‑softmax parametrisation) identical to
          the one used previously but with a slightly larger learning
          rate and occasional jitter.
    After each optimiser the result is mixed with a uniform
    distribution and a sparsification step (tiny entries are zeroed)
    in order to escape plateaus.

3.  Final check – the best sequence found is evaluated with the
    black‑box ``evaluate_sequence`` function, stored globally as
    ``height_sequence_1`` and returned.

The whole routine respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (peak of a*a)."""
    return _full_convolution_fft(a).max()


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implementation of the algorithm from Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ≈0).
    Returns best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4                       # a bit larger than the reference

    # initialise temperature (soft‑max smoothing)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)       # start with a small amount of smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (hard max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9995                # shrink smoothing quickly
        lr *= 0.9998                 # very slow decay

        # occasional jitter to escape plateaus
        if it % 600 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak


def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the simplex.
    Uses the exact sub‑gradient of the max‑convolution (τ=0).
    Returns the best sequence encountered and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    a = a0.copy()
    best_a = a.copy()
    best_peak = _max_convolution(a)

    it = 0
    step0 = S * 0.5   # initial step size (tuned experimentally)

    while time.time() < deadline:
        it += 1
        g, peak = _softmax_gradient(a, tau=0.0)

        # keep the best seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # decreasing step size
        step = step0 / np.sqrt(it)

        # gradient step + projection onto simplex
        a_tmp = a - step * g
        a = _project_simplex(a_tmp, S)

        # occasional small random jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            a = _project_simplex(a + jitter, S)

    return best_a, best_peak

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    raw = mix * edge + (1.0 - mix) * np.ones(n)
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)
    mix = rng.uniform(0.0, 1.0)
    raw = mix * np.exp(-lam * d) + (1.0 - mix) * np.ones(n)
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw


def _init_sinusoid(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    freq = rng.uniform(0.5, 5.0)          # number of half‑waves across the interval
    phase = rng.uniform(0.0, 2 * np.pi)
    amp = rng.uniform(0.5, 5.0)
    raw = amp * np.abs(np.sin(2 * np.pi * freq * idx / n + phase))
    raw += rng.uniform(0.0, 0.3, size=n)  # small background
    return raw


def _init_block(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few random high blocks plus a tiny uniform background."""
    raw = np.zeros(n)
    num_blocks = rng.integers(1, 5)
    for _ in range(num_blocks):
        start = rng.integers(0, n)
        length = rng.integers(1, max(2, n // 10))
        height = rng.uniform(0.5, 5.0)
        end = min(n, start + length)
        raw[start:end] = height
    raw += rng.uniform(0.0, 0.1, size=n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best and otherwise samples
    from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'sinusoid', 'block'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'sinusoid':
        raw = _init_sinusoid(n, rng)
    else:  # block
        raw = _init_block(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the time budget and stores the final construction in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 45 % of the total budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.45 * (budget_s - 5.0)
    while time.time() < explore_deadline:
        attempt += 1
        # Length selection – favour moderate sizes but also try larger ones
        if rng.random() < 0.7:
            n = int(rng.integers(2000, 10000))
        else:
            n = int(rng.integers(10000, 20000))
        a0 = _initial_vector(n, rng, base_seq)
        peak = _max_convolution(a0)               # internal score (sum normalised)
        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:5d} n={n:5d}  new best={best_score:.6f}")

    # Fallback (should never happen)
    if best_seq is None:
        n = 3000
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – scaling & long‑run refinement
    # ------------------------------------------------------------------
    # Candidate lengths (original and a few larger ones)
    lengths = [best_n]
    if best_n < 12000:
        lengths.append(min(20000, best_n * 2))
    if best_n < 8000:
        lengths.append(int(best_n * 1.5))
    lengths = sorted(set(lengths))

    for target_n in lengths:
        remaining = end_time - time.time()
        if remaining < 30.0:
            break

        # ----- prepare scaled start -----
        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # ----- sub‑gradient refinement -----
        slice_time = min(remaining * 0.35, 180.0)
        a_sub, peak_sub = _subgradient_optimize(a_scaled,
                                                time.time() + slice_time,
                                                rng)
        if peak_sub < best_score:
            best_score = peak_sub
            best_seq = a_sub.copy()
            best_n = target_n
            print(f"[subgrad] target_n={target_n:5d} improved={best_score:.6f}")

        # ----- mixing with uniform (helps escape shallow minima) -----
        uniform = np.full(target_n, S / target_n)
        a_mix = 0.9 * best_seq + 0.1 * uniform
        a_mix = _project_simplex(a_mix, S)
        peak_mix = _max_convolution(a_mix)
        if peak_mix < best_score:
            best_score = peak_mix
            best_seq = a_mix.copy()
            print(f"[mix]    target_n={target_n:5d} improved={best_score:.6f}")

        # ----- Adam refinement (more aggressive) -----
        remaining = end_time - time.time()
        if remaining < 30.0:
            break
        slice_time = min(remaining * 0.35, 180.0)
        a_adam, peak_adam = _adam_optimize(best_seq,
                                            time.time() + slice_time,
                                            rng)
        if peak_adam < best_score:
            best_score = peak_adam
            best_seq = a_adam.copy()
            print(f"[adam]   target_n={target_n:5d} improved={best_score:.6f}")

        # ----- sparsification (zero tiny entries) -----
        thr = np.max(best_seq) * 0.001
        sparse = best_seq.copy()
        sparse[sparse < thr] = 0.0
        sparse = _project_simplex(sparse, np.sqrt(2.0 * best_n))
        peak_sparse = _max_convolution(sparse)
        if peak_sparse < best_score:
            best_score = peak_sparse
            best_seq = sparse.copy()
            print(f"[sparse] target_n={target_n:5d} improved={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
