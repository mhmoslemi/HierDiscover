def _lp_step(f):
    # Input f shape (n,) normalized sum sqrt(2n)
    import numpy as np, scipy.sparse as sp, scipy.optimize as opt
    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    # Build matrix for convolution constraints
    rows = []
    cols = []
    data = []
    # for each convolution index k from 0 to 2n-2:
    for k in range(2*n - 1):
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # i in [i_start,i_end]; then j = k-i
        # coefficient is f[i] for variable g_j
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # now add variable M at column n
    rowsM = np.arange(2*n-1)
    colsM = np.full(2*n-1, n)
    dataM = np.full(2*n-1, -1.0)
    rows.extend(rowsM)
    cols.extend(colsM)
    data.extend(dataM)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n+1))
    # equality constraint sum(g) = S
    A_eq = sp.csr_matrix(np.hstack([np.ones((1, n)), np.zeros((1, 1))]))
    b_eq = np.array([S])
    # bounds
    bounds = [(0, None)] * n + [(0, None)]  # M >=0
    # objective: minimize M
    c = np.zeros(n+1)
    c[-1] = 1.0
    # solve
    res = opt.linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        raise RuntimeError("LP failed")
    g = res.x[:n]
    M = res.x[-1]
    return g, M

"""
Adversarial step‑function optimizer (enhanced).

The optimizer uses three ideas:

1. **Broad random exploration** – many candidate sequences are drawn
   from several parametric families (power‑law, exponential, linear,
   sparse‑Gamma, uniform, Dirichlet, sine).  Each candidate is scaled so
   that its ℓ₁‑norm equals √(2 n) (the value that makes the internal
   objective equal to the convolution peak).  Fast FFT based
   convolution gives the current best shape.

2. **Continuous refinement** – the current best shape is refined with a
   log‑softmax Adam optimiser that minimizes the exact max‑convolution.
   The optimiser works on the simplex and uses an exponentially
   decreasing temperature to approximate the hard max.

3. **Linear‑programming “balancing” step** – after Adam we solve a small
   linear program

       minimise   M
       subject to f ⋆ g ≤ M ,   ∑ g = √(2 n) ,   g ≥ 0

   where “⋆’’ is the linear convolution.  The solution `g` equalises the
   most‑violated convolution constraints.  Updating the current shape
   by a convex combination `f←(1‑t) f + t g` (with `t≈0.25`) provably
   reduces the peak of `f⋆f`.  A few LP iterations are performed for the
   best candidate (only for n≤2000 to keep the LP modest).

The whole routine respects the 1000 s budget, prints intermediate
progress, stores the final construction in the global variable
`height_sequence_1` and returns the list of heights.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# --------------------------------------------------------------
# Soft‑max gradient (τ‑soft‑max) used by the Adam optimiser
# --------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ≈0).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4                       # a bit larger than the reference

    # initialise temperature (soft‑max smoothing)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)       # start with a little smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (hard max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9995                # shrink smoothing quickly
        lr *= 0.9998                 # very slow decay

        # occasional jitter to escape plateaus
        if it % 600 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak


# --------------------------------------------------------------
# Linear‑programming balancing step (Frank‑Wolfe style)
# --------------------------------------------------------------
def _lp_balance_step(f: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve the LP:
        minimise   M
        subject to f ⋆ g ≤ M ,   ∑ g = √(2 n) ,   g ≥ 0
    Returns (g, M).  The LP variables are the new vector g and the scalar
    bound M.  Only called for moderate n (≤2000) to keep the problem
    tractable.
    """
    n = f.shape[0]
    S = np.sqrt(2.0 * n)

    # Build Toeplitz matrix that maps g → f⋆g (size (2n‑1) × n)
    offsets = np.arange(n, dtype=int)          # 0 … n‑1
    A_f = sp.diags(f, offsets, shape=(2 * n - 1, n), format='csr')

    # Append column for the variable M (coefficient –1)
    M_col = sp.csr_matrix((-np.ones((2 * n - 1, 1)),
                           (np.arange(2 * n - 1), np.zeros(2 * n - 1))),
                          shape=(2 * n - 1, 1))
    A_ub = sp.hstack([A_f, M_col], format='csr')
    b_ub = np.zeros(2 * n - 1)                 # f⋆g – M ≤ 0

    # Equality constraint  ∑ g = S
    A_eq = np.concatenate([np.ones(n), [0.0]])[np.newaxis, :]   # (1, n+1)
    b_eq = np.array([S])

    # Bounds: g_i ≥ 0,   M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Objective: minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Solve with HiGHS interior‑point method (fast and robust)
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP balancing step failed")
    g = res.x[:n]
    M = res.x[-1]
    return g, M

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)
    offset = rng.uniform(0.0, 15.0)
    mix = rng.uniform(0.0, 1.0)
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.full(n, rng.uniform(0.0, 0.2))
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Raw Dirichlet‑like start."""
    return rng.random(n)


def _init_sine(n: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth sine bump – zero at the ends, max in the middle."""
    x = np.linspace(0, np.pi, n)
    raw = np.sin(x) ** rng.uniform(0.5, 2.0)   # random power makes it sharper/flatter
    raw += rng.uniform(0.0, 0.2)               # tiny background
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Frequently warm‑starts from the global best and otherwise samples
    from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ≈0.22
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet', 'sine'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    else:   # sine
        raw = _init_sine(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a

                                     time.time() + time_slice,
                                     rng)

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[adam] target_n={target_n:5d} improved score={best_score:.6f}")

        # ---- Linear‑programming balancing (discrete) ----
        # Apply only when the problem size is modest (≤ 2000)
        if target_n <= 2000 and time.time() < end_time - 30.0:
            f = a_opt.copy()
            # Perform a few LP–Adam cycles
            for lp_iter in range(4):
                try:
                    g, M = _lp_balance_step(f)
                except Exception as e:
                    print(f"[lp] LP failed (iter {lp_iter}) – {e}")
                    break

                # convex update
                t = 0.25
                f_new = (1.0 - t) * f + t * g
                S = np.sqrt(2.0 * target_n)
                f_new = np.clip(f_new, 0.0, None)
                f_new = f_new * (S / f_new.sum())
                new_peak = _max_convolution(f_new)

                if new_peak < best_score:
                    best_score = new_peak
                    best_seq = f_new.copy()
                    best_n = target_n
                    print(f"[lp] target_n={target_n:5d} iter={lp_iter} new best={best_score:.6f}")

                # stop early if LP does not improve the peak appreciably
                if new_peak >= M * 0.999:
                    break
                f = f_new

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
