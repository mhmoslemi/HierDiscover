def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 2.0

    # retrieve warm start
    base_seq = globals().get('height_sequence_1')
    if base_seq is not None:
        try:
            base_seq = np.asarray(base_seq, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf

    # exploration
    explore_until = end_time - 300.0  # allocate 300s for refinement at end

    attempts = 0
    while time.time() < explore_until:
        attempts += 1
        # Choose n
        if rng.random() < 0.6:
            n = int(rng.integers(2000, 8000))
        else:
            n = int(rng.integers(8000, 25000))

        a0 = _initial_vector_n(n, rng, base_seq)  # we define this to use many families

        # compute max conv
        peak = _max_convolution(a0)
        if peak < best_score:
            best_score = peak
            best_seq = a0
            best_n = n
            print(f"[explore] attempt={attempts} n={n} new best={best_score:.6f}")

    # After exploration, if best_seq None fallback
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, np.sqrt(2 * n) / n)
        best_score = _max_convolution(best_seq)
        best_n = n

    # multi-scale refinement
    # define lengths list: start best_n, then up to max_n (maybe 24000)
    lengths = [best_n]
    max_n = min(24000, best_n * 8)  # maybe larger
    # Add scaling steps: multiply by 1.5,2,3,4...
    cur = best_n
    while cur < max_n:
        cur = int(cur * 1.7)  # factor
        if cur > max_n:
            cur = max_n
        if cur not in lengths:
            lengths.append(cur)
    lengths = sorted(set(lengths))

    # For each target length, perform multiple (e.g., 2) Adam runs
    for target_n in lengths:
        remaining = end_time - time.time()
        if remaining < 30.0:
            break

        # Determine time slice for this length: allocate proportionally to n
        slice_time = min(remaining * 0.25, 150.0)  # use at most 150 s per length

        a_scaled = _scale_sequence(best_seq, target_n)
        # Normalize sum to sqrt(2 n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        # Run a few Adam restarts
        num_restarts = 3
        best_local_seq = a_scaled
        best_local_score = _max_convolution(a_scaled)

        for r in range(num_restarts):
            # add small jitter to avoid same start
            jitter = rng.normal(scale=S*0.02, size=target_n)
            jitter = np.maximum(jitter, 0.0)
            start_seq = best_local_seq + jitter
            start_seq = start_seq.clip(min=0)
            start_seq = start_seq * (S / start_seq.sum())

            # allocate time for this run
            run_deadline = time.time() + slice_time / num_restarts
            opt_seq, opt_peak = _adam_optimize(start_seq, run_deadline, rng)

            if opt_peak < best_local_score:
                best_local_score = opt_peak
                best_local_seq = opt_seq.copy()

        # After restarts, compare with global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n} improved global score={best_score:.6f}")

        # Optional local hill-climbing descent on best_seq for a few seconds
        # using small random mass moves

        remaining_local = end_time - time.time()
        if remaining_local > 5.0:
            # allocate short time for hill climb
            hc_deadline = time.time() + min(remaining_local * 0.1, 20.0)

"""
Adversarial step‑function optimizer (enhanced).

The algorithm proceeds in three stages:
1. Random exploration of many parametric families (power‑law, exponential,
   triangular, double‑edge, sparse block, etc.).  Each candidate is
   normalised to satisfy ∑a_i = √(2 n) and its self‑convolution peak
   max(a∗a) is measured.  The best candidate seen so far is kept.
2. Multi‑scale refinement – the current best candidate is interpolated to
   a series of larger lengths (up to ≈ 2·10⁴).  For each length a few
   independent Adam optimiser runs are performed on the simplex
   (log‑softmax parametrisation) using the exact sub‑gradient of the
   max‑convolution (τ‑soft‑max with a decreasing temperature).  The best
   refined candidate replaces the global best.
3. A short stochastic hill‑climbing phase performs elementary mass‑move
   operations on the best vector to shave off any remaining peak.
Finally the sequence is returned and stored in the global variable
``height_sequence_1``.  The whole routine respects the supplied time
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ≈0).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.6                       # slightly larger initial learning rate

    # Initialise temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)       # start with a modest amount of smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # Recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # Convolution & gradient (soft max)
        g, peak = _softmax_gradient(a, tau)

        # Keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # Gradient with respect to logits (chain rule)
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # Optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Schedules
        tau *= 0.9995                # shrink smoothing quickly
        lr *= 0.9998                 # very slow decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)          # exponent
    offset = rng.uniform(0.0, 15.0)    # shift of the denominator
    mix = rng.uniform(0.0, 1.0)        # blending factor
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.03)   # decay rate
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)                     # height at the edge

    mix = rng.uniform(0.0, 1.0)

# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing to polish a candidate
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a very cheap local search: move a small amount of mass from a
    random index to another random index if the max‑convolution improves.
    """
    n = len(seq)
    S = seq.sum()
    best_seq = seq.copy()
    best_score = _max_convolution(seq)

    # step size as a small fraction of the total ℓ₁ norm
    step = max(S * 0.01, 1e-6)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # maximum amount transferable from i
        max_delta = min(best_seq[i], step)
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _max_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
    return best_seq, best_score

"""
Adversarial step‑function optimizer (enhanced).

The algorithm proceeds in three stages:
1. Random exploration of many parametric families (power‑law, exponential,
   triangular, double‑edge, sparse block, etc.).  Each candidate is
   normalised to satisfy ∑a_i = √(2 n) and its self‑convolution peak
   max(a∗a) is measured.  The best candidate is retained.
2. Multi‑scale refinement – the current best vector is interpolated to a
   series of larger lengths (up to ≈ 2.4·10⁴).  For each length a few
   independent Adam optimiser runs are performed on the simplex
   (log‑softmax parametrisation) using the exact sub‑gradient of the
   max‑convolution (τ‑soft‑max with a decreasing temperature).  The best
   refined candidate replaces the global best.
3. A short stochastic hill‑climbing phase performs elementary mass‑move
   operations on the best vector to shave off any remaining peak.
Finally the sequence is stored in the global variable ``height_sequence_1``
and returned.  The routine respects the supplied time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask ∗ a_rev)
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ≈0).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.6                       # a bit larger than the reference

    # initialise temperature (soft‑max smoothing)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)       # start with a modest amount of smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (soft max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9995                # shrink smoothing quickly
        lr *= 0.9998                 # very slow decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.5)          # exponent
    offset = rng.uniform(0.0, 15.0)    # shift of the denominator
    mix = rng.uniform(0.0, 1.0)        # blending factor
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.03)   # decay rate
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)                     # height at the edge
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))  # controls steepness
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n) * rng.uniform(0.0, 0.2)
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.random(n)
    return raw


def _init_double_edge(n: int, rng: np.random.Generator) -> np.ndarray:
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    exp = rng.uniform(2.0, 5.0)
    offset = rng.uniform(0.0, 12.0)
    edge = (d + offset) ** (-exp)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_block_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    block_frac = rng.uniform(0.05, 0.3)
    block_len = max(1, int(block_frac * n))
    start = rng.integers(0, n - block_len + 1)
    raw = np.full(n, rng.uniform(0.0, 0.1))
    raw[start:start + block_len] = 1.0
    return raw


def _initial_vector_n(n: int,
                     rng: np.random.Generator,
                     warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best and otherwise samples
    from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with modest probability
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0)
            total = n
        a = a * (S / total)
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet',
                        'double_edge', 'block'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    elif style == 'dirichlet':
        raw = _init_dirichlet(n, rng)
    elif style == 'double_edge':
        raw = _init_double_edge(n, rng)
    elif style == 'block':
        raw = _init_block_sparse(n, rng)
    else:
        raw = _init_uniform(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Cheap local search: move a small amount of mass between two random indices
    if it reduces the max‑convolution.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_score = _max_convolution(seq)

    # step size as a modest fraction of the total ℓ₁ norm
    step = max(best_seq.sum() * 0.01, 1e-9)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = min(best_seq[i], step)
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _max_convolution(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
    return best_seq, best_score
