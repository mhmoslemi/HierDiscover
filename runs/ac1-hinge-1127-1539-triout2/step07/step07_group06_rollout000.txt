"""
Propose a step‑function (list of non‑negative heights) that minimizes
the evaluation function defined in the challenge.

Algorithm
---------
1.  For several problem sizes (n ≈ 2000–5000) perform many short
    gradient‑descent runs on a smooth soft‑max approximation of the
    max‑convolution.  Each run starts from a random power‑law
    distribution and runs for a few seconds, using projection onto the
    ℓ¹‑simplex Σa = √(2·n) after every step.
2.  The best sequence found for each size is kept.  After the allotted
    time is exhausted a short incremental local‑search (mass‑transfer)
    polish is applied to the overall best candidate.
3.  The final sequence is truncated of trailing zeros and returned.

The code is fully self‑contained, uses only NumPy and the standard
library, and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (assumes Σa = √(2·n))."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value for a (max of self‑convolution)."""
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Soft‑max gradient of max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (gradient, full convolution array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # probability distribution over b

    # gradient = 2 * (a_rev * p) evaluated at the n positions
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Annealed gradient descent (single run)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_single(a0: np.ndarray,
                                      deadline: float,
                                      rng: np.random.Generator):
    """
    Projected gradient descent on the soft‑max approximation.
    Runs until `deadline` and returns the best sequence seen.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1e6
    step_base = target_sum * 0.25   # relatively aggressive step size
    it = 0

    while time.time() < deadline:
        # gradient of soft‑max
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0.0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        # projection onto the ℓ1‑simplex Σa = √(2·n)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow local minima
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # evaluate current max‑convolution
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD (n={n}) β={beta:.2f} best={best_val:.6f}")

        # schedule β upwards
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental mass‑transfer local search (used for final polish)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer an amount `delta` from `donor` to `receiver`.
    Updates the coefficient vector `a` and its full convolution `conv`
    exactly (including quadratic `δ²` terms).
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    a_before = a.copy()

    # linear part of the update
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic part
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # update coefficients
    a[donor] -= delta
    a[receiver] += delta
    if a[donor] < 0.0:
        a[donor] = 0.0

    # keep convolution non‑negative (tiny negatives may appear due to rounding)
    np.maximum(conv, 0.0, out=conv)


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Fast incremental local‑search (mass‑transfer + smoothing + jitter).
    Returns the best sequence and its max‑convolution.
    """
    target = np.sqrt(2.0 * n)

    # initialise
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        # annealing schedules
        max_frac = 0.5 * (0.005 / 0.5) ** tau            # max fraction of donor mass
        temp = 0.5 * (1e-5 / 0.5) ** tau                # temperature for Metropolis

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass transfer (≈80 % of moves)
        # --------------------------------------------------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]                  # contribution of each index
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            # bias receiver toward small coefficients
            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())

            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            # Metropolis acceptance
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] LS new best max={best_max:.6f}")
            else:
                # revert
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing window (≈10 %)
        # --------------------------------------------------------------
        elif move_type < 0.95:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_window + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] LS smoothing improved max={best_max:.6f}")
                else:
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) Gaussian jitter (≈10 %)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] LS jitter improved max={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the given evaluation metric.

    The routine performs many short annealed gradient‑descent runs for a
    selection of sequence lengths, keeps the best candidate and then
    applies a brief incremental local‑search polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # keep a safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Optional warm‑start from a previously discovered sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n0 = seq0.size
        if time.time() < deadline - 5.0:
            run_deadline = time.time() + min(20.0,
                                             (deadline - time.time()) * 0.15)
            seq, score = _local_search(n0, run_deadline, rng, init_seq=seq0)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] warm‑start n={n0} -> score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Gradient‑descent phase on several lengths
    # ------------------------------------------------------------------
    # Choose a modest set of lengths; larger n gives more flexibility.
    lengths = [2500, 3500, 4500]   # can be tuned
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # allocate a slice of the remaining time
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.3, 180.0)   # up to three minutes per n
        slice_deadline = time.time() + slice_time

        # run many short restarts within the slice
        while time.time() < slice_deadline - 5.0:
            # random power‑law exponent
            p = rng.uniform(0.3, 0.9)
            target = np.sqrt(2.0 * n)
            a0 = _power_law_sequence(n, p, target)

            # short jitter before optimisation (helps diversify)
            a0 += rng.normal(scale=target * 0.001, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target)

            # each GD run gets a few seconds
            gd_deadline = time.time() + min(8.0,
                                            slice_deadline - time.time() - 5.0)
            seq_candidate, cand_score = _annealed_gradient_descent_single(
                a0, gd_deadline, rng)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq_candidate.copy()
                print(f"[{time.time():.1f}] new global best n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing with incremental local search (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n_best = best_seq.size
        polish_deadline = time.time() + (deadline - time.time()) * 0.25
        seq_polished, polished_score = _local_search(
            n_best, polish_deadline, rng, init_seq=best_seq)
        if polished_score < best_score:
            best_score = polished_score
            best_seq = seq_polished.copy()
            print(f"[{time.time():.1f}] polished best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Trim trailing zeros (they do not affect the objective)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: a tiny uniform sequence (guarantees a finite score)
        n = 10
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best ≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
