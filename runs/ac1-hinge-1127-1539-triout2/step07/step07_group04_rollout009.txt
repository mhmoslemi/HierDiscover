"""
Greedy hill‑climbing step‑function optimizer.

The evaluator `evaluate_sequence` simply returns the maximum entry of the
self‑convolution, because the ℓ₁–normalisation forces the prefactor to 1.
Hence the task is: minimise  maxₖ (a∗a)ₖ  subject to aᵢ≥0 and Σaᵢ=√(2 n).

The algorithm proceeds in stages of increasing length.  For every length
`n`

* initialise a candidate vector on the ℓ₁‑simplex (either by up‑sampling the
  best solution from the previous stage or by a random family);
* compute its full linear convolution `b = a∗a`;
* run a **deterministic greedy hill‑climber** that repeatedly
  – finds the worst convolution index `M`;
  – chooses a source index `i` that contributes most to `b[M]`;
  – chooses a destination index `j` among the currently smallest entries;
  – moves as much mass as possible from `i` to `j` while keeping the new
    maximum as low as possible (exact O(n) update of the whole convolution);
  – accepts the move only if it strictly improves the current maximum;
  – falls back to a small random move if no greedy improvement exists.
  The convolution is kept up‑to‑date using the exact O(n) formula, and
  recomputed from scratch every few hundred moves to avoid numerical drift.
* after the greedy phase a short high‑β projected‑gradient descent (the
  log‑sum‑exp surrogate) is applied for polishing.

The whole search respects a wall‑clock budget (`budget_s`).  The best
vector found before the deadline is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, trimmed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (various families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)
    using a mixture of popular families.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # sine half‑wave
        a = np.sin(np.pi * (idx + 0.5) / n) ** 2 + 1e-12

    elif family == 5:                   # Gaussian bump
        mu = n / 2.0
        sigma = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-12

    else:                               # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, then project onto the
    appropriate ℓ¹‐simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used only for the final polish
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a (full convolution).
    Returns (grad, b) with grad = 2·(a ⋆ p), pₖ ∝ exp(β·bₖ).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1   # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth log‑sum‑exp surrogate.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.3

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Greedy deterministic hill‑climbing
# ----------------------------------------------------------------------


def _apply_move(a: np.ndarray, b: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Apply a move that transfers `delta` mass from index i to j.
    Both ``a`` and ``b`` are updated in‑place.
    """
    n = a.size
    # update convolution (linear part)
    b[i:i + n] -= 2.0 * delta * a
    b[j:j + n] += 2.0 * delta * a

    # quadratic correction
    k2i = 2 * i
    k2j = 2 * j
    kij = i + j
    if k2i < b.size:
        b[k2i] += delta * delta
    if k2j < b.size:
        b[k2j] += delta * delta
    if kij < b.size:
        b[kij] -= 2.0 * delta * delta

    # finally update the vector a itself
    a[i] -= delta
    a[j] += delta


def _greedy_hill_climb(a: np.ndarray, b: np.ndarray, deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Deterministic hill‑climbing that always moves mass from the index most
    responsible for the current worst convolution entry to a low‑mass index.
    The convolution `b` is kept up‑to‑date with O(n) updates.
    """
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    moves = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # locate the worst convolution entry
        # --------------------------------------------------------------
        M = int(np.argmax(b))
        cur_val = b[M]

        # a quick second‑largest value (mask out the maximum)
        mask = np.ones(b.shape[0], dtype=bool)
        mask[M] = False
        second_max = b[mask].max()

        # --------------------------------------------------------------
        # pick a source index that contributes most to entry M
        # --------------------------------------------------------------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            # no mass to move – abort this stage
            break

        # --------------------------------------------------------------
        # destination candidates (few smallest entries + a few random)
        # --------------------------------------------------------------
        k_small = min(12, n)
        small = np.argpartition(a, k_small)[:k_small]
        randoms = rng.integers(0, n, size=5)
        cand = np.unique(np.concatenate((small, randoms)))
        cand = cand[cand != i]

        # --------------------------------------------------------------
        # evaluate all candidates and keep the best improvement
        # --------------------------------------------------------------
        best_new_max = cur_val
        best_j = None
        best_delta = 0.0

        for j in cand:
            # gradient of b[M] w.r.t. moving mass from i to j
            val_i = a[M - i] if 0 <= M - i < n else 0.0
            val_j = a[M - j] if 0 <= M - j < n else 0.0
            grad = 2.0 * (val_j - val_i)          # < 0  → reduces the max

            if grad >= -1e-15:
                continue  # not helpful for this destination

            # we would like to bring the current maximum down towards
            # second_max (or a little below it)
            target = second_max
            needed = (target - cur_val) / grad    # both numerator & grad negative → positive
            delta = min(a[i], needed) if needed > 0 else a[i] * 0.5
            if delta <= 0:
                continue

            # compute the new maximum after the tentative move (O(n))
            b_tmp = b.copy()
            b_tmp[i:i + n] -= 2.0 * delta * a
            b_tmp[j:j + n] += 2.0 * delta * a
            k2i = 2 * i
            k2j = 2 * j
            kij = i + j
            if k2i < b_tmp.size:
                b_tmp[k2i] += delta * delta
            if k2j < b_tmp.size:
                b_tmp[k2j] += delta * delta
            if kij < b_tmp.size:
                b_tmp[kij] -= 2.0 * delta * delta

            new_max = b_tmp.max()
            if new_max < best_new_max:
                best_new_max = new_max
                best_j = j
                best_delta = delta

        # --------------------------------------------------------------
        # perform the best deterministic move if it improves the maximum
        # --------------------------------------------------------------
        if best_j is not None and best_delta > 0 and best_new_max < cur_max - 1e-12:
            _apply_move(a, b, i, best_j, best_delta)
            cur_max = best_new_max
            moves += 1
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] greedy improvement: max={best_max:.6f}")
            # occasional full recomputation to keep numerical errors in check
            if moves % 400 == 0:
                b = _full_convolution(a)
            continue

        # --------------------------------------------------------------
        # fallback: a small random move (similar to the original code)
        # --------------------------------------------------------------
        # source drawn proportionally to its contribution to M
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i_rand = rng.choice(idx, p=probs)
        else:
            i_rand = rng.integers(0, n)

        j_rand = rng.integers(0, n)
        if i_rand == j_rand or a[i_rand] < 1e-14:
            # nothing sensible to do – give up
            break

        max_delta = a[i_rand] * 0.45
        delta = rng.random() * max_delta
        _apply_move(a, b, i_rand, j_rand, delta)
        cur_max = b.max()
        moves += 1
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time():.1f}] random‑move improvement: max={best_max:.6f}")

        if moves % 400 == 0:
            b = _full_convolution(a)

    return best_a, _full_convolution(best_a)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best sequence
    found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional seeding from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – progressively larger dimensions
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 40000, 60000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # initialise candidate for the current length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target)

        # --------------------------------------------------------------
        # Greedy hill‑climbing phase (the main workhorse)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        # allocate roughly 55 % of the remaining budget to the greedy phase
        hc_deadline = time.time() + min(remaining * 0.55, 500.0)

        # compute the initial full convolution
        b_init = _full_convolution(a_init)

        a_hc, b_hc = _greedy_hill_climb(a_init, b_init, hc_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] greedy hill‑climb improvement for n={n} score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Light polishing with high‑β gradient descent (if time permits)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:

            score_ref = _max_convolution_fft(a_refine)
            if score_ref < best_score:

    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        n = best_seq.size
        target = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target)
        a = _gradient_descent_opt(a, deadline)          # high‑β run till the end
        final_score = _max_convolution_fft(a)
        if final_score < best_score:
