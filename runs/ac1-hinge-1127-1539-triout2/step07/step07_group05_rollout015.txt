def reduce_max(seq, deadline):
    a = seq.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    # ensure sum = S
    a = a * (S / np.sum(a))  # scaling to maintain sum

    while time.time() < deadline:
        b = np.fft.ifft(np.fft.fft(a, n=next_pow2(2*n-1))**2).real[:2*n-1]
        max_idx = int(np.argmax(b))
        max_val = b[max_idx]

        # compute contributions for each i that can pair: i in [0,n), j = max_idx - i must be int in [0,n)
        # We'll compute product contributions vector p_i = a_i * a_j (if valid else 0)
        j_vals = max_idx - np.arange(n)
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_vals[mask]]

        # find index with largest product
        i_max = int(np.argmax(contrib))
        j_max = int(max_idx - i_max)

        # ensure both indices valid
        if a[i_max] <= 0.0 or a[j_max] <= 0.0:
            # fallback random move
            i_max = np.random.randint(0, n)
            j_max = max_idx - i_max
            if not (0 <= j_max < n):
                continue

        # Determine move amount: delta = alpha * a[i_max] where alpha in (0, 0.5)
        # Move from the larger of the two indices to a random other position with low contribution
        if a[i_max] > a[j_max]:
            src = i_max
        else:
            src = j_max

        delta = a[src] * 0.2  # move 20% mass
        # choose destination index dest with small product contributions at this k_max
        # compute contributions for possible dest indices
        # we can just pick random dest where a[dest] is low
        low_indices = np.where(a < np.percentile(a, 20))[0]
        if low_indices.size == 0:
            dest = np.random.randint(0, n)
        else:
            dest = np.random.choice(low_indices)

        # Transfer mass delta
        a[src] -= delta
        a[dest] += delta

        # Ensure nonnegativity (should hold)
        # Renormalize sum (should still be same within rounding)
        # Show improvement
        # compute new max after move (or do incremental update)
        b_new = np.fft.ifft(np.fft.fft(a, n=next_pow2(2*n-1))**2).real[:2*n-1]
        new_max = b_new.max()
        if new_max > max_val:
            # revert change (maybe accept probabilistically)
            a[src] += delta
            a[dest] -= delta
        else:
            # accept improvement
            # optionally reproject onto simplex if sum drift small due to numerical errors
            a = a * (S / np.sum(a))

    return a

"""
Propose a non‑negative step function (a list of heights) that minimises
``evaluate_sequence``.  The objective is 
        2 * n * max(conv(a,a)) / (sum(a)**2)
which, after normalising the ℓ¹‑norm to √(2 n), is simply 
        max(conv(a,a)).
The search proceeds in three stages:

1. **Random family generation** – power‑law, exponential, linear,
   quadratic, and a “spike” family (few random non‑zero entries).  The
   vectors are projected onto the ℓ¹‑simplex Σa = √(2 n).

2. **Continuous refinement** – projected gradient descent on a
   log‑sum‑exp surrogate of the maximum (smooth‑max) followed by a
   *mass‑transfer* local optimiser that repeatedly moves a fraction of
   mass from a pair contributing strongly to the current peak to a
   low‑contributing index.  Both steps are constrained to the simplex.

3. **Permutation annealing** – finally the ordering of the multiset is
   improved by simulated‑annealing swaps.

The routine respects a 1000 s budget, keeps an elite list of the best
candidates, and returns the best sequence found (as a plain ``list``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σ a = s , a ≥ 0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s} (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a * a  (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # log‑sum‑exp probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient = 2·(a ⋆ p)
    # we need convolution of a (reversed) with p
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = _next_pow2(conv_len)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent on the smoothed max.
    Returns a refined vector (still on the ℓ¹‑simplex).
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9        # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            # step size inversely proportional to norm and to current beta
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # project back onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 8e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta (sharpen surrogate)
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Local optimiser that moves mass from a high‑contributing entry
# to a low‑contributing one (still respecting the simplex).
# ----------------------------------------------------------------------
def _mass_transfer_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly transfers a fraction of the value from an index that
    participates strongly in the current maximal convolution entry to an
    index with a weak contribution.  Each move is accepted only if it
    does not increase the maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()

    # helper that returns (max_value, max_index, full_conv)
    def _state(vec):
        b = np.fft.irfft(np.fft.rfft(vec, n=_next_pow2(2 * n - 1))**2,
                         n=_next_pow2(2 * n - 1))[: 2 * n - 1]
        k = int(np.argmax(b))
        return float(b.max()), k, b

    cur_max, cur_k, cur_b = _state(a)

    while time.time() < deadline:
        # compute contributions of each i to the current maximal entry
        j = cur_k - np.arange(n)
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j[mask]]

        # source index = the larger participant of the largest product
        i_src = int(np.argmax(contrib))
        # choose the partner just for information (not used directly)
        # j_src = cur_k - i_src

        # destination – pick from the 20 % smallest entries
        low_mask = a < np.percentile(a, 20)
        if low_mask.any():
            i_dst = int(rng.choice(np.where(low_mask)[0]))
        else:
            i_dst = rng.integers(0, n)

        # amount to move (30 % of source)
        delta = 0.3 * a[i_src]
        if delta <= 0:
            # nothing to move – break out early
            break

        a[i_src] -= delta
        a[i_dst] += delta

        # Project back onto simplex (keeps the total sum exact)
        a = _project_to_simplex(a, target_sum)

        new_max, new_k, _ = _state(a)

        if new_max <= cur_max:
            cur_max, cur_k = new_max, new_k
        else:
            # reject – revert the move
            a[i_dst] -= delta
            a[i_src] += delta
            a = _project_to_simplex(a, target_sum)   # restore feasibility

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    The ℓ¹‑norm stays unchanged; only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to the current best score
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = cur_score <= best_score or (T > 0 and rng.random() < np.exp(-dE / T))

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny global jitter (2 % of positions reshuffled)
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if mask.any():
                shuffled = a[mask].copy()
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Families for random initialisation
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale ``v`` so that its ℓ¹‑norm equals ``target`` (preserving non‑negativity)."""
    s = np.sum(v)
    if s == 0:
        # avoid division by zero – give all mass to first entry
        v[0] = target
        return v
    a = v * (target / s)
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _quadratic_sequence(n: int, a0: float, b: float, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, a0 - b * idx - c * idx * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _spike_sequence(n: int, k: int) -> np.ndarray:
    """
    k random non‑zero entries (spikes).  The spike magnitudes are drawn from
    a uniform distribution and the whole vector is projected onto the simplex.
    """
    rng = np.random.default_rng()
    a = np.zeros(n, dtype=float)
    positions = rng.choice(n, size=min(k, n), replace=False)
    # give each spike a random positive weight
    weights = rng.random(positions.shape[0]) + 0.1
    a[positions] = weights
    return _project_to_simplex(a, np.sqrt(2.0 * n))
