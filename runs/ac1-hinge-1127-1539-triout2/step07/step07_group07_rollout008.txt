def local_search(a, time_deadline):

    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = max_convolution(best_a)
    T0 = best_score * 0.05 + 1e-4
    T = T0
    cooling = 0.9999
    rng = np.random.default_rng()
    while time.time() < time_deadline:
        # pick i with a_i > epsilon
        i = rng.integers(0, n)

def max_convolution(a):
    n = a.size
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)
    return float(conv[:2*n-1].max())
```

def _max_convolution(a: np.ndarray) -> float:
    """Return max entry of the full linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two

    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s , x ≥ 0   (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ_k exp(β·b_k)  where b = a*a (full convolution).

    Returns tuple (g, b) where
        g – gradient w.r.t. a (same shape as a)
        b – the full convolution (length 2·n‑1)
    """
    n = a.shape[0]

    # ---- full convolution a*a ---------------------------------------
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # ---- soft‑max weighting -----------------------------------------
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)          # shape (2n‑1,)

    # ---- gradient: g = 2·(a ⋆ p) ------------------------------------
    # convolution of a (reversed) with p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1                # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Runs a projected Adam optimiser for the log‑sum‑exp surrogate.
    Stops when the supplied deadline (unix time) is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 5e6                       # higher temperature for tighter surrogate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate that keeps the ℓ¹‑norm roughly stable
        lr = 0.5 * target_sum / ((beta + 1.0) * (np.sqrt(v_hat).mean() + 1e-8))

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps sum = √(2 n))
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal the soft‑max temperature
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Continuous simulated‑annealing: transfer a small amount of mass
# ----------------------------------------------------------------------


def _mass_transfer_sa(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated‑annealing on the *values* of a fixed length sequence.
    A move transfers a random ε from one index to another (preserving the ℓ¹‑norm).
    The acceptance rule is based on the true maximal convolution.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution(best)

    # initial temperature – a fraction of the current peak
    T = best_score * 0.03 + 1e-8
    cooling = 0.9997
    rng = np.random.default_rng()
    # tolerance for considering a move “small”
    eps_scale_initial = 0.25          # up to 25 % of source value in early stage

    while time.time() < deadline:
        # pick source and target indices
        i = rng.integers(0, n)
        if best[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # propose an ε proportional to the source magnitude
        eps = rng.random() * best[i] * eps_scale_initial

        # apply move
        a_new = best.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # compute new objective
        cur_score = _max_convolution(a_new)

        # Metropolis acceptance
        if cur_score <= best_score or rng.random() < np.exp(-(cur_score - best_score) / T):
            best = a_new
            best_score = cur_score

        # cool down
        T *= cooling

        # slowly shrink the step‑size scale so we converge
        eps_scale_initial *= 0.99995
        if eps_scale_initial < 1e-4:
            eps_scale_initial = 1e-4

    return best


# ----------------------------------------------------------------------
# Pure permutation annealing (keeps multiset unchanged)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Swap‑only simulated annealing on the ordering of a fixed multiset."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        cur_score = _max_convolution(a)

        if cur_score <= best_score or rng.random() < np.exp(-(cur_score - best_score) / T):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny random shuffles to escape plateaus
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.015
            if np.any(mask):
                sub = a[mask].copy()
                rng.shuffle(sub)
                a[mask] = sub
                cur_score = _max_convolution(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Helper: generate a random Dirichlet‑like vector
# ----------------------------------------------------------------------


def _random_dirichlet_like(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n whose entries sum to √(2 n).
    Distribution is Gamma(0.5,1) – many small values and a few larger spikes.
    """
    raw = rng.gamma(shape=0.5, scale=1.0, size=n)
    target = np.sqrt(2.0 * n)
    vec = raw * (target / raw.sum())
    return vec


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence``.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin
    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 0.  Seed with the external construction if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        ext = np.asarray(height_sequence_1, dtype=float)
        n_ext = ext.size
        target = np.sqrt(2.0 * n_ext)
        ext = ext * (target / np.sum(ext))
        score = _max_convolution(ext)
        best_seq, best_obj = ext.copy(), score
        print(f"[{time.time():.1f}] external seed n={n_ext} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 1.  Random / parametric multi‑restart phase
    # ------------------------------------------------------------------
    lengths = [5000, 6000, 7500, 8000, 9000, 10000, 12000]
    # decide how many independent trials we can afford
    remaining = deadline - time.time()
    max_trials = max(1, int(remaining // 80))      # ~80 s per trial on average
    for trial in range(max_trials):
        if time.time() >= deadline:
            break
        n = int(rng.choice(lengths))
        target_sum = np.sqrt(2.0 * n)

        # random Dirichlet‑like start
        start = _random_dirichlet_like(n, rng)

        # allocate time: 40 % gradient, 55 % mass‑transfer SA, 5 % spare
        now = time.time()
        grad_deadline = now + (deadline - now) * 0.40
        seq = _adam_gradient_descent(start, grad_deadline)

        # mass‑transfer SA on the remaining budget
        seq = _mass_transfer_sa(seq, deadline)

        # evaluate
        cur_obj = _max_convolution(seq)
        if cur_obj < best_obj:
            best_obj, best_seq = cur_obj, seq.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL best (trial {trial}) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2.  Final permutation annealing on the current best sequence
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _permutation_annealing(best_seq, deadline)
        best_obj = _max_convolution(best_seq)
        print(f"[{time.time():.1f}] after final permutation annealing obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3.  Cosmetic trimming (remove trailing near‑zero entries)
    # ------------------------------------------------------------------
    if best_seq is None:
        # absolute fallback – uniform distribution (guaranteed feasible)
        n = 4000
        val = np.sqrt(2.0 * n) / n
        best_seq = np.full(n, val)
        best_obj = _max_convolution(best_seq)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # 4.  Final verification via evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] final evaluate_sequence = {ev:.6f}")
    except Exception as exc:                       # pragma: no cover
        print("evaluate_sequence raised:", exc)

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
