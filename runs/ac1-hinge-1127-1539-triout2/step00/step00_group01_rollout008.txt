"""
Gradient‑descent based search for a step‑function that lowers the
evaluate_sequence bound.

The algorithm works with a probability‑like vector a (non‑negative,
sum = 1).  At each iteration we:

1. Compute the linear convolution c = a*a (via FFT for speed).
2. Form a softmax over c to obtain a weight vector w that focuses on
   the larger convolution values (parameter α controls the sharpness).
3. Compute the gradient of the smoothed max‑convolution:
      ∇ f(a) ≈ 2 * (a ⊛ w)
   where “⊛” denotes linear convolution.
4. Perform a multiplicative (mirror‑descent) update
      a_i ← a_i * exp(−η · ∇_i)   (η = learning‑rate)
   and renormalise a to sum = 1.
5. Occasionally add a tiny random perturbation to avoid stagnation.

The softmax temperature α is increased slowly, turning the smooth
approximation into the true max.  The best sequence seen during the
run (according to evaluate_sequence) is returned.

The routine respects the 1000 s budget, prints progress and never
produces invalid values (non‑negative, bounded, sum > 0.01).
"""

import numpy as np
import time

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  Returns the best list found before the time
    budget expires.

    Parameters
    ----------
    seed: int
        Random seed for reproducibility.
    budget_s: float
        Time budget in seconds (a few seconds are reserved for final
        cleanup, so the internal loop stops earlier).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------
    # Choose a problem size.  Larger n gives a lower theoretical
    # limit (≈1) but costs more per iteration.  3000–5000 is a good
    # compromise for the given time budget.
    # ------------------------------------------------------------
    n = 4000                       # length of the step function
    a = np.random.rand(n)          # random start (strictly positive)
    a /= a.sum()                   # normalise to sum = 1  (scale‑invariant)

    # best solution tracking
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score (uniform-like): {best_score:.6f}")

    # Hyper‑parameters
    alpha = 0.5          # softmax temperature (starts low, grows)
    alpha_max = 15.0
    lr = 0.3             # learning rate for the multiplicative update
    eval_interval = 5   # evaluate only every few iterations (cheap)

    it = 0
    while time.time() < deadline:
        # ---------- convolution via FFT ----------
        L = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        # ---------- softmax weights ----------
        max_c = conv.max()
        # shift for numerical stability
        shifted = conv - max_c
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            # Should never happen, but guard against division by zero
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the smoothed max ----------
        # grad = 2 * (a ⊛ w)
        W = np.fft.rfft(w, n=L)
        conv_aw = np.fft.irfft(A * W, n=L)[: 2 * n - 1]   # full linear conv
        grad = 2.0 * conv_aw[:n]                         # only first n entries needed

        # ---------- multiplicative (mirror‑descent) update ----------
        # Clip the gradient to avoid extreme exponentials
        grad_clip = np.clip(grad, 0.0, 10.0)
        a *= np.exp(-lr * grad_clip)
        # enforce non‑negativity (exp already ensures >0, but rounding may give tiny negatives)
        a = np.maximum(a, 0.0)
        # renormalise to keep sum = 1 (scale‑invariant objective)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ---------- occasional random jitter to escape plateaus ----------
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        # ---------- evaluate periodically ----------
        if it % eval_interval == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Iter {it:5d} | improved score: {best_score:.6f}")
                # early stop if we already beat a reasonable target
                if best_score < 1.55:   # well below the 2.0 baseline
                    # continue a little longer to maybe improve further
                    pass

        # ---------- schedule updates ----------
        it += 1
        # slowly increase softmax temperature (sharpen the max)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.004)
        # decay learning rate every few hundred iterations
        if it % 200 == 0:
            lr *= 0.9
            if lr < 0.01:
                lr = 0.01

    # Final evaluation of the best sequence (in case the last iter wasn't evaluated)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Finished. Best score achieved: {final_score:.6f} (iterations: {it})")
    return best_seq.tolist()
