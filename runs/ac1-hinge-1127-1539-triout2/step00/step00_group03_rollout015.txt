"""
Geometric step sequence generator.

The evaluation function is 2 * n * max(conv(a, a)) / (sum(a)^2).
If we choose a geometric progression a_i = w * q**i with
    q = 1 - C / n   (0 < C < 1)          and
    w = sqrt(2*n) * C / n,
then the sum of the sequence is exactly sqrt(2*n) (the normalisation
used by the evaluator) and the maximum of the convolution occurs at
the centre k = n‑1 with value

    max_b = n * w**2 * q**(n-1) = 2 * C**2 * (1-C/n)**(n-1)
           ≈ 2 * C**2 * exp(-C)  (for large n).

Thus the score ≈ 2·C²·e⁻ᶜ, which can be made arbitrarily small by
choosing a modest C (e.g. C = 0.1 gives a score ~0.018).  
The algorithm simply tries a handful of (n, C) pairs and returns the
sequence with the lowest score found within the time budget.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Build a geometric sequence a_i = w * q**i.
    Several (length, C) candidates are evaluated; the best one is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    best_seq = None
    best_score = float('inf')

    # Lengths to try – all comfortably fast for np.convolve
    length_candidates = [2000, 4000, 6000, 8000, 10000]

    # C values (the parameter controlling the decay, 0 < C < 1 works well)
    C_candidates = [0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50, 0.75, 1.00]

    for n in length_candidates:
        for C in C_candidates:
            # Stop early if we ran out of time
            if time.time() > deadline:
                print("Time limit reached, returning best sequence found.")
                return best_seq if best_seq is not None else []

            # geometric parameters
            alpha = C / n                    # step size for q = 1 - alpha
            q = 1.0 - alpha
            w = np.sqrt(2.0 * n) * alpha    # guarantees sum == sqrt(2n)

            # build the sequence efficiently (no overflow concerns for these n)
            seq = [float(w * (q ** i)) for i in range(n)]

            # evaluate – any exception means an invalid sequence (very unlikely)
            try:
                score = evaluate_sequence(seq)
            except Exception:               # pragma: no cover
                continue

            # keep the best one
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"New best: n={n}, C={C:.3f}, score={score:.6f}")

    # final report
    if best_seq is not None:
        print(f"Best sequence length = {len(best_seq)}, final score = {best_score:.6f}")
    else:
        print("No valid sequence generated.")
    return best_seq if best_seq is not None else []
