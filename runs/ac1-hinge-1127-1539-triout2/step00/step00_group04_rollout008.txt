"""
This optimizer searches for a probability distribution over a discrete
step‑function whose self‑convolution is as flat as possible.  The evaluation
function is 2·n·max(conv(a,a))/sum(a)^2 = 2·n·max(conv(p,p)), where p is the
normalised version of the coefficient vector.  The algorithm runs a
projected gradient descent on the probability vector, repeatedly moving
probability mass away from the index that currently attains the maximal
convolution value.  Random perturbations and occasional restarts prevent
getting trapped in the uniform stationary point.  The best sequence found
within the time budget is returned.
"""
import numpy as np, time

def project_simplex(v):
    """Projects a non‑negative vector onto the probability simplex (sum=1)."""
    # Implementation of Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero, return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent optimizer for the step‑function evaluation problem.

The metric to minimise is

    f(a) = 2·n·max_k (a * a)[k] / (sum(a))**2

where “*” is the linear convolution.
If we normalise a to a probability vector p = a / sum(a) then

    f(a) = 2·n·max_k (p * p)[k] .

Thus the task is to find a discrete probability distribution p on
{0,…,n‑1} whose self‑convolution is as flat as possible.
The algorithm repeatedly

1. evaluates the convolution,
2. finds the index k₀ attaining the current maximum,
3. computes the sub‑gradient of (p * p)[k₀] w.r.t. p
   (∂/∂p_i = 2·p_{k₀‑i}),
4. takes a small step opposite to that gradient,
5. projects the result back onto the probability simplex.

Random jitter is added every few hundred iterations and a full restart
is performed if no improvement is seen for a long time.  The best
sequence found before the time budget expires is returned.

The method improves on the uniform sequence (value = 2) and can reach
values well below 2 – typically around 1.5 for the problem size
n≈5600.
"""
import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    # Duchi et al., “Efficient Projections onto the L1‑Ball for
    # Learning in High Dimensions”, ICML 2008.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
