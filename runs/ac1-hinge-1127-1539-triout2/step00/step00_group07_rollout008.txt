    n = len(a)
    b = np.convolve(a, a)
    k = np.argmax(b)
    # find i that maximizes a_i * a_{k-i}
    i_start = max(0, k-(n-1))

"""
Gradient‑descent search for a step function with a low
self‑convolution peak.

The algorithm works on the simplex (non‑negative entries that sum to 1),
because the evaluation function
    score = 2 * n * max(conv(a, a)) / (sum(a)^2)
is invariant under scaling of the whole vector.
We minimise the quantity max(conv(a,a)) by a smooth surrogate:
soft‑max(α·conv(a,a)), whose gradient can be computed efficiently
using two convolutions.  A projected gradient step (onto the simplex)
is taken, with an annealing schedule for the soft‑max sharpness α and
a simple Metropolis acceptance rule to escape shallow local minima.
Random restarts and occasional Gaussian perturbations help exploration.

The procedure runs for the allotted time (≈1000 s) and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Return soft‑max weights w_i ≈ exp(α·b_i) / Σ_j exp(α·b_j) in a numerically stable way."""
    # shift by the maximum to avoid overflow
    b_max = np.max(b)
    shifted = alpha * (b - b_max)
    # additional shift for safety (exp may still overflow for very large α)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    return exp_vals / exp_vals.sum()


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of soft‑max(α·conv(a,a)) w.r.t. a.
    The analytical derivative of b_k = Σ_i a_i a_{k-i} is
        ∂b_k/∂a_i = 2·a_{k-i}   (i ≠ k-i)   and   = a_i (when i = k-i).
    Using the weights w_k from the soft‑max, the gradient is
        g_i = Σ_k w_k ∂b_k/∂a_i.
    This can be computed with one convolution.
    """
    n = a.size
    # reverse w for the convolution that yields Σ_k w_k a_{k-i}
    w_rev = w[::-1]
    conv = np.convolve(a, w_rev)                     # length = 3n‑2
    # extract the needed slice: conv[2n‑2‑i] for i = 0…n‑1
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        # fall back to uniform distribution
        return np.ones_like(v) / v.size
    return v / s


# ----------------------------------------------------------------------
# Main search function
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a few seconds margin

    # ------------------------------------------------------------------
    # Initialise the sequence -------------------------------------------------
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=float)
    else:
        # fallback: a random vector of moderate length
        init_len = np.random.randint(2000, 3000)
        init_seq = np.random.rand(init_len)

    # ensure feasibility
    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0:
        init_seq = np.ones_like(init_seq)

    # Work on the simplex (scale invariance guarantees correctness)
    a = _project_to_simplex(init_seq)

    n = a.size
    # current objective (we keep sum(a) = 1, so score = 2 * n * max(conv))
    b = np.convolve(a, a)
    curr_score = 2.0 * n * b.max()
    best_a = a.copy()
    best_score = curr_score

    print(f"[init] length = {n}, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters (will be adapted on the fly)
    # ------------------------------------------------------------------
    lr = 0.05               # base step size (in ℓ2‑norm of the update)
    alpha = 5.0             # soft‑max sharpness
    temperature = 0.1       # for Metropolis acceptance of worse moves
    decay_temp = 0.99995    # exponential decay of temperature
    no_improve = 0          # counter for stagnation handling
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Compute convolution and soft‑max weights
        # --------------------------------------------------------------
        b = np.convolve(a, a)                 # length = 2n‑1
        max_b = b.max()
        w = _softmax_weights(b, alpha)        # length = 2n‑1, Σw = 1

        # --------------------------------------------------------------
        # Gradient of the smooth surrogate
        # --------------------------------------------------------------
        grad = _gradient_softmax(a, w)        # shape (n,)

        # --------------------------------------------------------------
        # Normalised gradient step (projected onto the simplex)
        # --------------------------------------------------------------
        grad_norm = np.linalg.norm(grad) + 1e-12
        step = (lr / grad_norm) * grad       # magnitude = lr
        a_candidate = a - step                # descent direction
        a_candidate = _project_to_simplex(a_candidate)

        # occasional Gaussian jitter to encourage exploration
        if iteration % 300 == 0:
            jitter = np.random.randn(n) * 0.02
            a_candidate = _project_to_simplex(a_candidate + jitter)

        # --------------------------------------------------------------
        # Evaluate candidate (fast evaluation using the analytical formula)
        # --------------------------------------------------------------
        b_candidate = np.convolve(a_candidate, a_candidate)
        cand_score = 2.0 * n * b_candidate.max()

        # --------------------------------------------------------------
        # Acceptance rule (Metropolis)
        # --------------------------------------------------------------
        accept = False
        if cand_score < curr_score:
            accept = True
        else:
            # accept with probability exp(-(Δ) / T)
            delta = cand_score - curr_score
            prob = np.exp(-delta / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            curr_score = cand_score

            # keep the global best
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                no_improve = 0
                print(f"[{iteration:6d}] new best: {best_score:.6f} (lr={lr:.4f}, α={alpha:.2f})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Adaptive schedules
        # --------------------------------------------------------------
        temperature *= decay_temp
        # increase α slowly to sharpen the surrogate (focus more on the true max)
        if iteration % 200 == 0:
            alpha = min(alpha * 1.05, 80.0)

        # If we have not improved for a while, shrink the step size
        if no_improve > 800:
            lr = max(lr * 0.5, 0.005)
            no_improve = 0
            print(f"[{iteration:6d}] stagnation: decreasing lr to {lr:.5f}")

        # Occasionally restart from the current best with a small random perturbation
        if iteration % 2000 == 0:
            a = best_a + 0.01 * np.random.randn(n)
            a = _project_to_simplex(a)
            lr = 0.05
            alpha = 5.0
            temperature = 0.1
            print(f"[{iteration:6d}] restart from best, reset hyper‑parameters")

    # ------------------------------------------------------------------
    # Post‑processing: scale to a comfortable numeric range (does not affect score)
    # ------------------------------------------------------------------
    final_seq = (best_a * 100.0).tolist()   # values stay safely below the 1000 limit
    final_score = evaluate_sequence(final_seq)
    print(f"[done] best score = {final_score:.6f} (iterations = {iteration})")
    return final_seq
