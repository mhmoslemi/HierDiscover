    a = np.clip(seq, 0.0, 1000.0)

        start = i

def evaluate_fast(seq):
    a = np.clip(seq, 0.0, 1000.0)
    S = a.sum()
    if S < 0.01:
        return np.inf
    n = len(a)
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    maxb = conv.max()
    return 2.0 * n * maxb / (S * S)
```

def perturb_sequence(seq, sigma=0.01):
    n = len(seq)
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j:
        return seq
    max_transfer = seq[i] * sigma
    delta = np.random.rand() * max_transfer
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    # Ensure nonnegative
    if new_seq[i] < 0:
        new_seq[i] = 0
    return new_seq
```

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0
    n = 500  # could be changed
    # start with uniform
    seq = np.full(n, np.sqrt(2 * n) / n, dtype=float)
    # random perturb initial few times
    for _ in range(5):
        seq = perturb_sequence(seq, sigma=0.1)
    best_seq = seq.copy()
    best_score = evaluate_fast(best_seq)
    iter_count = 0
    while time.time() < deadline:
        # compute normalized f for LP
        sum_f = seq.sum()
        if sum_f < 0.01:
            break
        f_norm = seq * np.sqrt(2 * n) / sum_f  # ensures sum sqrt(2n)
        # compute bound M (max conv of f_norm with itself)
        M = np.max(np.convolve(f_norm, f_norm))
        # solve LP
        g = solve_convolution_lp(f_norm, M)
        if g is not None:
            # normalize g
            sum_g = g.sum()
            if sum_g < 1e-6:
                # fallback
                continue
            g_norm = g * np.sqrt(2 * n) / sum_g
            # line search for mixing
            # define candidate t's
            t_candidates = [0.0, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0]
            best_local_seq = seq
            best_local_score = best_score
            for t in t_candidates:
                new_seq = (1 - t) * seq + t * g_norm
                score = evaluate_fast(new_seq)
                if score < best_local_score:
                    best_local_score = score
                    best_local_seq = new_seq.copy()
            seq = best_local_seq
            if best_local_score < best_score:
                best_score = best_local_score
                best_seq = seq.copy()

        # occasional random perturb to avoid stagnation
        if iter_count % 20 == 0:
            if np.random.rand() < 0.3:
                seq = perturb_sequence(seq, sigma=0.1)
        iter_count += 1
    return best_seq.tolist()

"""
Propose a candidate step–function sequence.

The algorithm works with a moderate problem size (n≈500) so that the
linear program used in the original state‑of‑the‑art method can be solved
quickly with a sparse matrix.  Starting from a uniform sequence we
introduce a small random perturbation to break symmetry, then repeatedly:

1.  Normalise the current sequence to have Σa = √(2n).
2.  Compute the maximum self‑convolution M = max(a*a).
3.  Solve the linear programme
        maximise Σg   s.t.  a * g ≤ M ,  g ≥ 0
   (the convolution constraints are expressed with a sparse Toeplitz
    matrix).  The LP returns a feasible direction g that can have a larger
    total mass than a while respecting the same convolution bound.
4.  Normalise g to the same Σ = √(2n) and mix it with the current
    sequence.  A short line‑search over several mixing factors t∈[0,1]
    selects the mixture that gives the smallest value of the original
    evaluation function.
5.  Keep the best sequence seen so far.

Random perturbations are occasionally applied to avoid stagnation.
All heavy work (convolution, LP) is done with NumPy / SciPy; the
evaluation used during the search is a fast FFT‑based implementation of
the provided `evaluate_sequence` (identical up to clipping and the
security checks).  The routine respects the 1000‑second budget and
returns a list of non‑negative floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# Cache for the sparse index pattern of the convolution matrix
_lp_index_cache = {}

def _build_conv_matrix_indices(n):
    """Create row/col index arrays for the Toeplitz convolution matrix."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    idx = 0
    arange_n = np.arange(n, dtype=np.int32)
    for i in range(n):
        rows[idx:idx + n] = i + arange_n        # rows i+j
        cols[idx:idx + n] = arange_n            # columns j
        idx += n
    return rows, cols


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise Σ g_j
        s.t.   (f_seq * g)_k ≤ rhs  for all k
               g_j ≥ 0
    The convolution constraints are encoded with a sparse matrix.
    Returns the optimal g as a NumPy array, or None on failure.
    """
    n = len(f_seq)
    # Build (or reuse) the index pattern for this size
    global _lp_index_cache
    if n not in _lp_index_cache:
        _lp_index_cache[n] = _build_conv_matrix_indices(n)
    rows, cols = _lp_index_cache[n]

    # The data of the convolution matrix = repeat each f_i exactly n times
    data = np.repeat(f_seq, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, rhs, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                 # maximise Σg  ⇒ minimise −Σg
    bounds = [(0, None)] * n                           # g ≥ 0

    # Use the HiGHS interior point solver (fast for moderate size problems)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'time_limit': 5.0})
    except Exception:          # any numerical issue – treat as failure
        return None

    if res.success:
        return res.x
    return None


def fast_evaluate(seq):
    """
    Fast version of evaluate_sequence (identical result for valid inputs).
    Uses an FFT‑based convolution which is O(n log n) instead of O(n²).
    """
    a = np.asarray(seq, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)   # the security checks of evaluate_sequence
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = a.shape[0]
    # length for a power‑of‑two FFT (at least 2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
    M = conv.max()
    return float(2.0 * n * M / (S * S))


def perturb_sequence(seq, sigma=0.05):
    """
    Small random transfer of mass between two indices while preserving the total sum.
    Only non‑negative values are kept.
    """
    seq = seq.copy()
    n = len(seq)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return seq
    max_transfer = seq[i] * sigma
    if max_transfer <= 0:
        return seq
    delta = np.random.rand() * max_transfer
    seq[i] -= delta
    seq[j] += delta
    # enforce non‑negativity (numerical safety)
    if seq[i] < 0:
        seq[i] = 0.0
    return seq


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided evaluate_sequence() function.  The routine runs for at most
    `budget_s` seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    # --- problem size -------------------------------------------------
    # A modest n keeps the LP tractable while still allowing a good bound.
    n = 500

    # start from a uniform sequence; its sum is √(2n) to match the usual normalisation
    uniform_val = np.sqrt(2.0 * n) / n
    seq = np.full(n, uniform_val, dtype=np.float64)

    # introduce a tiny random perturbation to create slack for the first LP
    for _ in range(5):
        seq = perturb_sequence(seq, sigma=0.2)

    best_seq = seq.copy()
    best_score = fast_evaluate(best_seq)

    iteration = 0
    while time.time() < deadline:
        # ---- normalise for the LP (sum = √(2n)) -------------------------
        sum_seq = seq.sum()
        if sum_seq < 1e-12:
            break
        f_norm = seq * (np.sqrt(2.0 * n) / sum_seq)

        # current convolution bound M
        conv_f = np.convolve(f_norm, f_norm)
        M = conv_f.max()

        # ---- solve the LP ------------------------------------------------
        g = solve_convolution_lp(f_norm, M)
        if g is None:
            # LP failed – apply a random perturbation and continue
            seq = perturb_sequence(seq, sigma=0.1)
            continue

        # normalise the LP solution to the same sum √(2n)
        sum_g = g.sum()
        if sum_g <= 0:
            seq = perturb_sequence(seq, sigma=0.1)
            continue
        g_norm = g * (np.sqrt(2.0 * n) / sum_g)

        # ---- line‑search mixing -----------------------------------------
        # try a handful of mixing coefficients and keep the best outcome
        candidate_ts = [0.0, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0]
        local_best = seq
        local_score = fast_evaluate(seq)
        for t in candidate_ts:
            mixed = (1.0 - t) * seq + t * g_norm
            sc = fast_evaluate(mixed)
            if sc < local_score:
                local_score = sc
                local_best = mixed

        # adopt the best local mixture
        seq = local_best

        # ---- keep global best -------------------------------------------
        if local_score < best_score:
            best_score = local_score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] iter {iteration:5d}  new best = {best_score:.6f}")

        # occasional random shake‑up to escape possible plateaus
        if iteration % 20 == 0 and np.random.rand() < 0.3:
            seq = perturb_sequence(seq, sigma=0.15)

        iteration += 1

    # Convert to a plain Python list as required by the evaluation harness
    return best_seq.tolist()
