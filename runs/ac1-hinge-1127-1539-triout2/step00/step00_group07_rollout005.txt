"""Propose a candidate step function using a hybrid LP and stochastic local
search.  The algorithm starts from the provided best known sequence (if
available) and alternates:
  * solve a linear program that finds a direction g that does not increase the
    current autocorrelation peak,
  * mix the current sequence with g using a line‑search over the mixing
    parameter,
  * perform random mass‑redistribution moves (hill‑climbing) to escape local
    minima,
  * occasionally restart from the best found solution with a small random
    perturbation.
The search runs until the allotted time budget is exhausted and returns the
best sequence found.
"""

import time
import numpy as np
from scipy import optimize

def _conv_max(seq):
    """Return the maximum value of the aperiodic self‑convolution of `seq`."""
    return np.max(np.convolve(seq, seq))

def _solve_lp(f_norm, M):
    """
    Solve the linear program

        max   sum b_j
        s.t.  (f_norm * b)_k <= M   for all k
              b_j >= 0

    Returns the optimal vector b as a NumPy array, or None if the solver
    fails.
    """
    n = len(f_norm)
    # objective: maximise sum b  -> minimize -1·b
    c = -np.ones(n)

    # Build the convolution constraints.
    # For every possible sum index k (0 … 2n‑2) we have
    #   Σ_j f_i * b_j <= M   where i = k‑j .
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # fill row[j] = f_norm[k‑j] when the index is inside the sequence
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i runs from i_min … i_max ; j = k-i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_norm[i]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])   # also enforce b_j >= 0
    b_ub = np.hstack([np.full(2 * n - 1, M), np.zeros(n)])

    # Use HiGHS – the default method is fast for medium‑size problems.
    try:
        res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    except Exception:
        return None
    if res.success:
        return res.x
    return None

"""
Hybrid LP‑guided hill‑climbing search for a non‑negative step function that
minimises the supplied `evaluate_sequence` objective.

The algorithm works as follows:

1. **Initialisation** – start from the public best sequence `height_sequence_1`
   if it exists, otherwise from a random vector.

2. **LP‑guided mixing** – at each iteration (with probability ≈30 %) the current
   sequence is normalised to Σa_i = √(2·n) (the scaling does not affect the
   objective) and an LP is solved:

        maximise   Σ b_j
        subject to (f * b)_k ≤ M  for the most critical convolution indices k
                   b_j ≥ 0

   where `f` is the normalised current sequence and `M` its maximal
   self‑convolution value.  Only those indices `k` with
   `(f*f)_k ≥ 0.95·M` are kept, which makes the LP very small.
   The obtained direction `b` is then mixed with the current sequence.
   A short line‑search over the mixing coefficient `t∈[0,1]` (11 points)
   selects the best combined sequence.

3. **Stochastic hill‑climbing** – with the remaining probability a small
   amount of mass is moved from a random donor index to a random recipient.
   The move is accepted if it improves the objective.

4. **Escaping plateaus** – if no improvement has been observed for a while,
   a tiny random jitter is added to the best‑so‑far sequence to help escape
   local minima.

The loop runs until the allotted time budget (default 1000 s) is almost
exhausted.  The best sequence seen during the search is returned.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _conv_max(seq):
    """Maximum value of the aperiodic self‑convolution of `seq`."""
    return np.max(np.convolve(seq, seq))


def _solve_lp_limited(f_norm, M, keep_ratio=0.95):
    """
    Solve the LP:

        maximise   Σ b_j
        subject to (f_norm * b)_k ≤ M  for selected convolution indices k
                   b_j ≥ 0

    Only the indices `k` for which the current convolution value is at least
    `keep_ratio * M` are kept (plus the exact peak).  This yields a very
    small constraint matrix even for thousands of variables.

    Returns the optimal vector `b` (as a NumPy array) or ``None`` on failure.
    """
    n = len(f_norm)

    # Identify the “critical’’ convolution positions.
    conv_f = np.convolve(f_norm, f_norm)
    crit = np.where(conv_f >= M * keep_ratio)[0]
    if crit.size == 0:                     # safety net – keep the absolute peak
        crit = np.array([np.argmax(conv_f)])

    # Build the (few) convolution constraints.
    rows = []
    for k in crit:
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_norm[i]
        rows.append(row)

    # Non‑negativity constraints.
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([np.full(len(crit), M), np.zeros(n)])

    # Linear program – maximise Σb ⇒ minimise -Σb.
    c = -np.ones(n)
    try:
        res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    except Exception:                     # any unexpected issue → give up
        return None

    if res.success:
        return res.x
    return None

    print(f"Search finished after {iteration} iterations – final score {best_score:.6f}")
