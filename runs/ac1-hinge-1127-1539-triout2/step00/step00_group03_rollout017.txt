def optimize_for_n(n, time_budget):
    target_sum = math.sqrt(2*n)
    best_a = None
    best_obj = float('inf')
    # list of init methods:
    init_methods = []
    # uniform
    init_methods.append(np.full(n, target_sum / n))
    # analytic shape
    i = np.arange(n)
    a_analytic = 1.0 / np.sqrt(i+1)  # i+1 to avoid zero, offset 0 odd
    a_analytic = a_analytic * (target_sum / a_analytic.sum())
    init_methods.append(a_analytic)
    # random small
    for _ in range(2):
        a_rand = np.random.rand(n)
        a_rand = a_rand * (target_sum / a_rand.sum())
        init_methods.append(a_rand)
    # run each init
    for a0 in init_methods:
        a = a0.copy()
        # compute current objective
        conv = convolve_fft(a)
        maxb = conv.max()
        obj = maxb  # as sum normalized
        # set step size
        lr = target_sum * 0.1 / n  # maybe.
        start = time.time()
        while time.time() - start < time_budget:
            # compute gradient at max
            idx = conv.argmax()
            grad = np.zeros_like(a)
            # compute reversed indices
            rev = idx - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            grad[mask] = 2 * a[rev[mask]]
            # trial step
            a_new = a - lr * grad
            # ensure nonnegative
            a_new = np.clip(a_new, 0, None)
            # renorm
            sum_new = a_new.sum()
            if sum_new < 1e-12:
                break
            a_new = a_new * (target_sum / sum_new)
            # compute new obj
            conv_new = convolve_fft(a_new)
            maxb_new = conv_new.max()
            # accept if improvement
            if maxb_new < maxb - 1e-9:
                a = a_new
                conv = conv_new
                maxb = maxb_new
                # increase lr slowly up to limit
                lr *= 1.02
            else:
                # reduce lr
                lr *= 0.5
                # possibly random perturb if lr very small
                if lr < 1e-8:
                    # random jitter
                    i = np.random.randint(n)
                    delta = (np.random.rand() - 0.5) * target_sum * 0.01
                    a[i] = max(0, a[i] + delta)
                    a = a * (target_sum / a.sum())
                    conv = convolve_fft(a)
                    maxb = conv.max()
                    lr = target_sum * 0.05 / n
            # track best
            if maxb < best_obj:
                best_obj = maxb
                best_a = a.copy()
            # occasionally random shuffle if stagnating
            if np.random.rand() < 0.001:
                # small random perturb globally
                a += np.random.randn(n) * a.mean() * 0.01
                a = np.clip(a, 0, None)
                a = a * (target_sum / a.sum())
                conv = convolve_fft(a)
                maxb = conv.max()
        # after inner loop, record best
    return best_a, best_obj

def conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 2).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    conv = np.maximum(conv, 0)   # avoid negative zero due to rounding
    return conv
```

def compute_convolution(a):
    n = a.shape[0]
    size = 1 << ((2*n - 2).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n - 1]
    return conv
```

Now gradient function:

```
def compute_gradient(a, conv, eps=1e-12):
    max_val = conv.max()
    active = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        rev = k - idx
        mask = (rev >= 0) & (rev < n)
        grad[mask] += 2 * a[rev[mask]]
    return grad, max_val
```

"""
Propose a step‑function (non‑negative sequence) that minimises the
evaluation function:

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Analytic starter** – a discretised version of the continuous
   function f(x)=x⁻¹ᐟ², which yields a near‑constant self‑convolution and
   an objective close to π/2 ≈ 1.57 (much better than the uniform 2).

2. **Random convex combinations** – a small basis of representative
   sequences (analytic, uniform and a few power‑law variants) is mixed
   with random weights (Dirichlet draw).  Each mixture is projected back
   onto the required L₁‑norm and evaluated.  The best mixtures are kept.

3. **Local gradient descent** – starting from the best candidate,
   a simple projected gradient descent is performed on the
   ‖a∗a‖∞ objective.  The gradient of the active (maximal) convolution
   entry is `∇a max(conv) = 2 * a_reversed`, where the reversal is taken
   with respect to the index of the active convolution entry.  After a
   step the vector is clipped to non‑negative values and re‑scaled to the
   required sum (√(2n)).  Step size adapts automatically.

The whole search respects the 1000 s time budget and returns the best
sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper functions (top‑level, no closures)
# -------------------------------------------------------------

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution of a real vector via FFT (real output)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    # very small negative round‑off may appear – clamp to 0
    conv = np.maximum(conv, 0.0)
    return conv


def _gradient_max_conv(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12):
    """
    Gradient of the current maximal convolution value.
    Returns (grad, current_max).
    """
    n = a.shape[0]
    max_val = conv.max()
    # all indices within eps of the max are considered active
    active = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        rev = k - idx
        mask = (rev >= 0) & (rev < n)
        grad[mask] += 2.0 * a[rev[mask]]
    return grad, max_val


def _analytic_initial(n: int, target_sum: float) -> np.ndarray:
    """
    Analytic starter a_i ∝ 1/√(i+1) (discrete version of x⁻¹ᐟ²).
    The vector is scaled so that its L₁‑norm equals target_sum.
    """
    i = np.arange(n)
    raw = 1.0 / np.sqrt(i + 1.0)
    raw *= target_sum / raw.sum()
    return raw


def _local_optimize(a_start: np.ndarray,
                    target_sum: float,
                    max_iter: int = 300) -> (np.ndarray, float):
    """
    Simple projected gradient descent on  max(conv(a, a))
    starting from a_start.  Returns the best vector seen and its
    maximal convolution value.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_obj = _conv_fft(a).max()

    # initial step magnitude – proportional to typical entry size
    step_factor = target_sum * 0.02 / n

    for _ in range(max_iter):
        conv = _conv_fft(a)
        grad, cur_max = _gradient_max_conv(a, conv)

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        # step size inversely proportional to gradient norm
        lr = step_factor / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)

        # re‑project to the required L₁‑norm
        s = a_new.sum()
        if s < 1e-12:                     # numerical safety – re‑initialise
            a_new = np.random.rand(n)
            a_new *= target_sum / a_new.sum()
        else:
            a_new *= target_sum / s

        new_max = _conv_fft(a_new).max()

        # accept improvement, otherwise shrink step
        if new_max < cur_max:
            a = a_new
            step_factor *= 1.03
        else:
            step_factor *= 0.5
            # occasional random jitter to escape plateaus
            if np.random.rand() < 0.02:
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * target_sum * 0.02
                a[idx] = max(0.0, a[idx] + delta)
                a *= target_sum / a.sum()

        # keep the best ever seen
        if new_max < best_obj:
            best_obj = new_max
            best_a = a.copy()

    return best_a, best_obj


# -------------------------------------------------------------
# Main entry point required by the competition harness
# -------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a low‑evaluation‑score step‑function.
    Returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # ---------------------------------------------------------
    # Choose a problem size – larger n gives lower theoretical bound
    # ---------------------------------------------------------
    # We settle on n ≈ 2000 (fits comfortably into memory and FFT time)
    n = 2000
    target_sum = math.sqrt(2.0 * n)

    # ---------------------------------------------------------
    # Build a small basis of reasonable starting vectors
    # ---------------------------------------------------------
    basis = []

    # analytic (x⁻¹ᐟ²) starter – usually the strongest
    basis.append(_analytic_initial(n, target_sum))

    # uniform vector (the old 2.0 bound)
    basis.append(np.full(n, target_sum / n))

    # a few power‑law shapes with exponents different from 0.5
    for p in (0.2, 0.3, 0.4, 0.6):
        i = np.arange(n)
        raw = 1.0 / np.power(i + 1.0, p)
        raw *= target_sum / raw.sum()
        basis.append(raw)

    # if the global `height_sequence_1` exists and has the right length,
    # add it to the basis (it may already be a good local optimum)
    try:
        if len(height_sequence_1) == n:
            basis.append(np.array(height_sequence_1, dtype=float))
    except NameError:
        pass

    # ---------------------------------------------------------
    # Random convex‑combination search + occasional local optimisation
    # ---------------------------------------------------------
    best_seq = None
    best_obj = float('inf')

    # small helper to evaluate the objective (max convolution)
    def _obj(a_vec):
        return _conv_fft(a_vec).max()

    # Main search loop
    while time.time() < deadline:
        # ---- random convex combination of basis vectors -----------------
        w = np.random.rand(len(basis))
        w /= w.sum()
        a = sum(wi * bi for wi, bi in zip(w, basis))

        # small random noise (helps to explore the neighbourhood)
        if np.random.rand() < 0.35:
            a += np.random.randn(n) * a.mean() * 0.008
            a = np.clip(a, 0.0, None)
            a *= target_sum / a.sum()

        cur_val = _obj(a)
        if cur_val < best_obj:
            best_obj = cur_val
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.1f}s] Random mix improved: "
                  f"obj={cur_val:.6f}")

        # ---- short local optimisation from this point -----------------
        # use a tiny time budget (≈0.3 s) per refinement to stay within total budget
        remaining = deadline - time.time()
        if remaining < 0.5:
            break
        # limit the number of GD steps; 300 steps are cheap (FFT each step)
        a_opt, obj_opt = _local_optimize(a.copy(),
                                         target_sum,
                                         max_iter=300)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.1f}s] Gradient refined: "
                  f"obj={obj_opt:.6f}")

    # ---------------------------------------------------------
    # Fallback – should never happen, but guarantees a valid list
    # ---------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # final re‑normalisation (numerical safety)
    best_seq = best_seq.astype(float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq *= target_sum / s

    return best_seq.tolist()
