"""
Propose a high‑quality step‑function sequence.

The evaluation function is
    E(a) = 2·n·max(a*a) / (sum(a)^2)

For non‑negative a this quantity is ≥2 for a uniform sequence, but it can be
drastically reduced by using a power‑law shape:
    a_i ∝ (i/n)^p ,   p > 0

In the continuous limit
    E(p) = 2·(p+1)^2·B(p+1,p+1)
where B is the Beta function.
For p = 4 we already obtain E ≈ 0.08; for p = 10 we get
E ≈ 1·10⁻⁴.  The value of E is scale‑invariant, so we normalise the
sequence to the interval [0, 1] to avoid the 0 – 1000 clipping limits.

The algorithm simply tries a few (length, power) pairs, evaluates them
with the provided ``evaluate_sequence`` and returns the best one found
within the time budget.
"""

import time
import numpy as np

def _power_sequence(n: int, p: int) -> list[float]:
    """
    Return a non‑negative step‑function of length ``n`` with shape (i/n)^p.
    The maximum entry is exactly 1, guaranteeing no clipping by the
    evaluator.
    """
    # indices 1 … n (avoid the exact 0‑th power which would be 0)
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = (idx / n) ** p
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑evaluation sequence using power‑law shapes.

    Parameters
    ----------
    seed : int
        Random seed (not heavily used, kept for compatibility).
    budget_s : int
        Approximate time budget in seconds.
    **kwargs : dict
        Ignored – present for the caller’s signature.

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # Candidate parameters – they run extremely fast, so a few dozen are fine.
    lengths = [500, 1000, 1500, 2000, 2500]          # various n
    powers  = [2, 3, 4, 5, 6, 7, 8, 9, 10]          # exponent p

    best_seq = None
    best_score = float("inf")

    # Try deterministic grid first
    for n in lengths:
        for p in powers:
            if time.time() >= deadline:
                break
            seq = _power_sequence(n, p)
            try:
                score = evaluate_sequence(seq)
            except Exception:   # safety – should never happen
                continue
            print(f"Tested n={n}, p={p:2d} → score={score:.6f}")
            if score < best_score:
                best_score, best_seq = score, seq

    # If some time remains, sprinkle a few random variations around the
    # best candidate found so far – tiny Gaussian noise does not change
    # the evaluation much but might discover an even better local shape.
    while time.time() < deadline:
        if best_seq is None:
            # fallback: uniform sequence (guaranteed to be valid)
            n = 1000
            seq = [1.0] * n
        else:
            # perturb the current best – keep length, perturb values a bit
            n = len(best_seq)
            seq = np.array(best_seq, copy=True)
            # apply a small relative Gaussian noise (std 1 %)
            noise = np.random.randn(n) * 0.01
            seq = np.maximum(0.0, seq * (1.0 + noise))
            # re‑normalise to keep max ≤ 1 (avoid clipping)
            seq /= np.max(seq) if np.max(seq) > 0 else 1.0
            seq = seq.tolist()

        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue
        print(f"Random perturbation → score={score:.6f}")
        if score < best_score:
            best_score, best_seq = score, seq

    # Final report
    print(f"Best sequence: length={len(best_seq)} with score={best_score:.6f}")
    return best_seq
