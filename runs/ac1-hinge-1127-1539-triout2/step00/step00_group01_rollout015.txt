def simplex_projection(v):
    # v: numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np, random
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0
    best_seq = None
    best_score = float('inf')
    
    # try multiple n values
    candidate_ns = [100, 200, 400, 800]
    for n in candidate_ns:
        # initialize p random Dirichlet
        p = np.random.dirichlet(np.ones(n))
        # scale: sum will be 1; we will later scale to sqrt(2n) when evaluating
        # parameters for gradient descent
        beta = 1.0
        eta = 0.1
        iter = 0
        while time.time() < deadline and iter < 5000:
            # compute convolution
            c = np.convolve(p, p)
            max_c = c.max()
            score = 2 * n * max_c   # because sum=1
            # update best
            if score < best_score:
                best_score = score
                best_seq = (p * np.sqrt(2*n)).tolist()
                print(f"n={n}, iter={iter}, score={score:.6f}")
            # smoothed max
            # update beta gradually
            if iter % 200 == 0 and beta < 50:
                beta *= 1.5
            # compute softmax weights
            # shift to avoid overflow
            shifted = c - c.max()
            w = np.exp(beta * shifted)
            w = w / w.sum()
            # compute gradient g = 2 * sum_j p_j * w_{i+j}
            # use vectorized: create matrix of w[i+j]
            idx_i = np.arange(n)[:, None]
            idx_j = np.arange(n)[None, :]
            w_ij = w[idx_i + idx_j]   # shape (n,n)
            g = 2 * (w_ij @ p)   # (n,) because we sum over j: Σ_j w_{i+j} p_j
            # exponentiated gradient descent
            p = p * np.exp(-eta * g)
            # project onto simplex
            p = simplex_projection(p)
            # optionally decay eta
            eta = eta * 0.9995
            iter += 1

        # after finishing n, maybe break if best_score sufficiently low (<1.55)
        if best_score < 1.55:
            break

    # fallback: if no improved sequence, return uniform of moderate length
    if best_seq is None:
        n = 100
        best_seq = [np.sqrt(2*n)/n] * n
    return best_seq

import numpy as np
import time

def simplex_projection(v):
    # Project onto simplex: v >=0, sum = 1
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? return uniform
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Search for a probability vector that reduces the max of its self‑convolution.
    Uses a smoothed max (log‑sum‑exp) and exponentiated‑gradient updates,
    running several random restarts with different lengths.
    Returns the raw heights (scaled to sum = sqrt(2n)) of the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a safety margin for the caller
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float('inf')

    # list of candidate lengths; try increasing sizes for better potential
    lengths = [100, 200, 400, 800]
    # optionally include the provided SOTA length as fallback
    if 'height_sequence_1' in globals():
        # length of known sequence
        lengths.append(len(height_sequence_1))

    for n in lengths:
        if time.time() > deadline:
            break
        # Number of iterations we allow for this length; proportional to n
        max_iters = int(2000 * (n / 200))   # scaling
        # random initial probability vector, Dirichlet with small concentration -> more spread
        p = np.random.dirichlet(np.ones(n) * 0.5)
        # ensure we are not exactly uniform (that would be a stationary point)
        # add a tiny perturbation
        p += np.random.randn(n) * 1e-6
        p = simplex_projection(p)

        # parameters for the optimisation
        eta = 0.5               # base learning rate for exponentiated gradient
        beta = 1.0              # temperature for soft‑max approximation of max
        beta_max = 50.0
        beta_inc = (beta_max / max_iters)  # linear increase

        for it in range(max_iters):
            if time.time() > deadline:
                break

            # convolution of p with itself (linear, size 2n-1)
            c = np.convolve(p, p)
            # current objective (exact max) – used for bookkeeping
            max_c = c.max()
            score = 2.0 * n * max_c      # because sum(p)=1
            if score < best_score:
                best_score = score
                # scale to the convention used by evaluate_sequence (sum = sqrt(2 * n))
                a = p * np.sqrt(2.0 * n)
                best_seq = a.tolist()
                print(f"[{time.time()-start_time:.1f}s] n={n} iter={it} score={score:.6f}")

            # soft‑max weights
            # shift by the maximum to avoid overflow
            shift = c.max()
            w = np.exp(beta * (c - shift))
            w_sum = w.sum()
            if w_sum == 0:
                # numerical failure, fall back to hard max
                w = np.zeros_like(c)
                w[np.argmax(c)] = 1.0
                w_sum = 1.0
            w = w / w_sum

            # compute gradient g_i = 2 * sum_j p_j * w_{i+j}
            # we build the matrix of w_{i+j} via broadcasting – O(n²) but cheap for our sizes
            idx_i = np.arange(n)[:, None]          # shape (n,1)
            idx_j = np.arange(n)[None, :]          # shape (1,n)
            w_ij = w[idx_i + idx_j]                # shape (n,n)
            g = 2.0 * (w_ij @ p)                   # (n,)

            # exponentiated‑gradient update (mirrors mirror descent on the simplex)
            p = p * np.exp(-eta * g)
            # projection back onto the probability simplex
            p = simplex_projection(p)

            # anneal parameters
            beta = min(beta + beta_inc, beta_max)
            # slowly decay the step size to stabilise
            eta *= 0.9995

        # after finishing this length, if we already reached a very good bound we can stop early
        if best_score < 1.55:      # target is 1.503 – stop a little earlier to save time
            break

    # if nothing succeeded, fall back to the uniform construction (guaranteed to give 2.0)
    if best_seq is None:
        n = 200
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a.tolist()
    return best_seq

"""
Optimise a step‑function heights sequence by directly minimising the
maximum value of its self‑convolution.

The evaluation function simplifies (after scaling) to
    score = 2 * n * max_k (p * p)_k
where p is the normalised probability vector  (sum(p)=1) and n = len(p).
Thus we only have to reduce the infinity‑norm of the discrete auto‑convolution.

The optimiser works on the probability vector p.
It performs several random restarts with different lengths n.
For each run it starts from a random Dirichlet vector (so we are not
initialised at the uniform fixed point) and then runs an
exponentiated‑gradient (mirror‑descent) loop.

A smoothed approximation of the max (log‑sum‑exp) is used:
    L(p) = (1/β) log Σ_k exp(β * c_k) ,   c = p ∗ p
Its gradient on the simplex is
    ∇L_i = 2 Σ_j p_j w_{i+j},
where w_k = softmax(β * c_k) (the soft‑max weights).
The update is    p ← p ∘ exp(−η ∇L)   (∘ = element‑wise product)
followed by a projection onto the probability simplex.
β is increased gradually so the smooth objective approaches the true
max.  The learning rate η is slowly annealed.

Whenever a new best score is found the current sequence (scaled to the
convention used by ``evaluate_sequence`` – i.e. sum = sqrt(2n) –) is
saved and reported.

The whole search respects the given time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  Returns a list of floats (the heights).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds
        before this limit).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float('inf')

    # a few different lengths – larger n gives more freedom
    lengths = [100, 200, 400, 800]
    # if the historic solution exists, also try its length as fallback
    if 'height_sequence_1' in globals():
        lengths.append(len(height_sequence_1))

    for n in lengths:
        if time.time() >= deadline:
            break

        # number of optimiser iterations – scaled with problem size
        max_iters = int(2000 * (n / 200.0))

        # random start (Dirichlet with concentration 0.5 → moderately spread)
        p = np.random.dirichlet(np.ones(n) * 0.5)
        # tiny perturbation to avoid exact uniform start
        p += np.random.randn(n) * 1e-6
        p = _simplex_projection(p)

        # pre‑compute the index‑sum matrix used for the gradient
        sum_idx = np.add.outer(np.arange(n), np.arange(n))   # shape (n,n)

        eta = 0.5                # base learning rate for the EG update
        beta = 1.0               # temperature for the soft‑max (smooth max)
        beta_max = 50.0
        beta_inc = (beta_max - beta) / max_iters if max_iters > 0 else 0.0

        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # convolution (linear) of the probability vector with itself
            c = np.convolve(p, p)                 # length 2n‑1
            max_c = c.max()
            score = 2.0 * n * max_c               # exact evaluation (sum(p)=1)

            # keep the best sequence seen so far
            if score < best_score:
                best_score = score
                # scale to the convention used by evaluate_sequence (sum = sqrt(2n))
                a = p * np.sqrt(2.0 * n)
                best_seq = a.tolist()
                print(f"[{time.time() - start_time:.1f}s] n={n} it={it} score={score:.6f}")

            # ------------------------------------------------------------
            #   Smooth max via softmax, then compute gradient on the simplex
            # ------------------------------------------------------------
            shift = max_c
            w = np.exp(beta * (c - shift))        # still safe numerically
            w_sum = w.sum()
            if w_sum == 0.0:                      # numerical safeguard
                w = np.zeros_like(c)
                w[np.argmax(c)] = 1.0
                w_sum = 1.0
            w /= w_sum

            # gradient: g_i = 2 * Σ_j p_j * w_{i+j}
            # use the pre‑computed index‑sum matrix to index w
            w_mat = w[sum_idx]                    # shape (n,n)
            g = 2.0 * (w_mat @ p)                 # vector of length n

            # exponentiated‑gradient (mirror descent) step
            p = p * np.exp(-eta * g)
            p = _simplex_projection(p)

            # occasional tiny random jitter to escape tiny plateaus
            if it % 50 == 0:
                idx = np.random.choice(n, size=min(5, n), replace=False)
                p[idx] += np.random.randn(len(idx)) * 1e-5
                p = _simplex_projection(p)

            # schedule updates
            beta = min(beta + beta_inc, beta_max)
            eta *= 0.9995   # slow decay for stability

        # early exit if we already have a very good bound
        if best_score < 1.55:    # target is 1.503, stop a bit earlier to save time
            break

    # fallback – uniform sequence (guarantees score = 2.0)
    if best_seq is None:
        n = 200
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = a.tolist()

    return best_seq
