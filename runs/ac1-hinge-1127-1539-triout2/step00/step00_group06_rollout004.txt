        best_seq = list(height_sequence_1)  # noqa: F821
        best_score = _fast_evaluate(best_seq)
        print(f"[init] Using provided height_sequence_1, score = {best_score:.6f}")
    except Exception:
        best_seq = np.ones(n, dtype=float) / n
        best_score = _fast_evaluate(best_seq)
        print(f"[init] No reference sequence, uniform start, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper for generating a random binary mask of given density.
    # ------------------------------------------------------------------
    def _random_mask(density):
        m = max(2, int(density * n))
        idx = np.random.choice(n, size=m, replace=False)
        seq = np.zeros(n, dtype=float)
        seq[idx] = 1.0
        return seq

    # ------------------------------------------------------------------
    # Search loop.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a little margin
    it = 0
    target = 1.503  # the target bound we would like to beat

    while time.time() < deadline:
        # ---- 1) Random dense mask ------------------------------------------------
        # Choose a density roughly between 0.3 and 0.7 (empirically good)
        d = np.random.uniform(0.30, 0.70)
        cand = _random_mask(d)
        cand_score = _fast_evaluate(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.tolist()
            print(f"[{it:05d}] New best (mask)  density={d:.3f}  score={best_score:.6f}")

        # ---- 2) Small local perturbation of the current best --------------------
        # Occasionally try a few bit‑flips around the current best solution.
        if np.random.rand() < 0.07:   # ~7 % of iterations
            cur = np.array(best_seq, copy=True)
            # Number of positions to toggle (1–5)
            flips = np.random.randint(1, 6)
            flip_idx = np.random.choice(n, size=flips, replace=False)
            cur[flip_idx] = 1.0 - cur[flip_idx]   # toggle 0↔1

            # Ensure we still have at least one non‑zero entry
            if cur.sum() < 0.5:
                cur[np.random.randint(n)] = 1.0

            cand2_score = _fast_evaluate(cur)
            if cand2_score < best_score:
                best_score = cand2_score
                best_seq = cur.tolist()
                print(f"[{it:05d}] New best (local) score={best_score:.6f}")

        # ---- 3) Early exit if we have already beaten the target ---------------
        if best_score <= target:
            # keep searching a bit more in case we can go even lower
            # but break if very little time is left
            if time.time() > deadline - 5.0:
                break

        it += 1

    print(f"Search finished after {it} iterations. Best score = {best_score:.6f}")
    return best_seq
