def project_to_simplex(v):
    """Project v onto simplex S = {x: x_i >=0, sum x_i = 1} using efficient algorithm."""
    # algorithm from Duchi et al. 
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that (hopefully) beats the current
upper bound 2.0.

Algorithm
---------

* The evaluation function is scale‑invariant, therefore we keep the sum of all
  heights equal to 1 (the simplex) during optimisation.
* For a fixed vector `a` we denote `c = a * a` (full convolution).  The quantity
  we want to minimise is `max(c)`.  Its sub‑gradient at the active lag `k*`
  (where the maximum is attained) is
         ∂ max(c) / ∂ a_i = 2·a_{k*‑i}   (if 0 ≤ k*‑i < n, otherwise 0).

* We use an **exponentiated‑gradient (mirror descent) on the simplex**:
      a ← a * exp( - η·grad )
      a ← a / sum(a)
  This respects non‑negativity automatically and stays on the simplex.

* To break symmetry we start from a random Dirichlet vector in each restart.
  Several independent restarts are performed until the time budget is exhausted.
* Occasionally a tiny Gaussian perturbation is added (simulated‑annealing style)
  to help escape shallow local minima.
* The best sequence seen (according to the true `evaluate_sequence` call)
  is kept and finally returned.

The method is cheap (O(n log n) per iteration with `np.convolve`) and
fits easily into the 1000 s budget for moderate lengths (a few thousand
elements).  The returned sequence can be any length; lengths around a few
thousand give a theoretical lower bound close to 1, so a value below 2.0 is
quite attainable.

"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------

def _exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform one exponentiated‑gradient step on the simplex.
    a is already non‑negative and sums to 1.
    """
    # Multiplicative update
    a = a * np.exp(-eta * grad)
    # Re‑normalise to the simplex
    s = a.sum()
    if s == 0.0:                     # should never happen, but guard against it
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    return a


def _max_convolution(a: np.ndarray):
    """
    Return the full convolution a*a, its maximum value, and the index where
    the maximum is attained.
    """
    conv = np.convolve(a, a)
    max_val = conv.max()
    idx = conv.argmax()
    return conv, max_val, idx


def _random_simplex_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point uniformly from the probability simplex (Dirichlet)."""
    v = rng.random(n) + 1e-12   # avoid exact zeros
    v = v / v.sum()
    return v


# ----------------------------------------------------------------------
# Core optimisation routine (single restart)
# ----------------------------------------------------------------------


def _optimise_one_restart(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 5000,
    start_eta: float = 0.5,
    eta_decay: float = 0.9995,
    perturb_prob: float = 0.02,
    perturb_scale: float = 0.01,
):
    """
    Perform a mirror‑descent optimisation for a given length `n`.

    Returns
    -------
    best_vec : np.ndarray
        The best vector (on the simplex) found during this restart.
    best_est : float
        The corresponding estimator `2 * n * max(conv)`.  (Since sum = 1.)
    """
    a = _random_simplex_vector(n, rng)

    best_vec = a.copy()
    # quick estimator (exact for sum = 1)
    _, max_b, _ = _max_convolution(a)
    best_est = 2.0 * n * max_b

    eta = start_eta

    # theoretical lower bound for the given length
    theoretical_lb = 2.0 * n / (2 * n - 1)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # occasionally add a tiny random jitter – helps with symmetry breaking
        if rng.random() < perturb_prob:
            noise = rng.normal(scale=perturb_scale, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()

        conv, max_b, k = _max_convolution(a)

        # build sub‑gradient of the max‑convolution at the active lag k
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start <= i_end:
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            grad[i_idx] = 2.0 * a[j_idx]

        # Exponentiated‑gradient update on the simplex
        a = _exponential_update(a, grad, eta)

        # decay the step size slowly
        eta *= eta_decay

        # keep track of the best discovered vector
        est = 2.0 * n * max_b            # exact because sum(a) == 1
        if est < best_est:
            best_est = est
            best_vec = a.copy()
            # early exit if we are essentially at the theoretical limit
            if best_est <= theoretical_lb + 1e-8:
                break

    return best_vec, best_est
