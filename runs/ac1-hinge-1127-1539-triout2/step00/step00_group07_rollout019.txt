Pseudo code for projection onto simplex (Algorithm from "Efficient Projections onto the l1-ball for Learning in High Dimensions"):

def simplex_projection(v):
    # v is a np.ndarray, we want to find w >=0, sum w = 1, minimizing ||w - v||^2.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1 # index starting from 0
    theta = cssv[cond][-1] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    # Ensure sum(v) = 1 and v >= 0.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.arange(1, n+1)
    cond = u - cssv / rho_arr > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Choose length n (maybe 300)
    n = 300
    # initial random simplex vector
    a = np.random.rand(n)
    a = a / np.sum(a)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    t = 0
    # hyperparameters
    eta0 = 0.5  # initial step size
    # we can adapt with decreasing schedule: eta = eta0 / sqrt(iter+1)
    while time.time() < deadline:
        conv = np.convolve(a, a)
        M = np.max(conv)
        k_max = np.where(conv == M)[0]  # indices where max occurs

        # compute gradient (subgradient) as sum of gradients for each max index
        grad = np.zeros_like(a)
        for k in k_max:
            # contribution for each i where 0 <= k - i < n
            # grad_i += 2 * a[k - i]
            i_min = max(0, k - (n-1))
            i_max = min(n-1, k)
            # vectorized: for i in range(i_min, i_max+1):
            #   grad[i] += 2 * a[k - i]
            # we can compute by iterating
            for i in range(i_min, i_max+1):
                grad[i] += 2 * a[k - i]

        # average gradient if multiple k_max
        if len(k_max) > 1:
            grad /= len(k_max)

        # step size schedule
        eta = eta0 / np.sqrt(t + 1)
        # update
        a_new = a - eta * grad
        # project onto simplex
        a_new = simplex_projection(a_new)

        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"Iter {t}, new best {best_score:.6f}")

        a = a_new
        t += 1

    print(f"Final best score: {best_score}")
    return best_seq.tolist()

"""
propose_candidate: minimise 2·n·max(conv(a,a))/sum(a)^2 by a simple
projected sub‑gradient descent with occasional random jitter.
The algorithm works on a probability simplex (a_i ≥ 0, Σa_i = 1) and
repeatedly moves mass between coordinates while keeping the simplex
constraints via a Euclidean projection.  The sub‑gradient of the
∞‑norm of the self‑convolution is easy to compute:
∂(max_k (a∗a)[k]) / ∂a_i = 2·a_{k_max−i} for every index i that
contributes to a maximal convolution entry k_max.  By stepping
against this gradient and projecting back onto the simplex the
maximum convolution value is reduced.  Random jitter is injected
periodically to escape plateaus.  The search stops when the time
budget is exhausted or a score better than the target (1.503) is
found.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the probability simplex
# -------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1  # cumulative sum minus 1
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection (should not happen here)
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# -------------------------------------------------------------

    on the simplex together with occasional random jitter to avoid local minima.
    Multiple problem sizes are tried; the best sequence found before the time
    budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # leave a tiny safety margin

    best_score = float("inf")
    best_seq = None

    # try a few different lengths – short enough to keep each iteration cheap,
    # long enough to allow a value below the target.
    lengths = [50, 100, 200, 300, 500]

    for n in lengths:
        # ---- initialise a random point on the simplex ----
        a = np.random.rand(n)
        a = a / a.sum()          # Σ a_i = 1  (sum is > 0.01, so evaluate_sequence is safe)

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        iteration = 0
        # ---- main optimisation loop for this length ----
        while time.time() < deadline:
            # compute the convolution and its maximum
            conv = np.convolve(a, a)
            max_val = conv.max()
            # indices where the maximum is attained (could be more than one)
            max_idx = np.flatnonzero(conv == max_val)

            # ----- sub‑gradient of the max‑convolution -----
            grad = np.zeros_like(a)
            for k in max_idx:
                # valid i satisfy 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
                i_min = max(0, k - (n - 1))
                i_max = min(n - 1, k)
                # contribution 2·a_{k‑i}
                for i in range(i_min, i_max + 1):
                    grad[i] += 2.0 * a[k - i]
            if len(max_idx) > 1:
