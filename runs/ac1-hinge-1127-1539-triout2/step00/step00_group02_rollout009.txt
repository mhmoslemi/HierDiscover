def project_simplex(v):
    # ensure v is non-negative and sum(v)=1
    # from https: //stackoverflow.com/a/42871490
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    # Find next power of 2 for speed
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=N)

    return conv

    n = len(f_sequence)
    # objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    # constraints: for each k: sum_j f_i * b_j <= rhs
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        rows.append(row)
    A_ub = np.vstack(rows)

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'time_limit': 10.0})

"""
Propose a step‑function (list of non‑negative heights) that minimises the
`evaluate_sequence` score.

The algorithm is a hybrid of three cheap heuristic moves:

1. **Multiplicative (exponentiated‑gradient) update** –  reduces the
   entries that contribute most to the current maximal convolution entry.

2. **Local hill‑climb (mass transfer)** –  moves a small amount of mass from
   an index with large gradient to a random index; the move is accepted
   only if the objective improves (simulated‑annealing style fallback).

3. **LP‑based boost** –  every few iterations solves the linear programme
   from the reference solution, then mixes the current vector with the LP
   solution using a line‑search on the mixing coefficient `t`.  This can
   dramatically lower the maximal autocorrelation while preserving the
   ℓ₁‑norm.

All steps keep the vector non‑negative and normalised to sum = 1 (the
objective is scale‑invariant).  The method runs for the allocated time
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def _project_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if np.sum(v) == 0:
        # uniform fallback
        return np.full_like(v, 1.0 / len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def solve_convolution_lp(f_seq, rhs):
    """
    Linear programme:
        maximise   sum(b_j)
        subject to  conv(f_seq, b) <= rhs   (for every k)
                    b_j >= 0
    Returns the optimal b (numpy array) or None on failure.
    """
    n = len(f_seq)
    # objective – minimise -sum(b)  == maximise sum(b)
    c = -np.ones(n)

    # build constraints A_ub * b <= rhs
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, rhs)

    # non‑negativity via bounds
    bounds = [(0, None) for _ in range(n)]

    # Use the modern HiGHS solver; a 10 s time limit keeps us safe.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': 10.0, 'disp': False})
    if res.success:
        return res.x
    return None


def _exponential_gradient_step(a, it):
    """
    Multiplicative update that reduces the entries responsible for the
    current maximal convolution value.
    """
    n = len(a)
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # gradient of b_k_max w.r.t. a_i is 2*a_{k_max-i} (if in range)
    idx = np.arange(n)
    rev = k_max - idx
    mask = (rev >= 0) & (rev < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[rev[mask]]

    # step size decays like O(1/√it)
    eta = 0.5 / (np.sqrt(it) + 1.0)

    # prevent overflow in the exponent
    grad = np.clip(grad, 0, 10.0)

    mult = np.exp(-eta * grad)
    a_new = a * mult
    a_new = _project_simplex(a_new)
    return a_new


def _hill_climb_step(a):
    """
    Small random transfer of mass from an index that contributes heavily to
    the current maximal convolution entry to a random destination.
    """
    n = len(a)
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # contribution weights for each i (larger => more likely to be decreased)
    idx = np.arange(n)
    rev = k_max - idx
    mask = (rev >= 0) & (rev < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[rev[mask]]

    if contrib.sum() == 0:
        # fallback to uniform random choice
        i = np.random.randint(n)
    else:
        prob = contrib / contrib.sum()
        i = np.random.choice(n, p=prob)

    j = np.random.randint(n)
    if i == j:
        return a  # no meaningful move

    # move at most half of the source mass
    delta = np.random.rand() * 0.5 * a[i]
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new


def _lp_improvement(a):
    """
    Solve the LP (maximise sum(b) under the current max‑convolution bound)
    and mix the result with the current vector using a short line‑search
    over the mixing coefficient t∈[0,1].
    """
    n = len(a)
    sum_a = np.sum(a)
    if sum_a == 0:
        a = np.full_like(a, 1.0 / n)
        sum_a = 1.0
    # normalise to the conventional sum = sqrt(2n) (the exact scale does not matter)
    scale = np.sqrt(2.0 * n) / sum_a
    f_norm = a * scale

    # current bound
    max_b = np.max(np.convolve(f_norm, f_norm))

    # LP for a better direction
    g = solve_convolution_lp(f_norm.tolist(), float(max_b))
    if g is None:
        return a  # LP failed – return unchanged

    g = np.array(g)
    sum_g = np.sum(g)
    if sum_g == 0:
        return a
    g_norm = g * np.sqrt(2.0 * n) / sum_g

    # line‑search on t
    best = a
    best_score = evaluate_sequence(a.tolist())
    for t in (0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
        cand = (1.0 - t) * a + t * g_norm
        cand = np.clip(cand, 0, None)          # safety
        # re‑normalise – not required for the objective but keeps numbers sane
        cand = _project_simplex(cand)
        try:
            sc = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    return best


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function that minimises evaluate_sequence.
    Returns the best list of heights found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # leave a little margin

    # ------------------------------------------------------------------
    # 1) initialise a starting vector
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = list(height_sequence_1)
        # limit length for speed – keep a contiguous block of ~2000 entries
        if len(base) > 2000:
            start = np.random.randint(len(base) - 2000 + 1)
            a = np.array(base[start:start + 2000], dtype=float)
        else:
            a = np.array(base, dtype=float)
    else:
        # completely random start if nothing else is available
        n = np.random.randint(800, 2000)
        a = np.random.rand(n)

    # normalise to a probability vector (scale does not affect the score)
    a = _project_simplex(a)

    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    print(f"Start – length {len(a)} – score {best_score:.6f}")

    it = 0
    last_lp = 0

    # ------------------------------------------------------------------
    # 2) main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # Every ~30 iterations we try the expensive LP boost
        if it - last_lp > 30 and np.random.rand() < 0.6:
            a_candidate = _lp_improvement(a)
            last_lp = it
        else:
            # 70 % exponentiated‑gradient, 30 % simple hill‑climb
            if np.random.rand() < 0.7:
                a_candidate = _exponential_gradient_step(a, it)
            else:
                a_candidate = _hill_climb_step(a)

        # Occasionally add a tiny random jitter to escape plateaus
        if it % 200 == 0 and np.random.rand() < 0.2:
            idx = np.random.randint(len(a_candidate))
            a_candidate[idx] = max(0.0, a_candidate[idx] + np.random.randn() * 0.01)
            a_candidate = _project_simplex(a_candidate)

        # Normalise again – keeps the vector in the simplex
        a_candidate = _project_simplex(a_candidate)

        # ------------------------------------------------------------------
        # 3) evaluate
        # ------------------------------------------------------------------
        try:
            cur_score = evaluate_sequence(a_candidate.tolist())
        except Exception:
            continue

        # Accept if improvement (always keep the best ever seen)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            print(f"Iter {it:5d} – new best score {best_score:.6f}")

        # For the next iteration we keep the candidate (greedy walk)
        a = a_candidate

    print(f"Finished – best score {best_score:.6f} – length {len(best_seq)}")
    return best_seq.tolist()
