"""
Propose a step‑function (non‑negative sequence) that minimises the
evaluation function:

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The score is scale‑invariant, so we may normalise the sequence to have
sum = 1.  For a normalised vector a, the denominator disappears and the
objective reduces to `2 * n * max(conv(a, a))`.  By choosing a
power‑law distribution `a_i ∝ i**p` (with p≥0) the auto‑convolution
becomes very flat; the peak is roughly `C(p) / n` where
`C(p) = (p+1)**2 * (p!**2) / (2p+1)!`.  For p ≈ 6 and n≈5 000 the peak
is dominated by the squared maximum entry `a_n**2 ≈ (p+1)²/n²`,
giving a score of about `2*(p+1)²/n ≈ 0.02`, far below the target
1.503.

The function tries a few integer exponents p (0‥8), evaluates the score
with the provided `evaluate_sequence`, and returns the best sequence
found.
"""

import numpy as np

def _power_sequence(n: int, p: int) -> list[float]:
    """
    Return a length‑n list where the i‑th element (1‑based) is i**p,
    normalised to have sum = 1.
    """
    # compute raw powers as float to avoid overflow of Python ints in the sum
    idx = np.arange(1, n + 1, dtype=np.float64)
    raw = np.power(idx, p, dtype=np.float64)
    total = raw.sum()
    # normalise – sum will be exactly 1.0 (within floating error)
    seq = (raw / total).tolist()
    return seq

    The function evaluates a few exponents and returns the sequence with
    the smallest score according to `evaluate_sequence`.
    """
    # Use the length of the previously found best sequence if it exists,
    # otherwise fall back to a reasonable size.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        if n < 2:
            n = 5000
    except Exception:
        n = 5000

    best_seq = None
    best_score = float("inf")

    # Test several exponents; higher exponents flatten the convolution
    # but eventually the self‑product of the largest entry dominates.
    # Exponents 0..8 cover the useful range for n≈5k.
    for p in range(0, 9):
        seq = _power_sequence(n, p)

        # Evaluate – any exception or inf is ignored.
        try:

        except Exception:
            continue

        print(f"p={p:<2d}, length={n}, score={score:.8f}")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq

    # As a safety‑net, if for some reason none of the candidates worked,

        print("All candidates failed; falling back to uniform sequence.")
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print(f"fallback score = {best_score}")

    print(f"Chosen exponent gives score {best_score:.8f}")
