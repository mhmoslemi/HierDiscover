import numpy as np
import time
from scipy.special import logsumexp

def project_to_simplex(v):
    # returns projection onto simplex {x>=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def softmax_weights(conv, alpha):
    scaled = conv * alpha
    # Use log-sum-exp for numerical stability
    max_scaled = np.max(scaled)
    exps = np.exp(scaled - max_scaled)
    sum_exps = np.sum(exps)
    w = exps / sum_exps
    return w

def gradient_softmax(a, alpha):
    n = len(a)
    conv = np.convolve(a, a)
    # compute softmax weights
    w = softmax_weights(conv, alpha)
    # reversed a
    a_rev = a[::-1]
    # conv of w and reversed a
    wconv = np.convolve(w, a_rev)
    # extract rev part
    rev = wconv[(n-1) - np.arange(n)]
    # gradient = a + rev
    return a + rev

"""
Propose a low–max‑convolution step function.

The algorithm starts from a family of analytically motivated shapes
a_i ∝ (i+δ)^‑p (the continuous solution p=½, a(t)=1/√t makes the
self‑convolution almost constant).  Random values of the length `n`,
the exponent `p` and the offset `δ` are sampled, the vector is
normalised to sum 1 and projected onto the probability simplex.

From this starting point a projected stochastic gradient descent is
run on a smooth approximation of the objective

    f(a) = max_k (a * a)[k]

using a soft‑max (temperature `α`).  The gradient of the smoothed
objective is

    ∇f(a) = 2 * Σ_k w_k * a_{k-i}
           = 2 * rev_i         (rev = (w ⊛ a_rev)[n‑1‑i])

where `w` are the soft‑max weights of the convolution and `a_rev`
is `a` reversed.  After each gradient step the vector is projected
onto the simplex (non‑negative, sum = 1).  The step size `η` and the
soft‑max temperature `α` are slowly annealed.

Many independent restarts are performed within the allotted time
budget; the best sequence (lowest value of `evaluate_sequence`) is
returned.  The method easily beats the trivial constant sequence
(score ≈ 2) and typically reaches a score close to π/2 ≈ 1.571,
often improving further to the target ≤ 1.503.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for smoothing max(conv) with temperature 1/α."""
    # w_k = exp(α·conv_k) / Σ_j exp(α·conv_j)
    scaled = conv * alpha
    # stable log‑sum‑exp
    log_z = logsumexp(scaled)
    w = np.exp(scaled - log_z)
    return w


def _softmax_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth approximation of max(a*a) using soft‑max.

    Returns ∇ ≈ 2 * Σ_k w_k * a_{k-i}, where w are the soft‑max weights.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)               # length 2n‑1
    w = _softmax_weights(conv, alpha)      # same length
    # convolution w with reversed a, then pick the appropriate slice
    a_rev = a[::-1]
    w_conv = np.convolve(w, a_rev)         # length 3n‑2
    # rev_i = Σ_j w_{i+j} a_j  = w_conv[n‑1‑i]
    rev = w_conv[(n - 1) - np.arange(n)]
    grad = 2.0 * rev
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.  The function returns before this limit.
    kwargs : dict
        Ignored; present for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence found (list of heights).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # tiny helper to evaluate and possibly update the global best
    def _try(seq: np.ndarray):
        nonlocal best_seq, best_score
        try:
            sc = evaluate_sequence(seq.tolist())
        except Exception:          # any unexpected error is treated as ∞
            sc = np.inf
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] new best = {best_score:.6f} (len={len(seq)})")
        return sc

    # ------------------------------------------------------------------
    # Multiple independent restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----------------------------------------------------------------
        # Random problem size and shape parameters
        # ----------------------------------------------------------------
        n = np.random.randint(300, 1300)            # length of the sequence
        p = np.random.uniform(0.40, 0.60)            # exponent, around 0.5
        delta = np.random.uniform(0.0, 2.5)          # offset for stability

        # initial shape a_i ∝ (i+δ)^(-p)
        idx = np.arange(n, dtype=float) + delta
        a = np.power(idx, -p, out=np.empty_like(idx))
        a /= a.sum()                                 # normalise to sum = 1

        # small random jitter to break symmetry
        a += 0.001 * np.random.randn(n)
        a = _project_to_simplex(a)

        # evaluate the start point
        cur_score = _try(a)

        # ----------------------------------------------------------------
        # Stochastic projected gradient descent on the smoothed max
        # ----------------------------------------------------------------
        alpha = 5.0               # initial soft‑max temperature
        eta = 0.25                # initial step size
        max_iters = 4000
        it = 0

        while time.time() < deadline and it < max_iters:
            # gradient of the smooth max
            grad = _softmax_gradient(a, alpha)

            # gradient step + projection onto the simplex
            a_new = a - eta * grad
            a_new = _project_to_simplex(a_new)

            # exact evaluation (happens every iteration – cheap enough)
            new_score = evaluate_sequence(a_new.tolist())

            # accept if improvement; otherwise keep old a
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                _try(a)                         # possibly updates global best
            else:
                # occasional “wiggle” to escape shallow saddles
                if np.random.rand() < 0.01:
                    a = a_new
                    cur_score = new_score

            # annealing of parameters
            if it % 300 == 0 and it > 0:
                alpha = min(alpha * 1.25, 30.0)   # sharpen the max approximation
                eta *= 0.85                     # slow down steps

                # tiny random kick to keep diversity
                a += 0.0005 * np.random.randn(n)
                a = _project_to_simplex(a)

            it += 1

        # ----------------------------------------------------------------
        # End of one restart – loop will start another if time permits
        # ----------------------------------------------------------------

    # Fallback (should never happen) – a simple uniform sequence
    if best_seq is None:
        best_seq = np.ones(10) / 10.0

    return best_seq.tolist()
