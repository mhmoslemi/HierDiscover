def greedy_set(n,g):
    # random ordering for positions
    indices = np.arange(n)
    np.random.shuffle(indices)
    S = []
    counts = np.zeros(2*n-1, dtype=int)  # ordered pair counts
    for idx in indices:
        feasible = True
        # create temporary increments
        # count self contribution
        if counts[2*idx] + 1 > g:
            continue
        # For each i in S:
        for i in S:
            s = i + idx
            if counts[s] + 2 > g:
                feasible = False
                break
        if feasible:
            # add
            S.append(idx)
            counts[2*idx] += 1
            for i in S[:-1]:  # existing ones before added idx
                s = i + idx
                counts[s] += 2
    return np.array(S)

"""
Greedy Sidon‑like construction.

The evaluation function (after normalisation) equals
    2 * len(seq) * max(conv(seq,seq)) / (sum(seq)**2).

If we use a binary sequence (values 0/1) then
    sum = m   (number of 1‑entries)
    max(conv) = C   (maximum ordered pair count for a sum index).

Hence the score reduces to 2·n·C / m².  By keeping C small while
maximising m we can drive the score well below the uniform bound 2.
The greedy algorithm builds a set S ⊂ {0,…,n‑1} such that no sum index
receives more than `g_target` ordered pairs.  For a given length n and
target C=g_target the theoretical limit is m ≈ √(2·n·g_target), which
gives a score ≈ 1.  We repeatedly try random lengths and targets,
keep the best sequence, and finally truncate trailing zeros to keep n
as small as possible.
"""

import numpy as np
import time

def greedy_set(n: int, g_target: int) -> np.ndarray:
    """
    Build a set of indices S ⊆ [0, n) such that the ordered pair count for
    any sum lies ≤ g_target.
    Returns the array of chosen indices (unsorted).
    """
    # ordered pair count for each possible sum index 0 … 2*n-2
    counts = np.zeros(2 * n - 1, dtype=np.int16)
    S = []                     # chosen indices
    order = np.arange(n)
    np.random.shuffle(order)   # random order of candidate positions

    for idx in order:
        # check self‑pair (idx,idx) contributes 1
        if counts[2 * idx] + 1 > g_target:
            continue

        # check all pairs with already chosen indices – each contributes 2
        ok = True
        for i in S:
            s = i + idx
            if counts[s] + 2 > g_target:
                ok = False
                break
        if not ok:
            continue

        # add idx to the set and update the count array
        counts[2 * idx] += 1                     # self‑pair
        for i in S:
            s = i + idx
            counts[s] += 2                       # (i,idx) and (idx,i)
        S.append(idx)

    return np.asarray(S, dtype=np.int32)

    evaluate_sequence().  The search consists of many independent greedy
    constructions with random length n and a small collision bound g_target.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small guard margin

    # keep the best sequence and its score
    best_seq = None

    # optionally start from the provided SOTA construction

    # main stochastic search
    while time.time() < deadline:
        # random length (keep it moderate for speed)
        n = np.random.randint(800, 3000)

        # random collision bound – a small integer > 1
        g_target = np.random.randint(2, 6)   # try 2 … 5

        # greedy construction
        S = greedy_set(n, g_target)
        m = len(S)
        if m < 2:
            continue

        # compute the maximum ordered‑pair count directly from the count array
        # (we can reuse the counts from greedy_set, but recompute for clarity)
        a = np.zeros(n, dtype=np.int8)
        a[S] = 1
