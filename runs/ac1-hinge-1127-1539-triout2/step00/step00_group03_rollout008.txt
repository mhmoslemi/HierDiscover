"""
Local‑search optimizer for the step‑function evaluation.

The objective is
    2·n·max(conv(a,a)) / (sum(a))**2
where `conv` is the full linear convolution.
The function is invariant under scaling of `a`, therefore we
only care about the *shape* of the sequence.

The algorithm starts from a simple decreasing exponential shape
(which already improves the naïve uniform bound 2 → ≈1.8) and
repeatedly reduces the peak of the self‑convolution by stepping
opposite to its gradient.  Random perturbations and occasional
re‑initialisations are used to escape local minima.

A fast re‑implementation of `evaluate_sequence` is used inside the
search to avoid the costly second convolution that the original
routine performs.
"""

import time
import numpy as np

def _fast_eval(seq):
    """Fast replica of the provided `evaluate_sequence` (no security checks)."""
    arr = np.array(seq, dtype=np.float64)
    # apply the same clipping as the reference implementation
    np.clip(arr, 0.0, 1000.0, out=arr)
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(arr, arr)
    return float(2.0 * len(arr) * conv.max() / (s * s))


def _gradient_multi_peak(seq, conv, max_val, rel_thr=0.98):
    """
    Gradient of the maximum convolution entry.
    If several entries are close to the maximum (within `rel_thr`),
    their contributions are summed – this smoothes the descent.
    """
    n = len(seq)
    # indices whose value is near the maximum
    peaks = np.where(conv >= rel_thr * max_val)[0]

    grad = np.zeros_like(seq)
    idx = np.arange(n)

    for k in peaks:
        j = k - idx                     # partner indices for shift k
        mask = (j >= 0) & (j < n)
        # derivative of a_i*a_{k-i} w.r.t a_i is 2*a_{k-i}
        grad[mask] += 2.0 * seq[j[mask]]
    return grad

    restart_prob = 0.001     # occasional fresh restart
    iteration = 0

    print(f"Initial best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # convolution and location of the current maximum
        conv = np.convolve(cur, cur)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # gradient that points *up* the current peak; we move *against* it
        grad = _gradient_multi_peak(cur, conv, max_val, rel_thr=0.98)

        improved = False

        # try several step sizes in a simple line‑search
        for f in step_factors:
            # scale step to the magnitude of the gradient (avoid overflow)
            scale = np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else 1.0
            step = f * max_val / scale
            cand = cur - step * grad
            np.maximum(cand, 0.0, out=cand)          # enforce non‑negativity

            # keep the sum comfortably away from zero (scaling does not matter)
            if cand.sum() < 0.01:
                cand *= (0.1 / max(cand.sum(), 1e-12))

            cand_score = _fast_eval(cand)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                cur = cand
                improved = True
                if iteration % 100 == 0:
                    print(f"Iter {iteration:05d}: improvement → {best_score:.6f}")
                break   # accept the first improvement found

        # ------------------------------------------------------------------
        # If no gradient step helped, perform a small random perturbation.
        # ------------------------------------------------------------------
        if not improved and np.random.rand() < perturb_prob:
            idx = np.random.randint(n)
            # multiplicative Gaussian noise (relative to current magnitude)
            noise = np.random.randn() * 0.05 * max(cur[idx], 1e-12)
            cand = cur.copy()
            cand[idx] = max(0.0, cand[idx] + noise)

            # optional tiny global renormalisation to avoid tiny sums
            if cand.sum() < 0.01:
                cand *= (0.1 / max(cand.sum(), 1e-12))

            cand_score = _fast_eval(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                cur = cand
                if iteration % 100 == 0:
                    print(f"Iter {iteration:05d}: random tweak → {best_score:.6f}")

        # ------------------------------------------------------------------
        # Occasionally restart from a fresh exponential with a random decay.
        # ------------------------------------------------------------------
        if np.random.rand() < restart_prob:
            delta = np.random.uniform(0.0001, 0.001)
            cur = np.exp(-delta * np.arange(n))
            cur += np.random.rand(n) * 1e-6
            np.clip(cur, 0.0, 1000.0, out=cur)
            cur_score = _fast_eval(cur)
            print(f"Iter {iteration:05d}: restart (δ={delta:.6f}) score={cur_score:.6f}")
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
