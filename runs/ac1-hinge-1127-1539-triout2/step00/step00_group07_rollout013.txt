def project_simplex(v):
    # from https://gist.github.com/...
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def _project_onto_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        return np.ones_like(v) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _make_uniform(n):
    return np.full(n, 1.0/n)

    w = 1.0 / ((i + 1) * (n - i))
    w /= w.sum()
    return w

"""
Projected‑gradient optimisation on the probability simplex.

The metric returned by ``evaluate_sequence`` equals
    2·n·max(conv(f,f)) / (sum(f)^2) .
With the sum fixed, minimizing the infinity‑norm of the self‑convolution
makes the bound as small as possible.  The algorithm works with a
probability vector ``w`` (non‑negative, sum = 1) and repeatedly:

1. Computes the convolution ``c = np.convolve(w, w)``.
2. Forms a smooth approximation of ``max(c)`` by a soft‑max with
   temperature ``α``:  p_k ∝ exp(α·(c_k−max(c))).
3. Uses the gradient of this smooth max:
      ∇ = w + (p * w_rev)  (where w_rev = w[::-1]).
   The convolution ``np.convolve(p, w_rev)`` yields the second term.
4. Takes a small step ``w ← w – η·∇`` and projects back onto the
   probability simplex (Euclidean projection).
5. Occasionally injects a tiny random jitter to avoid stagnation.

Several simple initialisations are tried (uniform, a U‑shape
1/((i+1)*(n−i)), a sqrt‑shape 1/√((i+1)*(n−i)) and a random start).
The best sequence found before the time limit is returned.
The actual scale of the vector is irrelevant – the evaluation function

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a real vector onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to the centre of the simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            shifted = conv - max_c
            exp_vals = np.exp(alpha * shifted)
            p = exp_vals / exp_vals.sum()          # probabilities, sum = 1

            # ------------------- gradient of smooth max ------------------
            w_rev = w[::-1]
            term = np.convolve(p, w_rev)           # length 3n‑2
            grad = w + term[n - 1 : n - 1 + n]      # the needed slice

            # ------------------- gradient step + projection ------------
            w = w - eta * grad
            w = _project_onto_simplex(w)

            # ------------------- schedule & occasional jiggle ----------
            iteration += 1
            if iteration % 200 == 0:
                eta = max(1e-4, eta * 0.9)          # decay step size
                alpha = min(50.0, alpha * 1.05)     # sharpen soft‑max

            if iteration % 500 == 0:
                # random jitter on a random coordinate
                idx = rng.integers(n)
                w[idx] = max(0.0, w[idx] + rng.normal(scale=0.01))

            # early exit if we have already crossed the target bound
            if best_score <= 1.503:
                break

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    if best_w is None:
        best_w = _make_uniform(n)

    # The evaluator is scale‑invariant, any positive scaling works.
    final_seq = (best_w * 1.0).tolist()
    print(f"Returning sequence of length {n} with final score {best_score:.6f}")
    return final_seq
