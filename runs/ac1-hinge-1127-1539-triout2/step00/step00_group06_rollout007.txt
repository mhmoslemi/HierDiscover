"""
Propose a step‑function (non‑negative sequence) that minimises

    E(a) = 2 * n * max(a * a) / (sum(a)^2)

where * denotes the full discrete convolution.
The algorithm is a simple stochastic projected‑gradient descent
on a smooth approximation of the max‑operator.
It repeatedly:

* picks a length n (small to medium, 2‑400)
* starts from a random point on the probability simplex
* repeatedly moves opposite to the gradient of   log‑sum‑exp(β·conv)
  (a smooth max) and projects back onto the simplex
* keeps the best sequence seen over the whole runtime.

The projection onto the simplex guarantees non‑negativity and a fixed
sum (the objective is scale‑invariant).  Random restarts and a short
final refinement are used to exploit the whole time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex {x >= 0, sum(x)=1}
def _project_simplex(v):
    v = np.asarray(v, dtype=float)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# One optimisation run for a fixed length n.
# It stops when the supplied deadline is reached.
def _optimise_length(n, deadline):
    # initialise a random point on the simplex
    a = np.random.rand(n)
    a /= a.sum()

    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())

    # hyper‑parameters (tuned empirically)
    lr = 0.5 / n            # initial learning rate
    beta = 1.0              # smooth‑max temperature
    decay_lr = 0.9995
    inc_beta = 1.002
    max_beta = 80.0

    while time.time() < deadline:
        # full convolution of the current sequence
        conv = np.convolve(a, a)                # length 2*n-1
        # soft‑max weights (smooth approximation of the max)
        z = beta * conv
        max_z = np.max(z)                       # for numerical stability
        exp_z = np.exp(z - max_z)
        w = exp_z / exp_z.sum()                # shape (2*n-1,)

        # gradient of the smooth max w.r.t. a:
        #   ∂/∂a_i  sum_k w_k * sum_j a_j a_{k-j}
        #   = 2 * sum_k w_k * a_{k-i}  (indices out of range give 0)
        rev_a = a[::-1]
        grad_full = np.convolve(w, rev_a)       # length 3*n-2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]  # keep the n relevant components

        # gradient step and projection back onto the simplex
        a = a - lr * grad
        # occasional tiny noise to avoid stagnation
        if np.random.rand() < 0.03:
            a += 1e-5 * np.random.randn(n)

        a = _project_simplex(a)

        # evaluate occasionally (cheap enough) and keep the best
        if np.random.rand() < 0.02:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # schedule updates
        lr *= decay_lr
        beta = min(beta * inc_beta, max_beta)

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1000, **kwargs):
