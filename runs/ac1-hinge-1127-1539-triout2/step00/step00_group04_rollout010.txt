def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1  # keep margin

    best_seq = None
    best_score = float('inf')
    iteration = 0
    while time.time() < deadline:
        # Choose length randomly between 300 and 2000
        n = np.random.randint(300, 2000)  # moderate size
        # Initialize sequence uniform or random
        a = np.random.rand(n)
        a = a / a.sum()
        # Gradient descent
        # hyperparameters
        beta = 5.0
        lr = 0.2
        decay = 0.999  # optional
        max_iters = 500  # limited per restart
        # optionally early stop if time is near deadline
        for it in range(max_iters):
            if time.time() >= deadline:
                break
            # compute convolution using np.convolve (full). This returns shape 2n-1
            conv = np.convolve(a, a)
            # compute smooth max
            conv_max = conv.max()
            # compute weights
            # use stable softmax
            exp_vals = np.exp(beta * (conv - conv_max))
            w = exp_vals / exp_vals.sum()
            # compute gradient using convolution of w with reversed a
            grad = np.convolve(w, a[::-1])
            # slice to length n: grad_slice = 2 * grad[n-1 : n-1 + n]
            grad_slice = 2.0 * grad[n-1 : n-1 + n]
            # gradient descent step
            a = a - lr * grad_slice
            # project nonnegative
            a = np.maximum(a, 0.0)
            # renormalize
            sum_a = a.sum()
            if sum_a < 1e-12:
                a = np.random.rand(n)
                a = a / a.sum()
                continue
            a = a / sum_a
            # optional adapt learning rates
            beta = min(beta * 1.01, 30.0)  # increase smoothness gradually
            lr *= 0.9995  # decrease step slowly
        # Evaluate final a using full evaluate function (to match official)
        score = evaluate_sequence(a.tolist())
        print(f"Restart {iteration}, n={n}, final score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"New best: {best_score}")
        iteration += 1
    # At end, return best_seq (list)
    return best_seq

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Gradient descent with smooth max objective to minimise the evaluation function.
    Starts from many random initializations, each run performs a few iterations
    of projected gradient descent on a smooth approximation of the infinity norm
    of the self‑convolution. The best sequence found within the time budget is
    returned.
    """
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_score = float('inf')
    best_seq = None
    run = 0

    while time.time() < deadline:
        # choose a length, bias towards values similar to existing construction
        n = np.random.randint(400, 2000)  # moderate size for speed
        # start from random Dirichlet distribution
        a = np.random.rand(n)
        a = a / a.sum()

        # hyper‑parameters for this run
        max_iters = np.random.randint(100, 300)  # number of GD steps
        beta = 1.0 + np.random.rand() * 3.0      # smoothing parameter start
        lr = 0.5 + np.random.rand() * 0.5        # learning rate

        # run gradient descent
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # convolution a*a
            conv = np.convolve(a, a)
            max_conv = conv.max()
            # smooth max (soft‑max) weights
            # use stable softmax
            exp_vals = np.exp(beta * (conv - max_conv))
            w = exp_vals / exp_vals.sum()

            # gradient = 2 * (w * a) convolution (w with reversed a)
            grad_full = np.convolve(w, a[::-1])
            grad = 2.0 * grad_full[n-1:n-1 + n]

            # gradient descent step
            a = a - lr * grad

            # small random jitter to escape plateaus
            if it % 20 == 0:
                jitter = np.random.randn(n) * 0.001
                a += jitter

            # project to feasible set (non‑negative, re‑normalise)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                # re‑initialise if we collapsed to zero
                a = np.random.rand(n)
                a = a / a.sum()
                continue
            a = a / s

            # slowly increase smoothing and decrease step size
            beta = min(beta * 1.02, 20.0)
            lr *= 0.995

        # final evaluation of this candidate
        score = evaluate_sequence(a.tolist())
        print(f"Run {run:3d}: n={n:4d}, iters={max_iters}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"  New best score: {best_score:.6f}")

        run += 1

    # If nothing found (should not happen) fall back to uniform sequence
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback uniform: score={best_score}")

    print(f"Best score after search: {best_score:.6f}")
    return best_seq

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Gradient‑descent based search for a step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    where ``conv`` is the linear convolution of the coefficient vector with itself.
    The method works as follows:

    1.  Randomly pick a length ``n`` (400 – 2000) and initialise a non‑negative
        vector ``a`` (Dirichlet / uniform density) and normalise it so that
        ``sum(a) == 1`` (the objective is scale‑invariant).

    2.  Run a small projected‑gradient loop.
        *   The infinity‑norm of the convolution is approximated by a smooth
            (soft‑max) version:
            ``w_k ∝ exp(beta * (conv_k - max(conv)))``.
        *   The gradient of this smooth objective w.r.t. ``a`` can be written
            analytically as a convolution:
            ``grad = 2 * (w * rev(a))`` where ``rev(a) = a[::-1]``.
        *   A simple gradient step ``a ← a - lr * grad`` is taken,
            followed by projection onto the feasible set (non‑negative, then
            re‑normalised to keep the sum equal to one).
        *   The smoothing parameter ``beta`` is slowly increased,
            while the learning rate ``lr`` is slowly decayed.
        *   Small random jitter is added every few iterations to help escape
            plateaus.

    3.  After the inner loop the candidate is evaluated with the provided
        ``evaluate_sequence`` function.  The best candidate seen so far is kept.

    4.  The outer loop repeats as many random restarts as the time budget allows.
        If, for any reason, no candidate is generated, a uniform sequence of
        length 1000 is returned as a safe fallback.

    The algorithm uses only NumPy (no external solvers) and runs well within
    the 1000 s limit.  Empirically it discovers sequences with scores
    considerably below the initial upper bound of 2.0 (target ~ 1.503).
    """
    import time
    import numpy as np

    # --------------------------------------------------------------------
    # Helper: fast evaluation (identical to evaluate_sequence for non‑negative)
    # --------------------------------------------------------------------
    def fast_score(vec):
        """Return the objective value for a non‑negative vector (sum may be ≠1)."""
        n = len(vec)
        conv = np.convolve(vec, vec)
        maxb = conv.max()
        s = np.sum(vec)
        # protect against tiny denominator (the real evaluate_sequence would
        # return np.inf, but we never let sum go to zero)
        return float(2.0 * n * maxb / (s * s))

    # --------------------------------------------------------------------
    # Search loop
    # --------------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave a tiny margin
    best_score = float("inf")
    best_seq = None
    run = 0

    while time.time() < deadline:
        # ----------------------------------------------------------------
        # 1) Random problem size and initial vector
        # ----------------------------------------------------------------
        n = np.random.randint(400, 2000)           # moderate size – fast conv
        a = np.random.rand(n).astype(np.float64)   # positive entries
        a /= a.sum()                               # normalise (scale‑invariant)

        # ----------------------------------------------------------------
        # 2) Gradient‑descent parameters (randomised per run for diversity)
        # ----------------------------------------------------------------
        max_iters = np.random.randint(120, 250)    # iterations per restart
        beta = 1.0 + 3.0 * np.random.rand()         # start smoothing
        lr = 0.5 + 0.5 * np.random.rand()           # initial step size

        # ----------------------------------------------------------------
        # 3) Main optimisation loop
        # ----------------------------------------------------------------
        for it in range(max_iters):
            # stop if we are out of time
            if time.time() >= deadline:
                break

            # convolution a*a (full length 2n‑1)
            conv = np.convolve(a, a)

            # ----------------------------------------------------------------
            # soft‑max approximation of the max (stable version)
            # ----------------------------------------------------------------
            mx = conv.max()
            # exponentials are ≤ 1 because conv‑mx ≤ 0
            exp_vals = np.exp(beta * (conv - mx))
            w = exp_vals / exp_vals.sum()          # weights sum to 1

            # ----------------------------------------------------------------
            # gradient = 2 * (w * rev(a))  (full convolution then slice)
            # ----------------------------------------------------------------
            grad_full = np.convolve(w, a[::-1])
            grad = 2.0 * grad_full[n - 1:n - 1 + n]   # length‑n gradient

            # gradient step
            a -= lr * grad

            # occasional tiny random perturbation (helps escape flat regions)
            if (it % 20) == 0:
                a += np.random.randn(n) * 0.001

            # ------------------------------------------------------------
            # Project back onto the feasible set: a ≥ 0, sum(a) = 1
            # ------------------------------------------------------------
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:          # degenerated – re‑initialise
                a = np.random.rand(n)
                a /= a.sum()
                continue
            a /= s

            # slowly tighten the soft‑max and shrink the step size
            beta = min(beta * 1.02, 20.0)
            lr *= 0.995

        # ----------------------------------------------------------------
        # 4) Evaluate the final candidate of this restart
        # ----------------------------------------------------------------
        cur_score = float(evaluate_sequence(a.tolist()))   # official check
        print(f"Run {run:4d}: n={n:5d}, iters={max_iters:3d}, score={cur_score:.6f}")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"   *** New best (score={best_score:.6f}) ***")

        run += 1

    # --------------------------------------------------------------------
    # Fallback: should never happen but guarantees a valid return value
    # --------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence, score={best_score}")

    print(f"Best score after search: {best_score:.6f}")
    return best_seq
