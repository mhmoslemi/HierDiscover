    best_seq = current.copy()

    iteration = 0
    temperature = 1.0
    while time.time() < deadline:
        iteration += 1
        # choose move type
        if np.random.rand() < 0.3:
            # gradient step
            conv = np.convolve(current, current)
            k = np.argmax(conv)
            g = np.zeros_like(current)
            for i in range(len(current)):
                j = k - i
                if 0 <= j < len(current):
                    g[i] = current[j]
            # step size decays
            eps = temperature * 0.01  # small
            new = current - eps * g
            new = np.maximum(new, 0)

            new[j] += delta

            new = np.random.dirichlet(np.ones(len(current)))

    np.random.seed(seed)

    best_seq = None

    def update(seq):

        if score < best_score:

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1
    best_seq = None
    best_score = float('inf')
    restarts = 0
    while time.time() < deadline:
        # random n each restart
        n = np.random.randint(200, 800)
        a = np.random.dirichlet(np.ones(n))
        # optional smoothing (moving average)
        if np.random.rand() < 0.5:
            kernel = np.ones(3) / 3
            a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
        cur = a
        cur_score = evaluate_sequence(cur.tolist())
        # local descent
        temp = 1.0
        it = 0
        while time.time() < deadline and it < 2000:
            it += 1
            # propose move
            if np.random.rand() < 0.3:
                # gradient based
                conv = np.convolve(cur, cur)
                k = np.argmax(conv)
                g = np.zeros_like(cur)
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        g[i] = cur[j]
                eps = temp * 0.01
                new = cur - eps * g
                new = np.maximum(new, 0)
            else:
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = np.random.rand() * cur[i] * 0.5
                    new = cur.copy()
                    new[i] -= delta
                    new[j] += delta
                else:
                    continue
            new = new / new.sum()
            new_score = evaluate_sequence(new.tolist())
            if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                cur = new
                cur_score = new_score
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"New best {best_score:.6f} at restart {restarts}, length {n}")
                # early break if good enough
                if best_score < 1.503:
                    return best_seq.tolist()
            # temperature schedule
            if it % 500 == 0:
                temp *= 0.9
        restarts += 1
    return best_seq.tolist()

"""
Propose a coefficient sequence that (hopefully) beats the uniform 2.0 bound.

The algorithm is a simple stochastic search:
1. Randomly initialise a non‑negative vector on the simplex (Dirichlet draw,
   geometric patterns, or uniform).  The length is chosen randomly between 200
   and 800 – larger lengths give a lower theoretical floor.
2. Perform a short local optimisation for each restart:
   * compute the full convolution a*a,
   * locate the index k where the convolution attains its maximum,
   * form a gradient‑like direction g_i = a_{k-i} (zero if the index is out of
     range).  Moving mass opposite to g reduces the contribution to that
     maximal entry.
   * propose a move that either follows this gradient direction or simply
     transfers a random amount of mass from one coordinate to another.
   * renormalise the vector to sum to one (the evaluation function is
     homogeneous, the exact scale does not matter) and evaluate the score.
   * accept the move if it improves the score, otherwise accept it with a
     Metropolis probability that slowly cools (simulated annealing).
3. Keep the best sequence seen over all restarts.  The loop stops when the
   allotted time (≈1000 s) is exhausted or when a score better than the target
   1.503 is found.

The search is deliberately lightweight – each iteration uses only NumPy
operations and a single call to ``evaluate_sequence`` – so thousands of
iterations can be performed within the time limit.
"""

import time
import math
import numpy as np

def _evaluate(seq):
    """Wrapper for the provided evaluate_sequence (already imported)."""
    return evaluate_sequence(seq.tolist())

def _random_sequence(n):
    """Generate a random start vector of length n on the simplex."""
    choice = np.random.rand()
    if choice < 0.4:
        # Dirichlet (balanced)
        a = np.random.dirichlet(np.ones(n))
    elif choice < 0.7:
        # Dirichlet with concentration >1 (more uniform)
        alpha = np.random.uniform(1.5, 5.0)
        a = np.random.dirichlet(np.full(n, alpha))
    elif choice < 0.9:
        # Geometric decay (r close to 1)
        r = np.random.uniform(0.85, 0.99)
        w = r ** np.arange(n)
        a = w / w.sum()
    else:
        # Simple uniform vector
        a = np.full(n, 1.0 / n)
    return a

def _gradient_step(a, eps):
    """Take one step that tries to reduce the current maximal convolution entry."""
    conv = np.convolve(a, a)          # full convolution
    k = int(np.argmax(conv))          # index of the worst (largest) entry
    n = a.shape[0]
    g = np.zeros_like(a)
    # g_i = a_{k-i} when the partner index lies inside the vector
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            g[i] = a[j]
    # move opposite to the gradient and keep non‑negative entries
    new = a - eps * g
    new = np.maximum(new, 0.0)
    # renormalise to the simplex (scale does not affect the objective)
    s = new.sum()
    if s > 0:
        new = new / s
    else:
        # degenerate case – fall back to a fresh random vector
        new = _random_sequence(n)
    return new

def _random_transfer(a):
    """Transfer a random fraction of mass from one coordinate to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    if i == j or a[i] == 0:
        return a.copy()
    delta = np.random.rand() * a[i] * 0.5   # at most half of the donor mass
    new = a.copy()
    new[i] -= delta
    new[j] += delta
    # renormalise (the total mass is unchanged, but protect against round‑off)
    s = new.sum()
    if s > 0:
        new = new / s
    else:
        new = _random_sequence(n)
    return new

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence.  The function runs for approximately ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    best_seq = None
    best_score = float("inf")
    restarts = 0

    # ----------------------------------------------------------------------
    # Main outer loop – many independent restarts with a short local search.
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        restarts += 1
        # pick a problem size; larger n gives a lower theoretical floor
        n = np.random.randint(200, 800)

        # initialise a random feasible vector
        cur = _random_sequence(n)
        cur_score = _evaluate(cur)

        # keep the best seen in this restart so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[{time.time():.1f}] New global best {best_score:.6f} "
                  f"(len={n}, restart={restarts})")

        # ------------------------------------------------------------------
        # Local optimisation – simulated‑annealing style.
        # ------------------------------------------------------------------
        temperature = 1.0
        it = 0
        while time.time() < deadline and it < 2500:
            it += 1

            # either a gradient‑guided move or a simple random transfer
            if np.random.rand() < 0.35:
                # gradient direction based on the current maximal convolution entry
                eps = temperature * 0.02
                cand = _gradient_step(cur, eps)
            else:
                # random mass transfer
                cand = _random_transfer(cur)

            cand_score = _evaluate(cand)

            # acceptance rule (Metropolis)
            improve = cand_score < cur_score
            accept = improve or (np.random.rand() <
                                 math.exp(-(cand_score - cur_score) /
                                          max(temperature, 1e-12)))
            if accept:
                cur = cand
                cur_score = cand_score

                # record a new global optimum
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.copy()
                    print(f"[{time.time():.1f}] New global best {best_score:.6f} "
                          f"(len={n}, restart={restarts}, it={it})")
                    if best_score < 1.503:        # target reached – early exit
                        return best_seq.tolist()

            # temperature schedule – slowly cool down
            if it % 500 == 0:
                temperature *= 0.90

        # occasional extra exploration: start again from the current best with a
        # tiny perturbation (helps escaping shallow basins)
        if restarts % 5 == 0 and best_seq is not None:
            cur = best_seq.copy()
            perturb = np.random.randn(cur.shape[0]) * 0.005
            cur = np.maximum(cur + perturb, 0.0)
            if cur.sum() == 0:
                cur = _random_sequence(cur.shape[0])
            else:
                cur = cur / cur.sum()
            cur_score = _evaluate(cur)

    # ----------------------------------------------------------------------
    # Return the best sequence discovered.
    # ----------------------------------------------------------------------
    if best_seq is None:
        # very unlikely – fall back to a simple uniform vector
        best_seq = np.full(100, 1.0 / 100)
    print(f"Finished search: best score {best_score:.6f} "
          f"(length {len(best_seq)}) after {restarts} restarts.")
    return best_seq.tolist()
