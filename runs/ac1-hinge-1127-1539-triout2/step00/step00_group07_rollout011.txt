def gradient_step(seq, step=0.02):
    n = len(seq)
    a = np.array(seq, dtype=float)
    conv = np.convolve(a, a)
    M = conv.max()
    # find indices where conv close to M
    tol = 1e-12  # maybe tolerance
    max_idxs = np.where(np.abs(conv - M) < 1e-12)[0]  # keep all maxima exactly

    # compute gradient
    grad = np.zeros_like(a)
    for k in max_idxs:
        # for each i, if 0 <= i < n and 0 <= k-i < n:
        # add 2*a[k-i]
        # This can be vectorized:
        # create index array i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        grad[i_range] += 2 * a[j_range]
    # move opposite direction
    new_a = a - step * grad
    # ensure nonnegative
    new_a = np.maximum(new_a, 0.0)
    # renormalize sum to original sum (or to 1)
    s = new_a.sum()
    if s < 1e-6:  # avoid too small sum
        new_a = a
    else:
        new_a = new_a * (a.sum() / s)
    return new_a.tolist()
```

def gradient_step(seq, step=0.01):
    a = np.array(seq, dtype=np.float64)
    n = a.size
    conv = np.convolve(a, a)
    k = np.argmax(conv)
    # compute gradient
    grad = np.zeros_like(a)
    i_start = max(0, k - (n-1))
    i_end = min(n-1, k)
    # Build i_range
    i_range = np.arange(i_start, i_end+1)
    j_range = k - i_range
    grad[i_range] += 2 * a[j_range]
    # move opposite direction
    new_a = a - step * grad
    # clip to nonneg
    new_a = np.clip(new_a, 0, None)
    # renormalize to keep sum (preserve scale, not too small)
    s = new_a.sum()
    if s < 1e-8:
        # fallback: keep original
        new_a = a
        s = a.sum()
    new_a = new_a * (a.sum() / s)  # maintain original sum
    return new_a.tolist()
```

def random_perturb(seq, magnitude=0.05):
    a = np.array(seq, dtype=np.float64)
    n = a.size
    # choose a few indices
    num = max(1, int(0.01 * n))  # about 1% of indices
    idx = np.random.choice(n, size=num, replace=False)
    noise = np.random.randn(num) * magnitude * a.mean()
    a[idx] += noise
    # Clip nonneg
    a = np.clip(a, 0, None)
    # Ensure sum > 0.01
    if a.sum() < 0.01:
        a += 0.01 / n
    # Optionally renorm to keep average scale
    # Not necessary.
    return a.tolist()

def solve_convolution_lp(f_seq, rhs):
    import numpy as np
    from scipy.optimize import linprog
    n = len(f_seq)
    # objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    # Build constraint matrix for each k
    rows = []
    bounds = []
    for k in range(2*n-1):
        row = np.zeros(n)
        # for each i where j = k-i in [0,n-1]
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # fill row[j]= f_seq[i] where j = k-i
        for i in range(i_start, i_end+1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        bounds.append(rhs)
    # non-negativity constraints: b >= 0 -> -b <= 0
    A_ub = np.vstack(rows + [-np.eye(n)])
    b_ub = np.hstack(bounds + [np.zeros(n)])
    # solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None

"""
Propose a step‑function (non‑negative height vector) that minimizes
evaluate_sequence.  

The algorithm works in three phases:

1. **Initialisation** – try a few deterministic seeds (the provided SOTA
   sequence, a uniform vector, a perturbed uniform vector and a fully
   random vector) and keep the best one.

2. **Local optimisation loop** – repeatedly modify the current vector.
   The modifications are a mixture of:
     * a cheap projected‑gradient step that pushes mass away from the
       convolution index where the current maximum occurs;
     * small random perturbations (Gaussian noise on a few coordinates);
     * occasional mixing with a freshly generated random vector;
     * a very sparse simulated‑annealing acceptance rule to escape plateaus.

   After each modification the candidate is evaluated; if it improves the
   best score it is kept.  A simple temperature schedule allows occasional
   uphill moves.

3. **Periodic LP refinement** – every few hundred iterations a linear‑
   programming step (the same LP used in the reference SOTA code) is run
   to obtain a direction `g` that maximises the ℓ¹‑norm while keeping the
   convolution bound `‖f*g‖∞` ≤ current bound.  Several mixing ratios
   between the current vector and the normalised LP solution are tried;
   any improvement is accepted.

The loop stops just before the allotted budget (`budget_s` seconds) and
the best sequence found is returned.  All entries are kept non‑negative,
clipped to 1000 (the evaluator does this automatically) and the total
sum is never allowed to fall below 0.01, preventing the “inf” case.
"""
import time
import numpy as np
from scipy.optimize import linprog


def _solve_convolution_lp(f_seq, rhs):
    """Solve  max∑b_j  s.t.  conv(f,b) ≤ rhs,  b ≥ 0.
    Returns the optimal b as a NumPy array or None on failure."""
    n = len(f_seq)
    # objective: minimise -∑b  === maximise ∑b
    c = -np.ones(n)

    # Build A_ub • b ≤ b_ub  for each convolution index k
    rows = []
    ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # fill the non‑zero entries of the row:
        # for i in [i_start,i_end] we have j = k - i
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        ub.append(rhs)

    # non‑negativity:  -b ≤ 0
    rows.append(-np.eye(n))
    ub.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(ub)

    # use HiGHS (fast and robust)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if res.success:
        return res.x
    return None


def _gradient_step(seq, step=0.02):
    """Projected gradient step that reduces the current maximal convolution."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.size
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))

    # gradient of the active convolution entry:
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_range = np.arange(i_start, i_end + 1, dtype=int)
    j_range = k - i_range
    grad[i_range] += 2.0 * a[j_range]

    # move opposite to the gradient, keep non‑negative, renormalise sum
    new_a = a - step * grad
    new_a = np.clip(new_a, 0.0, None)

    s = new_a.sum()
    if s < 1e-12:                 # safeguard – should never happen
        new_a = a
        s = a.sum()
    new_a *= a.sum() / s           # keep the original ℓ¹‑norm
    return new_a.tolist()


def _random_perturb(seq, magnitude=0.05):
    """Add small Gaussian noise to a random 1 % of the coordinates."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.size
    num = max(1, int(0.01 * n))
    idx = np.random.choice(n, size=num, replace=False)
    noise = np.random.randn(num) * magnitude * a.mean()
    a[idx] += noise
    a = np.clip(a, 0.0, None)

    if a.sum() < 0.01:               # never let the total drop too low
        a += 0.01 / n
    return a.tolist()


def _random_restart(n):
    """Generate a fresh random vector with the usual scaling."""
    a = np.random.rand(n)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n)
        s = n
    # scale to the same ℓ¹‑norm used by the reference code
    a *= np.sqrt(2.0 * n) / s
    return a.tolist()


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence with a low value of evaluate_sequence().
    The function runs for at most `budget_s` seconds and returns the
    best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # a little safety margin

    # ------------------------------------------------------------------
    # 1) initialise – try a few simple candidates and keep the best one
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 2000

    initial_candidates = []

    # a) the provided state‑of‑the‑art sequence (if present)
    if "height_sequence_1" in globals():
        initial_candidates.append(list(height_sequence_1))

    # b) uniform vector (all ones)
    initial_candidates.append([1.0] * base_len)

    # c) a slightly noisy uniform vector
    initial_candidates.append([1.0 + np.random.randn() * 0.01 for _ in range(base_len)])

    # d) a purely random vector
    initial_candidates.append(np.random.rand(base_len).tolist())

    best_seq = None
    best_score = float("inf")
    for cand in initial_candidates:
        try:
            sc = evaluate_sequence(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
        except Exception:
            continue

    if best_seq is None:                     # fallback – should never trigger
        best_seq = [1.0] * base_len
        best_score = evaluate_sequence(best_seq)

    print(f"[init] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) main optimisation loop
    # ------------------------------------------------------------------
    curr_seq = best_seq.copy()
    curr_score = best_score

    iteration = 0
    no_improve = 0
    temperature = 1.0
    temp_decay = 0.99995

        r = np.random.rand()

        # ----- choose a local move ------------------------------------
        if r < 0.2:

            new_seq = _gradient_step(curr_seq, step=0.02)
        elif r < 0.4:

            new_seq = _random_perturb(curr_seq, magnitude=0.05)
        elif r < 0.6:

            other = _random_restart(len(curr_seq))
            t = np.random.rand()
            new_seq = [(1.0 - t) * x + t * y for x, y in zip(curr_seq, other)]
        else:

            new_seq = curr_seq.copy()

        # ----- evaluate ------------------------------------------------
        try:

        except Exception:

            continue

        if new_score < curr_score:
            curr_seq = new_seq
            curr_score = new_score
            no_improve = 0
        else:

            prob = np.exp(-(new_score - curr_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                curr_seq = new_seq
                curr_score = new_score
                no_improve += 1
            else:
                no_improve += 1

        # ----- keep the globally best sequence -------------------------
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"[iter {iteration}] new global best = {best_score:.6f}")

        # ------------------------------------------------------------------

            n = len(curr_seq)
            sum_seq = float(np.sum(curr_seq))
            if sum_seq > 0.0:
                # normalise to the same ℓ¹‑norm used by the reference code

                    if sum_g > 0.0:

                        # try a few mixing ratios

                            try:

                                    best_seq = mixed.copy()
                                    curr_seq = mixed
                                    curr_score = sc_mix
                                    print(f"[LP mix t={t}] improved to {best_score:.6f}")
                            except Exception:
                                continue

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        if no_improve > 5000:

            try:

            no_improve = 0

    print(f"[done] best score = {best_score:.6f} after {iteration} iterations")
    return best_seq
