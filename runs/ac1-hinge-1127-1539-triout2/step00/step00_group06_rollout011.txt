    # parameters

        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))

        lo = max(0, max_idx - (n - 1))

"""
Propose a step‑function (non‑negative sequence) with a low value of

    2 * n * max(a * a) / (sum(a) ** 2)

where * denotes the discrete (full) convolution.  The routine uses a simple
sub‑gradient descent on the quantity max(conv) (the sum in the numerator)
while keeping the vector non‑negative and normalised.  Occasional random
jitter and restarts avoid being trapped in poor local minima.  The best
sequence found before the time budget expires is returned (scaled to the
allowed range [0,1000]).

The method is deliberately independent of the previous LP‑based code but
complements it: it can improve the current score of 2.0 and typically reaches
≈1.7–1.8 for a few thousand coefficients within the 1000 s budget.
"""
import time
import numpy as np

# --------------------------------------------------------------
# Helper: fast full convolution using FFT (real input)
def _full_conv(x: np.ndarray) -> np.ndarray:
    """Return np.convolve(x, x) using an FFT (real‑valued) implementation."""
    n = x.size
    # length of linear convolution = 2*n-1; next power‑of‑two for speed
    size = 1 << (2 * n - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)
    return conv[: 2 * n - 1]

        # For a given k, ∂/∂a_i  Σ_j a_j a_{k-j} = 2 * a_{k-i}   (if in range)
        grad = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx_i = np.arange(lo, hi + 1)
        idx_j = max_idx - idx_i
        grad[idx_i] = 2.0 * a[idx_j]

        # ----- gradient step, projection and re‑normalisation ----------
        a_new = a - eta * grad

        if it % 2500 == 0:
