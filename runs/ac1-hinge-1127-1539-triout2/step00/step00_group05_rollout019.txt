    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def convolve(a,b):
    size = 1 << ((len(a)+len(b)-2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    c = np.fft.irfft(C, n=size)
    return c[:len(a)+len(b)-1]
```

def project_simplex(v):
    # Project onto simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def fft_convolve(a, b):
    size = 1 << ((len(a) + len(b) - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[:len(a)+len(b)-1]

def internal_score(x):
    n = len(x)
    conv = fft_convolve(x, x)
    return 2.0 * n * np.max(conv)

"""
Fast stochastic search for a step‑function that minimizes
evaluate_sequence(seq).  The objective is scale‑invariant, so we keep the
sequence on the probability simplex (non‑negative, sum = 1) during the
search.  The algorithm repeatedly:

1. Starts from a random simplex point.
2. Computes the self‑convolution (via FFT) and its maximum.
3. Moves a small amount of mass from the largest interior entry to one
   of the two edge positions (0 or n‑1).  This tends to flatten the
   convolution because the centre of the convolution receives fewer
   large contributions.
4. Projects the modified vector back onto the simplex.
5. Accepts the move only if it lowers the current score.
6. Repeats until the time budget is exhausted, keeping the best
   sequence ever seen.

The final vector is scaled so its largest entry is exactly 1000 –
the evaluation function clamps larger values, but scaling does not
change the objective because it is homogeneous of degree 0.
"""

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # condition u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fft_convolve(a, b):
    """Linear convolution using real FFT (fast for lengths up to a few thousand)."""
    size = 1 << ((len(a) + len(b) - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[:len(a) + len(b) - 1]


def _internal_score(x):
    """
    Fast internal version of evaluate_sequence for vectors that already sum to 1.
    Returns 2 * n * max(conv(x,x)).
    """
    n = len(x)
    conv = _fft_convolve(x, x)
    return 2.0 * n * np.max(conv)


def propose_candidate(seed=0, budget_s=1000, length=1500):
    """
    Search for a non‑negative step function of the given length that
    minimizes evaluate_sequence.  The routine runs for up to ``budget_s``
    seconds and returns the best sequence found (as a list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_score = float('inf')
    best_vec = None

    # ------------------------------------------------------------------
    # Main search loop – many random restarts, each followed by a short
    # directed hill‑climbing phase.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = length
        # start from a random point on the simplex
        x = np.random.rand(n)
        x = _project_simplex(x)

        # number of inner iterations – limited by remaining time
        max_iters = 4000
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # current objective
            cur_score = _internal_score(x)

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_vec = x.copy()
                print(f"[{time.time():.1f}] new best score {best_score:.6f}")

            # ---------------------------------------------------------
            #   Directed move: shift mass from the biggest interior entry
            #   toward one of the two edges.
            # ---------------------------------------------------------
            # interior indices (exclude the two extremes)
            interior = slice(1, n - 1)
            i = np.argmax(x[interior]) + 1          # index of largest interior entry
            if x[i] > 1e-12:                        # ensure we have something to move
                j = 0 if np.random.rand() < 0.5 else n - 1
                # move a modest fraction of the source mass
                eps = min(0.5 * x[i], 0.01 * np.random.rand())
                if eps > 0:
                    x_candidate = x.copy()
                    x_candidate[i] -= eps
                    x_candidate[j] += eps
                    x_candidate = _project_simplex(x_candidate)
                    new_score = _internal_score(x_candidate)
                    if new_score < cur_score:
                        x = x_candidate
                        continue   # successful move – go to next iteration

            # ---------------------------------------------------------
            #   Random perturbation (helps escaping plateaus)
            # ---------------------------------------------------------
            i, j = np.random.choice(n, 2, replace=False)
            if x[i] <= 1e-12:
                continue
            eps = min(0.5 * x[i], 0.005 * np.random.rand())
            x_candidate = x.copy()
            x_candidate[i] -= eps
            x_candidate[j] += eps
            x_candidate = _project_simplex(x_candidate)
            new_score = _internal_score(x_candidate)
            if new_score < cur_score:
                x = x_candidate

        # early exit if we have already beaten the target bound
        if best_score < 1.5:
            break

    # ------------------------------------------------------------------
    #  Produce the final sequence (scale so the largest entry is exactly 1000)
    # ------------------------------------------------------------------
    if best_vec is None:                     # safety fallback – uniform distribution
        best_vec = np.full(length, 1.0 / length)

    max_val = np.max(best_vec)
    scale = 1000.0 / max_val if max_val > 0 else 1.0
    final_seq = (best_vec * scale).tolist()

    # final check (optional – not required for the returned value)
    # from the surrounding infrastructure evaluate_sequence will be called later
    print(f"Finished search – best internal score {best_score:.6f}")
    return final_seq
