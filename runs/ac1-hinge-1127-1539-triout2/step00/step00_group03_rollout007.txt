    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()

    return c

"""
Propose a candidate step‑function (non‑negative heights) that improves the bound
on   2 * n * max(conv(f,f)) / (sum(f)^2).

The algorithm is a simple projected gradient descent on the equivalent
objective

    minimize   max_k (a * a)[k]    subject to  a_i ≥ 0 ,  Σ a_i = 1

where “*” is the linear convolution.  The objective is smooth‑approximated
by a soft‑max with a temperature parameter γ, whose gradient can be computed
by a single convolution (using the FFT).  After each gradient step the vector
is projected back onto the probability simplex (non‑negative and sum‑to‑1)
and forced to be symmetric (a_i = a_{n-1-i}) – this symmetry is natural for
the convolution and helps flatten the autocorrelation.

The initial shape is taken from the continuous function
    a(x) = 1 / (2√x)  on (0,1],
which gives a nearly constant self‑convolution (the “π/2” bound).  A
multiplicative Dirichlet perturbation and several random restarts let the
search escape local minima.  The routine runs until the time budget expires
(≈ 1000 s) and returns the best sequence seen.
"""

import numpy as np
import time
from typing import List

# ----------------------------------------------------------------------
# Helper: convolution using real FFT (full linear convolution)
def convolve_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of 1‑D real arrays x and y."""
    size = len(x) + len(y) - 1
    # size of FFT – next power of two for speed
    N = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, N)
    fy = np.fft.rfft(y, N)
    conv = np.fft.irfft(fx * fy, N)[:size]
    return conv


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    # Implementation from
    #   https://doi.org/10.1007/s10107-013-0700-8
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def gradient_step(a: np.ndarray,
                  gamma: float,
                  step: float) -> np.ndarray:
    """
    Perform one (projected) gradient step for the soft‑max approximation of
    max_k (a*a)[k].

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector (sum = 1, non‑negative, symmetric).
    gamma : float
        Soft‑max temperature – larger values make the approximation closer to
        the true max.
    step : float
        Learning rate (additive).

    Returns
    -------
    np.ndarray
        Updated vector (still non‑negative, not yet normalised).
    """
    n = len(a)

    # 1) self‑convolution
    c = convolve_full(a, a)                     # length 2n‑1

    # 2) soft‑max weights – stable formulation
    maxc = c.max()
    exp_weights = np.exp(gamma * (c - maxc))
    w = exp_weights / exp_weights.sum()         # length 2n‑1, sums to 1

    # 3) gradient of the soft‑max (g_i = 2 * Σ_j w_{i+j} a_j )
    #    This is a convolution of the reversed a with w.
    conv = convolve_full(a[::-1], w)            # length 3n‑2
    grad = 2.0 * conv[n - 1:n - 1 + n]          # shape (n,)

    # 4) gradient descent step (additive)
    a_new = a - step * grad

    # 5) enforce non‑negativity
    a_new = np.maximum(a_new, 0.0)

    # 6) enforce symmetry (optional but very helpful)
    a_new = (a_new + a_new[::-1]) * 0.5

    # 7) project back onto the simplex Σa = 1
    a_new = project_simplex(a_new)

    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a good sequence of non‑negative heights.

    The routine runs several random restarts, each performing a projected
    gradient descent with a slowly increasing soft‑max temperature.
    The best sequence (lowest value of ``evaluate_sequence``) encountered is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # Helper to evaluate and keep the best found
    def try_update(seq: np.ndarray):
        nonlocal best_seq, best_score
        # evaluate_sequence works with any mutable sequence; ensure Python list
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New best score: {best_score:.6f} (len={len(seq)})")

    # ------------------------------------------------------------------
    # Main optimisation loop – run as many restarts as time permits
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Choose a length.  500–2000 works well: larger n lowers the
        # average but does not change the limiting factor (the “π/2” bound).
        n = np.random.randint(500, 2000)

        # --------------------------------------------------------------
        # Initialise a with the 1/√x shape (symmetric, already close to the
        # continuous optimum) plus a small Dirichlet perturbation.
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt(i + 1) + 1.0 / np.sqrt(n - i)
        a = a + 0.1 * np.random.dirichlet(alpha=np.ones(n))   # tiny random jitter
        a = project_simplex(a)        # ensure sum = 1
        a = (a + a[::-1]) * 0.5       # enforce symmetry

        # --------------------------------------------------------------

        step = 0.05                   # learning rate
        gamma = 2.0                   # initial soft‑max temperature
        max_iters = 3000

        try_update(a)

        # --------------------------------------------------------------
        for it in range(max_iters):
            if time.time() >= deadline:

            # one gradient descent step
            a = gradient_step(a, gamma, step)

            # keep track of the best sequence

                try_update(a)

            if it % 200 == 0 and it > 0:

                step *= 0.85                     # smaller steps as we converge

        # End of one restart – one more evaluation for safety
        try_update(a)

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen) return a tiny uniform seq
    if best_seq is None:
        best_seq = np.ones(10) / 10.0

    # Convert to plain Python list before returning
    return best_seq.tolist()
