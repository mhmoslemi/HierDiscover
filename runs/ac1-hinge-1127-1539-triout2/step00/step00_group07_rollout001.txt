def simplex_projection(v, s):
    # v list of floats
    n = len(v)
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a sequence of non‑negative heights that makes the self‑convolution as flat
as possible.  The continuous arcsine distribution satisfies

    (arcsine) * (arcsine) = uniform  on  [0,2]

so a discrete approximation of the arcsine density yields a convolution that is
nearly constant, i.e. max(b) ≈ sum(a)^2 / (2 n – 1).  Since the evaluation
function

    2 * n * max(b) / (sum(a)^2)

is invariant under scaling of the whole sequence, any (positive) discretisation
of the arcsine law is a near‑optimal candidate.  The algorithm therefore:

1.   Builds several arcsine‑shaped sequences of different lengths (the longer the
     flatter the convolution, but convolution cost grows quadratically).
2.   Keeps the best one seen.
3.   (Optional) performs a few cheap projected‑gradient steps that try to reduce
     the current maximal convolution entry while preserving the ℓ₁‑norm.

All operations stay well within the 1000 s time budget and use only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def generate_arcsine_sequence(n: int) -> np.ndarray:
    """
    Return a length‑n array whose entries follow the (unnormalised) arcsine density
    1 / sqrt(x (1‑x)) on the interval (0,1).  The points are taken at the mid‑points
    x_i = (i + 0.5) / n, guaranteeing x_i∈(0,1) and thus finite values.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # arcsine density; the values are finite and O(sqrt(n)) at the ends.
    a = 1.0 / np.sqrt(x * (1.0 - x))
    return a


def project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector `v` onto the simplex { w | w_i >= 0, sum w_i = s }.
    Implements the algorithm from
    Wang & Carreira‑Perpiñán, "Projection onto the probability simplex: An efficient
    algorithm with a simple proof, and an application", 2013.
    """
    if s < 0:
        raise ValueError("target sum must be non‑negative")

    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find rho = max {j | u_j - cssv_j / j > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def flatten_step(a: np.ndarray, lr: float, target_sum: float) -> np.ndarray:
    """
    One iteration of a simple projected‑gradient attempt to reduce the maximal entry
    of the self‑convolution.  `lr` is a short learning rate.
    """
    n = a.shape[0]
    # convolution a * a
    conv = np.convolve(a, a)
    # index of the current maximum
    k = int(np.argmax(conv))

    # gradient of b_k = (a * a)_k w.r.t. a
    grad = np.zeros_like(a)
    # first contribution: a_{k-i}
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] += a[j[mask]]
    # second contribution: a_{i-k}
    j2 = i - k
    mask2 = (j2 >= 0) & (j2 < n)
    grad[mask2] += a[j2[mask2]]

    # gradient descent step (note that we minimize max, so we move opposite to gradient)
    a_new = a - lr * grad
    # keep non‑negative before projection
    a_new = np.clip(a_new, 0.0, None)
    # project back onto the simplex to keep sum constant
    a_new = project_to_simplex(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Generate a candidate step‑function (list of non‑negative floats) that minimizes
    `evaluate_sequence`.  The method builds arcsine‑shaped sequences of several
    lengths, picks the best one, then performs a few cheap projected‑gradient
    refinements.

    Parameters
    ----------
    seed : int
        Random seed for any stochastic perturbations.
    budget_s : float
        Total time budget (seconds).  The function returns before this deadline.
    **kwargs : dict
        Ignored – present for compatibility with the surrounding harness.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    # small safety margin before the hard deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Try a family of arcsine sequences of increasing length.
    # ------------------------------------------------------------------
    # We start from modest sizes and grow; longer sequences give a theoretical
    # bound closer to 1.0 but also cost more time for the convolution.
    lengths = [500, 1000, 2000, 4000, 6000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break
        a = generate_arcsine_sequence(n)
        # Convert to plain Python list before evaluation (evaluate_sequence expects list)
        seq = a.tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New best after arcsine (n={n}): {score:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional local refinement: projected gradient to flatten the peak.
    # ------------------------------------------------------------------
    if best_seq is not None:
        # Use the sum of the current best as the simplex target (scale‑invariant)
        target_sum = float(np.sum(best_seq))
        # Learning rate – chosen adaptively based on current magnitude.
        # We start with a modest step and shrink if progress stalls.
        lr = 0.05 * target_sum / max(1.0, np.max(best_seq))
        iteration = 0
        while time.time() < deadline and iteration < 500:
            iteration += 1
            a_new = flatten_step(best_seq, lr, target_sum)
            seq_new = a_new.tolist()
            score_new = evaluate_sequence(seq_new)
            # Accept if we strictly improve; otherwise keep the old sequence.
            if score_new < best_score:
                best_score = score_new
                best_seq = a_new.copy()
                print(f"[{time.time():.2f}] Improved by gradient (iter={iteration}): {score_new:.6f}")
                # If improvement is good we can afford a slightly larger step
                lr *= 1.05
            else:
                # No improvement – reduce step size to stay stable
                lr *= 0.7

    # ------------------------------------------------------------------
    # 3) Fallback: if for any reason we have no best sequence yet, use a trivial
    #    constant sequence (which yields exactly 2.0).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(100, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback to constant sequence, score =", best_score)

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
