def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    n = 2500   # choose length moderate
    seq = np.sqrt(np.arange(1, n+1)) - np.sqrt(np.arange(0, n))
    seq = seq * np.sqrt(2 * n) / seq.sum()  # normalized sum sqrt(2n); optional
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial eval {best_score}')

    # parameters for SA
    temperature = 1.0
    decay = 0.9995

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose delta proportion of min of seq[i], seq[j]
        max_delta = min(best_seq[i], best_seq[j]) * 0.2
        if max_delta <= 0:
            continue
        delta = (np.random.rand() - 0.5) * 2 * max_delta   # in [-max_delta, max_delta]

        new_seq = best_seq.copy()
        new_seq[i] += delta
        new_seq[j] -= delta
        # ensure non-negative
        if new_seq[i] < 0 or new_seq[j] < 0:
            continue

        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        # metropolis criterion
        if score < best_score or np.random.rand() < np.exp((best_score - score) / temperature):
            best_seq = new_seq
            best_score = score
            # optional print
            print(f'Iter new best {best_score:.6f}')
        temperature *= decay
    return best_seq.tolist()

"""
Propose a candidate step‑function that improves on the trivial bound (value 2).

The algorithm starts from the analytically motivated sequence
    a_i = √(i+1) – √i      (i = 0 … n‑1)

which corresponds (after normalisation) to the continuous function
f(x)=c/√x whose self‑convolution is (asymptotically) constant.  Its
evaluation is ≈ π/2 ≈ 1.57, already better than 2.

To push the value further we perform a cheap simulated‑annealing /
gradient‑like local search that repeatedly:
* finds the shift k where the current convolution attains its maximum,
  i.e. the “worst” sum,
* identifies the pair of indices (i,j) that contributes most to this
  maximum (largest product a_i·a_j with i+j=k),
* identifies a pair (p,q) that contributes the least to the
  convolution (smallest product, often zero),
* transfers a small amount of mass from (i,j) to (p,q) while keeping the
  total sum unchanged and all entries non‑negative.

The move is accepted if it improves the score, otherwise it is kept
with a tiny probability to avoid getting stuck.  Occasionally a random
jitter is applied and, after many non‑improving steps, the algorithm
re‑initialises from the analytic starting point.

The whole search runs for the allotted time (≈ 1000 s) and returns the
best sequence encountered.  All operations are O(n log n) thanks to an
FFT‑based convolution, so the method works comfortably for n≈2500–3000.
"""

import numpy as np
import time

# ------------------------------------------------------------
# Helper: fast max‑convolution using FFT (returns max, argmax, full array)
def _max_convolution(seq: np.ndarray):
    """Return (max value, index of max, full convolution array) for seq*seq."""
    n = seq.size
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << (conv_len - 1).bit_length()
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)[:conv_len]
    # numerical noise may produce tiny negative values – clip safely
    conv = np.maximum(conv, 0.0)
    idx_max = int(np.argmax(conv))
    return float(conv[idx_max]), idx_max, conv


# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: float = 950.0,
                      n: int = 2500,
                      step_factor: float = 0.05) -> list[float]:
    """
    Search for a non‑negative sequence that minimises
    evaluate_sequence(seq).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Seconds allocated to the search (a few seconds are kept as a safety margin).
    n : int
        Length of the sequence.  2500‑3000 is a good compromise.
    step_factor : float
        Base proportion of mass moved in each local‑optimisation step.

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 5.0)  # keep a small margin

    # --------------------------------------------------------
    # Analytic starting point: a_i = sqrt(i+1) - sqrt(i)
    base = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(np.arange(0, n, dtype=np.float64))
    # Normalise to the conventional sum √(2 n) – not required for the score,
    # but keeps the numbers well‑scaled.
    base *= (np.sqrt(2.0 * n) / base.sum())

    curr_seq = base.copy()
    best_seq = curr_seq.copy()
    best_score = float('inf')
    # compute the first score
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    print(f'Initial score (analytic start): {best_score:.6f}')

    # --------------------------------------------------------
    # Main optimisation loop
    no_improve = 0          # steps without improvement
    iter_cnt = 0
    curr_score = best_score

    while time.time() < deadline:
        iter_cnt += 1

        # ---- convolution information -------------------------------------------------
        max_val, k_max, conv = _max_convolution(curr_seq)
        # also look at the smallest (often zero) entry – useful as a sink for mass
        k_min = int(np.argmin(conv))

        # ---- most harmful pair (i_max, j_max) for the current peak --------------------
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k_max - i_range
        products = curr_seq[i_range] * curr_seq[j_range]
        idx_peak = int(np.argmax(products))
        i_max = int(i_range[idx_peak])
        j_max = int(j_range[idx_peak])

        # ---- weakest pair (i_min, j_min) – typically a zero contribution -------------
        i_start2 = max(0, k_min - (n - 1))
        i_end2 = min(n - 1, k_min)
        i_range2 = np.arange(i_start2, i_end2 + 1, dtype=np.int64)
        j_range2 = k_min - i_range2
        products2 = curr_seq[i_range2] * curr_seq[j_range2]
        idx_weak = int(np.argmin(products2))
        i_min = int(i_range2[idx_weak])
        j_min = int(j_range2[idx_weak])

        # ---- move a small chunk of mass from the worst pair to the weakest pair ------
        eps = min(curr_seq[i_max], curr_seq[j_max]) * step_factor
        if eps > 1e-12:
            new_seq = curr_seq.copy()
            new_seq[i_max] -= eps
            new_seq[j_max] -= eps
            new_seq[i_min] += eps
            new_seq[j_min] += eps

            # Guard against numerical negatives (might appear for the diagonal case)
            if new_seq[i_max] < 0 or new_seq[j_max] < 0:
                # fallback: small random perturbation instead of the structured move
                new_seq = curr_seq.copy()
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.01 * curr_seq[idx]
                new_seq[idx] = max(0.0, new_seq[idx] + delta)
                # keep total sum unchanged by adjusting another random entry
                idx2 = (idx + np.random.randint(1, n)) % n
                new_seq[idx2] = max(0.0, new_seq[idx2] - delta)

            # ---- evaluate -----------------------------------------------------------
            try:
                new_score = evaluate_sequence(new_seq.tolist())
            except Exception:
                new_score = float('inf')

            # ---- acceptance criterion ------------------------------------------------
            # deterministic improvement or tiny Metropolis walk
            accept = False
            if new_score < curr_score:
                accept = True
            elif np.random.rand() < 0.001:   # rare uphill moves
                accept = True

            if accept:
                curr_seq = new_seq
                curr_score = new_score

                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    no_improve = 0
                    print(f'Iter {iter_cnt:6d}  NEW BEST  score={best_score:.6f}')
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            # ---- if eps is too tiny (unlikely) perform a tiny random jitter -----------
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.001 * curr_seq[idx]
            new_seq = curr_seq.copy()
            new_seq[idx] = max(0.0, new_seq[idx] + delta)
            # adjust another entry to keep the sum essentially unchanged
            idx2 = (idx + np.random.randint(1, n)) % n
            new_seq[idx2] = max(0.0, new_seq[idx2] - delta)

            try:
                new_score = evaluate_sequence(new_seq.tolist())
            except Exception:
                new_score = float('inf')

            if new_score < curr_score or np.random.rand() < 0.001:
                curr_seq = new_seq
                curr_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    no_improve = 0
                    print(f'Iter {iter_cnt:6d}  NEW BEST (jitter)  score={best_score:.6f}')
                else:
                    no_improve += 1
            else:
                no_improve += 1

        # ---- occasional restart if we seem stuck ------------------------------------
        if no_improve > 5000:
            curr_seq = base.copy()
            curr_score = best_score
            no_improve = 0
            print(f'Restarted after {iter_cnt} iterations – current best {best_score:.6f}')

    print(f'Finished after {iter_cnt} iterations. Best score: {best_score:.6f}')
    return best_seq.tolist()
