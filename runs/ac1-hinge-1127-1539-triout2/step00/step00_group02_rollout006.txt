def solve_lp(f_seq, rhs):

    n = len(f_seq)

    # Variables: b0...bn-1

    # Objective: minimize -sum(b) => maximize sum(b)
    c = -np.ones(n)

    # Build A_ub: for each k in 0..2n-2, we have sum_i f_i * b_{k - i} <= rhs, with b_j >=0.

    # Represent as matrix: rows (2n-1) x n

    # for each k, row_j = f_{k-j} if 0 <= k - j < n else 0

    # Build using loops.

    a_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # i from max(0,k-(n-1)) to min(n-1,k)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # row_j = f_i where j = k - i.
        # For each i in range, set row[ k - i ] = f[i]
        # Equivalent to fill row indices j
        j_indices = np.arange(i_min, i_max+1)
        i_vals = j_indices
        row_idx = k - i_vals
        a_ub[k, row_idx] = f_seq[i_vals]
    # Nonnegativity constraints via bounds param: bounds=(0,None) for each variable

    result = linprog(c, A_ub=a_ub, b_ub=np.full(2*n-1, rhs), bounds=[(0, None)]*n, method='highs', options={'time_limit': 5.0})

    if result.success:
        return result.x
    else:
        return None
