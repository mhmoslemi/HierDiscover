"""
Propose a low‑value step‑function sequence.

The evaluation metric is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the full convolution of the sequence with itself.
For a sequence of the form

    a_i = (i + 1) ** p      (i = 0 … n‑1)

the convolution at the centre (the largest value) can be computed in closed
form and the metric simplifies to

    2 * (p + 1) ** 2 * B(p + 1, p + 1),

where `B` is the Beta‑function.
Choosing a modest power `p` (e.g. `p = 5`) makes this value
≈ 0.026 – far smaller than the target 1.503.

The sequence is scaled so that the greatest element equals the allowed
maximum 1000; scaling cancels out of the metric, so the analytical bound
remains unchanged.

The function tries a few powers and returns the sequence that gives the
lowest observed score.
"""

import numpy as np
import time

def _make_sequence(length: int, power: int) -> list[float]:
    """
    Return a sequence a_i = ((i+1) ** power) scaled so that a_{length-1}=1000.
    All entries are clipped to [0, 1000] to satisfy the security checks.
    """
    # scaling factor α so that the last element equals 1000
    alpha = 1000.0 / (length ** power)
    seq = [(alpha * ((i + 1) ** power)) for i in range(length)]
    # safeguard against any floating‑point overshoot
    seq = [min(1000.0, max(0.0, x)) for x in seq]
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a polynomially increasing step function.
    The routine evaluates a handful of powers (2 … 6) and returns the
    sequence with the smallest evaluation score.
    """
    np.random.seed(seed)                 # not strictly needed, but kept for reproducibility
    deadline = time.time() + budget_s - 2   # leave a tiny safety margin

    # Length of the sequence – any value ≤ ~8000 is fine.
    # Larger length gives a nicer “construction” but does not change the score.
    length = 2000

    best_seq = None
    best_score = float("inf")

    # Try a few different polynomial powers.
    for power in range(2, 7):            # 2,3,4,5,6
        seq = _make_sequence(length, power)

        # Compute the score (may raise if something unexpected happens)
        try:
            score = evaluate_sequence(seq)
        except Exception as e:
            print(f"Evaluation failed for power {power}: {e}")
            continue

        print(f"Power {power}: score = {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq

        # stop early if we run out of time
        if time.time() >= deadline:
            break

    # As a final safety‑net, if for some reason none of the candidates
    # succeeded, fall back to a simple uniform sequence (which yields score 2).
    if best_seq is None:
        print("All attempts failed – returning a uniform fallback.")
        best_seq = [1.0] * length
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback score = {best_score}")

    print(f"Chosen power yields score {best_score:.6f} with length {len(best_seq)}")
    return best_seq
