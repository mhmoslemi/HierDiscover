def solve_convolution_lp(f_sequence, rhs, time_limit=2.0):
    import numpy as np
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix, vstack, diags, lil_matrix

    n = len(f_sequence)
    rows = 2 * n - 1
    # Build convolution constraint matrix A_ub (rows x n)
    # For each k, row[k, j] = f[i] where i + j = k, i in [0, n-1], j in [0, n-1].
    # Equivalent to: row[k, j] = f_sequence[k - j] if 0 <= k - j < n else 0
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # iterate over j = i (i variable)?? Actually we need j such that i + j = k. For each i, j = k - i.
        # Let's set i iterates over i_min..i_max, then set j = k - i.
        # So for each i, we have entry at column j = k - i with value f[i].
        for i in range(i_min, i_max + 1):
            j = k - i
            row_ind.append(k)
            col_ind.append(j)
            data.append(f_sequence[i])
    # Build nonnegative constraints: -I * b <= 0
    # Append identity rows as additional constraints.
    total_rows = rows + n
    A_data = data + [-1.0] * n
    A_row = row_ind + list(range(rows, total_rows))
    A_col = col_ind + list(range(n))
    A_ub = csr_matrix((A_data, (A_row, A_col)), shape=(total_rows, n))
    b_ub = np.concatenate([np.full(rows, rhs), np.zeros(n)])
    c = -np.ones(n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None),
                     method='highs', options={'time_limit': time_limit})
    if result.success:
        return result.x
    return None

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, random
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    np.random.seed(seed)
    random.seed(seed)

    # parameters
    min_n = 200
    max_n = 500
    n = np.random.randint(min_n, max_n+1)

    # initial sequence: uniform
    target_sum = np.sqrt(2 * n)
    seq = np.full(n, target_sum / n, dtype=float)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Start n={n}, initial score {best_score:.6f}")

    deadline = time.time() + budget_s - 2.0

    # helper to compute max conv
    def max_convolution(a):
        return np.max(np.convolve(a, a))

    # helper to solve LP (sparse)
    def solve_lp(f, rhs, time_limit=1.0):
        n = len(f)
        rows = 2 * n - 1
        # Build convolution matrix A_ub with shape (rows, n)
        # Use lists for data
        data = []
        row_ind = []
        col_ind = []
        # for each k row
        for k in range(rows):
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # iterate i, compute column j = k - i
            for i in range(i_min, i_max + 1):
                j = k - i
                data.append(f[i])
                row_ind.append(k)
                col_ind.append(j)
        A_ub = csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
        b_ub = np.full(rows, rhs)
        c = -np.ones(n)  # maximize sum, so minimize negative sum
        # bounds enforce nonnegativity
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'time_limit': time_limit, 'presolve': True})
        except Exception:
            return None
        if res.success:
            return res.x
        return None

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # Normalise current sequence to target sum for LP
        sum_seq = seq.sum()
        if sum_seq == 0:
            # avoid division by zero, reinitialize uniformly
            seq = np.full(n, target_sum / n)
            sum_seq = seq.sum()
        f_norm = seq * (target_sum / sum_seq)

        rhs = max_convolution(f_norm)

        # solve LP

            # scale g to target sum
            g_sum = g.sum()
            if g_sum == 0:

                g_sum = g.sum()

                try:
                    score = evaluate_sequence(new_seq.tolist())
                except Exception:
                    continue
                if score < best_score - 1e-9:
                    best_score = score
                    best_seq = new_seq.copy()
                    seq = new_seq  # continue from improved

                    break
            else:

                new_seq = seq.copy()
                new_seq[idx] = max(0.0, new_seq[idx] + delta)

            # optional clip

"""
propose_candidate
-----------------

A self‑contained stochastic optimiser that tries to push the value of
``evaluate_sequence`` below the current best bound of 2.0.

The algorithm works on a moderate sized step function (n≈200–500) so that
the linear program used in the state‑of‑the‑art method stays small enough
to be solved many times within the time budget.

Main ideas
~~~~~~~~~~
*   Keep the sum of the heights equal to ``sqrt(2·n)`` – after this
    normalisation the evaluation reduces to the maximal entry of the
    self‑convolution.
*   Given a current sequence *f*, solve a linear program that finds a
    non‑negative sequence *g* with a **larger** ℓ₁‑norm while guaranteeing
    ``conv(f,g) ≤ M`` where ``M = max(conv(f,f))``.
    (The LP is built as a sparse matrix to keep memory low.)
*   Scale *g* back to the required sum and form a convex combination
    ``h = (1‑t)·f + t·g``.  For any ``t∈(0,1]`` the new sequence has the same
    sum as *f* but a strictly smaller maximal convolution, therefore a
    smaller evaluation.
*   If the LP fails or no improvement is obtained, fall back to a tiny
    random perturbation (simulated‑annealing style) to escape possible
    stagnation.

The loop runs until the supplied ``budget_s`` seconds are exhausted (a
few seconds are kept as safety margin).  The best sequence seen so far is
returned as a plain Python list – ready for the external grader.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _max_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a * a)[k] using numpy.convolve."""
    return float(np.max(np.convolve(a, a)))


def _solve_lp(f: np.ndarray, rhs: float, time_limit: float = 0.5):
    """
    Solve
        max   sum b_j
        s.t.  conv(f, b) ≤ rhs   (for every shift)
              b_j ≥ 0

    The constraints are linear in b, therefore a standard LP.
    ``f`` is the (already normalised) current sequence, ``rhs`` is the bound
    M = max(conv(f,f)).  ``time_limit`` limits the solver wall‑time
    (HiGHS respects this option).

    Returns the optimal b as a 1‑d NumPy array, or ``None`` on failure.
    """
    n = len(f)
    rows = 2 * n - 1                     # one row per convolution shift
    data = []
    row_ind = []
    col_ind = []

    # Build the (rows × n) matrix A such that (A @ b)[k] = (f * b)[k]
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i                     # column index that multiplies f[i]
            data.append(f[i])
            row_ind.append(k)
            col_ind.append(j)

    A_ub = csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    b_ub = np.full(rows, rhs, dtype=float)

    # maximise sum b ⇒ minimise -sum b
    c = -np.ones(n, dtype=float)
    bounds = [(0, None)] * n             # b_j ≥ 0

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_limit, "presolve": True},
        )
    except Exception:
        return None

    if res.success:
        return np.asarray(res.x, dtype=float)
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the provided ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    **kwargs
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Choose a modest problem size (small enough for the LP)
    # ------------------------------------------------------------------
    n_min, n_max = 200, 500
    n = int(np.random.randint(n_min, n_max + 1))

    # Desired ℓ₁‑norm for the normalised problem:  Σ a_i = sqrt(2·n)
    target_sum = float(np.sqrt(2.0 * n))

    # ------------------------------------------------------------------
    # 2. Initialise with a uniform distribution (value = target_sum / n)
    # ------------------------------------------------------------------
    seq = np.full(n, target_sum / n, dtype=float)

    # Keep the best solution seen so far
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[Init] n={n}, uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    end_time = time.time() + budget_s - 2.0   # keep a small safety margin
    iteration = 0

    while time.time() < end_time:
        iteration += 1

        # Normalise current sequence for the LP (its ℓ₁‑norm becomes target_sum)
        cur_sum = seq.sum()
        if cur_sum == 0.0:
            seq = np.full(n, target_sum / n, dtype=float)
            cur_sum = seq.sum()
        f_norm = seq * (target_sum / cur_sum)

        # Current bound M = max(conv(f_norm, f_norm))
        M = _max_convolution(f_norm)

        # --------------------------------------------------------------
        # 4. Solve the LP: maximise sum(b) under conv(f_norm, b) ≤ M
        # --------------------------------------------------------------
        g_raw = _solve_lp(f_norm, M, time_limit=0.5)

        if g_raw is not None:
            # Scale g so that it also has ℓ₁‑norm = target_sum
            g_sum = g_raw.sum()
            if g_sum == 0.0:
                # fallback to a tiny uniform perturbation
                g = np.full(n, target_sum / n, dtype=float)
            else:
                g = g_raw * (target_sum / g_sum)

            # Try a few mixing factors t ∈ (0,1] and keep the best improvement
            improved = False
            for t in (0.01, 0.05, 0.10, 0.20, 0.50):
                new_seq = (1.0 - t) * seq + t * g
                # Clamp to the admissible domain (non‑negative, ≤1000)
                new_seq = np.clip(new_seq, 0.0, 1000.0)

                try:
                    score = evaluate_sequence(new_seq.tolist())
                except Exception:
                    continue

                if score < best_score - 1e-12:
                    best_score = score
                    best_seq = new_seq.copy()
                    seq = new_seq                     # continue from the improved point
                    improved = True
                    print(
                        f"[Iter {iteration:4d}] t={t:4.2f} → improved score={best_score:.6f}"
                    )
                    break

            if not improved:
                # No mixing yielded improvement – perform a tiny random jitter
                idx = np.random.randint(n)
                delta = np.random.randn() * 0.01 * max(seq[idx], 1e-8)
                seq[idx] = max(0.0, seq[idx] + delta)
        else:
            # LP failed (e.g. time‑out).  Apply a small random perturbation.
            idx = np.random.randint(n)
            delta = np.random.randn() * 0.05 * (target_sum / n)
            seq[idx] = max(0.0, seq[idx] + delta)

        # Occasionally re‑normalise to avoid drift caused by jitter
        if iteration % 50 == 0:
            seq = np.clip(seq, 0.0, 1000.0)
            total = seq.sum()
            if total > 0.0:
                seq = seq * (target_sum / total)

    print(f"[Done] Iterations: {iteration}, best score: {best_score:.6f}")
    return best_seq.tolist()
