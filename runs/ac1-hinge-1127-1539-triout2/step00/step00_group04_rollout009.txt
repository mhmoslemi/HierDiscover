import numpy as np
import time

def fast_evaluate(seq):
    """Fast version of evaluate_sequence using FFT."""
    # clamp values like evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    n = seq.shape[0]
    # length for FFT (next power of two for speed)
    L = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return float(2 * n * max_b / (sum_a ** 2))

def greedy_positions(n, max_allowed, rng):
    """Return a list of indices such that each unordered pair (i,j) (including i=j)
    occupies a sum i+j at most ``max_allowed`` times."""
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    positions = []
    order = rng.permutation(n)
    for p in order:
        # check self sum
        if sum_counts[2 * p] >= max_allowed:
            continue
        ok = True
        for q in positions:
            s = p + q
            if sum_counts[s] >= max_allowed:
                ok = False
                break
        if ok:
            # update counts
            for q in positions:
                sum_counts[p + q] += 1
            sum_counts[2 * p] += 1
            positions.append(p)
    return positions

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sparse binary step function that yields a small
    evaluate_sequence() value.  The algorithm builds a set of indices
    with a bounded number of unordered pair‑sums (a Sidon‑like construction)
    using a fast greedy routine, evaluates the resulting sequence with a
    FFT‑based proxy, and keeps the best result found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # Determine length of the sequence.  If a previous construction is
    # available we keep its length, otherwise we fall back to a moderate size.
    try:
        n = len(height_sequence_1)   # noqa: F821  (global provided by the framework)
        base_seq = list(height_sequence_1)
    except Exception:
        n = 3000
        base_seq = [0.0] * n

    # initialise best with the provided construction (if any)
    best_seq = np.array(base_seq, dtype=float)
    # fast initial score – we will also keep the true score for final reporting
    best_score = fast_evaluate(best_seq)

    # Do a quick sanity call with the official evaluator to avoid any
    # unexpected difference later.
    try:
        from __main__ import evaluate_sequence  # It will be injected by the runner.
    except Exception:
        # evaluate_sequence is already in the global namespace.
        pass
    # keep a reference
    eval_func = evaluate_sequence  # noqa: F821

    # store best score according to the official evaluator as well
    true_best_score = eval_func(best_seq.tolist())

    iteration = 0

        max_allowed = rng.choice([2, 3])

import numpy as np
import time

# --------------------------------------------------------------------
# Helper: fast evaluation (identical to evaluate_sequence but using FFT)
# --------------------------------------------------------------------
def _fast_evaluate(seq: np.ndarray) -> float:
    """Return the value of evaluate_sequence for a 1‑D numpy array.

    The implementation mirrors evaluate_sequence but uses FFT for a
    fast convolution.  All values are clipped to the same range as the
    official evaluator.
    """
    seq = np.clip(seq, 0.0, 1000.0)          # same protection as the original
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = seq.shape[0]
    # length for the FFT – next power of two speeds things up
    L = int(2 ** np.ceil(np.log2(2 * n - 1)))
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return float(2 * n * max_b / (total ** 2))


# --------------------------------------------------------------------
# Greedy construction of a “Sidon‑like” set.
# --------------------------------------------------------------------
def _greedy_positions(n: int, max_per_sum: int, rng: np.random.Generator) -> list[int]:
    """
    Build a list of indices such that each unordered pair (i, j) (including i=j)
    contributes to the sum i + j at most ``max_per_sum`` times.
    The algorithm walks through a random permutation of {0,…,n‑1} and keeps
    an index whenever it does not violate the bound.
    """
    # count of unordered pairs that already use each possible sum
    sum_cnt = np.zeros(2 * n - 1, dtype=np.int16)
    positions = []

    order = rng.permutation(n)
    for p in order:
        # self‑pair (p,p) would increase sum 2p
        if sum_cnt[2 * p] >= max_per_sum:
            continue

        # check the other pairs (p, q) with q already selected
        ok = True
        for q in positions:
            s = p + q
            if sum_cnt[s] >= max_per_sum:
                ok = False
                break
        if not ok:
            continue

        # accept p – update the counters
        for q in positions:
            sum_cnt[p + q] += 1
        sum_cnt[2 * p] += 1
        positions.append(p)

    # a second pass often squeezes a few more indices in
    if len(positions) < n:
        remaining = np.setdiff1d(np.arange(n), positions, assume_unique=True)
        rng.shuffle(remaining)
        for p in remaining:
            if sum_cnt[2 * p] >= max_per_sum:
                continue
            ok = True
            for q in positions:
                s = p + q
                if sum_cnt[s] >= max_per_sum:
                    ok = False
                    break
            if ok:
                for q in positions:
                    sum_cnt[p + q] += 1
                sum_cnt[2 * p] += 1
                positions.append(p)

    return positions
