def gradient_step(seq, beta=20, lr=0.1):
    n = len(seq)
    conv = np.convolve(seq, seq)
    # compute smooth weights
    exp_vals = np.exp(beta * conv)
    w = exp_vals / np.sum(exp_vals)
    w_rev = w[::-1]
    # compute convolution for gradient
    L = 1 << ((n + len(w_rev) - 1).bit_length())
    a_fft = np.fft.rfft(seq, L)
    w_fft = np.fft.rfft(w_rev, L)
    grad_full = np.fft.irfft(a_fft * w_fft, L)[:len(w_rev)+n-1]
    # extract needed slice
    idx_start = len(w_rev) - 1
    gradient = 2 * grad_full[idx_start - np.arange(n)]
    # perform gradient descent (move opposite)
    new_seq = seq - lr * gradient
    new_seq[new_seq < 0] = 0
    # renormalize
    new_seq = new_seq * (math.sqrt(2 * n) / np.sum(new_seq))
    return new_seq

"""
Simple heuristic search for a step function achieving low
evaluate_sequence value.  The algorithm works in three stages:

1.  A quick parametric sweep over a few families of symmetric
    non‑negative profiles (inverse power law and sinusoidal power).
    The best profile is kept as a starting point.

2.  Projected gradient descent on a smooth approximation of the
    max‑convolution objective.  The gradient is computed by
    convolving the current sequence with a softmax weight derived
    from the convolution values.  After each step the sequence is
    projected back onto the simplex of non‑negative vectors with
    the required L1‑norm (the sum is fixed to sqrt(2·n)).  This
    drives the auto‑convolution towards a flat profile and
    systematically lowers the evaluation score.

3.  A few calls to the linear‑programming “direction” used in the
    reference implementation.  For each direction a line‑search on
    the mixing coefficient is performed and the best mixture is
    accepted.

The whole routine respects the 1000 s budget and returns the best
sequence found.
"""

import math
import time
import random
import numpy as np
from scipy.optimize import linprog

def solve_convolution_lp(f_sequence, rhs, time_limit=5.0):
    """Linear program:
    maximise    sum(b)
    subject to   conv(f_sequence, b) <= rhs   ,   b >= 0

    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_sequence)
    # Objective: maximise sum(b) -> minimise -1·b
    c = -np.ones(n)
    # Build the convolution constraints
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # fill row so that (f * b)_k = sum_i f_i * b_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints: -b <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.concatenate([b_ub, np.zeros(n)])
    # solve with a modest time limit so the overall search stays fast
    res = linprog(c, A_ub=a_ub, b_ub=b_ub,
                  method='highs',
                  options={'time_limit': time_limit, 'disp': False})
    if res.success:
        return res.x
    return None


def line_search_mix(f, g, evaluate, max_iters=10):
    """
    Find a mixing coefficient λ∈[0,1] that minimises evaluate((1‑λ)·f+λ·g).
    A small grid is examined first and then the interval around the
    best point is refined.
    """
    best_lambda = 0.0
    best_score = evaluate(f.tolist())
    # coarse grid
    for lam in np.linspace(0.0, 1.0, 11):
        cand = (1.0 - lam) * f + lam * g
        sc = evaluate(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_lambda = lam
    # refinement
    left = max(0.0, best_lambda - 0.1)
    right = min(1.0, best_lambda + 0.1)
    for _ in range(max_iters):
        mids = np.linspace(left, right, 11)
        for lam in mids:
            cand = (1.0 - lam) * f + lam * g
            sc = evaluate(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_lambda = lam
        # shrink interval around the best λ
        left = max(0.0, best_lambda - (right - left) / 4)
        right = min(1.0, best_lambda + (right - left) / 4)
    return best_lambda, best_score


def gradient_step(seq, beta, lr):
    """
    One iteration of projected gradient descent on the smooth
    approximation of max(conv(seq, seq)).
    """
    n = len(seq)
    # convolution of seq with itself (full)
    conv = np.convolve(seq, seq, mode='full')
    # soft‑max weights that approximate the argmax
    w = np.exp(beta * conv)
    w /= w.sum()
    # reversed weight vector for the gradient expression
    w_rev = w[::-1]

    # length needed for a linear convolution of seq (len=n) with w_rev (len=2n-1)
    L = 1 << ((n + len(w_rev) - 1).bit_length())
    a_fft = np.fft.rfft(seq, L)
    w_fft = np.fft.rfft(w_rev, L)
    full = np.fft.irfft(a_fft * w_fft, L)[: n + len(w_rev) - 1]   # length 3n‑2

    # index that yields gradient component i
    idx_start = len(w_rev) - 1          # = 2n‑2
    grads = 2.0 * full[idx_start - np.arange(n)]

    # gradient descent step
    new_seq = seq - lr * grads
    new_seq = np.maximum(new_seq, 0.0)

    # projection onto the L1‑sphere with the required sum
    target_sum = math.sqrt(2.0 * n)
    s = new_seq.sum()
    if s > 0:
        new_seq *= (target_sum / s)
    else:
        # fall back to a uniform vector if everything vanished
        new_seq = np.full(n, target_sum / n, dtype=float)

    return new_seq


def parametric_initial(n, target_sum):
    """
    Scan a few simple parametric families and return the best
    sequence found together with its score.
    """
    best_seq = None
    best_score = float('inf')

    # family 1: inverse power law on distance to the nearest border
    for p in np.linspace(0.0, 2.0, 41):               # 0 … 2 in steps of 0.05
        i = np.arange(n, dtype=float)
        vals = (i + 1) ** (-p) + (n - i) ** (-p)
        # avoid the all‑zero case when p is very large
        if vals.sum() == 0.0:
            continue
        vals = vals * (target_sum / vals.sum())
        sc = evaluate_sequence(vals.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = vals.copy()

    # family 2: sinusoidal power (zero at the borders)
    for p in np.linspace(0.5, 4.0, 36):               # 0.5 … 4
        i = np.arange(n, dtype=float)
        vals = np.sin(np.pi * (i + 1) / (n + 1)) ** p
        if vals.sum() == 0.0:
            continue
        vals = vals * (target_sum / vals.sum())
        sc = evaluate_sequence(vals.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = vals.copy()

    # if nothing was produced (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())

    return best_seq, best_score

"""
Simple heuristic search for a step function achieving a low
evaluate_sequence value.  The algorithm works in three stages:

1.  A quick parametric sweep over a few families of symmetric
    non‑negative profiles (inverse power law and sinusoidal power).
    The best profile is kept as a starting point.

2.  Projected gradient descent on a smooth approximation of the
    max‑convolution objective.  The gradient is computed by
    convolving the current sequence with a softmax weight derived
    from the convolution values.  After each step the sequence is
    projected back onto the simplex of non‑negative vectors with
    the required L1‑norm (the sum is fixed to sqrt(2·n)).  This
    drives the auto‑convolution towards a flat profile and
    systematically lowers the evaluation score.

3.  A few calls to the linear‑programming “direction” used in the
    reference implementation.  For each direction a line‑search on
    the mixing coefficient is performed and the best mixture is
    accepted.

The routine respects the 1000 s budget and returns the best
sequence found.
"""

import math
import time
import random
import numpy as np
from scipy.optimize import linprog

def solve_convolution_lp(f_sequence, rhs, time_limit=5.0):
    """Linear program:
    maximise    sum(b)
    subject to   conv(f_sequence, b) <= rhs   ,   b >= 0

    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_sequence)
    # objective: minimise -sum(b)  (equivalent to maximise sum(b))
    c = -np.ones(n)
    # build convolution constraints
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints: -b <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.concatenate([b_ub, np.zeros(n)])
    # solve with a modest time limit
    res = linprog(c, A_ub=a_ub, b_ub=b_ub,
                  method='highs',
                  options={'time_limit': time_limit, 'disp': False})
    if res.success:
        return res.x
    return None


def line_search_mix(f, g, evaluate, max_iters=10):
    """
    Find a mixing coefficient λ∈[0,1] that minimises evaluate((1‑λ)·f+λ·g).
    A coarse grid is examined first and then the interval around the
    best point is refined.
    """
    best_lambda = 0.0
    best_score = evaluate(f.tolist())
    # coarse grid
    for lam in np.linspace(0.0, 1.0, 11):
        cand = (1.0 - lam) * f + lam * g
        sc = evaluate(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_lambda = lam
    # refinement
    left = max(0.0, best_lambda - 0.1)
    right = min(1.0, best_lambda + 0.1)
    for _ in range(max_iters):
        mids = np.linspace(left, right, 11)
        for lam in mids:
            cand = (1.0 - lam) * f + lam * g
            sc = evaluate(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_lambda = lam
        # shrink around the current best λ
        width = (right - left) / 4.0
        left = max(0.0, best_lambda - width)
        right = min(1.0, best_lambda + width)
    return best_lambda, best_score


def gradient_step(seq, beta, lr):
    """
    One iteration of projected gradient descent on a smooth version of
    max(conv(seq, seq)).
    """
    n = len(seq)
    # full convolution of seq with itself
    conv = np.convolve(seq, seq, mode='full')
    # soft‑max weights approximating the arg‑max
    w = np.exp(beta * conv)
    w /= w.sum()
    # reversed weight vector for the gradient expression
    w_rev = w[::-1]

    # length required for a linear convolution of seq (len=n) with w_rev (len=2n‑1)
    L = 1 << ((n + len(w_rev) - 1).bit_length())
    a_fft = np.fft.rfft(seq, L)
    w_fft = np.fft.rfft(w_rev, L)
    full = np.fft.irfft(a_fft * w_fft, L)[: n + len(w_rev) - 1]   # size = 3n‑2

    idx_start = len(w_rev) - 1          # = 2n‑2
    grads = 2.0 * full[idx_start - np.arange(n)]

    # gradient step
    new_seq = seq - lr * grads
    new_seq = np.maximum(new_seq, 0.0)

    # project back onto the L1‑sphere with the required sum
    target_sum = math.sqrt(2.0 * n)
    s = new_seq.sum()
    if s > 0:
        new_seq *= (target_sum / s)
    else:  # fall back to uniform if everything vanished
        new_seq = np.full(n, target_sum / n, dtype=float)

    return new_seq


def parametric_initial(n, target_sum):
    """

    best_seq = None

    for p in np.linspace(0.0, 2.0, 41):               # 0 … 2 in steps of 0.05
        i = np.arange(n, dtype=float)
        vals = (i + 1) ** (-p) + (n - i) ** (-p)
        if vals.sum() == 0.0:
