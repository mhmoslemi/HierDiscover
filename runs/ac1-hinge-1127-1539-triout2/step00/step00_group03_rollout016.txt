import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

def fast_convolve(x, y):
    """Return the linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    return np.fft.irfft(X * Y, size)[:n]

def solve_convolution_lp_active(f, M, active_idxs):
    """Solve max sum(b) s.t. conv(f,b) <= M for selected active indices.
       Returns b on success or None.
    """
    n = len(f)
    c = -np.ones(n)                     # maximise sum => minimise -sum
    rows = len(active_idxs)
    if rows == 0:
        return None
    # Build sparse constraint matrix; each row corresponds to one offset k
    data = []
    rows_idx = []
    cols_idx = []
    for r, k in enumerate(active_idxs):
        # indices j where both f[i] and b[j] appear in conv at offset k
        start_j = max(0, k - (n - 1))
        end_j = min(n - 1, k)
        # j runs from start_j to end_j inclusive
        j_vals = np.arange(start_j, end_j + 1, dtype=int)
        i_vals = k - j_vals                # i = k - j
        vals = f[i_vals]                   # coefficient = f_i
        data.extend(vals.tolist())
        rows_idx.extend([r] * len(j_vals))
        cols_idx.extend(j_vals.tolist())
    A_ub = sp.csr_matrix((data, (rows_idx, cols_idx)), shape=(rows, n))
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n                # b_i >= 0
    # Use the HiGHS interior point implementation (fast for sparse problems)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Randomised iterative improvement using a sparse LP direction.
    Starts from a random non‑negative step function, repeatedly
    (i) evaluates its self‑convolution,
    (ii) solves a small linear program that maximises the L1 norm of a
         partner sequence while keeping the mixed convolution below the
         current maximum,
    (iii) mixes the partner into the current sequence with a modest step
         size, re‑normalises, and keeps the best sequence seen.
    The loop runs until the allotted time expires and returns the
    best sequence (list of floats) discovered.
    """
    np.random.seed(seed)
    # Length of the sequence – prefer the length of the supplied state if present.
    try:
        n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    except Exception:
        n = 2000

    # target L1 norm (the algorithm keeps this constant)
    target_sum = np.sqrt(2.0 * n)

    # initialise with a random non‑negative vector
    f = np.random.rand(n)
    f = f / f.sum() * target_sum
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial random score: {best_score:.6f}")

    deadline = time.time() + max(0.0, budget_s - 0.5)
    iteration = 0
    # parameters controlling the search
    mix_factor = 0.15           # base mixing weight t
    active_frac = 0.02          # fraction of convolution entries kept as active constraints
    min_active = 20
    max_active = 80

    while time.time() < deadline:
        iteration += 1
        # 1. self‑convolution and current maximum
        conv_ff = fast_convolve(f, f)
        M = conv_ff.max()
        # current objective value equals M (since sum is fixed)
        cur_score = float(M)

        # pick a small set of offsets that are close to the current maximum.
        num_active = int(max(min_active, min(max_active, active_frac * len(conv_ff))))
        # indices of the largest entries

        # 3. solve the LP to get a direction g
        g = solve_convolution_lp_active(f, M, top_idxs)

            # renormalise to keep L1 norm constant
            f *= target_sum / f.sum()

        g_sum = g.sum()

"""
Iterative LP‑guided optimisation for the step‑function inequality.

The routine works with a fixed L¹ norm (√(2·n)).  Starting from a random
non‑negative vector it repeatedly:

1. Evaluates the self‑convolution and records the current maximum M.
2. Selects a small set of “active’’ convolution indices – the largest
   values of the current convolution – and solves a linear programme

       maximise   Σ b_j
       subject to conv(f, b) ≤ M  (only for the active indices)
                  b_j ≥ 0

   This LP finds a direction `g` that can carry a larger total mass while
   keeping the mixed convolution below the current peak.
3. Normalises `g` to the same L¹ norm, mixes it with the current vector
   (`f ← (1‑t)·f + t·g` with a modest step size `t`), and re‑normalises.
4. Keeps the best sequence seen (lowest peak of the self‑convolution).

Only a few active constraints are used, so the LP matrix stays small
(even for a few‑thousand‑element sequence) and the whole search finishes
well within the allocated time.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp


def fast_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (real‑valued, fast for long vectors)."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    return np.fft.irfft(X * Y, size)[:n]


def solve_convolution_lp_active(f: np.ndarray, M: float, active_idxs: np.ndarray):
    """
    Solve   maximise Σ b_j   subject to conv(f, b)[k] ≤ M   for k in active_idxs,
    and b_j ≥ 0.  Returns the optimal b as a dense array or None on failure.
    """
    n = len(f)
    if active_idxs.size == 0:
        return None

    # objective: minimise -Σ b_j  (i.e. maximise Σ b_j)
    c = -np.ones(n, dtype=np.float64)

    # build a sparse constraint matrix; each row corresponds to one offset k
    data = []
    rows = []
    cols = []
    for r, k in enumerate(active_idxs):
        # indices j for which f_i * b_j participates in conv at offset k
        start_j = max(0, k - (n - 1))
        end_j = min(n - 1, k)
        j_vals = np.arange(start_j, end_j + 1, dtype=int)
        i_vals = k - j_vals                     # i = k - j
        coeffs = f[i_vals]                      # coefficient = f_i
        data.extend(coeffs.tolist())
        rows.extend([r] * len(j_vals))
        cols.extend(j_vals.tolist())

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idxs), n))
    b_ub = np.full(len(active_idxs), M, dtype=np.float64)

    # non‑negativity bounds for all variables
    bounds = [(0.0, None)] * n

    # HiGHS works efficiently with sparse matrices
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None
