def evaluate(seq):
    n = len(seq)
    if n == 0:
        return np.inf
    a = np.asarray(seq, dtype=np.float64)
    # protect negative? ensure non-negative
    a = np.maximum(a, 0.0)
    # clipped: no large numbers
    a = np.clip(a, 0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    # compute convolution
    size = 1 << ((2*n-1).bit_length())
    fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft*fft, n=size)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a*sum_a)

"""
propose_candidate: a heuristic search for a non‑negative step function that
minimizes  2 * n * max(conv(a,a)) / (sum(a)**2).

The algorithm consists of three stages
1.  Build a family of analytically motivated shapes.
    The continuous function  f(x)=1/√x  (the “arcsine” density) gives a
    flat self‑convolution and leads to a score ≈π/2≈1.57.
    By adding a small offset δ to the denominator (1/√(i+δ)) and by
    symmetrising the shape we obtain several candidates.
2.  Pick the best candidate among the parametrised family.
3.  Refine the chosen sequence with a fast simulated‑annealing / hill‑climbing
    routine.  Each step moves a small amount of mass from an index that
    contributes heavily to the current maximal convolution entry to a random
    other index, preserving the total sum.  The objective is evaluated using
    an FFT‑based convolution (O(n log n)) which is much faster than the
    provided evaluate_sequence.
The whole search respects the 1000 s time budget and returns the best
sequence found as a plain Python list of floats.
"""

import time
import numpy as np


def _conv_max(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using an FFT (linear convolution)."""
    n = seq.size
    # size = next power of two ≥ 2*n‑1 for linear convolution
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[: 2 * n - 1]
    # small negative values may appear because of round‑off
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Exact same objective as evaluate_sequence (without the security checks)."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    m = _conv_max(seq)
    return 2.0 * seq.size * m / (s * s)


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a little safety margin

    # ------------------------------------------------------------------
    # 1) generate analytic initial guesses (arcsine‑type families)
    # ------------------------------------------------------------------
    n = 3000                     # length of the sequence (adjustable)
    best_seq = None
    best_val = np.inf

    # simple 1/√(i+δ) family
    for delta in np.linspace(0.0, 3.0, 31):
        idx = np.arange(1, n + 1) + delta          # avoid division by zero
        seq = 1.0 / np.sqrt(idx)
        seq = seq / seq.sum()                       # L1‑normalise (scale irrelevant)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[init] delta={delta:5.2f}  score={val:.6f}")

    # symmetrised version  f(i)=1/√i + 1/√(n‑i+1)
    for delta in np.linspace(0.0, 3.0, 31):
        idx = np.arange(1, n + 1) + delta
        base = 1.0 / np.sqrt(idx)
        seq = base + base[::-1]
        seq = seq / seq.sum()
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[init] sym‑delta={delta:5.2f}  score={val:.6f}")

    # ------------------------------------------------------------------
    # 2) local stochastic optimisation (hill‑climbing + simulated annealing)
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_val = best_val

    # hyper‑parameters (they decay slowly)
    step_factor = 0.07          # relative amount of mass moved each trial
    temperature = 0.1           # for occasional uphill moves
    decay = 0.9999

    # helper that performs a single trial move
    def trial_move(seq: np.ndarray) -> np.ndarray:
        """Create a new sequence by transferring a small amount of mass."""
        nloc = seq.size
        # 30 % of the time use a “targeted” move that looks at the current max‑peak
        if np.random.rand() < 0.3:
            # full convolution (needed only for the targeted move)
            size = 1 << ((2 * nloc - 1).bit_length())
            fft = np.fft.rfft(seq, n=size)
            full = np.fft.irfft(fft * fft, n=size)[: 2 * nloc - 1]
            kmax = int(np.argmax(full))

            i = np.arange(nloc)
            j = kmax - i
            mask = (j >= 0) & (j < nloc)
            # contribution of each index to the peak kmax
            contrib = np.zeros(nloc)
            contrib[mask] = seq[i[mask]] * seq[j[mask]]
            src = int(np.argmax(contrib))                 # index that hurts most
            trg = np.random.randint(nloc)                 # random destination
            while trg == src:
                trg = np.random.randint(nloc)

            delta = step_factor * seq[src] * np.random.rand()
            delta = min(delta, seq[src])                  # stay non‑negative
            new = seq.copy()
            new[src] -= delta
            new[trg] += delta
            return new
        else:
            # plain random transfer between two distinct indices
            i, j = np.random.randint(nloc, size=2)
            if i == j or seq[i] <= 0:
                return seq
            delta = step_factor * seq[i] * np.random.rand()
            delta = min(delta, seq[i])
            new = seq.copy()
            new[i] -= delta
            new[j] += delta
            return new

    # main optimisation loop
    while time.time() < deadline:

        cand_val = _score(cand)

        # accept if better, otherwise accept with a small probability
        if cand_val < cur_val or np.random.rand() < np.exp(

                )

        # annealing / step‑size decay
        temperature *= decay
        step_factor *= decay

    # ------------------------------------------------------------------
    # 3) return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
