```
def project_simplex(v):
    # from literature (Duchi et al.)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Simple gradient‐descent on a smoothed max‑convolution objective.

The evaluation of a sequence a (non‑negative, sum > 0) is

    E(a) = 2·n·max(conv(a,a)) / (sum(a))**2

Because the expression is homogeneous we normalize the sequence to a
probability vector p (sum(p)=1).  Then

    E = 2·n·max(conv(p,p))

so we only need to minimise the maximal entry of the self‑convolution.
We approximate the non‑smooth max by a soft‑max (log‑sum‑exp) with a
temperature α.  The gradient of the smoothed objective can be written
as a convolution, which we evaluate efficiently with FFTs.
After a gradient step we project back onto the probability simplex.
The algorithm runs for the whole time budget, keeps the best sequence
seen (according to the exact evaluation) and finally returns it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al. 2008)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the last index where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:         # all entries are zero or negative
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: convolution using real FFT (fast for many iterations)
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors via FFT (returns length len(a)+len(b)-1)."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Initial distribution: discrete arcsine (Beta(0.5,0.5)) – a good warm start
def _arcsine_distribution(n: int) -> np.ndarray:
    """Return a probability vector approximating the continuous arcsine law."""
    # centre the indices to avoid zero at the boundaries
    idx = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    w /= w.sum()
    return w

    probability vector and uses a smoothed max‑convolution objective
    together with a simplex projection.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Choose a working length.  Use the length of the provided SOTA

    except Exception:

    # a size around 2000 gives a good trade‑off between speed and quality
    n = min(base_len, 2000)
    if n < 100:                                 # sanity
        n = 2000

    # initialise with the arcsine law (good empirical start)
    p = _arcsine_distribution(n)

    # store the best solution seen so far
    conv = _conv_fft(p, p)
    best_val = 2.0 * n * conv.max()
    best_p = p.copy()

    # current state
    curr_p = p.copy()
    curr_val = best_val

    # hyper‑parameters (they adapt slowly during the run)
    eta0 = 0.6               # base step size for the gradient step
    alpha_start = 4.0        # initial soft‑max temperature
    alpha_end = 60.0         # final temperature
