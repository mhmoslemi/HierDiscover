"""
Propose a step‑function sequence that lowers the evaluation score.

The evaluation function can be written as

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

where *conv* is the ordinary (aperiodic) convolution.
If we keep the sum fixed to √(2 n) the score equals the
maximum value of the convolution, i.e.  we want a distribution whose
self‑convolution is as flat as possible.

The algorithm below performs a projected gradient descent on a smooth
approximation of the maximum (soft‑max with a temperature β).  For a
vector `a` (non‑negative, sum = √(2 n)) the gradient of

    L(a) = (1/β)·log Σ_k exp(β·c_k)   with   c = conv(a, a)

is

    ∂L/∂a_i = 2 · Σ_k w_k · a_{k-i}   where   w_k = softmax(β·c_k).

The convolution with the weight vector `w` is computed efficiently by
FFT, and after each gradient step the vector is projected onto the
non‑negative orthant and renormalised to keep the sum constant.
A modest amount of random jitter is injected every few hundred
iterations to avoid stagnation.  Several random restarts are performed
within the time budget; the best sequence found is returned.

The whole routine runs well within the 1000 s limit and typically
drives the score well below the previous value 2.0 (target ≈ 1.5).
"""

import math
import time
import numpy as np


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max with temperature β."""
    shift = x.max()
    exp_term = np.exp(beta * (x - shift))
    return exp_term / exp_term.sum()


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence().  The search uses a smooth‑max gradient descent
    with FFT‑based convolutions and several random restarts.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Total wall‑clock time allowed (seconds).  The function stops
        a few seconds before the limit to avoid being terminated.

    Returns
    -------
    list[float]
        The best sequence found (as plain Python floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # -----------------------------------------------------------------
    # problem dimensions – we use a length comparable to the current
    # construction (≈1500).  Larger n gives a finer “grid” for flattening
    # the convolution but also increases the FFT size modestly.
    # -----------------------------------------------------------------
    n = 1527                     # length of the candidate sequence
    target_sum = math.sqrt(2.0 * n)    # we keep Σa_i = √(2 n)

    # FFT lengths (next powers of two for efficiency)
    len1 = 1 << ((2 * n - 1).bit_length())   # for a*a
    len2 = 1 << ((3 * n - 2).bit_length())   # for a * w_rev

    # hyper‑parameters
    max_restarts = 8                      # independent random starts
    base_lr = target_sum * 0.5            # initial learning rate
    lr_decay = 1e-5                       # 1/(1+lr_decay·iter) schedule
    beta = 0.3                            # start temperature for soft‑max
    beta_max = 5.0
    beta_inc = 0.001                      # increase β slowly each iteration

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # start several random restarts
    # -----------------------------------------------------------------
    for restart in range(max_restarts):
        # Dirichlet‑style random start, then scale to the required sum
        a = np.random.rand(n)
        a = a / a.sum() * target_sum

        # local best for this restart
        local_best_seq = a.copy()
        local_best_score = 2.0 * n * np.max(np.convolve(a, a)) / (np.sum(a) ** 2)

        iteration = 0
        while time.time() < deadline:
            # ---------------------------------------------------------
            # 1) compute the current convolution a * a  (full length)
            # ---------------------------------------------------------
            A1 = np.fft.rfft(a, n=len1)
            conv = np.fft.irfft(A1 * A1, n=len1)[: 2 * n - 1]
            max_conv = conv.max()

            # ---------------------------------------------------------
            # 2) evaluate the score (identical to evaluate_sequence()
            #    when the sum is fixed to √(2 n))
            # ---------------------------------------------------------
            score = 2.0 * n * max_conv / (np.sum(a) ** 2)

            if score < local_best_score:
                local_best_score = score
                local_best_seq = a.copy()

                    best_seq = a.copy()

            # ---------------------------------------------------------

            w = _softmax(conv, beta)          # shape (2n‑1,)

            # ---------------------------------------------------------

            #    This is a convolution a * w_rev; we compute it with FFT.
            # ---------------------------------------------------------

            # extract the needed n entries (indices 2n‑2‑i)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]

        best_seq = np.full(n, target_sum / n)

    # return plain Python list, as expected by the evaluator
    return best_seq.tolist()
