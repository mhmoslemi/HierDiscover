Define function solve_convolution_lp(f_seq, bound):

```
n = len(f_seq)
c = -np.ones(n)  # maximize sum
A = []
b = []
# For each shift k: sum_i f_i * x_{k-i} <= bound
for k in range(2*n-1):
    row = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            row[j] = f_seq[i]
    A.append(row)
    b.append(bound)
# non-negativity: -I x <= 0
A = np.vstack([A, -np.eye(n)])
b = np.concatenate([b, np.zeros(n)])
res = linprog(c, A_ub=A, b_ub=b, method='highs', options={'time_limit':5})
if res.success:
    return res.x
else:
    return None
```

"""
Simulated annealing for the step‑function bound.

The evaluation of a sequence `a` is

    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

(the larger the denominator the better).  The objective does **not**
depend on a global scale, therefore we keep the L1‑norm of the
candidate fixed (here 1.0) and only move probability mass between
entries.  Starting from a random non‑negative vector we repeatedly
transfer a small amount of mass from one position to another.  The
new sequence is accepted if it improves the bound, otherwise it may
be accepted with a probability decreasing with the temperature
(Simulated Annealing).  Periodically the algorithm restarts with a
different length and a fresh random vector to avoid getting stuck.
The best sequence found before the time budget expires is returned.

The method is completely self‑contained, uses only ``numpy`` and the
already‑imported ``evaluate_sequence`` function and respects the
1000‑second time budget.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper – evaluate a NumPy array (the external evaluate_sequence expects a list)
def _eval(seq: np.ndarray) -> float:
    """Wrapper for evaluate_sequence that works with NumPy arrays."""
    return evaluate_sequence(seq.tolist())

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that yields a low
    value of ``evaluate_sequence``.  The function runs for at most
    ``budget_s`` seconds and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    **kwargs
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    # ------------------------------------------------------------------
    # initialisation
    random.seed(seed)
    np.random.seed(seed)

    # we keep the L1‑norm equal to 1.0 (any scale is equivalent)
    target_norm = 1.0

    # time control
    deadline = time.time() + budget_s - 0.5   # a little safety margin

    # bookkeeping ----------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    it_counter = 0

    # temperature schedule (linear cooling)
    T_start = 1.0       # initial temperature
    T_end   = 1e-4      # final temperature
    max_iters_for_cool = 300_000   # after this many iterations temperature ≈ T_end

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # (re)initialise a candidate when necessary
        if it_counter == 0 or it_counter % 30_000 == 0:
            # random length, moderately sized (80 … 180)
            n = random.randint(80, 180)
            # start from a random probability vector
            cur = np.abs(np.random.randn(n))
            cur /= cur.sum()                     # normalise to unit L1‑norm
            cur *= target_norm
            cur_score = _eval(cur)
            # keep the global best if this random start is already better
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[init] n={n}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # annealing temperature (linear in the iteration counter)
        T = T_start * max(0.0, 1.0 - it_counter / max_iters_for_cool)
        T = max(T, T_end)

        # pick two distinct indices and move a small amount of mass
        i, j = random.sample(range(cur.size), 2)

        # how much mass can we safely move from i to j ?
        # we take a fraction (up to 10 %) of the source entry
        max_transfer = 0.1 * cur[i]
        if max_transfer <= 0.0:
            it_counter += 1
            continue
        delta = random.uniform(0.0, max_transfer)   # amount moved from i → j

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # evaluate the modification
        new_score = _eval(new)

        # --------------------------------------------------------------
        # acceptance rule
        improve = new_score < cur_score
        if improve:
            accept = True
        else:
            # Metropolis criterion – downhill moves are accepted with a
            # probability that decays with the temperature
            prob = np.exp(-(new_score - cur_score) / T)
            accept = random.random() < prob

        if accept:
            cur = new
            cur_score = new_score

            # keep the overall best sequence
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()
                print(f"[iter {it_counter:>7}] new best = {best_score:.6f} (n={new.size})")

        it_counter += 1

    # ------------------------------------------------------------------
    # final report
    if best_seq is None:
        # very unlikely – fallback to a tiny uniform vector
        best_seq = np.ones(3) / 3.0
        best_score = _eval(best_seq)
        print("[fallback] using uniform three‑element sequence")

    print(f"Finished after {it_counter} iterations, best score = {best_score:.6f}, "
          f"length = {best_seq.size}")

    # return as a plain Python list (the harness expects a list)
    return best_seq.tolist()
