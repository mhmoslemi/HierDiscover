def improve(a, steps, epsilon):
    n = len(a)
    best = a.copy()
    best_score = evaluate_sequence(a)
    for _ in range(steps):
        conv = np.convolve(a, a)
        max_val = conv.max()
        k = conv.argmax()
        # determine indices contributing to k
        low = max(0, k-(n-1))
        high = min(n-1, k)
        # indices i in [low, high] contribute
        # Choose random i from this range weighted by a_i maybe
        i = np.random.randint(low, high+1)
        # choose j outside this range uniformly
        # outside range = [0, low) U (high, n)
        if low>0 and high < n-1:
            # choose side randomly
            side = np.random.choice(['left','right'])
            if side == 'left':
                j = np.random.randint(0, low)
            else:
                j = np.random.randint(high+1, n)
        elif low>0:
            j = np.random.randint(0, low)
        elif high < n-1:
            j = np.random.randint(high+1, n)
        else:
            # all indices in range (should not happen except n small)
            j = (i+1) % n

        # transfer epsilon proportion from i to j
        delta = epsilon * a[i]
        if delta > a[i]:
            delta = a[i]
        a[i] -= delta
        a[j] += delta

        # ensure nonneg, but a[i] might go negative if epsilon too big
        # compute new score
        score = evaluate_sequence(a)
        if score < best_score:
            best_score = score
            best = a.copy()
        else:
            # optionally revert if not improved (Simulated annealing)

"""
propose_candidate

Search for a step‑function (non‑negative sequence) whose evaluation value
2·n·max(a*a)/(sum(a)²) is as small as possible.

The main idea is to use a *Sidon* (B₂) set: a set of indices S ⊂ {0,…,n‑1}
such that all unordered sums i+j (i≤j, i,j∈S) are distinct.
With equal weights on the indices of S the self‑convolution has at most
two terms for any shift, so the maximal convolution value is
    max_b = 2·w²  with  w = √(2·n)/|S|
and the evaluation score equals max_b.  Choosing |S| large enough makes
the score drop below the target 1.503.

The algorithm works in two phases:

1. **Sidon construction** – repeatedly try random greedy constructions of
   a Sidon set until a set of size ≥ d_min is found, where
   d_min = ceil(sqrt(4·n/target)) guarantees a score < target.
   The resulting sequence has the non‑zero positions given by the set
   and all non‑zero entries equal to w.

2. **Local refinement** – starting from the best sequence so far, perform
   a fast hill‑climbing / simulated‑annealing style mass‑transfer move:
   move a small fraction of the weight from an index that contributes to
   the currently maximal convolution shift to an index outside that range.
   Accept only moves that improve the score (or occasionally worse moves
   with a decreasing temperature).

The whole search runs for the allotted time budget (≤ budget_s seconds)
and returns the best sequence discovered.
"""

import math
import random
import time
from typing import List, Optional

import numpy as np

# ----------------------------------------------------------------------
# Helper: build a Sidon (B2) set by a greedy random algorithm
# ----------------------------------------------------------------------
def _greedy_sidon_set(N: int, rnd: random.Random) -> List[int]:
    """Return a Sidon set S ⊆ {0,…,N‑1} built greedily in random order."""
    candidates = list(range(N))
    rnd.shuffle(candidates)
    S: List[int] = []
    used_sums = set()            # stores i+j for all i ≤ j already in S
    for x in candidates:
        # check for collision with existing sums
        clash = False
        for y in S:
            if x + y in used_sums:
                clash = True
                break
        if clash or (x + x) in used_sums:
            continue
        # add x to the set and record its new sums
        for y in S:
            used_sums.add(x + y)     # unordered sum (x,y) with x>y handled once
        used_sums.add(x + x)         # self‑sum
        S.append(x)
    return S

def _find_sidon_set(N: int, d_target: int, time_deadline: float) -> Optional[List[int]]:
    """
    Try many random greedy constructions until a Sidon set of at least
    d_target elements is obtained or the time runs out.
    """
    rnd = random.Random()
    best_set: List[int] = []
    while time.time() < time_deadline:
        S = _greedy_sidon_set(N, rnd)
        if len(S) > len(best_set):
            best_set = S
        if len(S) >= d_target:
            return S
    # if we never hit d_target, return the largest we have (may be None)
    return best_set if best_set else None

# ----------------------------------------------------------------------
# Helper: simple hill‑climbing refinement
# ----------------------------------------------------------------------
def _refine_sequence(seq: np.ndarray,
                     deadline: float,
                     init_temp: float = 1.0,
                     step_frac: float = 0.02) -> np.ndarray:
    """
    Perform a cheap hill‑climbing / simulated‑annealing style refinement.
    Moves a small fraction of mass from an index that participates in the
    current maximal convolution shift to an index outside that shift.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    cur = seq.copy()

    while time.time() < deadline:
        conv = np.convolve(cur, cur)
        max_idx = int(conv.argmax())
        max_val = float(conv.max())

        # range of indices that take part in the maximal shift
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)

        if low > high:
            continue

        # pick i inside [low,high] – weighted by current magnitude
        block = cur[low:high + 1]
        if block.sum() == 0.0:
            i = np.random.randint(low, high + 1)
        else:
            probs = block / block.sum()
            i = np.random.choice(np.arange(low, high + 1), p=probs)

        # pick j outside the block
        outside = list(range(0, low)) + list(range(high + 1, n))
        if outside:
            j = random.choice(outside)
        else:
            # very unlikely, fall back to any other index
            j = (i + 1) % n

        # move a small fraction from i to j
        delta = step_frac * cur[i]
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta

        new_score = evaluate_sequence(cur.tolist())

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_seq = cur.copy()
            # cheap progress output
            print(f"[refine] time={time.time():.1f}s  score={best_score:.6f}")

        # accept only improvements; occasional worse moves with temperature
        if new_score > best_score:
            # linear cooling temperature
            remaining = max(0.0, deadline - time.time())
            total = max(1.0, deadline - (deadline - (deadline - remaining)))
            T = init_temp * (remaining / total)
            if np.random.rand() >= math.exp(-(new_score - best_score) / (T + 1e-12)):
                # reject – revert
                cur[i] += delta
                cur[j] -= delta
    return best_seq

# ----------------------------------------------------------------------

    """
    Return a sequence of non‑negative heights with a low evaluation score.
    The search uses a Sidon‑set based construction and a short local
    refinement.  The function respects the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    # Length of the problem – use the known construction if available
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 1527

    # Starting point: the current best (if any) – copy to avoid mutation
    if 'height_sequence_1' in globals():
        best_seq = np.array(list(height_sequence_1), dtype=float)
    else:
        best_seq = np.full(n, math.sqrt(2 * n) / n)   # uniform fallback

    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}  initial_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1: try to obtain a Sidon set which already beats the target
    # ------------------------------------------------------------------
    target_score = 1.503
    # minimal size of Sidon set needed to reach the target (theoretical bound)
    d_min = int(math.ceil(math.sqrt(4 * n / target_score))) + 1   # safety margin
    print(f"[sidon] need size ≥ {d_min}")

    # allocate a portion of the budget to the Sidon search
    sidon_deadline = time.time() + max(0.0, budget_s * 0.3)  # 30 % of budget
    sidon_set = _find_sidon_set(N=n, d_target=d_min, time_deadline=sidon_deadline)

    if sidon_set is not None and len(sidon_set) >= d_min:
        d = len(sidon_set)
        weight = math.sqrt(2 * n) / d
        cand = np.zeros(n, dtype=float)
        cand[sidon_set] = weight
        cand_score = evaluate_sequence(cand.tolist())
        print(f"[sidon] found size {d}  score={cand_score:.6f}")
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
    else:
        print("[sidon] did not reach required size; continue with refinement.")

    # ------------------------------------------------------------------
    # Phase 2: local refinement of the best sequence found so far
    # ------------------------------------------------------------------
    remaining_budget = max(0.0, budget_s - (time.time() - (sidon_deadline - budget_s * 0.3)))
    refine_deadline = time.time() + remaining_budget
    refined = _refine_sequence(best_seq, deadline=refine_deadline)
    refined_score = evaluate_sequence(refined.tolist())
    print(f"[final] refined_score={refined_score:.6f}")

    if refined_score < best_score:
        best_seq = refined
        best_score = refined_score

    # Return a plain Python list of floats (as required by the evaluator)
    return best_seq.tolist()
