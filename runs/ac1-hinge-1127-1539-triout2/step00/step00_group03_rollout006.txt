def smooth_max(conv, lam):
    # lam >0 large
    # Use log-sum-exp
    max_val = np.max(conv)
    # For numerical stability: subtract max_val
    exp_terms = np.exp(lam * (conv - max_val))
    sm = np.log(np.sum(exp_terms)) / lam + max_val

def fft_convolve(x, y):
    n = len(x) + len(y) - 1
    sz = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, sz)
    Y = np.fft.rfft(y, sz)
    z = np.fft.irfft(X * Y, sz)[:n]
    return z

The method works on the *smoothed* max (log‑sum‑exp) to obtain a useful
gradient, then performs a multiplicative update (a_i ← a_i·e^{-η·grad_i}),
projects back onto the probability simplex and iterates.

The algorithm:
1. Choose a moderately large length (default 5000).  Initialise a with a small
   random perturbation of the uniform distribution (this breaks the symmetry
   of the uniform stationary point).
2. While time permits:
   * Compute the self‑convolution of a with FFT (O(n log n)).
   * Compute a smoothed max weight vector w using a temperature λ.
   * The sub‑gradient of the smoothed max w.r.t. a is
         g = 2·conv(w[::-1], a)   (taken from the appropriate slice).
   * Multiplicative update a ← a·exp(-η·g), clip negatives, and renormalise
     to keep sum(a)=1.
   * If the new sequence improves the true evaluation, keep it and slightly
     increase the learning rate η; otherwise shrink η and occasionally add a
     tiny random perturbation to escape flat regions.
   * Slowly raise λ so the optimisation focuses more on the true max.
3. Keep the best sequence seen during the run (including the optional
   ``height_sequence_1`` if it is already better) and return it.

The procedure is entirely self‑contained, uses only NumPy and the already
available ``evaluate_sequence`` for a final sanity check, and respects the
single‑CPU / 1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: real FFT based convolution (full length)
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[:n]
    return conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length and initialise a (sum = 1)
    n = 5000                      # a good trade‑off between quality & speed
    a = np.full(n, 1.0 / n)       # start from uniform
    # add a tiny random perturbation to break symmetry
    a += rng.normal(scale=1e-3, size=n) * a
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum                  # enforce sum = 1

    # ------------------------------------------------------------------
    # 2. Optionally start from the previously found construction
    best_a = a.copy()
    conv = _fft_convolve(a, a)
    best_E = 2.0 * n * conv.max()
    # try to improve with the globally available height_sequence_1
    if "height_sequence_1" in globals():
        try:
            seq1 = np.asarray(globals()["height_sequence_1"], dtype=float)
            if seq1.size > 0:
                s1_sum = seq1.sum()
                if s1_sum > 0.0:
                    conv1 = _fft_convolve(seq1, seq1)
                    E1 = 2.0 * len(seq1) * conv1.max() / (s1_sum ** 2)
                    if E1 < best_E:
                        # normalise to sum=1 for a fair comparison
                        best_a = seq1 / s1_sum
                        best_E = E1
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3. Gradient‑descent parameters
    lam = 2.0          # smoothing temperature (will increase)
    lam_max = 30.0
    eta = 0.5          # learning rate for the multiplicative update
    eta_min = 0.001
    iteration = 0

    print(f"Start optimisation: n={n}, init_E={best_E:.6f}")

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- current evaluation -----
        conv = _fft_convolve(a, a)
        max_conv = conv.max()
        cur_E = 2.0 * n * max_conv

        # keep the best seen so far
        if cur_E < best_E:
            best_E = cur_E
            best_a = a.copy()
            print(f"Iter {iteration:5d} – new best E = {best_E:.6f}")

        # early exit if we have already passed the target
        if best_E <= 1.503:
            print("Target achieved – stopping early.")
            break

        # ----- smoothed max weights -----
        # log‑sum‑exp trick for numerical stability
        shift = max_conv
        exps = np.exp(lam * (conv - shift))
        w = exps / exps.sum()          # shape (2n‑1,)

        # ----- gradient of the smoothed max -----
        # g = 2 * conv(w_rev, a)  (restricted to the n positions of a)
        w_rev = w[::-1]
        tmp = _fft_convolve(w_rev, a)          # length = (2n‑1) + n – 1
        start = len(w) - 1                     # index where the first a‑overlap occurs
        grad = 2.0 * tmp[start:start + n]      # shape (n,)

        # ----- multiplicative (exponentiated) update -----
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= s

        # ----- accept / reject -----
        conv_new = _fft_convolve(a_new, a_new)
        E_new = 2.0 * n * conv_new.max()
        if E_new < cur_E:
            a = a_new
            eta = min(eta * 1.02, 5.0)          # gently increase step size on success
        else:
            eta = max(eta * 0.5, eta_min)       # shrink step size on failure
            # occasional random jitter to avoid getting stuck
            if iteration % 500 == 0:
                jitter = rng.normal(scale=1e-4, size=n) * a
                a = np.maximum(a + jitter, 0.0)
                a /= a.sum()

        # ----- schedule for the smoothing parameter -----
        lam = min(lam * 1.001, lam_max)

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered
    print(f"Finished after {iteration} iterations. Best evaluation = {best_E:.6f}")
    # The evaluation routine expects a Python list of floats.
    return best_a.tolist()
