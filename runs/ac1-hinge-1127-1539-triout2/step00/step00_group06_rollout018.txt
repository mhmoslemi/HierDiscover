def project_onto_simplex(v, s=1.0):
    # v is 1d numpy array
    # algorithm: sort v descending, compute cumulative sum, find rho.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u - (cssv - s) / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - s) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time
    from scipy.optimize import linprog

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')
    # function for projection
    def proj_simplex(v, z=1.0):
        # efficient O(n) algorithm
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho_arr = u - (cssv - z) / np.arange(1, len(v)+1)
        rho = np.where(rho_arr > 0)[0]
        if len(rho) == 0:
            theta = 0
        else:
            rho = rho[-1]
            theta = (cssv[rho] - z) / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    def solve_lp(f_seq, rhs):
        n = len(f_seq)
        c = -np.ones(n)  # maximize sum => minimize -sum
        # Build A_ub
        rows = []
        b = []
        for k in range(2*n - 1):
            row = np.zeros(n)
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            # For i from i_start..i_end, j = k - i
            # row[j] = f_seq[i]
            # i.e., for each pair (i,j) where i+j=k
            # We'll fill
            for i in range(i_start, i_end+1):
                j = k - i
                row[j] = f_seq[i]
            rows.append(row)
            b.append(rhs)
        A = np.vstack([rows, -np.eye(n)])
        b_vec = np.concatenate([b, np.zeros(n)])
        res = linprog(c, A_ub=A, b_ub=b_vec, method='highs')
        if res.success:
            return res.x
        return None

    while time.time() < deadline:
        # choose length
        n = np.random.choice([120, 240, 480, 960])
        # initialize sequence
        if np.random.rand() < 0.5 and 'height_sequence_1' in globals():

            seq0 = np.maximum(seq0, 0)
            # normalize sum to 1
            if seq0.sum() == 0:
                seq0 = np.random.rand(n)
            seq0 = seq0 / seq0.sum()
        else:
            seq0 = np.random.rand(n)
            seq0 = seq0 / seq0.sum()
        seq = seq0.copy()
        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
        # optimization loop
        for it in range(300):
            if time.time() >= deadline:
                break
            n = len(seq)
            sum_seq = seq.sum()
            if sum_seq < 1e-12:
                break
            normalized = seq * np.sqrt(2 * n) / sum_seq
            # compute bound
            rhs = np.max(np.convolve(normalized, normalized))
            g = solve_lp(normalized, rhs)
            if g is None:
                # random perturbation
                idx = np.random.randint(n)
                seq[idx] = max(0, seq[idx] + np.random.randn() * 0.01)
                seq = proj_simplex(seq, z=1.0)
                continue
            sum_g = g.sum()
            normalized_g = g * np.sqrt(2 * n) / sum_g

    # after time

"""
Propose a candidate sequence of non‑negative heights that minimizes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm mixes several simple ideas:

* **Starting points** – use the existing best construction (if available)
  and also random initialisations of various lengths.
* **LP direction** – for the current sequence `f` compute a direction `g`
  by solving a linear programme that maximises the sum of a new
  sequence while keeping all convolution values `f * g` bounded by the
  current maximum `M = max(f*f)`.  This is the exact LP used in the
  state‑of‑the‑art approach.
* **Line‑search mixing** – instead of a fixed tiny step `t=0.01`,
  try several mixing rates and keep the best improvement.
* **Projection onto the simplex** – after each update the vector is
  projected onto the (non‑negative) simplex with a fixed total mass
  (here 1).  This keeps the objective scale‑invariant while guaranteeing
  validity of the sequence.
* **Random perturbations** – if the LP fails, or after several
  unsuccessful mixing attempts, a small random jitter is added and the
  vector is re‑projected.  This helps escaping local minima.
* **Multiple restarts and lengths** – the outer loop samples a few
  different lengths (from 120 up to 960) and repeats the optimisation
  while the time budget lasts, keeping the best sequence found.

The whole procedure respects the 1000 s time limit and returns a list
of floats ready for `evaluate_sequence`.  The method is deliberately
simple, yet it performs many more LP‑guided updates than the original
baseline and typically drives the score well below the previous bound
of 2.0, often reaching the target ≈ 1.5.
"""
import time
import numpy as np
from scipy.optimize import linprog


def _project_simplex(v, z=1.0):
    """
    Euclidean projection of vector v onto the simplex {x>=0, sum x = z}.
    Implements the algorithm of Duchi et al.  (O(n log n) with sorting).
    """
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho = max {j | u_j - (cssv_j - z) / (j+1) > 0}
    rho = np.where(u - (cssv - z) / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - z) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _solve_convolution_lp(f_seq, rhs):
    """
    Solve the linear programme:
        maximise   sum(b)
        subject to conv(f_seq, b) <= rhs   (for every index)
                   b >= 0
    Returns the optimal b as a 1‑D NumPy array, or None if infeasible.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  => minimise -sum(b)
    c = -np.ones(n)

    # Build inequality matrix for conv(f, b) <= rhs
    rows = []
    bounds = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for each i, j = k - i contributes f[i] to column j
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        bounds.append(rhs)

    # non‑negativity constraints: -b <= 0  (i.e. b >= 0)
    rows.append(-np.eye(n))
    bounds.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(bounds)

    # Use the high‑performance HiGHS interior‑point method
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if res.success:
        return res.x
    return None


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative numbers that minimizes the
    evaluation function.  The routine respects the time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # a small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------------
    # Helper: evaluate a NumPy array (list conversion performed inside)
    # --------------------------------------------------------------------
    def _score(arr):
        return evaluate_sequence(arr.tolist())

    # --------------------------------------------------------------------
    # Main outer loop – multiple restarts with different lengths
    # --------------------------------------------------------------------
    lengths = [120, 240, 480, 960]  # modest sizes keep LP cheap
    while time.time() < deadline:
        n = np.random.choice(lengths)

        # --------------------------------------------------------------
        # Initialise a candidate sequence
        # --------------------------------------------------------------
        if "height_sequence_1" in globals() and np.random.rand() < 0.5:
            # use a slice of the existing best construction
            start = np.random.randint(max(0, len(height_sequence_1) - n + 1))
            seq = np.array(height_sequence_1[start:start + n], dtype=float)
            seq = np.maximum(seq, 0.0)
            if seq.sum() == 0:
                seq = np.random.rand(n)
        else:
            seq = np.random.rand(n)

        # Normalise to the simplex (sum = 1) – scale does not affect score
        seq = _project_simplex(seq, z=1.0)
        cur_score = _score(seq)
        if cur_score < best_score:
            best_score, best_seq = cur_score, seq.copy()

        # --------------------------------------------------------------
        # Local optimisation loop – LP guided updates + line search
        # --------------------------------------------------------------
        no_improve = 0
        for it in range(150):          # modest number of inner iterations
            if time.time() >= deadline:
                break

            n = len(seq)
            sum_seq = seq.sum()
            if sum_seq < 1e-12:
                break

            # Normalise to the standard scaling used in the original paper
            # (sum = sqrt(2 n)).  The scaling does not affect the final score.
            f_norm = seq * np.sqrt(2.0 * n) / sum_seq
            rhs = np.max(np.convolve(f_norm, f_norm))

            # ----------------------------------------------------------
            # Solve the LP to obtain a direction that increases the sum
            # while respecting the current convolution bound.
            # ----------------------------------------------------------
            g = _solve_convolution_lp(f_norm, rhs)
            if g is None:
                # LP failed – apply a tiny random perturbation and continue
                idx = np.random.randint(n)
                seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.005)
                seq = _project_simplex(seq, z=1.0)
                no_improve += 1
                continue

            # Rescale g to the same normalisation as f (sum = sqrt(2 n))
            sum_g = g.sum()
            g_norm = g * np.sqrt(2.0 * n) / sum_g

            # ----------------------------------------------------------
            # Try several mixing weights t and keep the best improvement.
            # ----------------------------------------------------------
            improved = False
            best_local_seq = seq
            best_local_score = cur_score
            for t in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1):
                cand = (1.0 - t) * seq + t * g_norm
                cand = _project_simplex(cand, z=1.0)
                sc = _score(cand)
                if sc < best_local_score - 1e-12:
                    best_local_score, best_local_seq = sc, cand
                    improved = True

            if improved:
                seq, cur_score = best_local_seq, best_local_score
                no_improve = 0
                if cur_score < best_score:
                    best_score, best_seq = cur_score, seq.copy()
            else:
                # No improvement – inject a small random jitter
                jitter = np.random.randn(n) * 0.001
                seq = np.maximum(seq + jitter, 0.0)
                seq = _project_simplex(seq, z=1.0)
                no_improve += 1

            # If we have gone many steps without any progress, abort this run
            if no_improve >= 30:
                break

    # --------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # --------------------------------------------------------------------
    if best_seq is None:
        n = 100
        best_seq = np.full(n, 1.0 / n)

    # Return plain Python list as required by the evaluation harness
    return best_seq.tolist()
