def step_gradient(a, alpha):
    n = len(a)
    # compute conv
    conv = np.convolve(a, a)
    k_max = np.argmax(conv)
    grad = np.zeros_like(a)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]  # derivative
    # Now update a
    a_new = a - alpha * grad
    # Clip to >=0
    a_new = np.maximum(a_new, 0.0)
    # Renormalize to sum 1
    s = np.sum(a_new)
    if s < 1e-12:
        # reinitialize
        a_new = np.full_like(a, 1.0/len(a))
    else:
        a_new = a_new / s
    return a_new

"""
Propose a sequence of non‑negative heights that gives a low value of
evaluate_sequence.  The algorithm is a two‑stage heuristic:

1)  An alternating linear‑program (LP) phase – starting from a uniform
    step function we repeatedly solve a linear program that maximises the
    total mass of a new sequence g under the constraint
    conv(f, g) ≤ M, where M is the current peak of the auto‑convolution of
    f.  After scaling g to the same ℓ¹‑norm as f we replace f by g whenever
    the new peak decreases.  This phase quickly drives the peak down from
    the trivial value (ratio ≈ 2).

2)  A gradient–descent refinement – the peak of the convolution is almost
    always attained at a single shift k*.  The sub‑gradient of the peak
    with respect to the heights a_i is   ∂c[k*]/∂a_i = 2·a_{k*-i}  (if the
    index is in range).  By moving a small step opposite to this gradient,
    renormalising to keep the ℓ¹‑norm constant, and accepting only improving
    steps we flatten the convolution values further.  The gradient step is
    performed with an adaptive step‑size and occasional random perturbations
    to escape shallow local minima.

The whole search runs for the allotted time (default 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: compute the peak of the auto‑convolution using FFT (fast O(n log n))
def _max_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a * a)[k] (no scaling)."""
    n = a.shape[0]
    # length of the linear convolution
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    L = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:conv_len]
    return float(conv.max())

# LP: maximise sum(b) subject to conv(f, b) <= M and b >= 0
def _solve_convolution_lp(f: np.ndarray, M: float):
    """
    Solve   maximise 1ᵀ b   s.t.   (f * b)[k] ≤ M  (for all k) ,  b ≥ 0.
    Returns the optimal b as a np.ndarray or None if the solver fails.
    """
    n = f.shape[0]
    # objective: minimise -1ᵀb  (i.e. maximise 1ᵀb)
    c = -np.ones(n)

    # Build the matrix for the (2n-1) convolution constraints.
    # Row k corresponds to sum_i f[i] * b[k-i] ≤ M .
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over all indices where both i and k-i are valid
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        row[j_vals] = f[i_vals]          # a_i appears as coefficient of b_j
        rows.append(row)
    A_ub = np.vstack(rows)

    # Non‑negativity constraints (b ≥ 0) are expressed as -I·b ≤ 0
    A_ub = np.vstack([A_ub, -np.eye(n)])
    b_ub = np.hstack([np.full(2 * n - 1, M), np.zeros(n)])

    # Use the HiGHS interior‑point solver – it is fast for moderate sizes.
    # Limit the solver time to a few seconds so that a single iteration never
    # stalls the whole search.
    res = linprog(
        c, A_ub=A_ub, b_ub=b_ub,
        method='highs',
        options={'time_limit': 5.0, 'presolve': True, 'disp': False}
    )
    if res.success:
        # Numerical noise may produce tiny negatives – clip to zero.
        b = np.maximum(res.x, 0.0)
        return b
    return None

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative numbers) with a small
    evaluation score.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0            # leave a little margin

    # ------------------------------------------------------------------
    # 1)  Choose a problem size.  Large enough to make the theoretical
    #     bound small (< 2) but not so large that each iteration becomes
    #     prohibitively expensive.
    n = 800                                            # can be tuned

    # Target ℓ¹‑norm is sqrt(2 n) – this is the normalisation used in the
    # original paper and matches the scaling inside evaluate_sequence.
    target_sum = np.sqrt(2.0 * n)

    # Start from a uniform distribution (the trivial solution with score 2).
    f = np.full(n, target_sum / n)

    # ------------------------------------------------------------------
    # 2)  LP improvement phase.
    #    We iterate until the peak of the auto‑convolution cannot be reduced
    #    any more (or we run out of time).
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    max_peak = _max_convolution(f)

    iteration = 0
    while time.time() < deadline:
        # Solve the LP: find a vector g with larger ℓ¹‑mass while keeping
        # conv(f, g) ≤ current peak.
        g = _solve_convolution_lp(f, max_peak)
        if g is None:
            break

        # Scale g to the required ℓ¹‑norm.
        sum_g = np.sum(g)
        if sum_g <= 1e-12:
            break
        g = g * (target_sum / sum_g)

        # Compute the new peak after replacing f by g.
        new_peak = _max_convolution(g)

        # Accept the replacement only if the peak diminishes.
        if new_peak < max_peak * 0.999:                 # a tiny tolerance
            f = g
            max_peak = new_peak
            cur_score = evaluate_sequence(f.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
        else:
            # No improvement – terminate the LP phase.
            break

        iteration += 1
        if iteration % 5 == 0:
            print(f"[LP] iter {iteration}, peak {max_peak:.6e}, best score {best_score:.6f}")

        # Safety break to keep the LP phase short if it stalls.
        if iteration >= 30:
            break

    # ------------------------------------------------------------------
    # 3)  Gradient‑descent refinement.
    #    We minimise the peak of the convolution directly while keeping the
    #    ℓ¹‑norm fixed (by renormalisation after each step).
    a = best_f.copy()
    cur_peak = _max_convolution(a)
    cur_score = 2.0 * n * cur_peak                     # sum = target_sum ⇒ score = 2 n peak/target_sum² = 2 n peak

    step = 0.2                                         # initial gradient step size
    min_step = 1e-6
    iter_gd = 0
    while time.time() < deadline:
        # Compute convolution and locate the maximal shift.
        conv_len = 2 * n - 1
        L = 1 << (conv_len - 1).bit_length()
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:conv_len]
        k_max = int(np.argmax(conv))
        peak = float(conv[k_max])

        # Gradient of the peak shift (sub‑gradient for the max‑operator).
        grad = np.zeros_like(a)
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        i_idx = np.arange(i_min, i_max + 1)
        j_idx = k_max - i_idx
        # derivative: ∂c[k]/∂a_i = 2·a_{k-i}
        grad[i_idx] = 2.0 * a[j_idx]

        # Adaptive line search – try to decrease the peak.
        eta = step
        improved = False
        while eta > min_step:
            a_candidate = a - eta * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            s = a_candidate.sum()
            if s < 1e-12:
                break
            a_candidate = a_candidate / s * target_sum
            cand_peak = _max_convolution(a_candidate)
            cand_score = 2.0 * n * cand_peak
            if cand_score < cur_score:
                a = a_candidate
                cur_peak = cand_peak
                cur_score = cand_score
                step = eta          # keep the successful step size
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best_f = a.copy()
                break
            else:
                eta *= 0.5

        if not improved:
            # Random exploration to escape plateaus.
            noise = np.random.normal(scale=0.02, size=n)
            a_candidate = a + noise
            a_candidate = np.maximum(a_candidate, 0.0)
            s = a_candidate.sum()
            if s > 0:
                a_candidate = a_candidate / s * target_sum
                cand_peak = _max_convolution(a_candidate)
                cand_score = 2.0 * n * cand_peak
                # Accept if it improves, otherwise keep the current point.
                if cand_score < cur_score or np.random.rand() < 0.001:
                    a = a_candidate
                    cur_peak = cand_peak
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_f = a.copy()
            # Reduce step size slowly when no progress.
            step = max(step * 0.99, min_step)

        iter_gd += 1
        if iter_gd % 2000 == 0:
            print(f"[GD] iter {iter_gd}, cur_score {cur_score:.6f}, best_score {best_score:.6f}, step {step:.4e}")

        # Very early exit if we have already passed the target.
        if best_score <= 1.55:          # a little margin above the official target
            break

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list of floats).
    # The caller will feed it to evaluate_sequence; all numbers are already
    # non‑negative and lie well within the allowed range.
    return best_f.tolist()
