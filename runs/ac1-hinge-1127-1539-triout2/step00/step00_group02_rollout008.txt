def solve_min_max_conv(f):
    import numpy as np
    from scipy.optimize import linprog
    n = len(f)
    s = np.sum(f)
    if s == 0: return None, None
    f_norm = f / s

    # variables: n g's + 1 t
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    rows = 2*n - 1
    A_ub = np.zeros((rows, n+1))
    b_ub = np.zeros(rows)
    for k in range(rows):
        # fill A_ub[k, :n]
        for j in range(max(0, k - (n-1)), min(k, n-1) + 1):
            i = k - j
            # i in [0, n-1]
            A_ub[k, j] = f_norm[i]
        A_ub[k, -1] = -1.0
    # equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # bounds: g_j >= 0, t >= 0
    bounds = [(0, None)] * (n+1)
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 2.0})
    if not res.success:
        return None, None
    g = res.x[:n]
    t = res.x[-1]
    return g, t

def project_simplex(v):
    # based on algorithm from https://arxiv.org/abs/1309.1541
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works with a probability vector (sum = 1, all entries ≥0) and
repeatedly solves a tiny linear programme that reduces the current
convolution peak.  A linear programme of the form

    minimise  t
    s.t.      (f * g)_k ≤ t      for all k
              sum(g) = 1
              g ≥ 0

is solved (the convolution with a fixed f is linear in g).  The new g has a
strictly smaller peak, and we replace the current distribution f by g.
Occasionally a random mass‑transfer perturbation is performed to escape
possible stagnation.  The whole search runs until the time budget expires,
keeping the best sequence observed.  The returned list can be any scale
(because the evaluation function is scale‑invariant); we return the
probability vector directly.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_to_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    # Implementation of the algorithm from:
    #   Efficient Projections onto the L1‑Ball for Learning in High Dimensions
    #   (Duchi et al., ICML 2008)
    if np.all(v <= 0):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
def _solve_min_max_convolution(f):
    """
    Given a fixed non‑negative vector f with sum(f) = 1,
    solve the linear programme that finds a new distribution g (sum(g)=1)
    minimising the maximum entry of the convolution f * g.

    Returns (g, t) where t is the optimal peak value.
    If the LP fails, (None, None) is returned.
    """
    n = len(f)
    # Linear programme variables: g_0 … g_{n‑1}, t
    # Objective: minimise t
    c = np.zeros(n + 1)
    c[-1] = 1.0                               # coefficient of t

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Inequalities: (f * g)_k ≤ t   for each k = 0 … 2n‑2
    rows = 2 * n - 1
    A_ub = np.zeros((rows, n + 1))
    b_ub = np.zeros(rows)

    # Build each row of A_ub.
    # (f * g)_k = Σ_i f_i * g_{k-i}  with the convention g_j = 0 if j∉[0,n‑1].
    for k in range(rows):
        # g_j appears with coefficient f_{k-j} whenever 0 ≤ k-j < n.
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            A_ub[k, j] = f[i]          # coefficient of g_j
        A_ub[k, -1] = -1.0             # bring t to the left‑hand side

    # Bounds: all g_j ≥ 0, t ≥ 0
    bounds = [(0, None)] * (n + 1)

    # Solve.  Use the high‑performance HiGHS solver with a small time limit.
    try:
        res = linprog(c,
                      A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': 2.0})
    except Exception:                     # pragma: no cover – safety net
        return None, None

    if not res.success:
        return None, None

    g = res.x[:n]
    t = res.x[-1]

    # Numerical clean‑up
    g = np.maximum(g, 0.0)
    g_sum = np.sum(g)
    if g_sum <= 0:
        return None, None
    g /= g_sum
    t = max(t, 0.0)

    return g, t


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **_):
    """
    Search for a sequence with a low evaluation score.
    The function respects the time budget and returns the best list found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a few seconds for cleanup

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    def _score_of(v):
        """Exact score from the provided evaluate_sequence (scale‑invariant)."""
        # The evaluation function caps values, but our vectors are already
        # normalised (sum = 1) and bounded, so the direct formula works.
        conv = np.convolve(v, v)
        return float(2 * len(v) * np.max(conv))

    # ------------------------------------------------------------------
    # Lengths to try – a trade‑off between flexibility and speed.
    lengths = [200, 400, 800]

    for n in lengths:
        # Several random restarts for each length.
        for restart in range(3):
            # Initial distribution: a random probability vector.
            a = np.random.rand(n)
            a = _project_to_simplex(a)

            # Main optimisation loop.
            iteration = 0
            while time.time() < deadline:
                # Current score
                cur_score = _score_of(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.1f}s] "
                          f"New best – n={n}, restart={restart}, iter={iteration}, score={best_score:.6f}")

                # ---------------------------------------------------------
                # 1) Try a linear‑programming step that strictly reduces the peak.
                g, t = _solve_min_max_convolution(a)
                if g is not None:
                    new_score = 2.0 * n * t
                    # Accept only if we have a genuine improvement.
                    if new_score < cur_score - 1e-9:
                        a = g
                        iteration += 1
                        continue
                    # Occasionally blend to explore a larger neighbourhood.
                    if np.random.rand() < 0.15:
                        a = 0.5 * a + 0.5 * g
                        a = _project_to_simplex(a)
                        iteration += 1
                        continue

                # ---------------------------------------------------------
                # 2) Random local perturbation (mass transfer between two bins).
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
                # Transfer a small amount δ from i to j.
                delta = min(a[i] * 0.5, np.random.rand() * a[i])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                # Re‑project to the simplex to guarantee feasibility.
                a = _project_to_simplex(a)

                iteration += 1

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep the validator happy.
    if best_seq is None:
        best_seq = np.ones(100, dtype=float) / 100.0

    # Convert to a plain Python list as required by the evaluator.
    return best_seq.tolist()
