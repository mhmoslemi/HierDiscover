    from scipy.special import beta
    deadline = time.time() + budget_s - 0.5

    if best_seq is not None:
        # local random perturbation hill climbing
        n = len(best_seq)

"""
Propose a step‑function (non‑negative list of heights) that beats the
current bound (≈2.0) and approaches the theoretical optimum.
The idea is to use a power‑law decreasing shape

    a_i = (i + offset + 1)^{-α}   ,   i = 0 … n‑1

which makes the self‑convolution (the “sum of two independent draws”)
almost flat on the central part of the support.  For a given exponent α
the maximal convolution value is ≈C(α)/n (C(α)= (1‑α)^2·B(1‑α,1‑α)).
Choosing α≈0.6 gives C≈0.723 and a constant contribution
2·C≈1.44 to the evaluation function, already below the target 1.503.
A small offset reduces the first element a_0, therefore the endpoint
convolution term a_0² becomes negligible and the overall score is
governed by the central constant.

The function searches a modest grid of (length n, exponent α, offset)
combinations, keeps the best sequence found, and finally performs a
short hill‑climbing phase with random local perturbations to squeeze a
few more percent out of the score.
All operations respect the security checks in `evaluate_sequence`.
"""

import time
import numpy as np
from scipy import special  # only for the beta function in comments (not used)

def _make_sequence(n: int, alpha: float, offset: int) -> list[float]:
    """Generate a decreasing power‑law sequence of length *n*."""
    # (i+offset+1) prevents a_0 from being too large when offset>0
    return [(i + offset + 1) ** (-alpha) for i in range(n)]

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights whose `evaluate_sequence` score
    is as low as possible (the lower the better).  The routine respects
    the 1000 s time budget.
    """
    # -----------------------------------------------------------------
    # 1.  Basic parameters & time handling
    # -----------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5        # keep a small margin
    np.random.seed(seed)

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # 2.  Coarse grid search over (n, α, offset)
    # -----------------------------------------------------------------
    # reasonable ranges obtained from the analytical discussion
    lengths = list(range(400, 2001, 200))          # 400, 600, …, 2000
    alphas  = np.linspace(0.52, 0.78, 14)         # 0.52 … 0.78
    offsets = list(range(0, 12, 2))               # 0,2,4,6,8,10

    for n in lengths:
        for offset in offsets:
            for alpha in alphas:
                if time.time() > deadline:
                    break
                seq = _make_sequence(n, alpha, offset)
                try:
                    score = evaluate_sequence(seq)
                except Exception:               # safety net
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    print(f"New best: n={n}, α={alpha:.3f}, off={offset}, "
                          f"score={score:.6f}")

    # -----------------------------------------------------------------
    # 3.  Optional local refinement (very cheap hill‑climbing)
    # -----------------------------------------------------------------
    if best_seq is not None:
        cur_seq = best_seq[:]                     # start from the best grid point
        n = len(cur_seq)

        while time.time() < deadline:
            # pick a random index and propose a tiny relative change
            i = np.random.randint(n)
            old = cur_seq[i]
            # perturb by up to ±1 % of the current value (gaussian)
            delta = np.random.randn() * 0.01 * old
            cur_seq[i] = max(0.0, old + delta)

            try:
                score = evaluate_sequence(cur_seq)
            except Exception:
                cur_seq[i] = old                     # revert on error
                continue

            if score < best_score:
                best_score = score
                best_seq = cur_seq[:]                # keep a copy
                print(f"Improved (hill‑climb): n={n}, score={score:.6f}")

            else:
                # undo the change most of the time
                cur_seq[i] = old

            # occasional random restart to escape a poor local basin
            if np.random.rand() < 0.001:
                n = np.random.randint(500, 2001)
                alpha = np.random.uniform(0.5, 0.8)
                offset = np.random.randint(0, 12)
                cur_seq = _make_sequence(n, alpha, offset)

    # -----------------------------------------------------------------
    # 4.  Return the best sequence found
    # -----------------------------------------------------------------
    if best_seq is None:
        # In the very unlikely case nothing was evaluated, fall back to a safe constant.
        best_seq = [1.0] * 1000
        print("Fallback: constant sequence used (should not happen).")
    else:
        print(f"Finished. Best score = {best_score:.6f} "
              f"with length {len(best_seq)}.")

    return best_seq
