Now hill climbing algorithm:

```
def hill_climb(initial_seq, time_budget):
    a = np.array(initial_seq, dtype=float)
    b = np.convolve(a, a)
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    start = time.time()
    n = len(a)
    while time.time() - start < time_budget:
        # Random move
        i = np.random.randint(n)
        # propose new value for a[i] within [0,1000]
        # propose delta as Gaussian with sigma = 0.1 * a[i] + 0.01 (to allow zero)
        sigma = 0.05 * max(1.0, a[i])
        delta = np.random.randn() * sigma
        new_val = a[i] + delta
        # clamp
        new_val = max(0.0, min(1000.0, new_val))
        delta = new_val - a[i]
        if abs(delta) < 1e-12:
            continue
        # apply delta to a and b temporarily (store old values)
        a_old = a[i]
        # update
        apply_delta(a, b, i, delta)
        # compute score
        score = 2 * n * b.max() / (a.sum()**2)
        # if better, keep; else revert
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # Accept move
        else:
            # revert: apply -delta to undo
            # Use apply_delta again with -delta (since a already updated)
            apply_delta(a, b, i, -delta)
    return best_seq.tolist()
```

import numpy as np
import time

def generate_valley(n, c):
    idx = np.arange(n, dtype=float)
    dist = np.minimum(idx, n-1 - idx)
    a = np.exp(-c * dist / n)
    return a

def random_hill_climb(seq, time_budget):
    a = np.array(seq, copy=True)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    n = len(a)
    start = time.time()
    while time.time() - start < time_budget:
        # choose random index, propose random change
        i = np.random.randint(n)
        # propose multiplicative change
        delta = (np.random.rand() - 0.5) * 0.2 * a[i]  # up to +-10%
        new_val = a[i] + delta
        if new_val < 0:
            new_val = 0.0
        if new_val > 1000.0:
            new_val = 1000.0
        old_val = a[i]
        if abs(new_val - old_val) < 1e-12:
            continue
        a[i] = new_val
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        else:
            # revert move
            a[i] = old_val
    return best_seq.tolist(), best_score

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # keep margin
    # Determine length
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000
    print(f"Target length n = {n}")
    best_seq = None
    best_score = float('inf')
    # search over valley shape parameter c
    # try linear and log spaced range
    cs = np.concatenate([np.linspace(0.2, 5.0, 25), np.logspace(-2, 1, 15)])
    cs = np.unique(cs)
    for c in cs:
        if time.time() > deadline:
            break
        seq = generate_valley(n, c).tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"New best from valley c={c:.4f}: score={score:.6f}")
    # Optionally start from uniform random
    # apply hill climbing on best_seq
    remaining = deadline - time.time()
    if remaining > 20 and best_seq is not None:
        print(f"Starting hill climbing with {remaining:.1f}s remaining")
        # allocate half time for hill climbing
        hill_time = remaining * 0.5
        seq_hc, score_hc = random_hill_climb(best_seq, hill_time)
        if score_hc < best_score:
            best_score = score_hc
            best_seq = seq_hc
            print(f"Hill climbing improved score to {best_score:.6f}")
    # If no best found yet (unlikely), fallback to uniform
    if best_seq is None:
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback uniform score {best_score}")
    print(f"Final best score = {best_score}")
    return best_seq

"""
Optimiser for the step‑function evaluation problem.

The algorithm explores a simple one‑parameter family of “valley” sequences:
    a_i = exp( -c * min(i, n-1-i) / n ),   c > 0

These sequences are symmetric, high at the ends and low in the middle.
For this shape the convolution peak is much smaller than for a uniform
sequence while the total sum stays large, giving a much lower evaluation
value (theoretical analysis predicts values near 1 for modest c).

The routine scans a range of c values, keeps the best sequence and then
performs a short stochastic hill‑climbing refinement (random local
perturbations accepted only if they improve the score).  The whole search
fits comfortably inside the 1000 s budget.
"""

import numpy as np
import time

# evaluate_sequence is assumed to be imported in the execution environment.


def _valley_sequence(n: int, c: float) -> np.ndarray:
    """Generate the symmetric exponential “valley” sequence of length n."""
    idx = np.arange(n, dtype=float)
    dist = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
    a = np.exp(-c * dist / n)                    # a_i = exp(-c*dist/n)
    return a


def _hill_climb(seq: np.ndarray, time_budget: float) -> np.ndarray:
    """
    Simple stochastic hill climbing:
    repeatedly tweak a random entry; keep the change only if the score improves.
    """
    best = seq.copy()
    best_score = evaluate_sequence(best.tolist())
    n = len(best)
    start = time.time()

    while time.time() - start < time_budget:
        i = np.random.randint(n)
        # propose a multiplicative perturbation (up to ±15%)
        scale = 1.0 + (np.random.rand() - 0.5) * 0.30
        new_val = best[i] * scale
        new_val = max(0.0, min(1000.0, new_val))

        if abs(new_val - best[i]) < 1e-12:
            continue

        old = best[i]
        best[i] = new_val
        cur_score = evaluate_sequence(best.tolist())

        if cur_score < best_score:
            best_score = cur_score
            # keep the change
        else:
            # revert
            best[i] = old

    return best

    linear_cs = np.linspace(0.2, 5.0, 30)
