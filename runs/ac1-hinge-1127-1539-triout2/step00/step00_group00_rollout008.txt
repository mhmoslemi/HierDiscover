def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum, linprog minimises c^T x.
    # Build the constraints: for each k, sum_i f_i * b_{k-i} <= rhs
    # a_ub shape (2n-1, n)
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
    a_ub = np.vstack(rows + [-np.eye(n)])  # also b >= 0 => -b <= 0
    b_ub = np.hstack([np.full(2*n-1, rhs), np.zeros(n)])
    # Solve
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return np.maximum(res.x, 0.0)  # just in case numeric small negative
    else:
        return None
```

"""
Propose a low‑score step‑function using a simple meta‑heuristic.

Algorithm
---------
1. Initialise a random non‑negative vector `f` of length `n` (chosen
   randomly between 120 and 500).  The vector is scaled so that
   sum(f) = sqrt(2·n) – the same normalisation that makes the evaluation
   function equal to the maximal convolution value.

2. Repeatedly improve the current vector:
   * Compute the maximal self‑convolution `M = max(conv(f,f))`.
   * Solve a linear programme:
        maximise   Σ b_i
        subject to conv(f,b) ≤ M   (2·n‑1 inequalities)
                   b_i ≥ 0
     The LP finds a direction `g` that can increase the total mass while
     keeping the convolution bound.
   * Normalise the LP solution to the same sum as `f`.
   * For the convex combination `h(t) = (1‑t)·f + t·g` the convolution
     at any offset is a quadratic function of `t`.  The maximum of these
     quadratics is a convex function of `t`.  We locate the optimal
     `t∈[0,1]` by a short ternary search (≈30 evaluations) and obtain
     the best combined vector `h`.
   * If the new vector improves the score, keep it as the current
     solution and also store it as the best‑ever sequence.

3. If the LP ever fails (or after many unsuccessful steps) perform a
   cheap random perturbation that moves a little mass from one index to
   another – this helps escape possible stagnation.

4. When the allotted time is almost exhausted, return the best sequence
   found.

The method start‑s from many random seeds and different lengths, so it
does not rely on the previous constant‑height construction and is able
to find sequences with a maximal convolution well below the trivial
value `2.0`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------

def _target_sum(n: int) -> float:
    """Return the normalisation sum sqrt(2·n)."""
    return np.sqrt(2.0 * n)


def _normalise(seq: np.ndarray) -> np.ndarray:
    """Scale seq so that its L1‑norm equals sqrt(2·len(seq))."""
    n = len(seq)
    tgt = _target_sum(n)
    s = seq.sum()
    if s == 0.0:
        # avoid division by zero – fall back to a constant sequence
        return np.full(n, tgt / n)
    return seq * (tgt / s)


def _max_convolution(seq: np.ndarray) -> float:
    """Maximum entry of linear convolution of seq with itself."""
    return np.max(np.convolve(seq, seq))


def _solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Linear programme:
        maximise   Σ b_i
        subject to conv(f,b) ≤ rhs   (2·n‑1 inequalities)
                   b_i ≥ 0
    Returns the optimal vector b (as a 1‑D numpy array) or ``None`` on
    failure.
    """
    n = len(f_seq)
    # objective: minimise -Σ b_i   (linprog does minimisation)
    c = -np.ones(n)

    # Build the matrix for the convolution constraints.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # row[j] = f_i for every (i,j) with i + j = k
        # equivalently, j runs over all indices that appear in the sum.
        # We can fill it by scanning i
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i in [i_min, i_max]; then j = k - i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    # non‑negativity constraints: -b_i ≤ 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([np.full(2 * n - 1, rhs), np.zeros(n)])

    # Solve with the high‑performance HiGHS solver.
    # The time limit option is ignored on older scipy versions – that's fine.
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            method="highs",
            options={"presolve": True, "time_limit": 5.0},
        )
    except Exception:                     # pragma: no cover
        return None

    if res.success:
        # Numerical noise may produce tiny negative values; clip them.
        return np.maximum(res.x, 0.0)
    return None


def _optimal_t(conv_ff: np.ndarray,
               conv_fg: np.ndarray,
               conv_gg: np.ndarray,
               iterations: int = 30) -> float:
    """
    Find t∈[0,1] that minimises
        max_k ( (1‑t)²·conv_ff[k] + 2·t·(1‑t)·conv_fg[k] + t²·conv_gg[k] )
    using a ternary search (the objective is convex).
    """
    lo, hi = 0.0, 1.0
    for _ in range(iterations):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        # vectorised evaluation
        combo1 = (1.0 - m1) ** 2 * conv_ff + 2.0 * m1 * (1.0 - m1) * conv_fg + (m1 ** 2) * conv_gg
        combo2 = (1.0 - m2) ** 2 * conv_ff + 2.0 * m2 * (1.0 - m2) * conv_fg + (m2 ** 2) * conv_gg
        f1 = np.max(combo1)
        f2 = np.max(combo2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
