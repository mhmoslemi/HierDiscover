    """
    Returns a length‑n non‑negative sequence a_i = K * r**i
    with sum(a) = sqrt(2*n).  The decay is controlled by c:
        r = exp(-c / n)
    """
    S = np.sqrt(2.0 * n)                 # required ℓ1‑norm
    r = np.exp(-c / n)                   # 0 < r < 1
    # r**n = exp(-c)  – use that directly for numerical stability
    d = 1.0 - np.exp(-c)                  # denominator 1 - r**n
    K = S * (1.0 - r) / d                # scaling factor
    idx = np.arange(n, dtype=np.float64)
    seq = K * (r ** idx)
    return seq.tolist()


# --------------------------------------------------------------
# Helper: simple linear decreasing shape   a_i ∝ (n-i)
def linear_seq(n: int) -> list[float]:
    """Linear decreasing shape, normalized to the required ℓ1‑norm."""
    S = np.sqrt(2.0 * n)
    # weights decreasing linearly from n … 1
    w = np.arange(n, 0, -1, dtype=np.float64)
    seq = w / w.sum() * S
    return seq.tolist()


# --------------------------------------------------------------
# Local stochastic refinement (pairwise mass transfer)
def hill_climb(seq: np.ndarray, best_score: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a cheap hill‑climbing search around `seq`.
    Small mass is moved between two random positions.
    Returns the (possibly) improved sequence and its score.
    """
    n = len(seq)
    step_scale = max(seq) * 0.02          # typical magnitude of a transfer
    cur_seq = seq.copy()
    cur_score = best_score

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving eps from i to j
        eps = np.random.randn() * step_scale
        # clip to keep entries non‑negative
        eps = max(-cur_seq[j], min(eps, cur_seq[i]))
        if abs(eps) < 1e-12:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps

        # evaluate the new candidate
        try:
            val = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue
        if val < cur_score:
            cur_seq = new_seq
            cur_score = val
            # tighten step_scale when we improve
            step_scale *= 0.99
        else:
            # slowly enlarge step_scale to escape plateaus
            step_scale *= 1.001

    return cur_seq, cur_score


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence with a low evaluation value.
    The routine tries several lengths, geometric families,
    a linear family and finishes with a short hill‑climbing
    refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1) try a handful of lengths (including the known construction length)
    lengths = [500, 1000, 2000, 4000, 7903]
    # include the length of the provided construction if it exists
    try:
        lengths.append(len(height_sequence_1))
    except Exception:
        pass
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # 2) for each length explore geometric families (c‑grid)
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"--- Exploring length {n}")

        # grid of decay parameters; they are chosen loosely around the
        # empirically good region 0.5 … 3.0
        c_grid = np.linspace(0.4, 3.2, 70)

        for c in c_grid:
            if time.time() >= deadline:
                break
            seq = geometric_seq(n, c)
            try:
                score = evaluate_sequence(seq)
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"New best (geom) n={n} c={c:.3f} score={score:.6f}")

        # also try a simple linear decreasing shape
        seq_lin = linear_seq(n)
        try:
            score_lin = evaluate_sequence(seq_lin)
            if score_lin < best_score:
                best_score = score_lin
                best_seq = seq_lin
                print(f"New best (linear) n={n} score={score_lin:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) if we found something, run a short hill‑climbing refinement
    if best_seq is not None:
        seq_arr = np.array(best_seq, dtype=np.float64)
        # allocate a few seconds for refinement (≈5 % of total budget)
        refine_deadline = min(deadline, time.time() + max(2.0, 0.05 * budget_s))
        print(f"Refining best candidate of length {len(seq_arr)}")
        refined_seq, refined_score = hill_climb(seq_arr, best_score, refine_deadline)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.tolist()
            print(f"Refinement improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: return a uniform sequence (the SOTA baseline)
        n = 1000
        vals = [np.sqrt(2.0 * n) / n] * n
        best_seq = vals
        print("Fallback to uniform sequence.")

    print(f"Returning sequence of length {len(best_seq)} with score {best_score:.6f}")
    return best_seq
