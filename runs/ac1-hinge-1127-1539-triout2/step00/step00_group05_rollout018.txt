"""
Gradient‑flattening search for a low‑evaluation step‑function.

The evaluation metric is
    f(a) = 2·n·max_k (a*a)_k / (∑ a_i)²
where (a*a) denotes the linear convolution of the non‑negative vector a.

A perfectly flat convolution would give f = 1.  The classic uniform vector
(a_i = const) gives f = 2.  The continuous function w(x)=1/√x on [0,∞)
produces a convolution that is (almost) constant on the interior of the
discrete index range, yielding f ≈ π/2 ≈ 1.57.  We start from this
discrete analogue and iteratively flatten the convolution by moving mass
away from the index where the convolution is maximal and towards the
index where it is minimal.  The update direction is
    ∇ = a_{k_max‑i} – a_{k_min‑i}
and a simple gradient step (with a decaying learning rate) is applied.
The sequence is kept non‑negative and re‑scaled after each step – scaling
does not affect the evaluation but keeps the numbers in a well‑behaved
range.

The routine runs until the allotted time expires (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    `evaluate_sequence`.  The algorithm uses a gradient‑based flattening of the
    auto‑convolution, starting from the analytically motivated sequence
    a_i = 1 / sqrt(i+1).  It runs for at most `budget_s` seconds and returns
    the best sequence encountered.
    """
    # -------------------------------------------------------------
    # 1. Initial data
    # -------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5          # reserve a few seconds

    # Length – use the length of the previously found construction if
    # available, otherwise pick a convenient size.
    try:
        n = len(height_sequence_1)               # noqa: F821
    except Exception:
        n = 1527                                 # default length from the statement

    # Start from the classic 1/sqrt(i+1) shape (good theoretical start)
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    # Keep a reference sum for later rescaling (scale does not affect the score)
    target_sum = np.sum(a)

    # Keep the best sequence seen so far
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # -------------------------------------------------------------
    # 2. Helper: compute one step of the flatten‑gradient update
    # -------------------------------------------------------------
    def flatten_step(x, lr):
        """Perform a single gradient step that reduces max‑min disparity."""
        # Convolution (full length 2n‑1)
        conv = np.convolve(x, x)

        # Work only on the central part to avoid edge artefacts
        start = n // 4
        end = 3 * n // 4
        central = conv[start:end]

        # Indices of the maximal and minimal values in the central region
        rel_max = int(np.argmax(central))
        rel_min = int(np.argmin(central))
        k_max = start + rel_max
        k_min = start + rel_min

        idx = np.arange(n, dtype=np.int32)

        # Partners for the max‑index
        j_max = k_max - idx
        mask_max = (j_max >= 0) & (j_max < n)
        term_max = np.zeros_like(x)
        term_max[mask_max] = x[j_max[mask_max]]

        # Partners for the min‑index
        j_min = k_min - idx
        mask_min = (j_min >= 0) & (j_min < n)
        term_min = np.zeros_like(x)
        term_min[mask_min] = x[j_min[mask_min]]

        # Gradient (decrease contributions to the max, increase those to the min)
        grad = term_max - term_min

        # Gradient descent step
        y = x - lr * grad
