import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Generate a decreasing polynomial step function.
    a_i = C * (1 - i/(n-1))^p, with p >= 2.
    The resulting evaluation_score is analytically ~
    2 * (p+1)^2 * (p!^2) / (2p+1)!  (< 1.5 for p>=2).
    We try a few small p values and keep the best.
    """
    np.random.seed(seed)

    # Length of the step function: use the previous construction length
    n = 7390

    # Upper bound for each height after the security clipping.
    max_val = 1000.0

    best_seq = None
    best_score = float('inf')

    # Try several exponent values; higher p gives lower bound.
    for p in range(2, 7):  # p = 2 .. 6
        # Build the sequence using a vectorised operation for speed.
        # (1 - i/(n-1)) is in [0,1]; raising to a power keeps it non‑negative.
        idx = np.arange(n, dtype=np.float64)
        factors = (1.0 - idx / (n - 1)) ** p
        seq = (max_val * factors).tolist()

        # Evaluate – the function is already imported in the runner.
        score = evaluate_sequence(seq)

        print(f"p={p}, score={score}")

        if score < best_score:
            best_score = score
            best_seq = seq

    # In the unlikely event that something went wrong, fall back to a uniform vector.
    if best_seq is None:
        best_seq = [max_val] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback: uniform seq score={best_score}")

    print(f"Best score {best_score} obtained with p={p}")
    return best_seq
