        # compute score using evaluate_sequence for safety (fast enough)
        return evaluate_sequence(seq)

        return np.max(np.convolve(seq, seq))

        S = sum(best_seq)

        f_norm = [x * scale for x in best_seq]

        n = len(f_norm)

        A = []
        b = []

"""
Propose a candidate step‑function sequence that (almost) minimizes
evaluate_sequence(sequence).

Algorithm
---------
1.  Initialise a feasible sequence.
    * If a global variable ``height_sequence_1`` exists we take a short
      slice of it (so the LP stays tractable).  Otherwise we start from a
      uniform sequence of modest length (≈200).

2.  Normalise the current sequence to the standard
    ``∑ a_i = sqrt(2·n)`` (the inequality that is optimised in the
    reference paper).  Let ``M`` be the maximal value of the
    self‑convolution of the normalised sequence.

3.  Solve a linear programme

        maximise   Σ b_i
        subject to  (f_norm * b)_k ≤ M   for all k
                    b_i ≥ 0

    where ``f_norm`` is the normalised current sequence.
    The LP finds a new sequence ``b`` with a *larger* ℓ₁‑norm while
    keeping the convolution bound unchanged.

4.  Rescale ``b`` so that its ℓ₁‑norm again equals ``sqrt(2·n)``.
    For several mixing factors ``α∈{0.2,0.5,0.8}`` build candidates

        cand = (1‑α)·f + α·b

    and keep the candidate with the smallest value of
    ``evaluate_sequence`` (i.e. the smallest bound).

5.  If the LP fails (or does not improve the bound) perform a cheap
    random perturbation that keeps the total sum approximately constant.

6.  Every few hundred iterations the algorithm may restart from a fresh
    random sequence to avoid being trapped in a local minimum.

7.  The loop runs until the allocated time budget (≈ budget_s seconds)
    is exhausted; the best sequence encountered is returned.

The implementation uses only NumPy and SciPy (high‑level “highs” LP
solver) and stays well below the memory limits by keeping the problem
size modest (n≈200–500).  Even though the theoretical optimum of the
objective is 2 (the uniform sequence), the search is fully general and
can discover any improvement that might be hidden by numerical effects.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper utilities – they are defined at top level (no closures)
# ------------------------------------------------------------

def _normalise(seq):
    """Scale ``seq`` so that its ℓ₁‑norm equals sqrt(2·len(seq))."""
    n = len(seq)
    target = np.sqrt(2.0 * n)
    s = np.sum(seq)
    if s == 0:
        # avoid division by zero – return a uniform tiny sequence
        return np.full(n, target / n)
    factor = target / s
    return seq * factor


def _max_convolution(seq):
    """Return max_{k} (seq * seq)[k] (full convolution)."""
    return np.max(np.convolve(seq, seq))


def _build_lp_matrix(f):
    """
    Build the LP matrices for a given normalised sequence ``f``.
    Returns ``A_ub`` (2n-1 + n rows) and ``b_ub``.
    """
    n = len(f)
    rows = []
    rhs = []

    # Convolution constraints: (f * b)_k ≤ M, where M = max_conv(f)
    M = _max_convolution(f)

    # For each output index k of the full convolution
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over indices of f that overlap with b at position k
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # fill the row: coefficient of b_j is f_i where j = k - i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
        rhs.append(M)

    # Non‑negativity constraints b_i ≥ 0  →  -b_i ≤ 0
    rows.extend(-np.eye(n))
    rhs.extend([0.0] * n)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs, dtype=float)

    return A_ub, b_ub, M


def _solve_lp(f):
    """
    Solve the linear programme that maximises Σ b_i under
    (f * b) ≤ M and b ≥ 0.
    Returns the optimal b‑vector (as a NumPy array) or ``None`` on failure.
    """
    A_ub, b_ub, M = _build_lp_matrix(f)

    n = len(f)
    c = -np.ones(n)                     # maximise sum → minimise -sum
    # Use the HiGHS interior‑point solver (fast and robust)
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"presolve": True, "time_limit": 8.0},
        )
    except Exception:  # pragma: no cover – safety net
        return None

    if res.success:
        return res.x
    return None


def _mix_candidates(f, g, alphas):
    """
    Produce mixed sequences (1‑α)·f + α·g for each α in ``alphas``.
    The returned list contains NumPy arrays.
    """
    mixes = []
    for α in alphas:
        cand = (1.0 - α) * f + α * g
        # Apply the same clipping that evaluate_sequence performs
        cand = np.clip(cand, 0.0, 1000.0)
        mixes.append(cand)
    return mixes
