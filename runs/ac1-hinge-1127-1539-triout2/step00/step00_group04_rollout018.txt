def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    # parameters
    n = 2000  # perhaps adapt
    a = 1.0 / np.sqrt(np.arange(1, n+1))
    # normalize sum to 1 (optional)
    # but we can keep as is (makes sum ~2*sqrt{n})
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    start_time = time.time()
    # annealing parameters
    temperature = 1.0
    cooling = 0.9995
    # step size factor
    step_factor = 0.01
    while time.time() - start_time < budget_s - 0.5:
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount to move
        delta = step_factor * a[i]
        if delta <= 0:
            continue
        # apply move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonneg
        if a_new[i] < 0:
            continue
        # evaluate
        cur_score = evaluate_sequence(a_new.tolist())
        if cur_score < best_score or np.random.rand() < np.exp(-(cur_score - best_score) / temperature):
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
        # update temperature
        temperature *= cooling
        # occasionally re-normalize to keep sum stable (to avoid drift)
        # but scaling doesn't affect score anyway.
    return best_a.tolist()

"""
Propose a good step‐function sequence.

The algorithm starts from the analytically motivated shape a_i = 1/sqrt(i+1)
(which already gives a score ≈ π/2 ≈ 1.57, well below the trivial value 2).
From there a simple simulated‑annealing loop moves a small amount of mass
between two random positions while keeping all entries non‑negative.
Moves that improve the evaluation score are always accepted; otherwise they
are accepted with a probability that decreases with a temperature schedule.
Occasionally a linear‑programming direction (the same LP used by the
state‑of‑the‑art) is queried and the current sequence is mixed towards that
direction – this helps to escape plateaus.

The routine respects the overall time budget (`budget_s`).  It repeatedly
creates random problem sizes (200 – 2000 items) and performs the local
search, keeping the best sequence found.  The returned list contains only
non‑negative floats, ready for `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Linear‑programming direction (identical to the reference implementation)
# ----------------------------------------------------------------------
def solve_convolution_lp(f_sequence, rhs):
    """
    Solve the LP:
        maximize   Σ b_j
        subject to conv(f, b)_k ≤ rhs   for all k
                  b_j ≥ 0
    Returns the optimal b (as a NumPy array) or None if the LP fails.
    """
    n = len(f_sequence)
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # build convolution constraints (k = 0 … 2n‑2)
    rows = []
    bounds = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        rows.append(row)
        bounds.append(rhs)

    # non‑negativity constraints: -b ≤ 0
    rows.append(-np.eye(n))
    bounds.append(np.zeros(n))

    A = np.vstack(rows)
    b = np.hstack(bounds)

    # use a modest time limit to keep the whole search fast
    res = optimize.linprog(
        c, A_ub=A, b_ub=b,
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(sequence):
    """
    Given the current sequence f, solve the LP to obtain a direction g
    that keeps the convolution bound and has maximal L1‑mass.
    The returned sequence is a small convex combination of f and the
    normalised g (t = 0.5 by default).
    """
    n = len(sequence)
    seq = np.asarray(sequence, dtype=float)

    # normalise so that Σ a_i = sqrt(2n) (the standard normalisation)
    sum_seq = seq.sum()
    if sum_seq == 0:
        return None
    norm_seq = seq * math.sqrt(2 * n) / sum_seq

    # current bound M = max(conv(f,f))
    rhs = np.max(np.convolve(norm_seq, norm_seq))

    # solve the LP for g
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None

    # normalise g to the same L1‑norm as the reference normalisation
    sum_g = g.sum()
    if sum_g == 0:
        return None
    g_norm = g * math.sqrt(2 * n) / sum_g

    # move partway from f to g (larger step than the original 0.01)
    t = 0.5
    new_seq = (1 - t) * norm_seq + t * g_norm
    return new_seq.tolist()


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    The routine repeatedly:
      * picks a random length n (200‑2000),
      * initialises a_i = 1/sqrt(i+1),
      * performs a short simulated‑annealing loop that moves mass between
        two positions,
      * optionally mixes the current point towards an LP direction,
    keeping the best sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.2  # leave a tiny safety margin
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # helper: evaluate and keep the best sequence so far
    # ------------------------------------------------------------------
    def update_best(candidate):
        nonlocal best_seq, best_score
        try:
            sc = evaluate_sequence(candidate)
            if sc < best_score:
                best_score = sc
                best_seq = candidate.copy()
                print(f"New best score: {best_score:.6f} (len={len(candidate)})")
        except Exception:  # safety – the evaluator should never raise
            pass

    # ------------------------------------------------------------------
    # main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # random problem size (still fast to evaluate)
        n = np.random.randint(200, 2000)

        # start from the inverse‑sqrt law (good analytic baseline)
        seq = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))

        # optional small random perturbation to diversify starts
        eps = 0.01 * np.mean(seq)
        seq += eps * np.random.randn(n)
        seq = np.clip(seq, 0.0, None)

        # normalise – scaling does not affect the evaluation, but keeps numbers
        # in a comfortable range
        seq = seq / np.sum(seq) * np.sqrt(2 * n)

        cur_score = evaluate_sequence(seq.tolist())
        update_best(seq.tolist())

        # ----- simulated annealing on this instance -----
        temperature = 1.0
        cooling = 0.9995
        step_factor = 0.03          # relative amount of mass moved each trial
        max_iter = 1500

        for it in range(max_iter):
            if time.time() >= deadline:
                break

            i, j = np.random.randint(n, size=2)
            if i == j:
                continue

            delta = step_factor * seq[i]
            if delta <= 0:
                continue

            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            if new_seq[i] < 0:
                continue  # keep non‑negative

            new_score = evaluate_sequence(new_seq.tolist())

            # accept if better, otherwise with Metropolis probability
            if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / temperature):
                seq = new_seq
                cur_score = new_score
                update_best(seq.tolist())

            temperature *= cooling

        # ----- occasional LP‑based jump -----
        # after the annealing phase we ask the LP for a promising direction
        direction = get_good_direction_to_move_into(seq.tolist())
        if direction is not None:
            dir_score = evaluate_sequence(direction)
            update_best(direction)
            # keep the direction if it improves the current point
            if dir_score < cur_score:
                seq = np.asarray(direction, dtype=float)
                cur_score = dir_score

    # final report
    print(f"Finished search – best score = {best_score:.6f}")
    return best_seq if best_seq is not None else [1.0]  # fallback (should never happen)
