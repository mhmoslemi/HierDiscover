def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small margin

    # start with uniform or previous best
    if 'height_sequence_1' in globals():
        best_seq = np.array(height_sequence_1, dtype=float)
        if len(best_seq) < 1000:
            # optional expand length
            pass
    else:
        # fallback uniform
        n = 1527
        best_seq = np.full(n, np.sqrt(2*n)/n)

    # Ensure sum normalization to sqrt(2n)
    n = len(best_seq)
    target_sum = np.sqrt(2*n)
    best_seq = best_seq * (target_sum / best_seq.sum())
    best_score = evaluate_sequence(best_seq.tolist())
    # simulation annealing variables
    T0 = 0.1 * best_score  # temperature proportional
    T = T0
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # propose move
        # Choose random i
        i = np.random.randint(n)
        # delta proportion of current value maybe also zero if value small
        # sample delta from normal with sigma = 0.05 * avg
        avg = best_seq.mean()
        sigma = avg * 0.05
        delta = np.random.randn() * sigma
        # ensure delta not larger than a_i (can't go negative)
        if delta > best_seq[i]:
            delta = best_seq[i] * np.random.rand()  # bring within
        # pick j random different index
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # apply move: a_i -= delta, a_j += delta
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure nonneg
        if new_seq[i] < 0:
            # fix by moving remaining mass back
            new_seq[i] = 0
        # compute score
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_seq = new_seq
            best_score = new_score
            T = T0 * (0.9995 ** iteration)  # decay slowly
        else:
            # accept with probability
            if np.random.rand() < np.exp(-(new_score - best_score) / max(T, 1e-12)):
                best_seq = new_seq
                best_score = new_score
                T = T0 * (0.9995 ** iteration)
        # occasionally reheat
        if iteration % 10000 == 0:
            T = T0
            print(f"Iter {iteration}, best_score {best_score:.6f}, T {T:.6f}")

    return best_seq.tolist()

def project_onto_simplex(v, s=1.0):
    # ensure v is a numpy array
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[rho-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def local_optimize(a, target_sum, max_iters, deadline):
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    eta0 = target_sum * 0.5  # step size scaling
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        k = np.argmax(conv)
        # gradient
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2 * a[idx[mask]]
        # step size
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        # project
        a = project_onto_simplex(a, target_sum)
        # evaluate randomly maybe every some steps
        if t % 50 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                # early exit if good
                if best_score < 1.503:
                    break
    return best_a, best_score

"""
Propose a candidate step‑function height sequence.

The evaluation is invariant under scaling, therefore we fix the L1‑norm of any
candidate to sqrt(2·n).  The problem is to minimise the maximal entry of the
convolution a*a under this normalisation.

We use a stochastic projected gradient descent on a smooth approximation of
the max‑operator:
    L(a) = (1/β)·log ∑_k exp(β·(a*a)[k])      (β large)

∂L/∂a_i = 2·∑_k w_k·a_{k‑i} with w_k = softmax_k(β·(a*a)[k]).
This gradient can be computed efficiently via a single convolution.

The optimisation loop:
    * start from many random initialisations (Dirichlet, exponential,
      power‑law, mixed) of different lengths,
    * run a few hundred projected gradient steps,
    * keep the best sequence seen.

All iterates are projected onto the probability simplex (sum = sqrt(2·n),
non‑negative) after each update.  The routine stops when the time budget
expires or a sequence with score ≤ 1.503 is found.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x >= 0, sum x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)                # 1‑based indexing
    # Find the largest rho where u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All values would be projected to zero; return uniform distribution
        return np.full_like(v, s / n)
    rho = np.nonzero(cond)[0][-1]            # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_descent(a0: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    beta: float = 20.0,
                    max_iters: int = 10_000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth “soft‑max” loss.
    Returns the best sequence found and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = None

    # initial step–size; proportional to the norm of feasible region
    eta0 = target_sum * 0.5

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution a*a (full length 2n‑1)
        conv = np.convolve(a, a)

        # soft‑max weights over the convolution entries
        m = conv.max()
        # subtract m for numerical stability
        exp_vals = np.exp(beta * (conv - m))
        w = exp_vals / exp_vals.sum()                     # shape (2n‑1,)

        # gradient = 2 * (a_rev convolve w) restricted to n entries
        a_rev = a[::-1]
        conv_rev = np.convolve(a_rev, w)                  # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]             # shape (n,)

        # a step with decreasing step size  η_t = η0 / sqrt(t)
        eta = eta0 / np.sqrt(it)
        a -= eta * grad

        # project back onto the simplex (non‑negative, fixed L1)
        a = _project_onto_simplex(a, target_sum)

        # occasional evaluation to keep the best result
        if it % 20 == 0:
            score = evaluate_sequence(a.tolist())
            if best_score is None or score < best_score:
                best_score = score
                best_a = a.copy()
                # early exit if we hit the target bound
                if best_score <= 1.503:
                    break

    # final evaluation (in case loop ended without a recent check)
    final_score = evaluate_sequence(a.tolist())
    if best_score is None or final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Entry‑point used by the tester.

    The function performs many random restarts, each followed by a short
    projected gradient optimisation using the smooth max‑approximation.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # keep the overall best sequence
    best_sequence = None
    best_score = np.inf

    # a few different lengths are tried; they must be moderate for speed
    length_choices = [200, 500, 1000, 1527, 2000]

    # main search loop – keep launching random restarts while time permits
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) choose problem size and generate an initial vector
        # --------------------------------------------------------------
        n = int(np.random.choice(length_choices))
        target_sum = np.sqrt(2.0 * n)

        init_mode = np.random.choice(["dirichlet", "exp", "pow", "mixed"])

        if init_mode == "exp":
            # exponential decay a_i = r**i
            r = np.random.uniform(0.990, 0.99995)
            a0 = np.power(r, np.arange(n, dtype=float))
        elif init_mode == "pow":
            # power‑law a_i = (i+1)^(-p)
            p = np.random.uniform(0.2, 2.5)
            a0 = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
        elif init_mode == "mixed":
            # product of exponential and power‑law
            r = np.random.uniform(0.95, 0.9995)
            p = np.random.uniform(0.1, 2.0)
            a0 = np.power(r, np.arange(n, dtype=float)) / (np.arange(1, n + 1, dtype=float) ** p)
        else:
            # Dirichlet (uniform random on the simplex before scaling)
            a0 = np.random.dirichlet(np.ones(n))

        # optional random permutation – often helps breaking symmetry
        if np.random.rand() < 0.5:
            np.random.shuffle(a0)

        # scale to the required L1 norm
        a0 = a0 * (target_sum / a0.sum())

        # --------------------------------------------------------------
        # 2) run a short smooth gradient descent
        # --------------------------------------------------------------
        # allocate a per‑restart budget (e.g. 0.2 s); the loop will also stop
        # if the global deadline is reached.
        sub_deadline = min(deadline, time.time() + 0.18)
        a_opt, score = _smooth_descent(a0, target_sum, sub_deadline)

        # --------------------------------------------------------------
        # 3) keep the best sequence ever seen
        # --------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"[{time.time() - start_time:.1f}s] New best: {best_score:.6f} "
                  f"(n={n}, init={init_mode})")
            # Stop early if we already beat the target bound
            if best_score <= 1.503:
                break

    # If nothing was found (should not happen) fall back to the previous best.
    if best_sequence is None:
        raise RuntimeError("Search failed to produce any candidate.")

    return best_sequence.tolist()
