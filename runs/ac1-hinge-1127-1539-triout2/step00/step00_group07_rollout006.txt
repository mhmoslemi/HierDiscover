"""
Propose a (near‑optimal) non‑negative step function.

The algorithm is a refined version of the published alternating‑LP scheme:

1. Keep a sequence `a` whose sum is fixed to `sqrt(2·n)`.  With this
   normalization the evaluation function reduces to `max(conv(a,a))`.

2. Given the current `a`, compute `M = max(conv(a,a))` (fast FFT).

3. Solve a linear program:
        maximise   sum(b)
        s.t.       conv(a,b) ≤ M   (element‑wise)
                   b ≥ 0
   The LP yields a direction `g` that can increase the total sum while
   keeping all cross‑convolution entries ≤ `M`.

4. Renormalise `g` to the same total mass as `a`.

5. Mix the two sequences:
        a(t) = (1‑t)·a + t·g ,   0 ≤ t ≤ 1
   and choose the mixing coefficient `t` that gives the smallest
   `max(conv(a(t),a(t)))`.  A cheap line‑search using a few candidate
   `t`‑values (with an optional golden‑section refinement) is performed.
   The mixed sequence is re‑normalised to keep the sum fixed.

6. Keep the best sequence found (lowest value of `evaluate_sequence`).
   Occasionally:
   * perform a random perturbation of a single entry,
   * combine two of the best sequences found so far,
   * or restart from a fresh random vector.

The loop runs until the allotted time (default ≈ 990 s) is exhausted.
All heavy operations – convolution and the LP – are implemented with
NumPy (FFT) and SciPy (sparse matrices + HiGHS) for speed.

The returned sequence is a list of non‑negative floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def conv_max(seq):
    """Return max_{k} (seq * seq)_k using a fast FFT based convolution."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # size as next power‑of‑two ≥ 2n‑1
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    # numerical noise may produce tiny negatives – ignore them
    return float(np.max(conv))


def normalize_sequence(seq):
    """Scale `seq` so that its sum equals sqrt(2·len(seq)) and clip negatives."""
    n = len(seq)
    s = float(np.sum(seq))
    if s == 0.0:
        return [0.0] * n
    factor = np.sqrt(2.0 * n) / s
    return [max(0.0, float(x * factor)) for x in seq]


def solve_convolution_lp(f_seq, rhs):
    """
    Solve: maximise sum(b)  s.t.  conv(f_seq, b) ≤ rhs,  b ≥ 0.

    The constraints are linear because each entry of conv(f,b) is a
    linear combination of the components of b with coefficients taken
    from `f_seq`.  A sparse matrix (size (2n‑1)×n) is built and handed
    to HiGHS via scipy.optimize.linprog.
    """
    n = len(f_seq)
    # Build the (2n‑1) × n constraint matrix
    rows = []
    cols = []
    data = []
    # For each i (index in f_seq) and each j (index in b) the product
    # contributes to output index k = i + j.
    for i, fi in enumerate(f_seq):
        for j in range(n):
            rows.append(i + j)          # output index k
            cols.append(j)              # variable index
            data.append(fi)             # coefficient f_i
    a_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, rhs, dtype=np.float64)

    # maximise sum(b)  ⇒ minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS is the default fast interior‑point solver in recent SciPy.
    res = linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if res.success:
        return res.x
    return None


def compute_g_direction(seq):
    """
    Given a normalised sequence `seq` (sum = sqrt(2n)),
    compute a partner sequence `g` from the LP that respects the current
    maximal convolution bound.
    """
    n = len(seq)
    M = conv_max(seq)
    raw_g = solve_convolution_lp(seq, M)
    if raw_g is None:
        return None
    sum_g = float(np.sum(raw_g))
    if sum_g == 0.0:
        return None
    # Re‑scale to the required total sum sqrt(2·n)
    factor = np.sqrt(2.0 * n) / sum_g
    return [float(x * factor) for x in raw_g]


def best_mixing_t(a, g, t_candidates):
    """
    Choose `t` from `t_candidates` (list of floats) that minimises
    max(conv((1‑t)·a + t·g, (1‑t)·a + t·g)).
    Returns (best_t, best_value).
    """
    a_arr = np.asarray(a, dtype=np.float64)
    g_arr = np.asarray(g, dtype=np.float64)
    best_t = 0.0
    best_val = conv_max(a_arr)  # current value

    for t in t_candidates:
        mixed = (1.0 - t) * a_arr + t * g_arr
        val = conv_max(mixed)
        if val < best_val:
            best_val = val
            best_t = t

    # Optional small refinement via golden‑section around the best candidate
    if best_t > 0.0:
        lo = 0.0
        hi = min(1.0, best_t * 2.0 + 0.05)  # a modest interval
        phi = (1 + 5 ** 0.5) / 2
        for _ in range(6):  # a few iterations suffice
            t1 = hi - (hi - lo) / phi
            t2 = lo + (hi - lo) / phi
            mixed1 = (1 - t1) * a_arr + t1 * g_arr
            mixed2 = (1 - t2) * a_arr + t2 * g_arr
            v1 = conv_max(mixed1)
            v2 = conv_max(mixed2)
            if v1 < v2:
                hi = t2
                if v1 < best_val:
                    best_val = v1
                    best_t = t1
            else:
                lo = t1
                if v2 < best_val:
                    best_val = v2
                    best_t = t2
    return best_t, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a step‑function sequence with a low value of `evaluate_sequence`.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default 1000). The function stops a few
        seconds before this limit to avoid a hard timeout.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Initialise – use the supplied SOTA sequence if it exists,
    # otherwise fall back to a random dense vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        start_seq = list(height_sequence_1)
    else:
        start_seq = [np.random.rand() for _ in range(500)]

    # Normalise the start vector to the required sum.
    start_seq = normalize_sequence(start_seq)
    n = len(start_seq)

    best_seq = start_seq[:]
    best_score = evaluate_sequence(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    current_seq = best_seq[:]
    # Keep a short list of top sequences for occasional mixing.
    top_seqs = [(best_score, best_seq)]

    deadline = time.time() + budget_s - 10.0  # safety margin

    # Pre‑define a small set of candidate t‑values for the line‑search.
    t_candidates = [0.0, 0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) Obtain a promising partner `g` via the LP.
        # --------------------------------------------------------------
        g_seq = compute_g_direction(current_seq)

        if g_seq is None:
            # LP failed – fall back to a tiny random perturbation.
            idx = np.random.randint(n)
            # Relative perturbation (≈1 % of the current entry)
            delta = np.random.randn() * 0.01 * max(1.0, current_seq[idx])
            new_val = max(0.0, current_seq[idx] + delta)
            current_seq[idx] = new_val
            current_seq = normalize_sequence(current_seq)
            continue

        # --------------------------------------------------------------
        # 2) Choose a mixing coefficient t that reduces the max‑convolution.
        # --------------------------------------------------------------
        best_t, _ = best_mixing_t(current_seq, g_seq, t_candidates)
        if best_t > 0.0:
            # Apply the mixing and renormalise (tiny numerical drift may appear).
            mixed = [(1.0 - best_t) * a + best_t * b for a, b in zip(current_seq, g_seq)]
            current_seq = normalize_sequence(mixed)
        else:
            # No improvement found – perform a small random shake‑up.
            idx = np.random.randint(n)
            delta = np.random.randn() * 0.005 * max(1.0, current_seq[idx])
            current_seq[idx] = max(0.0, current_seq[idx] + delta)
            current_seq = normalize_sequence(current_seq)
            continue

        # --------------------------------------------------------------
        # 3) Evaluate the new sequence and possibly record a new best.
        # --------------------------------------------------------------
        try:
            cur_score = evaluate_sequence(current_seq)
        except Exception:
            cur_score = float("inf")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = current_seq[:]
            print(f"[{iteration:05d}] new best {best_score:.6f}")
            # keep a short sorted list of top performers
            top_seqs.append((best_score, best_seq))
            top_seqs.sort(key=lambda x: x[0])
            top_seqs = top_seqs[:5]

        # --------------------------------------------------------------
        # 4) Occasionally blend two of the elite sequences – this can jump
        #    out of shallow local minima.
        # --------------------------------------------------------------
        if len(top_seqs) > 1 and np.random.rand() < 0.08:
            (s1, seq1), (s2, seq2) = top_seqs[0], top_seqs[np.random.randint(1, len(top_seqs))]
            t_mix = np.random.rand()
            mixed = [(1 - t_mix) * a + t_mix * b for a, b in zip(seq1, seq2)]
            mixed = normalize_sequence(mixed)
            mixed_score = evaluate_sequence(mixed)
            if mixed_score < best_score:
                best_score = mixed_score
                best_seq = mixed[:]
                current_seq = mixed[:]
                top_seqs.append((best_score, best_seq))
                top_seqs.sort(key=lambda x: x[0])
                top_seqs = top_seqs[:5]
                print(f"[{iteration:05d}] mixed improvement {best_score:.6f}")

        # --------------------------------------------------------------
        # 5) Random restart with a small probability to diversify the search.
        # --------------------------------------------------------------
        if np.random.rand() < 0.004:
            rnd = np.random.rand(n)
            current_seq = normalize_sequence(rnd)
            print(f"[{iteration:05d}] random restart")

    print(f"[final] best score obtained: {best_score:.6f}")
    return best_seq
