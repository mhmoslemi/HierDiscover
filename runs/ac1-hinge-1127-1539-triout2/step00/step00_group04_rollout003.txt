def local_optimize(a, time_limit):
    start = time.time()
    best_local = a.copy()
    best_local_score = evaluate_sequence(a)
    iteration = 0
    while time.time() - start < time_limit:
        # compute conv using np.convolve
        b = np.convolve(a, a)
        max_idx = np.argmax(b)
        max_val = b[max_idx]
        # compute gradient direction g:
        # g_i = a[max_idx - i] if in range else 0
        # vectorized:
        idx_range = np.arange(len(a))
        counterpart = max_idx - idx_range
        mask = (counterpart >= 0) & (counterpart < len(a))
        g = np.zeros_like(a)
        g[mask] = a[counterpart[mask]]
        # step size based on multiplier
        # compute squared norm of g
        g_norm2 = np.dot(g, g)
        if g_norm2 == 0:
            # random perturb
            a += np.random.randn(len(a))*0.001
            a = np.clip(a, 0, None)
            a *= S / np.sum(a)
            continue
        # want to reduce max_b; language: derivative of b_k = 2*g; so step = eta * 2g
        # Choose eta such that a_i stays nonnegative likely
        # choose eta = step_factor * (max_val) / (2 * g_norm2)
        step_factor = 0.5   # maybe use decreasing schedule
        eta = step_factor * max_val / (2 * g_norm2 + 1e-12)

        a_new = a - eta * (2 * g)  # because gradient of max_b w.r.t a = 2*g

        a_new = np.maximum(a_new, 0)
        # renorm to sum S
        sum_new = np.sum(a_new)
        if sum_new > 1e-12:
            a_new *= S / sum_new

            a_new *= S / np.sum(a_new)
        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_local_score:

"""
Propose a candidate step‑function sequence.

The algorithm works in three stages:
1.  Choose several diverse initial sequences (uniform, exponential, sparse).
2.  For each start run a simple projected gradient descent that
    repeatedly reduces the current maximal convolution value.
    The gradient of the peak b_k = (a * a)[k] w.r.t. a_i is 2*a_{k-i},
    therefore we move in direction –a_{k-i} (with a safe step size)
    and re‑project onto the simplex (non‑negative, fixed L1‑norm).
3.  Keep the best sequence found within the allotted time.
The method is cheap (a single convolution per iteration) and
exploits the homogeneity of the objective: scaling a does not change
the score, so we constantly re‑normalise to the target sum
sqrt(2 n).  Random perturbations are added periodically to escape
flat stationary points such as the uniform distribution.
"""

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the
    current bound of 2.0.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑clock budget (seconds).  The function stops a few
        seconds before this limit to avoid termination.

    Returns
    -------
    List[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Basic problem parameters
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # use the known length when available
    except Exception:
        n = 5675                           # fallback length (same as the reference)
    target_sum = np.sqrt(2.0 * n)          # S = √(2n)

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate (list → float)
    # ------------------------------------------------------------------
    def score_of(arr):
        """Return evaluate_sequence result, inf on error."""
        try:
            return evaluate_sequence(arr.tolist())
        except Exception:
            return float('inf')

    # ------------------------------------------------------------------
    # Keep the best sequence seen so far
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    # try the existing construction, if any
    try:
        init = np.array(height_sequence_1, dtype=float)
        sc = score_of(init)
        if sc < best_score:
            best_seq = init.copy()
            best_score = sc
            print(f'Initial score from height_sequence_1: {best_score:.6f}')
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Local optimiser (projected gradient on the current maximal conv.)
    # ------------------------------------------------------------------
    def local_optimize(a0, max_seconds):
        """Run a simple gradient descent on a0 for at most max_seconds.
        Returns (best_vector, best_score) found during the run."""
        a = a0.copy()
        # normalise to the required L1 norm
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a = np.full_like(a, target_sum / len(a))

        best_local = a.copy()
        best_local_score = score_of(a)

        deadline = time.time() + max_seconds
        it = 0
        step_factor = 0.5                     # safety factor for step size

        while time.time() < deadline:
            # 1) full linear convolution
            conv = np.convolve(a, a)
            kmax = int(np.argmax(conv))
            max_val = float(conv[kmax])

            # 2) gradient of the active peak: grad = 2 * g,  g_i = a_{kmax-i}
            idx = np.arange(len(a))
            partner = kmax - idx
            mask = (partner >= 0) & (partner < len(a))
            g = np.zeros_like(a)
            g[mask] = a[partner[mask]]

            g_norm2 = np.dot(g, g)
            if g_norm2 == 0.0:

                a += np.random.randn(len(a)) * 1e-3

                a_sum = a.sum()

        else:  # sparse random support

            idx = np.random.choice(n, size=k, replace=False)
            a0[idx] = np.random.rand(k)
            a0 *= (target_sum / a0.sum())

        # ----- run the local optimiser -----------------------------------------
        cand_seq, cand_score = local_optimize(a0, slice_budget * 0.9)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f'New global best: {best_score:.6f} (type={init_type})')

        attempts -= 1

    # ------------------------------------------------------------------
    # Fallback: return the best found (or a uniform sequence if nothing)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Convert to plain Python list (as required by the evaluator)
    return best_seq.tolist()
