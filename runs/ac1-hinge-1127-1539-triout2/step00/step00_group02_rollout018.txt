            if score < best_score:

                best_seq = a.copy()
            # compute weights using smooth max:
            alpha = 10.0 # maybe increase gradually
            w = np.exp(alpha * b)
            w /= w.sum()
            # compute gradient
            grad = np.zeros_like(a)
            for k in range(len(b)):
                wk = w[k]
                # for i where k-i in [0,n-1]
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                # correspond j = k - i
                # contribution = 2 * wk * a_{j}
                # So for i in [i_start,i_end]:
                #   grad[i] += 2 * wk * a[k - i]
                # vectorize: slice
                i_idx = np.arange(i_start, i_end+1)
                grad[i_idx] += 2 * wk * a[k - i_idx]
            # gradient descent update
            eta = 0.01
            a_new = a - eta * grad
            # enforce nonnegative
            a_new = np.maximum(a_new, 0)
            # renormalize sum to 1
            sum_a = a_new.sum()
            if sum_a < 1e-12:
                continue
            a = a_new / sum_a
    return best_seq

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Search for a step function minimizing the evaluation ratio.
    Uses a combination of parametric families, random hill‑climbing and a
    Frank‑Wolfe style linear‑programming update.
    """
    import numpy as np, time, math
    from scipy.optimize import linprog

    # time budget
    end_time = time.time() + budget_s - 5  # leave safety margin

    best_seq = None
    best_score = float('inf')
    # helper to evaluate and update
    def try_sequence(seq):
        nonlocal best_seq, best_score
        # convert to list for evaluate_sequence, ensure non-negative
        try:
            score = evaluate_sequence(seq.tolist() if isinstance(seq, np.ndarray) else list(seq))
        except Exception:
            return
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best: {best_score:.6f} length {len(seq)}")
    
    # 1. try parametric families
    for n in [256, 512, 1024]:
        if time.time() > end_time:
            break
        x = np.arange(n)
        # families: uniform, linear up/down, quadratic powers, sine, cosine, gaussian
        families = []
        # uniform
        families.append(np.ones(n))
        # linear increasing
        families.append(x+1)
        # linear decreasing
        families.append(n - x)
        # power series
        for p in [-2, -1.5, -1, -0.5, 0.5, 1, 1.5, 2]:
            families.append(np.power(x+1, p))
            families.append(np.power(n-x, p))
        # sine
        families.append(np.sin(np.pi * (x + 0.5) / n))
        families.append(np.sin(np.pi * (x + 1) / (n+1)))  # alternative
        # cosine (Hann window)
        families.append(0.5 - 0.5*np.cos(2*np.pi * x/(n-1)))
        # gaussian
        for sigma_factor in [0.1, 0.2, 0.3, 0.4]:
            sigma = sigma_factor * n
            gauss = np.exp(-0.5 * ((x - (n-1)/2)/sigma)**2)
            families.append(gauss)
        # for each raw family:
        for raw in families:
            if time.time() > end_time:
                break
            seq = raw.astype(float)
            # ensure non-negative
            seq[seq < 0] = 0.0
            # normalize sum to 1 (scale invariant)
            s = seq.sum()
            if s < 1e-12:
                continue
            seq = seq / s
            try_sequence(seq)
    
    # 2. Random hill climbing on best candidate (or start from random if none)
    if best_seq is None:
        # fallback random
        n = 512
        seq = np.random.rand(n)
        seq = seq / seq.sum()
        best_seq = seq
        best_score = evaluate_sequence(seq.tolist())
        print(f"Initial random score: {best_score:.6f}")
    # local refinement
    seq = best_seq.copy()
    n = len(seq)
    # parameters
    step = 0.1
    while time.time() < end_time:
        # pick two positions i,j
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # perturb amount
        delta = step * np.random.rand()
        if seq[i] < delta:
            delta = seq[i] * np.random.rand()
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure non-negative
        new_seq = np.maximum(new_seq, 0)
        new_seq = new_seq / new_seq.sum()
        try_sequence(new_seq)
        # maybe accept improvement
        if evaluate_sequence(new_seq.tolist()) < evaluate_sequence(seq.tolist()):
            seq = new_seq
        # gradually reduce step
        step *= 0.9999
        if step < 1e-6:
            step = 0.1
    
    # 3. Optional LP refinement (few rounds)
    # we can do a few outer iterations with LP if time remains
    while time.time() < end_time - 5:
        n = len(best_seq)
        # normalize to sum=1
        f = best_seq / best_seq.sum()
        # compute bound M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Solve LP
        # Build matrix A_ub for constraints conv(f, b) <= M
        # using SciPy's linprog
        # we use form: maximize sum(b) -> minimize -sum(b)
        c = -np.ones(n)
        # build convolution constraint matrix
        # each row k corresponds to conv(f, b)[k] <= M
        a_ub = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            # for i in range(n):
            # j = k - i
            # if 0 <= j < n -> coefficient f[i]
            # so row k entries correspond to f[i] where i satisfies condition
            # i from max(0, k-(n-1)) to min(n-1, k)
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            # for i in start..end: coeff = f[i]
            # corresponding j = k - i (position in b)
            # set a_ub[k, j] = f[i]
            # Since j = k - i
            # We can fill:
            a_ub[k, k - np.arange(i_start, i_end+1)] = f[i_start:i_end+1]
        b_ub = np.full(2*n-1, M)
        # non-negativity constraints: -I b <= 0
        a_ub = np.vstack([a_ub, -np.eye(n)])
        b_ub = np.concatenate([b_ub, np.zeros(n)])
        # solve
        try:
            res = linprog(c, A_ub=a_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': 5.0})
        except Exception as e:
            break
        if not res.success:
            break
        g = res.x
        # normalize g to sum 1
        g = g / g.sum()
        # mix best_seq and g with a few t values
        for t in [0.01, 0.05, 0.1, 0.2, 0.4]:
            new_seq = (1 - t) * best_seq + t * g
            new_seq = np.maximum(new_seq, 0)
            new_seq = new_seq / new_seq.sum()
            try_sequence(new_seq)
        # replace best_seq if improvement
        if evaluate_sequence(g.tolist()) < best_score:
            best_seq = g
            best_score = evaluate_sequence(g.tolist())
            print(f"LP improvement: {best_score:.6f}")
        else:
            # reduce iteration
            break
    
    return best_seq.tolist()

def generate_sidon_set(n, m, max_trials=1000):
    for _ in range(max_trials):
        indices = np.random.choice(n, m, replace=False)  # random distinct
        sums = {}
        ok = True
        for i_idx in range(m):
            for j_idx in range(i_idx, m):
                s = indices[i_idx] + indices[j_idx]
                if s in sums:
                    ok=False
                    break
                sums[s] = True
            if not ok:
                break
        if ok:
            return indices
    return None

Pseudo:

```
def compute_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

"""Step‑function optimizer.

The evaluator is
    2·n·max(conv(a,a)) / (sum(a)²)

which is scale‑invariant.  The goal is to find a non‑negative vector a
with a low value of this expression.  Theory shows that a decreasing
power‑law a[i] ∝ (i+1)^{-p} (p≈0.6) improves the bound substantially
(compared with the uniform sequence which gives a value of 2).  The
search therefore consists of a quick sweep over the exponent p for a
few problem sizes, followed by a light hill‑climbing refinement.
All heavy lifting (convolution) is performed with an FFT based
implementation for speed.
"""

import time
import math
import numpy as np
from scipy import optimize

def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score using an FFT based convolution."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # length of linear convolution is 2*n-1 – pick a power of two for FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _make_power_sequence(n: int, p: float) -> np.ndarray:
    """Return a non‑negative vector a[i] ∝ (i+1)^{-p}, normalized to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    # avoid division by zero when p=0 (uniform); `np.ones` works as well
    if a.sum() == 0.0:
        a = np.ones_like(a)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Search for a step‑function with a low evaluation score.

    The algorithm:
    1. Sweep a few problem sizes (n) and power‑law exponents (p) to get a
       good starting sequence.
    2. Refine the best candidate with a simple random hill‑climbing
       (moving a tiny amount of mass from one index to another) while
       keeping the vector non‑negative and renormalised.
    3. Return the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    end_time = time.time() + budget_s - 5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Parameter sweep (power‑law families)
    # ------------------------------------------------------------------
    for n in (400, 800, 1600, 3200):
        if time.time() > end_time:
            break
        # coarse sweep over the exponent p
        for p in np.linspace(0.05, 0.90, 18):   # 0.05, 0.10, …, 0.90
            seq = _make_power_sequence(n, p)
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(
                    f"New best from sweep: n={n} p={p:.3f} score={score:.6f}"
                )
        # a tiny local refinement around the best p for this n
        # (simple golden‑section search on a narrow interval)
        if best_seq is not None:
            # find the p that gave the current best for this n
            # (re‑evaluate around it)
            lo, hi = max(0.0, p - 0.1), min(1.0, p + 0.1)
            for _ in range(12):
                p1 = lo + (hi - lo) * 0.382
                p2 = lo + (hi - lo) * 0.618
                s1 = _fast_score(_make_power_sequence(n, p1))
                s2 = _fast_score(_make_power_sequence(n, p2))
                if s1 < s2:
                    hi = p2
                else:
                    lo = p1
                if time.time() > end_time:
                    break
            # take the midpoint as the refined exponent
            p_mid = (lo + hi) / 2.0
            seq = _make_power_sequence(n, p_mid)
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(
                    f"Refined best: n={n} p≈{p_mid:.4f} score={score:.6f}"
                )

    # ------------------------------------------------------------------
    # 2) Light hill‑climbing from the best candidate found so far
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of modest length
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _fast_score(best_seq)

    seq = best_seq.copy()
    n = seq.shape[0]
    step = 0.2                     # initial fraction of mass to move
    while time.time() < end_time:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # propose moving a small amount of mass from i to j
        delta = step * np.random.rand()
        if seq[i] < delta:
            delta = seq[i] * np.random.rand()
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # enforce non‑negativity and renormalise (scaling does not matter)
        new_seq = np.maximum(new_seq, 0.0)
        total = new_seq.sum()
        if total < 1e-12:
            continue
        new_seq /= total
        new_score = _fast_score(new_seq)
        # accept only improvements (simple greedy descent)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq
            print(
                f"Hill‑climb improvement: score={new_score:.6f} step={step:.4f}"
            )
        else:
            # occasionally accept a worse move to escape plateaus
            if np.random.rand() < 0.001:
                seq = new_seq
        # slowly anneal the step size
        step *= 0.9999
        if step < 1e-5:
            step = 0.2

    # ------------------------------------------------------------------
    # 3) Final reporting
    # ------------------------------------------------------------------
    print(f"Finished search – best score = {best_score:.6f} (length {len(best_seq)})")
    # the caller expects a plain list of floats
    return best_seq.tolist()
