def propose_candidate(seed=...):
    import numpy as np, time, random
    np.random.seed(seed)
    n = 2000 (or given length maybe from height_sequence_1 length).
    # initialize with uniform or random
    a = np.random.rand(n)
    # ensure positive
    a = np.abs(a)
    # maybe apply smoothing to avoid spikes.
    # optionally start from uniform: a = np.ones(n)

    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # Simulated annealing parameters
    start_temp = 1.0
    end_temp = 0.001
    total_iters = 20000
    start_time = time.time()
    deadline = start_time + 990

    for iteration in range(int(total_iters)):
        if time.time() > deadline:
            break
        # temperature schedule
        t = start_temp * (end_temp/start_temp) ** (iteration/total_iters)
        # propose move: choose indices i,j distinct
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # delta = random fraction of a[i]
        delta = np.random.exponential(scale=0.1) * a[i]
        # ensure delta <= a[i]
        delta = min(delta, a[i] * 0.5) # cap
        # propose new vector
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate candidate
        score = evaluate_sequence(a_new.tolist())
        # accept if better or with probability exp((best_score - score)/t)
        if score < best_score or np.random.rand() < np.exp((best_score - score)/t):
            a = a_new
            if score < best_score:
                best_score = score
                best = a_new.copy()
                print(f"Iter {iteration}, new best {best_score:.6f}")
    # after loop, return best.tolist()
    return best.tolist()

propose_candidate

Searches for a non‑negative step sequence that gives a low value of
evaluate_sequence.  The method is a cheap multi‑stage heuristic:

1.  Try a handful of analytically‑motivated families (geometric,
    symmetric‑geometric, power‑law, simple linear ramps).  The evaluation
    of each candidate is cheap and provides good initial points.

2.  Starting from the best candidate found so far, perform a simulated‑annealing /
    hill‑climbing phase.  In each iteration a small amount of mass is moved
    from one index to another (respecting non‑negativity) and the new
    sequence is accepted if it improves the score (or with a tiny
    probability for a worse score to escape flat regions).  Because the
    objective is scale‑invariant we keep the raw values – only non‑negativity
    matters.

The whole search runs until the allotted budget (≈ 1000 s) is exhausted and
returns the best sequence encountered.
"""

    return evaluate_sequence(seq.tolist())


def _try_candidate(seq: np.ndarray, best_seq, best_score):
    """Evaluate a candidate and update the best if it improves."""
    score = _evaluate(seq)
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
        print(f"  New best score: {best_score:.6f}")
    return best_seq, best_score


def _geometric_seq(n: int, r: float) -> np.ndarray:
    """Monotonically decreasing geometric sequence a_i = r**i."""
    return np.power(r, np.arange(n, dtype=float))


def _sym_geometric_seq(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric tails:
    a_i = r**i + r**(n-1-i)  (no normalisation – irrelevant for the score)."""
    idx = np.arange(n, dtype=float)
    return np.power(r, idx) + np.power(r, n - 1 - idx)


def _power_law_seq(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)^{-p}."""
    return 1.0 / np.power(np.arange(1, n + 1, dtype=float), p)


def _v_shape_seq(n: int, width: int) -> np.ndarray:
    """Piecewise linear V‑shape: high at both ends, low in the middle."""
    if width * 2 >= n:
        # fall back to a simple linear ramp
        return np.linspace(1.0, 0.0, n)
    left = np.linspace(1.0, 0.0, width, endpoint=False)
    middle = np.zeros(n - 2 * width)
    right = left[::-1]
    return np.concatenate([left, middle, right])


def _perturb(seq: np.ndarray, max_step: float = 0.1) -> np.ndarray:
    """Move a small random amount of mass from one index to another."""
    n = seq.size
    i, j = np.random.randint(n, size=2)
    if i == j:
        return seq  # nothing to do
    # amount proportional to the source value, limited by max_step
    amount = seq[i] * max_step * np.random.rand()
    if amount > seq[i]:
        amount = seq[i]
    new_seq = seq.copy()
    new_seq[i] -= amount
    new_seq[j] += amount
    # enforce the non‑negative guard that evaluate_sequence also applies
    new_seq[i] = max(0.0, new_seq[i])
    new_seq[j] = max(0.0, new_seq[j])
    return new_seq
