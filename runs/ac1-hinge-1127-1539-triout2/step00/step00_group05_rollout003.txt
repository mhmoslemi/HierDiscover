def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    """Projects vector v onto the probability simplex (nonnegative and sum=1)."""
    n = v.shape[0]
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def optimize_for_n(n, time_deadline, seed):
    np.random.seed(seed)
    a = np.random.rand(n)
    a /= a.sum()
    cur_obj = objective(a)
    best_obj = cur_obj
    best_a = a.copy()
    step = 0.5
    max_step = 5.0
    alpha = n  # smoothing parameter
    no_improve = 0
    iteration = 0
    while time.time() < time_deadline:
        iteration += 1
        # compute gradient
        B = np.convolve(a, a)
        M = B.max()
        # smooth weights
        w = np.exp(alpha * (B - M))  # M is max, subtraction to keep <=1
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
        T = np.correlate(w, a, mode='valid')
        diag = w[2*np.arange(n)] * a
        g = 2 * T - diag
        # step
        a_new = a - step * g
        a_new = project_simplex(a_new)
        new_obj = objective(a_new)
        if new_obj < cur_obj:
            a = a_new
            cur_obj = new_obj
            no_improve = 0
            # increase step a bit
            step = min(step * 1.02, max_step)
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a_new.copy()
                print(f"n={n}, iter={iteration}, score={best_obj:.6f}")
        else:
            # no improvement, reduce step or random jitter
            step *= 0.5
            no_improve += 1
            if step < 1e-6 or no_improve > 200:
                # random restart for this n
                a = np.random.rand(n)
                a /= a.sum()
                step = 0.5
                no_improve = 0
    return best_a, best_obj

The algorithm is a simple projected gradient descent on a
smoothed version of the max‑convolution (soft‑max).  At each
iteration we

1. compute the self‑convolution,
2. form soft‑max weights w_k ∝ exp(α·conv_k)    (α ≃ n),
3. obtain the gradient of the smoothed max w.r.t. a
   (g_i = 2·∑_k w_k a_{k-i} – w_{2i}·a_i),
4. take a step a←a−η·g and project back onto the probability simplex
   (non‑negative, sum = 1).

The step size η is adapted: it grows after a successful move and
shrinks otherwise; after many consecutive failures we randomise the
current point.  The routine runs several independent restarts with
different lengths (up to a few thousand entries) while respecting the
overall time budget.  The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, sum_i x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]

    # Fast path for already feasible vectors
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()

    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence (sum of a is assumed >0).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_conv = conv.max()
    s = a.sum()
    return 2.0 * n * max_conv / (s * s)


def smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution:
        S(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∇S(a) (same length as a).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    m = conv.max()
    # soft‑max weights (stable formulation)
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation w * a (valid mode gives length n)
    T = np.correlate(w, a, mode='valid')
    # diagonal correction term
    diag = w[2 * np.arange(n)] * a
    # gradient: 2·T – diag
    g = 2.0 * T - diag
    return g


def optimize_for_length(
    n: int,
    deadline: float,
    rng_seed: int,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Run projected gradient descent on a sequence of length n.
    Returns the best sequence discovered (as a numpy array) and its
    objective value.
    """
    rng = np.random.default_rng(rng_seed)

    # ------------------------------------------------------------------
    # Initial point
    # ------------------------------------------------------------------
    if init_seq is not None and init_seq.shape[0] == n:
        a = init_seq.copy()
        a = project_simplex(a)          # ensure feasibility
    else:
        a = rng.random(n)
        a = a / a.sum()

    cur_val = objective(a)
    best_a = a.copy()
    best_val = cur_val

    # hyper‑parameters (tuned empirically)
    step = 0.5               # initial step size
    max_step = 5.0
    alpha = float(n)         # smoothing parameter α ≈ n
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # compute gradient of smoothed max
        g = smooth_gradient(a, alpha)

        # tentative update + projection
        a_new = a - step * g
        a_new = project_simplex(a_new)

        new_val = objective(a_new)

        if new_val < cur_val:
            # successful move
            a = a_new
            cur_val = new_val
            no_improve = 0
            step = min(step * 1.02, max_step)

            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(
                    f"[len {n}] iter {iteration} – new best "
                    f"obj={best_val:.6f}  step={step:.4f}"
                )
        else:
            # failed move
            step *= 0.5
            no_improve += 1

            if step < 1e-6 or no_improve > 200:
                # random restart for robustness
                a = rng.random(n)
                a = a / a.sum()
                step = 0.5
                no_improve = 0

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimizes the provided evaluation function.  The search respects
    the given time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so that we always finish in time
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Starting point – try the previously known construction if available
    # ------------------------------------------------------------------
    try:
        best_seq = list(height_sequence_1)  # type: ignore[name-defined]
        best_score = evaluate_sequence(best_seq)
        print(f"Initial score from height_sequence_1: {best_score:.6f}")
    except Exception:
        best_seq = None
        best_score = np.inf

    # also keep a simple uniform sequence as a cheap baseline
    uniform_len = 1527
    uniform = np.full(uniform_len, 1.0 / uniform_len)
    uniform_score = evaluate_sequence(uniform.tolist())
    if uniform_score < best_score:
        best_seq = uniform.tolist()
        best_score = uniform_score
        print(f"Uniform baseline score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – longer sequences have a better theoretical bound
    # ------------------------------------------------------------------
    lengths = [500, 1000, 1500, 2000, 2500, 3000]

    # Run independent optimisation runs for the chosen lengths
    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining time to this run
        remaining = deadline - time.time()
        alloc = min(remaining * 0.18, 120.0)   # at most 2 min per length
        sub_deadline = time.time() + alloc

        # initialise from the current best if lengths match, otherwise random
        init = None
        if best_seq is not None and len(best_seq) == n:
            init = np.asarray(best_seq, dtype=float)

        seq_opt, opt_val = optimize_for_length(
            n=n,
            deadline=sub_deadline,
            rng_seed=seed + idx + 12345,
            init_seq=init,
        )

        # evaluate with the official routine (extra safety)
        try:
            seq_list = seq_opt.tolist()
            real_score = evaluate_sequence(seq_list)
        except Exception:
            real_score = np.inf

        if real_score < best_score:
            best_score = real_score
            best_seq = seq_list
            print(
                f"*** NEW GLOBAL BEST *** length={n}  score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final polishing: small random perturbations around the best point
    # ------------------------------------------------------------------
    while time.time() < deadline:
        if best_seq is None:
            break
        cur = np.asarray(best_seq, dtype=float)
        n = cur.shape[0]

        # perturb a few coordinates
        idx = np.random.randint(0, n)
        delta = np.random.randn() * 0.02
        cur[idx] = max(0.0, cur[idx] + delta)

        cur = project_simplex(cur)
        cand_score = evaluate_sequence(cur.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cur.tolist()
            print(f"Polish improvement → score {best_score:.6f}")

    # safety fallback – should never happen
    if best_seq is None:
        best_seq = [1.0] * 1527

    print(f"Returning best sequence (length={len(best_seq)}) with score {best_score:.6f}")
    return best_seq
