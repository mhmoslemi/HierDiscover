def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np, random
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 5  # margin

    # Choose length:
    n = 2000 # could adapt based on time.

    # Init uniform
    a = np.full(n, 1.0 / n)

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    cur = a.copy()
    cur_score = best_score

    # Precompute FFT size:
    size = 1 << ((2*n-1).bit_length())

    iter_cnt = 0
    last_print = start

    while time.time() < deadline:
        # compute conv with FFT
        fft_a = np.fft.rfft(cur, n=size)
        conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
        k_max = int(np.argmax(conv))
        # contributions
        contrib = np.zeros(n)
        # using vectorized indexing
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >=0) & (partner < n)
        contrib[mask] = cur[idx[mask]] * cur[partner[mask]]
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))

        # step fraction
        delta = 0.05 * cur[i_max]  # 5% move
        if delta <= 1e-12:
            delta = 1e-12
        new = cur.copy()
        new[i_max] -= delta
        new[i_min] += delta

        # Ensure non-negativity
        if new[i_max] < 0:
            new[i_max] = 0.0
        # sum remains constant (since we moved delta)
        # compute score
        new_score = evaluate_sequence(new.tolist())
        iter_cnt += 1

        # Accept move if better than current
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        else:
            # with small probability accept fallback random move
            # do nothing else
            pass

        # occasional random exploration
        if np.random.rand() < 0.01:
            i, j = np.random.randint(n, size=2)
            if i != j:
                epsilon = 0.02 * cur[i]  # small fraction
                new2 = cur.copy()
                transfer = min(epsilon, cur[i])
                new2[i] -= transfer
                new2[j] += transfer
                sc2 = evaluate_sequence(new2.tolist())
                if sc2 < cur_score:
                    cur = new2
                    cur_score = sc2
                    if sc2 < best_score:
                        best = new2.copy()
                        best_score = sc2

        # periodic logging
        now = time.time()
        if now - last_print > 10:
            print(f"Time {now-start:.1f}s iter {iter_cnt} best {best_score:.6f}")
            last_print = now

    print(f"Finished. Best score: {best_score:.6f}")
    return best.tolist()

"""
Propose a candidate step‑function (list of non‑negative heights) that tries to
minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear (zero‑padded) convolution used in
`evaluate_sequence`.  The algorithm is a cheap hill‑climber:

* start from a random normalised vector (sum == 1);
* repeatedly compute its auto‑convolution via an FFT;
* locate the index `k_max` where the convolution is maximal;
* among all pairs (i, k_max-i) pick the pair with the largest product
  `a[i] * a[k_max-i]` – this pair contributes the most to the current peak;
* move a small fraction of mass from this index to another index that does
  **not** participate in the current peak (its product is zero, or minimal);
* accept the move only if it strictly reduces the objective.
* keep the best sequence seen so far; stop early if the target score
  (1.503) is reached or the time budget expires.

The method works entirely with NumPy and runs in O(n log n) per iteration
thanks to the FFT‑based convolution, so even a few hundred thousand iterations
fit easily into the 1000 s budget for lengths of several thousand elements.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_max_and_arg(seq: np.ndarray):
    """
    Compute the full linear convolution of ``seq`` with itself,
    return (max_value, arg_of_max, full_convolution_array).
    Uses real FFT for speed.
    """
    n = seq.size
    size = _next_pow2(2 * n - 1)
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values; clamp to zero.
    conv = np.maximum(conv, 0.0)
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return max_val, max_idx, conv


def _objective(seq: np.ndarray, max_conv: float = None) -> float:
    """
    Return the value of the objective used by ``evaluate_sequence``.
    The sum of the sequence is assumed to be positive; if ``max_conv`` is
    provided we reuse it to avoid a second convolution.
    """
    n = seq.size
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")
    if max_conv is None:
        max_conv = _conv_max_and_arg(seq)[0]
    return 2.0 * n * max_conv / (s * s)

# Main candidate generator
# ----------------------------------------------------------------------

    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a few seconds margin
    log_interval = 10.0
    next_log = start_time + log_interval
    step_fraction = 0.15          # proportion of mass moved each trial
    min_fraction = 0.001          # lower bound to keep steps sensible
    target_score = 1.503

    it = 0
    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------------
        # Identify the offset where the convolution is maximal.
        # ----------------------------------------------------------------
        cur_max, cur_kmax, cur_conv = _conv_max_and_arg(cur)

        # ----------------------------------------------------------------
        # For that offset compute the contribution a[i] * a[k_max-i] of each
        # index i (zero if the partner index is out of range).
        # ----------------------------------------------------------------
        idx = np.arange(n)
        partner = cur_kmax - idx
        valid = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=cur.dtype)
        contrib[valid] = cur[idx[valid]] * cur[partner[valid]]

        # ----------------------------------------------------------------
        # Choose the index that contributes most (i_max) and an index that
        # contributes the least (i_min).  If there are many zero‑contribution
        # indices we pick one of them at random – moving mass there does not
        # affect the current peak.
        # ----------------------------------------------------------------
        i_max = int(np.argmax(contrib))
        zero_mask = contrib == 0
        if np.any(zero_mask):
            i_min = int(np.random.choice(np.flatnonzero(zero_mask)))
        else:
            i_min = int(np.argmin(contrib))

        # avoid a degenerate move where source and destination are identical
        if i_max == i_min:
            candidates = np.delete(np.arange(n), i_max)
            i_min = int(np.random.choice(candidates))

        # ----------------------------------------------------------------
        # Propose a small transfer of mass from i_max to i_min.
        # ----------------------------------------------------------------
        delta = max(step_fraction * cur[i_max], min_fraction * cur[i_max])
        delta = min(delta, cur[i_max])          # never become negative

        new_seq = cur.copy()
        new_seq[i_max] -= delta
        new_seq[i_min] += delta

        # --------------------------------------------------------------
        # Evaluate the new candidate.  Because the sum of the vector stays
        # unchanged, the objective can be computed directly from the new
        # maximal convolution value.
        # --------------------------------------------------------------
        new_max, _, _ = _conv_max_and_arg(new_seq)
        new_score = _objective(new_seq, new_max)

        # --------------------------------------------------------------
        # Accept the move only if it strictly improves the objective.
        # Keep the globally best sequence seen.
        # --------------------------------------------------------------
        if new_score < cur_score:
            cur = new_seq
            cur_score = new_score
            if new_score < best_score:
                best = new_seq.copy()
                best_score = new_score

        # --------------------------------------------------------------
        # Occasionally perform a larger random perturbation to escape local
        # minima.  The probability is low and the perturbation magnitude is
        # modest to keep the search focused.
        # --------------------------------------------------------------
        if np.random.rand() < 0.005:
            # random redistribution of a modest amount of mass
            i, j = np.random.choice(n, size=2, replace=False)
            transfer = min(cur[i], 0.1 * cur[i])
            if transfer > 0:
                cur[i] -= transfer
                cur[j] += transfer
                # re‑evaluate after the perturbation
                cur_max, _, _ = _conv_max_and_arg(cur)
                cur_score = _objective(cur, cur_max)
                if cur_score < best_score:
                    best = cur.copy()
                    best_score = cur_score

        # --------------------------------------------------------------
        # Logging – every few seconds print the current best value.
        # --------------------------------------------------------------
        now = time.time()
        if now >= next_log:
            elapsed = now - start_time
            print(
                f"[{elapsed:6.1f}s] it={it:,d} best={best_score:.6f} "
                f"cur={cur_score:.6f}"
            )
            next_log = now + log_interval

        # --------------------------------------------------------------
        # Early termination if we have already reached the target.
        # --------------------------------------------------------------
        if best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found, as a plain Python list.
    # ------------------------------------------------------------------
    print(f"Search finished after {it:,d} iterations. Best score: {best_score:.6f}")
    # Ensure the returned list satisfies the input checks of evaluate_sequence
    # (non‑negative, finite, not all zeros).  Normalisation is not required,
    # but keeping the sum ~= 1 keeps the numbers well‑behaved.
    return best.tolist()
