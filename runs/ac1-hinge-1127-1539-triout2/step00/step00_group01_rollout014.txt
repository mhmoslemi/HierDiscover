def convolve_full(a,b):
    size = len(a) + len(b) - 1
    nfft = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(fa*fb, n=nfft)[:size]
    return conv

import numpy as np
import time

def conv_full(x, y):
    size = len(x) + len(y) - 1
    nfft = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, n=nfft)[:size]
    return conv

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    
    The algorithm works with a fixed L1 norm (``sqrt(2*n)``) so that the
    objective reduces to minimising the maximum entry of the self‑convolution.
    Starting from a random vector we repeatedly:
        * compute the full convolution,
        * build a soft‑max weight emphasising the current largest coefficient,
        * take a gradient descent step on the weighted sum,
        * project onto the non‑negative simplex (renormalise),
        * keep the best sequence seen.
    A small random jitter is added from time to time to escape possible
    plateaux.  The whole procedure runs until the allotted time runs out.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    # Length of the sequence – try to use the length of the known construction
    # if it exists, otherwise fall back to a moderate size.
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000               # reasonable default

    # target L1 norm (chosen because evaluate_sequence simplifies to max_b)
    target_sum = np.sqrt(2.0 * n)

    # ----- initialise a random feasible point -----
    a = np.random.exponential(scale=1.0, size=n)
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_score = np.max(conv_full(a, a))
    print(f"Start: n={n}, initial max={best_score:.6f}")

    # hyper‑parameters
    lambda_param = 10.0               # softness of the max
    step_factor = 0.5                 # scales the step size
    jitter_interval = 50
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # current convolution and score
        b = conv_full(a, a)
        cur_max = b.max()
        # keep best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"Iter {iteration:5d}: new best max={best_score:.6f}")

        # soft‑max weights concentrated on the largest coefficients
        w = np.exp(lambda_param * (b - cur_max))   # max => exponent 0, others < 0
        w_sum = w.sum()
        if w_sum == 0:
            # numerical safety – fall back to uniform weights
            w = np.ones_like(b) / b.size
        else:
            w = w / w_sum

        # gradient of the weighted max (∂/∂a_i of Σ w_k (a*a)_k)
        a_rev = a[::-1]
        grad_full = conv_full(w, a_rev)          # length 3n‑2
        grad = 2.0 * grad_full[n-1:n-1+n]        # slice of length n

        # a simple normalised step
        norm_grad = np.linalg.norm(grad) + 1e-12
        step = step_factor * (target_sum / norm_grad)

        # take a gradient step, project back to the simplex
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new * (target_sum / s)
        else:
            # all zero – re‑initialise
            a_new = np.random.exponential(scale=1.0, size=n)
            a_new = a_new * (target_sum / a_new.sum())

"""
Search for a non‑negative step‑function sequence that minimises
evaluate_sequence(sequence).  The evaluation is scale‑invariant; after fixing
the L1‑norm to sqrt(2·n) the objective becomes the maximum entry of the
self‑convolution.  The algorithm works with a fixed norm and performs a
gradient‑descent on a smoothed version of the max (a weighted sum that puts
most weight on the current largest convolution entries).  The gradient is
computed efficiently with FFTs, the iterates are projected back onto the
non‑negative simplex, occasional random jitters are added to avoid plateaus,
and the best sequence seen during the allotted time is returned.
"""

import time
import numpy as np


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    """
    size = len(x) + len(y) - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, n=nfft)[:size]
    return conv


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative numbers (the heights of the step function)
    that gives a low value of evaluate_sequence.  The routine runs for
    ``budget_s`` seconds (minus a small safety margin) and keeps the best
    sequence encountered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Length of the sequence.
    # Use the length of the known construction if it exists, otherwise a
    # reasonable default.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # provided by the framework
    except Exception:
        n = 2000

    # Target L1‑norm.  With this norm the objective equals the maximal entry
    # of the self‑convolution.
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initial feasible point: exponential random values, then renormalised.
    # ------------------------------------------------------------------
    a = np.random.exponential(scale=1.0, size=n)
    a = a * (target_sum / a.sum())

    best_a = a.copy()
    best_score = np.max(_conv_full(a, a))
    print(f"[init] n={n}, initial max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    lambda_param = 5.0            # softness of the “max” weight
    step_factor = 0.5            # controls the absolute step size
    jitter_every = 70            # iterations between random jitters
    max_iter_without_improve = 0

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- current convolution and score -----
        b = _conv_full(a, a)
        cur_max = b.max()

        # keep the best ever seen
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            max_iter_without_improve = 0
            print(f"[iter {it:5d}] new best max={best_score:.6f}")
        else:
            max_iter_without_improve += 1

        # ----- construct a weight that concentrates on the largest entries -----
        # Using a soft‑max with a modest lambda avoids numerical underflow.
        # For large n the gaps between b‑values are huge, therefore we fall
        # back to a binary mask when the exponential collapses.
        w_exp = np.exp(lambda_param * (b - cur_max))   # max gets 1.0
        if np.isfinite(w_exp).all() and w_exp.sum() > 0:
            w = w_exp / w_exp.sum()
        else:
            # binary mask of entries that are (almost) maximal
            tol = max(1e-12, cur_max * 1e-4)
            mask = (b >= cur_max - tol).astype(float)
            w = mask / mask.sum()

        # ----- gradient of the (weighted) max -----
        a_rev = a[::-1]
        grad_full = _conv_full(w, a_rev)                # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]        # slice of length n
        norm_grad = np.linalg.norm(grad) + 1e-12

        # ----- take a step, project back onto non‑negative simplex -----
        step = step_factor * (target_sum / norm_grad)
        a_candidate = a - step * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s > 0:
            a_candidate *= target_sum / s
        else:
            # degenerate case – re‑initialise
            a_candidate = np.random.exponential(scale=1.0, size=n)
            a_candidate *= target_sum / a_candidate.sum()

        # ----- occasional jitter to escape flat plateaus -----
        if it % jitter_every == 0:
            idx = np.random.randint(n)
            a_candidate[idx] += np.random.randn() * 0.02 * a_candidate.mean()
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate *= target_sum / a_candidate.sum()

        # ----- accept the candidate only if it does not increase the max -----
        b_candidate = _conv_full(a_candidate, a_candidate)
        max_candidate = b_candidate.max()
        if max_candidate <= cur_max:
            a = a_candidate
            # slowly increase the focus on the true max
            lambda_param = min(50.0, lambda_param * 1.01)
        else:
            # reject; shrink the step size a bit
            step_factor = max(0.1, step_factor * 0.7)

        # occasional aggressive reduction of step_factor if we stall
        if max_iter_without_improve > 400:
            step_factor = max(0.05, step_factor * 0.5)
            max_iter_without_improve = 0

    # ------------------------------------------------------------------
    # Final sanity check (optional, uses the supplied evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_a.tolist())
        print(f"[done] best value reported by evaluate_sequence = {final_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Could not call evaluate_sequence:", exc)

    return best_a.tolist()
