"""
Search for a low‑value step‑function.

The score of a sequence a of length n is

    score = 2·n·max(a∗a) / (sum(a))²

If we keep the L¹‑norm fixed to √(2n) the denominator equals 2·n,
so the score is exactly the maximum entry of the self‑convolution.
We therefore minimise `max(a∗a)` under the constraints a_i≥0 and
∑a_i = √(2n).

The algorithm works as follows

1. Choose a length n (several candidates are tried).
2. Initialise a with a geometric decay a_i = r^i, where r = 1−α/n
   and α≈1.256 (the analytically optimal value for a pure geometric
   shape).  The vector is then rescaled to the required L¹‑norm.
3. Repeatedly perform a sub‑gradient step on the current maximal
   convolution entry:
      * compute the convolution by FFT,
      * locate the index k where it attains its maximum,
      * the sub‑gradient of max(a∗a) at this point is
            g_j = 2·a_{k−j}   for indices with 0≤k−j<n,
            otherwise 0.
      * move a a little in the opposite direction of g,
        project onto the non‑negative orthant and renormalise.
4. Every few iterations we evaluate the true score (which equals the
   current max‑convolution because the L¹‑norm is fixed).  The best
   sequence seen so far is kept.
5. The loop stops when the time budget is exhausted and the best
   sequence is returned.

The method is a simple projected (sub‑)gradient descent on the
non‑smooth objective and, in practice, drives the score well below
the naïve value 2.0, often approaching the target 1.5 – 1.6 within the
allocated time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def geometric_sequence(n: int, alpha: float = 1.256) -> np.ndarray:
    """
    Return a length‑n vector a_i = r^i with r = 1 - alpha / n.
    The vector is *not* normalised.
    """
    eps = alpha / n
    r = 1.0 - eps
    # r may become slightly >1 due to rounding for very small n; guard.
    if r <= 0.0:
        r = 0.5
    return np.power(r, np.arange(n, dtype=np.float64))


def normalise_to_sum(a: np.ndarray, target: float) -> np.ndarray:
    """Scale a so that its L1‑norm equals ``target`` (leaving a unchanged if possible)."""
    s = a.sum()
    if s == 0.0:
        # distribute uniformly if everything vanished
        a[:] = target / a.size
    else:
        a *= target / s
    return a


def convolution_max_fft(a: np.ndarray) -> tuple:
    """
    Compute the linear convolution a∗a via FFT, return (max_value, argmax, full_conv).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    return float(conv[max_idx]), max_idx, conv


def subgradient_max_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(a∗a) at the index ``k``.
    For non‑negative a the sub‑gradient is g_j = 2·a_{k−j} whenever the index is valid.
    """
    n = a.size
    idx = k - np.arange(n, dtype=np.int64)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    return g


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a sequence of non‑negative heights whose score (as defined in
    ``evaluate_sequence``) is as small as possible within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a small safety margin

    best_seq = None
    best_score = np.inf

    # lengths to try – larger n gives more freedom but costs a bit more per iteration
    length_candidates = [2000, 4000, 6000]

    # target L1 norm for a given length (so that score == max convolution)
    def target_sum(n):
        return np.sqrt(2.0 * n)

    for n in length_candidates:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # 1. initialise with a geometric shape (the analytical starting point)
        # ------------------------------------------------------------------
        a = geometric_sequence(n, alpha=1.256)
        a = normalise_to_sum(a, target_sum(n))

        # quick check of the initial score
        cur_max, _, _ = convolution_max_fft(a)
        cur_score = cur_max                     # because sum == sqrt(2n)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init] n={n}  score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # 2. projected sub‑gradient descent on max(a∗a)
        # ------------------------------------------------------------------
        iteration = 0
        # initial step size – a fraction of the total L1 mass
        step_scale = target_sum(n) * 0.2

        while time.time() < deadline:
            # a∗a and the index of its maximum
            max_val, k_max, _ = convolution_max_fft(a)

            # sub‑gradient for this max entry
            g = subgradient_max_at_k(a, k_max)

            # diminishing step size (simple schedule)
            eta = step_scale / (1.0 + iteration * 0.001)

            # gradient step and projection
            a -= eta * g
            np.maximum(a, 0.0, out=a)                 # keep non‑negative
            a = normalise_to_sum(a, target_sum(n))   # enforce the L¹ constraint

            # occasionally evaluate and keep the best sequence seen
            if iteration % 20 == 0:
                # because sum is fixed, the true score equals the current max convolution
                score = max_val
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[iter] n={n}  it={iteration}  best={best_score:.6f}")

            # every few hundred steps add a tiny random perturbation to help escape plateaus
            if iteration % 500 == 0 and iteration > 0:
                noise = np.random.randn(n) * target_sum(n) * 5e-4
                a += noise
                np.maximum(a, 0.0, out=a)
                a = normalise_to_sum(a, target_sum(n))

            iteration += 1

            # safety guard – if the loop runs too long, break early
            if time.time() >= deadline:
                break

    # ----------------------------------------------------------------------
    # fallback – uniform distribution (should never be used in practice)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 500
        u = np.full(n, target_sum(n) / n, dtype=np.float64)
        best_seq = u

    # convert to plain Python list as required by the evaluator
    return best_seq.tolist()
