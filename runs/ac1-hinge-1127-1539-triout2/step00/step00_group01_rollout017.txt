def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # little safety

    # Determine length n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 7390

    # Initialize sequence: sqrt differences
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1) - np.sqrt(idx)    # shape length n
    # Normalize sum to sqrt(2n)
    target_sum = math.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())

    best_seq = a.copy()
    # compute initial score using evaluate_sequence
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    print(f"Initial score: {best_score}")

    # parameters for gradient descent
    iteration = 0
    while time.time() < deadline:
        # compute convolution via FFT
        L = 1 << ((2*n-1).bit_length())
        # compute real FFT
        fft_a = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fft_a * fft_a, n=L)[:2*n-1]
        max_b = conv.max()
        # evaluation (since sum normalized)
        curr_score = max_b
        if curr_score < best_score:
            best_score = curr_score
            best_seq = a.copy()
            print(f"Iter {iteration}: improved score {best_score:.6f}")
        # Determine indices near max (within tolerance)
        tol = 1e-8 * max_b  # relative tolerance
        ks = np.where(conv >= max_b - tol)[0]
        # compute subgradient sum over ks
        grad = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = k - idx_i
            grad[idx_i] += 2.0 * a[j]
        # learning rate schedule
        # base lr = 0.5 / (iteration+1) maybe.
        lr = 0.1 / (np.sqrt(iteration + 1))
        # update
        a = a - lr * grad
        # projection: nonnegative
        a = np.maximum(a, 0.0)
        # renormalize sum to target_sum
        s = a.sum()
        if s == 0:
            # refill small uniform to avoid zero
            a = np.full(n, target_sum / n)
        else:
            a *= (target_sum / s)
        # optional random jitter
        if iteration % 100 == 0 and iteration > 0:
            # add small random noise to escape
            noise = np.random.randn(n) * (target_sum * 0.001 / np.sqrt(n))
            a = np.maximum(a + noise, 0.0)
            a *= (target_sum / a.sum())
        iteration += 1

    print(f"Finished after {iteration} iterations, best score {best_score}")
    return best_seq.tolist()

```python
import numpy as np
import time
import math

def propose_candidate(seed=42, budget_s=1000):
    """Generate a step function sequence that aims to minimise the evaluation.
    Starts from the discrete derivative of sqrt (which gives a good
    theoretical bound ≈π/2) and then performs a projected sub‑gradient
    descent on the max‑convolution value while keeping the L1 norm fixed to
    sqrt(2·n).  The routine stops after the given time budget and returns the
    best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # keep a second margin

    # length of the sequence – we take a moderate size to keep LP/FFT cheap
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000
    if n < 500:          # ensure we have enough entries to beat the trivial bound
        n = 500

    # ----- initialise with the derivative of sqrt (known to give ≈π/2) -----
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1.0) - np.sqrt(idx)          # a_i = √(i+1)−√i
    target_sum = math.sqrt(2.0 * n)
    a *= target_sum / a.sum()                      # scale so that Σa_i = √(2n)

    # keep a copy of the best sequence seen so far
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    print(f"Initial score: {best_score:.6f}")

    # pre‑compute FFT size (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    iteration = 0
    while time.time() < deadline:
        # --- compute convolution using FFT (fast) ---
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2 * n - 1]
        max_b = conv.max()
        # because Σa = √(2n) the evaluation equals max_b
        current_score = float(max_b)

        # keep the best sequence
        if current_score < best_score:
            best_score = current_score
            best_seq = a.copy()
            print(f"Iter {iteration:5d}: new best {best_score:.6f}")

        # --- sub‑gradient of the max‑convolution ---
        # indices where the convolution is (almost) maximal
        eps = 1e-12 * max_b
        ks = np.where(conv >= max_b - eps)[0]
        grad = np.zeros_like(a)

        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = k - i
            grad[i] += 2.0 * a[j]

        # step size (decaying)
        lr = 0.15 / math.sqrt(iteration + 1.0)
        a = a - lr * grad

        # projection onto the feasible set: non‑negative & L1 = √(2n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:                # avoid division by zero – fall back to uniform
            a = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a *= (target_sum / s)

        # occasional small random jitter to escape flat regions
        if iteration % 100 == 0 and iteration > 0:
            noise = np.random.randn(n) * (target_sum * 1e-3 / math.sqrt(n))
            a = np.maximum(a + noise, 0.0)
            a *= (target_sum / a.sum())

        iteration += 1

    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    # convert to plain python list (evaluate_sequence expects list of floats)
    return best_seq.tolist()

"""
Optimisation of a non‑negative step function for the supplied evaluation.

The algorithm works in three stages:

1) **Initialisation** – we start from the discrete derivative of the
   square‑root function:
       a_i = sqrt(i+1) – sqrt(i)
   This sequence is known to give a self‑convolution whose maximum is
   ≈π/2 after normalising the L1‑norm to sqrt(2·n), i.e. a score around 1.57 –
   already significantly better than the naïve uniform construction (score 2).

2) **Projected sub‑gradient descent** – the score is the infinity‑norm of the
   self‑convolution.  At the current maximum shift `k*` the gradient of that
   maximum w.r.t. the coefficients is
       ∂max/∂a_i = 2·a_{k*‑i}   (when the index is valid)
   We move a small step opposite to that gradient, project onto the feasible
   set (non‑negative, sum fixed to sqrt(2·n)) and repeat.  The step size
   decays as 1/√iteration.

3) **Exploratory jitter** – every 100 iterations a tiny random perturbation
   (zero‑mean Gaussian) is added and the sequence is re‑normalised.  This
   helps to escape flat regions of the objective.

All heavy work (convolution, gradient) is performed with FFTs, so the
routine easily stays well within the 1000 s time budget even for a few
thousand coefficients.  The best sequence found is returned as a plain
Python list; the surrounding infrastructure will call `evaluate_sequence`
to obtain the final score.
"""

import time
import math
import numpy as np

def _conv_fft(a, L):
    """Self‑convolution of a real vector using a power‑of‑two FFT length."""
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]


def _add_gradient_for_k(grad, a, k):
    """Add the sub‑gradient contribution for shift k to array `grad`."""
    n = a.size
    if k < n:
        # indices 0 … k
        grad[: k + 1] += 2.0 * a[k :: -1]
    else:
        # indices k‑(n‑1) … n‑1
        start = k - n + 1
        length = 2 * n - 1 - k
        grad[start:] += 2.0 * a[: length][::-1]


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) minimises
    the provided `evaluate_sequence` function.

    Parameters
    ----------
    seed : int
        Random seed – makes the search deterministic.
    budget_s : int
        Time budget in seconds.  The function stops slightly before this
        limit to avoid being killed.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0          # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length.
    # If a previous construction is available we reuse its length,
    # otherwise we take a modest size that keeps the linear‑programming
    # alternative cheap (but we do not need the LP here).
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000                             # a comfortable default

    # ------------------------------------------------------------------
    # 1) Initialise with the sqrt‑difference sequence.
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1.0) - np.sqrt(idx)          # a_i = √(i+1) – √i
    target_sum = math.sqrt(2.0 * n)                # required L1‑norm
    a *= target_sum / a.sum()                      # normalise to √(2n)

    best_seq = a.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))  # type: ignore[name-defined]
    except Exception:
        best_score = float("inf")
    print(f"Initial score: {best_score:.6f}")

    # Pre‑compute FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    iteration = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Compute the self‑convolution and its maximum.
        # --------------------------------------------------------------
        conv = _conv_fft(a, L)
        max_b = conv.max()
        # Because we keep Σa = √(2n) the evaluation equals max_b.
        cur_score = float(max_b)

        # Keep the best ever seen.
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:5d} – new best {best_score:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient step for the active shift(s).
        # --------------------------------------------------------------
        eps = 1e-12 * max_b
        ks = np.where(conv >= max_b - eps)[0]       # usually a single index
        grad = np.zeros_like(a)

        for k in ks:
            _add_gradient_for_k(grad, a, k)

        # Decaying step size – 0.15 is a safe starting magnitude.
        lr = 0.15 / math.sqrt(iteration + 1.0)
        a -= lr * grad

        # Projection onto the feasible set: non‑negative and fixed L1‑norm.
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:                               # degenerate – fall back to uniform
            a = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a *= (target_sum / s)

        # --------------------------------------------------------------
        # Small random jitter every 100 iterations (helps to avoid local plateaus).
        # --------------------------------------------------------------
        if iteration % 100 == 0 and iteration > 0:
            # Gaussian noise with very small standard deviation.
            sigma = target_sum * 1e-3 / math.sqrt(n)
            noise = np.random.randn(n) * sigma
            a = np.maximum(a + noise, 0.0)
            a *= (target_sum / a.sum())

        iteration += 1

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
