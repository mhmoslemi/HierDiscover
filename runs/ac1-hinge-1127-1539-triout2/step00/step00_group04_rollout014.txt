def get_direction(f):

    sum_f = np.sum(f)
    # scale f to sqrt(2n) sum
    scale = np.sqrt(2*n) / sum_f if sum_f > 0 else 1.0
    f_norm = f * scale
    M = np.max(np.convolve(f_norm, f_norm))

    if g is None:
        return None
    # g sum may be small; normalize to sqrt(2n)
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    g_norm = g * (np.sqrt(2*n) / sum_g)
    return g_norm

"""
Propose a step‐function (list of non‑negative heights) that minimizes
evaluate_sequence.  The algorithm starts from a few simple seeds (including
the known optimal 2‑point profile) and then performs a Frank–Wolfe style
iteration:

* For the current sequence f we normalise it so that  Σf = sqrt(2·n)
  (the evaluation function is homogeneous, this normalisation is harmless).
* We solve a linear program that finds a direction g with a *larger* L1‑norm
  while keeping the convolution bound f∗g ≤ max(f∗f).  This is exactly the
  LP used by the reference implementation.
* Instead of a fixed tiny step we try several step sizes {0.01,…,0.6} and
  keep the one that yields the lowest evaluation.
* If the LP fails we perform a small random perturbation.
* Occasionally we restart from a fresh random vector (different length) to
  avoid stagnation.
* The best sequence seen so far is always remembered and returned at the
  end of the allotted budget.

The routine runs until the provided time budget (default 1000 s) is
exhausted and finally returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programme: maximise sum(g)   s.t.   (f * g)(k) ≤ rhs   for all k,
# g ≥ 0.  The programme is linear in g because f is fixed.
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """Return a feasible direction g (as a NumPy array) or None."""
    n = len(f_seq)
    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n)

    # Build the convolution constraints (2n‑1 of them)
    a_rows = []
    b_vals = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # row[j] = Σ_i f_i  with i + j = k  →  f_i contributes to g_j
        # i runs over all indices, j = k - i must be in range
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] += f_seq[i]
        a_rows.append(row)
        b_vals.append(rhs)

    # Non‑negativity constraints:  -g ≤ 0
    a_rows.append(-np.eye(n))
    b_vals.append(np.zeros(n))

    A_ub = np.vstack(a_rows)
    b_ub = np.concatenate(b_vals)

    # Use HiGHS – fast and robust for moderate sizes
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if res.success:
        # Numerical noise can produce tiny negatives – clip them.
        return np.clip(res.x, 0.0, None)
    return None


# ----------------------------------------------------------------------
# Obtain a feasible improving direction g for the current sequence f
# ----------------------------------------------------------------------
def get_improving_direction(f: np.ndarray):
    """
    Solve the LP described in the paper and return a normalised direction g.
    Both f and the returned g have Σ = sqrt(2·n).
    """
    n = len(f)
    sum_f = np.sum(f)
    if sum_f == 0.0:
        return None
    # Normalise to the standard sum √(2n)
    scale_f = np.sqrt(2.0 * n) / sum_f
    f_norm = f * scale_f

    # Upper bound on the convolution (the active maximum)
    rhs = np.max(np.convolve(f_norm, f_norm))

    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    sum_g = np.sum(g)
    if sum_g == 0.0:
        return None
    # Normalise g to the same Σ = √(2n)
    scale_g = np.sqrt(2.0 * n) / sum_g
    return g * scale_g


# ----------------------------------------------------------------------
# Helper – normalise a vector to a prescribed L1‑norm
# ----------------------------------------------------------------------
def normalise_to(vec, target_sum):
    """Scale vec so that its L1‑norm equals target_sum (preserves sign)."""
    s = np.sum(vec)
    if s == 0.0:
        # fallback: uniform distribution
        return np.full_like(vec, target_sum / len(vec))
    return vec * (target_sum / s)
