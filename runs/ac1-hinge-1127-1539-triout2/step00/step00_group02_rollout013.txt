    # w = (conv - t) * (conv > t)   # positive part
    w = np.maximum(conv - t, 0.0)
    # gradient g_i = 4 * sum_j a_j * w_{i + j}

def proj_simplex(v):
    # ensure non-negative sum to 1
    v = np.maximum(v, 0)
    if v.sum() == 0:
        # set uniform
        return np.full_like(v, 1.0 / len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence that lowers the value of

    2 * n * max(conv(a, a)) / (sum(a)**2)

by flattening the self‑convolution.  The algorithm works directly on a
probability vector `a` (sum = 1).  Starting from a slightly perturbed
uniform distribution it repeatedly

1. computes the full convolution via FFT,
2. builds a loss that penalises values of the convolution that lie above
   its average (the theoretical lower bound `1/(2n‑1)`),
3. takes a projected gradient step on that loss (gradient is obtained by
   another FFT‑based convolution), and
4. projects back onto the probability simplex.

The loss pushes mass from indices that contribute heavily to the current
maximum towards indices that are under‑represented, driving the
convolution towards a flat distribution.  Simple simulated‑annealing
style random restarts keep the search from stagnating.

The function runs for the allotted time (default 1000 s) and returns the
best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[: 2 * n - 1]
    return conv


def _gradient_of_flatness(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the loss  Σ_{k} max(0, conv[k] - avg)^2
    with respect to a (the probability vector).

    The derivative is 4 * sum_{k} (conv[k] - avg)_+ * a_{k-i}
    which is a cross‑correlation and can be computed with FFT.
    """
    n = a.shape[0]
    # average value of the convolution (for a probability vector sum=1)
    avg = 1.0 / (2 * n - 1)

    # positive part of (conv - avg)
    w = np.maximum(conv - avg, 0.0)

    # cross‑correlation a_rev * w  (see analysis)
    a_rev = a[::-1]
    size = _next_pow_two(len(w) + len(a_rev) - 1)
    A_rev = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(A_rev * W, n=size)  # length = 3n‑2
    start = n - 1
    g = corr[start : start + n]               # length n
    return 4.0 * g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >=0, sum x = 1}."""
    # Based on the algorithm of Duchi et al. (ICML 2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w = w / max(w.sum(), 1e-12)
    return w
