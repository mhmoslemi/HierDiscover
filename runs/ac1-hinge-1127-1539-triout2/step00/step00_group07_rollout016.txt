```
def project_onto_simplex(v):
    # returns projection to simplex sum(v)=1, v>=0
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:

    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)

def subgradient_step(a, lr):

    max_idx = int(np.argmax(conv))
    # compute gradient g_i = 2 * a[max_idx - i] if index valid, else 0

"""
Propose a sequence of non‑negative heights that minimizes the provided evaluation
function.  The algorithm is a projected sub‑gradient descent on the quantity
max(conv(f,f)), where f is forced to stay on the probability simplex
(∑f_i = 1, f_i ≥ 0).  Because the evaluation is scale‑invariant we may keep the
sum fixed to one.  At each iteration the index k* attaining the maximal
convolution value is identified; the sub‑gradient of conv_k* with respect to f
is simply 2·f_{k*‑i} (when the index is in range).  A step opposite to this
gradient lowers the maximal entry of the convolution.  After the step the
vector is projected back onto the simplex.  Several random restarts with
different Dirichlet initialisations are performed while time permits; the
best sequence encountered is returned.
"""

import time
import numpy as np

def _project_simplex(v):
    """Projection of a vector onto the probability simplex (sum==1, v>=0)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:        # all entries are zero – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a):
    """Linear convolution of a with itself using FFT (real input)."""
    n = a.shape[0]
    size = 2 * n - 1
    # rfft works with real data and is faster than full fft
    fa = np.fft.rfft(a, n=size)
    return np.fft.irfft(fa * fa, n=size)


def propose_candidate(seed=0, budget_s=1000):
    """
    Return a list of non‑negative floats that (empirically) gives a low value
    for evaluate_sequence.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    best_seq = None
    best_score = float('inf')

    # length of the step function – larger values give a better theoretical bound
    n = 2000

    restart = 0
    while time.time() < deadline:
        # ----- random initialisation ----------------------------------------
        # Dirichlet with a random concentration gives diverse shapes
        alpha = np.random.uniform(0.2, 2.0)
        a = np.random.dirichlet(np.full(n, alpha))
        # occasional uniform start (helps escaping pathological rests)
        if np.random.rand() < 0.05:
            a = np.full(n, 1.0 / n)

        # ----- sub‑gradient descent -----------------------------------------
        max_iters = 3000                     # how many inner steps per restart
        lr = 0.5                              # initial step size
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # convolution and the current maximal entry
            conv = _conv_self(a)
            k_star = int(np.argmax(conv))
            max_val = conv[k_star]

            # current evaluation (sum(a)=1 → 2·n·max_val)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] restart {restart} it {it}  "
                      f"new best = {best_score:.6f}")

            # sub‑gradient of the active convolution entry
            # g_i = 2 * a_{k* - i} wherever the index is valid
            g = np.zeros_like(a)
            i_low = max(0, k_star - (n - 1))
            i_high = min(n - 1, k_star)
            i_vals = np.arange(i_low, i_high + 1, dtype=int)
            j_vals = k_star - i_vals
            g[i_vals] = 2.0 * a[j_vals]

            # gradient step + projection onto simplex
            a = a - lr * g
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

            # simple learning‑rate schedule
            if (it + 1) % 200 == 0:
                lr *= 0.9

        restart += 1

    # fallback – should never happen, but keep code safe
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # convert to plain Python list (as required by the evaluator)
    return best_seq.tolist()
