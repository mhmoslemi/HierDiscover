def compute_score(a):
    n = len(a)
    # compute convolution via FFT
    m = 1 << ((2*n-1).bit_length())
    # real FFT
    fa = np.fft.rfft(a, n=m)

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, math

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Choose length
    n = 1527
    # Initialize with random positive values then normalize sum to 1
    a = np.random.rand(n)
    a = a / a.sum()   # sum = 1

    # compute initial score
    def score(arr):
        n = len(arr)
        size = 1 << ((2*n - 1).bit_length())
        fa = np.fft.rfft(arr, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
        maxb = conv.max()
        sum_a = arr.sum()
        if sum_a < 0.01:
            return np.inf
        return 2.0 * n * maxb / (sum_a * sum_a)

    cur_score = score(a)
    best_score = cur_score
    best_seq = a.copy()
    print(f"Initial score {cur_score:.6f}")

    # Parameters
    move_type_swap_prob = 0.5
    max_no_improve = 2000
    no_improve_counter = 0
    iteration = 0
    temperature_start = 0.01
    temperature_end = 0.0001

    while time.time() < deadline:
        iteration += 1
        # Temperature schedule linear decay
        elapsed = time.time() - start
        fraction = elapsed / (budget_s - 5.0)
        temperature = temperature_start * (1 - fraction) + temperature_end * fraction

        # Choose move
        if np.random.rand() < move_type_swap_prob:
            i, j = np.random.randint(n, size=2)
            # swap
            a[i], a[j] = a[j], a[i]
            moved = ('swap', i, j, None)
        else:
            i = np.random.randint(n)
            if a[i] == 0.0:
                continue
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            # transfer a fraction of a[i] to a[j]
            delta = a[i] * np.random.rand() * 0.2  # up to 20% of a[i]
            a[i] -= delta
            a[j] += delta
            moved = ('transfer', i, j, delta)

        new_score = score(a)

        # Acceptance
        if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / max(temperature, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                no_improve_counter = 0
                print(f"[{iteration}] Time {time.time() - start:.1f}s - New best {best_score:.6f}")
            else:
                no_improve_counter += 1
        else:
            # revert move
            typ = moved[0]
            if typ == 'swap':
                i, j = moved[1], moved[2]
                a[i], a[j] = a[j], a[i]
            else:
                i, j, delta = moved[1], moved[2], moved[3]
                a[i] += delta
                a[j] -= delta
            no_improve_counter += 1

        # occasional random restart if stuck
        if no_improve_counter >= max_no_improve:
            # reinitialize with small perturbation of best_seq
            a = best_seq.copy()
            # add small random noise
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(0.0, a + noise)
            a = a / a.sum()
            cur_score = score(a)
            no_improve_counter = 0
            print(f"Restarted after {iteration} iterations, cur_score {cur_score:.6f}")

    # Return best sequence as list of floats
    return best_seq.tolist()

import time
import math
import numpy as np

# ------------------------------------------------------------
#  Helper functions
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of a with itself using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power‑of‑2
    fa = np.fft.rfft(a, n=size)                     # FFT of zero‑padded a
    conv = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]
    # small negative values can appear because of rounding – clip them
    return np.clip(conv, 0.0, None)


def _score(a: np.ndarray) -> float:
    """
    The objective used by the competition:
        2 * n * max(conv) / (sum(a) ** 2)
    """
    n = a.size
    conv = _conv_fft(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _score_and_conv(a: np.ndarray):
    """Return (score, full convolution) – useful for the targeted move."""
    n = a.size
    conv = _conv_fft(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf, conv
    return 2.0 * n * max_b / (sum_a * sum_a), conv


# ------------------------------------------------------------
#  Main search routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Stochastic hill‑climbing / simulated‑annealing search for a sequence
    that minimises ``evaluate_sequence``.
    The algorithm works directly on the ratio
        2·n·max(conv)/sum²
    (the same quantity that ``evaluate_sequence`` computes) and therefore
    avoids expensive repeated calls to the supplied function.

    It starts from a random non‑negative vector (normalised to sum = 1) and
    repeatedly performs cheap moves:
        * swap two entries,
        * transfer a small amount of mass from one entry to another,
        * a *targeted* transfer that reduces the contribution to the current
          maximal convolution index.
    Improvements are always accepted; worsening moves are accepted with a
    slowly cooling temperature to help escape shallow local minima.

    The best sequence found before the time budget expires is returned as a
    plain Python list (the caller will evaluate it with ``evaluate_sequence``).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # leave a small safety margin

    # -----------------------------------------------------------------
    #  Initialise a random probability vector (sum == 1)
    # -----------------------------------------------------------------
    n = 1527                                        # length used by the current SOTA
    a = np.random.rand(n)
    a = a / a.sum()                                 # normalise – sum = 1

    cur_score = _score(a)
    best_score = cur_score
    best_seq = a.copy()
    print(f"[init] score = {cur_score:.6f}")

    # -----------------------------------------------------------------
    #  Parameters for the stochastic search
    # -----------------------------------------------------------------
    move_swap_prob = 0.55           # probability of performing a swap move
    max_no_improve = 3000          # after this many non‑improving steps we re‑perturb
    no_improve_cnt = 0
    it = 0
    # temperature schedule (linear from start to end)
    T_start = 0.02
    T_end = 1e-4

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 5.0)
        temperature = T_start * (1.0 - frac) + T_end * frac

        # -------------------------------------------------------------
        #  Decide which move to perform
        # -------------------------------------------------------------
        if np.random.rand() < move_swap_prob:
            # ---- simple swap -------------------------------------------------
            i, j = np.random.randint(n, size=2)
            a[i], a[j] = a[j], a[i]
            moved = ("swap", i, j, None)
        else:
            # ---- either a random transfer or a *targeted* transfer ----------
            if np.random.rand() < 0.7:
                # random transfer
                i = np.random.randint(n)
                if a[i] == 0.0:
                    continue
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * np.random.rand() * 0.2   # move up to 20 % of donor
                a[i] -= delta
                a[j] += delta
                moved = ("transfer", i, j, delta)
            else:
                # targeted transfer – reduces the biggest contribution to the
                # current maximal convolution index
                cur, conv = _score_and_conv(a)
                k_max = int(conv.argmax())
                # list of valid indices for which i + partner = k_max
                idx = np.arange(n)
                partner = k_max - idx
                mask = (partner >= 0) & (partner < n)
                # contribution of each ordered pair to the max entry
                contrib = a[idx[mask]] * a[partner[mask]]
                if contrib.size == 0:
                    # fallback to a random transfer
                    i = np.random.randint(n)
                    if a[i] == 0.0:
                        continue
                    j = np.random.randint(n - 1)
                    if j >= i:
                        j += 1
                    delta = a[i] * np.random.rand() * 0.2
                    a[i] -= delta
                    a[j] += delta
                    moved = ("transfer", i, j, delta)
                else:
                    # pick the pair with the largest product
                    flat_idx = np.argmax(contrib)
                    i1 = idx[mask][flat_idx]
                    i2 = partner[i1]
                    # move mass from the larger of the two to a random low‑weight index
                    donor = i1 if a[i1] > a[i2] else i2
                    receiver = np.random.randint(n - 1)
                    if receiver >= donor:
                        receiver += 1
                    delta = a[donor] * np.random.rand() * 0.15   # a little milder step
                    a[donor] -= delta
                    a[receiver] += delta
                    moved = ("transfer", donor, receiver, delta)

        # -------------------------------------------------------------
        #  Evaluate the new state
        # -------------------------------------------------------------
        new_score = _score(a)

        # -------------------------------------------------------------
        #  Acceptance rule (simulated annealing style)
        # -------------------------------------------------------------
        if new_score < cur_score or np.random.rand() < math.exp(
            (cur_score - new_score) / max(temperature, 1e-12)
        ):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                no_improve_cnt = 0
                print(
                    f"[{it:>6}] t={elapsed:5.1f}s  new best = {best_score:.6f}"
                )
            else:
                no_improve_cnt += 1
        else:
            # reject – revert the move
            mtype, i, j, delta = moved
            if mtype == "swap":
                a[i], a[j] = a[j], a[i]
            else:  # transfer
                a[i] += delta
                a[j] -= delta
            no_improve_cnt += 1

        # -------------------------------------------------------------
        #  Occasionally re‑perturb if we have been stuck for a while
        # -------------------------------------------------------------
        if no_improve_cnt >= max_no_improve:
            # start again from a small random perturbation of the best sequence
            a = best_seq.copy()
            a += np.random.normal(scale=0.02, size=n)   # tiny Gaussian noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()                                 # renormalise
            cur_score = _score(a)
            no_improve_cnt = 0
            print(
                f"[{it:>6}] re‑perturb (time {elapsed:5.1f}s)  cur = {cur_score:.6f}"
            )

    # -----------------------------------------------------------------
    #  Return the best sequence found (as plain Python list)
    # -----------------------------------------------------------------
    print(f"Finished – best score {best_score:.6f} after {it} iterations")
    return best_seq.tolist()
