"""
Propose a non‑negative sequence that minimizes

    2 * n * max(conv(f, f)) / (sum(f) ** 2)

The function is scale‑invariant, therefore we keep the L1‑norm fixed
to the natural value `sqrt(2*n)`.  The algorithm tries to make the
auto‑convolution as flat as possible:

1. Start from a random (or the provided) sequence and normalise the
   sum to `sqrt(2*n)`.
2. At each iteration compute the convolution `b = f * f`.
3. Define the residual `r = b - mean(b)`.  Minimising the squared
   residual drives the convolution towards a constant, which in turn
   reduces its maximum.
4. The gradient of ½‖r‖² w.r.t. `f` is

        grad = 2 * (f * r)[n-1 : 2*n-1]

   (the slice of the full convolution of `f` and `r`).  Perform a
   small projected gradient step, clip negatives to zero and renormalise
   the sum.
5. Occasionally add a tiny random jitter to avoid being trapped in a
   shallow local minimum and decay the step size.
6. Keep the best sequence seen so far (according to the true
   `evaluate_sequence` objective) and return it when the time budget
   expires.

The whole loop runs until the allotted time (default 1000 s) is almost
exhausted.  No external files or network access are used; only NumPy
and the already‑imported `evaluate_sequence` are required.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions ------------------------------------------------------
# ----------------------------------------------------------------------


def _convolution_max(seq: np.ndarray) -> float:
    """Return the maximum value of the linear convolution of `seq` with itself."""
    return float(np.convolve(seq, seq).max())


def _gradient_of_flatness(seq: np.ndarray) -> np.ndarray:
    """
    Gradient of ½ * sum_k ( (seq*seq)_k - mean )²  w.r.t. `seq`.

    The formula is  grad_i = 2 * sum_j seq_j * (b_{i+j} - mean),
    where b = seq*seq.  This is obtained by a single convolution.
    """
    n = seq.shape[0]
    b = np.convolve(seq, seq)               # length 2n-1
    mean = b.mean()
    r = b - mean                            # residual, length 2n-1
    full = np.convolve(seq, r)              # length 3n-2
    # the needed slice corresponds to indices n-1 .. 2n-2 (inclusive)
    grad = 2.0 * full[n - 1 : 2 * n - 1]
    return grad


def _normalise_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale `seq` so that its L1‑norm equals `target_sum` (preserving sign)."""
    s = seq.sum()
    if s == 0.0:
        # re‑initialise to a tiny uniform vector to avoid division by zero
        seq = np.full_like(seq, target_sum / seq.size)
    else:
        seq = seq * (target_sum / s)
    return seq


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights (a step function) that tries to minimise
    the provided `evaluate_sequence` objective.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function stops a few tenths of a second
        before the limit to avoid being killed).
    kwargs
        Unused, present for compatibility with the surrounding framework.

    Returns
    -------
    List[float]
        The best sequence found during the allotted time.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Basic problem size – use the length of the known construction if
    # it exists, otherwise fall back to a moderate default.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)                 # e.g. 7442 in the benchmark
    else:
        n = 2000                                    # a safe fallback

    target_sum = np.sqrt(2.0 * n)                    # the natural normalisation

    # ------------------------------------------------------------------
    # Initialise a candidate.
    # 50 % chance to start from the supplied construction (if any),
    # otherwise start from a random positive vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and np.random.rand() < 0.5:
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)                      # ensure non‑negative
        a = _normalise_sum(a, target_sum)
    else:
        a = np.random.rand(n).astype(float)
        a = _normalise_sum(a, target_sum)

    # ------------------------------------------------------------------
    # Keep track of the best sequence seen.
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_score = _convolution_max(best_seq)

    # ------------------------------------------------------------------
    # Parameters for the projected gradient descent.
    # ------------------------------------------------------------------
    # Basic step size – a small fraction of the typical entry value.
    step = 0.02 * target_sum / n
    step_decay = 0.9995          # slowly shrink the step
    iteration = 0
    last_report = time.time()

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.2   # stop a bit early
    while time.time() < deadline:
        # 1) compute gradient that flattens the convolution
        grad = _gradient_of_flatness(a)

        # 2) gradient step with projection onto the non‑negative orthant
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)

        # 3) re‑normalise the L1 norm (scale invariance of the true objective)
        a_new = _normalise_sum(a_new, target_sum)

        a = a_new
        step *= step_decay
        iteration += 1

        # --------------------------------------------------------------
        # Periodically evaluate the true objective (using the fast max)
        # --------------------------------------------------------------
        if iteration % 10 == 0:
            cur_score = _convolution_max(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration:5d}] Improved max_b = {best_score:.6f}")

        # --------------------------------------------------------------
        # Small random jitter every few hundred iterations – helps escape
        # tiny local minima.
        # --------------------------------------------------------------
        if iteration % 250 == 0:

            num_perturb = max(1, n // 100)
            idx = np.random.choice(n, size=num_perturb, replace=False)
            noise = np.random.randn(num_perturb) * (0.005 * target_sum / n)
            a[idx] = np.maximum(a[idx] + noise, 0.0)
            a = _normalise_sum(a, target_sum)

        # --------------------------------------------------------------
        # Very occasional verbose progress report (does not affect timing)
        # --------------------------------------------------------------
        if time.time() - last_report > 5.0:
            print(f"Time elapsed {time.time() - (deadline - budget_s + 0.2):.1f}s – "
                  f"current best max_b = {best_score:.6f}")
            last_report = time.time()

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    return best_seq.tolist()
