def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')
    
    # Predefined lengths to try
    lengths = [500, 1000, 2000, 4000, 8000]
    # Maybe also try length of height_sequence_1 if defined
    if 'height_sequence_1' in globals():
        lengths.append(len(height_sequence_1))
    
    for n in lengths:
        # if time left insufficient break
        if time.time() >= deadline:
            break
        
        print(f"Starting optimization for length {n}")
        # initialize sequence: arcsine shape + small random noise
        i = np.arange(n)
        # arcsine shape: a_i ∝ 1/√((i+0.5)*(n-i-0.5))
        eps = 1e-12
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5) + eps)
        # Normalize sum to sqrt(2 * n) (the baseline sum)
        target_sum = np.sqrt(2 * n)
        a = a * (target_sum / a.sum())
        
        # Perform gradient descent
        beta = 20.0  # smooth max parameter
        eta0 = 0.5   # initial step size
        decay = 0.001
        
        max_iters = 5000
        for it in range(max_iters):
            if time.time() >= deadline:
                break
            # compute full convolution via FFT
            m = 1
            while m < 2 * n:
                m <<= 1
            A = np.fft.fft(a, n=m)
            conv = np.fft.ifft(A * A).real[:2*n-1]
            max_b = conv.max()
            
            # compute smooth weights
            # shift for numerical stability
            shift = conv.max()
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0:
                break
            w /= w_sum
            
            # compute G = sum_k w_k * a_{k - i}
            a_rev = a[::-1]
            # convolution of w and a_rev
            W = np.fft.fft(w, n=m)
            # we need FFT of a_rev padded to length m
            A_rev = np.fft.fft(a_rev, n=m)
            conv_wa_rev = np.fft.ifft(W * A_rev).real
            # conv_wa_rev length m, we want indices [n-1 : n-1+n]
            start = n-1
            G = conv_wa_rev[start:start+n]
            # compute gradient
            sum_a = a.sum()
            grad = 2.0 * G - 2.0 / sum_a
            # step size
            eta = eta0 / (1.0 + decay * it)
            a_new = a - eta * grad
            # ensure nonnegativity
            a_new = np.maximum(a_new, 0.0)
            # optional renorm? maybe keep sum not too small.
            # If sum is too small (e.g., near zero) break
            if a_new.sum() < 1e-6:
                break
            a = a_new
            
            # occasionally evaluate actual score
            if (it % 50) == 0:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.tolist()
                    print(f"Iter {it}, n={n}, score={score:.6f}")
        
        # after inner loop, evaluate final
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"Finished length {n}, final score={score:.6f}")
    
    print(f"Best score found: {best_score}")
    return best_seq

"""
Propose a candidate step‑function sequence that improves the evaluation score.

The algorithm works as follows (all operations are vectorised with NumPy and
FFT based convolutions):

1.  For a given length `n` we start from a smooth “arcsine” shaped sequence
    a_i ∝ 1/√((i+½)(n‑i‑½)).  This distribution is known to give a rather flat
    self‑convolution, i.e. a low value of `max(a*a)`.

2.  The sequence is normalised so that   Σ a_i = √(2 n).  With this normalisation
    the evaluation function reduces to `score = max(a*a)`, therefore the task
    becomes “make the convolution as flat as possible”.

3.  We perform an iterative optimisation that minimizes a smoothed version of
    the max‑convolution while keeping the total sum (≈√(2 n)) fixed.
    * Let `b = a * a` (full linear convolution, computed with FFT).
    * Approximate the maximum of `b` by a soft‑max:
          w_k = exp(β·(b_k−max(b))) / Σ exp(·)      (β≈30).
      The weight vector `w` concentrates on the indices where `b` is large.
    * The gradient of the smooth maximum w.r.t. `a` is
          ∇_a  smooth_max(b) = 2·G ,   G_i = Σ_k w_k·a_{k−i}.
      The term `G` is a cross‑correlation and can also be computed by FFT.
    * We move the mass *away* from positions with large `G_i`:
          d = G − mean(G)                (zero‑mean direction)
          a ← a − η·d                     (η is a decreasing step size)
      Afterwards we clip to non‑negative values and renormalise to the target
      sum √(2 n).  This preserves the denominator of the score while reducing
      the numerator.

4.  Every few iterations a full call to `evaluate_sequence` is made to keep the
    best sequence found so far.  Random jitter is occasionally added to help
    escape shallow local minima.

5.  The whole process is repeated for several lengths (500 … 8000 and the
    length of the previously supplied `height_sequence_1`).  The best sequence
    discovered before the time budget expires is returned.

The method needs only the public `evaluate_sequence` function, NumPy and the
standard library.  No file I/O or external solvers are used, and the total
runtime respects the provided `budget_s` (≈1000 s)."""

import time
import math
import numpy as np

# Helper functions (top‑level – no closures)

def fft_convolution_full(x, y):
    """
    Full linear convolution of two real 1‑D arrays using FFT.
    Returns a real array of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    m = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.fft(x, n=m)
    Y = np.fft.fft(y, n=m)
    conv = np.fft.ifft(X * Y).real[:n]
    return conv


def self_convolution(a):
    """Return a * a (full convolution) using FFT."""
    return fft_convolution_full(a, a)


def cross_correlation_w_a(w, a_rev):
    """
    Compute G_i = Σ_k w_k · a_{k‑i}  for i=0…len(a)-1.
    `a_rev` should be the reversed version of the original a.
    """
    # length of w is 2·n‑1, length of a_rev is n
    n = len(a_rev)
    conv = fft_convolution_full(w, a_rev)          # length = 3·n‑2
    start = n - 1                                   # position of index 0
    return conv[start:start + n]                    # slice of length n


def softmax_weights(b, beta):
    """
    Compute a soft‑max weight vector that approximates the arg‑max of `b`.
    The shift by max(b) prevents overflow.
    """
    shift = b.max()
    exp_vals = np.exp(beta * (b - shift))
    s = exp_vals.sum()
    if s == 0.0:
        # degenerate case – fall back to uniform weights
        return np.full_like(b, 1.0 / b.size)
    return exp_vals / s
