"""
Propose a low‑evaluation step‑function using a simple parametric family.

The sequence is generated from two parameters:
    p      – the power‑law exponent (0 < p < 1)
    gamma  – a small offset that suppresses the large first coefficient.

For i = 0 … n‑1 we set
    a_i = ((i + gamma + 1) ** p - (i + gamma) ** p)

and then normalise the whole vector so that its L1‑norm equals 1.
For p ≈ 0.4 the theoretical asymptotic value of the evaluation function is

    2 * p**2 * B(p, p)  ≈ 1.35  (< 1.503)

Choosing a modest offset gamma (≈10) makes the first coefficient small enough
that the maximum of the self‑convolution is given by the interior plateau,
hence the evaluation is close to the theoretical value.
The routine evaluates a few (p, gamma) pairs (within the time budget) and
returns the best sequence found; a tiny local stochastic refinement is also
performed.

The algorithm is completely deterministic apart from the optional random
perturbations (seeded for reproducibility) and respects all constraints of
`evaluate_sequence`.
"""

import time
import numpy as np
import math

def _generate_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a normalised non‑negative sequence of length n."""
    i = np.arange(n, dtype=float)
    # power‑law differences
    a = (np.power(i + gamma + 1.0, p) - np.power(i + gamma, p))
    total = np.power(n + gamma, p) - np.power(gamma, p)
    # normalise to L1 = 1 (required by the evaluation function)
    a /= total
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function sequence with evaluation < 1.503.

    The search explores a small grid of (p, gamma) values on a moderate length
    vector (n = 2000) to keep the runtime tiny, then builds the final sequence
    with the same parameters on a larger length (n = 5675) – the same size as
    the previous best construction.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a safety margin

    # --------------------------------------------------------------
    # 1. Coarse grid search on a short vector (fast)
    # --------------------------------------------------------------
    n_small = 2000
    best_score = float('inf')
    best_params = (0.5, 0.0)      # (p, gamma) placeholder

    p_vals = np.linspace(0.30, 0.50, 21)       # 0.30 … 0.50
    gamma_vals = list(range(0, 101, 5))        # 0 … 100 step 5

    for p in p_vals:
        for gamma in gamma_vals:
            seq = _generate_sequence(n_small, p, float(gamma))
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:               # safety – should never happen
                continue
            if score < best_score:
                best_score = score
                best_params = (p, float(gamma))
                print(f"[grid] new best score={score:.6f} (p={p:.3f}, gamma={gamma})")
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2. Build the final (large) sequence using the best parameters.
    # --------------------------------------------------------------
    n_big = 5675                         # length of the final construction
    p_best, gamma_best = best_params
    seq_big = _generate_sequence(n_big, p_best, gamma_best)

    # one cheap evaluation to know where we are
    best_score = evaluate_sequence(seq_big.tolist())
    print(f"[final] initial score with p={p_best:.3f}, gamma={gamma_best:.1f} -> {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Very light stochastic refinement (preserve L1 = 1)
    # --------------------------------------------------------------
    seq = seq_big.copy()
    # pre‑compute cumulative sum for fast projection (Manhattan projection onto simplex)
    while time.time() < deadline:
        # pick two distinct indices and transfer a tiny amount delta
        i, j = np.random.choice(len(seq), 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-4      # signed tiny step
        if seq[i] + delta < 0 or seq[j] - delta < 0:
            continue
        cand = seq.copy()
        cand[i] += delta
        cand[j] -= delta
        # keep the L1‑norm exactly 1 (the transfer keeps the sum unchanged)
        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            seq = cand
            print(f"[refine] improved score -> {best_score:.6f}")

    return seq.tolist()
