"""
Simple linear step construction.

The sequence a_i = i (i = 0, 1, …, n‑1) has a closed‑form evaluation:

    eval = 2 n · max(a*a) / (sum a)^2 = (4/3)·(n‑2)/(n‑1)

For n = 1000 this gives ≈ 1.332, well below the target bound 1.503.
All entries are ≤ 999 < 1000, so the security caps do not alter the shape.
The function returns this sequence (or the previously known best one if it is
already better).
"""

import time

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a step function that beats the current bound.

    A linear increasing sequence (a_i = i) gives an evaluation of about 1.33.
    The length is chosen as 1000 so that the largest element is 999, safely
    below the 1000‑value cap enforced by `evaluate_sequence`.
    """
    # length that respects the value cap (max element = 999)
    n = 1000

    # linear sequence a_i = i, as floats
    linear_seq = [float(i) for i in range(n)]

    # evaluate our candidate (only for logging, not required for correctness)
    try:
        linear_score = evaluate_sequence(linear_seq)
    except Exception as exc:          # pragma: no cover – safety net
        print("Error evaluating linear candidate:", exc)
        linear_score = float("inf")

    # try the previously found sequence, if it exists
    best_seq = linear_seq
    best_score = linear_score

    # `height_sequence_1` is assumed to be a global variable if present
    try:
        prev_seq = list(height_sequence_1)          # copy in case it is mutable
        prev_score = evaluate_sequence(prev_seq)
        if prev_score < best_score:
            best_seq = prev_seq
            best_score = prev_score
            print(f"Previous sequence is better: {prev_score:.6f}")
        else:
            print(f"Linear sequence is better: {linear_score:.6f}")
    except Exception:
        # No previous sequence available – just keep the linear one
        print(f"Using linear sequence only (score {linear_score:.6f})")

    print(f"Returning sequence of length {len(best_seq)} with score {best_score:.6f}")
    return best_seq
