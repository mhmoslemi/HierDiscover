def propose_candidate(seed=42, budget_s=1000):
    # Setup
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1

    # starting sequence: uniform of length n (choose n maybe 100)
    n = 200  # Choose moderate length
    f = np.full(n, 1.0 / n)  # sum = 1

    # compute initial cost
    best_f = f.copy()
    best_cost = compute_cost(f)
    print(f"Initial cost {best_cost}")

    while time.time() < deadline:
        # compute M = max conv
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # solve LP for g
        g = solve_convolution_lp(f, M)
        if g is None:
            # random perturbation
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn()*0.01)
            f = np.maximum(f, 0)
            f /= f.sum()
            continue

        sum_f = f.sum()
        sum_g = g.sum()
        # compute convolution terms
        conv_g = np.convolve(g, g)
        conv_fg = np.convolve(f, g)

        a = conv_f - 2*conv_fg + conv_g  # coefficient a_k
        b = 2*(conv_fg - conv_f)         # coefficient b_k
        c = conv_f

        # search t in [0,1]
        t_vals = np.linspace(0.0, 1.0, 201)
        denom = ((1 - t_vals) * sum_f + t_vals * sum_g) ** 2
        M_t = np.max(c[:,None] + b[:,None]*t_vals + a[:,None]*t_vals**2, axis=0)
        cost_t = 2 * n * M_t / denom
        idx_min = np.argmin(cost_t)
        t_opt = t_vals[idx_min]
        # optional refinement
        if idx_min > 0 and idx_min < len(t_vals)-1:
            # refine by ternary search around idx_min +/- step
            left = max(0.0, t_vals[idx_min-1])
            right = min(1.0, t_vals[idx_min+1])
            for _ in range(3):
                t_mid1 = left + (right-left)/3
                t_mid2 = right - (right-left)/3
                M1 = np.max(c + b*t_mid1 + a*t_mid1**2)
                M2 = np.max(c + b*t_mid2 + a*t_mid2**2)
                cost1 = 2 * n * M1 / (( (1 - t_mid1)*sum_f + t_mid1*sum_g )**2)
                cost2 = 2 * n * M2 / (( (1 - t_mid2)*sum_f + t_mid2*sum_g )**2)
                if cost1 < cost2:
                    right = t_mid2
                else:
                    left = t_mid1
            t_opt = (left+right)/2

        new_f = (1 - t_opt) * f + t_opt * g
        # normalize to avoid floating errors
        new_f = np.maximum(new_f, 0.0)
        new_f_sum = new_f.sum()
        if new_f_sum < 1e-12:
            # fallback
            new_f = f
        else:
            new_f = new_f / new_f_sum  # keep sum =1 for consistency

        new_cost = compute_cost(new_f)
        if new_cost < best_cost:
            best_cost = new_cost
            best_f = new_f.copy()
            print(f"Iter {time.time():.2f}, improved cost {best_cost:.6f}")

        # accept update if improves or with probability
        if new_cost < compute_cost(f) or np.random.rand() < 0.1:

        # occasional random restart if stuck
        if np.random.rand() < 0.01:
            f = np.random.rand(n)
            f = f / f.sum()

def propose_candidate(seed=0, budget_s=1000):

    from scipy.optimize import linprog

    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    n = 200   # length of sequence
    target_sum = np.sqrt(2.0 * n)
    # start with uniform distribution
    f = np.full(n, target_sum / n)   # sum = target_sum

    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())

    while time.time() < deadline:
        # compute current M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Build LP matrices
        # coeff matrix for constraints conv(f, g) <= M
        # We'll build as list of rows.
        rows = []
        for k in range(2 * n - 1):
            row = np.zeros(n)
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            # j runs from low to high, i = k - j
            row[low:high + 1] = f[k - np.arange(low, high + 1)]
            rows.append(row)
        A_ub = np.vstack(rows)
        b_ub = np.full(2 * n - 1, M)

        # maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs')
        if res.success:
            g = res.x
            sum_g = g.sum()
            if sum_g == 0:
                # shouldn't happen, break
                break
            # scale g to same target sum
            g = g * (target_sum / sum_g)

            # Precompute convolutions needed for line search
            conv_g = np.convolve(g, g)
            conv_fg = np.convolve(f, g)

            # coefficients for quadratic in t
            a = conv_f - 2 * conv_fg + conv_g
            b = 2 * (conv_fg - conv_f)   # note: c = conv_f

            # line search for t in [0,1]
            t_vals = np.linspace(0.0, 1.0, 101)
            Mt = np.max(conv_f[:, None] + b[:, None] * t_vals + a[:, None] * t_vals ** 2, axis=0)
            idx = np.argmin(Mt)
            t_opt = t_vals[idx]
            # refine using golden-section
            left = max(0.0, t_opt - 0.05)
            right = min(1.0, t_opt + 0.05)
            for _ in range(10):
                t1 = left + (right - left) / 3
                t2 = right - (right - left) / 3
                M1 = np.max(conv_f + b * t1 + a * t1 ** 2)
                M2 = np.max(conv_f + b * t2 + a * t2 ** 2)
                if M1 < M2:

                else:
                    left = t1
            t_opt = (left + right) / 2

            # ensure non-negative and renormalize to target_sum (should stay)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new * (target_sum / f_new.sum())

            if cur_score < best_score:
                best_score = cur_score
                best_f = f_new.copy()

import numpy as np
import time
from scipy.optimize import linprog

def build_A_ub(f):
    n = len(f)
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # j from low..high, coefficient f[k-j]
        row[low:high + 1] = f[k - np.arange(low, high + 1)]
        rows.append(row)
    return np.vstack(rows)

def solve_convolution_lp(f, M):
    """
    Solve the LP:
        max   sum(g)
        s.t.  conv(f,g) <= M   (for all shifts)
              g >= 0
    Returns the optimal g (numpy array) or None on failure.
    """
    n = len(f)
    A = build_A_ub(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)  # maximize sum(g) -> minimize -sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def optimal_mix_t(f, g, target_sum):
    """
    Given two sequences f and g having the same L1 sum (target_sum),
    find t in [0,1] that minimizes the peak of the convolution of
    (1-t)*f + t*g.  Returns the mixed sequence and the optimal t.
    """
    n = len(f)
    conv_f = np.convolve(f, f)
    conv_g = np.convolve(g, g)
    conv_fg = np.convolve(f, g)

    a = conv_f - 2 * conv_fg + conv_g
    b = 2 * (conv_fg - conv_f)   # coefficient of t
    # c = conv_f (constant term)

    # coarse grid search
    t_grid = np.linspace(0.0, 1.0, 101)
    M_grid = np.max(conv_f[:, None] + b[:, None] * t_grid + a[:, None] * t_grid**2, axis=0)
    t_opt = t_grid[np.argmin(M_grid)]

    # refine by ternary search in a small neighbourhood
    left = max(0.0, t_opt - 0.05)
    right = min(1.0, t_opt + 0.05)
    for _ in range(12):
        t1 = left + (right - left) / 3.0
        t2 = right - (right - left) / 3.0
        M1 = np.max(conv_f + b * t1 + a * t1 * t1)
        M2 = np.max(conv_f + b * t2 + a * t2 * t2)
        if M1 < M2:
            right = t2
        else:
            left = t1
    t_opt = (left + right) / 2.0

    mixed = (1 - t_opt) * f + t_opt * g

"""
search_step_function.py

A fast iterative optimiser for the “step‑function” evaluation problem.
The algorithm works with a fixed length `n` (chosen to keep the linear
program reasonably cheap) and repeatedly:

1. Normalises the current sequence `f` so that its L1‑norm equals
   sqrt(2·n) (the same scaling used in the reference implementation).

2. Solves the linear programme
        maximise   Σ g_i
        subject to conv(f, g) ≤ M   (for every convolution shift)
                   g_i ≥ 0
   where `M = max(conv(f, f))`.  The LP is linear because `f` is fixed.

3. Scales the LP solution `g` back to the same L1‑norm as `f` and then
   performs a *line search* on the mixing parameter `t∈[0,1]` that
   minimises the peak of the convolution of the convex combination
        h(t) = (1‑t)·f + t·g .
   The line search uses a coarse grid followed by a short ternary‑search
   refinement, which is much cheaper than a full non‑linear optimiser.

4. Replaces `f` by the best mixture `h(t_opt)` and keeps the best
   sequence seen so far (according to `evaluate_sequence`).

The loop runs until the allotted time budget expires.  The code uses only
NumPy and SciPy (no file or network I/O) and prints occasional progress
messages.  The returned list of heights is guaranteed to be non‑negative
and to have an L1‑norm ≥ 0.01, so `evaluate_sequence` will accept it.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def build_A_ub(f):
    """
    Build the inequality matrix A for the LP   conv(f, g) ≤ M .
    Each row corresponds to one shift `k` (0 … 2·n‑2) and contains the
    coefficients of the variables `g_j` in the expression
        Σ_i f_i * g_{k-i}.
    """
    n = len(f)
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # indices j that appear in this shift
        # coefficient for g_j is f_{k-j}
        row[low:high + 1] = f[k - np.arange(low, high + 1)]
        rows.append(row)
    return np.vstack(rows)


def solve_convolution_lp(f, M):
    """
    Solve the linear programme

        maximise   Σ g_i
        subject to conv(f, g) ≤ M  (∀ shift)
                   g_i ≥ 0

    Returns the optimal `g` as a NumPy array, or ``None`` on failure.
    """
    n = len(f)
    A = build_A_ub(f)               # shape (2·n‑1, n)
    b = np.full(2 * n - 1, M)       # upper bounds for every shift
    c = -np.ones(n)                 # minimise –Σ g_i  →  maximise Σ g_i
    bounds = [(0, None)] * n        # g_i ≥ 0

    # ``highs`` is the default high‑performance interior‑point method
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def optimal_mix_t(f, g, target_sum):
    """
    Given two sequences `f` and `g` that both have L1‑norm ``target_sum``,
    find the mixing coefficient `t∈[0,1]` that minimises the peak value
    of the convolution of  h(t) = (1‑t)·f + t·g.
    Returns the mixed sequence and the optimal `t`.
    """
    # Pre‑compute the three convolutions that appear in the quadratic form
    conv_f = np.convolve(f, f)          # length 2·n‑1
    conv_g = np.convolve(g, g)
    conv_fg = np.convolve(f, g)

    # Coefficients of the quadratic in t for each shift:
    #   C(t) = conv_f + b·t + a·t²
    a = conv_f - 2 * conv_fg + conv_g
    b = 2 * (conv_fg - conv_f)   # linear term (c = conv_f is the constant)

    # Coarse grid search
    t_grid = np.linspace(0.0, 1.0, 101)
    M_grid = np.max(conv_f[:, None] + b[:, None] * t_grid + a[:, None] * t_grid**2,
                    axis=0)
    t_opt = t_grid[np.argmin(M_grid)]

    # Refine around the best grid point with a short ternary search
    left, right = max(0.0, t_opt - 0.05), min(1.0, t_opt + 0.05)
    for _ in range(12):
        t1 = left + (right - left) / 3.0
        t2 = right - (right - left) / 3.0
        M1 = np.max(conv_f + b * t1 + a * t1 * t1)
        M2 = np.max(conv_f + b * t2 + a * t2 * t2)
        if M1 < M2:
            right = t2
        else:
            left = t1
    t_opt = (left + right) / 2.0

    # Form the mixture and renormalise to keep the exact L1‑norm
    mixed = (1.0 - t_opt) * f + t_opt * g
    mixed = np.maximum(mixed, 0.0)
    mixed = mixed * (target_sum / mixed.sum())
    return mixed, t_opt
