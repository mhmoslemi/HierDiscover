"""
Propose a candidate step‑function (list of non‑negative heights) that tries to
minimise the evaluation score

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The score is scale‑invariant, therefore we work with probability vectors
(p_i ≥ 0, Σ p_i = 1) and minimise max(conv(p,p)).  The algorithm is:

1. Determine the length `n` (normally the length of the provided

2. Build a small portfolio of simple initial distributions:
      – uniform,
      – a family p_i ∝ (i·(n+1‑i))^{-s} for several exponents `s`

        arcsine distribution).

   becomes the starting point.

   * compute the convolution by FFT (O(n log n));
   * locate the index `k_max` where the convolution attains its maximum;
   * the indices `i` that contribute to this entry satisfy
       i ∈ [max(0, k_max‑(n‑1)), min(n‑1, k_max)];

     outside this range, move a small amount of mass `δ` from `i` to `j`,
     keep the total sum unchanged;
   * accept the move only if it strictly reduces the maximal convolution
     value (occasionally a worse move is accepted with a tiny probability
     to escape plateaus).
   * repeat until the allotted time runs out.
4. Return the best distribution discovered as a list of plain floats.

The whole procedure respects the 1000 s budget, uses only NumPy, and never
writes to the filesystem or accesses the network.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: maximal value of the self‑convolution of a non‑negative vector.
# The function works with a probability vector (sum = 1) – the scale does
# not matter for the final score.
def max_self_convolution(vec, nfft):
    """
    Return max(conv(vec, vec)) using an FFT of length `nfft`.
    `nfft` must be a power of two ≥ 2·len(vec)-1.
    """
    # real FFT → product → inverse FFT (real part)
    f_hat = np.fft.rfft(vec, n=nfft)
    conv_full = np.fft.irfft(f_hat * f_hat, n=nfft)
    # only the first 2·len(vec)-1 entries are the linear convolution
    conv = conv_full[: 2 * len(vec) - 1]
    # Numerical noise may produce tiny negatives – clip them.
    return float(conv.max().real)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a small evaluation score.
    The returned value is a list of floats (non‑negative heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # keep a small safety margin

    # ------------------------------------------------------------------
    # Determine the length of the sequence we shall work with.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        # fall back to a moderate size if the global is missing
        n = 1000

    # FFT length: next power of two ≥ 2·n‑1 (fast convolution)
    nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1) Build a few deterministic starting distributions.
    candidates = []

    # uniform distribution
    uniform = np.ones(n, dtype=np.float64) / n
    candidates.append(uniform)

    # inverse‑product power‑law family: p_i ∝ (i·(n+1‑i))^{-s}
    # i runs from 1 … n  (1‑based indexing makes the formula neat)
    i_vals = np.arange(1, n + 1, dtype=np.float64)
    prod = i_vals * (n + 1 - i_vals)          # product i·(n+1‑i)

    for s in np.linspace(0.2, 2.0, 10):       # explore several exponents
        vec = prod ** (-s)
        vec /= vec.sum()
        candidates.append(vec)

    # a few random Dirichlet draws (concentration < 1 gives more spread)
    for alpha in (0.3, 0.7, 1.5):
        rnd = np.random.gamma(alpha, 1.0, size=n)
        rnd /= rnd.sum()
        candidates.append(rnd)

    # ------------------------------------------------------------------
    # 2) Pick the best of the initial candidates.
    best_vec = None
    best_obj = np.inf
    for vec in candidates:
        obj = max_self_convolution(vec, nfft)
        if obj < best_obj:
            best_obj = obj
            best_vec = vec.copy()
    print(f"[init] best initial max‑conv = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3) Greedy local refinement with a tiny amount of simulated‑annealing
    #    flavour.  The move consists of transferring a small chunk δ from an
    #    index that participates in the current maximal convolution entry to
    #    an index that does **not** participate.
    temperature = best_obj * 0.01   # a tiny temperature for occasional uphill steps
    min_delta = 1e-6               # avoid zero moves
    max_delta_frac = 0.02          # at most 2 % of the source weight per move

    while time.time() < deadline:
        # Compute convolution and locate its maximal index
        conv_full = np.fft.irfft(
            np.fft.rfft(best_vec, n=nfft) * np.fft.rfft(best_vec, n=nfft),
            n=nfft
        )
        conv = conv_full[: 2 * n - 1]
        k_max = int(np.argmax(conv))
        max_val = float(conv[k_max])

        # Indices that contribute to this entry (i + j = k_max)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        contrib_idx = np.arange(i_low, i_high + 1)          # inclusive range

        # Choose a source index inside the contributing set, weighted by its mass.
        src_weights = best_vec[contrib_idx]
        if src_weights.sum() == 0.0:
            # All zero (should not happen) – pick a random source.
            src = np.random.choice(contrib_idx)
        else:
            src_probs = src_weights / src_weights.sum()
            src = np.random.choice(contrib_idx, p=src_probs)

        # Choose a destination index outside the contributing set.
        all_idx = np.arange(n)
        mask_out = np.ones(n, dtype=bool)
        mask_out[contrib_idx] = False
        outside = all_idx[mask_out]
        if outside.size == 0:
            dst = np.random.randint(n)
        else:
            dst = np.random.choice(outside)

        # Determine a transfer amount δ.
        delta = min(best_vec[src] * max_delta_frac, 0.01)
        delta = max(delta, min_delta)

        # If source is too small, skip this iteration.
        if delta > best_vec[src]:
            continue

        # Propose the new vector.
        new_vec = best_vec.copy()
        new_vec[src] -= delta
        new_vec[dst] += delta

        # Compute its maximal convolution value.
        new_obj = max_self_convolution(new_vec, nfft)

        # Greedy acceptance with a very small chance of uphill moves.
        if new_obj < best_obj:
            best_obj = new_obj
            best_vec = new_vec
            print(f"[improve] new best max‑conv = {best_obj:.6f} at t={time.time()-start_time:.1f}s")
        else:
            # Very occasional uphill acceptance to escape flat plateaus.
            if np.random.rand() < math.exp(-(new_obj - best_obj) / max(temperature, 1e-9)):
                best_vec = new_vec

        # Cool the tiny temperature slowly.
        temperature *= 0.9999
        # Safety abort if we somehow hit an infinite (should not happen).
        if not np.isfinite(best_obj):
            break

    # ------------------------------------------------------------------
    # Convert to plain Python list of floats (the evaluator will re‑scale as it likes).
    result = [float(v) for v in best_vec]

    # Final diagnostic (optional – the caller may ignore prints).
    try:
        # The problem statement guarantees that `evaluate_sequence` is already imported.
        final_score = evaluate_sequence(result)
        print(f"[final] evaluate_sequence = {final_score:.6f}")
    except Exception:
        pass

    return result
