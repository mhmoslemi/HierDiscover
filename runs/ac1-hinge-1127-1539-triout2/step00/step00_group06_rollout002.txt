def compute_score(seq):
    a = np.array(seq, dtype=np.float64)
    n = a.size
    # compute convolution using FFT (next power of two length)
    s = 2*n - 1
    m = 1 << (s-1).bit_length()
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:s]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a**2)

def gradient_step(seq, step):
    a = seq
    n = len(a)
    # compute conv
    m = 1 << ((2*n-2).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa*fa, m)[:2*n-1]
    max_k = int(conv.argmax())
    max_b = conv[max_k]
    # compute dmax_i:
    # for each i compute counterpart j = max_k - i; if 0 <= j < n else 0 (since a_j outside range treat as 0)
    # note derivative only when i,j in range.
    # For i where j not in range, derivative is 0? Let's derive: conv_k includes only terms where both in range; for i out of range, there is no term, derivative is 0.
    # We'll compute derivative as:
    # if j in range:
    #   if i != j: dmax_i = 2*a_j
    #   else: dmax_i = a_i
    # else: dmax_i = 0
    dmax = np.zeros(n)
    low = max(0, max_k - (n - 1))
    high = min(n - 1, max_k)
    idxs = np.arange(low, high+1)
    js = max_k - idxs
    # compute contributions
    # where idxs == js:
    same = (idxs == js)
    dmax[idxs[same]] = a[idxs[same]]   # derivative for i=j
    # for i != j:
    notsame = ~same
    i_not = idxs[notsame]
    j_not = js[notsame]
    # For each i_not, derivative includes both a_j (from term where i=j) and also from term where i' = j_not? Wait we already accounted both sides? dmax_i = a_j + a_j = 2* a_j. So each i not self should have derivative = 2 * a_j
    dmax[i_not] = 2 * a[j_not]

    # Note that for each pair i,j both appear in loops; So each will get derivative from counterpart.

    sum_a = a.sum()

    grad = const * (dmax * sum_a - 2 * max_b)
    # now perform gradient descent step: new_a = a - step * grad

    # ensure non-negative
    new_a = np.maximum(new_a, 0)
    # renormalize to keep sum same (scale factor)
    s = new_a.sum()

    else:
        # if all zeros fallback random
        new_a = np.random.rand(n)
        new_a = new_a * (sum_a / new_a.sum())
    return new_a

"""
propose_candidate
-----------------

A fast gradient‑descent / simple hill‑climbing optimizer for the
step‑function evaluation problem.

The algorithm starts from the analytically motivated
inverse‑square‑root shape

    a_i = sqrt(i+1) – sqrt(i)          (i = 0 … n‑1)

which is known to give a low value (≈ π/2 ≈ 1.57).  The sequence is
scaled so that Σ a_i = √(2 n) – the same normalisation used by the
reference implementation – making the objective invariant to the
scale.

At each iteration we:

1. Compute the self‑convolution a * a by FFT (O(n log n)).
2. Locate the index k where the convolution attains its maximum.
3. Evaluate the sub‑gradient of the objective w.r.t. the entries that
   contribute to this maximum (the derivative of the max‑convolution
   is 2·a_{k−i} for i≠k−i and a_i for i=k−i).
4. Take a small step opposite to that gradient, clip negatives to
   zero and renormalise the sum.
5. Keep the best sequence seen so far; if a step does not improve,
   the step size is reduced, otherwise it is slightly increased.

The loop runs until the allotted time (≈ 1000 s) expires.  The final
sequence – a list of non‑negative floats – is returned.  The code
uses only numpy, time and the already imported ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast computation of the evaluation score (identical to
# evaluate_sequence but without the safety clipping).  This is used
# inside the optimiser for speed.
def _score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.size
    # length of the linear convolution
    conv_len = 2 * n - 1
    # zero‑pad to the next power of two for a fast FFT
    m = 1 << ((conv_len - 1).bit_length())
    ft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(ft * ft, n=m)[:conv_len]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed (default 42).
    budget_s : float
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Determine the length of the step function.
    # The global variable ``height_sequence_1`` is provided in the
    # competition environment; fall back to the length mentioned in the
    # description if it does not exist.
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        n = 6597

    # ------------------------------------------------------------------
    # 2. Initialise with the inverse‑square‑root shape.
    # a_i = sqrt(i+1) - sqrt(i)  (i = 0 … n‑1)
    # Scale so that Σ a_i = sqrt(2 n) – the same normalisation used by
    # the SOTA code.
    a = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(
        np.arange(0, n, dtype=np.float64)
    )
    target_sum = np.sqrt(2.0 * n)
    a *= target_sum / a.sum()      # now Σ a_i = √(2 n)

    best_seq = a.copy()
    best_val = _score(best_seq)
    print(f"[init] n={n}, initial score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop.
    deadline = time.time() + budget_s - 5.0   # leave a tiny safety margin
    step = 0.12          # initial step size for the gradient update
    iteration = 0
    # Parameters for the simple step‑size adaptation
    step_up = 1.07
    step_down = 0.65
    min_step = 1e-5

    while time.time() < deadline:
        iteration += 1

        # ---- 3.1 Compute convolution and locate the max entry ----------
        n = a.size
        conv_len = 2 * n - 1
        m = 1 << ((conv_len - 1).bit_length())
        ft = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(ft * ft, n=m)[:conv_len]

        k_max = int(conv.argmax())
        max_b = conv[k_max]
        sum_a = a.sum()

        # ---- 3.2 Sub‑gradient of the maximal convolution ---------------
        # derivative of conv_k w.r.t. a_i:
        #   * 2·a_{k−i}  if i ≠ k−i
        #   * a_i        if i = k−i
        dmax = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        jdx = k_max - idx

        # i == j case
        eq = idx == jdx
        dmax[idx[eq]] = a[idx[eq]]
        # i != j case
        neq = ~eq
        dmax[idx[neq]] = 2.0 * a[jdx[neq]]

        # Gradient of the whole objective (see analysis)
        const = 2.0 * n / (sum_a ** 3)
        grad = const * (dmax * sum_a - 2.0 * max_b)

        # ---- 3.3 Gradient‑descent step (projected on the simplex) -------
        new_a = a - step * grad
        # enforce non‑negativity
        new_a = np.maximum(new_a, 0.0)

        # renormalise to keep the sum unchanged (scale invariance)
        new_sum = new_a.sum()
        if new_sum > 0.0:
            new_a *= sum_a / new_sum
        else:
            # extremely unlikely – fall back to a tiny uniform jitter
            new_a = np.full_like(a, sum_a / n)

        new_val = _score(new_a)

        # ---- 3.4 Accept / reject & adapt step size --------------------
        if new_val < best_val:
            best_val = new_val
            best_seq = new_a.copy()
            a = new_a
            step = min(step * step_up, 1.0)   # gradual increase, keep bounded
        else:
            # reject the move but keep the current a; shrink step size
            step = max(step * step_down, min_step)

        # occasional cheap random jitter to escape tiny plateaus

                a = np.maximum(a + jitter, 0.0)
                a *= sum_a / a.sum()
            print(f"[iter {iteration:6d}] best={best_val:.6f} step={step:.5f}")

    # ------------------------------------------------------------------
    # 4. Final validation with the official evaluator (optional but
    #    useful for debugging – the return value does not depend on it).
    try:
        official = evaluate_sequence(best_seq.tolist())
        print(f"[final] official evaluate = {official:.6f}")
    except Exception:
        pass

    return best_seq.tolist()
