    seq_i_new = seq[i] - delta
    seq_j_new = seq[j] + delta

"""
Propose a candidate step‑function sequence.

The algorithm is a simple stochastic hill‑climber that works directly with the
convolution values.  At each iteration it tries one of three local moves:

1. **Single‑site perturbation** – a random coefficient is increased or
   decreased by a small amount.
2. **Transfer** – a random amount of mass is moved from one index to another.
3. **Zero‑to‑positive addition** – a zero entry is increased (this raises the
   total sum while typically only a tiny increase of the maximum convolution).

For each trial the convolution is updated analytically in O(n) time (no
re‑convolution) and the new evaluation score

    2 * n * max(conv) / (sum(a)^2)

is computed.  The move is accepted if it strictly improves the score.
The procedure runs until the allotted time expires, repeatedly restarting from
different random initialisations (including the provided `height_sequence_1`
if it exists).  The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy import optimize  # only needed if the LP fallback is ever used


def _add_to_conv(conv: np.ndarray, seq: np.ndarray, idx: int, delta: float) -> np.ndarray:
    """
    Return a copy of `conv` that corresponds to increasing `seq[idx]` by `delta`.
    The update uses the exact formula:
        conv'[k] = conv[k] + 2*delta*seq[k-idx]   for idx <= k < idx+len(seq)
        conv'[2*idx] additionally gets +delta**2
    """
    n = len(seq)
    new_conv = conv.copy()
    new_conv[idx:idx + n] += 2.0 * delta * seq
    new_conv[2 * idx] += delta * delta
    return new_conv


def _transfer_conv(conv: np.ndarray, seq: np.ndarray, i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a copy of `conv` after moving `delta` mass from index `i` to index `j`.
    `delta` must be non‑negative and `delta <= seq[i]`.  The operation is performed
    as two successive single‑site updates (first decreasing `i`, then increasing `j`).
    """
    # first, decrease i
    tmp_conv = conv.copy()
    tmp_conv[i:i + len(seq)] += 2.0 * (-delta) * seq
    tmp_conv[2 * i] += delta * delta

    # temporary sequence after the decrease (needed for the second update)
    tmp_seq = seq.copy()
    tmp_seq[i] -= delta

    # then, increase j using the updated temporary sequence
    tmp_conv[j:j + len(seq)] += 2.0 * delta * tmp_seq
    tmp_conv[2 * j] += delta * delta
    return tmp_conv

    seed : int

    budget_s : int

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # margin for finalisation

    # ------------------------------------------------------------------
    # Helpers for scoring
    def score_from_conv(conv, total):
        """Exact score used by evaluate_sequence (without the clipping steps)."""
        return 2.0 * n * conv.max() / (total * total)

    # ------------------------------------------------------------------
    # Keep the best solution ever seen
    best_seq = None
    best_score = float("inf")

    # If the previous SOTA sequence is available, evaluate it first.
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=float)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[init] height_sequence_1 score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Main optimisation loop – many independent short runs
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ---- random initialisation ------------------------------------
        n = np.random.randint(300, 800)          # moderate length
        seq = np.random.rand(n)                  # uniform [0,1)
        # randomly zero out a fraction to create “free slots”
        zero_frac = np.random.uniform(0.2, 0.5)
        mask = np.random.rand(n) < zero_frac
        seq[mask] = 0.0
        seq = np.clip(seq, 0.0, 1000.0)

        total = seq.sum()
        if total < 0.01:            # avoid the trivial invalid case
            continue

        conv = np.convolve(seq, seq)
        cur_score = score_from_conv(conv, total)

        # ---- local stochastic optimiser ---------------------------------
        local_iters = 0
        # simple cooling schedule for step sizes
        step_factor = 0.2
        while time.time() < deadline and local_iters < 8000:
            local_iters += 1

            # ----- decide which move to try ---------------------------
            move_type = np.random.rand()
            if move_type < 0.4:
                # 1) single‑site perturbation
                i = np.random.randint(n)
                # propose a relative change; sigma shrinks with time
                sigma = step_factor * (1.0 - (time.time() - start_time) / budget_s)
                delta = np.random.randn() * sigma * max(seq[i], 1e-3)
                new_val = max(0.0, seq[i] + delta)
                delta = new_val - seq[i]
                if abs(delta) < 1e-12:
                    continue
                new_conv = _add_to_conv(conv, seq, i, delta)
                new_total = total + delta
                new_score = 2.0 * n * new_conv.max() / (new_total * new_total)

                if new_score < cur_score:
                    seq[i] = new_val
                    conv = new_conv
                    total = new_total
                    cur_score = new_score

            elif move_type < 0.9:
                # 2) transfer mass from a “bad” site to a random one
                # choose a source index that contributes most to the current max
                k_max = np.argmax(conv)
                # contribution of each i to this k_max (ordered pairs counted twice)
                j = k_max - np.arange(n)
                valid = (j >= 0) & (j < n)
                contrib = np.zeros(n)
                contrib[valid] = seq[valid] * seq[j[valid]]
                # take only indices that actually have some mass
                cand_src = np.where((seq > 1e-12) & (contrib > 0))[0]
                if cand_src.size == 0:
                    continue
                i = cand_src[np.argmax(contrib[cand_src])]
                # pick a distinct destination
                j = np.random.randint(n)
                while j == i:
                    j = np.random.randint(n)

                # move up to 30 % of the source mass
                max_transfer = 0.3 * seq[i]
                if max_transfer <= 1e-12:
                    continue
                delta = np.random.rand() * max_transfer  # amount moved from i to j

                new_conv = _transfer_conv(conv, seq, i, j, delta)
                new_score = 2.0 * n * new_conv.max() / (total * total)

                if new_score < cur_score:
                    seq[i] -= delta
                    seq[j] += delta
                    conv = new_conv
                    cur_score = new_score

            else:
                # 3) add a little mass to a zero entry (if any)
                zero_idx = np.where(seq == 0.0)[0]
                if zero_idx.size == 0:
                    continue
                i = np.random.choice(zero_idx)
                # a tiny epsilon, scaled with current total mass
                eps = max(1e-4, total * 0.001) * np.random.rand()
                new_conv = _add_to_conv(conv, seq, i, eps)
                new_total = total + eps
                new_score = 2.0 * n * new_conv.max() / (new_total * new_total)

                if new_score < cur_score:
                    seq[i] += eps
                    conv = new_conv
                    total = new_total
                    cur_score = new_score

        # ---- after the local run, evaluate the exact score -------------
        final_score = evaluate_sequence(seq.tolist())
        if final_score < best_score:
            best_score, best_seq = final_score, seq.copy()
            print(f"[{iteration}] new global best = {best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep the interface safe
    if best_seq is None:
        best_seq = np.ones(10, dtype=float)

    # Return a plain list as required by the testing harness
    return best_seq.tolist()
