"""Gradient‑descent search for a step‑function minimizing
   2 n·max(conv(a,a))/sum(a)^2.

   The algorithm works with the scale‑invariant formulation
   sum(a)=1.  Starting from a uniform or a “U‑shaped’’ vector it
   repeatedly:
   1. computes the self‑convolution and its deviation from the
      constant target 1/(2n‑1),
   2. evaluates the exact gradient of the L2‑error
      (which also reduces the L∞‑error),
   3. performs a simple back‑tracking line‑search, projects onto
      the non‑negative simplex, and updates a.

   Multiple restarts (uniform, U‑shape, random) are tried within
   the given time budget and the best sequence seen (lowest
   evaluate_sequence value) is returned.
"""

import time
import numpy as np

def _gradient(a: np.ndarray, target: float) -> np.ndarray:
    """
    Compute ∇_a ½∑_k ( (a*a)_k - target )².
    The formula derived in the analysis gives:
        grad = 2 * ( (err  *  rev_a)[n-1:n-1+n]  +  (a * err)[:n] )
    where err = conv(a,a) - target and rev_a = a[::-1].
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                # length 2n‑1
    err = conv - target                     # same length
    # first term: sum_{k} err_k * a_{k‑i}
    first = np.convolve(err, a[::-1])[n - 1:n - 1 + n]
    # second term: sum_{k} err_k * a_{i‑k}
    second = np.convolve(a, err)[:n]
    return 2.0 * (first + second)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the probability simplex {x ≥ 0, sum(x)=1}.
    Standard O(n log n) algorithm.
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find threshold
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that lowers the evaluation metric.
    The function respects the overall time budget (budget_s seconds) and
    returns the best sequence (as a Python list) found so far.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    # Length of the sequence – use the supplied construction if available.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = np.random.randint(1500, 3000)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate quickly (same formula as evaluate_sequence)
    # ------------------------------------------------------------------
    def _score(vec):
        # vec is assumed to sum to 1 (or very close)
        max_conv = np.max(np.convolve(vec, vec))
        return float(2.0 * n * max_conv)   # equals evaluate_sequence for sum≈1

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    restarts = 3                     # number of different initial points
    best_seq = None
    best_val = float('inf')
    target = 1.0 / (2 * n - 1)       # target constant for sum(a)=1

    for restart in range(restarts):
        # ------- initialise a ------------------------------------------------
        if restart == 0 and 'height_sequence_1' in globals():
            a = np.asarray(height_sequence_1, dtype=float)
        elif restart == 1:
            # U‑shaped: larger values at the ends, smaller in the middle
            idx = np.arange(n, dtype=float)
            centre = (n - 1) / 2.0
            # quadratic bowl (minimum at centre)
            a = 1.0 + 4.0 * ((idx - centre) / centre) ** 2
        else:
            a = np.random.rand(n)

        # make it a probability vector
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # ------- gradient descent with simple back‑tracking line‑search ---
        step = 0.8            # initial step size
        decay = 0.9995        # gradual decay of the step size
        min_step = 1e-9

        while time.time() < deadline:
            # current convolution and its max
            conv = np.convolve(a, a)
            max_conv = conv.max()
            cur_val = 2.0 * n * max_conv

            # store best found
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"New best = {best_val:.6f} (restart {restart})")

            # compute gradient of the L2‑error (helps flatten the convolution)
            grad = _gradient(a, target)

            # ----- back‑tracking line‑search (monotone decrease of max_conv) -----
            step_local = step
            while step_local > min_step:
                cand = a - step_local * grad
                cand = _project_simplex(cand)      # keep feasibility
                new_max = np.max(np.convolve(cand, cand))
                if new_max < max_conv:
                    break
                step_local *= 0.5

            # update a and step size for the next iteration
            a = cand
            step = min(step_local * 1.1, 1.0)      # modest increase after a success
            step *= decay                         # global decay

            # occasional tiny random jitter to escape shallow plateaus
            if np.random.rand() < 0.01:
                idx = np.random.randint(n)
                jitter = np.random.randn() * step * 0.05
                a[idx] = max(0.0, a[idx] + jitter)
                a = _project_simplex(a)

        # end of this restart – one final evaluation
        final_val = _score(a)
        if final_val < best_val:
            best_val = final_val
            best_seq = a.copy()
            print(f"Restart {restart} finished with new best = {best_val:.6f}")

    print(f"Search finished. Best value = {best_val:.6f}")
    # Return as plain Python list (the external evaluation expects a list)
    return best_seq.tolist()
