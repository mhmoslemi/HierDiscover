"""Candidate proposer using hill‑climbing on the max‑convolution.
The algorithm keeps the total L1‐norm fixed (by construction) and iteratively
moves weight from indices that contribute most to the current peak of the
auto‑convolution to indices that contribute the least.  The convolution is
updated in O(n) using slice operations, allowing millions of cheap steps
within the time budget.  Occasionally a simulated‑annealing acceptance
criterion is used to escape local minima.  After the hill‑climbing phase a
few linear‑programming refinements (the same LP used in the reference code)
are applied to increase the L1‑norm while preserving the attained bound.
The best sequence found is returned.
"""

import time
import numpy as np
from scipy import optimize

def solve_convolution_lp(f_sequence, rhs):
    """Linear program: maximize sum(b) s.t. conv(f,b) <= rhs, b >= 0."""
    n = len(f_sequence)
    # Objective: maximize sum(b) => minimize -1*b
    c = -np.ones(n)
    # Build constraints for each convolution index k
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # For each i, j = k-i must be in [0,n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # Fill row[j] = f[i] where j=k-i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # Non‑negativity constraints: -b <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])
    # Solve with a modest time limit per call.
    res = optimize.linprog(
        c, A_ub=a_ub, b_ub=b_ub,
        bounds=(0, None),
        method='highs',
        options={'time_limit': 5.0, 'presolve': True, 'msg': False}
    )
    if res.success:
        return res.x
    return None

def improve_via_lp(sequence, t=0.05):
    """
    One LP‑based refinement step.
    Normalises ``sequence`` to sqrt(2n), solves the LP that maximises the
    L1‑norm under the current convolution bound and mixes the result with
    the original vector.
    """
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:
        return sequence
    normalized = sequence * np.sqrt(2 * n) / sum_seq
    rhs = np.max(np.convolve(normalized, normalized))
    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return sequence
    # Rescale g to the original normalisation
    g = g * np.sqrt(2 * n) / np.sum(g)
    # Convex combination
    new_seq = (1 - t) * sequence + t * g
    return new_seq

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step function with low evaluation value.
    The method consists of:
      1. Random initialisation with fixed L1‑norm.
      2. A fast hill‑climbing / simulated‑annealing loop that repeatedly
         moves weight from an index that contributes most to the current
         convolution peak to an index that contributes the least.
         The convolution is updated incrementally in O(n).
      3. A few calls to the linear‑programming refinement from the reference
         implementation to increase the L1‑norm while keeping the peak
         unchanged.
    The best sequence found before the time deadline is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # keep a small safety margin

    # Choose a length that balances expressive power and speed.
    n = 800
    # Initialise a non‑negative sequence with a fairly wide spread.
    a = np.random.exponential(scale=1.0, size=n)
    a = a / np.sum(a) * np.sqrt(2 * n)   # normalise to sqrt(2n)

    # Full auto‑convolution (float64) and auxiliary data.
    c = np.convolve(a, a)
    sum_a = np.sum(a)                    # = sqrt(2n) up to rounding

    best_a = a.copy()
    best_score = 2.0 * n * np.max(c) / (sum_a ** 2)

    # Simple simulated‑annealing parameters.
    temperature = best_score * 0.2
    cooling_rate = 0.9995   # exponential cooling each iteration

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # Identify the index of the current maximal convolution entry.
        k = int(np.argmax(c))

        # Range of i for which both a[i] and a[k-i] are defined.
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        jdx = k - idx
        contributions = a[idx] * a[jdx]

        # Pick a donor (large contribution) and a receiver (small contribution).
        i_max = idx[np.argmax(contributions)]
        i_min = idx[np.argmin(contributions)]
        if i_max == i_min:
            continue

        # Amount we are willing to move from i_max to i_min.
        max_delta = min(a[i_max], 1000.0 - a[i_min])
        if max_delta <= 0:
            continue
        # Randomise the step size (up to half of the feasible amount).
        delta = max_delta * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        # Keep copies of the current state (cheap, because vectors are small).
        a_old = a.copy()
        c_old = c.copy()

        # Apply the move to the coefficient vector.
        a[i_max] -= delta
        a[i_min] += delta

        # Update the convolution using the analytic formula.
        # Linear part: 2 * delta * (shift by i_min – shift by i_max)
        c[i_max:i_max + n] -= 2.0 * delta * a_old
        c[i_min:i_min + n] += 2.0 * delta * a_old
        # Quadratic part: delta^2 * (δ_{2i_max} + δ_{2i_min} - 2 δ_{i_max+i_min})
        c[2 * i_max] += delta * delta
        c[2 * i_min] += delta * delta
        c[i_max + i_min] -= 2.0 * delta * delta

        # Current score after the tentative move.
        cur_score = 2.0 * n * np.max(c) / (sum_a ** 2)

        # Acceptance test.
        if cur_score < best_score or np.random.rand() < np.exp(-(cur_score - best_score) / temperature):
            # Accept the move.
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            # Reduce temperature a little.
            temperature *= cooling_rate
        else:
            # Reject – restore previous state.
            a = a_old
            c = c_old
            temperature *= cooling_rate

        # Occasionally perform a global random perturbation to escape plateaus.
        if iteration % 5000 == 0:
            # Small Gaussian shuffle that respects non‑negativity.
            pert = np.random.randn(n) * 0.02 * np.mean(a)
            a = np.maximum(0.0, a + pert)
            # Renormalise to keep the L1‑norm (avoids drift caused by clipping).
            a = a * (sum_a / np.sum(a))
            c = np.convolve(a, a)

    # After the hill‑climbing stage, run a few LP refinements.
    refined = best_a.copy()
    for _ in range(3):
        refined = improve_via_lp(refined, t=0.08)
        # Keep the best among the refined candidates.
        score_ref = 2.0 * n * np.max(np.convolve(refined, refined)) / (np.sum(refined) ** 2)
        if score_ref < best_score:
            best_score = score_ref
            best_a = refined.copy()

    # Ensure the return type complies with the evaluator (list of floats).
    return best_a.tolist()

"""
Candidate generator for the step‑function optimisation problem.

The algorithm works in three stages:

1. **Random initialisation** – a non‑negative vector of length ``n`` is drawn
   (exponential distribution) and normalised so that ``sum(a)=sqrt(2n)``.
   The evaluation function is invariant under a common scaling, therefore the
   L1‑norm can be fixed without loss of generality.

2. **Hill‑climbing / simulated‑annealing** – the current auto‑convolution
   ``c = a * a`` is maintained.  At every iteration we locate the index ``k``
   where ``c`` attains its maximum.  For this ``k`` we compute the per‑index
   contributions ``a[i] * a[k-i]``.  An index ``i_max`` with a *large*
   contribution (a donor) and an index ``i_min`` with a *small* contribution
   (a receiver) are selected.  A small amount ``delta`` of weight is moved
   from ``i_max`` to ``i_min``.  The convolution can be updated in O(n) time
   using the analytic formula

        c' = c + 2·δ·(shift(i_min) – shift(i_max))
             + δ²·(e_{2i_min}+e_{2i_max}–2·e_{i_min+i_max})

   where ``shift(p)`` is the vector ``a`` shifted by ``p`` places.
   The move is accepted if it improves the current score or with a
   probability that follows a simple annealing schedule; otherwise it is
   rejected.  Periodic global Gaussian perturbations help to escape plateaus.

3. **LP refinement** – after the hill‑climbing phase a few iterations of the
   linear‑programming step from the reference implementation are performed.
   The LP maximises the L1‑norm under the present convolution bound, and the
   resulting vector is mixed with the current best one.

The best sequence found before the deadline is returned as a plain list of
floats, ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import optimize

def solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise   sum b_j
        subject to  (f * b)_k <= rhs  for all k
                    b_j >= 0
    Returns the optimal ``b`` as a NumPy array or ``None`` if the LP fails.
    """
    n = len(f_seq)
    # maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints.
    A = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)

    A = np.vstack(A)
    b = np.array(b)

    # Use the HiGHS solver (fast and reliable) with a modest per‑call time limit.
    res = optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"time_limit": 5.0, "presolve": True, "msg": False},
    )
    if res.success:
        return res.x
    return None


def improve_via_lp(seq, t=0.08):
    """
    One refinement step that mimics the reference algorithm:
    * normalise ``seq`` to sqrt(2n)
    * solve the LP that maximises the L1‑norm under the current convolution
      bound ``M``
    * rescale the LP solution back to the original normalisation
    * return the convex combination (1‑t)·seq + t·g
    """
    n = len(seq)
    total = np.sum(seq)
    if total < 1e-12:
        return seq
    norm = np.sqrt(2 * n)
    f = seq * (norm / total)                 # normalised version
    rhs = np.max(np.convolve(f, f))
    g = solve_convolution_lp(f, rhs)
    if g is None:
        return seq
    # bring g back to the original scaling
    g = g * (norm / np.sum(g))
    return (1 - t) * seq + t * g


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Main entry point used by the evaluator.
    Returns a list of non‑negative heights that (hopefully) yields a low
    ``evaluate_sequence`` value.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # a tiny safety margin

    # ------------------------------------------------------------------ #
    # 1) initialise ---------------------------------------------------- #
    # ------------------------------------------------------------------ #
    n = 800                                     # moderate size – fast updates
    a = np.random.exponential(scale=1.0, size=n)
    a = a / np.sum(a) * np.sqrt(2 * n)          # fix L1‑norm

    c = np.convolve(a, a)                       # full auto‑convolution
    sum_a = np.sum(a)                           # = sqrt(2n) up to rounding

    best_a = a.copy()
    best_score = 2.0 * n * np.max(c) / (sum_a ** 2)

    # current state (used by the annealing loop)
    cur_a = a.copy()
    cur_c = c.copy()
    cur_score = best_score

    # simple exponential cooling schedule
    temperature = best_score * 0.2
    cooling = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1

        # -------------------------------------------------------------- #
        # locate the current peak of the convolution and the contributors #
        # -------------------------------------------------------------- #
        k = int(np.argmax(cur_c))

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        jdx = k - idx
        contrib = cur_a[idx] * cur_a[jdx]

        i_max = idx[np.argmax(contrib)]
        i_min = idx[np.argmin(contrib)]

        if i_max == i_min:
            continue

        # -------------------------------------------------------------- #
        # propose moving a small amount of mass from i_max to i_min     #
        # -------------------------------------------------------------- #
        max_delta = min(cur_a[i_max], 1000.0 - cur_a[i_min])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        # keep copies of the current state (cheap – vectors are small)
        a_old = cur_a.copy()
        c_old = cur_c.copy()
        score_before = cur_score

        # perform the move
        cur_a[i_max] -= delta
        cur_a[i_min] += delta

        # update the convolution using the analytic formula
        # linear part: 2·δ·(shift(i_min) – shift(i_max))
        cur_c[i_max : i_max + n] -= 2.0 * delta * a_old
        cur_c[i_min : i_min + n] += 2.0 * delta * a_old
        # quadratic part: δ²·(δ_{2i_max}+δ_{2i_min}−2·δ_{i_max+i_min})
        cur_c[2 * i_max] += delta * delta
        cur_c[2 * i_min] += delta * delta
        cur_c[i_max + i_min] -= 2.0 * delta * delta

        # new score after the tentative move
        new_score = 2.0 * n * np.max(cur_c) / (sum_a ** 2)

        # -------------------------------------------------------------- #
        # simulated‑annealing acceptance                                  #
        # -------------------------------------------------------------- #
        accept = False
        if new_score < score_before:
            accept = True
        else:
            prob = np.exp(-(new_score - score_before) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
            temperature *= cooling
        else:
            # revert
            cur_a = a_old
            cur_c = c_old
            cur_score = score_before
            temperature *= cooling

        # occasional global perturbation that respects the L1‑norm
        if it % 5000 == 0:
            perturb = np.random.randn(n) * 0.02 * np.mean(cur_a)
            cur_a = np.maximum(0.0, cur_a + perturb)
            # renormalise to keep the same sum (avoids drift)
            cur_a = cur_a * (sum_a / np.sum(cur_a))
            cur_c = np.convolve(cur_a, cur_a)
            cur_score = 2.0 * n * np.max(cur_c) / (sum_a ** 2)

    # ------------------------------------------------------------------ #
    # 2) final LP refinement -------------------------------------------- #
    # ------------------------------------------------------------------ #
    refined = best_a.copy()
    for _ in range(3):
        refined = improve_via_lp(refined, t=0.08)
        tmp_score = 2.0 * n * np.max(np.convolve(refined, refined)) / (np.sum(refined) ** 2)
        if tmp_score < best_score:
            best_score = tmp_score
            best_a = refined.copy()

    # return a plain Python list as required by the evaluator
    return best_a.tolist()
