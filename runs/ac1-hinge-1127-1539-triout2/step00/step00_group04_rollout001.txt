def compute_score(seq):
    n = len(seq)
    S = np.sum(seq)
    if S < 0.01: return np.inf
    L = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (S * S)

import numpy as np
import time, math

def fast_score(seq):
    n = len(seq)
    S = seq.sum()
    if S < 1e-12:
        return np.inf
    L = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (S * S)

def generate_sparse_sequence(n, m):
    # generate m non-zero positions with equal weights sum=1
    positions = np.random.choice(n, size=m, replace=False)
    seq = np.zeros(n)
    seq[positions] = 1.0 / m
    return seq

def propose_candidate(seed=12345, budget_s=1000):
    """
    Hill‑climbing / simulated‑annealing search for a sparse non‑negative step
    function that minimises ``evaluate_sequence``.
    The idea is to keep only a relatively small number of non‑zero heights,
    which reduces the maximum self‑convolution while the total sum stays
    normalised (scale invariant).  Random moves transfer a small amount of
    mass between two coordinates and are accepted if they improve the score
    (or occasionally with a temperature‑dependent probability).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # leave a small margin

    # Choose a problem size.  Larger n gives a lower theoretical bound
    # (≈ 2 n/(2 n‑1) → 1) but also makes each iteration slower.
    # ∼ 4000 provides a good balance for the 1000 s limit.
    n = 4000

    # Desired number of non‑zero entries.  Theory suggests m ≈ √(c · 2‑n)
    # with a small collision factor c.  We use a generous factor 1.8.
    m_target = int(math.ceil(1.8 * math.sqrt(2.0 * n)))
    if m_target > n // 2:
        m_target = n // 2

    # Initialise a sparse vector with equal mass on m_target coordinates.
    seq = generate_sparse_sequence(n, m_target)

    # Keep a copy of the best sequence seen so far.
    best_seq = seq.copy()
    best_score = fast_score(seq)
    current_score = best_score

    # Pre‑compute the FFT length (next power of two) to reuse in the loop.
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Simple exponential cooling schedule.
    max_temp = 0.1
    # A small step size relative to the current value of a coordinate.
    base_step = 0.05

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # Pick two (possibly equal) indices, ensure we move mass from a non‑zero.
        i = np.random.randint(n)
        if seq[i] <= 1e-12:
            # pick a non‑zero source
            nz = np.where(seq > 1e-12)[0]
            if nz.size == 0:
                continue
            i = np.random.choice(nz)

        j = np.random.randint(n)
        # Avoid a trivial move
        if i == j:
            continue

        # Amount of mass to transfer.
        delta = base_step * np.random.rand() * seq[i]
        if delta <= 0:
            continue

        # Apply the move.
        seq[i] -= delta
        seq[j] += delta

        # Occasionally prune near‑zero entries to keep the representation sparse.
        if seq[i] < 1e-12:
            seq[i] = 0.0

        # Compute the new score using the fast FFT based routine.
        fft_seq = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(fft_seq * fft_seq, n=fft_len)[:2 * n - 1]
        max_b = conv.max()
        new_score = 2.0 * n * max_b / (seq.sum() ** 2)

        # Temperature that linearly decays to zero.
        t_frac = max(0.0, (deadline - time.time()) / budget_s)
        temperature = max_temp * t_frac

        # Acceptance rule.
        if new_score < current_score or np.random.rand() < math.exp(-(new_score - current_score) / (temperature + 1e-12)):
            current_score = new_score
            # Keep the move.
        else:
            # Undo the move.
            seq[i] += delta
            seq[j] -= delta
            # If the reversal creates a tiny negative (numerical), fix it.
            if seq[j] < 0:
                seq[j] = 0.0

        # Update the best solution seen.
        if current_score < best_score:
            best_score = current_score
            best_seq = seq.copy()
            # Print occasional progress.
            if iteration % 500 == 0:
                print(f"Iter {iteration}: best score = {best_score:.6f}")

    # At the end, return the best sequence found (as a plain Python list).
    # The caller will evaluate it with ``evaluate_sequence``.
    return best_seq.tolist()

"""
Sparse‑hill‑climb search for a step‑function with a low
evaluate_sequence() value.

The idea:
* The evaluation 2·n·max(conv(a,a))/sum(a)^2 is scale‑invariant.
* If only a few entries of a are non‑zero, each sum k receives only a
  handful of products a_i·a_j, so max(conv) can be kept very small while
  the total sum (normalised to 1) stays large.
* Starting from a random sparse vector we repeatedly transfer a small
  amount of mass from one coordinate to another (preserving the total
  sum).  Moves that improve the fast FFT‑based proxy of the score are
  always accepted; occasional uphill moves are allowed using a simple
  temperature schedule (simulated annealing).
* The best sequence seen during the run is returned.  The returned list
  consists of non‑negative floats and will be checked by the
  supplied evaluate_sequence() function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fast_score(seq: np.ndarray) -> float:
    """
    Compute the same objective as evaluate_sequence but using FFT‑based
    convolution for speed.  The formula is
        2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)
    If the sum is too small the function returns +inf.
    """
    n = len(seq)
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    # length for a tidy linear convolution (next power of two)
    L = 1 << ((2 * n - 1).bit_length())
    # real FFT → O(L log L)
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (total * total)


def _initial_sparse_sequence(n: int, m: int) -> np.ndarray:
    """
    Return a length‑n vector with exactly m non‑zero entries.
    The non‑zero entries are each 1/m (so the total sum equals 1).
    """
    seq = np.zeros(n, dtype=float)
    # choose m distinct positions uniformly at random
    pos = np.random.choice(n, size=m, replace=False)
    seq[pos] = 1.0 / m
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The algorithm is a simple hill‑climbing /
    simulated‑annealing loop on a sparse vector.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget (seconds).  The function returns before this
        limit (a small safety margin is kept).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin so the environment can finish cleanly
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a problem size.
    # Larger n gives a lower theoretical lower bound (≈ 2n/(2n‑1) → 1),
    # but each iteration becomes slower.  n≈4000 is a good compromise.
    # ------------------------------------------------------------------
    n = 4000

    # Desired number of non‑zero entries.
    # Theory suggests m ≈ √(c·2n); we use a generous factor.
    m_target = int(math.ceil(1.8 * math.sqrt(2.0 * n)))
    # Guard against absurdly large m
    if m_target > n // 2:
        m_target = n // 2

    # Initialise a sparse candidate (sum == 1)
    seq = _initial_sparse_sequence(n, m_target)

    # Keep the best solution seen so far
    best_seq = seq.copy()
    best_score = _fast_score(seq)
    current_score = best_score

    # Pre‑compute the FFT length (next power of two) – reused each loop.
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Simple linear cooling schedule
    max_temp = 0.1          # temperature at the very start
    base_step = 0.05        # proportion of a source entry that may be moved

    iter_no = 0
    stagnant = 0            # number of iterations without improvement

    while time.time() < deadline:
        iter_no += 1

        # --------------------------------------------------------------
        # Propose a small random transfer of mass.
        # --------------------------------------------------------------
        # pick a source index with non‑zero mass
        nz_indices = np.where(seq > 1e-12)[0]
        if nz_indices.size == 0:
            # degenerate case – re‑initialise
            seq = _initial_sparse_sequence(n, m_target)
            current_score = _fast_score(seq)
            continue
        i = np.random.choice(nz_indices)

        # pick a (possibly zero) target index different from i
        j = np.random.randint(n)
        if j == i:
            continue

        # amount to move (up to base_step * seq[i])
        delta = base_step * np.random.rand() * seq[i]
        if delta <= 0.0:
            continue

        # Apply the move (mass is conserved)
        seq[i] -= delta
        seq[j] += delta
        # prune tiny values to keep the representation sparse
        if seq[i] < 1e-12:
            seq[i] = 0.0

        # --------------------------------------------------------------
        # Evaluate the new candidate (fast FFT version)
        # --------------------------------------------------------------
        f = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
        max_b = conv.max()
        new_score = 2.0 * n * max_b / (seq.sum() ** 2)

        # --------------------------------------------------------------
        # Temperature for simulated annealing (linear decay)
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        t_frac = max(0.0, time_left / budget_s)   # 1 → start, 0 → end
        temperature = max_temp * t_frac

        # --------------------------------------------------------------
        # Acceptance rule
        # --------------------------------------------------------------
        if new_score < current_score or np.random.rand() < math.exp(
            -(new_score - current_score) / (temperature + 1e-12)
        ):
            # keep the move
            current_score = new_score
        else:
            # revert
            seq[i] += delta
            seq[j] -= delta
            # numerical safety – never let an entry become negative
            if seq[j] < 0.0:
                seq[j] = 0.0

        # --------------------------------------------------------------
        # Record the best solution
        # --------------------------------------------------------------
        if current_score < best_score:
            best_score = current_score
            best_seq = seq.copy()
            stagnant = 0
            # occasional progress report
            if iter_no % 500 == 0:
                print(f"[{time.time() - start:6.1f}s] iter {iter_no}, "
                      f"new best = {best_score:.6f}")
        else:
            stagnant += 1

        # --------------------------------------------------------------
        # If we have been stagnant for a long time, re‑seed the search.
        # --------------------------------------------------------------
        if stagnant > 5000:
            seq = _initial_sparse_sequence(n, m_target)
            current_score = _fast_score(seq)
            stagnant = 0

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # The host evaluates it with the official evaluate_sequence().
    # ------------------------------------------------------------------
    print(f"Search finished after {iter_no} iterations. "
          f"Best score seen: {best_score:.6f}")
    return best_seq.tolist()
