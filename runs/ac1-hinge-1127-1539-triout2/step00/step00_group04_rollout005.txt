def solve_convolution_lp(f_sequence, rhs):
    n = len(f_sequence)
    # Build A_ub as sparse: each constraint row corresponds to convolution at k.

    rows = []
    cols = []
    data = []
    # for each k in 0..2n-2:
    for k in range(2*n -1):
        # i from max(0,k-n+1) to min(k,n-1)
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # for each i, j = k - i
        for i in range(i_start, i_end+1):
            j = k - i
            # We need row coefficient for variable b_j = f[i]
            # Actually row[j] = f[i]
            rows.append(k)
            cols.append(j)
            data.append(f_sequence[i])
    # Now also add -I for non-negativity
    # Actually we add constraints b_j >=0 as -I b <= 0
    for j in range(n):
        rows.append(2*n-1 + j)
        cols.append(j)
        data.append(-1.0)

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1 + n, n))
    b_ub = np.concatenate([np.full(2*n-1, rhs), np.zeros(n)])

    # Objective: maximize sum b => c = -1 (since linprog minimizes)
    c = -np.ones(n)

    # Use method='highs' and option 'max_iter' maybe
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 5})
    if result.success:
        return result.x
    else:
        return None

def solve_convolution_lp(f_seq, rhs):
    import numpy as np, scipy.sparse as sp
    from scipy.optimize import linprog
    n = len(f_seq)
    # Build sparse Toeplitz matrix of size (2n-1) x n
    rows = []
    cols = []
    data = []
    # each constraint for k = 0 .. 2n-2
    for k in range(2*n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # For each i, j = k - i
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f_seq[i])
    # Nonnegativity constraints: -I * b <= 0
    for j in range(n):
        rows.append(2*n - 1 + j)
        cols.append(j)
        data.append(-1.0)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1 + n, n))
    b_ub = np.concatenate([np.full(2*n - 1, rhs, dtype=np.float64),
                           np.zeros(n, dtype=np.float64)])
    c = -np.ones(n, dtype=np.float64)  # maximize sum(b)
    # Use HiGHS interior, can set time limit 3 seconds.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  method='highs',
                  options={'presolve': True, 'time_limit': 3.0})
    if res.success:
        return res.x
    else:
        return None

"""
Hybrid Frank‑Wolfe optimizer for the self‑convolution bound.

The algorithm works with a fixed length `n` (default 1000).  
1.  Start from a simple geometric sequence whose shape is close to the
    theoretical optimum for the continuous problem.
2.  Repeatedly:
    * Normalise the current sequence so that  Σa = sqrt(2·n)
      (the evaluation becomes simply max(conv(a,a)) ).
    * Solve a linear programme that finds a non‑negative vector `g`
      with the *largest possible* ℓ¹‑norm while respecting the same
      convolution bound.  This is the classic Frank‑Wolfe direction.
    * Perform a short line‑search (α = 1, 0.5, 0.3, 0.2, 0.1, 0.05,
      0.02, 0.01) on the mixture (1‑α)·a + α·g and keep the best
      improvement.  The mixture stays normalised, therefore the
      evaluation never increases when a better α is found.
    * Occasionally apply a tiny random perturbation to escape possible
      stagnation.
3.  The best sequence seen during the whole run is returned.
   The LP is built as a sparse matrix (SciPy “highs” solver) so that
   even `n≈2000` remains fast.
"""

import time
import random
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _normalise(seq):
    """Scale ``seq`` so that its ℓ¹‑norm equals sqrt(2·len(seq))."""
    n = len(seq)
    total = float(np.sum(seq))
    if total == 0.0:
        return [0.0] * n
    factor = np.sqrt(2.0 * n) / total
    return [float(x * factor) for x in seq]


def _solve_lp(f, rhs, time_limit=3.0):
    """
    Solve the LP

        maximise   ∑ b_j
        subject to  (f * b)_k ≤ rhs   for every k
                    b_j ≥ 0

    ``f`` is a non‑negative list, ``rhs`` is a scalar.
    Returns ``None`` on failure, otherwise a NumPy 1‑d array of length n.
    """
    n = len(f)

    # ---- build the convolution constraints  (sparse) ----
    rows, cols, data = [], [], []
    # for each output index k of the convolution
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i                       # b_j participates with coefficient f_i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # non‑negativity: -b_j ≤ 0
    for j in range(n):
        rows.append(2 * n - 1 + j)
        cols.append(j)
        data.append(-1.0)

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1 + n, n))
    b_ub = np.concatenate([np.full(2 * n - 1, rhs, dtype=np.float64),
                           np.zeros(n, dtype=np.float64)])

    # objective (linprog minimises, so use -1 to maximise)
    c = -np.ones(n, dtype=np.float64)

    # use HiGHS (fast and accepts sparse matrices)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  method='highs',
                  options={'presolve': True,
                           'time_limit': time_limit})
    if res.success:
        return res.x
    return None


def _eval(seq):
    """Thin wrapper around the provided evaluate_sequence function."""
    return evaluate_sequence(seq)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000, **_):
    """
    Return a list of non‑negative heights (the step function) that
    minimises the evaluation function within the given time budget.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between optimisation power and LP cost.
    # 1000 works well on a single CPU in < 1 s per LP.
    # ------------------------------------------------------------------
    n = 1000
    sqrt_2n = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1) Initial candidate – a geometric progression close to the
    #    continuous optimum (the constant s solves e^s = 1 + 2s).
    # ------------------------------------------------------------------
    s_opt = 1.2564312086261697          # root of e**s = 1 + 2*s
    delta = s_opt / n
    r = max(0.0, 1.0 - delta)          # common ratio
    geom = [r ** i for i in range(n)]
    cur = _normalise(geom)             # now Σcur = sqrt(2n)

    best_seq = cur.copy()
    best_score = _eval(best_seq)

    print(f"[init] length={n}, ratio={r:.6f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # small safety margin
    iteration = 0

    # pre‑computed list of step sizes for the line‑search
    alphas = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # (a) Normalise current point – invariant for the objective
        # --------------------------------------------------------------
        cur = _normalise(cur)
        cur_score = _eval(cur)

        # --------------------------------------------------------------
        # (b) Compute the current convolution bound M
        # --------------------------------------------------------------
        conv_cur = np.convolve(cur, cur)
        M = float(np.max(conv_cur))

        # --------------------------------------------------------------
        # (c) Solve the LP to obtain a direction g
        # --------------------------------------------------------------
        g = _solve_lp(cur, M, time_limit=2.0)

        if g is None:
            # LP failed – small random perturbation and continue
            idx = np.random.randint(n)
            cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.01 * cur[idx])
            continue

        g = _normalise(g.tolist())

        # --------------------------------------------------------------
        # (d) Line‑search on mixtures (1‑α)·cur + α·g
        # --------------------------------------------------------------
        improved = False
        for α in alphas:
            new = [(1.0 - α) * x + α * y for x, y in zip(cur, g)]
            new = _normalise(new)
            try:
                new_score = _eval(new)
            except Exception:
                continue

            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = new.copy()
                print(f"[{iteration:05d}] new best {best_score:.6f} (α={α})")

            if new_score < cur_score - 1e-12:
                cur = new
                cur_score = new_score
                improved = True
                break    # accept the first improving α

        # --------------------------------------------------------------
        # (e) If none of the α values improved, perform a tiny perturb.
        # --------------------------------------------------------------
        if not improved:
            # perturb a few random coordinates (≤ 5 % of entries)
            n_perturb = max(1, n // 20)
            idxs = np.random.choice(n, size=n_perturb, replace=False)
            cur = cur.copy()
            for i in idxs:
                cur[i] = max(0.0, cur[i] + np.random.randn() * 0.005 * cur[i])
            # no need to re‑normalise – a tiny scaling error is harmless,
            # but we normalise for safety.
            cur = _normalise(cur)

        # --------------------------------------------------------------
        # (f) Periodic checkpoint – occasionally re‑evaluate the background
        #     sequence (if a global height_sequence_1 exists) to avoid
        #     getting stuck in a poor basin.
        # --------------------------------------------------------------
        if iteration % 50 == 0 and 'height_sequence_1' in globals():
            cand = list(height_sequence_1)
            # adapt length if necessary
            if len(cand) != n:
                if len(cand) > n:
                    cand = cand[:n]
                else:
                    cand = cand + [0.0] * (n - len(cand))
            cand = _normalise(cand)
            try:
                cand_score = _eval(cand)
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand
                    print(f"[{iteration:05d}] swapped in external seq, score={best_score:.6f}")
            except Exception:
                pass

    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
