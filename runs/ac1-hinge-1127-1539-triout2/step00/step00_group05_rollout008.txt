def improve_sequence(seq):
   n=len(seq)
   sum_seq = np.sum(seq)
   if sum_seq ==0:
       return seq
   normalized_f = seq * np.sqrt(2*n) / sum_seq
   M = np.max(np.convolve(normalized_f, normalized_f))
   g = solve_convolution_lp(normalized_f, M) # returns array g, may be None
   if g is None:
       return seq  # no improvement
   # g may not be normalized; compute sum_g
   sum_g = np.sum(g)
   if sum_g == 0: return seq
   normalized_g = g * np.sqrt(2*n) / sum_g
   best_seq = seq
   best_score = evaluate_sequence(seq)
   for t in [0.02,0.05,0.1,0.2,0.4]:
        new_seq = (1 - t) * seq + t * normalized_g
        # ensure nonnegative (already) and optionally clip
        new_seq = np.maximum(0, new_seq) # as already nonnegative
        # optionally rescale to sum=original sum? Not needed for objective but keep stable
        # Rescale to original sum for fairness:
        new_seq = new_seq * (sum_seq / np.sum(new_seq) if np.sum(new_seq)>0 else 1)
        new_score = evaluate_sequence(new_seq)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq
   return best_seq

def gradient_step(seq, step=0.1):
   n=len(seq)
   b = np.convolve(seq, seq)
   max_idx = int(np.argmax(b))
   # gradient: 2 * shifted seq: for each i, if 0 <= max_idx - i < n, grad_i = 2*seq[max_idx - i] else 0
   grad = np.zeros(n)
   for i in range(n):
         j = max_idx - i
         if 0 <= j < n:
            grad[i] = 2 * seq[j]
   # step gradient descent:

def gradient_step(seq, step):
    n=len(seq)
    b = np.convolve(seq, seq)
    max_idx = np.argmax(b)
    # compute gradient (subgradient) vector
    # gradient_i = 2*seq[max_idx - i] if complement in range else 0
    grad = np.zeros(n)
    for i in range(n):
        j = max_idx - i
        if 0 <= j < n:
            grad[i] = 2 * seq[j]
    # Move opposite direction
    new_seq = seq - step * grad
    # Clip to non-negative
    new_seq = np.maximum(new_seq, 0.0)
    # If all zero, reinitialize random
    total = np.sum(new_seq)
    if total < 1e-9:
        # random uniform
        new_seq = np.random.random(n)
        total = np.sum(new_seq)
    # Renormalize sum to 1 (or original sum)
    new_seq = new_seq * (1.0 / total)  # set sum =1
    return new_seq

"""
Propose a candidate step‑function height sequence that lowers the
objective  2·n·max(conv(a,a)) / (sum(a)^2).

The algorithm mixes two complementary search heuristics:

1. **LP‑driven direction** – Implements the “LP‑improvement” described in the
   statement.  A linear program finds a non‑negative vector g that maximises
   its L1‑norm while keeping the convolution with the current sequence f
   bounded by the present maximum M = ‖f * f‖∞.  After normalising g to the
   same L1‑norm as f we move a fraction `t` towards g.  This operation is
   proven to never increase M and usually shrinks it.

2. **Gradient‑based descent** – For the current maximal convolution index k*,
   the sub‑gradient of ‖a*a‖∞ w.r.t. a is `grad_i = 2·a_{k*‑i}` (zero when the
   partner index is out of range).  A small step opposite to this gradient
   reduces the current maximal entry.  After the step the sequence is
   projected onto the non‑negative orthant and re‑normalised to unit L1‑norm.

The search alternates between these two moves, occasionally applying tiny
random perturbations to escape shallow plateaus and occasionally restarting
with a fresh random sequence of a (randomly chosen) length.  The best
sequence seen so far is kept and returned.

The whole routine respects the 1000 s budget, prints occasional progress
messages, and finally returns the best list of non‑negative floats.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def solve_convolution_lp(f_sequence, rhs):
    """
    Linear programme:
        maximise   Σ_j b_j
        subject to conv(f, b)_k ≤ rhs  for all k
                  b_j ≥ 0
    Returns the optimal b as a NumPy array or None if the LP fails.
    """
    n = len(f_sequence)
    # objective (maximise sum => minimise -sum)
    c = -np.ones(n)

    # Build inequality rows for each convolution index k = 0 … 2n‑2
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints: -b_j ≤ 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    # Use the HiGHS solver (default); give it a modest time‑limit.
    result = linprog(c, A_ub=a_ub, b_ub=b_ub,
                     method='highs',
                     options={'time_limit': 8.0, 'disp': False})
    if result.success:
        return result.x
    return None


def get_good_direction_to_move_into(sequence, t=0.01):
    """
    Returns a new sequence obtained by moving a fraction `t` towards the LP
    direction described above.  If the LP fails, returns None.
    """
    n = len(sequence)
    seq = np.asarray(sequence, dtype=float)
    sum_seq = seq.sum()
    if sum_seq == 0:
        return None

    # Normalise to the conventional L1‑norm √(2n)
    factor = np.sqrt(2.0 * n) / sum_seq
    f_norm = seq * factor

    # Current bound M = max(conv(f,f))
    rhs = np.max(np.convolve(f_norm, f_norm))

    # Solve the LP for g
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    # Rescale g back to the same L1‑norm as the original sequence
    sum_g = g.sum()
    if sum_g == 0:
        return None
    g_norm = g * (np.sqrt(2.0 * n) / sum_g)

    # Convex combination (1‑t)·f + t·g
    new_seq = (1.0 - t) * seq + t * g_norm
    # Safety: enforce non‑negativity and re‑normalise to unit sum
    new_seq = np.maximum(new_seq, 0.0)
    tot = new_seq.sum()
    if tot > 0:
        new_seq /= tot
    else:
        new_seq = np.full_like(new_seq, 1.0 / n)

    return new_seq.tolist()


def gradient_step(seq, step):
    """
    One (projected) gradient‑descent step that reduces the current maximal
    convolution entry.  The step size `step` controls how far we move.
    """
    a = np.asarray(seq, dtype=float)
    n = len(a)

    # Locate the index k* where convolution attains its maximum
    b = np.convolve(a, a)
    k_star = int(np.argmax(b))

    # Sub‑gradient of max entry w.r.t. a_i: 2·a_{k*‑i} (if partner index in range)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]

    # Move opposite to the gradient, project to non‑negative orthant
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)

    # Re‑normalise to unit L1‑norm (the objective is homogeneous)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        # degenerate case – fall back to uniform
        a_new = np.full_like(a_new, 1.0 / n)

    return a_new.tolist()


def random_mutate(seq, sigma=0.01, n_mut=5):
    """Small Gaussian perturbations of a few coordinates, followed by re‑normalisation."""
    a = np.array(seq, dtype=float)
    n = len(a)
    for _ in range(n_mut):
        i = np.random.randint(n)
        a[i] = max(0.0, a[i] + np.random.randn() * sigma)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a.tolist()
