def project_onto_simplex(v, target):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - target) / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        # All zero? then just return uniform?
        theta = (cssv[-1] - target) / len(v)
        w = np.maximum(v - theta, 0)
        return w
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, z=1):
    # Projects v onto the simplex {x: x >= 0, sum x = z}
    if z < 0:
        raise ValueError
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def project_to_simplex(v, z):
    """Project vector v onto the simplex {x >=0, sum x = z}."""
    if z < 0:
        raise ValueError
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    # find rho
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros case
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    within the given time budget."""
    import time, numpy as np, random, math
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a little margin

    best_seq = None
    best_score = float('inf')

    # try several lengths (the factor n appears only in the objective,
    # but larger n offers more degrees of freedom)
    lengths = [120, 240, 480]   # modest sizes that are fast to evaluate
    # allocate roughly equal time per length
    per_len = (deadline - start_time) / len(lengths)

    for idx_len, n in enumerate(lengths):
        # target L1‑norm that makes the evaluation equal to the max entry
        target_sum = math.sqrt(2.0 * n)

        # → a few thousand iterations fit easily.
        max_iters = 8000

        a = np.full(n, target_sum / n, dtype=np.float64)

"""
Gradient‑descent search for a non‑negative step‑function that minimizes the
evaluation function.  The objective 2·n·max(conv(a,a))/sum(a)^2 is invariant
under scaling, so we fix ‖a‖₁ = sqrt(2·n).  The algorithm performs a projected
gradient descent on a smooth approximation of the max‑convolution (soft‑max)
and uses a simple annealing schedule plus occasional random perturbations.
Multiple sequence lengths are tried within the allotted time budget.
"""

import time
import math
import random
import numpy as np

def project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Project a vector `v` onto the simplex {x >= 0, sum(x) = z}.
    Implements the O(n log n) algorithm from:
    Duchi et al., “Efficient Projections onto the L1‑Ball for Learning in
    High Dimensions”, ICML 2008.
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1  # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # This can happen only if z == 0, return the zero vector
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) improves
    evaluate_sequence.  The routine runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # keep a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # Different lengths give different degrees of freedom.
    # The chosen values keep the per‑iteration cost modest.
    lengths = [120, 240, 480]

    # Allocate roughly equal time to each length.
    per_len = (deadline - start) / len(lengths)

    for li, n in enumerate(lengths):
        target_sum = math.sqrt(2.0 * n)        # fixes the scaling of the score
        # Record the moment we must leave this length.
        length_deadline = start + (li + 1) * per_len

        # Initial vector: uniform distribution (feasible by construction)
        a = np.full(n, target_sum / n, dtype=np.float64)

        # Parameters for the smooth‑max approximation
        alpha = 1.0              # “temperature”, will be increased gradually
        max_iters = 8000

        # Local best (helps with occasional restarts)
        local_best_score = None
        local_best_a = None

        for it in range(max_iters):
            if time.time() >= length_deadline:
                break

            # ----- evaluate current sequence -----
            conv = np.convolve(a, a)                # linear convolution
            max_conv = conv.max()
            score = max_conv                        # because sum(a)^2 = 2·n

            if score < best_score:
                best_score = float(score)
                best_seq = a.copy()
                print(f"[{time.time() - start:6.2f}s] New global best: "
                      f"{best_score:.6f} (n={n}, iter={it})")

            # Occasionally verify with the official evaluator (safe guard)
            if it % 250 == 0:
                try:

                    if eval_val < best_score:
                        best_score = eval_val
                        best_seq = a.copy()
                        print(f"[{time.time() - start:6.2f}s] Updated from "
                              f"evaluate_sequence: {best_score:.6f}")
                except Exception:
                    pass

            # ----- smooth‑max gradient -----
            # Soft‑max distribution over convolution entries (shifted to avoid overflow)
            shift = conv.max()
            w = np.exp(alpha * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:

            a_rev = a[::-1]
