"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(sequence,sequence)) / (sum(sequence)**2)

The search is based on the following intuition.

If the sequence is viewed as a probability distribution (its sum can be
scaled arbitrarily because the objective is homogeneous of degree 0),
the quantity to minimise becomes 2 · n · max(conv).  For a uniform
distribution the convolution is triangular and the maximum equals 1/n,
so the objective equals 2.  A smoother distribution, e.g. a discrete
Gaussian, spreads the mass and reduces the peak of the self‑convolution.
The continuous Gaussian gives the analytic estimate

    max(conv) ≈ 1 / (σ·√(4π))

so the objective is roughly 2·n/(σ·√(4π)).  Choosing σ ≈ n/2 gives an
objective near 1.13, well below the target 1.503.

The algorithm therefore

1. Takes the length *n* of the supplied `height_sequence_1`
   (or a default length if it is missing).
2. Scans a few values of σ = scale·n (scale in [0.2, 0.9]) and evaluates the
   corresponding normalised Gaussian sequences.
3. Keeps the best sequence found.
4. Performs a cheap simulated‑annealing / hill‑climbing phase for the
   remaining time: a small amount of mass is moved from one random index
   to another, and the move is kept if it improves the objective (with a
   tiny probability of accepting a worse move to escape shallow minima).

The whole procedure respects the 1000 s budget and never returns a
negative or non‑finite value.  The final list is returned as a plain
Python list of floats.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that
    minimises `evaluate_sequence`.  The method constructs and refines
    discrete Gaussian profiles.
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # a little safety margin

    # Length of the sequence to work with.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 1500                      # fallback length if the global is missing

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Grid search over Gaussian width (σ = scale * n)
    # ------------------------------------------------------------------
    # We try a modest number of scales; each evaluation is cheap.
    scales = np.linspace(0.2, 0.9, 15)        # σ = scale·n, 0.2·n … 0.9·n
    idx = np.arange(n)
    centre = (n - 1) / 2.0

    for scale in scales:
        sigma = scale * n
        if sigma <= 0:
            continue
        # Discrete Gaussian (symmetric)
        raw = np.exp(-((idx - centre) ** 2) / (2.0 * sigma * sigma))
        # Normalise to sum = 1 (objective is scale‑invariant)
        seq = raw / raw.sum()
        # Evaluate – convert to plain Python list to be safe
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
        # Early exit if we already beat the target comfortably
        if best_score < 1.3:
            break

    # ------------------------------------------------------------------
    # 2. Local stochastic improvement (simulated‑annealing style)
    # ------------------------------------------------------------------
    # If the grid search failed (should not happen) fall back to a uniform list.
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # Work on a mutable copy
    cur_seq = np.array(best_seq, dtype=float)

    # Temperature schedule (very simple linear cooling)
    start_temp = 0.02
    end_temp = 0.0005

    while time.time() < deadline:
        # Linear cooling based on remaining time
        remaining = max(deadline - time.time(), 0.0)
        frac = remaining / (budget_s - 5.0)
        temp = end_temp + (start_temp - end_temp) * frac

        # Propose moving a small amount of mass from i to j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Choose a delta that respects non‑negativity
        max_delta = cur_seq[i]
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * 0.05   # at most 5 % of the source entry
        # Apply the move
        cur_seq[i] -= delta
        cur_seq[j] += delta

        # Evaluate the perturbed sequence
        score = evaluate_sequence(cur_seq.tolist())

        # Accept if improvement, otherwise accept with small probability
        accept = False
        if score < best_score:
            best_score = score
            best_seq = cur_seq.tolist()
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(score - best_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if not accept:
            # revert the move
            cur_seq[i] += delta
            cur_seq[j] -= delta

        # Occasionally re‑normalise (purely cosmetic – scaling does not affect score)
        if rng.random() < 0.001:
            cur_seq /= cur_seq.sum()

    print(f"Best evaluation score achieved: {best_score:.6f}")
    return best_seq
