def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    n = 6000  # approximate length
    # generate initial sequence a_i = 1/sqrt(i+0.5)
    a = 1.0 / np.sqrt(np.arange(n) + 0.5)
    # normalize sum to sqrt(2n)
    target_sum = np.sqrt(2 * n)
    a *= target_sum / a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # gradient descent loop
    start = time.time()
    deadline = start + budget_s - 5
    gamma = 10.0
    lr = 0.1
    while time.time() < deadline:
        # compute convolution via FFT
        fft_a = np.fft.rfft(a, n*2)  # size to handle convolution 2n
        conv = np.fft.irfft(fft_a * fft_a)[:2*n-1]
        max_c = conv.max()
        # compute smooth weights
        # shift to avoid overflow:
        w = np.exp(gamma * (conv - max_c))  # note max_c subtraction to keep w <=1
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0
        w /= w_sum
        # gradient = 2 * conv(w, a_rev)
        grad = 2 * np.convolve(w, a[::-1])[n-1:2*n-1]
        # update a
        a -= lr * grad
        a = np.maximum(0, a)
        # renormalize sum
        a_sum = a.sum()
        if a_sum == 0:
            a = np.ones_like(a) * target_sum / n
            a_sum = target_sum
        a *= target_sum / a_sum
        # evaluate
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Improved score {best_score:.6f} at time {time.time()-start:.1f}s")
        # adjust learning rate adaptively
        if max_c < best_score:
            # maybe increase lr
            lr = min(lr * 1.05, 1.0)
        else:
            lr = max(lr * 0.95, 1e-4)
    return best_seq.tolist()

        af = np.fft.rfft(a, n=nfft1)

        w = np.exp(gamma * (conv - max_c))
        w_sum = w.sum()

        a = a - lr * grad

        a = np.maximum(a, 0.0)

        if s == 0:

"""
Propose a sequence that minimizes the evaluation function
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm:
1. Initialise a smooth decreasing sequence a_i = 1 / sqrt(i+0.5) and
   normalise it so that sum(a) = sqrt(2 * n).  This shape already gives a
   value close to the theoretical optimum (≈π/2 ≈ 1.57).

2. Iteratively minimise the smoothed maximum of the convolution.
   For a given a we compute the convolution c = a * a.  The (soft) maximum
   is approximated by a soft‑max with temperature γ:
        w_k = exp(γ * (c_k - max(c)))  ,  sum w = 1
   The gradient of this smooth max with respect to a is
        g = 2 * ( w  ⊛  a_rev )               (⊛ = cross‑correlation)
   where a_rev is a reversed copy of a.  A gradient step
        a ← a - η * g
   (followed by a projection onto the non‑negative orthant and renormalisation
   to keep sum(a)=sqrt(2n)) reduces the true maximum of the convolution.
   γ is slowly increased (focusing the optimisation on the true maximum)
   and the step size η is slowly decreased.

3. Occasionally inject a tiny random perturbation to escape possible
   plateaux.

The whole procedure runs for the allocated time budget (default 1000 s)
and the best sequence seen is returned.
"""

import numpy as np
import time

def _next_pow2(x: int) -> int:
    """Return the next power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _compute_score(a: np.ndarray) -> float:
    """Fast computation of the evaluation function (without the security wrapper)."""
    n = a.shape[0]
    # length needed for a*a convolution
    nfft = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[: 2 * n - 1]
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a sequence of non‑negative heights whose evaluation score is
    (hopefully) lower than the current best bound (≈2.0).  The method uses
    a smoothed max gradient descent starting from the analytically motivated
    a_i = 1 / sqrt(i+0.5) shape.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # a small safety margin

    # Use the length of the reference construction if it exists,
    # otherwise fall back to a reasonable size.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 5000

    # ------------------------------------------------------------
    # 1.  Initialise a_i = 1 / sqrt(i + 0.5) and normalise.
    # ------------------------------------------------------------
    a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
    target_sum = np.sqrt(2.0 * n)
    a *= target_sum / a.sum()

    # Keep track of the best sequence found.
    best_a = a.copy()
    best_score = _compute_score(best_a)

    # FFT sizes required repeatedly.
    nfft_conv = _next_pow2(2 * n - 1)          # for a*a
    nfft_corr = _next_pow2(3 * n - 2)          # for w ⊛ a_rev

    # Hyper‑parameters (will be slowly adapted).
    gamma = 5.0          # soft‑max temperature
    gamma_max = 80.0
    eta = 0.12           # gradient step size
    eta_min = 0.001

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------
        # 2. Compute convolution a * a  (real FFT, length 2n‑1)
        # --------------------------------------------------------
        af = np.fft.rfft(a, n=nfft_conv)
        conv = np.fft.irfft(af * af, n=nfft_conv)[: 2 * n - 1]

        max_conv = conv.max()

        # --------------------------------------------------------
        # 3. Build soft‑max weights over the convolution entries.
        # --------------------------------------------------------
        # Subtract max to avoid overflow; max weight will be 1.
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:                # numerical safety
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0
        w /= w_sum

        # --------------------------------------------------------
        # 4. Gradient of the smooth maximum:
        #    g = 2 * ( w ⊛ a_rev )
        # --------------------------------------------------------
        wf = np.fft.rfft(w, n=nfft_corr)
        af_rev = np.fft.rfft(a[::-1], n=nfft_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=nfft_corr)
        # The valid part starts at index n-1 and has length n.
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # --------------------------------------------------------
        # 5. Gradient step with projection onto the non‑negative orthant.
        # --------------------------------------------------------
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)

        # Renormalise to keep the sum fixed (scaling does not change the score).
        s = a_new.sum()
        if s == 0.0:
            # Extremely unlikely, but recover a uniform vector.
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new *= target_sum / s

        # Occasionally add a tiny random jitter to help escape flat regions.
        if iteration % 73 == 0:
            jitter = 1e-4 * np.random.randn(n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new *= target_sum / a_new.sum()

        # --------------------------------------------------------
        # 6. Evaluate the new candidate.
        # --------------------------------------------------------
        score_new = _compute_score(a_new)

        # Accept the new point if it improves (or with a small probability
        # for uphill moves – simulated‑annealing style).
        accept = False
        if score_new < best_score:
            accept = True
        else:
            # Simple annealing: temperature is proportional to eta.
            temp = eta
            if np.random.rand() < np.exp(-(score_new - best_score) / (temp + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
                elapsed = time.time() - start_time
                print(
                    f"Iter {iteration:6d}: new best score {best_score:.6f} "
                    f" (elapsed {elapsed:.1f}s)"
                )
        else:
            # reject – keep the old a unchanged
            pass

        # --------------------------------------------------------
        # 7. Update hyper‑parameters (slow cooling / focusing).
        # --------------------------------------------------------
        gamma = min(gamma * 1.003, gamma_max)
        eta = max(eta * 0.998, eta_min)

    # Return the best sequence discovered (as a plain Python list).
    return best_a.tolist()
