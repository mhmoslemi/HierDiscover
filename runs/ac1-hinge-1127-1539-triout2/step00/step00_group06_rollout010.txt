def convolve_self(a):
    n = len(a)
    size = 2*n - 1
    # nearest power of 2 for faster FFT
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:size]
    return conv

"""
propose_candidate
=================

A stochastic local‑search that tries to minimise

    f(a) = 2·n·max(conv(a,a)) / (sum(a))**2

for a non‑negative step function a.  
The score is scale‑invariant, therefore we keep the ℓ¹‑norm equal to 1
throughout the search and minimise   2·n·max(conv(a,a)) .

The optimiser works with several restarts, each of them consisting of:

* a random Dirichlet start (or the provided `height_sequence_1`);
* an annealing loop that repeatedly tweaks the vector:
  – with high probability a “targeted” move that removes mass from the
    pair of indices that contributes most to the currently maximal
    convolution coefficient;
  – otherwise a small random mass exchange between two positions.
* the convolution is evaluated with an FFT ( O(n log n) ) – fast enough
  for n up to a few ten‑thousands and many thousands of iterations.

Whenever a better score is found it becomes the new global best.
The routine respects the 1000 s time budget and returns the best
sequence discovered (as a plain ``list[float]``) before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself using a real FFT."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for the FFT – speeds up the transform
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:size]
    return conv


def _objective(a: np.ndarray) -> float:
    """Return the value that evaluate_sequence would return (sum(a) == 1)."""
    n = a.shape[0]
    max_b = _fft_convolve_self(a).max()
    return 2.0 * n * max_b  # sum(a) == 1, so denominator omitted


def _random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a random point on the probability simplex (α=1)."""
    x = rng.exponential(scale=1.0, size=n)
    return x / x.sum()


def _targeted_move(rng: np.random.Generator, a: np.ndarray, conv: np.ndarray,
                   max_idx: int, eps: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Produce a new vector by moving a small amount of probability mass away
    from the pair of indices that contributes most to the maximal
    convolution coefficient ``max_idx``.
    """
    n = a.shape[0]
    # admissible i such that i and j = max_idx-i lie inside the vector
    i_min = max(0, max_idx - (n - 1))
    i_max = min(n - 1, max_idx)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    jdx = max_idx - idx
    # contribution of each ordered pair (i, jdx[i])
    contrib = a[idx] * a[jdx]

    # choose the source index from the largest contribution
    src_choice = idx[np.argmax(contrib)]
    # also look at its partner – we will empty the larger of the two
    partner = max_idx - src_choice
    src = src_choice if a[src_choice] >= a[partner] else partner

    # destination is a random index different from src
    dst = rng.integers(0, n - 1)
    while dst == src:
        dst = rng.integers(0, n - 1)

    # amount to move (up to ``eps`` fraction of the source entry)
    max_delta = eps * a[src]
    if max_delta <= 0.0:
        return a, conv
    delta = rng.random() * max_delta

    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta

    # recompute the whole convolution (FFT, cheap enough)
    conv_new = _fft_convolve_self(a_new)
    return a_new, conv_new


def _random_move(rng: np.random.Generator, a: np.ndarray, eps: float) -> np.ndarray:
    """
    Small random exchange of mass between two distinct positions.
    """
    n = a.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a
    # amount to transfer from i to j (or opposite if negative)
    delta = (rng.random() - 0.5) * 2.0 * eps * a[i]   # up to ±eps·a[i]
    # clamp to keep both entries non‑negative
    if delta > a[i]:
        delta = a[i]
    if -delta > a[j]:
        delta = -a[j]
    if abs(delta) < 1e-15:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine spends at most ``budget_s`` seconds,
    respects the 1‑CPU constraint and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # possible lengths – they influence the objective linearly (2·n·M)
    # longer vectors give more room to reduce the max convolution.
    lengths = [500, 1000, 2000, 3000, 4000, 5000, 6000, 8000]

    # initialise with the provided SOTA sequence (if it exists)
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=float)
        init = init / init.sum()        # normalise to ℓ¹=1
        score = _objective(init)
        best_seq, best_score = init, score
        print(f"[init] height_sequence_1 score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – many restarts until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----------------------------------------------------------------
        # 1) pick a length and generate a start point on the simplex
        # ----------------------------------------------------------------
        n = int(rng.choice(lengths))
        a = _random_dirichlet(rng, n)          # sum == 1 by construction
        conv = _fft_convolve_self(a)
        cur_score = 2.0 * n * conv.max()
        # keep the best of this restart if it already beats global best
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[new best restart] n={n} score={best_score:.6f}")

        # ----------------------------------------------------------------
        # 2) annealing / local search inside this restart
        # ----------------------------------------------------------------
        # temperature schedule: linear decay from 1 → 0
        inner_deadline = time.time() + 2.0    # at most a few seconds per restart
        while time.time() < inner_deadline and time.time() < deadline:
            # linear temperature (0 … 1)
            t_frac = (deadline - time.time()) / (deadline - start_time)
            # eps controls how much mass we move – shrinks with temperature
            eps = 0.30 * t_frac + 0.01      # between ~0.01 and ~0.31

            # ----------------------------------------------------------------
            # 2a) decide move type (targeted vs random)
            # ----------------------------------------------------------------
            if rng.random() < 0.7:   # majority are targeted moves
                max_idx = int(conv.argmax())
                a_candidate, conv_candidate = _targeted_move(
                    rng, a, conv, max_idx, eps
                )
            else:
                a_candidate = _random_move(rng, a, eps)
                conv_candidate = _fft_convolve_self(a_candidate)

            cand_score = 2.0 * n * conv_candidate.max()

            # ----------------------------------------------------------------
            # 2b) Metropolis acceptance
            # ----------------------------------------------------------------
            if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / (0.001 + t_frac)):
                a, conv, cur_score = a_candidate, conv_candidate, cand_score
                if cur_score < best_score:
                    best_score, best_seq = cur_score, a.copy()
                    print(f"[improve] n={n} score={best_score:.6f}")

        # ----------------------------------------------------------------
        # occasionally break if we are very close to the budget
        # ----------------------------------------------------------------
        if time.time() > deadline - 1.0:
            break

    # --------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------------
    # Ensure positivity and normalise once more (tiny numerical drift)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq /= best_seq.sum() if best_seq.sum() > 0 else 1.0
    print(f"[final] best length={len(best_seq)} best score={best_score:.6f}")
    return best_seq.tolist()
