"""
Propose a step‐function (a list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear convolution used in ``evaluate_sequence``.
The method consists of three stages:

1. **Geometric seed** – a decreasing geometric sequence `r**i` (0≤i<n) is
   searched on a coarse grid to obtain a good initial bound (typically
   ~1.8, already better than the uniform 2.0).

2. **Local refinements** – a fast hill‑climbing / simulated‑annealing
   loop moves a little mass from indices that contribute most to the
   current maximal convolution towards random positions.  Occasionally
   random swaps are tried and, with low probability, a linear‑programming
   step (the same LP used in the reference implementation) is applied
   to obtain a better descent direction.

3. **Best‑ever record** – the best sequence found during the allotted
   time is returned.

The algorithm respects all security checks of ``evaluate_sequence``:
all entries stay non‑negative, never exceed 1000, and the sum stays
strictly positive.  No I/O, no closures, and everything runs on a
single CPU within the provided ``budget_s`` time limit.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise   sum(b)
        subject to  conv(f, b) <= rhs   (for every possible shift)
                    b >= 0
    Returns the optimal b as a NumPy array or ``None`` if the LP fails.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  --> minimise -sum(b)
    c = -np.ones(n)

    # build the constraint matrix for each convolution index k
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # b_j appears in conv_k with coefficient f_{k-j} if 0<=k-j<n
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # fill row[j] = f_seq[k-j] for the valid j range
        # we can vectorise this slice
        js = np.arange(i_min, i_max + 1)
        row[js] = f_seq[k - js]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints  b >= 0  -->  -b <= 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # ``highs`` is fast and supports a time limit via the ``max_iter`` option.
    try:
        res = linprog(
            c,
            A_ub=a_ub,
            b_ub=b_ub,
            method="highs",
            options={"presolve": True, "max_iter": 1000},
        )
    except Exception:          # pragma: no cover – safety net
        return None

    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(seq):
    """
    Given a current sequence ``seq`` return a new sequence obtained by the
    LP‑based direction used in the reference code.  ``None`` is returned if
    the LP does not succeed.
    """
    n = len(seq)
    seq = np.asarray(seq, dtype=float)
    total = np.sum(seq)
    if total == 0:                     # should never happen
        return None
    # normalise to the usual ``sqrt(2n)`` convention
    norm_seq = seq * np.sqrt(2 * n) / total
    rhs = np.max(np.convolve(norm_seq, norm_seq))
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None
    # renormalise the LP solution back to the original total mass
    g = np.asarray(g)
    g_total = np.sum(g)
    if g_total == 0:
        return None
    g = g * np.sqrt(2 * n) / g_total
    # small step towards the direction (t=0.01)
    t = 0.01
    new_seq = (1 - t) * seq + t * g
    return new_seq.tolist()
