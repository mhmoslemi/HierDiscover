    n = 500 # maybe
    target_sum = np.sqrt(2.0 * n)
    # Initialize a: random positive values, normalized sum
    a = np.random.rand(n)
    a = a / a.sum() * target_sum
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    beta = 10.0
    eta = 0.1 * target_sum / n # step size maybe.

    deadline = time.time() + budget_s - 2

    while time.time() < deadline:
        # compute conv using FFT
        size = 1 << ((2*n-1).bit_length())
        fft_a = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
        max_b = conv.max()
        # check current score (same as evaluate)
        score = 2.0 * n * max_b / (a.sum() ** 2)
        # compute softmax weights
        # subtract max to avoid overflow
        weighted = np.exp(beta * (conv - conv.max()))
        w = weighted / weighted.sum()
        # compute gradient
        g = 2.0 * np.correlate(w, a, mode='valid')
        # gradient descent step
        a_new = a - eta * g
        # enforce nonnegativity
        a_new = np.maximum(0, a_new)
        # re-normalize to target sum
        if a_new.sum() == 0:
            a_new = a # avoid division by zero
        else:
            a_new = a_new * (target_sum / a_new.sum())
        # evaluate new
        new_score = evaluate_sequence(a_new.tolist())
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"Iter {iter}: improved score {best_score}")
        # accept if improvement else maybe revert
        if new_score <= score:
            a = a_new
        # else keep old a (or random perturbation)
        # optionally adjust beta / eta adaptively

    return best_a.tolist()

"""
Propose a step‑function heights vector that beats the naïve constant bound (score 2).

The algorithm works with a moderate length (default 300) to keep each iteration
fast.  It uses a simple simulated‑annealing / hill‑climbing scheme:

* Initialise a random non‑negative sequence and normalise its ℓ₁‑norm to
  √(2·n) – this normalisation makes the evaluation function scale‑invariant.
* In each iteration compute the full self‑convolution (via an FFT) and its
  maximum value.
* With probability 0.7 a *targeted* move is performed:
      – Locate the shift kₘₐₓ where the convolution attains its maximum.
      – For that shift compute the contribution aᵢ·a_{kₘₐₓ-i} of every index.
      – Move a small fraction (≈10 %) of the entry that contributes most
        to a random entry that contributes the least.
  This directly reduces the dominant term of the max‑convolution.
* Otherwise perform a cheap random mass‑transfer between two positions.
* Accept the move if the score does not increase; otherwise keep the old
  sequence (a simple greedy descent, which is sufficient for the limited
  time budget).
* Keep the best sequence ever seen.

The loop runs until the allotted time (≈ 990 s) expires, then returns the
best sequence found.  The code avoids any filesystem or network I/O and
relies only on NumPy (and the already‑imported ``evaluate_sequence``)."""

import numpy as np
import time


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the score 2·n·max(b)/(sum(a)²)."""
    n = a.size
    # linear convolution via FFT (real‑to‑real)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = np.max(conv)
    s = a.sum()
    # protect against division by zero – the outer loop never lets this happen
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)

    The algorithm uses a simulated‑annealing style optimisation described above.

    while time.time() < end_time:

        n = a.size

        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
        current_score = 2.0 * n * np.max(conv) / (a.sum() ** 2)

        if np.random.rand() < prob_targeted:

            idx = np.arange(n, dtype=np.int64)
            j = k_max - idx
