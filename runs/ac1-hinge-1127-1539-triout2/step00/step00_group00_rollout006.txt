"""
Optimiser for the self‑convolution bound.

The algorithm works in three stages:

1. **Parametric sweep** – simple “U‑shaped’’ families (linear,
   quadratic and exponential increase towards the edges) are tried.
   The best of these shapes is kept as a starting point.

2. **Projected sub‑gradient descent** – for the current sequence the
   indices where the convolution attains its maximum are identified.
   The sub‑gradient of the max‑convolution is computed and a step
   orthogonal to the all‑ones direction is taken (so the ℓ₁‑norm,
   i.e. the total sum, stays unchanged).  After each step negatives
   are clipped and the vector is re‑scaled to the required sum
   ( √(2 n) ).  This directly reduces the maximal auto‑convolution.

3. **Random pairwise perturbations** – occasional transfers of a tiny
   amount of mass between two random positions are performed to jump
   out of possible plateaus.  If an improvement is observed the new
   sequence becomes the current best.

The whole search respects the time budget (≈1000 s) and always returns
the best sequence found.  All numbers are non‑negative, the sum is kept
positive and the length stays constant, satisfying the requirements of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures or lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using a power‑of‑two FFT.
    Returns the full convolution of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())            # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector to have the prescribed ℓ₁‑norm.
    Zero entries stay zero; if the whole vector is zero a uniform
    distribution is returned.
    """
    s = v.sum()
    if s > 0:
        return v * (target / s)
    # fallback – uniform distribution
    return np.full_like(v, target / v.shape[0])


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param controls the strength of the edge‑enhancement (≥0).
    The result is scaled so that the ℓ₁‑norm equals target_sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_to_sum(raw, target_sum)


def subgradient_step(
    a: np.ndarray,
    target_sum: float,
    eta: float,
    rel_tol: float = 0.0,
) -> np.ndarray:
    """
    One projected sub‑gradient step aimed at decreasing the maximum
    entry of the self‑convolution.
    * rel_tol* determines the size of the active set:
    all convolution entries ≥ (1‑rel_tol)·max are considered.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_val = conv.max()
    # active indices (those that are (almost) maximal)
    if rel_tol <= 0.0:
        act = np.array([np.argmax(conv)], dtype=np.int64)
    else:
        act = np.where(conv >= max_val * (1.0 - rel_tol))[0]

    # sub‑gradient of the max‑convolution
    g = np.zeros_like(a)
    idx = np.arange(n, dtype=np.int64)
    for k in act:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]              # contribution from pair (i, k‑i)

    # make the direction orthogonal to the all‑ones vector (preserve sum)
    direction = -g + np.mean(g)                  # sum(direction)=0
    a_new = a + eta * direction
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_to_sum(a_new, target_sum)
    return a_new


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small amount of mass from one random index to another.
    The total sum is preserved by construction.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # transfer a tiny fraction of the source entry
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    # re‑scale to keep the exact target sum (helps against rounding errors)
    return project_to_sum(a, target_sum)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The function runs for at most *budget_s*
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Choose a length – use the known construction if available
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)                               # keep the same size
    else:
        n = np.random.randint(7000, 9000)           # a large but manageable size

    target_sum = np.sqrt(2.0 * n)                    # the normalisation used in the paper

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep over simple U‑shapes
    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)            # uniform starting point
    best_score = float("inf")

    shape_kinds = ("linear", "quadratic", "exponential")
    # small grids for each shape family
    param_grids = {
        "linear": np.linspace(0.0, 4.0, 41),          # 0 … 4 in steps of 0.1
        "quadratic": np.linspace(0.0, 3.0, 31),      # 0 … 3
        "exponential": np.linspace(0.0, 2.0, 21),    # 0 … 2
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_U_shape(n, target_sum, kind=kind, param=p)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best after {kind} sweep (p={p:.2f}): {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    eta = 0.02 * (target_sum / n)          # initial step size (≈2 % of the average entry)
    rel_tol = 0.0                          # start with the strict max index
    no_improve = 0
    iteration = 0

    print(f"[{time.time():.2f}] Entering sub‑gradient descent (eta={eta:.5f})")
    while time.time() < deadline:
        a = subgradient_step(a, target_sum, eta, rel_tol=rel_tol)
        score = evaluate_sequence(a.tolist())
        iteration += 1

        if score + 1e-12 < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration:06d} – improved to {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # adapt step size and tolerance slowly
        if iteration % 500 == 0:
            eta *= 0.9                # gradual decay
            if eta < 1e-6:
                eta = 1e-6
            # after a while broaden the active set – makes the descent smoother
            rel_tol = min(0.05, rel_tol + 0.005)

        # occasional random pair transfer to escape plateaus
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.005)

        # if stagnation lasts too long, inject a tiny random jitter
        if no_improve >= 3000:
            jitter = np.random.randn(n) * 0.001
            a = np.clip(a + jitter, 0.0, None)
            a = project_to_sum(a, target_sum)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied")

    # ------------------------------------------------------------------
    # Phase 3 – a few final random perturbations (optional)
    # ------------------------------------------------------------------
    # (we keep this short; the main work is already done)
    print(f"[{time.time():.2f}] Sub‑gradient phase finished; best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
