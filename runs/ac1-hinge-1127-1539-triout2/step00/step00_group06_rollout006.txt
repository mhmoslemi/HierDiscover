def propose_candidate(seed=0, budget_s=1000):

    np.random.seed(seed)

    best_seq = None
    best_score = float('inf')

    # Determine candidate lengths
    lengths = [80, 120, 180, 250, 350]   # moderate
    # might adapt based on time

    for n in lengths:
        # initial distribution can be uniform, or random Dirichlet
        p = np.ones(n) / n
        # optionally perturb
        # set parameters
        alpha = 5.0
        lr = 0.3
        iteration = 0
        # track best for this n
        local_best_score = float('inf')
        local_best_p = p.copy()
        while time.time() < deadline and iteration < 20000:
            # compute conv and score
            conv = np.convolve(p, p)
            M = conv.max()
            score = 2 * n * M  # sum=1
            if score < local_best_score:
                local_best_score = score
                local_best_p = p.copy()
                if score < best_score:
                    best_score = score
                    best_seq = p.copy()
                    print(f"New global best: score {score:.6f} len {n} iter {iteration}")
            # compute gradient of smooth max
            # use increasing alpha schedule maybe
            conv = np.convolve(p, p)
            # compute q = softmax(alpha*conv)
            expvals = np.exp(alpha * conv - np.max(alpha * conv))  # to avoid overflow
            q = expvals / expvals.sum()
            # compute gradient
            # g_i = sum_{k} q_k * (p_{k-i} + delta{k=2i} * p_i)
            # we'll compute via loops
            g = np.zeros(n)
            # precompute for each i contributions

            for i in range(n):

                # j indexes from 0..n-1 such that k=i+j < L

                if 2*i < L:

            # exponentiated gradient update
            p = p * np.exp(-lr * g)
            # renormalize
            p_sum = p.sum()
            if p_sum == 0:
                p = np.ones(n) / n
            else:
                p = p / p_sum
            # optionally project small values to zero for sparsity?

    if best_seq is None:

"""
Propose a step‑function (non‑negative heights) that tries to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

The score is invariant to scaling, therefore we optimise a probability
vector `p` (sum(p)=1) and later return the scaled vector.
The optimisation is performed by a smooth approximation of the
`max`‑operator (soft‑max) together with an exponentiated‑gradient
update (mirror descent on the simplex).  The temperature of the
soft‑max (`alpha`) is increased and the learning rate is decreased
during the run – a simple annealing schedule.  Several problem sizes
(`n`) are tried and the best candidate found before the time budget
expires is returned.

The algorithm works entirely with NumPy, uses only O(n log n) work per
iteration (via FFT based convolution) and respects all the
constraints enforced by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def convolution(p: np.ndarray) -> np.ndarray:
    """Return the full linear convolution p * p."""
    return np.convolve(p, p)


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights q_k = exp(alpha * conv_k) / sum_k exp(alpha * conv_k)
    in a numerically stable way.
    """
    scaled = alpha * conv
    # subtract the maximum for stability
    scaled -= np.max(scaled)
    exp_vals = np.exp(scaled)
    return exp_vals / exp_vals.sum()


def gradient_exp(p: np.ndarray, q: np.ndarray, conv_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(p,p))
    with respect to the probability vector `p`.  The formula is

        g_i = Σ_k q_k (p_{k-i}) + q_{2i} * p_i

    where the first term is a convolution of q with the reversed p.
    """
    n = p.shape[0]

    # convolution of q with reversed p gives Σ_k q_k p_{k-i}
    rev_p = p[::-1]
    conv_q_rev = np.convolve(q, rev_p)          # length = (2n-1)+(n-1) = 3n-2

    # slice the part that corresponds to i = 0 … n‑1
    g = conv_q_rev[n - 1 : n - 1 + n].copy()

    # add the diagonal contribution q_{2i} * p_i
    diag_idx = 2 * np.arange(n)
    mask = diag_idx < conv_len
    g[mask] += q[diag_idx[mask]] * p[mask]

    return g


def exponentiated_update(p: np.ndarray, g: np.ndarray, lr: float) -> np.ndarray:
    """
    Perform a mirror‑descent step on the simplex using the
    exponentiated‑gradient rule: p_i ← p_i * exp(-lr * g_i) and re‑normalise.
    """
    # avoid overflow/underflow by clipping the exponent
    step = -lr * g
    step = np.clip(step, -50, 50)                # reasonable numeric limits
    p = p * np.exp(step)
    total = p.sum()
    if total <= 0 or np.isnan(total):
        # fall back to uniform if something went wrong
        n = p.shape[0]
        p = np.full(n, 1.0 / n)
    else:
        p /= total
    return p


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence` metric.  The routine runs a few
    independent optimisations for different lengths (n) using a
    smoothed max‑convolution objective and returns the best sequence
    found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # keep a tiny margin

    best_score = float("inf")
    best_vec = None

    # Candidate lengths – a modest range that fits comfortably into the time budget
    lengths = [80, 120, 180, 250, 350]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- initialise a probability vector --------------------------------
        p = np.full(n, 1.0 / n)
        # a tiny random jitter helps to break symmetry of the uniform start
        p += 0.001 * np.random.rand(n)
        p /= p.sum()

        # ---- hyper‑parameters (will be annealed) ----------------------------
        alpha = 3.0          # softness of the max‑approximation
        lr = 0.6             # learning rate for the exponentiated update
        max_iter = 30000    # upper bound – real stop comes from the time budget
        it = 0

        # ---- optimisation loop ---------------------------------------------
        while time.time() < deadline and it < max_iter:
            # convolution and score (note: sum(p)==1 → denominator = 1)
            conv = convolution(p)
            M = conv.max()
            score = 2.0 * n * M

            # keep the best candidate seen so far (global best)
            if score < best_score:
                best_score = score
                best_vec = p.copy()
                print(f"[{time.time():.1f}] New best → score={score:.6f}, n={n}, iter={it}")

            # soft‑max weights for the smooth max
            q = softmax_weights(conv, alpha)

            # gradient of the smoothed objective
            g = gradient_exp(p, q, conv_len=conv.shape[0])

            # mirrored (exponentiated) gradient step on the simplex
            p = exponentiated_update(p, g, lr)

            # ---- simple annealing schedule ---------------------------------
            if (it + 1) % 800 == 0:
                alpha *= 1.05          # sharpen the approximation to the true max
                lr *= 0.93             # slowly reduce the step size

            it += 1

    # ----------------------------------------------------------------------
    # Fallback in the very unlikely case that nothing was found
    if best_vec is None:
        best_vec = np.array([0.5, 0.5])   # a trivial two‑point sequence

    # Scale the vector (the score is scale‑invariant, the scaling only
    # prevents extremely tiny numbers that could be rounded to zero).
    best_vec = best_vec * 10.0

    # Return a plain Python list as required by the harness
    return best_vec.tolist()
