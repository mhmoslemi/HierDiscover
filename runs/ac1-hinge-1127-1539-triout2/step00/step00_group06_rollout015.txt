def build_sidon_sequence(n):
    # use greedy to produce Sidon set up to size k
    used = np.zeros(2*n-1, dtype=bool)
    pos = []
    for i in range(n):
        ok = True
        for p in pos:
            if used[p+i]:
                ok = False
                break
        if ok and not used[2*i]:
            # add
            for p in pos:
                used[p+i] = True
            used[2*i] = True
            pos.append(i)
    # assign value 1 to each position
    seq = np.zeros(n)
    seq[pos] = 1.0
    # scale to sum_target = sqrt(2n)
    sum_target = np.sqrt(2 * n)
    if np.sum(seq) > 0:
        seq = seq * (sum_target / np.sum(seq))
    # maybe also keep some small values for others (0)
    return seq.tolist()

"""
Propose a sequence of non‑negative heights that minimises the given
evaluation function.

The algorithm is a hybrid of three simple heuristics that run within the
allocated time budget:

1. **Baseline initialisation** – start either from the provided
   ``height_sequence_1`` or from a uniform sequence whose sum equals
   ``sqrt(2·n)`` (the normalisation used by the reference method).

2. **LP‑guided direction** – periodically solve a linear programme that
   finds a sequence `g` satisfying `conv(f,g) <= M` where `M` is the
   current maximal convolution of the candidate `f`.  Several mixing
   factors `t` are tried (`f←(1‑t)·f + t·g`).  Any resulting improvement
   is kept.

3. **Stochastic mass‑transfer moves** – repeatedly pick two positions
   `i` and `j` and move a random fraction of the mass from `i` to `j`
   while keeping the total sum constant.  Moves that decrease the
   evaluation are always accepted; worsening moves are accepted with a
   small Metropolis probability to escape local minima.

The loop runs until the time budget expires (leaving a safety margin),
and the best sequence encountered is returned.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Linear programming helpers
# ----------------------------------------------------------------------
def solve_convolution_lp(f_sequence, rhs, time_limit=5.0):
    """
    Solve the LP:
        maximise    sum(b)
        subject to   conv(f, b) <= rhs  for all offsets
                      b >= 0
    Returns the optimal b as a NumPy array, or ``None`` if the solver fails.
    """
    n = len(f_sequence)
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build convolution constraints: for each offset k we have
    #   sum_j f[i] * b[k-i] <= rhs.
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints b >= 0  →  -b <= 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # Use the high‑performance HiGHS solver; limit runtime to avoid hangs.
    options = {"time_limit": time_limit, "disp": False}
    res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub,
                          method="highs", options=options)

    if res.success:
        return res.x
    return None


def get_lp_direction(sequence, t_candidates=None):
    """
    Compute an LP direction `g` from the current ``sequence``.
    Returns a list of candidate mixtures (sequence mixed with g using
    various mixing factors) sorted by decreasing improvement, or ``None``
    if the LP fails.
    """
    if t_candidates is None:
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5]

    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:
        return None

    # Normalise to the standard sum sqrt(2·n) used by the reference code.
    target_sum = np.sqrt(2.0 * n)
    f_norm = np.array(sequence) * (target_sum / sum_seq)

    # Current maximal convolution (the bound to respect)
    rhs = float(np.max(np.convolve(f_norm, f_norm)))

    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    # Rescale g to the same target sum.
    g = g * (target_sum / np.sum(g))

    mixtures = []
    f = np.array(sequence, dtype=float)
    for t in t_candidates:
        new_seq = (1.0 - t) * f + t * g
        # Clip to non‑negative numbers (LP already guarantees >=0 but rounding may give -0)
        new_seq = np.clip(new_seq, 0.0, None)
        mixtures.append(new_seq.tolist())
    return mixtures


# ----------------------------------------------------------------------
# Simple Sidon construction (used as an alternative starting point)
# ----------------------------------------------------------------------
def build_sidon_sequence(n):
    """
    Greedy construction of a Sidon set (all pairwise sums distinct) inside
    ``[0, n)``.  The positions belonging to the Sidon set receive weight 1,
    all others 0, and the whole vector is finally normalised to the standard
    sum sqrt(2·n).
    """
    used = np.zeros(2 * n - 1, dtype=bool)
    positions = []
    for i in range(n):
        # check that adding i creates no sum collision
        collision = False
        for p in positions:
            if used[p + i]:
                collision = True
                break
        if collision or used[2 * i]:
            continue
        # accept i
        for p in positions:
            used[p + i] = True
        used[2 * i] = True
        positions.append(i)

    seq = np.zeros(n, dtype=float)
    seq[positions] = 1.0
    target_sum = np.sqrt(2.0 * n)
    if np.sum(seq) > 0:
        seq *= target_sum / np.sum(seq)
    return seq.tolist()


# ----------------------------------------------------------------------
# Core optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights with a low evaluation
    score.  The function runs for at most ``budget_s`` seconds and returns
    the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # leave a small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise a starting sequence
    # ------------------------------------------------------------------
    if np.random.rand() < 0.5 and 'height_sequence_1' in globals():
        # use the provided construction if available
        cur_seq = np.array(height_sequence_1, dtype=float)
        n = len(cur_seq)
    else:
        # start from a uniform sequence with a reasonable length
        n = np.random.randint(200, 800)          # moderate size → fast iterations
        target_sum = np.sqrt(2.0 * n)
        cur_seq = np.full(n, target_sum / n, dtype=float)

    # Evaluate the initial candidate
    try:
        best_score = evaluate_sequence(cur_seq.tolist())
    except Exception:
        best_score = float('inf')
    best_seq = cur_seq.tolist()

    # Occasionally try a Sidon‑based construction as a fresh start
    sidon_seq = np.array(build_sidon_sequence(n), dtype=float)
    sidon_score = evaluate_sequence(sidon_seq.tolist())
    if sidon_score < best_score:
        best_score = sidon_score
        best_seq = sidon_seq.tolist()
        cur_seq = sidon_seq.copy()

    # ------------------------------------------------------------------
    # 2. Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 2a) Occasionally ask the LP for a promising direction
        # --------------------------------------------------------------
        if np.random.rand() < 0.07:      # ~7 % of the iterations
            candidates = get_lp_direction(cur_seq.tolist())
            if candidates:
                # test each mixture and keep the best improvement
                for cand in candidates:
                    try:
                        sc = evaluate_sequence(cand)
                    except Exception:
                        continue
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand
                        cur_seq = np.array(cand, dtype=float)
                        # a good LP step usually yields a sizable drop;
                        # break to avoid unnecessary further tries
                        break
                continue  # move to next iteration after LP handling

        # --------------------------------------------------------------
        # 2b) Stochastic mass‑transfer move (simulated‑annealing style)
        # --------------------------------------------------------------
        i, j = np.random.choice(n, size=2, replace=False)
        if cur_seq[i] <= 0.0:
            continue
        # Decide how much to shift; up to 50 % of the source mass
        frac = np.random.rand() * 0.5
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # Guard against accidental negative values due to rounding
        if new_seq[i] < -1e-12:
            continue

        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue

        # Accept if the move improves the score.
        # Occasionally accept a worse move to escape plateaus.
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # temperature‑like acceptance probability (decays slowly)
            prob = np.exp(-(new_score - best_score) / (0.02 + 0.001 * iteration))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = new_seq
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()

        # --------------------------------------------------------------
        # 2c) Periodically try a completely new random start (exploration)
        # --------------------------------------------------------------
        if iteration % 5000 == 0:
            n_new = np.random.randint(200, 800)
            target_sum_new = np.sqrt(2.0 * n_new)
            candidate = np.full(n_new, target_sum_new / n_new, dtype=float)
            cand_score = evaluate_sequence(candidate.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = candidate.tolist()
                cur_seq = candidate
                n = n_new

        # Optional debug output (lightweight)
        if iteration % 2000 == 0:
            print(f"[{iteration}] best score so far: {best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – final best score: {best_score:.6f}")
    return best_seq
