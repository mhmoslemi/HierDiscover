def mirror_step(x, k, eta):
    n = len(x)
    g = np.zeros_like(x)
    # define start and end indices overlapped
    i_min = max(0, k - (n-1))
    i_max = min(k, n-1)
    # for i in [i_min, i_max], j = k - i
    idx_i = np.arange(i_min, i_max+1)
    idx_j = k - idx_i
    g[idx_i] = x[idx_j]   # factor 1; skip factor 2
    # update multiplicatively
    x_new = x * np.exp(-eta * g)
    # renormalize
    sum_x = x_new.sum()
    if sum_x == 0:
        # avoid zero sum; fallback uniform
        x_new = np.full_like(x, 1.0/x.shape[0])
    else:
        x_new /= sum_x
    return x_new

import time
import numpy as np

# ----------------------------------------------------------------------
# Propose a candidate sequence that (hopefully) beats the 1.503 bound.
#
# The search works in two stages:
#
# 1)  A cheap parametric sweep over families a_i = (i+σ)^(-p)
#     (σ > 0, p ≈ 0.5).  The sequence is truncated when the values become
#     smaller than a tiny cutoff, which keeps the length moderate.
#     The best sequence from this sweep is kept.
#
# 2)  A mirror–descent style refinement that repeatedly reduces the
#     largest convolution coefficient.  For the current vector x (which
#     is kept on the probability simplex) we locate the index k where
#     conv(x,x) attains its maximum.  The sub‑gradient of the ℓ∞‑norm
#     at that point is g_i = x_{k-i} (zero outside the overlap).  An
#     exponential update  x_i ← x_i·exp(‑η·g_i)  is performed and the
#     result is renormalised to sum to one.  Small random perturbations
#     are injected to avoid stagnation.
#
# Both phases are bounded by the global time budget (default 1000 s).
# The function returns the list of heights with the smallest score
# observed.
# ----------------------------------------------------------------------


def _generate_powerlaw(p: float, sigma: float, max_len: int = 5000,
                       eps: float = 1e-8) -> np.ndarray:
    """
    Return a truncated power‑law sequence a_i = (i + sigma) ** (-p).
    Truncation stops when the values fall below `eps` or `max_len` is
    reached.  The sequence is not normalised; scale invariance of the
    objective makes normalisation unnecessary.
    """
    # generate a reasonably long vector and then cut it
    i = np.arange(max_len, dtype=np.float64)
    a = (i + sigma) ** (-p)
    # keep only entries larger than eps
    mask = a > eps
    a = a[mask]
    return a


def _mirror_descent_step(x: np.ndarray, eta: float) -> np.ndarray:
    """
    One mirror‑descent iteration.
    x must be non‑negative and sum to one.
    The step reduces the largest coefficient of the convolution x*x.
    """
    n = x.shape[0]
    # compute convolution (full, length 2n‑1)
    conv = np.convolve(x, x)
    k = int(conv.argmax())               # index of the maximal entry
    # build the sub‑gradient  g_i = x_{k-i}   (zero outside)
    g = np.zeros_like(x)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx = np.arange(i_min, i_max + 1)
    jdx = k - idx
    g[idx] = x[jdx]                       # (factor 2 omitted – absorbed in η)
    # exponential (multiplicative) update, then project onto the simplex
    y = x * np.exp(-eta * g)
    # avoid the pathological case of a zero vector
    if y.sum() == 0.0:
        y = np.abs(np.random.randn(n))
    y = y / y.sum()
    return y


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    `evaluate_sequence`.  The routine respects the 1000 s time limit and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a little safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric sweep
    # ------------------------------------------------------------------
    max_len = 3000           # keep the length moderate for speed
    eps = 1e-8
    # exponents around the theoretically optimal 0.5
    p_vals = np.linspace(0.45, 0.55, 11)
    # sigma sampled log‑uniformly between 1e‑3 and 10
    sigma_vals = np.logspace(-3, 1, 30)

    print("=== Phase 1 : parametric sweep ===")
    for p in p_vals:
        for sigma in sigma_vals:
            if time.time() >= deadline:
                break
            seq = _generate_powerlaw(p, sigma, max_len=max_len, eps=eps)
            # safety: evaluate_sequence expects a Python list
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:      # the judge may raise unexpected errors
                continue
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.tolist()
                print(f'new best | p={p:.3f} sigma={sigma:.5f} len={len(seq)} '
                      f'score={best_score:.6f}')
            if best_score < 1.503:
                break
        if best_score < 1.503 or time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent refinement (only if we have a candidate)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a uniform sequence if everything else failed
        n_fallback = 500
        best_seq = [1.0] * n_fallback
        best_score = evaluate_sequence(best_seq)
        print('fallback to uniform sequence, score =', best_score)

    x = np.array(best_seq, dtype=np.float64)
    # normalise to the probability simplex (scale does not matter)
    x = x / x.sum()

    print("=== Phase 2 : mirror descent refinement ===")
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # step size schedule – slowly decreasing
        eta = 0.2 / np.sqrt(iteration)
        x = _mirror_descent_step(x, eta)

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.03:
            idx = np.random.randint(x.shape[0])
            delta = (np.random.rand() - 0.5) * 0.02 * x[idx]
            x[idx] = max(0.0, x[idx] + delta)
            x = x / x.sum()

        try:
            cur_score = evaluate_sequence(x.tolist())
        except Exception:
            continue

        if cur_score < best_score:
            best_score = float(cur_score)
            best_seq = x.tolist()
            print(f'iteration {iteration:5d} -> improved score = {best_score:.6f}')
            if best_score < 1.503:
                print('Target reached – stopping early.')
                break

    print('final best score:', best_score, 'length:', len(best_seq))
    return best_seq
