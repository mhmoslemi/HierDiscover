def project_onto_simplex(v):
    """Projects v onto the probability simplex (nonnegative and sum=1)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    # ensures sum(v) = 1, v >= 0
    # from https://github.com/tonyspiro/torchsad
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that minimizes
evaluate_sequence.  The algorithm uses a smooth approximation of the
max‑convolution (log‑sum‑exp) and performs projected gradient descent on
the simplex (∑a_i = 1, a_i ≥ 0).  Random restarts with different lengths
and occasional jitter avoid stagnation.  The best sequence found before
the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (sum = 1, non‑negative)
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluate_sequence value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate CPU time budget (seconds). The function will return
        before the budget is exhausted.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # leave a small safety margin

    best_seq = None
    best_score = float("inf")

    # a few different lengths – short enough to be fast,
    # long enough to give the optimiser room.
    lengths = [150, 200, 250, 300]
    np.random.shuffle(lengths)

    # ------------------------------------------------------------------
    # Main search loop (restarts with different lengths)
    while time.time() < deadline:
        # pick a length for this restart
        n = int(np.random.choice(lengths))
        # initialise a random point on the simplex (Dirichlet(1))
        a = np.random.rand(n)
        a = np.maximum(a, 1e-12)
        a /= a.sum()

        # tiny perturbation to break symmetry of the uniform point
        a += 1e-3 * (np.random.rand(n) - 0.5)
        a = np.maximum(a, 0.0)
        a /= a.sum()

        # gradient‑descent parameters
        step = 0.12            # initial step size
        beta = 15.0            # smoothness for log‑sum‑exp
        no_improve = 0        # counter for stagnation
        inner_iters = 5000    # hard limit per restart

        for _ in range(inner_iters):
            if time.time() >= deadline:
                break

            # ----- convolution and smooth max -----
            b = np.convolve(a, a)                 # full linear convolution
            max_b = b.max()
            # smooth (log‑sum‑exp) weights – numerically stable
            exp_term = np.exp(beta * (b - max_b))
            w = exp_term / exp_term.sum()         # length = 2*n-1, sums to 1

            # keep the best ever seen
