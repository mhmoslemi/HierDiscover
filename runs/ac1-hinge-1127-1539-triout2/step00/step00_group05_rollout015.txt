def subgradient_step(a, step):
    conv = np.convolve(a, a)
    max_val = conv.max()
    max_indices = np.where(conv == max_val)[0]
    g = np.zeros_like(a)
    for k in max_indices:
        # for each i where k-i in range
        i_valid = np.arange(max(0, k - (len(a)-1)), min(len(a)-1, k)+1)
        j = k - i_valid
        # then g[i_valid] += 2*a[j]
        g[i_valid] += 2 * a[j]
    # average if multiple indices
    g /= len(max_indices)
    # take negative step
    new_a = a - step * g
    # set negatives to zero then renormalize
    new_a[new_a < 0] = 0.0
    if new_a.sum() == 0:
        new_a = a
    else:
        new_a /= new_a.sum()   # normalize sum to 1
    return new_a
```

"""
Propose a candidate step‑function (a list of non‑negative heights) that improves
the objective 2·n·max(conv(a,a))/sum(a)^2.

The algorithm is a simple simulated‑annealing / stochastic hill‑climbing
procedure that works directly on the probability simplex (sum = 1, a_i ≥ 0).
It repeatedly

1. draws a random starting distribution (Dirichlet / exponential),
2. repeatedly transfers a small random amount of mass from one index to another,
   accepting the move if it improves the score or with a Metropolis probability,
3. slowly cools the temperature and shrinks the transfer size,
4. keeps the best sequence seen so far.

All arithmetic is performed with NumPy; the convolution needed for the score
is evaluated with ``np.convolve`` (identical to the provided ``evaluate_sequence``
behaviour for non‑negative inputs).  The routine respects the overall time
budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

def _score_from_array(arr: np.ndarray) -> float:
    """
    Compute the evaluation score for a probability vector ``arr``.
    The vector is assumed to have a positive sum (normally 1.0),
    otherwise the function would return ``inf`` – but the search never
    creates such a vector.
    """
    n = len(arr)
    conv = np.convolve(arr, arr)
    max_b = conv.max()
    # since sum(arr) == 1, the denominator is 1
    return float(2.0 * n * max_b)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Stochastic search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds. The function stops before this limit.
    **kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        A list of non‑negative heights (the best sequence found).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # safety margin

    # Length of the sequence – use the previously discovered construction
    # if it exists, otherwise fall back to a modest size.
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        n = 1500

    best_seq = None
    best_score = float("inf")

    # --- Simulated annealing parameters ---------------------------------
    # initial temperature (controls acceptance of uphill moves)
    T0 = 0.3
    # geometric decay per inner iteration
    temp_decay = 0.9999
    # initial proportion of mass moved in each step
    move_frac0 = 0.25
    move_decay = 0.99995
    # number of inner iterations per restart (tuned for the time budget)
    max_iter_per_restart = 2000
    # limit on total restarts – the loop also stops when time runs out
    max_restarts = 500

    restart = 0
    while time.time() < deadline and restart < max_restarts:
        restart += 1

        # ---- initialise a random point on the simplex --------------------
        # exponential draws give a Dirichlet(1,…,1) distribution.
        a = np.random.exponential(1.0, size=n)
        a /= a.sum()
        # mix a little bit of the uniform vector – it stabilises the early
        # search and guarantees a feasible starting point.
        a = 0.3 * a + 0.7 * (1.0 / n)
        a /= a.sum()

        cur_score = _score_from_array(a)
        temperature = T0
        move_frac = move_frac0

        # ---- inner stochastic optimisation loop -------------------------
        for it in range(max_iter_per_restart):
            # time‑out protection
            if time.time() >= deadline:
                break

            # keep the globally best solution
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"new best score {best_score:.6f} (restart {restart}, iter {it})")
                # we stop early if we reach the target zone
                if best_score < 1.51:
                    deadline = time.time()   # break outer loops
                    break

            # ---- propose a random mass transfer -------------------------
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # amount taken from i (a fraction of its current mass)
            delta = np.random.rand() * move_frac * a[i]
            if delta <= 0.0:
                continue

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # numerical safety – keep non‑negative and renormalise
            a_new = np.maximum(a_new, 0.0)
            a_new_sum = a_new.sum()
            if a_new_sum == 0.0:                # should never happen
                continue
            a_new /= a_new_sum

            new_score = _score_from_array(a_new)

            # ---- Metropolis acceptance ---------------------------------
            if new_score <= cur_score:
                a = a_new
                cur_score = new_score
            else:
                # probability to accept a worse move
                prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    a = a_new
                    cur_score = new_score

            # ---- annealing schedule ------------------------------------
            temperature *= temp_decay
            move_frac *= move_decay

        # after finishing the inner loop we start a fresh random restart
        # (the outer while‑loop will repeat until the time budget expires)

    # if nothing better than the default was found, fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Convert to a plain Python list as required by the evaluator.
    return best_seq.tolist()
