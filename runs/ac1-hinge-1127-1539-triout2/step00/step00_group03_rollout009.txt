def conv_max(a):
    # compute convolution using FFT
    n = len(a)
    Nfft = 1 << ((2*n-1).bit_length())
    Fa = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(Fa * Fa, n=Nfft)[:2*n-1]
    return conv

def gradient_step(a, eta):
    n = len(a)
    conv = conv_max(a)
    k = np.argmax(conv)
    # compute gradient
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >=0) & (partner < n)
    g = np.zeros_like(a)
    # regular contributions
    g[valid] = 2 * a[partner[valid]]
    # correct diagonal
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]   # replace 2*a[d] with a[d] (since derivative is a[d] not 2*a[d])
    # multiplicative update
    a = a * np.exp(-eta * g)
    # renorm
    total = a.sum()
    if total == 0:
        # reinit random
        a = np.random.rand(n)
        a = a / a.sum() * np.sqrt(2 * n)
    else:
        a *= (np.sqrt(2 * n) / total)
    return a, conv.max()

"""
Propose a candidate sequence that (hopefully) beats the baseline 2.0 bound.

The algorithm is a simple projected multiplicative gradient descent on the
∞‑norm of the self‑convolution.

* Initialise a non‑negative vector `a` (either from the existing
  ``height_sequence_1`` or from a random Dirichlet draw) and scale it so that
  Σa = sqrt(2·n).  With this normalisation the evaluation score returned by
  ``evaluate_sequence`` equals the maximal convolution entry, i.e.  max_k (a∗a)[k].

* At each iteration compute the linear convolution by an FFT (O(n log n)).
  Let `k_max` be the index where the convolution attains its maximum.

* The sub‑gradient of the maximal entry w.r.t. each coefficient a_j is
  `g_j = 2·a_{k_max−j}` (with the diagonal case `j = k_max/2` using a single
  term).  A multiplicative update `a ← a·exp(−η·g)` keeps all entries non‑negative.

* After the update the vector is re‑normalised to keep the L1‑norm fixed.
  The step size η is slowly decayed; occasional tiny random perturbations are
  injected to avoid stagnation.

* The best sequence seen during the allotted time is returned.

The method stays fully within the 1000 s budget, uses only NumPy, and makes no
filesystem or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Compute the linear convolution a*a using FFT, return the full convolution
    array, its maximum value and the index of that maximum.
    """
    n = a.shape[0]
    # length needed for linear convolution
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One multiplicative gradient step aimed at reducing the current maximal
    convolution entry.  The step size `eta` is a positive scalar.
    """
    n = a.shape[0]
    _, _, k = _conv_max(a)

    # compute sub‑gradient of the entry at index k
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)

    g = np.empty_like(a)
    g.fill(0.0)
    # regular off‑diagonal terms contribute 2*a[partner]
    g[valid] = 2.0 * a[partner[valid]]

    # diagonal case (i == j) where k is even
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]                     # derivative is a[d] (not doubled)

    # multiplicative update – guarantees non‑negativity
    a_new = a * np.exp(-eta * g)

    # re‑normalise to the required L1‑norm (sqrt(2·n))
    S = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total == 0.0:                     # extremely unlikely, but be safe
        a_new = np.full(n, S / n)
    else:
        a_new *= (S / total)

    return a_new


def _initial_sequence(n: int, seed: int) -> np.ndarray:
    """
    Produce an initial non‑negative sequence of length n, summed to sqrt(2·n).
    With 50 % probability the global `height_sequence_1` (if present) is used,
    otherwise a random Dirichlet draw is taken.
    """
    rng = np.random.default_rng(seed)

    S = np.sqrt(2.0 * n)

    # try to use the provided construction if it exists
    use_existing = False
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=float)
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.pad(base, (0, n - base.size), constant_values=0.0)
            use_existing = True
        except Exception:
            use_existing = False

    if use_existing and rng.random() < 0.5:
        a = a.copy()
    else:
        # Dirichlet(1,…,1) – i.e. uniform on the simplex
        a = rng.random(n)
    # normalise
    a = a / a.sum() * S
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def _optimize(n: int, seed: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the projected multiplicative gradient descent until `deadline`
    (seconds since the epoch).  Returns the best sequence found and its
    external score as reported by ``evaluate_sequence``.
    """
    a = _initial_sequence(n, seed)

    # external evaluation of the starting point
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, score={best_score:.6f}")

    # step‑size schedule
    eta = 0.5
    decay = 0.9999

    # occasional random “jitter” helps escape tiny plateaus
    jitter_prob = 0.01

    while time.time() < deadline:
        # one deterministic gradient step
        a = _gradient_step(a, eta)

        # fast internal objective (max convolution entry)
        _, cur_max, _ = _conv_max(a)

        # only call the expensive `evaluate_sequence` when we see any
        # improvement in the fast surrogate.
        if cur_max < best_score:
            ext_score = evaluate_sequence(a.tolist())
            if ext_score < best_score:
                best_score = ext_score
                best_seq = a.copy()
                print(f"[improve] time={time.time():.1f}s  score={best_score:.6f}")

        # step‑size decay
        eta *= decay

        # tiny random perturbation
        if np.random.random() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Required entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    `evaluate_sequence` score than the current best bound of 2.0.
    The function runs for at most ``budget_s`` seconds and prints progress.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a small safety margin

    # Prefer to keep the length of the known construction (5641) – larger
    # lengths give a slightly lower theoretical lower bound.
    n = 5641
    try:
        # if a global variable with that length exists, keep it, otherwise fall back
        if 'height_sequence_1' in globals():
            n = len(height_sequence_1)
    except Exception:
        pass

    # Run the optimiser (single run – the whole budget is spent here)
    best_seq, best_score = _optimize(n, seed, deadline)

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
