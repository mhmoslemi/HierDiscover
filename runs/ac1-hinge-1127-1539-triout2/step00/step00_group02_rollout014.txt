"""
Simple stochastic optimizer for the step‑function bound.

The objective is 2·n·max(conv(a,a)) / (sum(a)^2).  Scaling of the whole
vector does not change the value, therefore we keep the ℓ1‑norm fixed at
√(2·n).  The optimizer starts from a continuous analogue of the optimal
continuous solution a_i ∝ 1/√i (this gives a bound ≈π/2≈1.57) and then
performs cheap random mass‑transfer moves that preserve the total sum.
All moves that decrease the maximum convolution are kept; occasional
re‑starts with slightly different exponents give a quick diversification.
The search runs until the time budget expires and returns the best
sequence found.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (no closures, no lambdas as required)
# ------------------------------------------------------------

def _max_convolution(a: np.ndarray) -> float:
    """
    Return the maximum value of the linear convolution a*a.
    Uses an FFT of length a power of two for speed.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Compute the evaluation function used by the competition.
    The vector a must be non‑negative.
    """
    n = a.size
    max_b = _max_convolution(a)
    s = a.sum()
    # the safety checks performed inside evaluate_sequence are reproduced
    # here only for speed; the final candidate will be verified by the
    # supplied evaluate_sequence when the wrapper returns it.
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _initial_sequence(p: float, n: int) -> np.ndarray:
    """
    Produce a deterministic start sequence a_i ∝ (i+1)^{-p}
    and renormalise it to the required ℓ1‑norm √(2·n).
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    # renormalise to √(2n); this makes the evaluation function equal to the
    # max‑convolution value (the denominator cancels out)
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


def _local_search(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that preserves the total sum.
    Randomly selects two positions i, j and moves a fraction of the mass
    from i to j.  Only moves that strictly improve the score are kept.
    The loop stops when the time budget is exhausted.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    # a few parameters controlling the magnitude of the moves
    max_frac = 0.12          # at most 12 % of the source entry is moved
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick a source index that currently holds some mass
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # propose moving a random fraction of a[i] to a[j]
        frac = rng.random() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_val = _score(a)

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                # small progress report
                print(f"[{time.time():.1f}]  improved score = {best_val:.6f}")
        else:
            # reject the move – revert the change
            a[j] -= delta
            a[i] += delta

    return best_a, best_val


# ------------------------------------------------------------
# Main entry point required by the challenge
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence that minimises the supplied
    evaluate_sequence() function.  The routine spends the allotted time
    (budget_s seconds) performing a short stochastic optimisation that
    starts from power‑law profiles a_i ∝ (i+1)^{-p} (p≈0.5).  Several
    values of p and several lengths are tried; the best sequence found
    during the run is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a little margin

    # keep the overall best candidate
    best_seq = None
    best_score = np.inf

    # a few different lengths – shorter vectors give a smaller factor n,
    # but we also need enough points to approximate the continuous optimum
    lengths = [800, 1200, 1800, 2500]          # all comfortably below 10 000
    exponents = [0.45, 0.48, 0.5, 0.52, 0.55]

    for n in lengths:
        for p in exponents:
            # respect the global time budget
            if time.time() >= deadline:
                break

            # initialise a power‑law sequence and evaluate it
            a = _initial_sequence(p, n)
            cur_score = _score(a)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"  start (n={n}, p={p}) score = {cur_score:.6f}")

            # allocate a modest chunk of time for local refinement
            sub_deadline = min(deadline, time.time() + 5.0)
            a_opt, opt_score = _local_search(a, sub_deadline)

            if opt_score < best_score:
                best_score = opt_score
                best_seq = a_opt.copy()
                print(f"  after local search (n={n}, p={p}) best = {opt_score:.6f}")

        if time.time() >= deadline:
            break

    # fallback – should never happen, but keep the code robust
    if best_seq is None:
        best_seq = np.ones(1000, dtype=np.float64)
        best_seq *= np.sqrt(2.0 * best_seq.size) / best_seq.sum()

    # discard trailing zeros just in case
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # return a plain Python list as required by the outer harness
    return best_seq.tolist()
