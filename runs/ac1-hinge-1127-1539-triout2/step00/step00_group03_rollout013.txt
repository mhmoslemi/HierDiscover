def _linear_ramp_sequence(n: int) -> list[float]:
    """
    Build a linear ramp a_i = i (0‑based) and scale it so that
    sum(a) = sqrt(2 * n).  The result is a list of non‑negative floats.
    """

    total = idx.sum()                               # n*(n‑1)/2

    seq = (idx * scale).tolist()
    return seq


def _gradient_step(seq: list[float], step: float = 0.01) -> list[float]:
    """
    Perform a single projected sub‑gradient step that tries to lower the
    current maximal convolution entry.

    The gradient of the active convolution entry k* is
        g_i = 2 * a_{k* - i}   (if the index is in range, else 0).

    The step is projected onto the hyperplane Σ a_i = const by subtracting
    the mean of the gradient.  After the move the sequence is clipped to
    non‑negative values and renormalised.
    """
    a = np.array(seq, dtype=float)
    n = a.size

    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))
