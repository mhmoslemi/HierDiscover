def improve_sequence(seq, time_limit):
    start = time.time()
    n = len(seq)
    # Normalize sum to 1 (or any fixed) to keep scaling invariance; we can enforce sum=1 at each step.
    # Actually scaling doesn't affect evaluation but normalization helps numeric stability.
    # We'll keep sum=1 by scaling after modifications.
    seq = np.array(seq, dtype=float)
    total = seq.sum()
    if total == 0:
        seq = np.random.rand(n)
        total = seq.sum()
    seq = seq / total  # sum=1

    best_seq = seq.copy()
    best_val = compute_eval(seq)

    # Determine move step magnitude dynamic.
    step = 0.02  # fraction of total mass to move per step.

    # Precompute conv contributions? Not needed.

    while time.time() < time_limit:
        # compute conv and find k_max
        conv = np.convolve(seq, seq)
        k_max = np.argmax(conv)
        max_val = conv[k_max]

        # Determine indices contributing:
        # for i in 0..n-1, j = k_max - i if valid.
        # contributions = seq[i] * seq[j] if valid.
        # We'll pick i with max contribution.
        contributions = []
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contributions.append((seq[i] * seq[j], i, j))
        if not contributions:
            # no contributions? continue with random
            idx_from = np.random.randint(n)
        else:
            # choose pair with max product
            _, i_max, j_max = max(contributions)
            idx_from = i_max if seq[i_max] >= seq[j_max] else j_max
        # Determine idx_to: choose location with smallest seq value and far from idx_from? Or random 
        # To reduce max conv, choose idx_to where adding mass will minimally increase conv at k_max.
        # We could compute, for each candidate index k, the increase to conv at each shift.
        # Simpler: pick random index not equal idx_from.
        idx_to = np.random.randint(n)
        while idx_to == idx_from:
            idx_to = np.random.randint(n)

        # move mass
        delta = min(step, seq[idx_from])
        if delta <= 0:
            continue
        seq_new = seq.copy()
        seq_new[idx_from] -= delta
        seq_new[idx_to] += delta

        # normalization: after moving, sum unchanged (since we moved mass). So preserve sum.

        # compute new eval
        val_new = compute_eval(seq_new)

        # accept if improved or with probability
        if val_new < best_val:
            best_seq = seq_new.copy()
            best_val = val_new

        # Accept based on simulated annealing schedule
        # Determine T = annealing temperature decreasing over time
        elapsed = time.time() - start
        T = max(0.001, (1 - elapsed / (time_limit - start)) )  # linear cooling
        if val_new < compute_eval(seq) or np.random.rand() < np.exp((compute_eval(seq) - val_new) / (T + 1e-9)):
            seq = seq_new
        # else reject, seq stays

        # occasionally adjust step size
        if np.random.rand() < 0.01:
            step = max(0.001, step * (0.9 + 0.2 * np.random.rand()))  # random adjust

    return best_seq, best_val

"""
Efficient local search for the step‑function optimisation problem.

The algorithm works as follows:
1.  Initialise a non‑negative vector (either a slice of the known SOTA
    construction or a random one) and normalise its L1‑norm.
2.  Iteratively improve the vector:
    *   Compute the current autocorrelation and its maximal entry M.
    *   Solve a linear programme that maximises the total sum of a new
        vector `g` under the linear constraints  conv(f,g) ≤ M  (all
        entries non‑negative).  The LP returns a feasible direction that
        can increase the L1‑norm without raising the convolution bound.
    *   Normalise `g` so that its sum matches that of `f` (both = 1).
    *   Perform a cheap line‑search on the mixing coefficient `t` in
        (0, 0.5] to minimise the true objective after mixing
        `f←(1−t)·f + t·g`.  The line‑search uses the pre‑computed
        convolutions of `f`, `g` and `f*g`.
    *   Occasionally apply a small random jitter to escape local minima.
3.  Keep the best sequence seen for each tested length and finally return
    the overall best one.

All heavy work (convolutions, LP solves, line‑search) is performed with
NumPy / SciPy and stays well inside the allocated 1000 s budget.
"""

# Helper: fast evaluation that mirrors `evaluate_sequence` (without the
#          safety checks) – the objective is scale‑invariant.
def _eval_seq(seq: np.ndarray) -> float:
    total = seq.sum()
    if total < 0.01:
        return float('inf')
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return 2.0 * n * max_b / (total * total)

# Linear programme: maximise sum(b)  subject to  conv(f,b) ≤ rhs, b ≥ 0.
def _solve_lp(f: np.ndarray, rhs: float):
    n = len(f)
    c = -np.ones(n)                     # minimise -sum(b)  → maximise sum(b)
    rows = []
    bounds = []

    # constraints for each convolution index k = 0 … 2n‑2
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]               # coefficient of b_j in conv(f,b)[k]
        rows.append(row)
        bounds.append(rhs)

    rows.append(-np.eye(n))
    bounds.append(np.zeros(n))

    b_ub = np.hstack(bounds)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

# Find the mixing coefficient t∈[0,0.5] that yields the smallest objective.
def _mix_optimal_t(f, g, conv_f, conv_g, conv_fg):
    n = len(f)
    sum_f = f.sum()
    sum_g = g.sum()
    best_t = 0.0
    best_val = _eval_seq(f)

    # coarse scan
    for t in np.linspace(0.0, 0.5, 11)[1:]:
        mix_sum = (1 - t) * sum_f + t * sum_g
        mix_conv = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_g
        val = 2.0 * n * mix_conv.max() / (mix_sum * mix_sum)
        if val < best_val:
            best_val = val
            best_t = t

    # local refinement around the best coarse value
    if best_t > 0.0:
        for _ in range(3):
            lo = max(0.0, best_t - 0.05)
            hi = min(0.5, best_t + 0.05)
            for t in np.linspace(lo, hi, 11):
                mix_sum = (1 - t) * sum_f + t * sum_g
                mix_conv = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_g
                val = 2.0 * n * mix_conv.max() / (mix_sum * mix_sum)
                if val < best_val:
                    best_val = val
                    best_t = t
    return best_t, best_val

# Core improvement loop for a single sequence.
def _improve(seq: np.ndarray, deadline: float):
    n = len(seq)
    seq = seq.astype(float)
    if seq.sum() < 1e-6:
        seq = np.random.rand(n)
    seq = seq / seq.sum()                     # normalise: sum = 1 (scale‑invariant)

    best_seq = seq.copy()
    best_val = _eval_seq(seq)

    while time.time() < deadline:
        conv_f = np.convolve(seq, seq)
        M = conv_f.max()

        g_raw = _solve_lp(seq, M)
        if g_raw is None:
            # fallback: tiny random jitter
            i = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.02
            seq[i] = max(0.0, seq[i] + delta)
            seq = seq / seq.sum()
            continue

        sum_g_raw = g_raw.sum()
        if sum_g_raw <= 0:
            g = np.random.rand(n)
            g = g / g.sum()
        else:
            g = g_raw / sum_g_raw                # bring g to the same L1‑norm as f ( = 1)

        conv_g = np.convolve(g, g)
        conv_fg = np.convolve(seq, g)

        t, _ = _mix_optimal_t(seq, g, conv_f, conv_g, conv_fg)
        if t > 0.0:
            seq = (1 - t) * seq + t * g
            seq = np.maximum(seq, 0.0)          # clip possible tiny negatives
            seq = seq / seq.sum()

        cur_val = _eval_seq(seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] local improvement (n={n}): {best_val:.6f}")

        # occasional random escape move
        if np.random.rand() < 0.03:
            i, j = np.random.randint(n, size=2)
            delta = (np.random.rand() - 0.5) * 0.05
            seq[i] = max(0.0, seq[i] + delta)
            seq[j] = max(0.0, seq[j] - delta)
            seq = seq / seq.sum()

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height sequence with a low value of
    ``evaluate_sequence``.  The routine runs several independent attempts
    (different lengths, random starts or a slice of the known SOTA
    construction) and keeps the best result found before the time budget
    expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 5.0)   # small safety margin

    best_overall_seq = None
    best_overall_val = float('inf')

    # lengths to experiment with – modest sizes keep the LP cheap.
    lengths = [200, 400, 800, 1600]
    try:
        lengths.append(len(height_sequence_1))
    except Exception:
        pass
    lengths = sorted(set(lengths))

    for n in lengths:
        if time.time() >= deadline:
            break

        # choose initial vector
        if np.random.rand() < 0.5 and 'height_sequence_1' in globals():
            base = list(height_sequence_1)[:n]
            if len(base) < n:
                base += list(np.random.rand(n - len(base)))
            init_seq = np.array(base, dtype=float)
        else:
            init_seq = np.random.rand(n)

        if init_seq.sum() < 1e-6:
            init_seq = np.random.rand(n)
        init_seq = init_seq / init_seq.sum()

        # allocate a slice of remaining time to this attempt
        now = time.time()
        remaining = deadline - now
        attempt_deadline = now + max(2.0, 0.3 * remaining)   # at most ~30 % of what is left

        local_seq, local_val = _improve(init_seq, attempt_deadline)

        # final evaluation with the official routine (adds clipping etc.)
        try:
            final_val = evaluate_sequence(local_seq.tolist())
        except Exception:
            final_val = float('inf')

        print(f"Length {n}: internal best = {local_val:.6f}, final = {final_val:.6f}")

        if final_val < best_overall_val:
            best_overall_val = final_val
            best_overall_seq = local_seq.tolist()
            print(f"*** new overall best = {best_overall_val:.6f} (n={n}) ***")

    # fallback – uniform sequence if everything else fails
    if best_overall_seq is None:
        best_overall_seq = [1.0] * 100

    print(f"Returning sequence of length {len(best_overall_seq)} with eval {best_overall_val:.6f}")
    return best_overall_seq
