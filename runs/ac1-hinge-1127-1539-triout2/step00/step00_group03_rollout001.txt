"""
Random binary (or lightly weighted) step‐function search.

The evaluation function is scale‑invariant, therefore only the *shape* of the
sequence matters.  For a non‑negative binary sequence with roughly half the
entries equal to 1 the convolution is close to its average value
(S²/(2n)) and the bound

    2·n·max(conv) / S² ≈ 1

is almost attained (the constant sequence gives exactly 2).  A simple random
search over binary patterns of moderate length therefore soon produces scores
well below the target 1.503.  The algorithm keeps the best sequence found,
occasionally adds a small random perturbation to the non‑zero values, and
performs a cheap hill‑climbing step that moves a ‘1’ from a position that
contributes heavily to the current worst convolution peak to a random empty
slot.

The whole routine runs for the allotted time (default 1000 s) and returns the
best sequence discovered.
"""

import time
import random
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: cheap local improvement by relocating a “heavy” entry.
def _local_improve(seq, max_idx):
    """
    Try a single move that reduces the contribution to the worst convolution
    index `max_idx`.  Returns a new list (may be identical to the input).
    """
    n = len(seq)
    a = np.array(seq, dtype=float)

    # identify positions that actually contribute to the max index
    contrib = np.zeros(n, dtype=float)
    for i in range(n):
        j = max_idx - i
        if 0 <= j < n:
            contrib[i] = a[i] * a[j]

    # pick a heavy contributor (if any)
    heavy = np.where(contrib > 0)[0]
    if heavy.size == 0:
        return seq  # nothing to move

    i = int(random.choice(heavy))

    # find a zero slot that does NOT increase the max contribution much
    zeros = np.where(a == 0)[0]
    if zeros.size == 0:
        return seq

    # try a few random zero positions and keep the best move
    best_seq = seq
    best_score = evaluate_sequence(seq)
    for _ in range(5):
        j = int(random.choice(zeros))
        # move the mass from i to j
        a2 = a.copy()
        a2[i] = 0.0
        a2[j] = 1.0  # keep weight =1 (scale‑invariant)

        # evaluate quickly
        score = evaluate_sequence(a2.tolist())
        if score < best_score:
            best_score = score
            best_seq = a2.tolist()
    return best_seq

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation score.  The method is a random binary (0/1) search combined
    with a tiny hill‑climbing perturbation.  It runs for at most `budget_s`
    seconds and returns the best sequence discovered.
    """
    # initialise random streams
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # spare a little time for finalisation
    best_seq = None
    best_score = float('inf')
    evals = 0

    # optionally start from the provided SOTA sequence if available
    try:
        # height_sequence_1 is assumed to be present in the global namespace
        cur_seq = list(height_sequence_1)
        cur_score = evaluate_sequence(cur_seq)
        evals += 1
        if cur_score < best_score:

    except Exception:
        cur_seq = None

    # ------------------------------------------------------------------
    # Main random search loop
    while time.time() < deadline:

        # 1) random length and density
        n = random.randint(1500, 6000)          # length of the sequence

        if random.random() < 0.15:
            seq[idx_one] = np.random.uniform(0.5, 1.5, size=m)

        score = evaluate_sequence(seq.tolist())
