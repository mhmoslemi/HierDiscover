def evaluate_and_convolution(sequence):
    # returns score, convolution, sum, max
    n = len(sequence)
    # using fft for speed
    size = 1 << (2 * n - 1).bit_length()
    A = np.array(sequence, dtype=np.float64)
    fft_a = np.fft.rfft(A, n=size)
    B = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
    max_b = np.max(B)
    sum_a = np.sum(A)
    score = 2 * n * max_b / (sum_a ** 2)

Linear decreasing construction for the step-height sequence.

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)
