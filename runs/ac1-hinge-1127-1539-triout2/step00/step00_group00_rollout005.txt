The evaluation function is
    score = 2 * n * max(conv(a,a)) / (sum(a)^2)

Mathematically the Cauchy–Schwarz inequality guarantees that
    max(conv(a,a)) ≥ sum(a)^2 / n

Consequently the theoretical minimum of the score is 2, attained by any

        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # leave a tiny margin

    # ------------------------------------------------------------------
    # 1. Initialise from the known construction if it exists.
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` is a global that may already contain a good
        # solution (normally a uniform vector).  We copy it to avoid
        # accidental mutation of the global.
        current = list(height_sequence_1)          # type: ignore[name-defined]
        print("Using provided height_sequence_1 as starting point.")
    except Exception:
        # Fallback: create a fresh uniform sequence.
        # Length 10 000 is comfortably large while keeping convolution cheap.
        current = uniform_sequence(10_000, value=1.0)
        print("Created fresh uniform sequence of length", len(current))

    # Evaluate the initial candidate.
    best_seq = current
    best_score = evaluate(best_seq)
    print(f"Initial score = {best_score:.12f}")

    # ------------------------------------------------------------------
    # 2. Very light stochastic search – unlikely to improve on the uniform.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Pick a random index and perturb it slightly.
        idx = np.random.randint(len(current))
        # Small Gaussian bump; ensure non‑negative afterwards.
        new_val = max(0.0, current[idx] + np.random.randn() * 0.01)
        candidate = current.copy()
        candidate[idx] = new_val

        # Re‑evaluate.
        try:
            score = evaluate(candidate)
            if score < best_score:
                best_score = score
                best_seq = candidate
                current = candidate
                print(f"New best score = {best_score:.12f} (iteration {int(time.time() - start_time)}s)")
            else:
                # keep the old current sequence
                current = candidate
        except Exception:
            # Any unexpected failure – just ignore this candidate.
            continue

    print("Search finished.")
    print(f"Best score obtained = {best_score:.12f}")
    print(f"Final sequence length = {len(best_seq)}")
    return best_seq
