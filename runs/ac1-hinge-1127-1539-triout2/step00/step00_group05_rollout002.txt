"""
Linear‑decay construction for the convolution‑maximisation problem.

The evaluation function is invariant under scaling, therefore only the shape of
the sequence matters.  For a decreasing sequence a_i = (1 - i/(n-1)) (or any
power of this term) the maximal entry of the auto‑convolution occurs at the
centre and can be computed analytically.  With a linear decay (p = 1) the
asymptotic score tends to 4/3 ≈ 1.333, which is already below the required
target 1.503.

The proposer evaluates a few simple power‑law shapes
    a_i = (1 - i/(n‑1))**p   (p ∈ {0.5, 0.8, 1.0, 1.2, 1.5, 2.0})
on the same length as the previously supplied best construction and keeps
the one with the smallest `evaluate_sequence` value.  The algorithm is
deterministic, runs in O(k·n²) with a tiny constant (k = number of tested p‑
values) and finishes well within the 1000 s budget.
"""

import numpy as np
import time

# evaluate_sequence is assumed to be imported in the execution environment.


def _linear_decay_sequence(length: int, power: float = 1.0) -> list[float]:
    """
    Produce a non‑negative decreasing sequence of the given length.
    a_i = (1 - i/(length-1)) ** power, i = 0 … length-1.
    The values lie in [0, 1] and the sum is > 0, so the security checks
    inside `evaluate_sequence` will never reject them.
    """
    if length <= 1:
        # degenerate case – just a single positive entry
        return [1.0]

    idx = np.arange(length, dtype=float)
    seq = (1.0 - idx / (length - 1)) ** power
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: float = 1000,
                      **kwargs) -> list[float]:
    """
    Return a step‑function (list of non‑negative heights) that minimises the
    provided `evaluate_sequence` function.

    The method searches a tiny family of analytically motivated shapes
    (linear/power‑law decays) and returns the best one found.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Length of the candidate: try to stay close to the length that was
    # used in previous attempts (the global `height_sequence_1` may be
    # defined).  If it is absent, fall back to a moderate size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 1500                     # a reasonable default

    # ------------------------------------------------------------------
    # Candidate shapes to test.  Power = 1 corresponds to a pure linear
    # decay which theoretically yields a score of 4/3.
    # ------------------------------------------------------------------
    powers = [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]

    best_seq = None
    best_score = float("inf")

    start = time.time()
    for p in powers:
        # Stop early if we run out of time (margin of 5 s for safety).
        if time.time() - start > budget_s - 5:
            break

        seq = _linear_decay_sequence(n, power=p)

        # The evaluation function already clips negative numbers, caps large
        # values and checks the sum, so we can call it directly.
        try:
            score = evaluate_sequence(seq)
        except Exception:
            # In the extremely unlikely case the security checks raise,
            # treat it as an infinite score.
            score = float("inf")

        print(f"Tested power={p: .2f} → score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq

    # As a final safety net, if for any reason no sequence was accepted,
    # fall back to a uniformly distributed one (the constant sequence).
    if best_seq is None:
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)

    print(f"Chosen sequence length={len(best_seq)} with score={best_score:.6f}")
    return best_seq
