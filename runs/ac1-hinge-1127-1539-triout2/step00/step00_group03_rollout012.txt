def project_onto_simplex(v, z=1.0):
    # returns projection onto simplex sum = z, nonnegative
    if np.sum(v) == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

Note: yields sum(w) = z (within small epsilon).

The evaluation function  

```python
score = 2 * n * max(conv(f,f)) / (sum(f)**2)
```  

is *scale‑invariant*: multiplying the whole vector by a constant does not change the
score.  After normalising the vector so that `sum(f) = sqrt(2*n)` the score reduces to  

```
score = max(conv(f,f))
```  

Thus we only have to minimise the maximal entry of the self‑convolution while keeping
the entries non‑negative and the ℓ₁‑norm fixed.

A well‑known family of sequences that give a low maximal convolution are power‑law
vectors  

```
a_i  ∝  (i + c)^(-β)      (i = 0 … n‑1)
```  

with a small shift `c>0`.  
For `β≈½` the convolution becomes almost flat (the continuous analogue is the
inverse‑square‑root kernel whose autocorrelation is a constant).  By tuning `β` and
the shift `c` we can push the maximal convolution below the naïve bound `2.0`.

The algorithm works in three stages

1. **Baseline** – evaluate the provided construction (`height_sequence_1`).
2. **Parametric grid search** – enumerate a coarse grid of `(β,c)`, build the
   corresponding power‑law vectors, normalise them to `sqrt(2*n)`,
   evaluate the score and keep the best one.
   A finer local grid is explored around the best parameters.
3. **Local combinatorial refinement** – starting from the best vector,
   repeatedly try a random swap of two positions.  
   If the swap reduces the maximal convolution the change is kept.
   The process is stopped when the time budget expires.

All steps are cheap (`np.convolve` uses an FFT for vectors longer than a few
thousand elements), so many thousands of candidates can be examined well within
the 1000 s limit.

The final sequence (a Python `list[float]`) is returned by `propose_candidate`.
A short progress log is printed during the run.

```python
"""Propose a step‑function (non‑negative height vector) with a low
evaluation score.

The algorithm consists of:
  1. evaluating the supplied baseline (`height_sequence_1`);
  2. a grid search over power‑law shapes a_i ∝ (i + shift)^{-beta};
  3. a simple swap‑based local improvement to further lower the maximal
     self‑convolution.

The evaluation is scale‑invariant, therefore each candidate is
normalised to have ℓ₁‑norm sqrt(2·n) before calling `evaluate_sequence`.
"""

import time
import numpy as np
import math
import random

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project a vector onto the simplex {x | x_i ≥ 0, sum x_i = z}."""
    # Works for any real vector v, returns a non‑negative vector with sum z.
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return zeros
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _build_powerlaw(beta: float, shift: float, n: int, target_sum: float) -> np.ndarray:
    """
    Return a vector a of length n with
        a_i ∝ (i + shift)^{-beta}
    normalised to have ℓ₁‑norm = target_sum.
    """
    # avoid division by zero – enforce shift > 0
    if shift <= 0.0:
        shift = 1e-9
    idx = np.arange(n, dtype=np.float64)
    raw = (idx + shift) ** (-beta)
    total = raw.sum()
    if total == 0.0:
        return None
    a = raw * (target_sum / total)
    return a


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    # np.convolve uses an efficient FFT for vectors > ~1500 elements.
    conv = np.convolve(a, a)
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the given
    evaluation function.  The function respects the 1000 s time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1) initialise with the supplied construction if it exists
    # ------------------------------------------------------------------
    try:
        base_seq = list(height_sequence_1)   # global variable supplied by the runner
        n = len(base_seq)
        print(f"[init] Using supplied height_sequence_1 of length {n}")
    except Exception:
        # fallback: create a uniform vector of a reasonable size
        n = 4000
        base_seq = [1.0] * n
        print("[init] height_sequence_1 not available – using uniform fallback")

    # normalise to the required ℓ₁‑norm ( sqrt(2·n) )
    target_sum = math.sqrt(2.0 * n)
    base_arr = np.array(base_seq, dtype=np.float64)
    base_arr = base_arr * (target_sum / base_arr.sum())
    best_arr = base_arr.copy()
    best_score = _conv_max(best_arr)       # equal to evaluate_sequence after normalisation
    print(f"[init] baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Grid search over power‑law parameters (beta, shift)
    # ------------------------------------------------------------------
    # coarse grid
    betas = np.linspace(0.30, 0.80, 26)          # 0.30 … 0.80 step ≈0.02
    shifts = np.linspace(0.0, 50.0, 26)          # 0 … 50 step ≈2
    print("[grid] Starting coarse grid search")
    for beta in betas:
        for shift in shifts:
            if time.time() > deadline:
                break
            cand = _build_powerlaw(beta, shift, n, target_sum)
            if cand is None:
                continue
            score = _conv_max(cand)
            if score < best_score:
                best_score = score
                best_arr = cand.copy()
                print(f"[grid] New best: score={best_score:.6f}, beta={beta:.3f}, shift={shift:.1f}")

        if time.time() > deadline:
            break

    # fine grid around the current best parameters (if we have stored them)
    # The fine search explores a small neighbourhood of the current best beta/shift.
    # To keep the implementation simple we re‑use the same loops with finer steps.
    try:
        # Estimate the beta and shift that produced the current best vector:
        # Reverse‑engineer by fitting a power law to the first few entries.
        # This is only an approximation used for seeding the fine grid.
        # If it fails we simply continue with a generic fine sweep.
        first_vals = best_arr[:10]
        idx = np.arange(10) + 1.0
        # linear regression on log‑log to get beta, shift ≈ 0
        coeffs = np.polyfit(np.log(idx), np.log(first_vals + 1e-12), 1)
        beta_est = -coeffs[0]
    except Exception:
        beta_est = 0.5

    shift_est = 0.0
    betas_fine = np.linspace(max(0.1, beta_est - 0.2),
                             min(1.2, beta_est + 0.2), 21)
    shifts_fine = np.linspace(max(0.0, shift_est - 5.0),
                              shift_est + 5.0, 21)
    print("[grid] Starting fine grid search")
    for beta in betas_fine:
        for shift in shifts_fine:
            if time.time() > deadline:
                break
            cand = _build_powerlaw(beta, shift, n, target_sum)
            if cand is None:
                continue
            score = _conv_max(cand)
            if score < best_score:
                best_score = score
                best_arr = cand.copy()
                print(f"[fine] New best: score={best_score:.6f}, beta={beta:.3f}, shift={shift:.2f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3) Local combinatorial refinement (random swaps)
    # ------------------------------------------------------------------
    print("[refine] Starting random‑swap local improvement")
    # Number of swaps is limited by remaining time.
    # We keep the current best vector unchanged unless a swap yields a better score.
    while time.time() < deadline:
        # pick two distinct indices uniformly at random
        i, j = random.sample(range(n), 2)
        # try swapping the two values
        trial = best_arr.copy()
        trial[i], trial[j] = trial[j], trial[i]

        # score after swap
        trial_score = _conv_max(trial)

        if trial_score < best_score:
            best_score = trial_score
            best_arr = trial
            print(f"[refine] Accepted swap ({i},{j}) -> score={best_score:.6f}")

        # occasional break to avoid tight loop when time is nearly up
        if random.random() < 0.001:
            # a tiny pause allows the outer time check to happen less often
            pass

    # ------------------------------------------------------------------
    # 4) Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] Best score after search: {best_score:.6f}")
    return best_arr.tolist()
```
