def solve_lp_cutting(f, M, max_iters=10, tolerance=1e-6, time_limit=5.0):
    n = len(f)
    # initial active set: just the shift where max occurs (choose max index)
    conv_f = np.convolve(f, f)
    max_idx = int(np.argmax(conv_f))
    active = {max_idx}
    # maybe also add other large shift indices: those with conv >= M*0.9
    for k, val in enumerate(conv_f):
        if val >= M * 0.9:
            active.add(k)

    # Prepare nonnegativity constraints: -I * b <= 0 (b >= 0)
    # We'll include them once.

    for it in range(max_iters):
        # Build A_ub and b_ub
        A_rows = []
        b_rows = []
        for k in active:
            # row for shift k: sum_i f_i * b_{k-i} <= M
            row = np.zeros(n)
            # i iterates over indices where both i and j = k - i in range [0, n-1]
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # i_start .. i_end inclusive
            for i in range(i_start, i_end + 1):
                j = k - i
                row[j] = f[i]  # f[i] * b[j]
            A_rows.append(row)
            b_rows.append(M)
        # Add nonnegativity rows
        A_rows.append(-np.eye(n))
        b_rows.append(np.zeros(n))
        A_ub = np.vstack(A_rows)
        b_ub = np.hstack(b_rows)

        c = -np.ones(n)  # maximize sum(b) -> minimize -sum(b)
        try:
            res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': time_limit})
        except Exception as e:
            # if fails, return None
            return None
        if not res.success:
            return None
        b = res.x
        # Compute conv of f with b
        conv_fb = np.convolve(f, b)
        # Check for violations > M + tolerance
        violations = [k for k, val in enumerate(conv_fb) if val > M + tolerance]
        if not violations:
            # No violations, solution feasible
            return b
        # Add violating shifts to active set
        for k in violations:
            active.add(k)
    # after max_iter, return best feasible found (maybe None)
    return None

def max_convolution(seq):
    n = len(seq)
    size = 1 << (2*n-1).bit_length() # next power of two >= 2n-1
    fft_a = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    # conv length = size, but we need first 2n-1 elements
    max_val = np.max(conv[:2*n-1])
    return max_val

    # initial active shifts containing argmax of conv_f_f (or maybe all)
    conv_f_f = np.convolve(f, f)
    # identify shifts where conv_f_f >= M * 0.9
    active_shifts = set(np.where(conv_f_f >= M * 0.9)[0].tolist())
    # optionally always include shift where M occurs (max index)
    if not active_shifts:
        active_shifts.add(int(np.argmax(conv_f_f)))
    # Also include shift 0 maybe.
    active_shifts.add(0)

    # Preprocess negative constraints rows (b >=0)
    # We'll add them after loop.

    for iteration in range(max_iter):
        # Build A_ub matrix:
        rows = []
        b_vec = []
        for k in active_shifts:
            row = np.zeros(n)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # fill row for j = k - i
            # Loop i = i_start..i_end
            # Equivalent: row[j] = f[i] where j = k - i; we can assign directly
            for i in range(i_start, i_end+1):
                j = k - i
                row[j] = f[i]
            rows.append(row)
            b_vec.append(M)

        # add nonneg constraints
        rows.append(-np.eye(n))
        b_vec.append(np.zeros(n))
        A_ub = np.vstack(rows)
        b_ub = np.concatenate(b_vec)

        # objective
        c = -np.ones(n)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
        if not res.success:
            return None
        b = res.x

        # Compute conv_f_b
        conv_fb = np.convolve(f, b)
        # Identify violations where conv_fb > M + tol
        violation_indices = [k for k, val in enumerate(conv_fb) if val > M + tol]
        if not violation_indices:
            return b
        # Add violations to active set
        active_shifts.update(violation_indices)
        # If active set size grows overly large, maybe break to avoid huge matrix
        if len(active_shifts) > 5 * n: # unrealistic.
            break

    return None

def project_onto_simplex(v, s):
    # v: array, s: desired sum (>=0)
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # find rho = max i where u_i - (cssv_i)/i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v, s):
    # returns projection of v onto simplex {x >=0, sum(x) = s}
    # algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step–function using a cutting–plane linear program
to enlarge the L1‑norm while keeping the self‑convolution bounded.
The algorithm starts from a known sequence (height_sequence_1 if present)
and repeatedly:
  * computes the current maximum convolution value M,
  * solves  max sum(b)  s.t.  conv(f,b) ≤ M  (non‑negative b),
    using an active‑set of the most binding convolution constraints,
  * blends the current solution with the LP result using a line search,
  * keeps the best sequence seen.
A simple projected‑gradient fallback is used when the LP fails.
The whole search runs for the allotted time budget and returns the best
sequence found.  All operations respect the non‑negative, ≤1000 limits
imposed by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, given sum)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x >= 0, sum(x)=s} (a scaled simplex)."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    # Duchi et al. (2008) – O(n log n) implementation
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution (FFT) – used many times during the search
# ----------------------------------------------------------------------
def _max_convolution(seq: np.ndarray) -> float:
    """Return max value of the full (linear) convolution of seq with itself."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())         # next power of two
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)
    return float(np.max(conv[: 2 * n - 1]))


# ----------------------------------------------------------------------
# Cutting‑plane LP: maximize sum(b) with conv(f,b) ≤ M.
# Only a subset of the (2n‑1) convolution constraints is kept,
# growing the set until no violation is observed.
# ----------------------------------------------------------------------
def _solve_convolution_lp_cutting(f: np.ndarray, M: float, *,
                                 max_outer_iter: int = 8,
                                 tol: float = 1e-8,
                                 lp_time_limit: float = 4.0) -> np.ndarray | None:
    n = f.shape[0]

    # start with the most critical shifts (those already close to the bound)
    conv_f = np.convolve(f, f)
    active = {int(np.argmax(conv_f))}
    # also keep any shift whose convolution is at least 90 % of M
    high = np.where(conv_f >= 0.9 * M)[0]
    active.update(map(int, high))

    # ------------------------------------------------------------------
    # iterative refinement of the active set
    # ------------------------------------------------------------------
    for _ in range(max_outer_iter):
        # build the linear constraints for the current active set
        rows = []
        rhs = []
        for k in active:
            row = np.zeros(n, dtype=float)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # vectorised fill: j = k - i
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            row[j_idx] = f[i_idx]
            rows.append(row)
            rhs.append(M)

        # non‑negativity constraints  -b ≤ 0
        rows.append(-np.eye(n))
        rhs.append(np.zeros(n, dtype=float))

        A_ub = np.vstack(rows)
        b_ub = np.concatenate(rhs)

        # maximise sum(b)  → minimise -sum(b)
        c = -np.ones(n, dtype=float)

        try:
            res = optimize.linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                method="highs",
                options={"time_limit": lp_time_limit},
            )
        except Exception:
            return None

        if not res.success:
            return None

        b = res.x

        # check all convolution constraints; add any that violate the bound
        conv_fb = np.convolve(f, b)
        viol = np.where(conv_fb > M + tol)[0]
        if viol.size == 0:
            return b
        active.update(map(int, viol))

    # if we exit the loop without a feasible point, give up
    return None


# ----------------------------------------------------------------------
# Main search routine – the function required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # initialise – prefer the existing construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(list(height_sequence_1), dtype=float)
    else:
        # fallback: a short random start
        cur = rng.random(500, dtype=float)

    cur = np.clip(cur, 0.0, 1000.0)

    n = cur.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # ensure a reasonable sum (the evaluation is scale‑invariant,
    # but a very tiny sum would trigger the “sum too close to zero” guard)
    if cur.sum() < 0.01:
        cur = np.full_like(cur, target_sum / n)

    # normalise once – the algorithm works with a fixed sum,
    # which keeps the objective simply the maximum convolution.
    cur = cur * (target_sum / cur.sum())
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # iterative improvement loop
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # current bound
        cur_max = _max_convolution(cur)

        # ------------------------------------------------------------------
        # 1) try the cutting‑plane LP direction
        # ------------------------------------------------------------------
        g = _solve_convolution_lp_cutting(cur, cur_max, lp_time_limit=3.0)

        if g is not None:
            # re‑normalise the LP result to the same L1‑norm as ``cur``
            g = np.clip(g, 0.0, 1000.0)
            if g.sum() == 0.0:
                g = cur.copy()
            else:
                g = g * (target_sum / g.sum())

            # evaluate the raw LP point – it might already be better
            score_g = evaluate_sequence(g.tolist())
            if score_g < best_score:
                best_score = score_g
                best_seq = g.copy()
                cur = g.copy()
                print(f"[{it}] LP step improved directly to {best_score:.6f}")
                continue

            # -----------------------------------------------------------------
            # 2) line‑search blending between cur and g
            # -----------------------------------------------------------------
            improved = False
            for t in (0.02, 0.05, 0.10, 0.20, 0.33, 0.50):
                cand = (1.0 - t) * cur + t * g
                cand = cand * (target_sum / cand.sum())   # keep the sum fixed
                cand = np.clip(cand, 0.0, 1000.0)

                # a cheap approximation first (FFT); only call the full checker
                # if we see a promising improvement.
                cand_max = _max_convolution(cand)
                cand_score_est = 2.0 * n * cand_max / (cand.sum() ** 2)

                if cand_score_est < best_score - 1e-6:
                    cand_score = evaluate_sequence(cand.tolist())
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        cur = cand.copy()
                        improved = True
                        print(f"[{it}] blend t={t:.2f} improved to {best_score:.6f}")
                        break  # accept the first improving blend

            if improved:
                continue   # go to next iteration

        # ------------------------------------------------------------------
        # 3) fallback – small random perturbation (projected onto simplex)
        # ------------------------------------------------------------------
        # pick two indices and transfer a tiny amount of mass
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = 0.001 * target_sum / n   # a tiny chunk
        tmp = cur.copy()
        tmp[i] = max(0.0, tmp[i] - delta)
        tmp[j] = min(1000.0, tmp[j] + delta)
        # re‑project to keep the exact L1‑norm
        tmp = _project_onto_simplex(tmp, target_sum)
        cur = tmp

        # occasional full re‑evaluation (cheap check)
        cur_score_est = 2.0 * n * _max_convolution(cur) / (target_sum ** 2)
        if cur_score_est < best_score:
            cur_score = evaluate_sequence(cur.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[{it}] random tweak improved to {best_score:.6f}")

        # every few hundred iterations restart from the best known point
        if it % 300 == 0:
            cur = best_seq.copy()

    # final safety: clip to the allowed range and return as a plain list
    final = np.clip(best_seq, 0.0, 1000.0).tolist()
    return final
