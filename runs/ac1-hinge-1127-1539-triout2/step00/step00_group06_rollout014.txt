Project onto simplex algorithm:

def simplex_projection(v):
    """Projects a vector v onto the probability simplex {x >=0, sum x = 1}."""
    # ensure numpy array
    v = np.asarray(v)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Geometric‑decay construction for the step‑function evaluation.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

For a non‑negative sequence a of length n we can force the maximal
convolution value to be small by making a decay geometrically:
        a_i ∝ r**i ,   0 < r < 1
and then normalising the vector to have sum 1.

For such a sequence the dominant term of the convolution occurs near
shift k ≈ 1/(1‑r) and its height is approximately (1‑r)/e.
Consequently the score is roughly

    score ≈ 2·n·(1‑r)/e

Choosing 1‑r (called α) as α = target·e/(2·n) gives a score close to the
desired target.  Using a slightly smaller α (e.g. 95 % of the value)
gives a safe margin and a strictly lower score.

The function below builds several candidates with different safety
factors, evaluates them and returns the best one found before the
budget expires.
"""

import time
import numpy as np

def geometric_sequence(length: int, alpha: float) -> np.ndarray:
    """
    Return a normalised geometric sequence of given length.
    a_i = r**i  with r = 1‑alpha, then scaled so that sum(a)=1.
    """
    r = 1.0 - alpha
    # powers r**i, i = 0 .. length-1
    idx = np.arange(length, dtype=np.float64)
    raw = np.power(r, idx)
    total = raw.sum()
    # normalise to unit L1 norm (the evaluation function only cares about the
    # ratio, but a sum of exactly 1 simplifies the analysis)
    return raw / total


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function sequence with a very low evaluation score.

    The algorithm:
      * fix a length (the previously found construction used 6597 elements;
        this size gives plenty of degrees of freedom and fits comfortably
        in the time budget);
      * compute a geometric decay factor α that would give a score around
        1.5 (the current target);
      * try a few safety factors around this α, keep the best sequence;
      * return the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # keep a small safety margin

    # length of the construction – 6597 is the current size, keeping it
    # simplifies comparison with the reference solution.
    n = 6597

    # Desired score (target).  Anything below 1.5 is a win.
    desired_score = 1.503

    # Theoretical relation:  score ≈ 2·n·α/e   →   α ≈ score·e/(2·n)
    base_alpha = desired_score * np.e / (2.0 * n)

    # Test a few safety factors around the theoretical value.
    safety_factors = [0.6, 0.8, 0.95, 1.0, 1.2]
    best_seq = None
    best_score = float('inf')

    for factor in safety_factors:
        if time.time() >= deadline:
            break
        alpha = base_alpha * factor
        # α must be in (0, 1).  Guard against pathological values.
        if not (0.0 < alpha < 1.0):
            continue

        seq = geometric_sequence(n, alpha)
        # The evaluation function expects a plain list of Python floats.
        score = evaluate_sequence(seq.tolist())
        print(f"alpha={alpha:.6e}  factor={factor:.2f}  score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # In the unlikely case that nothing succeeded, fall back to a simple
    # uniform vector (score exactly 2) – this guarantees a valid return.
    if best_seq is None:
        print("Fallback to uniform sequence.")
        best_seq = np.full(n, 1.0 / n)

    print(f"Best score obtained: {best_score:.6f}")
    return best_seq.tolist()
