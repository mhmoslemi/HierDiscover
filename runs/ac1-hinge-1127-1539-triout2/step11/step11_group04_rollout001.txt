"""
Hybrid optimizer for the self‑convolution bound – version 2

Main ideas
~~~~~~~~~~
* Use a *larger* final dimension (up to 20 k) – more degrees of freedom give a lower
  maximal convolution value (the theoretical optimum is ~1 for large n).

* After a long Adam refinement we search for a *good permutation* of the values.
  The permutation is restricted to the family “place the sorted values at positions
  (i·step) mod n” for many coprime steps (including random ones).  This spreads the

* Finish with a short deterministic mass‑transfer (greedy Metropolis) phase.
* All phases respect the ℓ¹‑simplex constraint `∑a = √(2 n)` and non‑negativity.
* The routine works under the 1000 s budget imposed by the competition harness
  and stores the best found sequence in the global variable ``height_sequence_1``.

import time

# ----------------------------------------------------------------------
# Convolution utilities (FFT based) – identical to the reference code
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)² (for ∑a=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0 , Σ x_i = s }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _step_spread_order(seq: np.ndarray, step: int) -> np.ndarray:
    """Place sorted values at indices (i*step) mod n (step must be coprime to n)."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]   # descending
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

            continue
        # ensure coprime
        while np.gcd(step, n) != 1:
            step = (step + 1) % n or 1
        candidates.add(step)

    # a few random coprime steps
    rng = np.random.default_rng()
    while len(candidates) < 15 and time.time() < time_deadline:
        step = rng.integers(1, n)
        while np.gcd(step, n) != 1:
            step = rng.integers(1, n)
        candidates.add(step)

    for step in candidates:
        if time.time() >= time_deadline:
            break
        candidate = _step_spread_order(seq, step)
        val = _max_convolution_fft(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate

    return best_seq


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _adam_subgradient(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      base_lr: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the sub‑gradient of  max(conv(a,a)).
    The gradient is assembled from all indices attaining the current peak.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    t = 0
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # compute convolution and identify the maximal entries
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices where the peak is (with a tiny tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        # construct sub‑gradient
        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate scales with target_sum and decays with the iteration count
        lr = base_lr * target_sum / (np.sqrt(v_hat) + eps)

        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0003, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam sub‑grad it={t:5d} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (as in the reference code)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.3,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves with an exponential temperature schedule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate one of the maximal convolution entries
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasionally recompute the convolution (already done after each move)
        # and apply a tiny global jitter
        if time.time() - start_time > 30 and (np.random.rand() < 0.01):
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    Returns a plain list of floats (the final sequence).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1️⃣ Choose a target dimension – larger is better if we have time.
    # ------------------------------------------------------------------
    # Rough schedule:
    #   * < 300 s  → n≈  8000
    #   * 300‑600 s → n≈ 12000
    #   * > 600 s  → n≈ 20000
    remaining_total = deadline - time.time()
    if remaining_total > 600:
        n_final = 20000
    elif remaining_total > 300:
        n_final = 12000
    else:
        n_final = 8000

    target_final = np.sqrt(2.0 * n_final)

    # ------------------------------------------------------------------
    # 2️⃣ Build an initial candidate.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.array(globals()["height_sequence_1"], dtype=float)
        init_seq = _resize_sequence(base_seq, n_final)
    else:
        # create a modestly shaped start: power‑law with exponent ~0.55
        init_seq = _power_law_sequence(n_final, 0.55, target_final)

    # Guarantee feasibility
    init_seq = _project_to_simplex(init_seq, target_final)

    # ------------------------------------------------------------------
    # 3️⃣ Adam sub‑gradient refinement (most of the budget).
    # ------------------------------------------------------------------
    sub_deadline = deadline - 20.0        # keep ~20 s for ordering + final polishing
    a_refined, val_refined = _adam_subgradient(
        init_seq,
        deadline=sub_deadline,
        target_sum=target_final,
        base_lr=0.4,
        beta1=0.9,
        beta2=0.999,
        eps=1e-8,
    )
    best_seq = a_refined
    best_score = val_refined
    print(f"[{time.time():.2f}] After Adam refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Search for a better permutation (step‑spread orders).
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        order_deadline = deadline - 8.0
        best_seq = _best_step_order(best_seq, order_deadline)
        best_score = _score(best_seq)
        print(f"[{time.time():.2f}] After ordering score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final deterministic mass‑transfer (short).
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_deadline = deadline - 2.0
        a_final, val_final = _deterministic_mass_transfer(
            best_seq,
            deadline=mt_deadline,
            target_sum=target_final,
            start_temp=0.2,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] After final MT score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Trim trailing near‑zero entries for compactness.
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 7️⃣ Optional final evaluation (for logging only – does not affect the result).
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for possible future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
