"""
Optimisation of the non‑negative step‑function for the self‑convolution L∞‑norm.

Main ideas
----------
1. Choose a problem size `n` (≈15000 for a long budget) and set the L¹‑norm
   to `target = sqrt(2·n)`.  With this normalisation the evaluation function
   reduces to the maximal entry of the linear convolution `a*a`.

2. Build a rich pool of parametric shapes (power‑law, exponential,
   sine‑raised, discrete Gaussian).  Project each shape onto the simplex
   `{a≥0, Σa = target}` and keep the best few.

3. Refine the best candidates with an **Adam** optimiser applied to a
   soft‑max approximation of the maximum.  The temperature (β) is increased
   exponentially up to 5·10⁵, turning the smooth objective into the true
   max.  After each update the sequence is projected back onto the simplex.

4. The top refined candidates are averaged, projected again and run through a
   final Adam phase (more iterations, higher β) – this helps escape local
   minima.

5. A short sub‑gradient descent on the true max‑convolution follows, then a
   deterministic mass‑transfer (local annealing) phase.

6. The best sequence seen during the whole run is returned (trailing zeros
   trimmed).  The sequence is stored in the global variable
   `height_sequence_1` for warm‑starts in subsequent runs.

The whole pipeline respects the 1000 s budget by constantly checking the
deadline and shrinking the allocated time for each phase accordingly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (uses FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (scaled)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto {x ≥ 0, Σx = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (unnormalised)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """(i+1)^{-p}."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """exp(-lam·i)."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    """sin(π·(i+0.5)/n)^{p}."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at n/2."""
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    return np.exp(-0.5 * (x / sigma) ** 2)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv_vector).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)

    # convolution a*a
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    max_b = np.max(conv)

    # soft‑max probabilities
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (rev(a) * p) restricted to the central n entries
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------
def _adam_softmax_opt(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      max_beta: float = 5e5,
                      base_lr: float = 0.6,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation of the soft‑max approximation of the
    maximum self‑convolution.  Returns the best vector seen and its max‑conv.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    beta = 0.5
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        it += 1
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate adapts to current β (larger β ⇒ steeper landscape)
        lr = base_lr * target_sum / (beta + 1.0)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * jitter_scale / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d} β={beta:6.1f} best={best_val:.6f}")

        # increase β periodically (log‑schedule)
        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val

# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.55,
                         gamma: float = 0.95,
                         jitter_interval: int = 400,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent with momentum on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: 2*a[kmax - i] where index valid
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:6d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

# Deterministic mass‑transfer (local annealing)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.6,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer that tries to lower the current peak."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]                 # contribution of each pair

        donor = idx[np.argmax(w)]
        # pick a low‑contributing partner
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        # Metropolis‑type acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:6d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

        perm = base_seq.copy()
        rng.shuffle(perm)
        candidates.append(perm)

    # ------------------- quick screening -------------------
    scored = []
    for seq in candidates:
        val = _eval(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])

    top_k = 5
    best_seq = scored[0][1].copy()
    best_score = scored[0][0]
    print(f"[{time.time():.2f}] Initial best score from pool = {best_score:.6f}")

    # ------------------- Phase A: Adam soft‑max refinement -------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # allocate roughly 40 % of the remaining time to Adam on the top candidates
        adam_total = remaining * 0.4
        per_candidate = adam_total / top_k
        refinements = []
        for i in range(top_k):
            seq0 = scored[i][1]
            sub_dead = time.time() + min(per_candidate, deadline - time.time())
            a_opt, val_opt = _adam_softmax_opt(
                seq0,
                sub_dead,
                target_sum,
                max_beta=5e5,
                base_lr=0.6,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                jitter_scale=1e-3,
            )
            refinements.append((val_opt, a_opt))
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] Adam‑phase improved best to {best_score:.6f}")

        # keep the three best refined candidates for the averaging step
        refinements.sort(key=lambda x: x[0])
        top_refined = [x[1] for x in refinements[:3]]
    else:
        top_refined = [best_seq]

    # ------------------- Phase A2: averaging & final Adam -------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and len(top_refined) > 1:
        avg_seq = np.mean(np.stack(top_refined, axis=0), axis=0)
        avg_seq = _project_to_simplex(avg_seq, target_sum)

        a_final, val_final = _adam_softmax_opt(
            avg_seq,
            deadline,
            target_sum,
            max_beta=8e5,
            base_lr=0.4,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            jitter_scale=5e-4,
        )
        if val_final < best_score:
            best_score, best_seq = val_final, a_final.copy()
            print(f"[{time.time():.2f}] Averaged‑Adam improved best to {best_score:.6f}")

    # ------------------- Phase X: sub‑gradient descent -------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        a_sg, val_sg = _subgradient_descent(
            best_seq,
            target_sum,
            time.time() + remaining * 0.35,
            step_factor=0.55,
            gamma=0.95,
            jitter_interval=400,
            jitter_scale=1e-3,
        )
        if val_sg < best_score:
            best_score, best_seq = val_sg, a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best to {best_score:.6f}")

    # ------------------- Phase B: mass‑transfer refinement -------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            time.time() + remaining * 0.5,
            target_sum,
            start_temp=0.6,
            end_temp=1e-5,
            low_frac=0.04,
        )
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_score:.6f}")

    # ------------------- final polishing (short Adam) -------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_fin, val_fin = _adam_softmax_opt(
            best_seq,
            time.time() + remaining,
            target_sum,
            max_beta=1e6,
            base_lr=0.3,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            jitter_scale=2e-4,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing improved best to {best_score:.6f}")

    # ------------------- post‑process & store -------------------
    final_arr = np.array(best_seq, dtype=float)
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]
    final_seq = final_arr.tolist()

    # one extra evaluation for logging (the official scoring routine)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
