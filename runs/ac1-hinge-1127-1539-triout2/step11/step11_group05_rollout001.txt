evaluation function

    2·n·max(a*a) / (sum(a)^2)

The algorithm works in several stages inside the given time budget:

1. **Problem size** – a larger `n` (up to 20000) gives more freedom to lower
   the max‑convolution.  For a 1000 s budget we use `n = 20000`,
   otherwise we fall back to 12000/8000.

2. **Candidate pool** – a diverse set of analytic shapes (Gaussian,
   power‑law, exponential, sine‑power, tent, uniform) plus jittered copies.
   All candidates are projected onto the simplex
   `{x≥0, Σx = sqrt(2·n)}`.  The best few (≈6) are kept.

3. **Adam‑SGD on a soft‑max approximation** – we minimise a smooth
   approximation of `max(a*a)`:

        F_β(a) = (1/β)·log Σ exp(β·(a*a))

   The gradient is obtained by one FFT‐based convolution.
   Adam’s adaptive step is combined with a schedule that raises `β`
   (doubling every few hundred iterations) up to `β_max`.  A small
   Gaussian jitter is added occasionally to escape shallow minima.

4. **Exact sub‑gradient descent** – after the soft‑max phase we switch to the
   true max‑convolution, using the sub‑gradient that is non‑zero only at the
   current peak index.

5. **Deterministic mass‑transfer (local annealing)** – we greedily move mass
   from the most “responsible’’ pair of indices (the ones that realise the
   current maximum) to the least responsible ones, accepting moves with a
   temperature schedule.

6. **Final polishing** – a short Adam run with a very large `β` (≈5·10⁵).

The best sequence found is trimmed (trailing near‑zero entries removed),
stored globally as `height_sequence_1` for warm‑starts in subsequent runs,

import time

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto {x ≥ 0, Σx = s} using the O(n log n) algorithm."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _gaussian_sequence(n: int, sigma_ratio: float) -> np.ndarray:
    """Unnormalised Gaussian centred in the middle, sigma = sigma_ratio·n."""
    mu = (n - 1) / 2.0
    sigma = sigma_ratio * n
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Unnormalised (i+1)^(-p)."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Unnormalised exp(-lam·i)."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    """Unnormalised (sin(π·(i+0.5)/n))^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _tent_sequence(n: int) -> np.ndarray:
    """Symmetric triangular shape: max at centre, linearly to zero at ends."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    return np.minimum(i, n - 1 - i)


def _uniform_sequence(n: int) -> np.ndarray:
    """Uniform shape."""
    return np.ones(n, dtype=np.float64)


def _make_symmetric(seq: np.ndarray) -> np.ndarray:
    """Force exact symmetry around the centre (useful for odd/even n)."""
    n = seq.shape[0]
    half = n // 2
    if n % 2 == 0:
        left = seq[:half]
        right = left[::-1]
        return np.concatenate([left, right])
    else:
        left = seq[:half + 1]
        right = left[:-1][::-1]
        return np.concatenate([left, right])


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of F_β(a) = (1/β)·log Σ exp(β·conv) . Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    # avoid overflow
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)                     # softmax weights, length 2n‑1

    # grad = 2 * (rev(a) ⋆ p) restricted to central n entries
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on smooth max (soft‑max)
# ----------------------------------------------------------------------
def _adam_sgd_softmax(a0: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      beta_start: float = 0.5,
                      beta_max: float = 5e5,
                      beta_mul: float = 2.0,
                      beta_inc_it: int = 400,
                      learning_rate: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max approximation of max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = beta_start
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam β={beta:.1f} it={it:5d} best={best_val:.6f}")

        # increase β gradually
        if it % beta_inc_it == 0 and beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_it: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution (non‑smooth)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: g[i] = 2 * a[kmax - i] when index valid
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_it == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (local annealing)
# ----------------------------------------------------------------------
def _mass_transfer(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  start_temp: float = 0.6,
                  end_temp: float = 1e-5,
                  low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """Greedy mass moves from high‑weight to low‑weight pairs."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # location of current peak
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution weight of each pair (i, kmax‑i)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        # pick a receiver among low‑contributing indices
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # maximum transferable mass (keep positivity)
        max_delta = min(a[donor] * 0.5, a[donor] - 0.0)

        # also bound by the weight difference to avoid overshooting
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Mass‑transfer it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to keep diversity
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the evaluation metric.
    Returns a list of floats ready for `evaluate_sequence`.
    """
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose problem dimension
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        if budget_s >= 800:
            n = 20000          # larger size gives more flexibility
        elif budget_s >= 500:
            n = 15000
        elif budget_s >= 300:
            n = 12000
        else:
            n = 8000
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Helper for fast evaluation
    # --------------------------------------------------------------
    def _eval(arr: np.ndarray) -> float:
        return _max_convolution_fft(arr)

    # --------------------------------------------------------------
    # Build a diverse pool of initial shapes
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # Warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        candidates.append(prev)

    # Analytic families
    # 1) Gaussian (symmetric)
    for sigma_ratio in np.linspace(0.06, 0.22, 12):
        seq = _gaussian_sequence(n, sigma_ratio)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

        # jittered copy
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # 2) Power‑law
    for p in np.linspace(0.30, 0.90, 15):
        seq = _power_law_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # 3) Exponential decay
    for lam in np.linspace(3e-4, 0.015, 12):
        seq = _exponential_sequence(n, lam)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # 4) Sine‑power (symmetric)
    for p in np.linspace(0.2, 2.2, 10):
        seq = _sin_power_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # 5) Tent shape (piecewise linear)
    seq = _tent_sequence(n)
    seq = _project_to_simplex(seq, target_sum)
    candidates.append(seq)

    # 6) Uniform (baseline)
    seq = _uniform_sequence(n)
    seq = _project_to_simplex(seq, target_sum)
    candidates.append(seq)

    # Random dense vectors (few)
    for _ in range(4):
        rand = rng.random(size=n)
        rand = _project_to_simplex(rand, target_sum)
        candidates.append(rand)

    # --------------------------------------------------------------
    # Quick evaluation of all candidates and keep the best few
    # --------------------------------------------------------------
    scored = []
    for arr in candidates:
        scored.append((_eval(arr), arr))
    scored.sort(key=lambda x: x[0])

    top_k = 6
    best_seq = scored[0][1].copy()
    best_score = scored[0][0]
    print(f"[{time.time():.2f}] Initial best from pool = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on soft‑max (exploration)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        phase_time = remaining * 0.35
        per_cand = phase_time / top_k
        for i in range(top_k):
            if time.time() >= deadline - 0.5:
                break
            a0 = scored[i][1]
            sub_dead = min(time.time() + per_cand, deadline - 0.2)
            a_opt, val_opt = _adam_sgd_softmax(
                a0,
                target_sum,
                sub_dead,
                beta_start=0.5,
                beta_max=3e5,
                learning_rate=0.6,
                beta_inc_it=300
            )
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] Adam improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        sub_dead = time.time() + remaining * 0.25
        a_sg, val_sg = _subgradient_descent(best_seq, target_sum, sub_dead,
                                            step_factor=0.55, gamma=0.95)
        if val_sg < best_score:
            best_score, best_seq = val_sg, a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Deterministic mass transfer (local annealing)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        mt_dead = time.time() + remaining * 0.30
        a_mt, val_mt = _mass_transfer(best_seq, target_sum, mt_dead,
                                      start_temp=0.6,
                                      end_temp=1e-5,
                                      low_frac=0.04)
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Final polishing with high‑β Adam
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_fin, val_fin = _adam_sgd_softmax(
            best_seq,
            target_sum,
            time.time() + remaining,
            beta_start=5e5,
            beta_max=5e5,
            learning_rate=0.5,
            beta_inc_it=1000
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final post‑processing
    # --------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=float)
    # Trim trailing near‑zero entries (tolerance 1e-12)
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]

    final_seq = final_arr.tolist()

    # Evaluate with the provided function (if it exists)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # Store globally for warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
