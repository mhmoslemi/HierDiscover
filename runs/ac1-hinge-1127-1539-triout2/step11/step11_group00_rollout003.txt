def anneal_mass_transfer(a0, target_sum, deadline):
    a = a0.copy()
    n = len(a)
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()
    T0 = cur_val * 0.8  # bigger temperature
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # slower cooling
        temperature = max(0.0, T0 * (1.0 - frac**0.5))
        # pick donor and receiver
        # choose k_max with max b
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        idx = np.arange(lo, hi+1)
        # contributions to max for each i
        contrib = a[idx] * a[k_max - idx]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            donor = rng.choice(idx, p=contrib / total_contrib)
        else:
            donor = rng.integers(lo, hi+1)
        # receiver choose some low a
        # measure low a with probability inversely proportional to a
        prob = 1.0/(a+1e-12)
        prob = prob / prob.sum()
        receiver = rng.choice(n, p=prob)

        if donor == receiver:
            continue

        # delta size: allow up to 5% of donor mass or 0.5% of total sum, whichever smaller,
        # scaled by temperature ratio
        max_delta = min(a[donor]*0.05, target_sum*0.005)
        if max_delta <= 0:
            continue
        # scale factor: temperature/T0 or maybe sqrt
        scale = temperature / T0 if T0>0 else 0
        delta = rng.random() * max_delta * scale
        if delta <= 0:
            continue

        # compute b_new efficiently
        b_new = b.copy()
        # apply linear terms
        # For array indices: we need efficient vectorized add operations across slices.
        b_new[donor:donor+n] -= 2.0*delta*a
        b_new[receiver:receiver+n] += 2.0*delta*a
        # quadratic terms
        b_new[2*donor] += delta*delta
        b_new[2*receiver] += delta*delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0*delta*delta

        new_val = float(2.0*n*b_new.max()/(target_sum*target_sum))

        # acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob_accept = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob_accept:
                accept = True
        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

"""
Optimisation of a non‑negative step function for the benchmark
-----------------------------------------------------------------
The algorithm follows three successive stages:

1. **Multiplicative mirror descent** on a soft‑max surrogate of the
   max‑convolution.  This quickly drives the peak down while keeping the
   sum of the coefficients fixed (∑a = √(2 n)).

2. **Multiplicative sub‑gradient descent** on the true max‑convolution.
   The sub‑gradient is the exact derivative of the active peak and
   provides a sharper directional signal.

3. **Improved simulated‑annealing mass‑transfer**.
   * The Metropolis acceptance uses the *current* energy (not the
     historical best) which greatly improves exploration.
   * The temperature starts at a larger fraction of the current score
     (≈ 80 %) and cools **non‑linearly** (√‑schedule) to keep the early
     phase hot.
   * Transfer size is scaled with the temperature and allowed up to
     5 % of the donor mass (or 0.5 % of the total sum) which makes the
     moves larger when the temperature is high.
   * Donor atoms are sampled proportionally to their contribution to the
     current maximal convolution entry; receivers are sampled inversely
     proportional to their height.

All phases respect a global time budget (default 1000 s).  The current
best sequence is stored in the global variable `height_sequence_1`
to serve as a warm‑start in subsequent runs.

The returned value is a plain list of floats (trailing zeros are cut
off) ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective  2·n·max(a*a) / (sum(a)²).
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – multiplicative (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – multiplicative sub‑gradient descent on the true max‑conv.
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – improved simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Long‑run annealing that moves tiny amounts of mass from “bad’’ to “good’’
    indices.  Acceptance is based on the current energy (Metropolis) and
    the temperature decays with a √‑schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # start temperature at a large fraction of the current score
    T0 = cur_val * 0.80

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # non‑linear cooling: keep temperature high longer
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with relatively small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.05, target_sum * 0.005)
        if max_delta <= 0:
            continue
        # scale factor grows with temperature
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on *current* value
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 5000 == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

    # Warm‑start from the best known construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent (≈12 % of budget)
    # --------------------------------------------------------------
    exp_deadline = time.time() + 0.12 * (deadline - time.time())
    a_exp, val_exp = exponential_descent(init, target_sum, exp_deadline)
    best_seq, best_score = a_exp, val_exp
    print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent on the true max‑convolution (≈12 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.12 * (deadline - time.time())
        a_sub, val_sub = subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – long simulated‑annealing refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_seq, best_score = a_ann, val_ann
        print(f"[{time.time():.2f}] After annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
