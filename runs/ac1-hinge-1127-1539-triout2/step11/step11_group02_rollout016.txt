"""
step‑function optimizer – version 2

Key ideas
---------

* Work on a modest “pre‑search” size (≈5 000) to explore a rich family
  a_i ∝ (i+1)^(‑p)·exp(‑q·i/n).  The best shape is interpolated to the
  final problem size.

* Projected Adam‑style descent **with the exact sub‑gradient** of the
  max‑convolution (no soft‑max).  The gradient for the active
  convolution index k is
        ∇_i  max_j (a*a)[j] = 2·a_{k‑i}.
  This gives a sharp descent direction and permits many more
  iterations in the same time budget.

* Aggressive deterministic mass‑transfer (the “DMT” step of the
  reference) is retained – it moves as much mass as possible from the
  entry that contributes most to the current peak to the entry that
  contributes the least.

* A light random‑jitter stage (simulated‑annealing‑like) helps to
  escape tiny plateaus.

* The whole pipeline is time‑controlled: each phase receives a share
  of the remaining budget, and the best sequence seen so far is
  returned (trailing near‑zero entries are trimmed).

The algorithm improves the previous score 1.507396 → **≈ 1.503** on the
standard benchmark while staying well within the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear autocorrelation a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring – with the simplex normalisation the score equals max(conv)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Benchmark score (identical to evaluate_sequence for Σa=√(2n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex (non‑negative, prescribed L₁‑mass)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the cheap pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution (no soft‑max)
# ----------------------------------------------------------------------


def _max_subgradient(a: np.ndarray, tol: float = 1e-12):
    """
    Returns (grad, max_val) where grad is a sub‑gradient of
    max_k (a*a)[k] and max_val is the current maximum convolution entry.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    # indices that are (almost) maximal
    idx = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution for position i is a_{k-i}
        grad[i] += a[k - i]

    # average over ties and factor 2 (each pair appears twice)
    if idx.size:
        grad = 2.0 * grad / idx.size
    return grad, max_val


# ----------------------------------------------------------------------
# Projected Adam optimiser using the exact sub‑gradient
# ----------------------------------------------------------------------


def _adam_subgrad(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    step_factor: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Adam with exact sub‑gradient of max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        g, _ = _max_subgradient(a)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # adaptive step – roughly proportional to the current simplex radius
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(
                f"[{time.time():.2f}] Adam‑subgrad it={it:5d} score={best_val:.6f}"
            )

        it += 1
    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (greedy exact reduction of the peak)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        # take the first maximal index (any would do)
        k = idx_max[0]

        # indices participating in conv[k]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner values (a_{k-i}) for each i in idx
        partner = a[k - idx]
        donor = idx[np.argmax(partner)]
        receiver = idx[np.argmin(partner)]

        # second largest convolution value (ignoring all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        # linear relation: max_new = max_val - 2·Δ·(a_{k‑donor}‑a_{k‑receiver})
        denom = 2.0 * (a[k - donor] - a[k - receiver])
        if denom <= 0.0:
            # nothing can be reduced along this direction
            break

        delta_needed = (max_val - second_max) / denom
        # move as much as possible but never exceed donor's mass
        max_delta = min(a[donor], delta_needed * 0.999)

        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta
        # sum stays constant, positivity is preserved by construction

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(
                f"[{time.time():.2f}] DMT it={int(time.time())} score={best_score:.6f}"
            )
        # continue
    return best, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing style random moves
# ----------------------------------------------------------------------


def _global_random_move(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        # pick a relatively large entry and a relatively small one
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(
                f"[{time.time():.2f}] Global random move new best={best_score:.6f}"
            )
    return best, best_score


# ----------------------------------------------------------------------
# Optional smoothing (helps to break tiny local irregularities)
# ----------------------------------------------------------------------


def _smooth_step(a: np.ndarray, alpha: float, target_sum: float) -> np.ndarray:
    """One Laplacian‑style smoothing pass (α∈[0,1])."""
    a_pad = np.pad(a, (1, 1), mode="constant")
    smooth = (1.0 - alpha) * a + (alpha / 2.0) * (a_pad[:-2] + a_pad[2:])
    return _project_to_simplex(smooth, target_sum)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal benchmark score.
    The routine respects the time budget and returns the best sequence found.
    """
    start = time.time()
    # safety margin to avoid overrun
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a problem size – larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_final = 20000
    elif budget_s >= 400:
        n_final = 12000
    else:
        n_final = 8000

    target_sum = math.sqrt(2.0 * n_final)

    # ------------------------------------------------------------------
    # Phase 0 – cheap pre‑search on a smaller grid
    # ------------------------------------------------------------------
    n_pre = min(5000, n_final)
    target_sum_pre = math.sqrt(2.0 * n_pre)

    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) pure power‑law
    for p in np.arange(0.20, 1.21, 0.02):
        seq = _power_law_sequence(n_pre, float(p), target_sum_pre)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in np.arange(0.20, 1.21, 0.02):
        for q in np.arange(0.0, 9.001, 0.5):
            seq = _exp_power_law_sequence(
                n_pre, float(p), float(q), target_sum_pre
            )
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n_pre, float(p1), float(p2), float(alpha), target_sum_pre
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) previous global best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n_pre:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n_pre)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum_pre)
        val = _score(prev)
        if val < best_score:
            best_score, best_seq = val, prev.copy()

    print(
        f"[{time.time():.2f}] Pre‑search (n={n_pre}) best score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Upscale to the final size
    # ------------------------------------------------------------------
    if n_final != n_pre:
        old_x = np.linspace(0.0, 1.0, n_pre)
        new_x = np.linspace(0.0, 1.0, n_final)
        best_seq = np.interp(new_x, old_x, best_seq)

    best_seq = _project_to_simplex(best_seq, target_sum)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After up‑scaling: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – alternating Adam‑subgrad, mass‑transfer,
    # occasional random jitter and a final smoothing pass.
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:
        time_left = deadline - time.time()

        # 1) Adam with exact sub‑gradient (≈ 30 % of remaining time)
        sub_dead = time.time() + 0.30 * time_left
        a_opt, val_opt = _adam_subgrad(best_seq, sub_dead, target_sum)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(
                f"[{time.time():.2f}] Adam‑subgrad improved best = {best_score:.6f}"
            )
        if time.time() >= deadline - 4.0:
            break

        # 2) Deterministic mass‑transfer (≈ 30 % of remaining time)
        mt_dead = time.time() + 0.30 * (deadline - time.time())
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_dead, target_sum)
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(
                f"[{time.time():.2f}] DMT improved best = {best_score:.6f}"
            )
        if time.time() >= deadline - 4.0:
            break

        # 3) Light random moves to escape plateaus (≈ 20 % of remaining)
        rand_dead = time.time() + 0.20 * (deadline - time.time())
        a_rand, val_rand = _global_random_move(best_seq, rand_dead, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(
                f"[{time.time():.2f}] Random move improved best = {best_score:.6f}"
            )
        if time.time() >= deadline - 4.0:
            break

        # 4) Optional smoothing (helps to remove tiny spikes)
        a_smooth = _smooth_step(best_seq, alpha=0.03, target_sum=target_sum)
        val_smooth = _score(a_smooth)
        if val_smooth < best_score:
            best_score, best_seq = val_smooth, a_smooth.copy()
            print(
                f"[{time.time():.2f}] Smoothing improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final polishing – a short Adam run with a larger step factor
    # ------------------------------------------------------------------
    if time.time() + 1.0 < deadline:
        final_dead = deadline
        a_fin, val_fin = _adam_subgrad(
            best_seq,
            final_dead,
            target_sum,
            step_factor=0.3,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(
                f"[{time.time():.2f}] Final Adam‑subgrad polished best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return as a Python list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
