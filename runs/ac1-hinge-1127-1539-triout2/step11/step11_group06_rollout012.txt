import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Short multiplicative sub‑gradient refinement
# ----------------------------------------------------------------------


def _short_subgradient(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """A lightweight sub‑gradient descent (≈2000 iterations or until deadline)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    n = a.shape[0]
    it = 0

    while time.time() < deadline and it < 3000:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient of the (exact) max‑convolution entry
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # a simple multiplicative step
        eta = 0.25 * (1.0 - it / 3000.0) + 0.05
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad short it={it:4d} score={best_val:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Soft‑max gradient (used for the final polish)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of the max‑convolution entry."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    grad = np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                        np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = 2.0 * grad[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Final projected gradient on the soft‑max surrogate (high β)
# ----------------------------------------------------------------------


def _final_projected_gradient(a0: np.ndarray,
                             target_sum: float,
                             steps: int = 1500) -> np.ndarray:
    """A few projected‑gradient steps with a very large β (≈1e5)."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    beta = 1e5
    step_base = target_sum * 0.12

    for it in range(steps):
        g, _ = _gradient_softmax(a, beta)
        # simple multiplicative update (always moves away from the gradient)
        eta = step_base / (np.linalg.norm(g) + 1e-12)
        a = a * np.exp(-eta * g)
        a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Targeted simulated‑annealing (mass transfer)
# ----------------------------------------------------------------------


def _targeted_sa(a0: np.ndarray,
                 target_sum: float,
                 deadline: float) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA that always moves mass away from the current maximal entry."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)

    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0.0:
        return best_a, best_val

    # initial temperature – a fraction of the excess over the trivial lower bound (≈1)
    T0 = max(0.02, (best_val - 1.0) * 0.8)

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # --- donor (indices that actually contribute to the current max) ---
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idx_range] * a[k_max - idx_range]
        sum_contrib = contrib.sum()
        if sum_contrib <= 0.0:
            donor = rng.integers(lo, hi + 1)
        else:
            donor = rng.choice(idx_range, p=contrib / sum_contrib)

        # --- receiver (any other index) ---
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        # --- propose a transfer ------------------------------------------------
        max_delta = min(a[donor] * 0.20, target_sum * 0.0015)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # --- Metropolis acceptance --------------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_val:.6f}")

        # --- occasional global jitter -----------------------------------------
        if int(elapsed * 100) % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.3            # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (larger n often gives a better bound)
    # ------------------------------------------------------------------
    if budget_s >= 850:
        n = 25000
    elif budget_s >= 600:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law as a fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # ------------------------------------------------------------------
    # Generate a few cheap candidates (warm start + random shapes)
    # ------------------------------------------------------------------
    candidates = [init]

    rng = np.random.default_rng()
    # random power‑law exponents
    for p in (0.30, 0.55, 0.80, 1.10):
        cand = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        cand = cand * (target_sum / cand.sum())
        candidates.append(cand)

    # random exponentials
    for lam in (0.0002, 0.0006, 0.0010):
        cand = np.exp(-lam * np.arange(n, dtype=np.float64))
        cand = cand * (target_sum / cand.sum())
        candidates.append(cand)

    # keep the best of the cheap candidates
    best_seq = candidates[0]
    best_val = _score(best_seq)
    for cand in candidates[1:]:
        val = _score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
    print(f"[{time.time():.2f}] Warm‑start best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Short sub‑gradient refinement (uses at most a few seconds)
    # ------------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 6.0)
    refined_seq, refined_val = _short_subgradient(best_seq, target_sum, sub_deadline)
    if refined_val < best_val:
        best_val = refined_val
        best_seq = refined_seq
        print(f"[{time.time():.2f}] After short sub‑grad best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Main simulated‑annealing phase (dominates the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_seq, sa_val = _targeted_sa(best_seq, target_sum, deadline)
        if sa_val < best_val:
            best_val = sa_val
            best_seq = sa_seq
            print(f"[{time.time():.2f}] After SA best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polish with a few projected‑gradient steps on the soft‑max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq = _final_projected_gradient(best_seq, target_sum, steps=1200)
        final_val = _score(final_seq)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[{time.time():.2f}] After final GD best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and prepare the output
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Final external evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)   # supplied by the platform
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
