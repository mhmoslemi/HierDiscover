"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence(seq) == max_i (seq * seq)[i] when the ℓ¹‑norm is fixed
to √(2·n).  The algorithm mixes three ideas:

1. Randomised families give a diverse set of starting vectors.
2. An iterative linear‑program (LP) step:
   given the current vector f with max convolution M, solve
       maximise   Σ g_j
       subject to f * g ≤ M   (component‑wise) ,   g ≥ 0 .
   The optimum pushes mass to positions where f is small, which tends to
   lower the peak.  The LP is solved with SciPy’s high‑performance HiGHS
   interior‑point method.
   The new candidate is formed as a convex combination (1‑t)·f + t·g and
   projected onto the ℓ¹‑simplex (Σ = √(2·n)).  The step size t is
   reduced until the peak indeed drops.
3. A short Adam sub‑gradient phase on the exact max, followed by the
   deterministic hill‑climb from the reference code.

The search respects the 1000 s wall‑clock budget, tries several
dimensions (256 … 2048) and keeps the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project onto the simplex {x ≥ 0, Σx = s} by simple scaling.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a⋆a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a⋆a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(a) to new_n, followed by simplex projection.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)

# Random initialisers (several simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)

def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """Projected Adam where the gradient is the exact sub‑gradient of max(a*a)."""
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.2 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # convolution a_rev * p
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that attacks the current worst convolution entry."""
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose a destination with the smallest counterpart
        counterpart = np.zeros(n, dtype=float)
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (linear‑time update)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score

def _solve_lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   subject to   f * g ≤ M  (component‑wise) , g ≥ 0.
    Returns the optimal g (unscaled).  Uses SciPy's HiGHS LP solver.
    """
    n = f.shape[0]

    # Build the Toeplitz constraint matrix A (shape (2n‑1, n))
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n - 1, dtype=f.dtype)])
    A = toeplitz(col, row)                     # dense, size (2n‑1)×n

    c = -np.ones(n, dtype=float)               # maximise Σg  → minimise –Σg
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=float)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})
    if not res.success:
        # fallback: return a zero vector (no improvement possible)
        return np.zeros_like(f)
    return res.x


def _lp_iterative_improve(a0: np.ndarray, target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a few LP‑based improvement steps on a0.
    Returns the best vector found and its max‑convolution value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = _max_convolution_fft(a)

    while time.time() < deadline:
        # current peak
        M = best_M

        # 1) solve LP
        g = _solve_lp_step(a, M)
        if g.sum() == 0.0:
            break   # LP could not move

        # 2) try to mix f and g
        improved = False
        t = 0.5
        while t > 1e-4 and time.time() < deadline:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = _project_to_simplex_fast(a_candidate, target_sum)
            M_new = _max_convolution_fft(a_candidate)
            if M_new < best_M - 1e-12:
                a = a_candidate
                best_M = M_new
                best_a = a.copy()
                improved = True
                break
            t *= 0.5

        if not improved:
            # no progress – stop the LP loop
            break

    return best_a, best_M

            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                n0 = prev.size
                target0 = math.sqrt(2.0 * n0)
                prev = _project_to_simplex_fast(prev, target0)
                best_seq = prev.copy()
                best_M = _max_convolution_fft(best_seq)
                print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                      f"(n={n0}) M={best_M:.6f}")
        except Exception as e:
            print("Seeding error:", e)

    # ------------------------------------------------------------------
    # 1) size schedule – start small, go larger
    # ------------------------------------------------------------------
    schedule = [256, 512, 1024, 2048]   # keep total work modest

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # initialise – upsample previous best if sizes differ
        if best_seq is not None and best_seq.size != n:
            a_start = _upsample_sequence(best_seq, n)
        else:
            a_start = _random_initial_sequence(n, rng)

        a_start = _project_to_simplex_fast(a_start, target)

        # allocate a slice of the remaining budget for this size
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        per_n_deadline = time.time() + remaining * 0.25   # 25 % of what is left

        # ------------------------------------------------------------------
        # 1a) several restarts of the LP‑based improvement
        # ------------------------------------------------------------------
        n_restarts = max(1, int((per_n_deadline - time.time()) // 7.0))
        for r in range(n_restarts):
            if time.time() >= per_n_deadline:
                break
            start_vec = a_start.copy() if r == 0 else _random_initial_sequence(n, rng)

            # LP descent
            lp_deadline = min(per_n_deadline, time.time() + 5.0)   # keep each LP phase short
            a_lp, M_lp = _lp_iterative_improve(start_vec, target, lp_deadline)

            # short Adam sub‑gradient polishing (if time permits)
            if time.time() < per_n_deadline - 1.0:
                adam_deadline = min(per_n_deadline, time.time() + 2.0)
                a_lp = _adam_optimize_submax(a_lp, target, adam_deadline)
                M_lp = _max_convolution_fft(a_lp)

            if M_lp < best_M - 1e-12:
                best_seq = a_lp.copy()
                best_M = M_lp
                print(f"[{time.time():.1f}] Improvement n={n} restart={r} M={best_M:.6f}")

        # ------------------------------------------------------------------
        # 1b) deterministic hill‑climb (exact max) if time remains
        # ------------------------------------------------------------------
        if best_seq is not None and time.time() < deadline - 2.0:
            cur = _project_to_simplex_fast(best_seq, target)
            cur_conv = _full_convolution(cur)
            hc_deadline = min(deadline - 0.5, time.time() + (per_n_deadline - time.time()) * 0.2)
            a_hc, M_hc = _hill_climb_deterministic(cur, cur_conv, hc_deadline)
            if M_hc < best_M - 1e-12:
                best_seq = a_hc.copy()
                best_M = M_hc
                print(f"[{time.time():.1f}] Hill‑climb improved n={n} M={best_M:.6f}")

        # ------------------------------------------------------------------
        # 1c) tiny random tweaks (simulated‑annealing style)
        # ------------------------------------------------------------------
        while time.time() + 1.0 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            M_cand = _max_convolution_fft(cand)
            if M_cand < best_M - 1e-12:
                best_seq = cand
                best_M = M_cand
                print(f"[{time.time():.1f}] Random tweak improved n={n} M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 2) Final polishing on whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        # Adam sub‑gradient (the most effective local optimiser we have)
        best_seq = _adam_optimize_submax(best_seq, target, deadline - 1.0)
        best_M = _max_convolution_fft(best_seq)

        # final deterministic hill‑climb
        if time.time() < deadline - 1.0:
            a_final, M_final = _hill_climb_deterministic(best_seq,
                                                         _full_convolution(best_seq),
                                                         deadline - 0.5)
            if M_final < best_M - 1e-12:
                best_seq = a_final
                best_M = M_final
                print(f"[{time.time():.1f}] Final hill‑climb improvement M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 3) Clean‑up – drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 256
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4) Report the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size}, max_conv={best_M:.6f})")
    except Exception as exc:  # pragma: no cover
        print("Error calling evaluate_sequence:", exc)

    return best_seq.tolist()
