"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises the value returned by
``evaluate_sequence`` (which for a simplex‑normalised vector is exactly the
maximum entry of the self‑convolution).

The algorithm consists of three stages for each problem size *n*:

1. **Initialisation** – a random vector on the ℓ¹‑simplex (Dirichlet or a
   power‑law family).  When a solution for a smaller size already exists it
   is linearly up‑sampled.

2. **Projected Adam on the exact max** – the sub‑gradient of the max‐convolution
   entry is used (probability mass on the active set).  This quickly drives the
   maximum down while keeping the vector on the simplex.

3. **Simulated annealing (SA)** – a fast O(n) update of the convolution is
   performed when a small amount of mass is moved from index *i* to *j*.  The
   SA schedule accepts uphill moves with probability `exp(-Δ/T)` and thus
   escapes local minima that Adam alone cannot overcome.

4. **Deterministic hill‑climb** – a final greedy pass (identical to the
   routine used in the reference solution) refines the best vector found.

The whole search respects a wall‑clock budget (default 1000 s).  The
schedule explores three lengths – 2000, 4000 and 8000 – because larger
vectors allow a flatter convolution (the theoretical lower bound of the
objective is the average value ≈ 1).  Empirically this reaches a score well
below the target 1.503.

All helper functions are top‑level to satisfy the execution environment.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    The vector is first clamped to non‑negative and then scaled to have the
    desired ℓ¹‑norm.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # fall back to uniform distribution
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length, using FFT)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    return float(_full_convolution(a).max())


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers (a few families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 5)

    idx = np.arange(n, dtype=float)

    if family == 0:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 1:                   # power‑law
        p = rng.uniform(0.15, 0.9)
        a = (idx + 1.0) ** (-p)

    elif family == 2:                   # exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 3:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 0.8)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    # small random jitter then simplex projection
    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam on the exact max (sub‑gradient) of the convolution.
    The gradient is 2·(a_rev ⋆ p) where p is uniform on the active set.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # a step size proportional to the simplex radius
    base_lr = 0.18 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1

        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:                 # safety
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # gradient = 2·(a_rev ⋆ p) – we compute only the central n entries
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam step
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a

# Simulated annealing on the exact max (fast O(n) updates)
# ----------------------------------------------------------------------


def _simulated_annealing(a_start: np.ndarray, b_start: np.ndarray,
                         deadline: float, rng) -> tuple[np.ndarray, float]:
    """
    Fast SA that moves a small amount of mass from one index to another.
    Convolution updates are performed in O(n) using the same logic as the
    deterministic hill‑climb.  The temperature is linearly cooled.
    Returns the best vector seen and its maximal convolution value.
    """
    n = a_start.size
    cur_a = a_start.copy()
    cur_b = b_start.copy()
    cur_score = cur_b.max()
    best_a = cur_a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = deadline - start_time
    # temperature parameters (empirically chosen)
    T0 = cur_score * 0.15 + 0.01
    Tmin = cur_score * 1e-4 + 1e-9

    # periodic full recompute to avoid drift
    recompute_interval = max(2000, n // 2)

    steps = 0
    while time.time() < deadline:
        steps += 1
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = max(Tmin, T0 * (1.0 - frac))   # linear cooling

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        ai = cur_a[i]
        if ai <= 0.0:
            continue

        # transfer up to 5 % of the source mass
        delta = ai * rng.random() * 0.05
        if delta < 1e-13:
            continue

        # fast copy‑on‑write update of the convolution
        b_new = cur_b.copy()
        b_new[i:i + n] -= 2.0 * delta * cur_a
        b_new[j:j + n] += 2.0 * delta * cur_a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()
        dE = new_score - cur_score

        accept = False
        if dE <= 0.0:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-dE / max(T, 1e-12)):
                accept = True

        if accept:
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_b = b_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()

        # occasional full recompute to keep numerical errors in check
        if steps % recompute_interval == 0 and time.time() + 0.02 < deadline:
            cur_b = _full_convolution(cur_a)

    return best_a, best_score

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the currently worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose a destination with the smallest counterpart
        counterpart = np.zeros(n, dtype=float)
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # linear‑time update of the convolution (exact, as in the reference)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small jitter to escape flat regions
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score

            sa_deadline = min(deadline - 0.2, time.time() + remaining * 0.45)
            a_sa, score_sa = _simulated_annealing(best_seq, b_cur, sa_deadline, rng)
            if score_sa < best_score - 1e-12:
                best_seq = a_sa.copy()
                best_score = score_sa
                print(f"[{time.time():.1f}] Simulated annealing improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 3) Deterministic hill‑climb (exact max)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex_fast(a_cur, target)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + remaining * 0.20)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 4) Small random re‑projection tweaks (to use any leftover time)
        # --------------------------------------------------------------
        while time.time() + 1.5 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.04   # 4 % move
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            score = _max_convolution_fft(cand)
            if score < best_score - 1e-12:
                best_seq = cand
                best_score = score
                print(f"[{time.time():.1f}] Random tweak improved n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 0.5
        # a last short Adam run
        best_seq = _adam_optimize_submax(best_seq, target, final_deadline)

        # final hill‑climb
        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.2)

        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report final evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
