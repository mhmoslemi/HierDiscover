        if score < best_score:

def solve_lp_for_g(a, M):
    n = a.shape[0]
    # Build convolution matrix
    rows = []
    data = []
    # We'll use sparse matrix for efficiency
    from scipy import sparse
    row_inds = []
    col_inds = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i
            row_inds.append(k)
            col_inds.append(j)
            vals.append(a[i])
    A = sparse.csr_matrix((vals, (row_inds, col_inds)), shape=(2 * n - 1, n))
    # In cvxpy, we can pass A @ g <= M as constraint
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    return g.value

"""
Simple mixed search for the step‑function benchmark.

The core idea is to start from a reasonably good analytic family
(power‑law * exponential tail) and then improve the shape with a
linear‑programming refinement step that exploits the convex constraint

    (a * g)[k] ≤ M   for all k ,

where M is the current maximal convolution value of the candidate a.
The LP maximises the total mass of g under these linear constraints,
after which g is scaled back to the original ℓ₁‑norm and mixed with a.
A few deterministic mass‑transfer passes finish the polishing.

All helpers are top‑level (no closures) and the routine respects the
time budget (default 1000 s).  The returned list can be fed directly to
`evaluate_sequence`.
"""

import time
import math
import numpy as np
import cvxpy as cp
import scipy.sparse as sp


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast enough for our n)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark score 2·n·max(a*a)/(sum(a)²).  No scaling needed."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric family (power‑law × exponential) – used for a quick start
# ----------------------------------------------------------------------


def _exp_power_law_sequence(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-q·i/n) (no extra scaling – scale is irrelevant)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    # make sure we have at least one non‑zero entry
    if a.sum() == 0.0:
        a[0] = 1.0
    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (single iteration)
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray, M: float, target_sum: float, mix: float = 0.5):
    """
    Given current sequence ``a`` and its maximal convolution bound ``M``,
    solve the LP:

        maximise   Σ g_i
        subject to g_i ≥ 0
                   (a * g)[k] ≤ M   ∀k

    The solution ``g`` is then scaled to have ℓ₁‑norm = ``target_sum`` and
    mixed with ``a`` (convex combination with weight ``mix``).
    Returns the new sequence and the new maximal convolution.
    """
    n = a.size
    # build the convolution matrix A (shape (2n‑1, n))
    row = []
    col = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # i runs over lo…hi, and the partner index is j = k‑i
        for i in range(lo, hi + 1):
            row.append(k)
            col.append(k - i)          # index of g
            data.append(a[i])
    A = sp.csr_matrix((data, (row, col)), shape=(2 * n - 1, n))

    # LP variables
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # use ECOS (fast for dense problems); warm_start for speed
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)

    if g.value is None:
        # Infeasible – return the original a unchanged
        return a, M

    g_val = np.asarray(g.value, dtype=np.float64)
    # scale g to the required ℓ₁‑norm (the score is scale invariant)
    sum_g = g_val.sum()
    if sum_g <= 0:
        return a, M
    g_scaled = g_val * (target_sum / sum_g)

    # convex combination
    a_new = (1.0 - mix) * a + mix * g_scaled
    # recompute the maximal convolution
    M_new = _conv_fft(a_new).max()
    return a_new, M_new


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (aggressive exact reduction of the peak)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the current maximal convolution entry.
    Repeatedly moves as much mass as possible from the index that
    contributes most to the peak to the index that contributes least.
    Stops when time runs out.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 1.0:
        conv = _conv_fft(a)
        max_val = conv.max()
        # all positions attaining the maximum (take the first)
        k = np.argmax(conv)

        # indices that appear in conv[k]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions a_i * a_{k-i}
        contrib = a[idx] * a[k - idx]
        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        # second‑largest convolution value (excluding all maxima)
        conv2 = conv.copy()
        conv2[conv == max_val] = -np.inf
        second_max = conv2.max()

        # how much mass can we move to bring the peak down to second_max ?
        if a[donor] <= 0:
            break
        denom = 2.0 * (a[donor] - a[receiver])
        if denom <= 0:
            break
        delta_needed = (max_val - second_max) / denom
        delta = min(a[donor], delta_needed * 0.999)

        if delta <= 0:
            break

        a[donor] -= delta
        a[receiver] += delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT improved score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Searches for a non‑negative step‑function with a minimal benchmark score.
    The routine respects the supplied time budget and returns the best sequence
    found (trailing near‑zero entries are trimmed).
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    # problem size – a few thousand entries give a good trade‑off
    n = 1200                     # small enough for LP, large enough for quality
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 0 – quick parametric pre‑search (same families as the reference)
    # ------------------------------------------------------------------
    best_a = np.full(n, target_sum / n)    # start with uniform
    best_sc = _score(best_a)

    # (a) exponential‑power‑law grid (very coarse – cheap)
    p_vals = np.arange(0.2, 1.3, 0.3)   # 0.2,0.5,0.8,1.1
    q_vals = np.arange(0.0, 9.0, 3.0)   # 0,3,6
    for p in p_vals:
        for q in q_vals:
            if time.time() > deadline - 30.0:
                break
            cand = _exp_power_law_sequence(n, float(p), float(q))
            sc = _score(cand)
            if sc < best_sc:
                best_sc, best_a = sc, cand.copy()
                print(f"[{time.time():.2f}] grid p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP‑based refinement (few iterations, limited time)
    # ------------------------------------------------------------------
    # Use the current best as the starting point
    a = best_a.copy()
    M = _conv_fft(a).max()
    mix = 0.5
    for it in range(12):
        if time.time() > deadline - 15.0:
            break
        a, M = _lp_refine(a, M, target_sum, mix=mix)
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[{time.time():.2f}] LP iter {it} improved score={sc:.6f}")
        mix = max(0.1, mix * 0.85)   # gradually shorten the step

    # ------------------------------------------------------------------
    # Phase 2 – aggressive deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        a_ref, sc_ref = _deterministic_mass_transfer(best_a, deadline)
        if sc_ref < best_sc:
            best_sc, best_a = sc_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT final polish improved score={best_sc:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing zeros and return as plain list
    # ------------------------------------------------------------------
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]

    final_seq = best_a.tolist()
    # optional final evaluation (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
