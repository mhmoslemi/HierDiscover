            mask = (partner >=0) & (partner < n)

                j = rng.integers(n)

            a[receiver] += delta

            j = rng.integers(n)

                continue

            revert = (i, j)
            move_type = 'swap'

"""
Propose a low‑score non‑negative step function.

The algorithm works in three stages:

1. **Exploration** – many short projected‑gradient runs (soft‑max
   approximation of the max‑convolution) from a diverse pool of
   power‑law / exponential / random seeds.  The best intermediate
   sequence is kept.

2. **Refinement** – a simulated‑annealing loop that moves a small
   amount of mass from indices that contribute heavily to the current
   maximal convolution entry to indices that contribute little.
   Swaps of values are also tried.  The temperature is annealed from
   ~0.02 down to 1e‑6, allowing occasional uphill moves while focusing
   on improving the objective.

3. **Final polishing** – a short high‑β gradient‑descent pass.

All vectors are kept on the ℓ¹‑simplex
   {x_i ≥ 0, Σx_i = √(2 n)}.
The convolution and the objective are computed with FFT for speed.
The routine respects the 1000 s time budget; if time runs out the
current best sequence is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa fixed = √(2n) this reduces to the peak)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Create an exp(-lam*i) sequence, renormalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift a small amount of mass from a “donor’’
    (large contribution to the current maximal convolution entry) to a
    “receiver’’ (small contribution).  Moves are accepted with a
    Metropolis rule whose temperature decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass transfer + swaps)
# ----------------------------------------------------------------------


def _simulated_annealing_refine(a0: np.ndarray,
                                deadline: float,
                                start_temp: float = 0.02,
                                end_temp: float = 1e-6,
                                top_frac: float = 0.02,
                                low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style refinement that either transfers a small
    amount of mass from a heavily contributing index to a weak one,
    or swaps two values.  The temperature drops geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_conv = _conv_fft(a)
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # decide move type
        if rng.random() < 0.7:
            # ----- mass transfer move -----
            kmax = int(np.argmax(cur_conv))
            partner = kmax - np.arange(n)
            mask = (partner >= 0) & (partner < n)

            if not np.any(mask):
                # fallback to random swap if something unexpected happens
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                a[i], a[j] = a[j], a[i]
                move = ("swap", i, j)
            else:
                w = np.zeros(n)
                w[mask] = a[mask] * a[partner[mask]]

                donor = np.argmax(w)
                low_cnt = max(1, int(mask.sum() * low_frac))
                low_inds = np.where(mask)[0][np.argsort(w[mask])[:low_cnt]]
                receiver = rng.choice(low_inds)

                if donor == receiver or a[donor] <= 0:
                    it += 1
                    continue

                max_delta = a[donor] * 0.5
                delta = rng.random() * max_delta

                a[donor] -= delta
                a[receiver] += delta
                move = ("transfer", donor, receiver, delta)
        else:
            # ----- swap move -----
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            move = ("swap", i, j)

        new_conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            cur_conv = new_conv
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            typ = move[0]
            if typ == "transfer":
                _, donor, receiver, delta = move
                a[donor] += delta
                a[receiver] -= delta
            else:   # swap
                _, i, j = move
                a[i], a[j] = a[j], a[i]

        # occasional re‑projection to correct floating‑point drift
        if it % 2000 == 0 and it > 0:
            a = _project_to_simplex(a, a.sum())
            cur_conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimizing the benchmark score.
    The routine consists of:
      * exploration with many short soft‑max gradient runs,
      * a brief deterministic mass‑transfer pass,
      * a long simulated‑annealing refinement (mass moves + swaps),
      * a final high‑β gradient polishing, and
      * trimming of trailing zeros.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # small safety margin

    # --------------------------------------------------------------
    # Choose final length based on available time
    # --------------------------------------------------------------
    if budget_s >= 900:
        n_final = 12000
    elif budget_s >= 600:
        n_final = 10000
    else:
        n_final = 8000
    target_sum = np.sqrt(2.0 * n_final)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Build initial pool of candidate seeds
    # --------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.40, 0.70, 7):
        init_pool.append(_power_law_sequence(n_final, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 7):
        init_pool.append(_exponential_sequence(n_final, float(lam), target_sum))

    # random Dirichlet‑like points
    for _ in range(5):
        a = rng.random(n_final)
        a = a / a.sum() * target_sum
        init_pool.append(a)

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        prev = _resize_sequence(prev, n_final)
        prev = _project_to_simplex(prev, target_sum)
        init_pool.append(prev)

    # --------------------------------------------------------------
    # Exploration: many short gradient‑descent runs
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    early_deadline = start_time + (deadline - start_time) * 0.35

    while time.time() < early_deadline:
        seed = rng.choice(init_pool)
        # jitter to increase diversity
        jitter = rng.normal(scale=target_sum * 0.005, size=n_final)
        seed = np.maximum(seed + jitter, 0.0)
        seed = _project_to_simplex(seed, target_sum)

        run_dead = min(time.time() + 12.0, early_deadline)
        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Ensure we have a starting sequence
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_final, 0.55, target_sum)
        best_score = _score(best_seq)

    # --------------------------------------------------------------
    # Brief deterministic mass‑transfer refinement before SA
    # --------------------------------------------------------------
    if time.time() < early_deadline - 5.0:
        mt_dead = time.time() + min(5.0, early_deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.5, end_temp=1e-4, low_frac=0.04
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Pre‑SA MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated‑annealing refinement (mass transfers + swaps)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sa_dead = min(deadline - 2.0, time.time() + (deadline - time.time()) * 0.8)
        a_sa, val_sa = _simulated_annealing_refine(
            best_seq,
            sa_dead,
            start_temp=0.02,
            end_temp=1e-6,
            top_frac=0.02,
            low_frac=0.02,
        )
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] SA refined best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final high‑β gradient polishing (short)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        final_dead = deadline - 0.5
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            final_dead,
            target_sum,
            max_beta=2e6,
            step_factor=0.4,
            gamma=0.995,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing zeroes for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Store globally for possible reuse in subsequent runs
    globals()["height_sequence_1"] = final_seq

    # Optional final evaluation (uses the provided evaluate_sequence)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
