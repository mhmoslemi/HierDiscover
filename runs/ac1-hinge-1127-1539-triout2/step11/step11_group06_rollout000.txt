"""
Hybrid optimizer for the non‑negative step‑function benchmark.

The new version expands on the previous pipeline in several ways:
  * Larger problem size when time permits (up to n≈20000).
  * Multi‑start: a power‑law, a geometric decay, and the previous best
    (resized) are all examined and the best is taken as warm start.
  * More aggressive schedules for the exponential‑descent (β grows faster,
    larger learning‑rate) and the sub‑gradient phase.
  * Longer local‑refinement + simulated‑annealing phases with higher
    temperature and occasional “big” mass moves to escape deep local minima.
  * A final clean‑up step that truncates trailing zeros.

The routine respects the given time budget (default 1000 s) and returns
the best sequence found as a plain list of floats.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of the max‑convolution entry."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # avoid overflow/underflow
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    grad = np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                        np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]

    # take the central n entries (alignment of the two length‑n arrays)
    grad = 2.0 * grad[n - 1:n - 1 + n]
    return grad, b


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 2.0           # start a bit larger than before
    max_beta = 1e6
    base_eta = 0.04
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        # Adaptive step: larger for large beta to keep updates effective
        eta = base_eta / (beta**0.4 + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 150 == 0:
            # small jitter to avoid stagnation
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # accelerate β growth
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 2.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – subgradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent directly on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        # gradient of max entry wrt a_i = 2*a_{k_max-i}
        g[idx] = 2.0 * a[k_max - idx]

        # linear decay of learning rate from 2.5 → 0.2
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.5 * (1.0 - elapsed / total) + 0.2 * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 120 == 0:
            jitter = rng.normal(scale=target_sum * 4e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _gd_with_momentum(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max surrogate, with momentum."""
    a = _project_to_simplex(a0, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 2.0
    max_beta = 1e6
    step_base = target_sum * 0.48
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.91
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta**0.4 + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 150 == 0:
            jitter = rng.normal(scale=target_sum * 0.0012 / (beta**0.2 + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDmom it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 3.0, max_beta)

        it += 1

    return best_a, best_val


def _gd_restarts(init: np.ndarray,
                 target_sum: float,
                 deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts and keep the best result."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(25.0, remaining * 0.34)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.025, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _gd_with_momentum(start,
                                   target_sum,
                                   time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – exact O(n) local mass‑transfer refinement (hill‑climb)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climbing with occasional simulated‑annealing moves."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # start temperature based on the excess over the lower bound 1.0
    T0 = max(0.05, (best_val - 1.0) * 0.35)

    while time.time() < deadline:
        # locate current maximal convolution entry
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions for this shift
        prod = a[idx] * a[k_max - idx]
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # pick a receiver with little contribution to the max shift
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # Allow a somewhat larger mass move when we have "room"
        max_delta = min(a[donor] * 0.07, target_sum * 0.0015)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution (see original code)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic cheap re‑normalisation to fight accumulated drift
        if time.time() - start - elapsed > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)

        # occasional "big" jitter to shake out of plateaus
        if (int(elapsed * 100) % 2800) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.008, target_sum * 0.0006)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)   # recompute (still cheap enough rarely)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Big jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase E – simple simulated annealing (cheap O(n) updates)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA with O(n) convolution updates."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    T0 = max(0.05, (best_val - 1.0) * 0.45)

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # pick donor weighted by mass (more likely to move mass from large entries)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue
        # pick distinct receiver uniformly
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.09, target_sum * 0.0018)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) convolution update
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # occasional global jitter
        if (int(elapsed * 100) % 4600) == 0:
            jitter = rng.normal(scale=target_sum * 6e-4, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA global jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.4   # safety margin

    # -------------------------------- problem size ---------------------------------
    if budget_s >= 850:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 10000

    target_sum = math.sqrt(2.0 * n)

    # ---------------------- warm‑starts (several candidates) ----------------------
    candidates = []

    # 1) reuse previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        candidates.append(_resize_sequence(prev, n))

    # 2) simple decreasing power‑law
    cand2 = (np.arange(1, n + 1, dtype=np.float64) ** (-0.53))
    candidates.append(cand2)

    # 3) geometric decay
    r = 0.9995
    cand3 = r ** np.arange(n, dtype=np.float64)
    candidates.append(cand3)

    # Normalise all candidates to the required L1‑norm
    for i, c in enumerate(candidates):
        candidates[i] = c * (target_sum / c.sum())

    # pick the best of the three cheap evaluations as start
    best_seq = candidates[0]
    best_score = _score(best_seq)
    for c in candidates[1:]:
        sc = _score(c)
        if sc < best_score:
            best_seq = c
            best_score = sc
    print(f"[{time.time():.2f}] Warm‑start best score={best_score:.6f}")

    # --------------------------- Phase A – exponential descent ---------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.10 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_score:
            best_seq, best_score = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------- Phase B – subgradient descent ---------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.15 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------- Phase C – GD restarts --------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.20 * (deadline - time.time())
        a_gd, val_gd = _gd_restarts(best_seq, target_sum, gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------- Phase D – local refinement ---------------------------
    if time.time() < deadline:
        loc_deadline = time.time() + 0.30 * (deadline - time.time())
        a_loc, val_loc = _local_refine(best_seq, target_sum, loc_deadline)
        if val_loc < best_score:
            best_seq, best_score = a_loc, val_loc
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------- Phase E – simulated annealing -----------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
        print(f"[{time.time():.2f}] After simulated annealing best score={best_score:.6f}")

    # --------------------------- final clean‑up ------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (the external verifier)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for future warm‑starts
    globals()["height_sequence_1"] = final_seq

    return final_seq
