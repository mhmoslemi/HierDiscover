"""
Propose a step‑function (non‑negative sequence) that minimises the
maximum entry of its self‑convolution.

Approach
--------
1. **Problem size** – choose n≈12000 for the full 1000 s budget (smaller
   when less time is available) and fix the ℓ¹‑norm
   `target_sum = sqrt(2·n)`.  When the sum equals this value the
   evaluation function returns exactly `max(a * a)`, so we may work with
   the normalized problem.

2. **Rich initial pool** – generate many parametric families (power‑law,
   exponential, sin‑power, Gaussian, triangular) and jittered copies.
   Project each candidate onto the simplex {x≥0, Σx = target_sum} and keep
   the best few.

3. **Projected Adam on a soft‑max surrogate** – minimise a smooth
   approximation of `max(conv)`:
      `softmax_β(conv) = (1/β)·log Σ exp(β·conv)`.
   Its gradient is computed via FFT (see `_gradient_softmax`).  Adam
   updates are performed, after every step the vector is projected back
   onto the simplex.  The smoothing parameter `β` is slowly increased
   (annealed) up to 5·10⁵, allowing the optimiser to first see a smooth
   landscape and later focus on the true max.

   A small L2‑regularisation term (`λ·2·a`) is added to the gradient to
   discourage overly peaked sequences.

4. **Mass‑transfer annealing** – a fast, combinatorial local search.
   At each iteration the index pair that contributes most to the current
   convolution peak is identified; a small amount of mass is moved from
   the larger contributor to a low‑contributing index.  Moves are
   accepted with a Metropolis criterion using a temperature that decays
   exponentially.

5. **Final polishing** – a short second Adam run with a higher `β`
   (≈1 · 10⁶) to fine‑tune the solution.

All steps respect the 1000 s deadline, keep the sequence non‑negative,
and store the best sequence in the global variable `height_sequence_1`
for warm‑starts in subsequent runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto the probability simplex scaled by `s`."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (unnormalised)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    return np.exp(-0.5 * (idx / sigma) ** 2)


def _triangular_sequence(n: int) -> np.ndarray:
    idx = np.abs(np.arange(n, dtype=np.float64) - (n - 1) / 2.0)
    half = n / 2.0
    tri = np.maximum(0.0, 1.0 - idx / half)
    return tri


# ----------------------------------------------------------------------
# Simple low‑pass smoothing (three‑point kernel)
# ----------------------------------------------------------------------
def _smooth(a: np.ndarray, factor: float = 0.02) -> np.ndarray:
    """Blend a small amount of a three‑point moving‑average into `a`."""
    if factor <= 0.0:
        return a
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    sm = np.convolve(a, kernel, mode="same")
    return (1.0 - factor) * a + factor * sm


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(conv)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where grad = ∂ softmax_β(conv) / ∂ a.
    The formula uses the identity:
        grad = 2 * (reverse(a) ★ p)  (★ = convolution),
    where p is the soft‑max probability vector over the convolution entries.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection and β‑annealing
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray,
              deadline: float,
              target_sum: float,
              max_beta: float = 5e5,
              beta_start: float = 0.5,
              beta_factor: float = 2.0,
              step_factor: float = 0.2,
              l2_reg: float = 1e-3,
              smooth_every: int = 500,
              smooth_factor: float = 0.02):
    """
    Adam on the soft‑max surrogate of max‑convolution.
    Returns the best sequence found and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    beta = beta_start
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        # gradient of soft‑max + L2 regularisation
        grad, _ = _gradient_softmax(a, beta)
        grad += l2_reg * 2.0 * a

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size proportional to target_sum
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional smoothing to reduce spikiness
        if smooth_every and (t % smooth_every == 0):
            a_new = _smooth(a_new, factor=smooth_factor)
            a_new = _project_to_simplex(a_new, target_sum)

        # jitter to escape plateaus (very occasional)
        if t % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a_new += jitter
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[{time.time():.2f}] Adam it={t:6d} β={beta:8.1f} best={best_val:.6f}")

        # anneal β (log‑linear)
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic / simulated‑annealing mass transfer
# ----------------------------------------------------------------------
def _mass_transfer_anneal(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-5,
                         low_frac: float = 0.05):
    """
    Greedy mass‑transfer moves aimed at decreasing the current peak.
    Uses a simple temperature schedule; accepts uphill moves with
    Metropolis probability.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)
    cur_score = _max_convolution_fft(a)
    best_a, best_score = a.copy(), cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # exponential cooling
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # current peak index
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, kmax-i) to the peak
        contrib = a[idx] * a[kmax - idx]   # product, exact contribution
        if contrib.sum() == 0.0:
            # nothing to move – break early
            break

        # donor: index with largest contribution
        donor = idx[np.argmax(contrib)]

        # choose a receiver among low‑contributing indices (outside the pair set)
        # we prefer indices that are not part of the current peak pair
        outside = np.setdiff1d(np.arange(n), idx, assume_unique=True)
        if outside.size == 0:
            # fallback: any low‑contributing index inside the pair set
            low_candidates = idx[np.argsort(contrib)[:max(1, int(len(idx) * low_frac))]]
            receiver = rng.choice(low_candidates)
        else:
            # pick a receiver from the outside set uniformly
            receiver = rng.choice(outside)

        if donor == receiver:
            continue

        # maximum transferable mass (keep a[donor] non‑negative)
        max_delta = a[donor] * 0.5
        # also limit delta so that the contribution drop is not too abrupt
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # perform move
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # recompute convolution (costly but still fine for moderate iterations)
        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a.copy()
                print(f"[{time.time():.2f}] MT it={int(elapsed*1000)} best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)
            # conv already stale – recompute for next iteration
            conv = _conv_fft(a)

    return best_a, best_score
