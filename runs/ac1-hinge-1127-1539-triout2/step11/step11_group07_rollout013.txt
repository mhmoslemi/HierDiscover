"""
Propose a step‑function that minimizes `evaluate_sequence`.

Algorithm
---------
1.  For each length n in a short schedule (2000, 4000, 8000) generate
    a variety of *structured* candidates:
        * two‑sided exponential  a[i] = exp(-λ·min(i,n‑1‑i))
        * one‑sided exponential  a[i] = exp(-λ·i)
        * mixed power‑law × exponential
    λ (and the power‑law exponent) are swept on a small grid.
    Each candidate is projected onto the ℓ¹‑simplex Σa = √(2 n).

2.  Keep the best vector seen so far (lowest max‑convolution).

3.  Refine the current best by running a projected Adam optimiser that
    uses the exact sub‑gradient of the max‑convolution (the same
    routine used in the reference code).  Several short Adam runs are
    performed while time permits.

4.  A final polishing Adam run uses all remaining budget.

5.  Trailing near‑zero entries are trimmed, the sequence is evaluated
    with `evaluate_sequence`, and the list of floats is returned.

All helper functions are top‑level, there are no closures or file I/O.
The routine respects the wall‑clock budget (default 1000 s) and prints
intermediate scores.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}:
    simply clamp to non‑negative and renormalise the ℓ¹‑norm.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:                     # fall back to uniform distribution
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the (linear) convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Structured families
# ----------------------------------------------------------------------


def _shape_ends_exp(n: int, lam: float) -> np.ndarray:
    """Two‑sided exponential: a[i] = exp(-lam·min(i,n-1-i))."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)
    return np.exp(-lam * dist)


def _shape_one_exp(n: int, lam: float) -> np.ndarray:
    """One‑sided exponential: a[i] = exp(-lam·i)."""
    return np.exp(-lam * np.arange(n))


def _shape_pow_exp(n: int, p: float, lam: float) -> np.ndarray:
    """Mixed power‑law × exponential."""
    idx = np.arange(n)
    return (idx + 1.0) ** (-p) * np.exp(-lam * idx)


# ----------------------------------------------------------------------
# Random initialiser (reuse from reference code)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)

    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of
    max(a*a).  The active set (indices attaining the maximum) is used as
    the probability vector p.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.2 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1

        # full convolution and active mask
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:       # safety fallback
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # gradient = 2·(a_rev ⋆ p)  (central n entries)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

            print(f"[{time.time():.1f}] Final Adam polishing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 7) Official evaluation report
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:                     # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
