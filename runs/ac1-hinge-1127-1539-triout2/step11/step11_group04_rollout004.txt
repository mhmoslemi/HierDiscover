"""
Hybrid step‑function optimizer (≈2 × speed, refined local search).

The algorithm proceeds in 5 phases:

1. **Rapid exploration** – many short annealed‑gradient runs on a modest
   size (≈5 k).  A large diverse pool of seeds (power‑law, exponential,
   random simplex) is used.  The best low‑dimensional sequence is kept.

2. **Upscaling** – the best low‑dimensional solution is interpolated to a
   larger length (≈12–20 k) and re‑projected onto the ℓ¹‑simplex.
   Several deterministic reorderings (alternating, even‑spread) are
   examined and the one with the smallest self‑convolution peak is
   selected.

3. **Permutation search** – the (sorted) values are shuffled many times
   (including a few random permutations) while the convolution peak is
   evaluated via FFT.  The best ordering is retained.

4. **Projected sub‑gradient descent** – exact sub‑gradient steps on the
   true max‑convolution are performed with a decaying step‑size and
   momentum.  This refines the sequence on the large scale.

5. **Greedy mass‑transfer (simulated‑annealing)** – mass is moved from
   indices that contribute heavily to the current peak toward indices
   that contribute little.  Moves are accepted with a Metropolis rule
   whose temperature decays exponentially.

All phases respect the non‑negativity and ℓ¹‑simplex constraints.
The final sequence is trimmed of trailing near‑zero entries and
returned as a plain list of floats.
"""

import time
import math

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=√(2n) this equals max conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length (≈√n).
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # ----- gradient of soft‑max -----
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # ----- momentum & projection -----
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        # beta schedule
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices reaching (almost) the maximum
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.5,
                                      end_temp: float = 1e-4,
                                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves from a “donor’’ (large contribution to the peak)
    to a “receiver’’ (small contribution) with a Metropolis acceptance rule.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    conv = _conv_fft(a)          # current convolution
    n = a.shape[0]

    while time.time() < deadline:
        # exponential temperature decay
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate the current peak
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak (product a_i * a_{kmax-i})
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        # propose moving a small amount of mass
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # apply tentative move
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        # update convolution efficiently (full recompute – cheap enough)
        conv = _conv_fft(a)
        new_score = _score(a)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it=--- new best={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to shake out of plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random permutation search (fast heuristic)
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Try a number of random reorderings of `base_seq` (keeping the multiset
    of values fixed) and return the ordering with smallest max‑convolution.
    """
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    n = base_seq.shape[0]

    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows the five‑phase schedule described in the module
    docstring and never exceeds `budget_s` seconds.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – rapid exploration on a modest size
    # ------------------------------------------------------------------
    n_expl = 5000 if budget_s >= 800 else (4000 if budget_s >= 500 else 3000)
    target_expl = math.sqrt(2.0 * n_expl)

    # Build a diverse pool of seeds
    init_pool = []

    # power‑law family
    for p in np.linspace(0.1, 2.0, 12):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0005, 0.03, 10):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random simplex points
    for _ in range(12):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of all pool members (increase diversity)
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.03, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # Run short annealed‑gradient runs until a few seconds are left
    while time.time() < deadline - 15.0:
        # pick a seed (biased towards current best)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 12.0, deadline - 8.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_expl,
            max_beta=8e5, step_factor=0.9, gamma=0.98)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic refinement on the small scale
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        mt_deadline = time.time() + (deadline - time.time()) * 0.12
        a_ref, val_ref = _mass_transfer_simulated_annealing(
            best_seq, mt_deadline, target_expl,
            start_temp=0.4, end_temp=1e-4, low_frac=0.07)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] MT refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – upscale to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.5, target_expl)

    # decide final length based on remaining time
    remaining = deadline - time.time()
    if remaining > 120.0:
        n_final = 18000
    elif remaining > 80.0:
        n_final = 15000
    elif remaining > 50.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try deterministic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)

    print(f"[{time.time():.2f}] Upscaled to {n_final} elements, start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation search (random shuffles)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        perm_deadline = time.time() + min(8.0, deadline - time.time() - 5.0)
        trials = 30
        seq_big, _ = _permutation_search(seq_big, trials, perm_deadline)
        print(f"[{time.time():.2f}] After perm search max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – sub‑gradient refinement on the large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        sub_deadline = deadline - 4.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_deadline, target_final,
            step_factor=0.45, gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy mass‑transfer pass (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_deadline = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            seq_big, mt_deadline, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
