"""
Propose a step‑function sequence that minimises the maximal self‑convolution.
The algorithm consists of three phases:

1. **Initialisation** – start from a good known solution (if any) or a random
   family, then up‑sample to the final length (default 8000) and project onto
   the ℓ¹‑simplex Σa = √(2 n).

2. **Gradient optimisation** – a short Adam run on a smooth‐max surrogate
   (log‑sum‑exp) followed by Adam on the exact sub‑gradient of the true max.
   These phases give a decent starting point for a local search.

3. **Deterministic hill‑climb** – the dominant convolution entry is identified
   and a small amount of mass is moved from the contributing index with the
   largest product to an index that does *not* affect this entry (or, if none,
   to the weakest contributor).  The convolution is updated incrementally,
   and the move is accepted only if the maximal value strictly decreases.
   The process repeats until the time budget is exhausted.

All vector updates keep the sum exactly √(2 n); positivity is enforced by the
simplex projection after occasional jitter.  The heavy lifting (FFT, vector
updates) is performed with NumPy in compiled loops, allowing many thousands of
hill‑climb steps within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    The implementation simply rescales after clamping negatives.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # uniform fallback
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2 n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:  # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:  # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:  # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:  # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:  # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:  # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) where g = 2·(a ⋆ p) and
    p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    # convolution of a_rev with p
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.12 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        if t % 200 == 0:
            # tiny jitter to avoid stagnation
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of
    max(a*a). The active set (indices attaining the maximum) is used
    as the probability vector p.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.18 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1

        # full convolution and active mask
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            p = np.ones_like(b) / b.size
        else:
            p = mask / mask.sum()

        # gradient = 2·(a_rev ⋆ p) (central n entries)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a

def _hill_climb(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Aggressive greedy hill‑climb that repeatedly attacks the current worst
    convolution entry.  All updates are performed in‑place on the convolution
    array to avoid O(n log n) recomputation.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.size
    b = _full_convolution(a)                # length 2 n‑1
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()
    eps = 1e-14

    while time.time() < deadline:
        M = int(b.argmax())                 # index of current worst entry
        # source range that contributes to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)

        # contribution of each i to the max entry
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        # choose a destination j that does NOT affect entry M if possible
        if i_low > 0:
            # left side exists
            j_candidates = np.arange(0, i_low, dtype=int)
        elif i_high < n - 1:
            # right side exists
            j_candidates = np.arange(i_high + 1, n, dtype=int)
        else:
            # M spans the whole array; pick the index with the smallest counterpart
            counterpart = a[M - idx]
            mask = idx != i
            if np.any(mask):
                j = idx[mask][np.argmin(counterpart[mask])]
            else:
                # nothing to move – perform a tiny random jitter and recompute b
                jitter = rng.normal(scale=a.mean() * 1e-4, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)
                continue
        # when we have a side region, pick the element with smallest value there
        if i_low > 0 or i_high < n - 1:
            j = j_candidates[np.argmin(a[j_candidates])]

        if a[i] <= eps:
            # source depleted – abort this iteration
            break

        # try a relatively large move; we will backtrack if it hurts
        delta = a[i] * 0.55                     # empirical fraction
        if delta <= eps:
            break

        old_max = b.max()

        # ----- apply tentative move (in‑place) -----
        # linear part
        b[i:i + n] -= 2.0 * delta * a
        b[j:j + n] += 2.0 * delta * a

        # quadratic corrections
        if 2 * i < b.size:
            b[2 * i] += delta * delta
        if 2 * j < b.size:
            b[2 * j] += delta * delta
        if i + j < b.size:
            b[i + j] -= 2.0 * delta * delta

        new_max = b.max()

        if new_max < old_max - 1e-12:
            # accept move
            a[i] -= delta
            a[j] += delta
            # maintain feasibility (sum unchanged, non‑negativity)
            if a[i] < 0.0:
                a[i] = 0.0
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            continue

        # ----- revert the tentative move -----
        b[i:i + n] += 2.0 * delta * a
        b[j:j + n] -= 2.0 * delta * a
        if 2 * i < b.size:
            b[2 * i] -= delta * delta
        if 2 * j < b.size:
            b[2 * j] -= delta * delta
        if i + j < b.size:
            b[i + j] += 2.0 * delta * delta

        # ----- try a smaller step (binary search) -----
        improved = False
        delta2 = delta * 0.5
        while delta2 > eps:
            # apply
            b[i:i + n] -= 2.0 * delta2 * a
            b[j:j + n] += 2.0 * delta2 * a
            if 2 * i < b.size:
                b[2 * i] += delta2 * delta2
            if 2 * j < b.size:
                b[2 * j] += delta2 * delta2
            if i + j < b.size:
                b[i + j] -= 2.0 * delta2 * delta2

            new_max = b.max()
            if new_max < old_max - 1e-12:
                a[i] -= delta2
                a[j] += delta2
                if a[i] < 0.0:
                    a[i] = 0.0
                old_max = new_max
                improved = True
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                break   # keep this delta2
            # revert and shrink
            b[i:i + n] += 2.0 * delta2 * a
            b[j:j + n] -= 2.0 * delta2 * a
            if 2 * i < b.size:
                b[2 * i] -= delta2 * delta2
            if 2 * j < b.size:
                b[2 * j] -= delta2 * delta2
            if i + j < b.size:
                b[i + j] += 2.0 * delta2 * delta2
            delta2 *= 0.5

        if not improved:
            # random perturbation to shake the system
            jitter = rng.normal(scale=a.mean() * 2e-4, size=n)
            a += jitter
            a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
            b = _full_convolution(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine combines:
        • up‑sampling of a known good solution (if available),
        • short Adam phases on smooth‑max and exact max,
        • a long deterministic hill‑climb that moves mass away from the worst
          convolution entry,
    all respecting a wall‑clock budget (default 1000 s).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Target length – larger than the previous 2000 to exploit finer granularity.
    # ------------------------------------------------------------------
    target_n = 8000
    target_sum = math.sqrt(2.0 * target_n)

    # ------------------------------------------------------------------
    # Initialise – start from a known construction if present.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            a0 = _upsample_sequence(prev, target_n)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1, up‑sampled to n={target_n}")
        except Exception as e:
            print(f"[{time.time():.1f}] Seeding error ({e}), falling back to random.")
            a0 = _random_initial_sequence(target_n, rng)
    else:
        a0 = _random_initial_sequence(target_n, rng)
        print(f"[{time.time():.1f}] Random initial sequence, n={target_n}")

    a0 = _project_to_simplex_fast(a0, target_sum)

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam (quick exploration)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + min(30.0, remaining * 0.20)
    a1 = _adam_optimize(a0, target_sum, phase1_deadline)
    print(f"[{time.time():.1f}] Phase 1 finished (smooth‑max Adam)")

    # ------------------------------------------------------------------
    # Phase 2 – exact‑max Adam (sub‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase2_deadline = time.time() + min(30.0, remaining * 0.20)
    a2 = _adam_optimize_submax(a1, target_sum, phase2_deadline)
    print(f"[{time.time():.1f}] Phase 2 finished (sub‑max Adam)")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic hill‑climb (the heavyweight part)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_final, score = _hill_climb(a2, deadline - 1.0)
        print(f"[{time.time():.1f}] Hill‑climb finished, best max={score:.6f}")
    else:
        a_final = a2
        score = _max_convolution_fft(a_final)
        print(f"[{time.time():.1f}] Not enough time for hill‑climb, using Adam result.")

    # ------------------------------------------------------------------
    # Optional final polishing (few seconds of sub‑max Adam)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        polish_deadline = time.time() + 5.0
        a_final = _adam_optimize_submax(a_final, target_sum, polish_deadline)
        score = _max_convolution_fft(a_final)
        print(f"[{time.time():.1f}] Polishing Adam done, max={score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    a_final = np.asarray(a_final, dtype=float)
    nz = np.where(a_final > 1e-12)[0]
    if nz.size:
        a_final = a_final[: nz[-1] + 1]
    else:
        a_final = a_final[:1]

    # ------------------------------------------------------------------
    # Report final evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(a_final.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={a_final.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return a_final.tolist()
