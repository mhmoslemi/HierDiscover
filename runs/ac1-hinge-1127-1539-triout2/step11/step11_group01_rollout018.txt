"""
Propose a step‑function (non‑negative sequence) with a low max‑convolution value.
The algorithm consists of:

1. Choose a problem size n (larger n gives a finer approximation of the
   continuous optimum).  For a full 1000 s budget we use n≈30 000.

2. Fast parametric pre‑search:
   * pure exponential (a_i ∝ r^i),
   * power‑law + exponential tail,
   * the same families with a small integer offset of the index.
   All candidates are projected to the simplex Σa = √(2n) and the best
   (lowest max‑convolution) is kept.

3. Projected Adam optimisation that works on the *exact* sub‑gradient of the
   max‑convolution (no soft‑max smoothing).  The ADAM moments are updated
   and the iterate is projected onto the simplex after each step.

4. Deterministic aggressive mass‑transfer:
   repeatedly pick the index k where the convolution is maximal, find a
   donor index i with the largest contribution a_{k‑i} and a receiver j with the
   smallest contribution, and move the analytically required amount of mass
   from i to j (up to the whole donor mass).  This directly drives the peak
   down to the second‑largest value.

5. A short random mass‑transfer phase is used to escape possible plateaus.

The whole routine respects the given time budget (default 1000 s) and returns
the best sequence found (trailing near‑zero entries are removed).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑2 FFT (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value (with Σa = √(2n) the score equals max‑convolution)."""
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * a.size * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _exp_sequence(n: int, q: float, target_sum: float) -> np.ndarray:
    """Pure exponential a_i = r^i,  r = exp(-q/n)."""
    r = np.exp(-q / n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(
    n: int, p: float, q: float, target_sum: float, offset: int = 0
) -> np.ndarray:
    """a_i = (i+offset+1)^(-p) * exp(-q*i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_max(a: np.ndarray):
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a (exact, no smoothing)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    k = int(np.argmax(conv))                     # take first maximal index
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = k - i                                    # partner indices, also in [0,n)
    g = np.zeros_like(a)

    # i < j : off‑diagonal contributions
    mask_lt = i < j
    i_lt = i[mask_lt]
    j_lt = j[mask_lt]
    g[i_lt] += a[j_lt]
    g[j_lt] += a[i_lt]

    # i == j : diagonal contributions (derivative of a_i^2)
    mask_eq = i == j
    idx_eq = i[mask_eq]
    g[idx_eq] += 2.0 * a[idx_eq]

    return g, conv


# ----------------------------------------------------------------------
# Adam optimiser using the exact sub‑gradient
# ----------------------------------------------------------------------


def _adam_subgrad_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    step_factor: float = 0.6,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the exact max‑convolution sub‑gradient."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        g, conv = _gradient_max(a)

        # Adam moment updates
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)

        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam subgrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


def _adam_restart_search(
    start_vec: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Run a few Adam restarts and keep the best result."""
    best_a = start_vec.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        # jittered start for a new run
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        s = best_a + jitter
        s = _project_to_simplex(np.maximum(s, 0.0), target_sum)

        a_opt, val_opt = _adam_subgrad_descent(
            s,
            time.time() + min(30.0, (deadline - time.time()) * 0.35),
            target_sum,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray, deadline: float, target_sum: float):
    """Greedy reduction of the maximal convolution entry (no 0.7‑mass cap)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < 1e-12:
            break

        k = idx_max[0]                     # pick one maximal position
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}

        # donor: index with positive mass and largest partner contribution
        donor_mask = a[idx] > 0
        if not donor_mask.any():
            break
        donor_local = np.argmax(partner * donor_mask)
        donor = idx[donor_local]

        # receiver: index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            if idx[r_loc] != donor:
                receiver = idx[r_loc]
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        # move as much as possible (up to the whole donor mass)
        delta = min(a[donor], delta_needed)
        if delta <= 0.0:
            break

        a[donor] -= delta
        a[receiver] += delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray, deadline: float, target_sum: float):
    """Simple stochastic exchange of a small amount of mass."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        donor = np.argsort(-a)[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        receiver = np.argsort(a)[rng.integers(k_low)]

        if donor == receiver:
            continue

        delta = rng.random() * a[donor] * 0.1
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_score:
            best_score = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score
