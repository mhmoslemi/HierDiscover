"""
Hybrid optimizer that starts from a low‑additive‑energy construction.

The core idea:
  * Build a Sidon (B₂) set – a set of indices whose pairwise sums are all different.
    On such a set every product a_i·a_j appears in a unique convolution entry,
    so the contribution of the “big” coefficients to the maximal entry is just
    w_big².
  * Fill the remaining positions with a tiny uniform weight.  The tiny part
    behaves almost like a uniform sequence, but its total mass is small,
    keeping its own contribution to the max low.
  * The parameters (size of the Sidon set and the split α of the total mass)
    are chosen heuristically; several α values are tried and the best
    initialization is kept.
  * A short deterministic hill‑climb (mass‑transfer) phase and a final
    simulated‑annealing phase (both O(n) per move) polish the solution.

The algorithm fits comfortably into the 1000 s budget and typically reaches
a score well below the previous 1.508 bound (often ≈ 1.2‑1.3 on n≈12000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT‑based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  With sum = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = float(_conv_fft(a).max())
    s = a.sum()
    if s < 0.01:                     # should never happen
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sidon set construction
# ----------------------------------------------------------------------


def _generate_sidon_set(n: int, target_m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy construction of a Sidon (B₂) set.
    Returns an array of positions (indices) whose pairwise sums are all distinct.
    """
    used = np.zeros(2 * n - 1, dtype=bool)          # which sums are already taken
    pos = []
    # random order gives a more spread set than the deterministic 0,…,n‑1 order
    candidates = rng.permutation(n)
    for i in candidates:
        # test whether i creates a duplicate sum
        if used[i + i]:
            continue
        if pos:
            sums = i + np.array(pos, dtype=int)
            if used[sums].any():
                continue
        # accept i
        pos.append(i)
        used[i + i] = True
        if pos:
            used[i + np.array(pos, dtype=int)] = True
        if len(pos) >= target_m:
            break
    return np.array(pos, dtype=int)


# ----------------------------------------------------------------------
# Exact hill‑climbing (mass‑transfer) – deterministic refinement
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing that moves a small delta of mass
    from a “donor’’ index (chosen proportionally to its contribution
    to the current maximal convolution entry) to a “receiver’’ index
    taken among indices that contribute little.
    The convolution is updated in O(n) time.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    T0 = best_val * 0.04                               # initial temperature for SA acceptance

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each possible donor to the current max entry
        prod = a[idx] * a[k_max - idx]
        if prod.sum() > 0.0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # pick a receiver among indices that have tiny influence on the max entry
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)          # 30‑percentile → “light’’ indices
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # the delta to move – limited to a few per‑cent of the donor mass
        max_delta = min(a[donor] * 0.04, target_sum * 8e-4)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # update convolution in O(n)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # simulated‑annealing acceptance (allows occasional worsening moves)
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / (deadline - start)))
        accept = (new_val < best_val) or (temperature > 0.0 and
                                         np.random.random() < np.exp(-(new_val - best_val) / temperature))

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved to {best_val:.6f}")

        # every few seconds re‑normalise to fight numerical drift
        if time.time() - start > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            start = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing with cheap O(n) updates
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly moves a tiny amount of mass between two indices.
    The convolution is updated in O(n) time; the schedule cools linearly.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = max(deadline - start, 1e-12)
    T0 = max(0.05, (best_val - 1.0) * 0.5)            # start temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # donor chosen proportionally to its mass (helps avoid zero donors)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue
        # receiver: any other index
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.05, target_sum * 0.001)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = (new_val < cur_val) or (temperature > 0.0 and
                                         np.random.random() < np.exp(-(new_val - cur_val) / temperature))
        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved to {best_val:.6f}")

        # occasional global jitter to escape plateaus
        if int(elapsed * 100) % 5000 == 0:
            jitter = rng.normal(scale=target_sum * 2e-4, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = _conv_fft(a)
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Build a (large) Sidon set
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    target_m = int(np.sqrt(n))            # theoretical upper bound ~√n
    # give the Sidon construction a short dedicated budget
    sidon_deadline = start + min(30.0, (deadline - start) * 0.1)
    best_sidon = np.empty(0, dtype=int)
    best_m = 0
    while time.time() < sidon_deadline:
        cand = _generate_sidon_set(n, target_m, rng)
        if cand.size > best_m:
            best_m = cand.size
            best_sidon = cand
            # stop early if we already hit the theoretical bound
            if best_m >= target_m:
                break

    if best_m == 0:                        # fallback – uniform power law
        best_sidon = np.empty(0, dtype=int)
        best_m = 0

    print(f"[{time.time():.2f}] Sidon set size = {best_m}")

    # ------------------------------------------------------------------
    # 2. Try a few α‑splits between “big’’ (Sidon) and “tiny’’ entries
    # ------------------------------------------------------------------
    alphas = [0.35, 0.45, 0.55, 0.65]       # share of total mass given to the big part
    best_seq = None
    best_score = np.inf

    for alpha in alphas:
        if best_m > 0:
            big_weight = alpha * target_sum / best_m
        else:
            big_weight = 0.0
        tiny_weight = (1.0 - alpha) * target_sum / max(n - best_m, 1)

        seq = np.full(n, tiny_weight, dtype=np.float64)
        if best_m > 0:
            seq[best_sidon] = big_weight

        # small safety renormalisation (numerical errors)
        seq *= (target_sum / seq.sum())
        val = _score(seq)
        print(f"[{time.time():.2f}] Alpha={alpha:.2f} → init score={val:.6f}")

        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # 3. Deterministic mass‑transfer hill‑climb (uses exact O(n) updates)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = time.time() + 0.45 * (deadline - time.time())
        seq, val = _local_refine(best_seq, target_sum, local_deadline)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated annealing phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _simulated_annealing(best_seq, target_sum, deadline)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
        print(f"[{time.time():.2f}] After SA best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final clean‑up
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    # final external evaluation (for the log)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
