"""
A multi‑stage optimiser for the non‑negative step‑function problem.

Main ideas
----------
* Build a diverse initial pool (power‑law, exponential, Gaussian,
  random Dirichlet, linear, previous best).
* Short projected‑gradient runs on the moderate size (≈8 k) to get a
  solid seed.
* Upscale the seed (linear interpolation) to a larger length
  (≈15 k–20 k) – more degrees of freedom usually lower the peak.
* Heavy polishing:
    - high‑β soft‑max gradient descent,
    - deterministic “mass‑transfer’’ moves,
    - targeted permutation annealing.
* Final trimming of trailing zeros.

All steps respect the simplex ℓ¹‑constraint
Σ a_i = sqrt(2·n) and non‑negativity, so the evaluated score equals the
maximum entry of the auto‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa=√(2n) this reduces to max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Create an exp(-lam*i) sequence, renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma_frac: float, target_sum: float) -> np.ndarray:
    """
    Gaussian‑shaped sequence centered in the middle.
    sigma_frac is the standard deviation expressed as a fraction of n.
    """
    sigma = max(sigma_frac * n, 1e-9)
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Reorder by placing the largest value at the leftmost position,
    the second largest at the rightmost, the third largest next to the left,
    etc.
    """
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Same as _alternating_order but start with the largest on the right."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p   (⋆ = convolution)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flats
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase temperature gradually
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift a small amount of mass from a “donor’’
    (large contribution to the current maximal convolution entry) to a
    “receiver’’ (small contribution).  Moves are accepted with a
    Metropolis rule whose temperature decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / max(1e-12, (deadline - start_time))
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution contributed by each index to the peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style search that swaps indices which
    contribute heavily to the current maximal convolution entry.
    Only re‑orders the values – the ℓ¹‑sum stays unchanged.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / max(1e-12, (deadline - start_time))
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows the classic three‑phase plan (exploration → up‑scale
    → polishing) but adds Gaussian seeds, a longer final polish and a
    second permutation pass.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose exploration size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a rich pool of initial candidates
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.35, 0.75, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0015, 0.015, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # Gaussian family – try a handful of widths
    for sigma_frac in np.linspace(0.04, 0.18, 8):
        init_pool.append(_gaussian_sequence(n_expl, float(sigma_frac), target_expl))

    # random Dirichlet‑like points
    for _ in range(6):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # linear decreasing shape (just for variety)
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies to enlarge diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Exploration: short gradient‑descent runs on the moderate size
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # limit the number of seeds to keep total time reasonable
    seeds_to_try = min(len(init_pool), 30)
    pool_indices = rng.choice(len(init_pool), size=seeds_to_try, replace=False)

    while time.time() < deadline - 12.0:   # keep room for later phases
        # pick a seed (occasionally jitter around current best)
        if best_seq is not None and rng.random() < 0.5:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            idx = rng.choice(pool_indices)
            seed = init_pool[idx].copy()

        # run a short GD pass (≈10‑15 s)
        run_dead = min(time.time() + 12.0, deadline - 6.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_expl,
            max_beta=4e5,
            step_factor=0.6,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Deterministic mass‑transfer refinement at the moderate scale
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.12

            best_seq, mt_dead, target_expl,

        )

            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    elif remaining > 30.0:
        n_final = 15000

        n_final = 12000

        n_final = n_expl

    seq_big = _project_to_simplex(seq_big, target_final)

        sc = _score(cand)

            best_score = sc

    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------

            end_temp=8e-5,
            top_frac=0.03,
            low_frac=0.03,
        )
        if val_perm < best_score:
            best_score = val_perm

            print(f"[{time.time():.2f}] After first permutation search best={best_score:.6f}")

    if time.time() < deadline - 2.0:

        )
        if val_mt2 < best_score:
            best_score = val_mt2

            print(f"[{time.time():.2f}] After second MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:

        a_perm2, val_perm2 = _permutation_targeted_search(
            best_seq,
            perm2_dead,
            start_temp=0.09,
            end_temp=5e-5,
            top_frac=0.02,
            low_frac=0.02,
        )
        if val_perm2 < best_score:
            best_score = val_perm2
            best_seq = a_perm2.copy()
            print(f"[{time.time():.2f}] After second permutation best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 11. Final log (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
