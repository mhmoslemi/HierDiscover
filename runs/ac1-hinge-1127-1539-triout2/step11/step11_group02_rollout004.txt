"""
Improved step‑function optimizer.

Main ideas
----------

* **Richer parametric pre‑search** – in addition to the pure and
  exponential power‑law families we scan a *symmetric product* family
  `a_i ∝ ((i+1)(n-i))^{-p}` which concentrates mass at the two ends
  of the vector and is known to lower the central convolution peak.

* **Long Adam GD** – the same soft‑max based Adam optimiser as before,
  but with a larger final β (≈10⁶) and a finer learning‑rate schedule.

* **Exact sub‑gradient polishing** – after the Adam phase we run a
  projected sub‑gradient method on the *true* max‑convolution (no
  soft‑max approximation).  The gradient of the active convolution entry
  is cheap to compute (just a shifted copy of the current vector).

* **Aggressive deterministic mass‑transfer** – unchanged but run for a
  larger portion of the budget.

* **Time‑budget aware** – each phase receives a fraction of the total
  wall‑clock budget; the routine stops well before the hard deadline
  (0.5 s safety margin).

The routine respects the simplex constraint
    Σ a_i = √(2 n) ,   a_i ≥ 0
and returns the best sequence found within the allotted time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (full linear convolution)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring (the benchmark score equals max(a*a) because Σa = √(2 n))
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(n: int, p1: float, p2: float,
                               alpha: float, target_sum: float) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _sym_product_power_law_sequence(n: int, p: float,
                                    target_sum: float) -> np.ndarray:
    """Symmetric shape a_i ∝ ((i+1)*(n-i))^{-p} (edges heavy, centre light)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in Adam GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()                     # soft‑max weights

    # convolution of a[::-1] with p gives ∂/∂a of Σ p_k (a*a)[k]
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # each entry appears twice except boundaries
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float,
                          target_sum: float,
                          max_beta: float = 5e5,
                          step_factor: float = 0.8,
                          gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # tiny jitter every few hundred steps
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 /
                                (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} "
                  f"score={best_val:.6f}")

        # anneal β
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution (no soft‑max)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray):
    """Return (grad, max_val) for the active convolution entry."""
    n = a.size
    conv = _conv_fft(a)
    k = int(np.argmax(conv))

    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)

    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]          # derivative w.r.t. each active a_i
    return grad, float(conv[k])


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         target_sum: float,
                         step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        grad, max_val = _subgradient_max(a)

        # diminishing step size (≈ step0 / √t)
        step = step0 * target_sum / (math.sqrt(it + 1) + 1e-12)

        a_new = a - step * grad
        a_new = _project_to_simplex(a_new, target_sum)

        cur_val = _score(a_new)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} step={step:.6f} "
                  f"score={best_val:.6f}")

        a = a_new
        it += 1

        # early exit if step becomes negligible
        if step * np.linalg.norm(grad) < 1e-12:
            break

    return best_a, best_val

def _deterministic_mass_transfer(a0: np.ndarray, deadline: float,

                                 tol: float = 1e-12) -> tuple[np.ndarray, float]:

        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions a_{k-i}
        contrib = a[k - idx]

        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        # second‑largest convolution value (ignoring all maximal indices)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        denom = 2.0 * (a[donor] - a[receiver])
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={int(time.time())} "
                  f"score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (light exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray, deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 1.0:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the given wall‑clock budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ---------------------------------------------------------------
    # Choose problem size
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = math.sqrt(2.0 * n)                # Σa = √(2 n)

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (including new family)
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) pure power‑law
    for p in np.arange(0.20, 1.21, 0.01):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in np.arange(0.20, 1.21, 0.02):
        for q in np.arange(0.0, 8.001, 0.25):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum)
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) new symmetric product‑power‑law family
    for p in np.arange(0.10, 2.01, 0.05):
        seq = _sym_product_power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (e) occasional Dirichlet random samples (few, cheap)
    for _ in range(30):
        alpha = np.full(n, 0.5)                     # mildly sparse Dirichlet
        rand_vec = rng.dirichlet(alpha)
        seq = target_sum * rand_vec
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (f) include previous global best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent (long)
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + 0.35 * remaining
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec, gd_deadline, target_sum,
            max_beta=1e6, step_factor=0.7, gamma=0.995)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] GD restart improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – aggressive deterministic mass‑transfer
    # ---------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        # occasional random nudges to escape tiny plateaus
        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

        if time.time() > deadline - 6.0:
            break

    # ---------------------------------------------------------------
    # Phase C – exact sub‑gradient polishing (finetuning)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        sub_deadline = deadline - 1.0
        a_fin, val_fin = _subgradient_descent(
            best_seq, sub_deadline, target_sum, step0=0.4)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Sub‑gradient polish improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries, convert to list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
