"""
Propose a step‑function that minimises the maximal entry of its self‑convolution.

The search proceeds in the following stages:

1. **Problem size** – we keep the length used in the previous run
   (12 000 when enough time is available).  The ℓ¹‑norm is fixed to
   √(2 n) so that the evaluation function reduces to the maximal
   convolution value.

2. **Rich candidate pool** – parametric families (power‑law, exponential,
   sinusoidal, linear, quadratic, logarithmic) plus random Dirichlet
   draws.  Each shape is projected onto the simplex and a jittered copy
   is added.

3. **Quick screening** – the best few candidates (default 8) are kept.

4. **Refinement of each elite candidate**
   * **Adam optimiser on a soft‑max approximation** of the max‑convolution.
     The temperature `β` is increased geometrically (0.5 → 10⁶) while the
     Adam learning‑rate is scaled with the target ℓ¹‑norm.
   * **Exact sub‑gradient descent** on the true max‑convolution (as in the
     reference implementation).
   * **Simulated‑annealing mass‑transfer** – move a small amount of mass
     from a “donor’’ index (large contribution to the current peak) to a
     “receiver’’ index, accepting worsening moves with a temperature schedule.

5. **Symmetrisation** – the best sequence is averaged with its mirror image,
   projected onto the simplex and polished once more with Adam.

6. **Post‑processing** – trailing near‑zero entries are trimmed, the final
   sequence is stored globally (`height_sequence_1`) for warm‑starts and
   returned as a plain Python list.

All heavy work uses FFT (O(n log n)) and the algorithm respects the
global time budget (default 1000 s).  Prints provide progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (unnormalised)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _linear_sequence(n: int) -> np.ndarray:
    """Decreasing linear shape (from 1 to 0)."""
    return 1.0 - np.arange(n, dtype=np.float64) / (n - 1)


def _quadratic_sequence(n: int) -> np.ndarray:
    """Decreasing quadratic shape (1 - (i/(n-1))^2)."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    return 1.0 - t ** 2


def _log_sequence(n: int, offset: float = 2.0) -> np.ndarray:
    """Slowly decaying logarithmic shape 1 / log(i+offset)."""
    return 1.0 / np.log(np.arange(n, dtype=np.float64) + offset)


def _random_dirichlet_sequence(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex via Dirichlet(alpha)."""
    x = rng.dirichlet([alpha] * n)
    return x


# ----------------------------------------------------------------------
# Soft‑max gradient of max‑convolution
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv). Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (rev(a) ⋆ p) restricted to central n entries
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------
def _adam_softmax(a0: np.ndarray, deadline: float, target_sum: float,
                  max_beta: float = 5e5,
                  step_factor: float = 0.6,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    # main loop – stop when time runs out
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning rate scaled with target_sum (roughly the ℓ¹‑norm)
        lr = step_factor * target_sum / (np.sqrt(v_hat) + eps)

        a -= lr * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.0008 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        # increase β slowly (geometric schedule)
        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Exact sub‑gradient descent (as previously used)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

# Simulated‑annealing mass transfer
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         start_temp: float = 0.04,
                         end_temp: float = 1e-5,
                         max_step_fraction: float = 0.3) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_fft(a)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_val = cur_max

    start_time = time.time()
    total = deadline - start_time

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        # compute contributions w_i = a_i * a_{kmax-i}
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] * a[kmax - idx]

        # donor: prob. proportional to contribution (larger product -> more likely)
        if w.sum() == 0:
            donor = rng.choice(idx)
        else:
            probs = w / w.sum()
            donor = rng.choice(idx, p=probs)

        # receiver: biased toward small values (prefer low a)
        low_frac = 0.05
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(a[idx])[:low_cnt]]
        # occasionally allow any index to increase diversity
        if rng.random() < 0.2:
            receiver = rng.integers(0, n)
        else:
            receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * max_step_fraction
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta

        # propose move
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # projection is not needed because sum is preserved and we keep non‑negativity
        if a_new[donor] < 0:
            it += 1
            continue

        # evaluate
        conv_new = _conv_fft(a_new)
        new_max = float(conv_new.max())
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(new_max - cur_max) / max(temp, 1e-12)):
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_val:.6f}")

        it += 1

    return best_a, best_val
