"""
Hybrid optimizer for the non‑negative step‑function benchmark (improved).

The routine consists of the following stages:

1. **Warm‑start** – reuse a previously found construction if available,
   otherwise generate a set of simple parametric candidates (power‑law,
   inverse‑product, exponential and Gaussian) and keep the best.
2. **Mirror descent on a soft‑max surrogate** – multiplicative (exponential)
   updates on a smooth approximation of the max‑convolution term, with a
   slowly increasing temperature (beta) and occasional jitter.
3. **Sub‑gradient descent on the exact max‑convolution** – classic
   multiplicative sub‑gradient steps on the true maximal entry.
4. **Aggressive O(n) hill‑climbing** – repeatedly move a tiny amount of mass
   from an index that contributes heavily to the current maximal convolution
   entry to an index that contributes little, accepting worsening moves with
   a simple linear cooling schedule.
5. **Final simulated‑annealing pass** – cheap O(n) mass‑transfer moves with
   a slower temperature decay to polish the solution.

The time budget (default 1000 s) is split among the phases; more time is
dedicated to the cheap O(n) refinement where the bulk of the improvement
occurs.  The returned sequence satisfies all constraints of `evaluate_sequence`
(and is stored in the global name `height_sequence_1` for warm‑starts in later
calls).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (FFT based full convolution)
# ----------------------------------------------------------------------


def _fft_convolve_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_fft_convolve_full(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective value for a sequence a.
    For a correctly normalised sequence (sum = √(2n)) the score equals
    max(a * a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:          # guard against degenerate sums
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_len = seq.size
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used by mirror descent)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where conv is the full convolution a*a
    and grad is the gradient of the log‑sum‑exp approximation of max(conv).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_val = conv.max()
    exp_terms = np.exp(beta * (conv - max_val))
    p = exp_terms / exp_terms.sum()          # soft‑max weights

    # grad = 2 * reverse(a) convolve p
    a_rev = a[::-1]
    pf = np.fft.rfft(p, n=size)
    grad_full = np.fft.irfft(np.fft.rfft(a_rev, n=size) * pf, n=size)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1:n - 1 + n]  # keep only the n relevant entries
    return grad, conv


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    a = a0 * (target_sum / a0.sum())
    best = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1e6
    base_eta = 0.03
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * grad)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] ExpDes it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the exact max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    a = a0 * (target_sum / a0.sum())
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        conv = _fft_convolve_full(a)
        k_max = int(np.argmax(conv))

        n = a.size
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear decay of step size
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.2 * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 4e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Phase C – cheap O(n) hill‑climbing on the exact max‑convolution
# ----------------------------------------------------------------------


def _local_hill_climb(a0: np.ndarray,
                      target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Aggressive mass‑transfer hill‑climbing.
    At each step we move a small amount of mass from a donor index
    (chosen proportionally to its contribution to the current maximal entry)
    to a receiver index that contributes little.
    Acceptance follows a simple linear cooling schedule.
    """
    a = a0.copy()
    n = a.size
    b = _fft_convolve_full(a)
    cur_max = b.max()
    best = a.copy()
    best_val = float(2.0 * n * cur_max / (target_sum * target_sum))

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature as a fraction of the excess over the theoretical lower bound (≈1)
    T0 = max(0.02, (best_val - 1.0) * 0.3)

    iter_cnt = 0
    while time.time() < deadline:
        # --- identify the currently dominant convolution index ---
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)

        # contribution of each index to this max entry
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k_max - idx
        contrib = a[idx] * a[partner]          # non‑negative

        if contrib.sum() == 0.0:
            # degenerate case – pick random donor / receiver
            donor = rng.integers(n)
        else:
            donor = rng.choice(idx, p=contrib / contrib.sum())

        # choose a receiver that is *not* in the high‑contribution set
        # (i.e. from the complement of idx) – if empty fall back to random
        low_candidates = np.setdiff1d(np.arange(n), idx, assume_unique=True)
        if low_candidates.size:
            receiver = rng.choice(low_candidates)
        else:
            # all indices belong to the active set – pick a random different one
            receiver = rng.integers(n - 1)
            if receiver >= donor:
                receiver += 1

        # size of the transfer – a small fraction of the donor mass
        max_delta = min(a[donor] * 0.12, target_sum * 0.0015)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # ---- incremental update of the full convolution (O(n)) ----
        # donation
        b[donor:donor + n] -= 2.0 * delta * a
        # reception
        b[receiver:receiver + n] += 2.0 * delta * a
        # quadratic terms
        b[2 * donor] += delta * delta
        b[2 * receiver] += delta * delta
        if donor != receiver:
            b[donor + receiver] -= 2.0 * delta * delta

        # apply the mass move
        a[donor] -= delta
        a[receiver] += delta

        # evaluate new score (max entry only)
        new_max = b.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # temperature schedule (linear)
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        accept = False
        if new_val < cur_max:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_max) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
                print(f"[{time.time():.2f}] HillClimb iter={iter_cnt:7d} score={best_val:.6f}")

        # occasional re‑normalisation to cure numerical drift
        if iter_cnt % 8000 == 0:
            a = a * (target_sum / a.sum())
            b = _fft_convolve_full(a)
            cur_max = b.max()

        iter_cnt += 1

    return best, best_val

# Phase D – final simulated annealing (cheap mass swaps)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Standard SA with O(n) updates, slower cooling than the hill‑climb."""
    a = a0.copy()
    n = a.size
    b = _fft_convolve_full(a)
    cur_max = b.max()
    cur_val = float(2.0 * n * cur_max / (target_sum * target_sum))
    best = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = max(0.02, (cur_val - 1.0) * 0.4)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # donor weighted by current mass (helps avoid zeros)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-15:
            continue
        # receiver uniformly chosen, distinct from donor
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.2, target_sum * 0.002)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) convolution update
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
                print(f"[{time.time():.2f}] SA iter={it:7d} score={best_val:.6f}")

        # occasional jitter to shake out of plateaus
        if it % 9000 == 0:
            jitter = rng.normal(scale=target_sum * 3e-4, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _fft_convolve_full(a_j)
            val_j = float(2.0 * n * b_j.max() / (target_sum * target_sum))
            if val_j < best_val:
                best_val = val_j
                best = a_j.copy()
                a, b, cur_val = a_j, b_j, val_j
                print(f"[{time.time():.2f}] SA jitter improved score={best_val:.6f}")

        it += 1

    return best, best_val

# Parametric warm‑start generator
# ----------------------------------------------------------------------


def _generate_initial_candidate(n: int, target_sum: float) -> np.ndarray:
    """
    Scan a small family of parametric shapes (power‑law, inverse‑product,
    exponential, Gaussian) and return the best candidate according to
    the exact objective.
    """
    best = None
    best_val = np.inf
    rng = np.random.default_rng()

    # cache of candidate arrays to avoid recomputing the same shape
    candidates = []

    # 1) power‑law a_i ∝ (i+1)^(-p)
    for p in np.linspace(0.35, 0.85, 11):
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        candidates.append(a)

    # 2) inverse product a_i ∝ ((i+1)*(n-i))^(-p)  (U‑shaped)
    for p in np.linspace(0.2, 0.8, 7):
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = (idx * (n - idx + 1)) ** (-p)
        candidates.append(a)

    # 3) exponential decay a_i ∝ exp(-α i)
    for alpha in np.linspace(0.0005, 0.005, 9):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        candidates.append(a)

    # 4) Gaussian centred at the middle
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n * 0.05, n * 0.2, 6):
        a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        candidates.append(a)

    # evaluate all candidates (very cheap relative to the whole budget)
    for a in candidates:
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()

    # add a tiny random perturbation to avoid exact symmetry artefacts
    if best is not None:
        jitter = rng.normal(scale=target_sum * 1e-4, size=n)
        best = np.maximum(best + jitter, 0.0)
        best = best * (target_sum / best.sum())

    return best

# Main entry point

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The function respects the supplied time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ---- problem size -------------------------------------------------
    if budget_s >= 800:
        n = 10000          # a good trade‑off between quality and speed
    elif budget_s >= 500:
        n = 8000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # ---- warm‑start ---------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        init = _generate_initial_candidate(n, target_sum)

    # ---- Phase A : exponential descent (soft‑max) --------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.15 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(init, target_sum, phase_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After ExpDes best score={best_score:.6f}")

    # ---- Phase B : sub‑gradient descent --------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.15 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, phase_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After SubGrad best score={best_score:.6f}")

    # ---- Phase C : aggressive O(n) hill‑climbing ----------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.45 * (deadline - time.time())
        a_hc, val_hc = _local_hill_climb(best_seq, target_sum, phase_deadline)
        if val_hc < best_score:
            best_seq, best_score = a_hc, val_hc
        print(f"[{time.time():.2f}] After HillClimb best score={best_score:.6f}")

    # ---- Phase D : final simulated annealing ---------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
        print(f"[{time.time():.2f}] After SimulatedAnnealing best score={best_score:.6f}")

    # ---- final clean‑up ------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # drop trailing zeros for tidiness
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # evaluate with the external verifier (if import is available)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
