def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.1

    # set n_expl based on budget
    n_expl = 12000 if budget_s > 800 else 8000 if budget_s > 500 else 6000
    target_sum = math.sqrt(2.0 * n_expl)
    rng = np.random.default_rng()
    init = []
    # add families:
    # power law p in...
    for p in np.linspace(0.4,0.7,9):
        a = (np.arange(1,n_expl+1)**(-p))
        a = a * target_sum / a.sum()
        init.append(a)
    # exponent lam
    for lam in np.linspace(0.001,0.02,7):
        a = np.exp(-lam * np.arange(n_expl, dtype=float))
        a = a * target_sum / a.sum()
        init.append(a)
    # random Dirichlet
    for _ in range(8):
        a = rng.random(n_expl)
        a = a * target_sum / a.sum()
        init.append(a)
    # alternate order etc.
    for arr in list(init):
        alt = _alternating_order(arr)
        init.append(alt)
        rev = _reversed_alternating(arr)
        init.append(rev)

    # include previous height_sequence_1 if exists:
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], float)
        if len(prev) != n_expl:
            prev = _resize_sequence(prev, n_expl)
        prev = _project_to_simplex(prev, target_sum)
        init.append(prev)

    best_seq = None
    best_val = np.inf

    # run many independent runs until deadline - some buffer
    while time.time() < deadline - 10:
        seed = rng.choice(init)
        # jitter seed
        jitter = rng.normal(scale=target_sum*0.002, size=n_expl)
        a = seed + jitter
        a = np.maximum(a, 0)
        a = _project_to_simplex(a, target_sum)
        # apply subgradient descent for a short period (like 3 sec)
        a_opt, val_opt = _subgradient_descent(a, time.time() + min(4.0, deadline - time.time() - 5), target_sum)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New best subgrad: {best_val:.6f}")

        # optionally perform mass-transfer refinement
        if time.time() < deadline - 5:
            a_mt, val_mt = _deterministic_mass_transfer(a_opt, time.time() + min(2.0, deadline - time.time() - 5), target_sum,
                                                       start_temp=0.3, end_temp=1e-4, low_frac=0.05)
            if val_mt < best_val:
                best_val = val_mt
                best_seq = a_mt.copy()
                print(f"[{time.time():.2f}] New best after MT: {best_val:.6f}")

    # now upscale to larger length if time remains
    if time.time() < deadline - 5:
        # choose final length larger than n_expl maybe up to 20000
        n_final = int(min(20000, n_expl * 1.5 + 2000))
        target_final = math.sqrt(2.0 * n_final)
        seq_big = _resize_sequence(best_seq, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)
        # do final subgradient descent
        a_big_opt, val_big_opt = _subgradient_descent(seq_big, time.time() + min(12.0, deadline - time.time() - 5), target_final)
        if val_big_opt < best_val:
            best_val = val_big_opt
            best_seq = a_big_opt.copy()
            print(f"[{time.time():.2f}] Final big subgrad best: {best_val:.6f}")

        # final permutation search
        if time.time() < deadline - 2:
            a_perm, val_perm = _permutation_targeted_search(best_seq, time.time() + min(8.0, deadline - time.time() - 2),
                                                            start_temp=0.12, end_temp=1e-5, top_frac=0.03, low_frac=0.03)
            if val_perm < best_val:
                best_val = val_perm
                best_seq = a_perm.copy()
                print(f"[{time.time():.2f}] final perm best: {best_val:.6f}")

    # trim trailing zeros
    arr = best_seq
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1]+1]
    final_seq = arr.tolist()
    # log final
    try:
        print(f"Final score {evaluate_sequence(final_seq):.6f}")
    except Exception:
        pass
    globals()["height_sequence_1"] = final_seq
    return final_seq

def _subgradient_descent(a0, deadline, target_sum, step_factor=0.5, gamma=0.99):
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        # compute conv and max index
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        # compute gradient of max entry (subgradient)
        n = a.shape[0]
        grad = np.zeros_like(a)
        # contributions:
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        # for i in range(lo, hi+1):
        #   j=kmax-i
        #   grad[i] += a[j]
        #   if j != i: grad[i] += a[i] (the other term)?? Actually derivative formula above:
        # For i, derivative includes a_{k-i} if k-i in range and i term adds a_i if k-i != i else 0
        # We'll compute vectorized.
        idx = np.arange(lo, hi+1)
        partner = kmax - idx
        grad[idx] += a[partner]
        mask = partner != idx
        grad[idx[mask]] += a[idx[mask]]
        # The second part of derivative: a_i from term j = k-i.

        # Now gradient climb direction is +grad because increasing a inc conv, we want to reduce conv, so we move opposite.
        # Use momentum similar to GD
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            # step size scaling
            lr = step_factor * target_sum / (norm + 1e-12)  # adapt
            a -= lr * momentum
        # project to simplex
        a = _project_to_simplex(a, target_sum)

        # optionally jitter occasionally
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it} val={best_val:.6f}")

        it += 1
    return best, best_val

"""
Hybrid optimisation for the convolution‑peak minimisation problem.

The routine builds a diverse pool of candidate step‑functions (power‑law,
exponential, random Dirichlet), performs short soft‑max gradient
descent runs, up‑scales the best candidate, then refines it with a
sub‑gradient descent that directly targets the maximal convolution entry.
Afterwards a targeted simulated‑annealing permutation search and a final
deterministic mass‑transfer pass are applied.  All updates respect non‑negativity
and the ℓ¹‑simplex (∑a = √(2 n)).  The best sequence found before the time
budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a)/(∑a)².
    When ∑a = √(2 n) the score equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) sequence renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam*i) sequence renormalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple deterministic reorderings
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest left, 2nd largest right, 3rd next left, …."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Largest right, second left, …."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approx. to max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase temperature periodically
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy donor→receiver moves with a Metropolis schedule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Swap indices that heavily (or lightly) contribute to the current peak."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *exact* maximum convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.95) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on max(conv(a,a)).  The gradient of the
    current maximal entry is computed analytically; a momentum term and
    a Euclidean simplex projection keep the iterates feasible.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        n = a.shape[0]
        grad = np.zeros_like(a)

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx

        # term a_i * a_{k-i}
        grad[idx] += a[partner]                     # derivative from j = i
        mask = partner != idx
        grad[idx[mask]] += a[idx[mask]]             # derivative from j = k-i when distinct

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low convolution‑peak score.
    The algorithm proceeds through:
      1. diverse initial pool,
      2. short soft‑max GD explorations,
      3. up‑scaling,
      4. high‑precision sub‑gradient descent,
      5. permutation annealing,
      6. final deterministic mass‑transfer,
      7. trimming zeros.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # tiny safety margin

    # --------------------------------------------------------------
    # 1. Choose a moderate size for the exploratory phase
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 12000
    elif budget_s >= 600:
        n_expl = 10000
    else:
        n_expl = 8000
    target_expl = math.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 2. Build a pool of seeds
    # --------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.38, 0.68, 7):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.001, 0.015, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like points
    for _ in range(6):
        a = rng.random(n_expl)
        a = a * target_expl / a.sum()
        init_pool.append(a)

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if len(prev) != n_expl:
            prev = _resize_sequence(prev, n_expl)
        prev = _project_to_simplex(prev, target_expl)
        init_pool.append(prev)

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 3. Exploration: many short soft‑max GD runs
    # --------------------------------------------------------------
    while time.time() < deadline - 15.0:
        seed = rng.choice(init_pool)
        jitter = rng.normal(scale=target_expl * 0.003, size=n_expl)
        a = seed + jitter
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_expl)

        # run a brief gradient descent
        run_deadline = min(deadline - 5.0, time.time() + 5.0)
        a_opt, val_opt = _annealed_gradient_descent(
            a,
            run_deadline,
            target_expl,
            max_beta=5e5,
            step_factor=0.6,
            gamma=0.98,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best = {best_score:.6f}")

    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # --------------------------------------------------------------
    # 4. Upscaling to a larger size (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and budget_s >= 700:
        n_final = min(20000, n_expl + 8000)
    elif remaining > 15.0:
        n_final = min(15000, n_expl + 4000)
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Upscaled n={n_final} start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5. High‑precision sub‑gradient descent on the exact max
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            deadline - 4.0,
            target_final,
            step_factor=0.4,
            gamma=0.90,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Targeted permutation search
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        perm_dead = time.time() + min(12.0, deadline - time.time() - 1.0)
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.15,
            end_temp=1e-5,
            top_frac=0.04,
            low_frac=0.04,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Final deterministic mass‑transfer fine‑tune
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 0.5)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.25,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 8. Trim trailing zeros for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # 9. Optional final log of the exact evaluation (may be cheap)
    # --------------------------------------------------------------
    try:
        print(f"[{time.time():.2f}] Final evaluated score = {evaluate_sequence(final_seq):.6f}")
    except Exception:
        pass

    # expose for reuse in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
