"""
Hybrid optimizer with an additional shape‑parameter search.

The routine consists of three major stages:

1. **Shape parameter sweep** – explore simple families
   a_i = (i+1)^(-p)·exp(-λ·i)  (and truncated versions) and keep the
   best candidate.  This supplies a far better warm‑start than the
   fixed power‑law used previously.

2. **Gradient‑based refinement** – a short projected‑gradient / sub‑
   gradient descent phase followed by the deterministic “mass‑transfer”
   hill‑climber from the reference implementation.

3. **Final simulated‑annealing** – cheap O(n) updates that move a tiny
   amount of mass between two indices, accepting occasional worsening
   moves.

All phases respect the global time budget (default 1000 s) and the
best sequence found is stored in the global variable ``height_sequence_1``
for warm‑starts in subsequent calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used in mirror & GD phases)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of the max‑convolution entry."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    grad = np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                        np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = 2.0 * grad[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – subgradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent directly on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with random restarts (momentum version)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – exact local mass‑transfer refinement (deterministic hill‑climb)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase E – simulated annealing with cheap O(n) updates
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Simple SA that moves small mass between two indices, updating the
    convolution in O(n) time. Accepts occasional worsening moves."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    # initial temperature – a fraction of the current excess over 1.0
    T0 = max(0.05, (best_val - 1.0) * 0.5)

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.05, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # occasional global jitter
        if (int(elapsed * 100) % 5000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Additional simple Euclidean projected‑gradient descent
# ----------------------------------------------------------------------


def _proj_gradient_descent(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute sub‑gradient of the max entry
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # step size schedule
        eta = 1.0 / (np.sqrt(it + 1) + 1.0)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ProjGD it={it:5d} score={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # spare a tiny safety margin

    # ------------------------ problem size ---------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------ initialise best sequence ----------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        best_seq = _resize_sequence(np.array(prev, dtype=np.float64), n)
        best_seq = best_seq * (target_sum / best_seq.sum())
    else:
        # simple decreasing power‑law fallback
        indices = np.arange(1, n + 1, dtype=np.float64)
        best_seq = (indices) ** (-0.55)
        best_seq = best_seq * (target_sum / best_seq.sum())
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 1 – cheap shape parameter sweep (power‑law + exponential)
    # ----------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + min(30.0, remaining * 0.20)  # at most 20 % of remaining time
    indices = np.arange(1, n + 1, dtype=np.float64)

    rng = np.random.default_rng()

    # systematic grid search
    p_vals = np.linspace(0.30, 1.20, 21)          # 21 values
    lam_vals = np.linspace(0.0, 0.0015, 21)       # 21 values
    for p in p_vals:
        for lam in lam_vals:
            if time.time() >= phase1_deadline:
                break
            a_raw = np.power(indices, -p) * np.exp(-lam * (indices - 1.0))
            a_raw = a_raw * (target_sum / a_raw.sum())
            val = _score(a_raw)
            if val < best_score:
                best_score = val
                best_seq = a_raw.copy()
                print(f"[{time.time():.2f}] Grid p={p:.4f} λ={lam:.6f} → score={best_score:.6f}")
        if time.time() >= phase1_deadline:
            break

    # random exploration of a third parameter: truncated support
    while time.time() < phase1_deadline:
        p = rng.uniform(0.25, 1.30)
        lam = rng.uniform(0.0, 0.0020)
        support_frac = rng.uniform(0.55, 1.0)
        m = int(support_frac * n)
        a_raw = np.zeros(n, dtype=np.float64)
        a_raw[:m] = np.power(np.arange(1, m + 1, dtype=np.float64), -p) * \
                    np.exp(-lam * (np.arange(m) ))
        a_raw = a_raw * (target_sum / a_raw.sum())
        val = _score(a_raw)
        if val < best_score:
            best_score = val
            best_seq = a_raw.copy()
            print(f"[{time.time():.2f}] Random p={p:.4f} λ={lam:.6f} m={m} → score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 2 – projected gradient descent (short)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + (deadline - time.time()) * 0.25
        a_gd, val_gd = _proj_gradient_descent(best_seq, target_sum, gd_deadline)
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] Proj‑GD improved to {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 3 – original multi‑stage pipeline (exp‑GD, sub‑GD, GD‑restarts)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        # Exponential descent
        exp_deadline = time.time() + (deadline - time.time()) * 0.10
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_score:
            best_score = val_exp
            best_seq = a_exp.copy()
            print(f"[{time.time():.2f}] Exp‑GD improved to {best_score:.6f}")

    if time.time() < deadline:
        # Sub‑gradient descent
        sub_deadline = time.time() + (deadline - time.time()) * 0.15
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑GD improved to {best_score:.6f}")

    if time.time() < deadline:
        # GD restarts (momentum version)
        gd_restart_deadline = time.time() + (deadline - time.time()) * 0.15
        a_gd2, val_gd2 = _gradient_descent_with_restarts(best_seq,
                                                          target_sum,
                                                          gd_restart_deadline)
        if val_gd2 < best_score:
            best_score = val_gd2
            best_seq = a_gd2.copy()
            print(f"[{time.time():.2f}] GD‑restarts improved to {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 4 – deterministic local refinement (mass‑transfer)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = time.time() + (deadline - time.time()) * 0.20
        a_loc, val_loc = _local_refine(best_seq, target_sum, local_deadline)
        if val_loc < best_score:
            best_score = val_loc
            best_seq = a_loc.copy()
            print(f"[{time.time():.2f}] Local refine improved to {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 5 – final simulated annealing (cheap updates)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] Sim‑Anneal improved to {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Trim trailing near‑zero entries and final evaluation
    # ----------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (external verifier)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
