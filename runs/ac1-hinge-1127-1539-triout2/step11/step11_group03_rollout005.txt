def _permutation_sa(a, deadline, start_temp, end_temp):
    n = len(a)
    best = a.copy()
    best_val = _score(a)  # i.e., maxconv
    cur = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        # schedule temperature
        t = (time.time() - start) / (deadline - start)
        temp = start_temp * (end_temp/start_temp) ** t
        i,j = rng.integers(0,n, size=2)
        if i==j: continue
        # swap
        cur[i],cur[j] = cur[j],cur[i]
        new_val = _score(cur)
        # acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val)/max(temp,1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
                print(f"[{time.time():.2f}] PermSA new best {best_val}")
        else:
            # revert
            cur[i],cur[j] = cur[j],cur[i]
    return best, best_val

"""
Hybrid optimiser for the step‑function attack.

The algorithm works in several stages:

1.  Build a diverse pool of seed sequences (power‑law, exponential,
    random Dirichlet, previous best).  All are projected onto the
    ℓ¹‑simplex {x ≥ 0, Σx = √(2n)}.

2.  Fast projected sub‑gradient descent on the exact objective
    max{(a*a)_k}.  The sub‑gradient of the peak entry is
    g_j = 2·a_{k*−j} (zero outside the array).  A simple diminishing
    step‑size together with Euclidean simplex projection yields rapid
    reduction of the peak.

3.  Targeted simulated‑annealing on the permutation of the multiset
    (the ℓ¹‑sum stays unchanged).  Swaps are biased toward moving a
    large contributor to the peak towards a small contributor.

4.  A final polishing phase repeats the sub‑gradient descent,
    followed by a short deterministic mass‑transfer (Metropolis) pass.

5.  The best sequence is trimmed (trailing zeros removed) and returned.
   The global variable ``height_sequence_1`` stores the result for possible
   reuse in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a * a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Create an exp(-lam*i) sequence, renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest value leftmost, second largest rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Scoring (max‑convolution, identical to the benchmark when Σa=√(2n))
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return max entry of a*a (sum of a is assumed to be √(2n))."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_scale: float = 0.5,
                         momentum: float = 0.0) -> tuple[np.ndarray, float]:
    """
    Projected (optionally momentum‑accelerated) sub‑gradient descent
    for min max(a*a).  The step size is
        η = step_scale * target_sum / (‖g‖·√(t+1)).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    vel = np.zeros_like(a)  # momentum buffer

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(conv.argmax())
        # sub‑gradient of the peak entry
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[partner[mask]]

        # optional momentum
        if momentum > 0.0:
            vel = momentum * vel + grad
            g = vel
        else:
            g = grad

        norm = np.linalg.norm(g)
        if norm > 0:
            eta = step_scale * target_sum / (norm * (np.sqrt(it + 1) + 1.0))
            a -= eta * g

        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        # occasional tiny jitter to escape flat regions
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Metropolis moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy redistribution of a small amount of mass with a cooling schedule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(conv.argmax())
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak entry
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style swapping of indices that heavily
    contribute to the current peak.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(conv.argmax())

        # contribution of each position to the peak (a_i * a_{kmax-i})
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        # temperature schedule (exponential decay)
        tau = (time.time() - start) / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The pipeline consists of:

      * diverse seeding,
      * several short sub‑gradient runs,
      * a targeted permutation annealing,
      * a final polishing sub‑gradient run,
      * a tiny deterministic mass‑transfer sweep,
      * trimming of trailing zeros.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size based on available time
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 12000
    elif budget_s >= 600:
        n = 10000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a pool of initial candidates
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.40, 0.65, 7):
        init_pool.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.020, 6):
        init_pool.append(_exponential_sequence(n, float(lam), target_sum))

    # random Dirichlet‑like points
    for _ in range(6):
        a = rng.random(n)
        a = _project_to_simplex(a, target_sum)
        init_pool.append(a)

    # linear decreasing shape (for variety)
    lin = np.linspace(1.0, 0.0, n)
    lin = _project_to_simplex(lin, target_sum)
    init_pool.append(lin)

    # previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        init_pool.append(_project_to_simplex(prev, target_sum))

    # ------------------------------------------------------------------
    # Phase 1 – many short sub‑gradient runs (exploration)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    while time.time() < deadline - 30.0:            # keep reserve for later phases
        # choose a seed
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_sum * 0.004, size=n)
            seed = _project_to_simplex(seed, target_sum)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 12.0, deadline - 5.0)
        if run_dead <= time.time():
            break

        seq_opt, val_opt = _subgradient_descent(
            seed,
            target_sum,
            run_dead,
            step_scale=0.6,
            momentum=0.85,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted permutation annealing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        perm_dead = time.time() + min(18.0, deadline - time.time() - 2.0)
        seq_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.12,
            end_temp=1e-5,
            top_frac=0.03,
            low_frac=0.03,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = seq_perm.copy()
            print(f"[{time.time():.2f}] After permutation best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        final_dead = deadline - 3.0
        seq_final, val_final = _subgradient_descent(
            best_seq,
            target_sum,
            final_dead,
            step_scale=0.3,
            momentum=0.92,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"[{time.time():.2f}] Final polish best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – tiny deterministic mass‑transfer sweep
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        seq_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_sum,
            start_temp=0.25,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = seq_mt.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional re‑ordering (alternating) – cheap check
    # ------------------------------------------------------------------
    if best_seq is not None:
        alt_seq = _alternating_order(best_seq)
        alt_score = _score(alt_seq)
        if alt_score < best_val:
            best_val = alt_score
            best_seq = alt_seq
            print(f"[{time.time():.2f}] Alternating order improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Final evaluation logging (does not affect the returned sequence)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
