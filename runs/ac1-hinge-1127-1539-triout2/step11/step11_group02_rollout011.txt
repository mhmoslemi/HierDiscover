def _two_tier_sequence(n, k_frac, r, target_sum):
    """
    k_frac: fraction of entries in first tier (0<k_frac<1). Could be integer K = int(k_frac*n).
    r: ratio A/B >=1.
    Returns array.
    """
    K = max(1, int(round(k_frac*n)))
    # compute B = target_sum / (K*r + (n-K))
    B = target_sum / (K*r + (n-K))
    A = r * B
    a = np.full(n, B, dtype=np.float64)
    a[:K] = A
    return a

def _adam_subgradient_descent(a0, deadline, target_sum, step_factor=0.8, gamma=0.995):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    while time.time() < deadline:
        # compute convolution full via FFT
        a_fft = np.fft.rfft(a, n=size)
        conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_val = conv_full.max()
        # find first index of max
        k = np.argmax(conv_full)
        # gradient
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        if k % 2 == 0:
            i_eq = k // 2
            if 0 <= i_eq < n:
                grad[i_eq] = a[i_eq]
        # Adam update
        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)
        # maybe jitter every 200 it
        if it % 400 == 0:
            # add small jitter
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGradAD it={it} score={best_val:.6f}")
        it += 1
    return best_a, best_val

"""
Improved step‑function optimizer.

Main ideas
----------

1. **Very dense parametric pre‑search** – we enumerate several simple families
   (power‑law, power‑law+exponential, broken power‑law, two‑tier, linear
   decrease).  All candidates are scaled to the simplex ∑a = √(2 n) and the
   best one is kept.

2. **Projected Adam optimisation** – two complementary variants are used:
   *soft‑max* (as in the reference) and an exact *sub‑gradient* for the true
   max‑convolution.  Both run many short restarts with jittered starts,
   keeping the best result.

3. **Aggressive deterministic mass‑transfer** – after gradient descent we
   repeatedly move mass from the “donor” index that contributes most to the
   current maximal convolution entry to a “receiver’’ index that contributes the
   least, directly shrinking the peak.

4. **Final polishing** – a short Adam run with a very large temperature
   (soft‑max) or the exact sub‑gradient is performed.

All steps respect the non‑negativity and the simplex constraint and obey the
time budget (default 1000 s).  The best sequence found is returned and stored
in the global name ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (full linear convolution)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (the optimisation objective)."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _two_tier_sequence(n: int, k_frac: float, r: float, target_sum: float) -> np.ndarray:
    """
    First K = round(k_frac*n) entries have value A,
    remaining entries have value B, with A/B = r ≥ 1.
    """
    K = max(1, int(round(k_frac * n)))
    B = target_sum / (K * r + (n - K))
    A = r * B
    a = np.full(n, B, dtype=np.float64)
    a[:K] = A
    return a


def _linear_decrease_sequence(n: int, K: int, target_sum: float) -> np.ndarray:
    """
    Decreasing arithmetic progression on the first K positions,
    reaching zero at index K‑1, zero afterwards.
    """
    if K < 2:
        K = 2
    A = 2.0 * target_sum / K
    i = np.arange(K, dtype=np.float64)
    a = A * (1.0 - i / (K - 1))
    a = np.where(a > 0, a, 0.0)
    out = np.zeros(n, dtype=np.float64)
    out[:K] = a
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (already used in the reference code)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Adam optimisation (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Soft‑max Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimisation (exact max‑gradient / sub‑gradient)
# ----------------------------------------------------------------------


def _adam_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    step_factor: float = 0.8,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    n = a0.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        # full convolution to locate the maximal entry
        a_fft = np.fft.rfft(a, n=size)
        conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        k = int(np.argmax(conv_full))

        # sub‑gradient of max_{k} (a*a)[k]
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        if k % 2 == 0:
            ieq = k // 2
            if 0 <= ieq < n:
                grad[ieq] = a[ieq]  # replace the double‑counted term

        # Adam update
        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to avoid stagnation
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Sub‑grad Adam it={it:5d} k={k} score={best_val:.6f}"
            )

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        k = idx_max[0]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[k - idx]

        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        denom = 2.0 * (a[k - donor] - a[k - receiver])
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] DMT it={int(time.time())} score={best_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Light random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 1.0:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Global random move new best={best_score:.6f}"
            )
    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the time budget and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # problem size (larger n gives finer discretisation)
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)  # Σa is fixed to this value

    # ------------------------------------------------------------------
    # Phase 0 – ultra‑dense parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) pure power‑law
    for p in np.arange(0.20, 1.21, 0.02):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in np.arange(0.20, 1.21, 0.05):
        for q in np.arange(0.0, 8.001, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law
    for p1 in np.arange(0.20, 0.81, 0.2):
        for p2 in np.arange(0.80, 2.01, 0.4):
            for alpha in np.arange(0.1, 0.91, 0.2):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) two‑tier family
    for k_frac in np.linspace(0.05, 0.5, 10):
        for r in np.logspace(0.0, 2.0, 12):  # r ∈ [1,100]
            seq = _two_tier_sequence(n, float(k_frac), float(r), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (e) linear decreasing families
    for K in np.linspace(int(0.1 * n), n, 9, dtype=int):
        seq = _linear_decrease_sequence(n, int(K), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (f) reuse previous global best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(
        f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase A – Adam with restarts (mix of soft‑max and exact sub‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + 0.35 * remaining

    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        if rng.random() < 0.5:
            a_opt, val_opt = _adam_gradient_descent(
                start_vec,
                gd_deadline,
                target_sum,
                max_beta=5e5,
                step_factor=0.8,
                gamma=0.995,
            )
        else:
            a_opt, val_opt = _adam_subgradient_descent(
                start_vec,
                gd_deadline,
                target_sum,
                step_factor=0.7,
                gamma=0.998,
            )

        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(
                f"[{time.time():.2f}] Gradient restart improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase B – deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:

        )
