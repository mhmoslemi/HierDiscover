def _greedy_mass_transfer(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    # hyperparams
    reduction_factor = 0.5  # fraction of donor mass to move each step
    min_delta = target_sum * 1e-4
    max_delta_ratio = 0.02  # max fraction of target_sum to move
    it = 0
    while time.time() < deadline:
        # compute peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        contribs = a[idx] * a[k_max - idx]
        i_max = idx[np.argmax(contribs)]
        # choose receiver: the index with smallest product contribution
        # we can also use minimal a
        # compute contributions to that peak as well
        # To reduce effect on future peaks, pick r where product with counterpart is minimal
        # also prefer smallest a
        # compute counterpart contributions: a[r] * a[k_max - r]
        # Let's aim for minimal a[r] * a[k_max - r] and minimal a[r]
        receiver_candidates = np.arange(n)
        # compute candidate metric: a * a[k_max-i] maybe + epsilon
        rc_products = a[receiver_candidates] * a[np.clip(k_max - receiver_candidates,0,n-1)]
        # but need to handle out-of-range index for counterpart
        # for k_max - r out-of-range, we treat product as 0 (since a out of bounds)
        # In convolution, only indices i where partner exists produce contribution.
        # For out-of-range, product not part of current peak; so choose such r
        # Let's build mask where (k_max - r) < 0 or >= n => product zero
        mask = (k_max - receiver_candidates < 0) | (k_max - receiver_candidates >= n)
        # if any mask True, choose one with minimal a
        if np.any(mask):
            r_candidates = np.where(mask)[0]
            r_min = r_candidates[np.argmin(a[r_candidates])]
        else:
            # otherwise choose index with minimal product
            r_min = receiver_candidates[np.argmin(rc_products)]
        if i_max == r_min:
            # pick random other
            r_min = rng.integers(n)
            while r_min == i_max:
                r_min = rng.integers(n)
        donor = i_max
        receiver = r_min

        max_delta = min(a[donor] * reduction_factor, target_sum * max_delta_ratio)
        if max_delta <= 0:
            break
        delta = max_delta  # for now move full allowed
        # update convolution efficiently
        b_new = b.copy()
        # updates as earlier:
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a

        a[donor] -= delta
        a[receiver] += delta

            best_a = a.copy()

def propose_candidate(budget_s: int = 1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # random power law exponent
        rng = np.random.default_rng()
        p = rng.uniform(0.3, 0.9)
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-p))

    init = init * (target_sum / init.sum())

    # Phase A
    if time.time() < deadline:
        phase_end = time.time() + 0.10 * (deadline - time.time())
        a_exp, best_val = _exponential_descent(init, target_sum, phase_end)
        best_seq, best_score = a_exp, best_val
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    if time.time() < deadline:
        phase_end = time.time() + 0.12 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, phase_end)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # Phase C
    if time.time() < deadline:
        phase_end = time.time() + 0.18 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq, target_sum, phase_end)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # Phase D1 deterministic flatten
    if time.time() < deadline:
        phase_end = time.time() + 0.30 * (deadline - time.time())
        a_det, val_det = _deterministic_anneal(best_seq, target_sum, phase_end)
        if val_det < best_score:
            best_seq, best_score = a_det, val_det
        print(f"[{time.time():.2f}] After deterministic anneal best score={best_score:.6f}")

    # Phase D2 final anneal (optional)
    if time.time() < deadline:
        a_ref, val_ref = _mass_transfer_anneal(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref

"""
Improved search for a step‑function with low max‑convolution.

The algorithm follows the proven four‑phase pipeline from the previous
state‑of‑the‑art, but adds a dedicated *deterministic* annealing stage.
Instead of choosing the donor/receiver indices probabilistically, the
new stage always moves mass from the index that contributes most to the
current convolution peak to an index that contributes (almost) nothing.
The move size is still scaled by a temperature schedule, but the
selection is greedy, which typically reduces the peak much faster.
The final phase falls back to the original stochastic mass‑transfer
annealer for fine‑tuning.

All heavy lifting (FFT‑based convolutions, soft‑max surrogate gradients,
projected gradient descent, etc.) is re‑used from the reference code.
The routine respects the supplied time budget (default 1000 s) and
stores the best sequence in the global variable ``height_sequence_1`` for
warm‑starts in future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phase A and Phase C)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with random restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = _project_to_simplex(a0, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1.5e-3 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray, target_sum: float, deadline: float):
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start, target_sum,
                                            time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – deterministic mass‑transfer annealing (new)
# ----------------------------------------------------------------------


def _deterministic_anneal(a0: np.ndarray,
                         target_sum: float,
                         deadline: float):
    """
    Greedy annealing: always move mass from the index with the largest
    contribution to the current convolution peak to an index that contributes
    (almost) nothing to that peak.  The step size is still temperature‑scaled.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.30                # start temperature (~30 % of score)

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – index with largest contribution to that peak
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)]

        # receiver – index that contributes the *least* to the current peak
        partner = k_max - np.arange(n)
        valid = (partner >= 0) & (partner < n)
        prod = np.empty(n)
        prod[valid] = a[valid] * a[partner[valid]]
        prod[~valid] = 0.0
        min_prod = prod.min()
        cand = np.where(prod == min_prod)[0]
        receiver = cand[np.argmin(a[cand])]

        if donor == receiver:
            # fallback: pick the smallest‑height index different from donor
            cand2 = np.where(np.arange(n) != donor)[0]
            receiver = cand2[np.argmin(a[cand2])]

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.07, target_sum * 0.002)
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            # no effective move – just continue
            it += 1
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] DetAnneal improved score={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if time.time() - last_full > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute (det) improved score={best_val:.6f}")

        # occasional tiny jitter to keep the dynamics alive
        if it % 2000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _mass_transfer_anneal(a0: np.ndarray,
                         target_sum: float,
                         deadline: float):
    """Long‑run simulated‑annealing refinement (fallback after deterministic stage)."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.30                   # start at ~30 % of the current score

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – larger contribution ⇒ more likely to be chosen
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with relatively small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by the temperature)
        max_delta = min(a[donor] * 0.04, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if time.time() - last_full > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional tiny jitter to keep the dynamics alive
        if it % 2000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start from the best known construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # random power‑law fallback (exponent drawn uniformly)
        rng = np.random.default_rng()
        exp = rng.uniform(0.3, 0.9)
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-exp))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent (≈10 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.10 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq, best_score = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent on the true max‑convolution (≈12 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.12 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts (≈18 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.18 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D1 – deterministic annealing (≈30 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        det_deadline = time.time() + 0.30 * (deadline - time.time())
        a_det, val_det = _deterministic_anneal(best_seq, target_sum, det_deadline)
        if val_det < best_score:
            best_seq, best_score = a_det, val_det
        print(f"[{time.time():.2f}] After deterministic anneal best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D2 – stochastic annealing for polishing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _mass_transfer_anneal(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After final anneal best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
