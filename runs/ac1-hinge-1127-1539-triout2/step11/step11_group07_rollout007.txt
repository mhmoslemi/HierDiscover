"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  The objective after normalising the ℓ¹‑norm to
√(2 n) is simply the maximum entry of the self‑convolution.
The search combines several ideas:

* a rich set of random initialisers – especially *sparse* masks,
  random Fourier‑phase vectors and the families used previously;
* a smooth‑max (log‑sum‑exp) Adam optimiser for quick exploration;
* a sub‑gradient Adam optimiser that works on the exact max;
* a deterministic hill‑climb that moves mass from indices that
  contribute most to the current peak towards indices that contribute
  least;
* a very cheap random‑tweak loop (simulated‑annealing style) for the
  remaining budget.

All updates are projected onto the ℓ¹‑simplex Σa = √(2 n) (the
normalisation required by the evaluation function).  The routine is
budget‑aware: it stops before the allotted wall‑clock time (default
1000 s) and returns the best sequence found.
"""

# Helper linear‑algebra routines
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector onto the simplex {x ≥ 0, Σx = s}.
    This simple method first clamps to non‑negative and rescales the ℓ¹‑norm.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # fall back to uniform distribution
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers – a wider palette (including sparse families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 5:                   # sparse mask, equal amplitude
        frac = rng.uniform(0.1, 0.5)
        k = max(2, int(frac * n))
        mask = np.zeros(n)
        mask[rng.choice(n, size=k, replace=False)] = 1.0
        a = mask

    elif family == 6:                   # sparse mask with Dirichlet amplitudes
        frac = rng.uniform(0.1, 0.4)
        k = max(2, int(frac * n))
        idxs = rng.choice(n, size=k, replace=False)
        amps = rng.dirichlet(alpha=np.ones(k))
        a = np.zeros(n)
        a[idxs] = amps

    else:                               # random Fourier‑phase, take magnitude
        phi = rng.random(n) * 2.0 * np.pi
        A = np.exp(1j * phi)
        a = np.abs(np.fft.ifft(A).real)

    # tiny jitter for robustness
    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate (log‑sum‑exp).
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.12 * target_sum                # a little smaller for stability
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        # increase β slowly so that the smooth max turns into the true max
        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of max(a*a).
    The active set (indices attaining the maximum) is used as a probability
    vector p.
    """
    a = a0.copy()
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.22 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1

        # full convolution and active mask
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # gradient = 2·(a_rev ⋆ p)  (central n entries)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climb on the exact convolution
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # current worst index
        M = int(np.argmax(b))

        # indices that actually contribute to b[M]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # pick a destination j where the counterpart a[M‑j] is small
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart = a[M - j_range]

        order = np.argsort(counterpart)
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # move a fraction of the mass from i to j
        delta = a[i] * 0.5
        if delta <= 0.0:
            break

        # update convolution efficiently (linear time w.r.t. n)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # random jitter to break possible stagnation
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a,
                                             math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method mixes:
        – diverse random (including sparse) initialisers,
        – Adam on a smooth‑max surrogate,
        – Adam on the exact max (sub‑gradient),
        – deterministic hill‑climb,
        – cheap random pairwise tweaks.
    It respects a wall‑clock budget (default 1000 s) and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Seed from a previously discovered solution if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            n0 = prev.size
            target0 = math.sqrt(2.0 * n0)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={n0}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 1)  Length schedule – we start modestly and grow if time permits.
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000]   # feel free to adapt

    for n in schedule:
        if time.time() >= deadline:
            break

        target = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # 1a) Initialise a diverse pool of candidates.
        # --------------------------------------------------------------
        pool = []

        # up‑sample previous best (if any)
        if best_seq is not None and best_seq.size != n:
            pool.append(_upsample_sequence(best_seq, n))

        # a few random families (including the sparse ones)
        for _ in range(4):
            pool.append(_random_initial_sequence(n, rng))

        # random Fourier‑phase candidates
        for _ in range(3):
            phi = rng.random(n) * 2.0 * np.pi
            A = np.exp(1j * phi)
            a = np.abs(np.fft.ifft(A).real)
            a += rng.normal(scale=target * 0.001, size=n)
            pool.append(_project_to_simplex_fast(a, target))

        # --------------------------------------------------------------
        # 1b) Evaluate the pool and keep the champion.
        # --------------------------------------------------------------
        for cand in pool:
            sc = _max_convolution_fft(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] Initial pool improvement: "
                      f"n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # 1c) Allocate time for optimisation on this length.
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 15.0:
            break

        opt_budget = remaining * 0.55
        hc_budget = remaining * 0.30
        extra_budget = remaining * 0.15

        # --------------------------------------------------------------
        # 1d) Optimisation restarts (smooth‑max Adam → sub‑max Adam)
        # --------------------------------------------------------------
        n_restarts = max(1, int(opt_budget // 6.0))   # at least ~6 s each
        per_restart = opt_budget / n_restarts

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2,
                                 time.time() + per_restart)

            # start from either the best so far (first restart) or fresh random
            start_vec = best_seq.copy() if (r == 0 and best_seq is not None) \
                         else _random_initial_sequence(n, rng)

            # smooth‑max Adam (quick phase)
            mid_deadline = min(this_deadline,
                               time.time() + per_restart * 0.45)
            a_mid = _adam_optimize(start_vec, target, mid_deadline)

            # exact‑max Adam (sub‑gradient phase)
            a_opt = _adam_optimize_submax(a_mid, target, this_deadline)

            sc = _max_convolution_fft(a_opt)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] Optimisation restart {r+1}/{n_restarts} "
                      f"improved n={n} score={best_score:.6f}")

        # --------------------------------------------------------------
        # 1e) Deterministic hill‑climb on the current best.
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex_fast(a_cur, target)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + hc_budget)
            a_hc, sc_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if sc_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = sc_hc
                print(f"[{time.time():.1f}] Hill‑climb improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 1f) Tiny random pairwise tweaks (simulated‑annealing style)
        # --------------------------------------------------------------
        while time.time() + 1.0 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # move a small random fraction of mass from i to j
            frac = rng.uniform(0.01, 0.07)
            delta = cand[i] * frac
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            sc = _max_convolution_fft(cand)
            if sc < best_score - 1e-12:
                best_seq = cand.copy()
                best_score = sc
                print(f"[{time.time():.1f}] Random tweak improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 1g) Optional extra Adam run (if time remains)
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0 and extra_budget > 2.0:
            extra_deadline = deadline - 0.5
            a_ref = _adam_optimize_submax(best_seq, target, extra_deadline)
            sc_ref = _max_convolution_fft(a_ref)
            if sc_ref < best_score - 1e-12:
                best_seq = a_ref.copy()
                best_score = sc_ref
                print(f"[{time.time():.1f}] Post‑HC Adam improvement n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final polishing on whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 1.0
        best_seq = _adam_optimize_submax(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)

        if final_score < best_score - 1e-12:
            best_seq = best_seq
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Clean‑up: drop trailing near‑zeros (optional aesthetic step)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n,
                           math.sqrt(2.0 * n) / n,
                           dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4) Official evaluation report
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
