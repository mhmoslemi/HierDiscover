"""
Hybrid optimizer for the step‑function bound.

The algorithm proceeds in three stages:

1.  Take the best known construction (or a power‑law seed), resize it to a
    relatively large length (≈12‑16 k) and run a long projected‑gradient
    descent on a soft‑max approximation of the max‑convolution.  The descent
    uses momentum, a gradually increasing “temperature’’ β and an
    ℓ¹‑simplex projection, which respects the sum constraint exactly.

2.  Refine the result with a deterministic mass‑transfer Monte‑Carlo step
    that swaps a small amount of mass from indices that contribute heavily
    to the current peak to those that contribute little, accepting moves
    with a Metropolis rule.

3.  Perform a short simulated‑annealing permutation search that only
    re‑orders the values (the ℓ¹‑norm stays unchanged).  Finally try a few
    simple deterministic reorderings (alternating left/right) and keep the
    best sequence.

All phases respect the non‑negativity constraints, and the total runtime
is bounded by the provided budget (default 1000 s).  The returned list of
floats can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^{-p} sequence and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Create an exp(-lam*i) sequence, renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place values alternately left‑most/right‑most, large → small."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Alternating order but starting from the right."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max approximation of the max‑convolution (gradient)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (a reversed) ⋆ p   (⋆ = convolution)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on a soft‑max approximation of
    max(convolution).  Returns the best sequence seen and its score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # small jitter every few hundred iterations to escape flats
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Metropolis moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass transfers with an exponentially cooling temperature."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - (deadline - (deadline - time.time()))  # dummy to get fraction
        # linear schedule in log‑time
        tau = (deadline - time.time()) / (deadline - (deadline - (deadline - time.time())))
        temp = start_temp * (end_temp / start_temp) ** (1.0 - tau)

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on order only)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Re‑order the multiset of heights to reduce the convolution peak."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # contribution of each position to the peak
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass  # keep swap
        else:
            a[donor], a[receiver] = a[receiver], a[donor]  # revert

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.

    Strategy:
    * Start from the previously best construction (if any) or from a power‑law
      seed and up‑scale it to a fairly large length (≈12–16 k).
    * Run a long projected‑gradient descent on a soft‑max surrogate.
    * Apply a deterministic mass‑transfer refinement.
    * Run a permutation‑only simulated‑annealing search.
    * Try a few deterministic reorderings and return the best sequence.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a target length (larger lengths give more degrees of freedom)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n_target = 16000
    elif budget_s >= 700:
        n_target = 12000
    else:
        n_target = 8000
    target_sum = math.sqrt(2.0 * n_target)

    # ------------------------------------------------------------------
    # Initialise the candidate
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=float)
        a = _resize_sequence(init, n_target)
    else:
        # Power‑law heuristic; p≈0.55 has worked well historically
        a = _power_law_sequence(n_target, 0.55, target_sum)

    a = _project_to_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # 1. Long projected‑gradient descent (≈70 % of the budget)
    # ------------------------------------------------------------------
    gd_deadline = start_time + (deadline - start_time) * 0.70
    if gd_deadline > deadline - 5.0:
        gd_deadline = deadline - 5.0
    a_gd, val_gd = _annealed_gradient_descent(
        a,
        gd_deadline,
        target_sum,
        max_beta=5e6,
        step_factor=0.55,
        gamma=0.995,
    )
    best_seq = a_gd
    best_score = val_gd
    print(f"[{time.time():.2f}] GD finished score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Small exponent‑flattening tweak (helps escape shallow plateaus)
    # ------------------------------------------------------------------
    for alpha in (0.925, 0.95, 0.975):
        cand = best_seq ** alpha
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Exponent {alpha:.3f} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Deterministic mass‑transfer refinement (≈12 % of budget)
    # ------------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.12
    if mt_deadline > deadline - 2.0:
        mt_deadline = deadline - 2.0
    a_mt, val_mt = _deterministic_mass_transfer(
        best_seq,
        mt_deadline,
        target_sum,
        start_temp=0.4,
        end_temp=1e-5,
        low_frac=0.04,
    )
    if val_mt < best_score:
        best_score = val_mt
        best_seq = a_mt.copy()
        print(f"[{time.time():.2f}] MT after GD improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Permutation‑only simulated annealing (≈10 % of budget)
    # ------------------------------------------------------------------
    perm_deadline = time.time() + (deadline - time.time()) * 0.10
    if perm_deadline > deadline - 1.0:
        perm_deadline = deadline - 1.0
    a_perm, val_perm = _permutation_targeted_search(
        best_seq,
        perm_deadline,
        start_temp=0.15,
        end_temp=1e-5,
        top_frac=0.025,
        low_frac=0.025,
    )
    if val_perm < best_score:
        best_score = val_perm
        best_seq = a_perm.copy()
        print(f"[{time.time():.2f}] Permutation search improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Deterministic alternating re‑orderings (cheap, often beneficial)
    # ------------------------------------------------------------------
    for reorder_fn in (_alternating_order, _reversed_alternating):
        cand = reorder_fn(best_seq)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Re‑ordering ({reorder_fn.__name__}) gave {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Trim trailing zeros for a compact representation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 7. Final evaluation (just for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
