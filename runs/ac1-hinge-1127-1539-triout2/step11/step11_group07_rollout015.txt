    n = len(a)
    b = _full_convolution(a)
    best_a = a.copy()
    best_score = b.max()
    cur_score = best_score
    cur_a = a.copy()
    cur_b = b.copy()
    T0 = best_score * 0.1   # or (best_score - 1) maybe
    steps = 0
    while time.time() < deadline:
        steps += 1
        # temperature schedule
        T = T0 * (1 - (steps / 1000000)) # or compute based on elapsed time fraction
        # propose
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        delta = cur_a[i] * rng.uniform(0.01, 0.2) # small ratio
        if delta <= 0: continue
        # compute new b using incremental update (like hillclimb)
        b_new = cur_b.copy()
        # subtract contributions from i
        b_new[i:i+n] -= 2.0 * delta * cur_a
        # add contributions from j
        b_new[j:j+n] += 2.0 * delta * cur_a
        k2i = 2*i
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        k2j = 2*j
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        kij = i + j
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta
        new_score = b_new.max()
        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability
            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_a = _project_to_simplex_fast(cur_a, target) # ensure normalization? Actually sum unchanged if we exactly move mass but due to clipping we want to keep sum stable; moving mass keeps sum same, but potential clipping to zero? Avoid negative.

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = _next_pow2(2*n-1)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)
    # compute convolution a_rev * p
    a_rev = a[::-1]
    conv_len = n + (2*n-1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    # pick central n entries
    g = 2.0 * conv_rev[n-1:n-1+n]
    return g, b
```

"""
Multi‑phase optimisation for the step‑function challenge.

The algorithm works on a hierarchy of lengths (200‑2000‑4000‑8000‑16000).
For each size it runs many short Adam restarts on a smooth‑max surrogate,
followed by Adam on the exact max (sub‑gradient).  The best candidate is
polished with a deterministic hill‑climb that moves mass from the most
active convolution entry, and finally a short simulated‑annealing phase
that makes small stochastic mass‑exchanges.  Throughout the run every
vector is projected onto the ℓ¹‑simplex Σa = √(2 n) (the evaluation is
scale‑invariant, so fixing the sum is harmless).  The routine respects a
wall‑clock budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    This is just a non‑negative scaling (since we never need an
    optimisation‑level projection with ordering).
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation onto a new length, followed by simplex projection.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers (several simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


def _gaussian_initial_sequence(n: int, rng) -> np.ndarray:
    """Gaussian‑shaped start (discrete) projected onto the simplex."""
    target = math.sqrt(2.0 * n)
    centre = (n - 1) / 2.0
    sigma = n / 6.0                     # roughly 1/3 of the interval
    idx = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a += rng.normal(scale=target * 5e-4, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Gradient of smooth‑max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


# ----------------------------------------------------------------------
# Adam on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize_smooth(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.30 * target_sum          # a bit more aggressive than before
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        # beta schedule – grow quickly towards the true max
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam on the exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of max(a*a).
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.25 * target_sum          # slightly larger than the previous 0.2
    t = 0

    while time.time() < deadline:
        t += 1

        # full convolution and active mask
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        total = mask.sum()
        if total == 0.0:               # safety fallback
            p = np.ones_like(b) / b.size
        else:
            p = mask / total

        # gradient = 2·(a_rev ⋆ p)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climb on the exact convolution
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly attacks the worst convolution entry.
    Returns the best vector seen and its peak value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # pick a destination with the smallest counterpart
        counterpart = np.zeros(n, dtype=float)
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # linear‑time update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # jitter to escape shallow local minima
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score

# Simulated annealing (small stochastic moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap SA that moves a tiny fraction of mass between two positions.
    The acceptance probability follows exp(‑Δ/T) with a linear cooling schedule.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.size
    b = _full_convolution(a)
    cur_score = b.max()
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    T0 = cur_score * 0.05  # initial temperature (relative to current peak)

    steps = 0
    while time.time() < deadline:
        steps += 1
        elapsed = time.time() - start
        total = deadline - start
        T = max(0.0, T0 * (1.0 - elapsed / total))

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a modest fraction of the source entry
        delta = a[i] * rng.uniform(0.02, 0.15)
        if delta <= 0.0:
            continue

        # incremental update of the convolution (same formula as hill‑climb)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()
        accept = False
        if new_score < cur_score:
            accept = True
        elif T > 0.0:
            prob = math.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            # tiny numerical drift is harmless; enforce non‑negativity periodically
            if a[i] < 0.0:
                a[i] = 0.0
            b = b_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        # occasional re‑projection to keep the ℓ¹‑norm exact
        if steps % 8000 == 0:
            a = _project_to_simplex_fast(a, target_sum)
            b = _full_convolution(a)
            cur_score = b.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method combines:
        • random / Gaussian initial families,
        • Adam on a smooth‑max surrogate,
        • Adam on the exact max (sub‑gradient),
        • deterministic hill‑climb,
        • a short simulated‑annealing polishing,
    and respects a wall‑clock budget (default 1000 s).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1)  Seed from a previously found solution if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            target0 = math.sqrt(2.0 * prev.size)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={prev.size}) score={best_score:.6f}")
        except Exception as e:
            print(f"[{time.time():.1f}] Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Length schedule – gradually increase the dimension.
    # ------------------------------------------------------------------
    schedule = [500, 1000, 2000, 4000, 8000, 16000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # initialise (upsample previous best if possible)
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            # randomly decide between a pure random start and a Gaussian start
            if rng.random() < 0.6:
                a_init = _random_initial_sequence(n, rng)
            else:
                a_init = _gaussian_initial_sequence(n, rng)

        a_init = _project_to_simplex_fast(a_init, target)

        remaining = deadline - time.time()
        if remaining < 12.0:
            break

        # --------------------------------------------------------------
        # 2a)  Time allocation for this size
        # --------------------------------------------------------------
        opt_budget = remaining * 0.55
        hc_budget = remaining * 0.25
        sa_budget = remaining * 0.10
        extra_budget = remaining * 0.10

        # --------------------------------------------------------------
        # 2b)  Optimisation restarts (smooth‑max Adam → sub‑max Adam)
        # --------------------------------------------------------------
        min_restart = 6.0                # seconds per restart (≈2 iterations)
        n_restarts = max(1, int(opt_budget // min_restart))
        per_restart = opt_budget / n_restarts

        local_best = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)

            # first restart uses the upsampled a_init, later restarts are fresh
            start_vec = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)

            # Phase 1 – smooth‑max Adam (quick exploration)
            mid_deadline = min(this_deadline,
                               time.time() + per_restart * 0.45)
            a_mid = _adam_optimize_smooth(start_vec, target, mid_deadline)

            # Phase 2 – exact‑max Adam (sub‑gradient refinement)
            a_opt = _adam_optimize_submax(a_mid, target, this_deadline)

            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best = a_opt.copy()

        if local_best is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] New best after n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2c)  Deterministic hill‑climb (exact max)
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex_fast(a_cur, target)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + hc_budget)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2d)  Random jitter / tiny tweaks (simulated‑annealing style)
        # --------------------------------------------------------------
        while time.time() + 1.0 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            score = _max_convolution_fft(cand)
            if score < best_score - 1e-12:
                best_seq = cand
                best_score = score
                print(f"[{time.time():.1f}] Random tweak improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2e)  Short simulated annealing polishing (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline - 0.5 and sa_budget > 2.0:
            sa_deadline = time.time() + sa_budget
            a_sa, score_sa = _simulated_annealing(best_seq, target, sa_deadline)
            if score_sa < best_score - 1e-12:
                best_seq = a_sa.copy()
                best_score = score_sa
                print(f"[{time.time():.1f}] SA polishing improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2f)  Optional extra Adam run (if any time left)
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0 and extra_budget > 2.0:
            extra_deadline = deadline - 0.5
            a_ref = _adam_optimize_submax(best_seq, target, extra_deadline)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score - 1e-12:
                best_seq = a_ref.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑HC Adam improved n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing on whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 1.0
        best_seq = _adam_optimize_submax(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)

        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

        # a very short SA as a last touch
        if time.time() < deadline - 0.5:
            a_last, s_last = _simulated_annealing(best_seq, target, deadline - 0.3)
            if s_last < best_score - 1e-12:
                best_seq = a_last
                best_score = s_last
                print(f"[{time.time():.1f}] Final SA improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5) Official evaluation report
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.1f}] Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
