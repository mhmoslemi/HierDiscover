def _local_hill_climb(a0, target_sum, deadline, step_scale=0.02):
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    n = len(a)
    while time.time() < deadline:
        # pick donors more likely at higher a
        donor = rng.choice(n, p=a/np.sum(a))
        # random receiver != donor
        receiver = rng.integers(0, n-1)
        if receiver == donor:
            continue
        delta = a[donor] * step_scale * rng.random()
        if delta <= 0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        # maintain sum already same; ensure nonnegativity (a[donor] may become negative if delta > a[donor], but we limit delta)
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
    return best_a, best_val

"""
Advanced search for a low‑peak self‑convolution step function.

The algorithm builds a rich pool of parametric, symmetric and random
candidates, keeps the best few and refines them through several
gradient‑type phases:
  * soft‑max (smooth) momentum GD with annealed β,
  * exact sub‑gradient descent on the true max‑convolution,
  * deterministic greedy mass transfer,
  * simple hill‑climbing (pairwise mass moves),
  * a final high‑β polishing GD.

All operations use FFT‑based convolution (O(n log n)) and the Euclidean
projection onto the simplex {x ≥ 0, Σx = √(2 n)} to keep the ℓ¹‑norm
constant.  The sequence length n is chosen from the time budget:
larger n gives a finer discretisation of the continuous optimum and
normally yields a smaller maximal convolution value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (all produce non‑negative arrays)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """(i+1)^(-p)   i = 0,…,n‑1."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """exp(-lam·i)   i = 0,…,n‑1."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    """sin(π·(i+0.5)/n)^p   i = 0,…,n‑1."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _sym_power_law_sequence(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law (distance from centre)⁻ᵖ."""
    d = np.abs(np.arange(n, dtype=np.float64) - (n - 1) / 2.0)
    return (d + 1.0) ** (-p)


def _sym_exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay exp(-lam·distance)."""
    d = np.abs(np.arange(n, dtype=np.float64) - (n - 1) / 2.0)
    return np.exp(-lam * d)


def _arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine density ≈ 1/√(x·(1‑x))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid division by zero – endpoints are never exactly 0/1 because of +0.5
    return 1.0 / np.sqrt(x * (1.0 - x))


def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Beta(α,α) ∝ x^{α‑1} (1‑x)^{α‑1}
    return (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))


def _random_dirichlet_sequence(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet draw with concentration `conc` (≥0)."""
    if conc <= 0:
        # fallback to uniform
        v = np.ones(n, dtype=np.float64)
    else:
        v = rng.gamma(conc, 1.0, size=n)
    return v


# ----------------------------------------------------------------------
# Soft‑max gradient (used by momentum GD)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Soft‑max momentum GD (annealed β)
# ----------------------------------------------------------------------
def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 3e5,
                                   step_factor: float = 0.6,
                                   gamma: float = 0.995) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase X – Exact max sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.90,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active quadratic term
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic mass‑transfer (greedy annealing)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase C – Simple stochastic hill‑climbing (pairwise mass moves)
# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray,
                      target_sum: float,
                      deadline: float,
                      init_step: float = 0.02,
                      decay: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()
    n = a.shape[0]
    step = init_step

    while time.time() < deadline:
        # pick donor proportional to its mass (more impact)
        donor = rng.choice(n, p=a / np.sum(a))
        receiver = rng.integers(0, n)
        if receiver == donor:
            continue
        delta = a[donor] * step * rng.random()
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # keep the improvement
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta

        step *= decay  # slowly shrink step size

    return best_a, best_val
