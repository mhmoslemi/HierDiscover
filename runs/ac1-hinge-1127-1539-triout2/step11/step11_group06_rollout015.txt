                ok = False
                break

            # add sums with existing A and self

                sums_set.add(a + candidate)
            sums_set.add(candidate + candidate)

The benchmark asks for a non‑negative step‑function `a` (a list of floats) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2},
\]

where `a*a` is the linear convolution of `a` with itself.
If we normalise the vector to  

\[
\sum a = \sqrt{2n},
\]

the score is exactly the maximal entry of the convolution, i.e. we have to make the
self‑convolution as flat as possible.

The program below follows three ideas that together improve the previous
implementation:

1. **Sidon‑set warm‑start** –  
   A *Sidon* (or `B₂`) set is a collection of indices whose pairwise sums are all
   distinct.  If we put equal mass on the indices of a Sidon set, every convolution
   entry contains at most a single product `a_i·a_j`.  This gives a very good
   starting point whose score is roughly  

   \[
   \frac{2n}{m^2},
   \]

   where `m` is the size of the set.  A greedy construction of a maximal Sidon set
   for the given length `n` is cheap and yields a score already below 1.6.

2. **Soft‑max (log‑sum‑exp) mirror descent** –  
   Instead of minimising the raw `max` we minimise the smooth surrogate  

   \[
   \operatorname{lse}_\beta(b)=\frac{1}{\beta}\log\!\Bigl(\sum_k e^{\beta b_k}\Bigr)
   \]

   and gradually increase the temperature parameter `β`.  The gradient can be
   obtained by a single FFT, and we use a multiplicative mirror‑descent step
   followed by a Euclidean projection onto the simplex (the sum constraint).

3. **Targeted local refinement & simulated annealing** –  
   After the soft‑max phase we repeatedly:
   * locate the current maximal convolution entry,
   * pick a *donor* index that contributes most to it,
   * pick a *receiver* index that contributes little,
   * move a tiny amount of mass `δ` from donor to receiver,
   * update the convolution in **O(n)** time,
   * accept the move according to a simple annealing schedule.

   This hill‑climbing step eliminates the remaining spikes of the convolution.

The whole pipeline respects the supplied time budget (`budget_s`, default 1000 s)
and returns the best sequence found.  All helper functions are defined at the
module level, there are no closures or file I/O, and the code works with the
provided `evaluate_sequence` function.

```python
"""
Hybrid optimisation for the non‑negative step‑function benchmark.

The routine consists of:
  1. Greedy Sidon‑set warm‑start (almost flat convolution).
  2. Mirror‑descent on a log‑sum‑exp (soft‑max) surrogate of the max‑convolution.
  3. Targeted O(n) mass‑transfer hill‑climbing.
  4. Simple simulated‑annealing using the same O(n) updates.

All phases are timed so that the total run‑time never exceeds the supplied
budget (default 1000 s).  The final sequence is projected onto the simplex
(∑a = √(2 n)) and trimmed of trailing zeros before being returned.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_full_fft(a).max())


def _score(a: np.ndarray, target_sum: float) -> float:
    """Exact benchmark score 2·n·max(a*a) / (target_sum²)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    if target_sum < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (target_sum * target_sum))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a different length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# 1.  Sidon set warm‑start
# ----------------------------------------------------------------------


def _greedy_sidon(n: int, max_m: int) -> List[int]:
    """
    Greedy construction of a B₂ (Sidon) set inside {0,…,n‑1}.
    Returns a list of indices.  Stops once `max_m` elements are collected.
    """
    chosen = []
    sums = set()
    for cand in range(n):
        # test whether adding `cand` would create a duplicate sum
        ok = True
        # self‑sum
        if (cand + cand) in sums:
            continue
        for a in chosen:
            if (a + cand) in sums:
                ok = False
                break
        if not ok:
            continue
        # accept candidate
        for a in chosen:
            sums.add(a + cand)
        sums.add(cand + cand)
        chosen.append(cand)
        if len(chosen) >= max_m:
            break
    return chosen


def _sidon_initial(n: int, target_sum: float) -> np.ndarray:
    """
    Build a warm‑start based on a Sidon set.
    All selected positions receive equal mass, the rest are zero.
    """
    # Theoretical maximum size of a Sidon set inside [0,n) is about √(2n).
    # We ask the greedy routine for that many elements but never exceed n.
    max_m = int(math.sqrt(2.0 * n))
    sid = _greedy_sidon(n, max_m)
    m = len(sid)
    if m == 0:          # fallback – should never happen
        a = np.full(n, target_sum / n, dtype=np.float64)
        return a

    a = np.zeros(n, dtype=np.float64)
    mass = target_sum / m
    a[sid] = mass
    return a


# ----------------------------------------------------------------------
# 2.  Soft‑max (log‑sum‑exp) mirror descent
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float) -> Tuple[np.ndarray, np.ndarray]:
    """
    Gradient of `lse_beta(b)` where `b = a*a`.
    Returns (grad, b_vector) where `grad` has same length as `a`.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = b.max()
    # log‑sum‑exp in a numerically stable way
    exp_term = np.exp(beta * (b - max_b))
    p = exp_term / exp_term.sum()                     # soft‑max probabilities

    # grad = 2 * (reverse(a) *conv* p)
    a_rev = a[::-1]
    pf = np.fft.rfft(p, n=size)
    grad_full = np.fft.irfft(np.fft.rfft(a_rev, n=size) * pf, n=size)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad, b


def _mirror_descent(a0: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    beta_start: float = 1.0,
                    beta_max: float = 1e7,
                    eta_base: float = 0.02) -> Tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent on the soft‑max surrogate.
    Gradually raises `beta` to tighten the approximation of the true max.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best = a.copy()
    best_val = _score(a, target_sum)

    beta = beta_start
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)

        # step size shrinks with beta – this mirrors the derivation in
        # the original reference code.
        eta = eta_base / (beta + 1.0)

        # multiplicative update, then re‑project onto the simplex
        a = a * np.exp(-eta * grad)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape plateaux
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a, target_sum)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] mirror it={it:5d} β={beta:.1e} score={best_val:.6f}")

        # raise beta exponentially, but stop once we hit the cap
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * 2.0, beta_max)

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# 3.  Targeted local refinement (mass transfer)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> Tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing / simulated annealing using O(n) convolution updates.
    Each iteration moves a small amount of mass from a “donor” (large contribution
    to the current maximal entry) to a “receiver” that participates little.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_full_fft(a)
    best = a.copy()
    best_val = _score(a, target_sum)

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = best_val * 0.04                       # initial temperature

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions to the maximum entry:
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() == 0:
            donor = rng.integers(lo, hi + 1)
        else:
            donor = rng.choice(idx, p=contrib / contrib.sum())

        # pick a receiver among indices that are “light” for this max entry
        weight = np.zeros(n, dtype=np.float64)
        weight[lo:hi + 1] = contrib
        thresh = np.percentile(weight, 30)
        low = np.where(weight <= thresh)[0]
        receiver = rng.choice(low) if low.size else rng.integers(n)

        if donor == receiver:
            continue

        # move a tiny fraction of the donor’s mass
        max_delta = min(a[donor] * 0.03, target_sum * 5e-4)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution (see the original code for the exact formula)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_score = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # simple annealing acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))
        accept = new_score < best_val or (temperature > 0 and
                                         rng.random() < math.exp(-(new_score - best_val) / temperature))

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_score < best_val:
                best_val = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] local refine score={best_val:.6f}")

        # occasional re‑normalisation to compensate drift
        if time.time() - start - elapsed > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_full_fft(a)

    return best, best_val

# 4.  Simulated annealing (cheap O(n) moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> Tuple[np.ndarray, float]:
    """Simple SA that repeatedly moves a small amount of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_full_fft(a)
    cur_val = _score(a, target_sum)
    best = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0.0:
        return best, best_val

    T0 = max(0.05, (best_val - 1.0) * 0.5)

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # choose donor proportional to its mass (so we rarely pick a zero)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue

        # distinct receiver uniformly
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.04, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) convolution update
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_score = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_score < cur_val:
            accept = True
        elif temperature > 0.0:
            accept = rng.random() < math.exp(-(new_score - cur_val) / temperature)

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_score
            if new_score < best_val:
                best_val = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # periodic global jitter to escape deep plateaus
        if int(elapsed * 100) % 6000 == 0:
            jitter = rng.normal(scale=target_sum * 2e-4, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_full_fft(a_j)
            val_j = _score(a_j, target_sum)
            if val_j < best_val:
                best_val = val_j
                best = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA global jitter improved score={best_val:.6f}")

    return best, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Find a non‑negative step sequence with a low benchmark score.
    The function respects a time budget (seconds) and returns the best
    sequence discovered as a plain Python list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – a slight up‑scaling for large budgets
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 18000
    elif budget_s >= 700:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. warm‑start (Sidon set)
    # ------------------------------------------------------------------
    a0 = _sidon_initial(n, target_sum)
    best_seq = a0.copy()
    best_score = _score(best_seq, target_sum)
    print(f"[{time.time():.2f}] Warm‑start (Sidon) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. mirror descent on soft‑max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        md_deadline = time.time() + 0.20 * (deadline - time.time())
        a_md, v_md = _mirror_descent(best_seq, target_sum, md_deadline)
        if v_md < best_score:
            best_seq, best_score = a_md, v_md
        print(f"[{time.time():.2f}] After mirror descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. targeted local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lr_deadline = time.time() + 0.35 * (deadline - time.time())
        a_lr, v_lr = _local_refine(best_seq, target_sum, lr_deadline)
        if v_lr < best_score:
            best_seq, best_score = a_lr, v_lr
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. simulated annealing (cheap O(n) moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, v_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if v_sa < best_score:
            best_seq, best_score = a_sa, v_sa
        print(f"[{time.time():.2f}] After SA best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final clean‑up: trim trailing zeros and convert to list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final verification (the external scorer)
    try:
        final_val = evaluate_sequence(final_seq)           # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in a future call
    globals()["height_sequence_1"] = final_seq

    return final_seq
```
