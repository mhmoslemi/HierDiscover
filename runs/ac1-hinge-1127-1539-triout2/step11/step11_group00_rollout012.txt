import cvxpy as cp

def solve_lp_max_sum(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # Convolution constraints: for each k from 0 to 2n-2:
    # sum_i f[i] * g[k - i] where valid indices => <= M
    # Equivalent to sum_{i=max(0,k-(n-1))}^{min(k,n-1)} f[i] * g[k-i] <= M
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        idx_range = range(lo, hi+1)
        # Build linear expression: sum_{i} f[i] * g[k-i] <= M
        # Since g[k-i] variable, we can write as sum f[i] * g[k-i] = sum_j f[k-j] * g[j] for j in [0,n-1]
        # But easiest: loop through indices
        expr = 0
        for i in idx_range:
            expr += f[i] * g[k - i]
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return None
    return np.array(g.value)

    y = np.log(a0 + 1e-12)
    size = 1 << ((2*n - 1).bit_length())
    # define objective function using closure referencing n, size, target_sum, beta
    def fun(y):
        exp_y = np.exp(y - np.max(y))
        s = exp_y.sum()
        a = target_sum * exp_y / s

        # convolution a*a
        a_fft = np.fft.rfft(a, n=size)

        max_b = b.max()

"""
Improved step‑function search.

The algorithm consists of three stages:

1️⃣  A series of L‑BFGS optimisations on a smooth soft‑max
    surrogate of the max‑convolution.  The variables are the
    logarithms of the heights, guaranteeing positivity and an
    exact L1‑normalisation (∑a = √(2 n)).  A schedule of
    increasing “softness’’ parameters β gradually turns the
    surrogate into the true max‑convolution.

2️⃣  A long simulated‑annealing “mass‑transfer’’ phase that moves a
    tiny amount of mass from indices that contribute most to the
    current peak towards indices that are low.  The temperature
    starts at ~45 % of the current score and cools linearly to zero.

3️⃣  Final cleanup – truncate trailing zeros, evaluate the true
    objective and store the result for warm‑starting the next run.

The search adapts the problem size to the allotted budget,
uses the best known construction (`height_sequence_1`) when
available and stays well within the 1000 s time limit.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, power‑of‑2 size)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For Σa = √(2 n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS surrogate optimisation
# ----------------------------------------------------------------------


class _LBFGSObjective:
    """
    Callable objective for L‑BFGS.

    Variables are y = log(a) (up to an irrelevant additive constant).
    The true heights are a = S·exp(y) / Σexp(y)   with S = √(2 n).

    For a given β the objective is the smooth soft‑max
    Jβ(a) = (1/β)·log Σ_k exp(β·b_k)   where b = a*a.
    Its gradient w.r.t. y is computed analytically.
    """

    def __init__(self, n: int, target_sum: float, beta: float):
        self.n = n
        self.S = target_sum
        self.beta = beta
        # FFT size – the smallest power‑of‑2 covering the full convolution
        self.fft_size = 1 << ((2 * n - 1).bit_length())

    def __call__(self, y: np.ndarray):
        # ----- a from y (soft‑max normalisation) -----
        # shift to avoid overflow
        y_shift = y - np.max(y)
        e = np.exp(y_shift)
        Z = e.sum()
        a = self.S * e / Z

        # ----- convolution a*a (full, length 2n‑1) -----
        a_fft = np.fft.rfft(a, n=self.fft_size)
        b = np.fft.irfft(a_fft * a_fft, n=self.fft_size)[: 2 * self.n - 1]

        # ----- soft‑max of the convolution entries -----
        max_b = b.max()
        scaled = self.beta * (b - max_b)          # all ≤ 0
        exp_scaled = np.exp(scaled)
        sum_exp = exp_scaled.sum()
        J = max_b + np.log(sum_exp) / self.beta   # stable log‑sum‑exp

        # ----- gradient w.r.t. a -----
        p = exp_scaled / sum_exp                  # soft‑max weights, length 2n‑1
        a_rev = a[::-1]
        a_rev_fft = np.fft.rfft(a_rev, n=self.fft_size)
        p_fft = np.fft.rfft(p, n=self.fft_size)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=self.fft_size)[: 2 * self.n - 1]
        grad_a = 2.0 * conv_rev[self.n - 1:self.n - 1 + self.n]   # same size as a

        # ----- chain rule: ∂J/∂y = diag(a)·(∂J/∂a – (a·∂J/∂a)/S) -----
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot / self.S)

        return J, grad_y


def _lbfgs_optimize(a0: np.ndarray, target_sum: float,
                    beta: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Runs a limited L‑BFGS optimisation on the β‑soft‑max surrogate.
    The number of L‑BFGS iterations is bounded by the time left.
    Returns the (renormalised) heights and the true score.
    """
    n = a0.shape[0]
    y0 = np.log(a0 + 1e-12)          # small epsilon avoids log(0)

    # estimate a safe number of iterations (≈0.5–1 ms per iteration)
    time_left = max(0.0, deadline - time.time())
    max_iter = max(5, int(time_left / 0.006))   # 6 ms ≈ one iteration
    max_iter = min(max_iter, 400)               # hard cap

    obj = _LBFGSObjective(n, target_sum, beta)

    res = minimize(obj, y0,
                   method='L-BFGS-B',
                   jac=True,
                   options={'maxiter': max_iter,
                            'disp': False,
                            'gtol': 1e-9})

    a_opt = target_sum * np.exp(res.x) / np.exp(res.x).sum()
    val = _score(a_opt)
    return a_opt, val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer (very similar to the previous run,
# but with a larger initial temperature and a slightly larger delta)
# ----------------------------------------------------------------------


def _mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """Long‑run annealing that moves tiny amounts of mass from “bad’’ to “good’’ indices."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.45                 # start at ~45 % of the current score

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – larger contribution ⇒ more likely to be chosen
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – index with low height (inverse‑probability)
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by the temperature)
        max_delta = min(a[donor] * 0.06, target_sum * 0.0015)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0.0 else 0.0)
        if delta <= 0.0:
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if time.time() - last_full > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional tiny jitter to keep the dynamics alive
        if it % 2000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val

    elif budget_s >= 500:
        n = 16000
    else:
        n = 10000

    globals()["height_sequence_1"] = final_seq
    return final_seq
