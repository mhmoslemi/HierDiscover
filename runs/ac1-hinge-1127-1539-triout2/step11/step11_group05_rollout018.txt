    a = a0.copy()
    # ensure sum close to target_sum (maybe slight rounding)
    current = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = current

    n = a.shape[0]
    rng = np.random.default_rng()
    start_temp = 0.5  # high
    end_temp = 1e-4
    it = 0
    while time.time() < deadline:
        # schedule temperature
        t = (time.time() - start_time) / total_time
        temp = start_temp * (end_temp/start_temp)**t

        # choose donor index weighted by contribution to current max
        K = int(np.argmax(_conv_fft(a))) # could compute conv array but we only need conv for max maybe store conv.
        # we can compute conv each iteration anyway
        conv = _conv_fft(a) # O(n log n)
        K = int(np.argmax(conv))

        # Compute contributions w_i = a[i] * a[K-i] (if index in range). We'll use that to sample donor.
        lo = max(0, K - (n - 1))
        hi = min(K, n - 1)
        idxs = np.arange(lo, hi + 1)
        w = a[idxs] * a[K - idxs]  # product contributions

        # donor: choose i with probability proportional to w (or w^p)
        if w.sum() > 0:
            probs = w / w.sum()
            donor = rng.choice(idxs, p=probs)
        else:
            donor = rng.integers(0, n)

        # choose receiver uniformly random from all indices except donor
        receiver = rng.integers(0, n-1)
        if receiver >= donor:
            receiver += 1

        # Determine max delta
        max_delta = a[donor] * 0.5  # move up to half of donor mass
        # optionally limit based on product difference
        # Variation: draw delta in [0, max_delta] scaled by temperature
        delta = rng.random() * max_delta * (temp / start_temp)  # scale smaller as temp decreases

        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        # Ensure nonnegative (donor > delta)
        # compute new max conv:
        val_new = _max_convolution_fft(a_new)

        # accept
        if val_new < current or rng.random() < np.exp(-(val_new - current) / max(temp, 1e-12)):
            a = a_new
            current = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} new best={best_val}")

        it += 1

    return best_a, best_val

"""
Propose a non‑negative step‑function minimizing the maximal self‑convolution.

Algorithm
---------
1. Choose a length n (12000 for a long budget) and the target ℓ¹‑norm
   `target = sqrt(2 n)`.  The objective for a sequence with this exact sum
   equals `max(conv(a,a))`.
2. Build a diverse library of parametric shapes (power law, exponential,
   sine‑power).  Each shape is *scaled* (not projected) so that its sum equals
   `target`.  Small Gaussian jittered copies are added for diversity.
3. Quickly evaluate all candidates with an FFT based convolution and keep the
   best few.
4. Refine the best seeds with a custom simulated‑annealing optimizer that
   repeatedly transfers a random amount of mass from a “donor’’ (chosen with
   probability proportional to its contribution to the current peak) to a
   random “receiver’’.  Acceptance follows a temperature schedule and the
   best ever sequence is remembered.
5. After the time budget is exhausted the best sequence is trimmed,
   re‑scaled to the exact target sum, stored globally for warm‑starts and
   returned as a plain Python list.

The optimizer works directly on the simplex (mass is transferred, the total
sum stays constant) and uses only O(n log n) FFT calls, which comfortably fit
into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _scale_to_sum(v: np.ndarray, s: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ¹‑norm equals `s`."""
    total = v.sum()
    if total == 0:
        return v
    return v * (s / total)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape generators (unnormalised)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """(i+1)^{-p} for i=0…n‑1."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """exp(-lam·i)."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    """sin(π·(i+½)/n)^{p}."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


# ----------------------------------------------------------------------
# Simulated annealing optimiser
# ----------------------------------------------------------------------
def _simulated_annealing_opt(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing.
    Keeps the ℓ¹‑sum exactly `target_sum` by moving a random `delta`
    from a donor to a receiver.
    """
    a = _scale_to_sum(a0.copy(), target_sum)
    cur_val = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_val

    n = a.shape[0]
    rng = np.random.default_rng()

    # temperature schedule
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    start_temp = 0.12
    end_temp = 1e-6
    step_factor = 0.5          # max fraction of donor mass that can be moved

    # initial convolution & peak index
    conv_cur = _conv_fft(a)
    K = int(np.argmax(conv_cur))

    it = 0
    while time.time() < deadline:
        # ----- temperature (exponential cooling) -----
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * ((end_temp / start_temp) ** tau)

        # ----- choose donor (probability ∝ contribution to current peak) -----
        lo = max(0, K - (n - 1))
        hi = min(K, n - 1)
        idxs = np.arange(lo, hi + 1, dtype=np.int64)
        contributions = a[idxs] * a[K - idxs]
        if contributions.sum() > 0:
            probs = contributions / contributions.sum()
            donor = rng.choice(idxs, p=probs)
        else:
            donor = rng.integers(0, n)

        # ----- choose receiver (uniform, different from donor) -----
        rec = rng.integers(0, n - 1)
        if rec >= donor:
            rec += 1
        receiver = rec

        # ----- propose a transfer -----
        max_delta = a[donor] * step_factor
        if max_delta <= 0:
            # nothing to give – skip this iteration
            it += 1
            continue
        # scale the step size with temperature (larger steps early)
        delta = rng.random() * max_delta * (temp / start_temp)
        if delta <= 0:
            it += 1
            continue

        new_a = a.copy()
        new_a[donor] -= delta
        new_a[receiver] += delta
        # sum is unchanged, but numerical drift can appear – renormalise
        new_a = _scale_to_sum(new_a, target_sum)

        new_val = _max_convolution_fft(new_a)

        # ----- Metropolis acceptance -----
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_val:.6f}")

            # recompute full convolution only on acceptance (expensive)
            conv_cur = _conv_fft(a)
            K = int(np.argmax(conv_cur))

        # ----- periodic housekeeping -----
        if it % 500 == 0 and it > 0:
            # recompute peak information even if we didn't accept
            conv_cur = _conv_fft(a)
            K = int(np.argmax(conv_cur))
            cur_val = conv_cur.max()

        if it % 2000 == 0 and it > 0:
            # occasional global jitter to escape flat regions
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _scale_to_sum(a, target_sum)
            conv_cur = _conv_fft(a)
            K = int(np.argmax(conv_cur))
            cur_val = conv_cur.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA jitter improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with minimal max self‑convolution.
    Returns a plain Python list ready for `evaluate_sequence`.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        if budget_s >= 800:
            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Helper to evaluate a candidate quickly
    # --------------------------------------------------------------
    def _eval(a: np.ndarray) -> float:
        return _max_convolution_fft(a)

    # --------------------------------------------------------------
    # Build a pool of parametric candidates
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # Warm‑start (if any)
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if warm.shape[0] != n:
            warm = _resize_sequence(warm, n)
        warm = _scale_to_sum(warm, target_sum)
        candidates.append(warm)

    # Power‑law family
    for p in np.linspace(0.45, 0.78, 67):  # fine grid around the theoretical optimum
        raw = _power_law_sequence(n, p)
        seq = _scale_to_sum(raw, target_sum)
        candidates.append(seq)

        # jittered copy
        jitter = rng.normal(scale=target_sum * 0.0015, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _scale_to_sum(seq_j, target_sum)
        candidates.append(seq_j)

    # Exponential decay family
    for lam in np.linspace(2e-4, 0.014, 45):
        raw = _exponential_sequence(n, lam)
        seq = _scale_to_sum(raw, target_sum)
        candidates.append(seq)

        jitter = rng.normal(scale=target_sum * 0.0015, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _scale_to_sum(seq_j, target_sum)
        candidates.append(seq_j)

    # Sine‑power family
    for p in np.linspace(0.2, 2.5, 36):
        raw = _sin_power_sequence(n, p)
        seq = _scale_to_sum(raw, target_sum)
        candidates.append(seq)

        jitter = rng.normal(scale=target_sum * 0.0015, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _scale_to_sum(seq_j, target_sum)
        candidates.append(seq_j)

    # Random permutations of a few promising seeds (adds diversity)
    for base in candidates[:8]:
        perm = base.copy()
        rng.shuffle(perm)
        candidates.append(perm)

    # --------------------------------------------------------------
    # Quick evaluation of the pool
    # --------------------------------------------------------------
    scored = []
    for seq in candidates:
        val = _eval(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])

    top_k = min(6, len(scored))
    best_seq, best_val = scored[0][1].copy(), scored[0][0]
    print(f"[{time.time():.2f}] Initial best from pool = {best_val:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing refinement
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0 and top_k > 0:
        # allocate roughly equal time per seed
        per_seed = remaining * 0.85 / top_k
        for i in range(top_k):
            cur_dead = time.time() + min(per_seed, deadline - time.time())
            a0 = scored[i][1]
            a_opt, val_opt = _simulated_annealing_opt(a0, target_sum, cur_dead)
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] SA improved best to {best_val:.6f}")

        # use any leftover time for a final polish on the current best
        remaining = deadline - time.time()
        if remaining > 4.0:
            a_opt, val_opt = _simulated_annealing_opt(best_seq, target_sum, deadline)
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] Final SA polish improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Post‑process & return
    # --------------------------------------------------------------
    # Re‑scale to eliminate tiny numerical drift
    best_seq = _scale_to_sum(best_seq, target_sum)

    # Trim trailing near‑zero entries (compact representation)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    final_seq = best_seq.tolist()

    # Optional final evaluation for logging
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
