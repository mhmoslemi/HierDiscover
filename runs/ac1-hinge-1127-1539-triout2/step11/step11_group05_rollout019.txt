"""
Propose a low‑max‑self‑convolution step function.

The algorithm is a lightweight but aggressive multi‑phase search:

1. **Problem size** – chooses a length `n` (≈12000 for a full budget) and
   fixes the normalising ℓ¹‑norm `S = sqrt(2·n)`.  For this `S`
   the evaluation objective equals the maximal entry of the self‑convolution.

2. **Rich candidate pool** – a handful of analytic families (power law,
   exponential, sinusoid, “U‑shape’’ ((i·(n‑i))⁻ᵖ), and a regularised
   arcsine shape) are created, projected onto the simplex {x≥0, Σx=S},
   jittered, and also added in random permutations.

3. **Coarse filtering** – all candidates are evaluated with an FFT
   convolution and the best `K` are kept.

4. **Soft‑max momentum GD** – each of the `K` survivors is refined by a
   momentum gradient descent on a smooth approximation of `max(conv)`.
   The inverse temperature `β` is annealed to 3·10⁵.

5. **Exact sub‑gradient descent** – a few passes of true max‑convolution
   sub‑gradient descent further lower the peak.

6. **Deterministic mass‑transfer** – greedy moves of mass from
   “high‑contribution’’ indices to “low‑contribution’’ indices are
   accepted if they reduce the peak (simulated‑annealing style).

7. **Final polishing** – another short soft‑max GD with a higher
   temperature (`β = 5·10⁵`).

8. **Mix‑in optimisation** – the best sequence is mixed with a few
   simple shapes (uniform and the U‑shape) by a line search over the
   mixing weight `ε`.  The convolution of the mixture can be written in
   closed form, so the line search is performed with only O(n) work.

All phases respect the global time‑budget (≈1000 s).  The final
sequence is stored in ``height_sequence_1`` for warm‑starts and is
returned as a plain ``list`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution a*b (both length n) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """(i+1)⁻ᵖ, monotone decreasing."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """exp(-λ·i)."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    """sin(π·(i+0.5)/n)ᵖ."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _u_shape_sequence(n: int, p: float) -> np.ndarray:
    """
    U‑shape roughly proportional to ((i+1)*(n-i))⁻ᵖ.
    Peaks at the two ends, low in the middle.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n + 1 - i)
    return prod ** (-p)


def _arcsine_like_sequence(n: int, eps: float = 1e-6) -> np.ndarray:
    """
    Regularised arcsine shape: 1 / sqrt( (i+½)*(n‑i‑½) + eps ).
    The epsilon prevents a singularity at the boundaries.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    val = 1.0 / np.sqrt(i * (n - i) + eps)
    return val


def _uniform_sequence(n: int) -> np.ndarray:
    """Constant sequence (will be scaled later)."""
    return np.ones(n, dtype=np.float64)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (used in GD)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float) -> Tuple[np.ndarray, np.ndarray]:
    """
    Gradient of the soft‑max approximation of max(conv). Returns (grad, conv).
    The gradient equals 2·(rev(a) ⋆ p) restricted to the central n entries,
    where p is the soft‑max distribution over the convolution entries.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Soft‑max momentum GD on the soft‑max
# ----------------------------------------------------------------------
def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 3e5,
                                   step_factor: float = 0.6,
                                   gamma: float = 0.995) -> Tuple[np.ndarray, float]:
    """Momentum GD on a soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase X – Exact max sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> Tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

# Phase B – Deterministic mass‑transfer (local annealing)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> Tuple[np.ndarray, float]:
    """Greedy mass transfer that reduces the current peak."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

# Mixing with an auxiliary shape (exact objective via pre‑computation)
# ----------------------------------------------------------------------
def _mix_optimal(a: np.ndarray,
                 v: np.ndarray,
                 deadline: float) -> Tuple[np.ndarray, float]:
    """
    Find the best non‑negative weight ε for the mixture a + ε·v,
    where v is normalised to have total mass 1.
    The convolution of the mixture can be written as:
        conv(ε) = conv_aa + 2·ε·conv_av + ε²·conv_vv
    The function is evaluated cheaply for any ε.
    """
    n = a.shape[0]

    # Normalise v to unit ℓ¹ mass (so ε is the total extra mass)
    v = v / np.maximum(v.sum(), 1e-12)

    conv_aa = _conv_fft(a)
    conv_av = _conv_fft_two(a, v)
    conv_vv = _conv_fft(v)

    a_max = a.max()
    v_max = v.max()
    # hard cap to stay inside evaluate_sequence’s 0‑1000 range
    eps_limit = (1000.0 - a_max) / max(v_max, 1e-12)
    eps_limit = max(eps_limit, 0.0)

    # Reasonable upper bound – we never need to add more than a few
    # hundred units of total mass because the objective then tends back to 2.
    eps_limit = min(eps_limit, 0.5 * a.sum())

    if eps_limit <= 0:
        return a, _max_convolution_fft(a)

    # Golden‑section search on [0, eps_limit]
    phi = (math.sqrt(5) - 1) / 2  # 0.618...
    lo, hi = 0.0, eps_limit
    # initial points
    x1 = lo + (1 - phi) * (hi - lo)
    x2 = lo + phi * (hi - lo)

    def _objective(eps: float) -> float:
        conv = conv_aa + 2.0 * eps * conv_av + (eps * eps) * conv_vv
        max_conv = conv.max()
        sum_a = a.sum() + eps
        return 2.0 * n * max_conv / (sum_a * sum_a)

    f1 = _objective(x1)
    f2 = _objective(x2)

    # iterate until time runs out or enough precision reached
    itr = 0
    while time.time() < deadline and (hi - lo) > 1e-6:
        if f1 > f2:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + phi * (hi - lo)
            f2 = _objective(x2)
        else:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + (1 - phi) * (hi - lo)
            f1 = _objective(x1)
        itr += 1
        if itr % 30 == 0:
            # occasional early‑exit check
            pass

    eps_opt = (x1 + x2) / 2.0
    new_seq = a + eps_opt * v
    new_seq = np.clip(new_seq, 0.0, 1000.0)   # safety
    val = _max_convolution_fft(new_seq)
    return new_seq, val

def propose_candidate(budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the
    maximal entry of the self‑convolution.  Returns a plain Python list
    ready for `evaluate_sequence`.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem dimension
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        if budget_s >= 800:
            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Helper: fast evaluation (max convolution)
    # --------------------------------------------------------------
    def _eval(a: np.ndarray) -> float:
        return _max_convolution_fft(a)

    # --------------------------------------------------------------
    # Build a diverse pool of analytic candidates
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # Warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        prev = _project_to_simplex(prev, target_sum)
        candidates.append(prev)

    # Power‑law family
    for p in np.linspace(0.30, 0.90, 31):
        seq = _power_law_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        # jittered copy
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Exponential decay family
    for lam in np.linspace(4e-4, 0.015, 30):
        seq = _exponential_sequence(n, lam)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Sine‑power family
    for p in np.linspace(0.2, 2.5, 22):
        seq = _sin_power_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # U‑shape family (ends‑heavy)
    for p in np.linspace(0.10, 0.45, 16):
        seq = _u_shape_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Regularised arcsine‑like shape
    seq = _arcsine_like_sequence(n, eps=1e-6)
    seq = _project_to_simplex(seq, target_sum)
    candidates.append(seq)
    jitter = rng.normal(scale=target_sum * 0.004, size=n)
    seq_j = np.maximum(seq + jitter, 0.0)
    seq_j = _project_to_simplex(seq_j, target_sum)
    candidates.append(seq_j)

    # Random permutations of the first few promising candidates
    for base_seq in candidates[:8]:
        perm = base_seq.copy()
        rng.shuffle(perm)
        candidates.append(perm)

    # --------------------------------------------------------------
    # Quick evaluation & keep the best few
    # --------------------------------------------------------------
    scored = []
    for seq in candidates:

    scored.sort(key=lambda x: x[0])

    top_k = 8

    print(f"[{time.time():.2f}] Initial best score from pool = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Soft‑max GD on top candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        gd_time = remaining * 0.35

        per_candidate = gd_time / top_k
        for i in range(top_k):

            sub_dead = time.time() + min(per_candidate, gd_deadline - time.time())
            a_opt, val_opt = _annealed_gradient_descent_ext(
                seq0, sub_dead, target_sum,
                max_beta=3e5, step_factor=0.6, gamma=0.995
            )
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] GD improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase X – Sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_dead = time.time() + remaining * 0.25
        a_sg, val_sg = _subgradient_descent(
            best_seq, target_sum, sub_dead,
            step_factor=0.55, gamma=0.95
        )
        if val_sg < best_score:
            best_score, best_seq = val_sg, a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        mt_dead = time.time() + remaining * 0.30
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.6, end_temp=1e-5, low_frac=0.04
        )
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – Final polishing with a higher‑β GD
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        a_fin, val_fin = _annealed_gradient_descent_ext(
            best_seq,
            time.time() + remaining,
            target_sum,
            max_beta=5e5,
            step_factor=0.5,
            gamma=0.998,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase M – Mixing with auxiliary shapes (exact line search)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        mix_deadline = time.time() + remaining
        # two auxiliary shapes: uniform and a low‑mass U‑shape
        aux_shapes = []

        uniform = _uniform_sequence(n)
        aux_shapes.append(uniform)

        u_low = _u_shape_sequence(n, p=0.30)   # gentle U‑shape
        aux_shapes.append(u_low)

        for shape in aux_shapes:
            if time.time() >= mix_deadline:
                break
            mixed_seq, mixed_val = _mix_optimal(best_seq, shape, mix_deadline)
            if mixed_val < best_score:
                best_score, best_seq = mixed_val, mixed_seq.copy()
                print(f"[{time.time():.2f}] Mixing improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑process & final evaluation
    # --------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=float)
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]

    final_seq = final_arr.tolist()

    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in possible future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
