def _adam_optimize(start_a, target_sum, deadline):
    n = len(start_a)
    eps = 1e-12
    # initialize y
    y = np.log(start_a + eps)  # shape n
    # initialize Adam parameters
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # schedule for learning rate? we can set base_lr such that step size ~ target_sum * 0.1 maybe.

    # schedule for beta (softmax temperature)
    beta_temp = 0.5
    beta_max = 3e5  # similar to prior
    beta_mul = 1.5
    iter = 0
    best_a = start_a.copy()
    best_val = _max_convolution_fft(best_a)
    while time.time() < deadline:
        # compute grad w.r.t a via softmax
        grad_a, _ = _gradient_softmax(_softmax_a(y, target_sum), beta_temp)
        # compute prob = a / target_sum
        a = _softmax_a(y, target_sum)  # to compute prob
        prob = a / target_sum

        avg = np.dot(grad_a, prob)
        grad_y = target_sum * prob * (grad_a - avg)

        # Adam update
        iter += 1
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** iter)
        v_hat = v / (1 - beta2 ** iter)
        lr = 0.1 / (np.sqrt(v_hat) + eps_adam)  # per-parameter step
        # maybe also multiply by scale factor
        y = y - lr * m_hat

        # compute a for evaluation
        a = _softmax_a(y, target_sum)
        # compute true max conv
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={iter} β={beta_temp:.0f} best={best_val:.6f}")

        # schedule beta increase
        if iter % 500 == 0 and beta_temp < beta_max:
            beta_temp = min(beta_temp * beta_mul, beta_max)

        # occasional jitter
        if iter % 2000 == 0:
            rnd = np.random.default_rng()
            y += rnd.normal(scale=0.01, size=n)

    return best_a, best_val

"""
Optimisation of a non‑negative step function for the self‑convolution max‑norm.

The objective for a sequence `a` of length `n` is
    f(a) = 2·n·max(a*a) / (sum(a)**2)

For a fixed `n` the objective is invariant to scaling of `a`; we simply
enforce the ℓ¹‑norm `sum(a) = sqrt(2·n)` and minimise `max(a*a)`.
The algorithm combines several ideas:

1. **Parametric pool** – power‑law, exponential, sine‑power shapes and a few
   random symmetric candidates are generated, projected onto the simplex and
   quickly evaluated with an FFT based convolution. The best few are retained.

2. **Symmetric Adam optimisation** – the retained candidates are refined by
   Adam on a soft‑max surrogate of the max‑convolution.  The variables are
   the log‑softmax parameters `y` (so positivity and the ℓ¹‑constraint are
   satisfied automatically).  After each Adam update `y` is symmetrised
   (`y ← (y+y[::-1])/2`) to restrict the search to symmetric unimodal shapes,
   which are known to be close to optimal for this problem.

3. **Deterministic mass‑transfer** – a greedy annealing that moves mass from
   positions that contribute most to the current peak to low‑contributing
   positions.

4. **Final polishing** – a high‑β soft‑max gradient descent (momentum GD) and
   a short exact sub‑gradient descent on the true max‑convolution.

All phases respect the global time budget (default 1000 s).  The best sequence
found is stored in the global variable `height_sequence_1` for warm‑starts in
later runs and returned as a plain Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (unnormalised)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _random_symmetric_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random symmetric non‑negative sequence and project onto the simplex."""
    half = (n + 1) // 2
    vals = rng.random(half)  # uniform (0,1)
    vals = np.sort(vals)[::-1]  # decreasing for a nice unimodal shape
    if n % 2 == 0:
        seq = np.concatenate([vals, vals[::-1]])
    else:
        seq = np.concatenate([vals, vals[-2::-1]])  # middle element not duplicated
    return _project_to_simplex(seq, target_sum)


# ----------------------------------------------------------------------
# Soft‑max surrogate of the max‑convolution and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where grad has shape (n,) and corresponds to d/d a
    of the soft‑max loss.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max probabilities (log‑sum‑exp trick)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (rev(a) ⋆ p) restricted to the central n entries
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Soft‑max / Adam optimiser on the simplex (via log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   max_beta: float = 3e5,
                   beta_increase_interval: int = 500,
                   beta_multiplier: float = 1.5,
                   base_lr: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation of a.

    The parameters are `y` such that a = softmax(y) * target_sum.
    After each Adam update `y` is symmetrised (y ← (y + y[::-1]) / 2)
    to stay in the space of symmetric sequences.
    """
    n = start_a.shape[0]
    eps = 1e-14

    # initialise y from start_a (avoid log(0))
    y = np.log(start_a + eps)

    # Adam moments
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # soft‑max temperature schedule
    beta = 0.5
    it = 0
    best_a = start_a.copy()
    best_val = _max_convolution_fft(best_a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- compute current a (softmax) -----
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)          # stabilised
        prob = exp_y / np.sum(exp_y)        # this already sums to 1
        a = prob * target_sum

        # ----- gradient w.r.t a (soft‑max surrogate) -----
        grad_a, _ = _gradient_softmax(a, beta)

        # chain rule: ∂L/∂y = target_sum * prob * (grad_a - <grad_a, prob>)
        avg = np.dot(grad_a, prob)
        grad_y = target_sum * prob * (grad_a - avg)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        y -= step

        # ----- enforce symmetry on y -----
        y = (y + y[::-1]) * 0.5

        # ----- occasional jitter to escape plateaus -----
        if it % 2000 == 0:
            y += rng.normal(scale=0.01, size=n)

        # ----- update temperature -----
        if it % beta_increase_interval == 0 and beta < max_beta:
            beta = min(beta * beta_multiplier, max_beta)

        # ----- evaluate true objective periodically -----
        if it % 200 == 0:
            a_eval = _project_to_simplex(a, target_sum)   # just to be safe
            cur_val = _max_convolution_fft(a_eval)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_eval.copy()
                print(f"[{time.time():.2f}] Adam it={it} β={beta:.0f} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (same as in the previous pipeline)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it} new best={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Momentum GD on the soft‑max surrogate (used for a final polish)
# ----------------------------------------------------------------------
def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 5e5,
                                   step_factor: float = 0.5,
                                   gamma: float = 0.998) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it} β={beta:.0f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the maximum
    entry of its self‑convolution.  A combination of parametric pool,
    Adam optimisation on a soft‑max surrogate, deterministic mass‑transfer,
    and final polishing is used.  The best found sequence is returned as a
    list of Python floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives a richer search space)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        # adapt size to the available budget
        if budget_s >= 800:
            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Helper for fast evaluation (max of self‑convolution)
    # --------------------------------------------------------------
    def _eval(a: np.ndarray) -> float:
        return _max_convolution_fft(a)

    # --------------------------------------------------------------
    # Build a diversified pool of candidates
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # Warm‑start from any previous solution
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        candidates.append(prev)

    # Parametric families
    for p in np.linspace(0.35, 0.85, 31):
        seq = _power_law_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

    for lam in np.linspace(5e-4, 0.012, 25):
        seq = _exponential_sequence(n, lam)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

    for p in np.linspace(0.2, 2.0, 20):
        seq = _sin_power_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

    # Random symmetric seeds
    for _ in range(12):
        candidates.append(_random_symmetric_sequence(n, target_sum, rng))

    # Add jittered copies of the first few deterministic candidates
    for base in candidates[:6]:
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        copy = np.maximum(base + jitter, 0.0)
        copy = _project_to_simplex(copy, target_sum)
        candidates.append(copy)

    # --------------------------------------------------------------
    # Quick evaluation of the pool
    # --------------------------------------------------------------
    scored = []
    for seq in candidates:
        val = _eval(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])

    best_seq = scored[0][1].copy()
    best_score = scored[0][0]
    print(f"[{time.time():.2f}] Initial best from pool = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on the soft‑max surrogate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        top_k = min(4, len(scored))
        per_candidate = remaining * 0.35 / top_k
        for i in range(top_k):
            seq0 = scored[i][1]
            sub_dead = time.time() + min(per_candidate, deadline - time.time())
            a_opt, val_opt = _adam_optimize(seq0, target_sum, sub_dead)
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] Adam improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        mt_dead = time.time() + remaining * 0.30
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                    target_sum,
                                                    start_temp=0.6,
                                                    end_temp=1e-5,
                                                    low_frac=0.04)
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Final polishing with high‑β GD
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        a_fin, val_fin = _annealed_gradient_descent_ext(
            best_seq,
            time.time() + remaining,
            target_sum,
            max_beta=5e5,
            step_factor=0.5,
            gamma=0.998,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final GD improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑process: trim trailing near‑zeros and evaluate formally
    # --------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=float)
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:

    final_seq = final_arr.tolist()

    # Safety evaluation (may be omitted if unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
