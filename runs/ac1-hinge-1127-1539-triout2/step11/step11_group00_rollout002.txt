def propose_candidate(budget_s=1000):

    start = time.time()
    deadline = start + budget_s - 0.5

    # Choose n based on budget
    if budget_s >= 800: n = 20000
    elif budget_s >= 500: n = 15000
    else: n = 10000

    target_sum = np.sqrt(2.0 * n)

    # Initialize with known sequence if exists else power-law.
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"]), n)
    else:
        # power law with exponent 0.6
        init = np.arange(1, n+1)**(-0.6)
    init = init / init.sum() * target_sum

    # store best solution.
    best_a = init.copy()
    best_val = _score(best_a)

    # Phase A: Mirror descent (softmax) approx
    ph_a_deadline = time.time() + 0.12 * (deadline - time.time())
    a_a, val_a = _exponential_descent(best_a, target_sum, ph_a_deadline)
    if val_a < best_val: best_a, best_val = a_a, val_a

    # Phase B: additive subgradient descent with line-search
    ph_b_deadline = time.time() + 0.12 * (deadline - time.time())
    a_b, val_b = _additive_subgradient(best_a, target_sum, ph_b_deadline)
    if val_b < best_val: best_a, best_val = a_b, val_b

    # Phase C: population-based moves (random mass transfer + jitter)
    ph_c_deadline = time.time() + 0.20 * (deadline - time.time())
    a_c, val_c = _population_search(best_a, target_sum, ph_c_deadline)
    if val_c < best_val: best_a, best_val = a_c, val_c

    # Phase D: final mass transfer anneal with extended temperature
    ph_d_deadline = deadline
    a_d, val_d = _mass_transfer_anneal(best_a, target_sum, ph_d_deadline)
    if val_d < best_val: best_a, best_val = a_d, val_d

    # Clean up: remove trailing near-zero entries
    arr = np.array(best_a)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()
    # Evaluate final
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"Final evaluated score = {final_val}")
    except Exception as e:
        print(f"Error evaluating final seq: {e}")

    globals()["height_sequence_1"] = final_seq

    return final_seq

"""
Advanced mixed‑search for the step‑function benchmark.

The algorithm consists of five stages:

0. *Parameter sweep*: try a few power‑law shapes (a_i ∝ (i+1)^‑α) and keep the
   best as a warm start.
1. *Soft‑max mirror descent* (phase A) – fast reduction of the peak using a
   smooth approximation of the max‑convolution.
2. *Additive sub‑gradient descent* (phase B) – true sub‑gradient on the
   current peak with a decreasing step size and Euclidean projection onto the
   simplex Σa = √(2 n).
3. *Population‑based stochastic search* (phase C) – keep a small pool of
   candidates, apply random mass‑transfer mutations or convex combinations,
   and keep the best.
4. *Mass‑transfer simulated annealing* (phase D) – long‑run annealing that
   moves tiny amounts of mass from donors that heavily contribute to the
   current peak to low‑height receivers, with a linear cooling schedule.

All phases respect the global time budget (default 1000 s).  The final
sequence is projected onto the non‑negative simplex, trimmed of trailing
zeros and stored as ``height_sequence_1`` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max entry of the full self‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    For a normalized sum (≈√(2n)) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _powerlaw_warm_start(n: int, target_sum: float, deadline: float):
    """Try a few power‑law exponents and keep the best sequence."""
    best_a = None
    best_val = np.inf
    rng = np.random.default_rng()
    exponents = np.linspace(0.45, 0.85, 9)  # 0.45,0.50,…,0.85
    for alpha in exponents:
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=np.float64) ** (-alpha)
        a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # tiny jitter to avoid pathological alignment
        a += rng.normal(scale=target_sum * 1e-4, size=n)
        a = np.maximum(a, 0.0)
        a = a * (target_sum / a.sum())
    if best_a is None:          # fallback (should never happen)
        best_a = np.ones(n, dtype=np.float64) * (target_sum / n)
    print(f"[{time.time():.2f}] Power‑law start best score={best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------


def _additive_subgradient(a0: np.ndarray, target_sum: float, deadline: float):
    """Additive sub‑gradient descent with Euclidean projection onto the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # current peak
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient of the peak (2·a_{k−i} as in the original code)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # diminishing step size (≈ η₀ / √t)
        eta0 = target_sum * 0.12
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        a = a_new

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 6e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AddSubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _population_search(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Maintain a modest population of candidates.
    Mutations consist of random mass transfers and convex combinations.
    """
    pop_size = 5
    n = a0.shape[0]
    rng = np.random.default_rng()

    # initialise population: the warm start + random Dirichlet samples
    population = [a0.copy()]
    scores = [_score(a0)]
    for _ in range(pop_size - 1):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        population.append(rnd)
        scores.append(_score(rnd))

    best_idx = int(np.argmin(scores))
    best_a = population[best_idx].copy()
    best_val = scores[best_idx]

    it = 0
    while time.time() < deadline:
        # pick a parent (biased towards good fitness)
        probs = np.exp(-np.array(scores) / (np.mean(scores) + 1e-12))
        probs /= probs.sum()
        parent = population[rng.choice(pop_size, p=probs)]

        # mutation type
        if rng.random() < 0.6:
            # random mass‑transfer mutation (similar to anneal step but larger)
            donor = rng.integers(n)
            receiver = rng.integers(n)
            if donor == receiver:
                continue
            max_delta = min(parent[donor] * 0.06, target_sum * 0.002)
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            child = parent.copy()
            child[donor] -= delta
            child[receiver] += delta
        else:
            # convex combination of two random parents
            i, j = rng.choice(pop_size, size=2, replace=False)
            alpha = rng.random()
            child = alpha * population[i] + (1 - alpha) * population[j]

        # project back to the simplex
        child = np.maximum(child, 0.0)
        child = _project_to_simplex(child, target_sum)

        val = _score(child)

        # replace the worst individual if child is better
        worst_idx = int(np.argmax(scores))
        if val < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = val
            if val < best_val:
                best_val = val
                best_a = child.copy()
                print(f"[{time.time():.2f}] PopSearch it={it:5d} new best={best_val:.6f}")

        it += 1

    return best_a, best_val

def _mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """Long‑run annealing that moves tiny amounts of mass from “bad’’ to “good’’ indices."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.30                   # start temperature ≈30 % of current score

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – proportional to contribution (more contribution → higher chance)
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – prefers a small height
        inv = 1.0 / (a + 1e-12)
        inv /= inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by the temperature)
        max_delta = min(a[donor] * 0.06, target_sum * 0.002)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if time.time() - last_full > 7.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional tiny jitter (swap a tiny bit of mass)
        if it % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.004, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start: either from previous best or a quick power‑law sweep
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        init = init * (target_sum / init.sum())
    else:
        # cheap power‑law sweep (0 % of budget initialisation)
        wl_deadline = time.time() + 0.03 * (deadline - time.time())
        init, _ = _powerlaw_warm_start(n, target_sum, wl_deadline)

    best_seq, best_score = init, _score(init)
    print(f"[{time.time():.2f}] Initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent (≈10 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_dead = time.time() + 0.10 * (deadline - time.time())
        a_a, val_a = _exponential_descent(best_seq, target_sum, a_dead)
        if val_a < best_score:
            best_seq, best_score = a_a, val_a
        print(f"[{time.time():.2f}] After ExpDescent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – additive sub‑gradient descent on the true max‑convolution (≈12 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        b_dead = time.time() + 0.12 * (deadline - time.time())
        a_b, val_b = _additive_subgradient(best_seq, target_sum, b_dead)
        if val_b < best_score:
            best_seq, best_score = a_b, val_b
        print(f"[{time.time():.2f}] After AddSubGrad best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – population‑based stochastic search (≈20 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        c_dead = time.time() + 0.20 * (deadline - time.time())
        a_c, val_c = _population_search(best_seq, target_sum, c_dead)
        if val_c < best_score:
            best_seq, best_score = a_c, val_c
        print(f"[{time.time():.2f}] After population search best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – long simulated‑annealing refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_d, val_d = _mass_transfer_anneal(best_seq, target_sum, deadline)
        if val_d < best_score:
            best_seq, best_score = a_d, val_d
        print(f"[{time.time():.2f}] After annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
