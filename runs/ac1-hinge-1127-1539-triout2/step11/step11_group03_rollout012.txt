def _lp_iterative_optimization(initial_f, target_sum, time_deadline, t0=0.3, t_decay=0.99, max_iter=1000):
    f = initial_f.copy()
    best_f = f.copy()
    best_score = _score(f)
    t = t0
    iteration = 0
    while time.time() < time_deadline and iteration < max_iter:
        # compute M
        conv_f = _conv_fft(f)
        M = float(conv_f.max())
        # build A
        n = len(f)
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            start = i
            end = i + n
            if end > rows: end = rows
            length = end - start
            A[start:end, i] = f[:length]
        b_ub = np.full(rows, M, dtype=np.float64)
        c = -np.ones(n)
        # Solve LP
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                     bounds=(0, None), method='highs')
        if not res.success:
            print(f"LP failed at iteration {iteration}, exit loop")
            break
        b = res.x
        sum_b = b.sum()
        # if sum_b is extremely low, break
        if sum_b <= 1e-12:
            break
        g = target_sum * b / sum_b
        # combine
        f_new = (1 - t) * f + t * g
        # evaluate score
        score_new = _score(f_new)
        if score_new < best_score - 1e-9:
            best_score = score_new
            best_f = f_new.copy()
            print(f"[{time.time():.2f}] LP it={iteration} t={t:.4f} new best {best_score:.6f}")
        f = f_new
        # decay t
        t *= t_decay
        # optional jitter if no improvement over many iter
        iteration += 1
        # maybe random restart if stagnates
    return best_f, best_score

def _lp_phase(f, target_sum, time_deadline, t0=0.2, t_decay=0.98, max_iter=200):
    best_f = f.copy()
    best_score = _score(f)
    t = t0
    it = 0
    n = len(f)
    rows = 2*n - 1
    while time.time() < time_deadline and it < max_iter:
        # compute M
        conv_f = _conv_fft(f)
        M = float(conv_f.max())
        # Build A
        A = np.zeros((rows, n), dtype=np.float64)
        # fill columns
        for i in range(n):
            start = i
            end = i + n
            if end > rows:
                end = rows
            length = end - start
            A[start:end, i] = f[:length]
        b_ub = np.full(rows, M, dtype=np.float64)
        c = -np.ones(n)
        # Solve LP
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            # maybe break
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = target_sum * b / sum_b
        # combine
        f_new = (1 - t) * f + t * g
        # recompute score (max conv)
        score_new = _score(f_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_f = f_new.copy()
            print(f"[{time.time():.2f}] LP it={it} t={t:.4f} best={best_score:.6f}")
        # update f and t
        f = f_new
        t *= t_decay
        it += 1
    return best_f, best_score

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _annealed_gradient_descent(a0, deadline, target_sum, max_beta=1e6, step_factor=0.5, gamma=0.995):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1
    return best_a, best_val

"""
Hybrid optimizer with a small‑scale linear‑programming refinement.

1.  Build a modest (≈1‑2 k) step function and run a few LP
    iterations that shrink the maximal auto‑convolution while keeping the
    ℓ¹‑norm fixed.  This phase is cheap and often yields a shape that is
    already better than a plain power‑law.

2.  Interpolate the shape to a larger length (≈8‑12 k) and project onto the
    ℓ¹‑simplex (∑a = √(2·n)).  The larger problem is tackled by the already
    proven pipeline:
       – short projected‑gradient descents on a soft‑max approximation,
       – deterministic “mass‑transfer’’ refinements,
       – a targeted permutation annealing that only re‑orders the heights,
       – a final tiny mass‑transfer polish.

All numbers are forced to be non‑negative, the sum is always normalised
to √(2·n) and the final list is trimmed of trailing zeros before being
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy for the linear‑programming refinement
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _HAS_LINPROG = True
except Exception:               # pragma: no cover
    _HAS_LINPROG = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Objective (the benchmark score)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a)/(∑a)² – for normalised sum this is just max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the ℓ¹‑simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) sequence, renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam·i) sequence, renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old to new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest … ordering."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Same as _alternating_order but start on the right."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of soft‑max(max(conv(a,a))) w.r.t. a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Momentum projected GD with annealed soft‑max temperature."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy donor‑receiver moves with an exponential temperature schedule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Swap heavy and light contributors to the current peak."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Small‑scale LP refinement (the new ingredient)
# ----------------------------------------------------------------------


def _lp_refine_phase(f0: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     t0: float = 0.2,
                     t_decay: float = 0.98,
                     max_iter: int = 80) -> tuple[np.ndarray, float]:
    """
    Repeated linear‑programming step.
    At each iteration we maximise Σb subject to (f*b)_k ≤ M,
    then convexly mix the solution back into f with a small step size.
    """
    if not _HAS_LINPROG:
        return f0, _score(f0)

    f = f0.copy()
    best_f = f.copy()
    best_val = _score(f)
    t = t0
    n = f.shape[0]
    rows = 2 * n - 1
    it = 0

    while time.time() < deadline and it < max_iter:
        # current maximal auto‑convolution
        M = _max_convolution_fft(f)

        # build the convolution matrix A (rows × n)
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            start = i
            end = i + n
            if end > rows:
                end = rows
            length = end - start
            A[start:end, i] = f[:length]

        b_ub = np.full(rows, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σb  ⇔ minimise -Σb

        # solve the LP
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=(0, None),
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # if the solver fails we abort the LP phase
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # scale to the required ℓ¹‑norm
        g = target_sum * b / sum_b

        # convex update and re‑project (tiny numerical drift)
        f = (1.0 - t) * f + t * g
        f = _project_to_simplex(f, target_sum)

        val = _score(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[{time.time():.2f}] LP it={it:3d} t={t:.4f} score={best_val:.6f}")

        t *= t_decay
        it += 1

    return best_f, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 1. Small‑scale LP refinement (≈1‑2 k points)
    # ------------------------------------------------------------------
    n_small = 1200
    target_small = np.sqrt(2.0 * n_small)

    # a simple power‑law seed
    f_small = _power_law_sequence(n_small, 0.55, target_small)

    # give the LP phase up to ~180 s (or less if overall budget is small)
    lp_deadline = min(deadline - 5.0, start_time + 180.0)
    if lp_deadline > time.time():
        f_small, _ = _lp_refine_phase(f_small, target_small, lp_deadline,
                                      t0=0.25, t_decay=0.97, max_iter=70)

    # ------------------------------------------------------------------
    # 2. Upscale to the final problem size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and budget_s >= 800:
        n_big = 12000
    elif remaining > 15.0 and budget_s >= 500:
        n_big = 10000
    else:
        n_big = 8000

    target_big = np.sqrt(2.0 * n_big)
    seq_big = _resize_sequence(f_small, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # ------------------------------------------------------------------
    # 3. Build a modest pool of starting points for the gradient phase
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    init_pool = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big)
    ]

    # a few jittered variants to increase diversity
    for _ in range(4):
        jitter = rng.normal(scale=target_big * 0.01, size=n_big)
        seq_j = seq_big + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_big)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 4. Exploration via short projected‑gradient runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 10.0:
        # pick a seed (biased towards the current best)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_big * 0.004, size=n_big)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_big)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 22.0, deadline - 5.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_big,
            max_beta=2e6,
            step_factor=0.6,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        mt_dead = time.time() + min(6.0, (deadline - time.time()) * 0.15)
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_big,
            start_temp=0.5, end_temp=1e-4, low_frac=0.05
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final polishing with a high‑β gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        a_fin, val_fin = _annealed_gradient_descent(
            best_seq,
            deadline - 3.0,
            target_big,
            max_beta=5e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Targeted permutation search (re‑ordering only)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        perm_dead = time.time() + min(12.0, deadline - time.time() - 1.0)
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.12,
            end_temp=1e-5,
            top_frac=0.03,
            low_frac=0.03,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny final mass‑transfer pass
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        mt2_dead = time.time() + min(5.0, deadline - time.time() - 0.5)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq, mt2_dead, target_big,
            start_temp=0.3, end_temp=1e-5, low_frac=0.05
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = seq_big

    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Final evaluation (optional logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)        # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # make the sequence available for future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
