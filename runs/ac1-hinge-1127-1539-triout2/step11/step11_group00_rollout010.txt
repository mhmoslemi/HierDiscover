def _search_parametric_shape(n, target_sum, deadline):

    rng = np.random.default_rng()
    i = np.arange(n, dtype=np.float64)
    best_score = np.inf

    iter_count = 0
    while time.time() < deadline:
        # sample parameters
        p = rng.uniform(0.45, 0.65)  # exponent around 0.5
        lam = rng.uniform(0.0, 1e-4)  # exponential decay
        offset = rng.uniform(0.0, 5.0)  # offset

def _mass_transfer_anneal(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = best_val * 0.30

    while time.time() < deadline:
        # compute current max index
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # compute donor proportional to contributions
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx_range] * a[k_max - idx_range]
        contrib_sum = contrib.sum()
        if contrib_sum > 0:
            donor = rng.choice(idx_range, p=contrib / contrib_sum)
        else:
            donor = rng.integers(lo, hi + 1)

        # prepare receiver
        # indices not in [lo, hi] do not affect current peak
        all_idx = np.arange(n)
        out_mask = (all_idx < lo) | (all_idx > hi)
        if np.any(out_mask):
            # choose uniformly among out-of-range (or weighted by low a)
            # Use uniform
            receiver = rng.choice(all_idx[out_mask])
        else:
            # all indices affect peak, choose one with small future contribution
            pair_idx = k_max - all_idx  # may be out of bounds but all_idx within lo..hi ensures within bounds
            # actually for all_idx in [lo,hi], pair_idx in [0,n-1]
            pair_vals = a[pair_idx]
            # weights inversely proportional to product of a[idx] and pair_vals
            # to avoid division by zero, add epsilon
            eps = 1e-12
            w = 1.0 / ( (a + eps) * (pair_vals + eps) )
            # maybe also incorporate current a small benefit
            w = w / w.sum()
            receiver = rng.choice(all_idx, p=w)

        if donor == receiver:
            continue

        # size of transfer
        max_delta = min(a[donor] * 0.04, target_sum * 0.001)
        if max_delta <= 0:
            continue
        # temperature schedule
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - frac))
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        # update convolution efficiently as before
        b_new = b.copy()
        # update linear terms
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic term updates
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic recompute
        if time.time() - start > 10.0 and (time.time() - start) % 10.0 < 0.001: # approximate every 10 secs
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional jitter
        # we may skip jitter for simplicity.

    return best_a, best_val

"""
Optimisation of a non‑negative step function for the max‑convolution benchmark.

Main ideas
==========

*  A low‑dimensional parametric family  a_i = (i+γ)^{-p}·exp(-λ·i)
   is explored by random sampling.  The parameters (p, λ, γ) are
   tuned so that the normalised sum Σa_i = √(2 n) and the resulting
   objective 2·n·max(a*a)/Σa² (which equals max(a*a) for the normalisation)
   is small.

*  The best candidate from the parametric search is refined by a
   specialised simulated‑annealing “mass‑transfer’’ routine.
   The move set is identical to the previously used Phase D but the
   receiver is chosen more intelligently: indices that do **not**
   contribute to the current peak of the convolution are preferred,
   otherwise a probability inversely proportional to the product of the
   two terms that would increase the peak is used.

*  The algorithm automatically adapts the problem size `n` to the
   allotted time (larger `n` when more time is available) and stores
   the best sequence globally under the name `height_sequence_1` for
   warm‑starting future runs.

The whole search runs within the given budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Convolution utilities (FFT based)
# -------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).  For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# -------------------------------------------------------------
# Simple helpers
# -------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# -------------------------------------------------------------
# Phase 1 – random search in a 3‑parameter family
# -------------------------------------------------------------


def _search_parametric_shape(
    n: int, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Randomly explores a_i = (i+γ)^(-p)·exp(-λ·i)."""
    rng = np.random.default_rng()
    i = np.arange(n, dtype=np.float64)

    best_a = None
    best_val = np.inf
    it = 0

    while time.time() < deadline:
        # parameter ranges (tuned heuristically)
        p = rng.uniform(0.45, 0.62)          # around 0.5
        lam = rng.uniform(0.0, 1.0e-4)       # tiny exponential decay
        gamma = rng.uniform(0.0, 5.0)        # offset to soften the singularity

        a = (i + gamma) ** (-p) * np.exp(-lam * i)

        # normalise to the required L1‑norm
        s = a.sum()
        if s == 0.0:
            continue
        a = a * (target_sum / s)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] param‑search it={it:5d} "
                f"p={p:.4f} λ={lam:.2e} γ={gamma:.3f} score={best_val:.6f}"
            )
        it += 1

    return best_a, best_val


# -------------------------------------------------------------
# Phase 2 – mass‑transfer annealing (improved version)
# -------------------------------------------------------------


def _mass_transfer_anneal(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that repeatedly moves a tiny amount of mass
    from a “donor’’ (large contribution to the current peak) to a
    “receiver’’ that is unlikely to increase that peak.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                       # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature ≈ 30 % of the current score
    T0 = best_val * 0.30

    last_full = start
    it = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # locate current peak of the convolution
        # ------------------------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)

        # ------------------------------------------------------------------
        # donor – proportional to its contribution to the peak
        # ------------------------------------------------------------------
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # ------------------------------------------------------------------
        # receiver – prefer indices that do NOT affect the current peak
        # ------------------------------------------------------------------
        all_idx = np.arange(n, dtype=np.int64)
        out_of_range = (all_idx < lo) | (all_idx > hi)

        if np.any(out_of_range):
            receiver = rng.choice(all_idx[out_of_range])
        else:
            # every index influences the peak; choose one with a small
            # product a[j]·a[k_max‑j] (i.e. a low marginal effect)
            pair_idx = k_max - all_idx                      # always in‑range now
            pair_vals = a[pair_idx]
            eps = 1e-12
            w = 1.0 / ((a + eps) * (pair_vals + eps))       # inverse of the product
            w = w / w.sum()
            receiver = rng.choice(all_idx, p=w)

        if donor == receiver:
            continue

        # ------------------------------------------------------------------
        # transfer size (scaled by the temperature)
        # ------------------------------------------------------------------
        max_delta = min(a[donor] * 0.04, target_sum * 0.001)
        if max_delta <= 0.0:
            continue

        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - frac))

        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # exact O(n) update of the convolution after the move
        # ------------------------------------------------------------------
        b_new = b.copy()
        # linear terms: 2·a·Δ (donor removal, receiver addition)
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic terms (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved score={best_val:.6f}")

        # ------------------------------------------------------------------
        # periodic full recomputation to avoid numerical drift
        # ------------------------------------------------------------------
        if time.time() - last_full > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved score={best_val:.6f}")
            last_full = time.time()

        # ------------------------------------------------------------------
        # tiny random jitter every few thousand iterations (helps escape)
        # ------------------------------------------------------------------
        if it % 3000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.004, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ---------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # ---------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # ---------------------------------------------------------
    # Warm‑start from previously discovered construction if available
    # ---------------------------------------------------------
    best_a = None
    best_score = np.inf

    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.shape[0] != n:
            seq0 = _resize_sequence(seq0, n)
        seq0 = seq0 * (target_sum / seq0.sum())
        score0 = _score(seq0)
        best_a = seq0
        best_score = score0
        print(f"[{time.time():.2f}] warm‑start score={best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 1 – random search in the parametric family
    # ---------------------------------------------------------
    phase1_deadline = time.time() + 0.20 * (deadline - time.time())
    a_ps, score_ps = _search_parametric_shape(n, target_sum, phase1_deadline)
    if a_ps is not None and score_ps < best_score:
        best_a = a_ps
        best_score = score_ps
        print(f"[{time.time():.2f}] after param‑search best score={best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 2 – refined mass‑transfer simulated annealing
    # ---------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _mass_transfer_anneal(best_a, target_sum, deadline)
        if score_ref < best_score:
            best_a = a_ref
            best_score = score_ref
            print(f"[{time.time():.2f}] after anneal best score={best_score:.6f}")

    # ---------------------------------------------------------
    # Final clean‑up, evaluation and storage
    # ---------------------------------------------------------
    arr = np.array(best_a, dtype=np.float64)
    # drop trailing near‑zero entries
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
