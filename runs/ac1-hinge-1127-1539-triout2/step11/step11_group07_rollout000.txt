"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Strategy
========
* Work on several lengths (200 → 400 → 800 → 1600 → 3200 → 6400) and keep the
  best solution found so far.  The objective after the ℓ¹‑simplex projection
  (Σa = √(2 n)) equals the maximal entry of the self‑convolution, so we
  minimise ``max(a*a)``.
* For each length:
    – Initialise with a diverse family of random monotone shapes.
    – Run a few *soft‑max* Adam phases (smooth approximation of the max)
      with an exponentially increasing smoothing factor β.
    – Switch to a *sub‑gradient* Adam phase that uses the exact active set
      of maximal convolution entries.
    – Refine with a deterministic greedy hill‑climb that moves mass from the
      most harmful index to a more benign one.
    – Perform cheap random swaps while time permits.
* The whole routine respects the wall‑clock budget (default 1000 s).  It
  prints occasional progress messages and finally returns the best vector
  as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    The implementation simply rescales after clamping to non‑negative.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # uniform fallback
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT (returns length 2·n‑1 array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Random families for initial vectors (all projected onto the simplex)
# ----------------------------------------------------------------------


def _random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random non‑negative vector on Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.2, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise constant + exponential tail
        split = rng.integers(int(0.15 * n), int(0.45 * n))
        plateau = rng.uniform(0.8, 1.2) * target / n
        lam = rng.exponential(scale=0.02)
        a = np.concatenate([np.full(split, plateau),
                            np.exp(-lam * np.arange(n - split))])

    elif family == 5:                   # two‑segment linear decay
        split = rng.integers(int(0.2 * n), int(0.8 * n))
        left = rng.uniform(0.5, 1.5) * target / n * np.linspace(1.0, 0.5, split)
        right = rng.uniform(0.5, 1.5) * target / n * np.linspace(0.5, 0.1, n - split)
        a = np.concatenate([left, right])

    else:                               # near‑uniform jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex(a, target)


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to length new_n followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return _project_to_simplex(a_new, math.sqrt(2.0 * new_n))


# ----------------------------------------------------------------------
# Adam optimiser – smooth‑max surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size_fft = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size_fft)
    b = np.fft.irfft(a_fft * a_fft, n=size_fft)[: 2 * n - 1]

    # softmax probabilities over convolution entries
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev * p  (only the central n entries are needed)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    A_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(A_fft * P_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_softmax(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    The smoothing parameter β is doubled every few hundred steps.
    """
    a = a0.copy()
    n = a.size
    beta = 1.5
    beta_max = 1e7
    beta_factor = 2.0

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.12 * target_sum
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1

        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-5, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # increase smoothing
        if t % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_submax(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of
    max(a*a).  The active set (indices attaining the maximum) defines p.
    """
    a = a0.copy()
    n = a.size

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.18 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1

        # full convolution
        b = _full_convolution(a)
        max_b = b.max()
        # active mask (ties broken by tolerance)
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # gradient = 2·(a_rev ⋆ p)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

    return a

# Deterministic hill‑climb (greedy reduction of the current maximal entry)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly moves a small amount of mass from the
    most damaging position to a less damaging one.  Returns the best vector
    encountered and its maximal convolution value.
    """
    a = a0.copy()
    n = a.size
    target = math.sqrt(2.0 * n)

    b = _full_convolution(a)
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # locate the current maximal convolution index
        k = int(np.argmax(b))

        # the index pair (i, j) that contributes most to this entry:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        indices = np.arange(i_low, i_high + 1, dtype=int)
        contributions = a[indices] * a[k - indices]
        i = indices[np.argmax(contributions)]

        # choose a destination j that interacts weakly with the complement
        # of i in the same window
        counterpart = a[k - indices]          # a[j'] where j' = k - i'
        # sort counterpart increasingly
        order = np.argsort(counterpart)
        j = None
        for cand in indices[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # amount to relocate (a fraction of a[i])
        delta = a[i] * 0.7
        if delta <= 0.0:
            break

        # compute the updated convolution using a O(n) incremental update
        b_new = b.copy()
        # effect of decreasing a[i] by delta:
        b_new[i:i + n] -= 2.0 * delta * a
        # effect of increasing a[j] by delta:
        b_new[j:j + n] += 2.0 * delta * a

        # update the three self‑terms that change quadratically:
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            # accept the move
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a reduced step via binary search
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small random jitter to escape stagnation
                jitter = rng.normal(scale=target * 1e-6, size=n)
                a += jitter
                a = _project_to_simplex(a, target)
                b = _full_convolution(a)

    return best_a, best_score
