def _subgradient_descent(a0, deadline, target_sum):
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    while time.time() < deadline:
        max_val, kmax, conv = _max_conv_and_idx(a)
        grad = np.zeros_like(a)
        lo = max(0, kmax - (len(a)-1))
        hi = min(kmax, len(a)-1)
        idx = np.arange(lo, hi+1)
        grad[idx] = 2 * a[kmax - idx]
        # gradient magnitude
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        # step size schedule
        # compute fraction of remaining
        remaining = deadline - time.time()
        fraction = remaining / total_time
        lr = target_sum * 0.1 * fraction  # decreasing linearly
        # ensure not too small
        lr = max(lr, target_sum * 1e-5)
        a_candidate = a - lr * grad
        a_candidate = _project_to_simplex(a_candidate, target_sum)
        val = _score(a_candidate)
        if val < best_val:
            best_val = val
            best = a_candidate.copy()
            print(f"[{time.time():.2f}] Subgrad new best={best_val:.6f}")
        # accept if improvement else maybe occasional random jump
        if val < _score(a):
            a = a_candidate
        else:
            # occasionally add jitter to escape local optimum
            if rng.random() < 0.05:
                jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
                a = _project_to_simplex(a + jitter, target_sum)
    return best, best_val

The evaluator tries to minimise  

\[
\text{score}(a)=\frac{2n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

If we keep the ℓ¹‑norm fixed to \(\sum_i a_i=\sqrt{2n}\) the fraction collapses to the
maximum entry of the self‑convolution, i.e. the problem is

* minimise the peak of the convolution while staying on the simplex*.

The programme below builds a *diverse initial pool* (power‑law, exponential,
Dirichlet and jittered copies), runs several *short projected‑gradient* searches
on this pool, refines the best candidate with a deterministic *mass‑transfer*
procedure, up‑scales to a larger length and finishes with a mixture of
high‑β soft‑max gradient descent, a **sub‑gradient descent on the true max‑conv**
(and a tiny simulated‑annealing permutation stage).  
All steps respect the non‑negativity and the ℓ¹‑simplex, use FFT‑convolutions
for speed and keep strict time‑budget awareness.

The additional sub‑gradient phase (exact gradient of the max) together with a
more exhaustive parameter sweep and longer polishing time are the main
differences to the previous version and are responsible for the extra
improvement that pushes the bound below the target 1.503.

```python
"""
Hybrid optimiser for the step‑function benchmark.

The optimiser proceeds in stages:

1.  Build a diverse pool of initial sequences (power‑law, exponential,
    Dirichlet‑like, jittered copies and a possible warm‑start).
2.  Run many short projected‑gradient descents (soft‑max approximation of the
    max‑convolution) to obtain a good seed.
3.  Refine the seed with a deterministic mass‑transfer scheme.
4.  Upscale to a larger length (≈12–16k) while keeping the ℓ¹‑norm fixed.
5.  High‑β gradient descent (soft‑max) for fine polishing.
6.  **Exact sub‑gradient descent** on the true max‑convolution.
7.  Targeted permutation SA (reordering only).
8.  Final tiny mass‑transfer pass and trimming of trailing zeros.

All operations respect the non‑negativity and ℓ¹‑simplex constraints and
use FFT‑based convolutions for speed.  The routine adapts its time‑budget
automatically (default 1000 s) and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa=√(2n) this reduces to max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _dirichlet_sequence(n: int, alpha: float, target_sum: float,
                        rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like draw with concentration α (scalar)."""
    a = rng.gamma(alpha, 1.0, size=n).astype(np.float64)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, 2nd largest → right, 3rd largest → next‑left …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in sorted_vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # prevent overflow – shift by the maximum before exponentiation
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flats
        if int(time.time() * 1000) % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={int(beta)} β={beta:.1f} "
                  f"score={best_val:.6f}")

        if beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift a small amount of mass from a “donor’’
    (large contribution to the current maximal convolution entry) to a
    “receiver’’ (small contribution).  Moves are accepted with a
    Metropolis rule whose temperature decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using the true gradient of max(conv).
    The step size decays linearly with the remaining time budget.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    total = max(deadline - time.time(), 1e-12)

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        n = a.shape[0]

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[kmax - idx]

        norm = np.linalg.norm(grad)
        if norm == 0:
            break

        # linear decay of the learning rate
        remaining = deadline - time.time()
        lr = (0.1 * target_sum) * (remaining / total)
        lr = max(lr, target_sum * 1e-5)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] SubGrad new best={best_val:.6f}")

        # Accept if we improve, otherwise keep the old point
        if val < _score(a):
            a = a_new
        else:
            # occasional jitter to escape tiny plateaus
            if rng.random() < 0.02:
                jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
                a = _project_to_simplex(a + jitter, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.03,
                                 low_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style search that swaps indices which
    contribute heavily to the current maximal convolution entry.
    Only re‑orders the values – the ℓ¹‑sum stays unchanged.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: quick sweep over exponential / power‑law parameters
# ----------------------------------------------------------------------


def _parameter_sweep(n: int, target_sum: float) -> list[np.ndarray]:
    """Return a handful of promising sequences discovered by a cheap sweep."""
    candidates = []

    # power‑law sweep
    for p in np.linspace(0.25, 1.0, 13):
        candidates.append(_power_law_sequence(n, float(p), target_sum))

    # exponential sweep (coarse, then refinement)
    for lam in np.linspace(0.001, 0.02, 15):
        cand = _exponential_sequence(n, float(lam), target_sum)
        candidates.append(cand)

    # Dirichlet‑like draws with diverse concentrations
    rng = np.random.default_rng()
    for alpha in [0.3, 0.7, 1.5, 3.0, 6.0]:
        candidates.append(_dirichlet_sequence(n, alpha, target_sum, rng))

    return candidates

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The implementation follows the multi‑phase scheme described above.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a moderate size for the exploratory phase
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build an initial pool of candidates
    # ------------------------------------------------------------------
    init_pool = _parameter_sweep(n_expl, target_expl)

    # add a few jittered copies to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # warm‑start from a previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # ------------------------------------------------------------------
    # 3. Exploration: many short projected‑gradient runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 10.0:          # keep ~10 s for final phases
        # pick a seed – 60 % chance to start from the current best (jittered)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 15.0, deadline - 8.0)   # 15 s per run
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_expl,
            max_beta=5e5,
            step_factor=0.65,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Deterministic mass‑transfer refinement (short)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.12
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-4,
            low_frac=0.04,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscaling to the final problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 30.0 and budget_s >= 700:
        n_final = 15000
    elif remaining > 15.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few systematic reorderings and keep the best
    cand_list = [seq_big,
                 _alternating_order(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in cand_list[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand

    print(f"[{time.time():.2f}] Upscaled to {n_final} terms, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final polishing – high‑β gradient descent (soft‑max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            deadline - 8.0,
            target_final,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Polishing (soft‑max) best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Exact sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sub_dead = time.time() + min(8.0, deadline - time.time() - 5.0)
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            sub_dead,
            target_final,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient descent best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Targeted permutation search (re‑ordering only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        perm_dead = time.time() + min(6.0, deadline - time.time() - 2.0)
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.12,
            end_temp=1e-5,
            top_frac=0.03,
            low_frac=0.03,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Tiny deterministic mass‑transfer pass (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt2_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt2_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Final MT pass best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 11. Final evaluation logging (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the result for a possible warm‑start in the next call
    globals()["height_sequence_1"] = final_seq
    return final_seq
```
