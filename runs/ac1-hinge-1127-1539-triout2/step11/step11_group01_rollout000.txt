"""
Propose a step‑function (non‑negative sequence) with a minimal max‑convolution.

Algorithm overview
-------------------
1. **Problem size** – choose `n` based on the time budget (larger *n* gives
   more degrees of freedom, at the cost of slower iterations).

2. **Initialisation** –  
   * If a previous solution (`height_sequence_1`) is available, interpolate it
     to the current length.  
   * Otherwise perform a fast parametric sweep over several families:
       – pure power‑law `a_i ∝ i^{-p}`
       – power‑law with exponential tail `a_i ∝ i^{-p}·exp(-q·i/n)`
       – sinusoidally modulated power‑law `a_i ∝ i^{-p}(1+r·sin(2πi/n))`  
     The best candidate is kept.

3. **Projected Adam gradient descent** – optimise a smooth soft‑max
   approximation of the max‑convolution.  The optimizer works on the
   simplex `{a_i ≥ 0, Σ a_i = √(2n)}` and uses an increasing temperature
   parameter `β` (up to 1e6) to get ever closer to the true maximum.

4. **Aggressive deterministic mass‑transfer** – repeatedly locate the
   current peak of the convolution and move exactly the amount of mass
   needed (subject to non‑negativity) from a “donor” index that contributes
   strongly to the peak to a “receiver” that contributes little.  The move
   is computed analytically, i.e.  `Δ = (M₁-M₂) / (2·(a_{k‑i}−a_{k‑j}))`,
   where `M₁` is the current maximal entry and `M₂` the second‑largest.
   This step is far more aggressive than the previous 0.7‑fraction cap.

5. **Escaping plateaus** – a cheap global random swap phase is run
   intermittently.

6. **Final polish** – a short Adam run with a very large `β` (≈1e7)
   refines the solution.

All steps respect the overall time budget (default 1000 s).  The best
sequence found is truncated to remove trailing near‑zero entries and
returned as a plain Python list.

The objective after normalising the sum to `√(2n)` is simply the maximal
convolution entry, so minimising `max(a*a)` directly minimises the score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective value (for a given normalisation Σa = √(2n) the score equals
    max(a*a)).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = S}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the cheap pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_modulated_sequence(n: int, p: float, r: float,
                            target_sum: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p) * (1.0 + r * np.sin(2.0 * np.pi * i / n))
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where grad is the gradient of the soft‑max of the
    convolution w.r.t. a, and conv is the raw convolution vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)                # soft‑max probabilities

    # conv_rev = conv of reversed a with p (gives sub‑gradient)
    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # only the first n entries matter
    return grad, conv


# ----------------------------------------------------------------------
# Projected Adam gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 1e6,
                           step_factor: float = 0.7,
                           gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the soft‑max approximation of
    the max‑convolution.  `beta` is increased every few hundred iterations
    up to `max_beta` to tighten the approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    inc_every = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)

        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % inc_every == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(start_vec: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Multiple short Adam runs with random perturbations.
    """
    best_a = start_vec.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=1e6,
                                         step_factor=0.7,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_aggressive(a0: np.ndarray,
                                           deadline: float,
                                           target_sum: float,
                                           tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    For the current peak index `k`, we move exactly the amount of mass
    required (subject to the donor's available mass) from a donor index
    with large partner value a_{k‑i} to a receiver with small partner value.
    This version does **not** artificially limit the transferred mass to
    0.7·donor – we use the full analytically required amount (capped only
    by the donor's remaining mass).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-15)[0]

        # second largest value (ignore all current maximal entries)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                       # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                 # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor: largest partner among indices with positive mass
        donor_local = np.argmax(partner * donor_mask)
        donor = idx[donor_local]

        # receiver: smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver_local = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver_local = r_loc
                break
        if receiver_local is None:
            break

        partner_donor = partner[donor_local]
        receiver = idx[receiver_local]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed)   # move at most what we have
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Aggressive DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic swaps: move a random fraction of a large entry to a
    small entry, then re‑project onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal max‑convolution.
    The routine respects the supplied time budget (seconds) and returns
    the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------- choose problem size ------------------------------------------------
    if budget_s >= 900:
        n = 18000
    elif budget_s >= 600:
        n = 14000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scale‑invariant)

    # ------- initial candidate ---------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # (a) use previous solution if available, interpolated to current length
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            seq = np.interp(new_x, old_x, prev)
        else:
            seq = prev.copy()
        seq = _project_to_simplex(seq, target_sum)
        val = _score(seq)
        best_seq = seq
        best_score = val

    # (b) parametric sweeps (power‑law, exp‑tail, sinusoid modulated)
    p_grid = np.linspace(0.30, 0.85, 21)          # fewer points → faster
    q_grid = np.linspace(0.0, 6.0, 21)
    r_grid = np.linspace(0.0, 0.18, 9)

    # pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # sinusoid‑modulated power‑law
    for p in p_grid:
        for r in r_grid:
            seq = _sin_modulated_sequence(n, float(p), float(r), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # small jittered population around current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric search best score = {best_score:.6f}")

    # ------- Phase A: Adam‑style projected gradient descent with restarts -------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

            print(f"[{time.time():.2f}] Aggressive DMT improved best = {best_score:.6f}")

    globals()["height_sequence_2"] = final_seq

    return final_seq
