"""
Low‑max‑convolution step‑function search.

The algorithm consists of four stages:

0.  Fast parametric scouting – a few thousand candidates are generated from
    several families (pure power‑law, power‑law+exponential tail and a
    two‑piece power‑law with an optional exponential factor).  The best
    candidate is kept.

A.  Projected Adam gradient descent on a soft‑max approximation of the
    max‑convolution.  Multiple short restarts with random jitter are used.

B.  Two deterministic mass‑transfer refinements:
    – the original “Exact DMT2” routine.
    – a more exhaustive version that tries several donor/receiver pairs
      per iteration in order to escape situations where the simple rule
      cannot move any mass.

C.  Simulated‑annealing exploration – random mass moves are accepted
    with a Metropolis rule whose temperature decays slowly.  The move is
    evaluated exactly (full FFT convolution) but only a few thousand
    proposals are made, which fits easily into the remaining budget.

Finally the best sequence is trimmed of trailing near‑zero entries and
returned.  All steps respect the supplied time budget (default 1000 s)
and only non‑negative numbers are produced.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective used by the competition – after the standard normalisation it equals max(a*a)."""
    # sum is kept at sqrt(2·n) by construction, therefore the returned value is the max‑convolution.
    max_b = _max_convolution_fft(a)
    return float(max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _two_piece_power_law_sequence(
    n: int,
    p1: float,
    p2: float,
    cutoff: int,
    q: float,
    target_sum: float,
) -> np.ndarray:
    """Early indices use exponent p1, later ones use p2 (optionally with an exponential tail)."""
    i = np.arange(n, dtype=np.float64)
    a = np.empty_like(i)
    early = i < cutoff
    # first segment
    a[early] = (i[early] + 1) ** (-p1)
    # second segment – continuous at the junction
    scale = (cutoff + 1) ** (-p1)
    a[~early] = scale * ((i[~early] + 1) / (cutoff + 1)) ** (-p2)
    if q != 0.0:
        a *= np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑style projected GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the smooth max = softmax(β·c) where c = a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    c = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = np.max(c)
    exps = np.exp(beta * (c - max_c))
    p = exps / np.sum(exps)                     # soft‑max probabilities

    # derivative of c w.r.t a: 2 * (a * reversed(p))   (see derivation in analysis)
    a_rev = a[::-1]
    Pr_fft = rfft(p, n=size)
    Ar_fft = rfft(a_rev, n=size)
    conv_rev = irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, c


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            # small jitter to escape flat regions
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Run a few short Adam restarts and keep the best result."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (two variants)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray, deadline: float, target_sum: float, tol: float = 1e-12
) -> tuple[np.ndarray, float]:
    """Greedy reduction of the max‑convolution using a single donor/receiver pair."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]  # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver with minimal partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}"
            )
        it += 1

    return best_a, best_score


def _deterministic_mass_transfer_exhaustive(
    a0: np.ndarray, deadline: float, target_sum: float, tol: float = 1e-12
) -> tuple[np.ndarray, float]:
    """
    Same goal as the simple DMT but tries several donor/receiver pairs per iteration.
    It is still cheap (O(n log n) per iteration) and can break dead‑locks that
    the original routine encounters.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # second largest value
        conv2 = conv.copy()
        conv2[conv == max_val] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick one peak index
        k = np.flatnonzero(conv == max_val)[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        # donor candidates – top‑K by contribution a_i * partner_i
        contrib = a[idx] * partner
        topk = min(30, idx.size)
        donor_order = np.argsort(-contrib)[:topk]
        # receiver candidates – bottom‑K by partner
        recv_order = np.argsort(partner)[:topk]

        moved = False
        for di in donor_order:
            donor = idx[di]
            if a[donor] <= 0.0:
                continue
            p_donor = partner[di]
            for rj in recv_order:
                receiver = idx[rj]
                if receiver == donor:
                    continue
                p_recv = partner[rj]
                denom = 2.0 * (p_donor - p_recv)
                if denom <= 0.0:
                    continue
                delta_needed = (max_val - second_max) / denom
                max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
                if max_delta <= 0.0:
                    continue
                a[donor] -= max_delta
                a[receiver] += max_delta
                moved = True
                break
            if moved:
                break

        if not moved:
            # as a last resort try a random (donor,receiver) pair
            donor = rng.choice(idx[a[idx] > 0])
            receiver = rng.choice(idx)
            if donor != receiver:
                p_donor = partner[np.where(idx == donor)[0][0]]
                p_recv = partner[np.where(idx == receiver)[0][0]]
                denom = 2.0 * (p_donor - p_recv)
                if denom > 0.0:
                    delta_needed = (max_val - second_max) / denom
                    max_delta = min(a[donor] * 0.5, delta_needed * 0.95)
                    if max_delta > 0.0:
                        a[donor] -= max_delta
                        a[receiver] += max_delta
                        moved = True

        if not moved:
            # cannot improve this peak any more
            break

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Exhaustive DMT it={it:5d} score={best_score:.6f}"
            )
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing exploration
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    init_temp: float = 0.05,
    cooling: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑style annealing that moves a small amount of mass
    from a high‑value index to a low‑value index.  The acceptance rule
    uses the true max‑convolution (no surrogate), so it can escape
    local minima that the deterministic steps cannot.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    T = init_temp * cur_score  # temperature proportional to current objective

    # Pre‑compute sets of high/low indices (updated lazily)
    while time.time() < deadline:
        # pick donor from the top fraction
        top_fraction = 0.02
        low_fraction = 0.02
        k_top = max(1, int(top_fraction * n))
        k_low = max(1, int(low_fraction * n))

        donor = np.argsort(-a)[rng.integers(k_top)]
        receiver = np.argsort(a)[rng.integers(k_low)]
        if donor == receiver:
            continue

        max_delta = a[donor] * 0.08
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        # sum is preserved, non‑negativity guaranteed by construction
        new_score = _score(a_new)
        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time():.2f}] SA new best={best_score:.6f}  T={T:.4e}"
                )
        T *= cooling
        if T < 1e-9:
            T = init_temp * cur_score  # occasional reheating to keep moves possible

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal max‑convolution.
    The routine respects the given time budget and returns the best sequence
    (as a plain Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Choose problem size – larger size gives more freedom but costs a bit more time.
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)  # Σa is fixed to this value (scale‑invariant)

    # --------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (including a 2‑piece family)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # (a) pure power‑law
    for p in np.linspace(0.30, 0.85, 31):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in np.linspace(0.30, 0.85, 23):
        for q in np.linspace(0.0, 6.0, 23):
            seq = _exp_power_law_sequence(
                n, float(p), float(q), target_sum
            )
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) two‑piece power‑law with optional exponential factor
    for _ in range(1500):
        p1 = rng.uniform(0.20, 0.65)
        p2 = rng.uniform(0.65, 1.30)
        cutoff = rng.integers(int(0.05 * n), int(0.45 * n))
        q = rng.uniform(0.0, 4.0)
        seq = _two_piece_power_law_sequence(
            n, p1, p2, int(cutoff), q, target_sum
        )
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (d) reuse a previously found construction if available
    if "height_sequence_1" in globals():
        prev = np.array(
            globals()["height_sequence_1"], dtype=np.float64
        )
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (e) small jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(
        f"[{time.time():.2f}] Initial best after parametric phase = {best_score:.6f}"
    )

    # --------------------------------------------------------------
    # Phase A – Adam gradient descent with restarts
    # --------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.28
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.012, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(
            start_vec, target_sum, gd_deadline
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer refinements
    # --------------------------------------------------------------
    while time.time() < deadline - 12.0:
        # first variant (original)
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(
                f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}"
            )
        # second, more exhaustive variant
        a_ref2, val_ref2 = _deterministic_mass_transfer_exhaustive(
            best_seq, deadline, target_sum
        )
        if val_ref2 < best_score:
            best_score = val_ref2
            best_seq = a_ref2.copy()
            print(
                f"[{time.time():.2f}] Exhaustive DMT improved best = {best_score:.6f}"
            )
        if time.time() > deadline - 12.0:
            break

    # --------------------------------------------------------------
    # Phase C – Simulated‑annealing exploration
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            deadline,
            target_sum,
            init_temp=0.04,
            cooling=0.9996,
        )
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(
                f"[{time.time():.2f}] SA improved best = {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase D – Final polish with a high‑β Adam run
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(
                f"[{time.time():.2f}] Final polish best = {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
