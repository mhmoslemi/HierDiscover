def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    while time.time() < deadline:
        # compute b via FFT
        b_full = _conv_fft(a)    # length 2n-1
        max_val = b_full.max()
        # early exit if max_val is close to second max (can't reduce)
        # but we continue
        k = int(b_full.argmax())
        # compute gradient g: size n
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices i in [lo, hi], partner index = k - i
        # g[i] = 2 * a[k-i]
        # we can fill as:
        i_range = np.arange(lo, hi+1)
        g[i_range] = 2.0 * a[k - i_range]
        # step size
        step = target_sum / (np.sqrt(it+1) + 1e-12) * 0.5   # tune factor
        a = a - step * g
        # projection
        a = _project_to_simplex(a, target_sum)
        # maybe clip small negative to zero (project already)
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_score:.6f}")
        it += 1
```

"""
Propose a step‑function with a very low max‑convolution value.

Algorithm outline
-----------------
1. **Problem size** – pick n according to the time budget (larger n gives
   more freedom, but stays well within the 1000 s budget).

2. **Fast parametric pre‑search** – explore a rich family of
   power‑law / exponential / piece‑wise power‑law shapes (≈3‑4 k candidates)
   and keep the best sequence (all are projected onto the simplex
   Σa = √(2 n)).

3. **Sub‑gradient descent on the max‑convolution** – the objective
   `max(a*a)` is convex; we minimise it by a classic sub‑gradient method:
   at each iteration the sub‑gradient is computed by a convolution of the
   current sequence with the indicator of the maximal convolution entries.
   A decreasing step‑size and Euclidean projection onto the simplex keep
   the iterates feasible.

4. **Projected Adam (soft‑max) optimisation** – a short Adam run on the
   smooth soft‑max approximation of the max‑convolution (the code from the
   previous state‑of‑the‑art is kept, but it now starts from the improved
   sub‑gradient point).

5. **Exact deterministic mass‑transfer** – greedy reduction of the
   current peak (identical to the former `DMT2` step).

6. **Global random mass‑transfer** – occasional random moves to escape
   plateaus.

7. **Final high‑β polishing** – a last Adam run with a very large soft‑max
   temperature.

The routine respects the global 1000‑second budget and returns the best
sequence found (trailing near‑zero entries are trimmed).  All numbers are
non‑negative, and the objective value equals the maximal convolution entry
because the sequence is kept normalised to Σa = √(2 n).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value: 2·n·max(b) / (sum(a)²). For Σa = √(2n) this equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families used in the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _piecewise_power_law_sequence(
    n: int, p1: float, p2: float, r: float, target_sum: float
) -> np.ndarray:
    """Two‑segment power‑law: first r·n entries use p1, the rest use p2."""
    cut = int(np.clip(r, 0.0, 1.0) * n)
    cut = max(1, min(cut, n - 1))
    a = np.empty(n, dtype=np.float64)
    a[:cut] = (np.arange(1, cut + 1, dtype=np.float64) ** (-p1))
    a[cut:] = (np.arange(1, n - cut + 1, dtype=np.float64) ** (-p2))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (convex optimisation)
# ----------------------------------------------------------------------


def _subgradient(a: np.ndarray):
    """
    Returns a sub‑gradient of  max(b)  where  b = a*a .
    The gradient is the average of the gradients of all maximal entries.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    # full convolution
    b = _conv_fft(a)                     # length 2n‑1
    max_b = b.max()
    # indicator of maximal entries (tolerance to avoid floating errors)
    w = np.zeros_like(b, dtype=np.float64)
    w[np.where(b >= max_b - 1e-12)[0]] = 1.0
    cnt = w.sum()
    # reverse a for convolution with w
    a_rev = a[::-1]
    A_rev_fft = np.fft.rfft(a_rev, n=size)
    W_fft = np.fft.rfft(w, n=size)
    grad_full = np.fft.irfft(A_rev_fft * W_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    if cnt > 1:
        grad /= cnt
    return grad, max_b


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Simple diminishing‑step sub‑gradient method on max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    # constant factor controlling the initial step magnitude
    step_const = target_sum * 0.05

    while time.time() < deadline:
        grad, _ = _subgradient(a)
        norm = np.linalg.norm(grad) + 1e-12
        step = step_const / (np.sqrt(it + 1) * norm)
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

# Exact deterministic mass‑transfer (improved version)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner coefficients a_{k‑i}
        partner = a[k - idx]               # shape == idx.shape

        # donor must have positive mass
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        # coefficients for the chosen donor / receiver
        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        # amount required to bring the peak down to the second‑largest value
        delta_needed = (max_val - second_max) / denom

        # never move more than a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score

def _parametric_presearch(n: int,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan a diverse family of analytic shapes (power‑law,
    power‑law + exponential, piecewise power‑law) and keep the best.
    """
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # modest deterministic grids
    p_grid = np.linspace(0.30, 0.85, 21)      # 21 values
    q_grid = np.linspace(0.0, 6.0, 21)       # 21 values
    r_grid = np.linspace(0.2, 0.8, 7)        # 7 values

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) piecewise power‑law
    for p1 in p_grid:
        for p2 in p_grid:
            for r in r_grid:
                seq = _piecewise_power_law_sequence(n, float(p1), float(p2), float(r), target_sum)
                val = _score(seq)
                if val < best_score:
                    best_score = val
                    best_seq = seq.copy()

    # (d) random draws (limited by remaining time)
    while time.time() < deadline - 5.0:
        p = rng.uniform(0.30, 0.85)
        q = rng.uniform(0.0, 6.0)
        r = rng.uniform(0.2, 0.8)
        typ = rng.integers(0, 3)
        if typ == 0:
            seq = _power_law_sequence(n, p, target_sum)
        elif typ == 1:
            seq = _exp_power_law_sequence(n, p, q, target_sum)
        else:
            p2 = rng.uniform(0.30, 0.85)
            seq = _piecewise_power_law_sequence(n, p, p2, r, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
