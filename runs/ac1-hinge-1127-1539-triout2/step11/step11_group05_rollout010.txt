def _adam_descent(initial, deadline, target_sum, max_beta=3e5, beta_schedule_factor=2.0, beta_start=0.5, step0=0.1, epsilon=1e-8):
    a = _project_to_simplex(initial.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    beta = beta_start
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 * (target_sum) / (np.sqrt(v_hat) + epsilon)
        a = a - step * m_hat
        # projection
        a = _project_to_simplex(a, target_sum)
        # update best
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # beta update
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * beta_schedule_factor, max_beta)
    return best_a, best_val

"""
Propose a step‑function sequence that (approximately) minimises the
maximum entry of its self‑convolution.

Algorithm
---------
1. Choose a problem size `n` (larger when the time budget allows) and the
   ℓ¹‑norm target Σa = √(2 n).
2. Build a diverse pool of parametric shapes (power‑law, exponential,
   sine‑power) together with a few Dirichlet‑sampled random vectors.
   Every candidate is projected onto the simplex {x≥0, Σx=target}.
3. Keep the best few candidates (by the fast FFT max‑convolution) as
   seeds.
4. Refine each seed with several optimisation phases, each limited by the
   remaining time budget:
   * **Adam‑style GD** on a soft‑max approximation of the max‑convolution,
     with an annealed β (inverse temperature) – fast and globally
     explorative.
   * **Exact sub‑gradient descent** on the true max entry – a local
     improvement step.
   * **Deterministic mass‑transfer** that moves mass from the pair that
     contributes most to the current peak to a low‑contributing pair.
   * **Random annealing** (simulated‑annealing style) for final fine‑tuning.
   * One final short Adam polishing.
5. Trim trailing near‑zero entries, store the result globally for warm‑starts,
   and return it as a plain Python list.

All heavy operations use FFT (O(n log n)) and the code constantly checks the
deadline, guaranteeing a return before the 1000 s budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Unnormalised power‑law (i+1)⁻ᵖ."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Unnormalised exponential decay e^{‑λ·i}."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    """Unnormalised (sin(π·(i+0.5)/n))ᵖ."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv). Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (rev(a) ⋆ p) restricted to the central n entries
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style gradient descent on the soft‑max objective
# ----------------------------------------------------------------------
def _adam_descent(initial: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  max_beta: float = 3e5,
                  beta_start: float = 0.5,
                  beta_factor: float = 2.0,
                  step0: float = 0.05,
                  epsilon: float = 1e-8) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of max‑convolution."""
    a = _project_to_simplex(initial.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    t = 0
    beta = beta_start
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 * m_hat / (np.sqrt(v_hat) + epsilon)
        a -= step
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} best={best_val:.6f}")

        if t % 250 == 0 and beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.95,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient: g[i] = 2 * a[kmax - i] when valid
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------
def _deterministic_mass_transfer_refine(a0: np.ndarray,
                                        deadline: float,
                                        target_sum: float,
                                        low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """Greedy mass transfer that explicitly reduces the current peak."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partners = kmax - idx
        w = a[idx] * a[partners]

        donor = idx[np.argmax(w)]
        # pick a low‑contributing index among the worst few
        low_cnt = max(1, int(len(idx) * low_frac))
        receiver = rng.choice(idx[np.argsort(w)[:low_cnt]])

        if donor == receiver:
            it += 1
            continue

        # bound on the transferable mass
        delta_max = a[donor] * 0.5
        diff = a[partners[np.argmax(w)]] - a[partners[np.argmin(w)]]
        if diff > 0:
            delta_max = min(delta_max, diff * 0.25)
        delta = delta_max * 0.5
        if delta <= 0:
            it += 1
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # projection is unnecessary – we kept the total sum and stayed non‑negative
        new_val = _max_convolution_fft(a_new)

        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            a = a_new
            print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_val:.6f}")
        else:
            # occasionally accept a worsening move to escape plateaus
            if rng.random() < 0.001:
                a = a_new

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            cur = _max_convolution_fft(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT jitter improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Randomised mass‑transfer simulated annealing
# ----------------------------------------------------------------------
def _random_mass_transfer_anneal(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.6,
                                end_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """Stochastic mass move accepting worsening moves with a temperature decay."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    cur_val = best_val
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # pick donor proportionally to its mass (more mass → more influence)
        donor = rng.choice(n, p=a / a.sum())
        receiver = rng.integers(0, n)
        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        new_val = _max_convolution_fft(a_new)

        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

    return best_a, best_val

            n = 11000
        else:
            n = 8000

    target_sum = math.sqrt(2.0 * n)

    rng = np.random.default_rng()
    candidates = []

    # --------------------------------------------------------------
    # Warm‑start (if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        prev = _project_to_simplex(prev, target_sum)
        candidates.append(prev)

    # --------------------------------------------------------------
    # Parametric families
    # --------------------------------------------------------------
    # Power‑law
    for p in np.linspace(0.40, 0.85, 35):
        seq = _power_law_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Exponential decay
    for lam in np.linspace(5e-4, 0.012, 30):
        seq = _exponential_sequence(n, lam)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Sine‑power
    for p in np.linspace(0.2, 2.0, 24):
        seq = _sin_power_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

    # Dirichlet random vectors (moderately sparse)
    for _ in range(12):
        alpha = np.full(n, 0.5, dtype=np.float64)
        seq = rng.dirichlet(alpha) * target_sum
        candidates.append(seq)

    # --------------------------------------------------------------
    # Quick evaluation of the pool
    # --------------------------------------------------------------
    scored = []
    for seq in candidates:
        val = _max_convolution_fft(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])

    top_k = min(8, len(scored))
    best_seq = scored[0][1].copy()
    best_val = scored[0][0]
    print(f"[{time.time():.2f}] Initial best score from pool = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam GD on soft‑max (explorative)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase_dead = time.time() + remaining * 0.35
        per_candidate = (phase_dead - time.time()) / top_k
        for i in range(top_k):
            if time.time() >= phase_dead - 0.5:
                break
            seq0 = scored[i][1]
            sub_dead = time.time() + min(per_candidate, phase_dead - time.time())
            a_opt, val_opt = _adam_descent(seq0, sub_dead, target_sum,
                                            max_beta=3e5,
                                            step0=0.05)
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] Adam improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        sub_dead = time.time() + remaining * 0.20
        a_sg, val_sg = _subgradient_descent(best_seq, target_sum, sub_dead,
                                            step_factor=0.45, gamma=0.95)
        if val_sg < best_val:
            best_val, best_seq = val_sg, a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑grad improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        mt_dead = time.time() + remaining * 0.30
        a_mt, val_mt = _deterministic_mass_transfer_refine(best_seq, mt_dead, target_sum)
        if val_mt < best_val:
            best_val, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Random annealing (fine‑tuning)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_ann, val_ann = _random_mass_transfer_anneal(best_seq, deadline, target_sum,
                                                      start_temp=0.6,
                                                      end_temp=1e-5)
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[{time.time():.2f}] Annealing improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – Final short Adam polishing
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        a_fin, val_fin = _adam_descent(best_seq,
                                        time.time() + remaining,
                                        target_sum,
                                        max_beta=5e5,
                                        step0=0.03)
        if val_fin < best_val:
            best_val, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polishing improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Post‑process, final evaluation, store globally
    # --------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=float)
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]
    final_seq = final_arr.tolist()

    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Final evaluation failed: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
