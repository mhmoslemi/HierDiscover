def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5
    # Choose size based on budget
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 14000
    else:
        n = 10000
    target_sum = np.sqrt(2.0 * n)  # normalization

    # Warm start:
    if 'height_sequence_1' in globals():
        init = _resize_sequence(np.array(globals()['height_sequence_1'], dtype=np.float64), n)
    else:
        init = np.power(np.arange(1, n+1, dtype=np.float64), -0.6)
    init *= target_sum / init.sum()

    # Phase 1: mirror descent
    a, val = _exponential_descent_phase(init, target_sum, time.time() + 0.10*(deadline - time.time()))

    # Phase 2: subgradient descent
    a, val = _subgradient_phase(a, target_sum, time.time() + 0.20*(deadline - time.time()))

    # Phase 3: annealing
    a, val = _deterministic_anneal(a, target_sum, deadline)

    # Trim trailing zeros
    arr = np.array(a)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1]+1]
    seq = arr.tolist()
    final_val = evaluate_sequence(seq) # evaluate final
    print(f"Final score {final_val}")
    globals()['height_sequence_1'] = seq
    return seq

def _deterministic_anneal(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.4  # larger temperature to allow bigger moves

    it = 0
    last_full = start

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate peak
        k_max = int(np.argmax(b))
        # compute donor contributions
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions for each index in idx
        # We need a[idx] and a[k_max - idx]
        contrib = a[idx] * a[k_max - idx]

        # choose donor as argmax contrib
        donor = idx[int(np.argmax(contrib))]

        # choose receiver as argmin a

        # choose among n candidates
        receiver = int(np.argmin(a))

"""
Proposes a step‑function sequence with low max‑self‑convolution.
The search runs within a given time budget (default 1000 s) and
consists of three stages:

1. **Mirror descent on a soft‑max surrogate** – quickly reduces the
   peak of the convolution using a multiplicative update derived from the
   gradient of a smooth approximation of the max operation.

2. **Sub‑gradient descent on the true max entry** – directly attacks the
   objective using the exact sub‑gradient of the max‑convolution entry.

3. **Deterministic simulated‑annealing mass transfer** – repeatedly moves
   a small amount of mass from the index that contributes most to the
   current peak towards the globally smallest entry, accepting uphill
   moves with a temperature that decays linearly to zero.

The algorithm starts from the best known construction (`height_sequence_1`)
if it exists, otherwise from a simple power‑law, and respects the
normalisation ∑a = √(2·n).  The final sequence is trimmed of trailing zeros
and stored back into the global variable `height_sequence_1` for warm‑starts
in future calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    max_b = _max_convolution(a)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phase A)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – multiplicative mirror descent on the surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase smoothing temperature
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient of max entry
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear cooling of step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.05
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # jitter occasionally
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – deterministic mass‑transfer simulated annealing
# ----------------------------------------------------------------------


def _deterministic_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Long‑run annealing that moves mass from the index that contributes most
    to the current peak towards the globally smallest entry.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # start temperature ≈ 40 % of current best (allows larger moves)
    T0 = best_val * 0.40

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – index with the largest contribution
        donor = int(idx[np.argmax(contrib)])

        # receiver – index with the smallest height
        receiver = int(np.argmin(a))

        if donor == receiver:
            # if coincident, pick a random different index
            receiver = rng.integers(n)
            if receiver == donor:
                it += 1
                continue

        # maximal transferable amount (scaled by local size and global budget)
        max_delta = min(a[donor] * 0.5, target_sum * 0.005)

        # temperature‑scaled random transfer
        if T0 > 0.0:
            delta = rng.random() * max_delta * (temperature / T0)
        else:
            delta = 0.0

        if delta <= 1e-14:
            it += 1
            continue

        # ---- trial update of the convolution (copy‑on‑write) ----
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_full > 6.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional tiny random jitter to keep dynamics alive
        if it % 5000 == 0 and it > 0:
            i, j = rng.integers(n, size=2)
            if i != j and a[i] > 1e-12:
                eps = min(a[i] * 0.1, target_sum * 0.001)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.

    The search consists of three phases:
    1) exponential mirror descent on a soft‑max surrogate,
    2) sub‑gradient descent on the true max‑convolution,
    3) deterministic annealing mass‑transfer.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem size (more points when more time is available)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 22000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start from the best known construction if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.58))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent
    # --------------------------------------------------------------
    time_A_deadline = time.time() + 0.08 * (deadline - time.time())
    a_best, best_score = _exponential_descent(init, target_sum, time_A_deadline)
    print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_B_deadline = time.time() + 0.20 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(a_best, target_sum, time_B_deadline)
        if val_sub < best_score:
            a_best, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After sub‑gradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – deterministic annealing mass‑transfer
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = _deterministic_anneal(a_best, target_sum, deadline)
        if val_ann < best_score:
            a_best, best_score = a_ann, val_ann
        print(f"[{time.time():.2f}] After deterministic anneal best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(a_best, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)     # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
