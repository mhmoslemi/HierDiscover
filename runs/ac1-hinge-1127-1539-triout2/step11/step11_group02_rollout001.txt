        rows = []
        cols = []
        data = []
        b_ub = []
        # constraints conv[idx] + 2* sum_j a[idx - j] * Δ_j <= t
        for row_i, k in enumerate(idxs):

"""
Step‑function optimizer for the max‑convolution benchmark.

The score is exactly the maximal entry of the linear convolution a*a,
because the sum of a is fixed to √(2 n).  The algorithm proceeds in
four phases:

1. **Coarse parametric pre‑search** – scans a few families
   (pure power law, power law with exponential cut‑off, broken power law)
   on a moderate size n to obtain a good starting shape.

2. **Projected Adam gradient descent** – minimises a soft‑max
   approximation of the max‑convolution while keeping a on the simplex
   Σa = √(2 n).  The temperature β is annealed exponentially.

3. **Linear‑programming mass‑transfer refinement** – for the current
   maximal convolution entries we solve a small LP that moves mass from
   many “donor’’ indices to many “receiver’’ indices simultaneously.
   The LP enforces a decrease of the targeted peak (and keeps all a_i
   non‑negative and the total mass unchanged).  This typically beats the
   greedy single‑donor moves used before.

4. **Final short Adam polish** – a quick high‑β Adam run to clean up
   any remaining imperfections.

All phases respect the time budget (default 1000 s) and the simplex
constraint.  The best sequence found is trimmed of trailing near‑zeros
and returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score – with the proper normalisation it equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (used in the dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming based mass‑transfer (global refinement)
# ----------------------------------------------------------------------


def _lp_refine(
    a0: np.ndarray,
    deadline: float,
    top_k: int = 60,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Solve a small LP that moves mass from many donors to many receivers
    in order to lower the current maximal convolution entries.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # pick the K largest convolution indices
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        # build sparse constraint matrix
        rows = []
        cols = []
        data = []
        b_ub = []

        # constraints for selected convolution peaks: conv[k] + 2*a*Δ <= t
        for row_i, k in enumerate(idxs):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j_range = np.arange(lo, hi + 1)
            coeffs = 2.0 * a[k - j_range]
            rows.extend([row_i] * len(j_range))
            cols.extend(j_range.tolist())
            data.extend(coeffs.tolist())

            # coefficient for t (last variable)
            rows.append(row_i)
            cols.append(n)          # t column index
            data.append(-1.0)
            b_ub.append(-conv[k])

        # positivity constraints: -Δ_i <= a_i
        for i in range(n):
            row_i = len(idxs) + i
            rows.append(row_i)
            cols.append(i)
            data.append(-1.0)
            b_ub.append(a[i])

        A_ub = coo_matrix((data, (rows, cols)), shape=(len(b_ub), n + 1)).tocsc()
        b_ub = np.array(b_ub, dtype=np.float64)

        # equality: Σ Δ_i = 0
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0], dtype=np.float64)

        # objective: minimise t
        c = np.zeros(n + 1, dtype=np.float64)
        c[n] = 1.0

        # variable bounds
        bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]

        # solve
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if res.status != 0:
            # infeasible or solver failure – stop refining
            break

        delta = res.x[:n]
        t_val = res.x[n]

        new_a = a + delta
        # numerical safety
        new_a[new_a < 0] = 0.0
        new_score = _score(new_a)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = new_a.copy()
            a = new_a
            print(
                f"[{time.time():.2f}] LP refine improved score={best_score:.6f} (t={t_val:.6f})"
            )
        else:
            # no improvement – stop
            break

    return best_a, best_score

    best_seq = None
    best_score = float("inf")

    # (a) pure power‑law (coarse grid)
    for p in np.arange(0.2, 1.21, 0.03):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in np.arange(0.2, 1.21, 0.04):
        for q in np.arange(0.0, 7.01, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (very few combos)
    for p1 in [0.3, 0.5, 0.7]:
        for p2 in [1.0, 1.5, 2.0]:
            for alpha in [0.2, 0.4, 0.6]:
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) initialise from previous global best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(
        f"[{time.time():.2f}] After pre‑search: best score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase A – projected Adam gradient descent (moderate duration)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    gd_deadline = time.time() + 0.30 * time_left  # 30 % of remaining time
    rng = np.random.default_rng()

    while time.time() < gd_deadline - 2.0:
        jitter = rng.normal(scale=target_sum * 0.012, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            gd_deadline,
            target_sum,
            max_beta=8e5,
            step_factor=0.75,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(
                f"[{time.time():.2f}] Adam GD restart improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase B – linear‑programming global mass transfer
    # ------------------------------------------------------------------
    lp_deadline = deadline - 4.0  # keep a couple seconds for final polish
    a_ref, val_ref = _lp_refine(best_seq, lp_deadline, top_k=80)
    if val_ref < best_score:
        best_score, best_seq = val_ref, a_ref.copy()
        print(
            f"[{time.time():.2f}] LP refinement improved best = {best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase C – final short high‑β Adam polish (few seconds)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(
                f"[{time.time():.2f}] Final Adam polish improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries and convert to list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final external evaluation (harmless if not present)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
