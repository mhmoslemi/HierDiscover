"""
Hybrid optimizer for the non‑negative step‑function benchmark.

The new version adds an explicit linear‑programming “mass‑balancing’’
phase that works on a down‑sampled copy of the sequence.  For a fixed
maximum convolution value M the constraints

    (f * g)[k] ≤ M ,   k = 0,…,2n‑2

are linear in the unknown vector g, so we can maximise the total mass
∑g_i subject to these constraints (LP).  The optimum pushes mass into
indices where f is small, thereby flattening the autocorrelation.
The LP is solved on a modest size (≈2000) problem, the result is
up‑sampled to the full size and refined with the original multi‑phase
gradient / simulated‑annealing pipeline.

The whole routine respects the given time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used in mirror & GD phases)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of the max‑convolution entry."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    grad = np.fft.irfft(
        np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size), n=size
    )[: 2 * n - 1]
    grad = 2.0 * grad[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – subgradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent directly on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with random restarts (momentum version)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – exact local mass‑transfer refinement (deterministic hill‑climb)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase E – simulated annealing with cheap O(n) updates
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Simple SA that moves small mass between two indices, updating the
    convolution in O(n) time. Accepts occasional worsening moves."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    T0 = max(0.05, (best_val - 1.0) * 0.5)

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.05, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        if (int(elapsed * 100) % 5000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# New Phase – LP‑based mass equalisation on a reduced problem
# ----------------------------------------------------------------------


def _lp_equalization(a0: np.ndarray,
                     target_sum: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a small linear program:
        maximise   1ᵀ g
        subject to (a * g) ≤ M   (M = max(a*a))
                  g ≥ 0
    and mix the result back into the current vector.
    The LP works on a down‑sampled copy of the sequence, keeping the
    dimension modest (≈2000) so that a dense constraint matrix fits
    comfortably into memory.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        M = _max_convolution_fft(a)

        # Build the (2n‑1) × n constraint matrix for the convolution a * g.
        # Row k contains the segment a[k‑j] for j = 0…n‑1 where defined.
        offsets = np.arange(n, dtype=np.int32)[:, None]        # (n,1)
        cols = np.arange(n, dtype=np.int32)                   # (n,)
        rows = cols + offsets                                 # (n,n)
        rows_flat = rows.ravel()
        cols_flat = np.tile(cols, n)
        data = np.repeat(a, n)                                # length n²
        A = sparse.csr_matrix((data, (rows_flat, cols_flat)),
                              shape=(2 * n - 1, n))

        # Linear programme: maximise sum(g)  ↔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Give a few seconds for the solver; clip the overall remaining time.
        time_left = max(0.0, deadline - time.time())
        opts = {"time_limit": time_left}
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs', options=opts)

        if not res.success:
            print(f"[{time.time():.2f}] LP failed ({res.message}); aborting LP phase.")
            break

        g = res.x
        if g.sum() == 0:
            break

        # Renormalise to the required total mass.
        g = g * (target_sum / g.sum())

        # Mix a fraction of g into the current vector.
        t = 0.3 if it < 5 else 0.1
        a = (1.0 - t) * a + t * g

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP iter {it} improved score={best_val:.6f}")

        it += 1
        if it >= 12:      # a small, bounded number of LP solves
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ---- problem size -------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # ---- warm‑start ---------------------------------------------------
    if "height_sequence_1" in globals():
        init_full = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
    else:
        init_full = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
    init_full = init_full * (target_sum / init_full.sum())

    best_seq = init_full.copy()
    best_score = _score(best_seq)

    # ---- Phase A: quick exponential descent ---------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.08 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_score:
            best_seq, best_score = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # ---- Phase B: LP‑based equalisation on a reduced problem ----------
    if time.time() < deadline:
        # down‑sample to a manageable size for the LP
        n_small = min(2000, max(500, n // 5))
        small_seq = _resize_sequence(best_seq, n_small)
        small_target = np.sqrt(2.0 * n_small)
        small_seq = small_seq * (small_target / small_seq.sum())

        lp_deadline = time.time() + 0.20 * (deadline - time.time())
        a_lp, val_lp = _lp_equalization(small_seq, small_target, lp_deadline)

        # up‑sample back to the full dimension and renormalise
        up_seq = _resize_sequence(a_lp, n)
        up_seq = up_seq * (target_sum / up_seq.sum())
        val_up = _score(up_seq)

        if val_up < best_score:
            best_seq, best_score = up_seq, val_up
        print(f"[{time.time():.2f}] After LP equalisation best score={best_score:.6f}")

    # ---- Phase C: gradient‑descent restarts (soft‑max) ---------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # ---- Phase D: exact local refinement ------------------------------
    if time.time() < deadline:
        local_deadline = time.time() + 0.30 * (deadline - time.time())
        a_loc, val_loc = _local_refine(best_seq, target_sum, local_deadline)
        if val_loc < best_score:
            best_seq, best_score = a_loc, val_loc
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # ---- Phase E: final simulated annealing ---------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
        print(f"[{time.time():.2f}] After simulated annealing best score={best_score:.6f}")

    # ---- final clean‑up ------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (the external verifier)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
