    n = a.shape[0]

def _max_after_move(a, b, donor, receiver, delta):
    n = a.shape[0]
    # Avoid invalid if delta=0 returns original max (b.max())
    if delta <= 0:
        return b.max()
    b_new = b.copy()
    b_new[donor:donor + n] -= 2.0 * delta * a
    b_new[receiver:receiver + n] += 2.0 * delta * a
    b_new[2 * donor] += delta * delta
    b_new[2 * receiver] += delta * delta
    if donor != receiver:
        b_new[donor + receiver] -= 2.0 * delta * delta
    return b_new.max()

"""
Greedy mass‑transfer refinement for the step‑function benchmark.

The search contains four stages:
* A – multiplicative mirror descent on a soft‑max surrogate of the max‑convolution.
* B – sub‑gradient descent on the true max‑convolution.
* C – projected gradient descent with random restarts.
* D – deterministic greedy mass transfer: repeatedly move a small amount of mass
      from the index that contributes most to the current peak towards the
      smallest indices, using an inexpensive O(n) update of the full
      convolution and a tiny line‑search (few trial deltas).  This stage
      typically pushes the score below the previous 1.507 benchmark.

All helpers are top‑level; the routine respects the given time budget
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phases A and C)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _annealed_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = _project_to_simplex(a0, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray, target_sum: float, deadline: float):
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start, target_sum,
                                            time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# Phase D – deterministic greedy mass‑transfer refinement
# ----------------------------------------------------------------------


def _max_after_move(a: np.ndarray, b: np.ndarray,
                    donor: int, receiver: int, delta: float) -> float:
    """Return the maximum entry of the convolution after moving `delta`."""
    if delta <= 0.0:
        return b.max()
    n = a.shape[0]
    b_new = b.copy()
    b_new[donor:donor + n] -= 2.0 * delta * a
    b_new[receiver:receiver + n] += 2.0 * delta * a
    b_new[2 * donor] += delta * delta
    b_new[2 * receiver] += delta * delta
    if donor != receiver:
        b_new[donor + receiver] -= 2.0 * delta * delta
    return float(b_new.max())


def _apply_move(a: np.ndarray, b: np.ndarray,
                donor: int, receiver: int, delta: float):
    """Perform the move in‑place and return the updated (a,b)."""
    n = a.shape[0]
    # update convolution (linear + quadratic parts)
    b[donor:donor + n] -= 2.0 * delta * a
    b[receiver:receiver + n] += 2.0 * delta * a
    b[2 * donor] += delta * delta
    b[2 * receiver] += delta * delta
    if donor != receiver:
        b[donor + receiver] -= 2.0 * delta * delta
    # update heights
    a[donor] -= delta
    a[receiver] += delta
    return a, b


def _greedy_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Greedy deterministic mass‑transfer: repeatedly move a small amount
    of mass from the index that contributes most to the current peak
    towards the smallest indices.  A tiny line‑search (few trial deltas)
    is used to guarantee an actual improvement.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)

    best_a = a.copy()
    best_val = _score(a)
    print(f"[{time.time():.2f}] Greedy start score={best_val:.6f}")

    rng = np.random.default_rng()
    it = 0
    no_improve = 0

    while time.time() < deadline:
        # current peak
        k_max = int(np.argmax(b))
        # indices that actually contribute to that peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)]

        # candidate receivers – the five smallest heights (excluding donor)
        rcands = np.argsort(a)[:5]
        rcands = rcands[rcands != donor]
        if rcands.size == 0:
            rcands = np.array([rng.integers(0, n)])

        improved = False
        best_local_val = best_val
        best_local_delta = 0.0
        best_pair = (donor, donor)  # dummy

        for receiver in rcands:
            if receiver == donor:
                continue
            # allow up to 10 % of donor’s mass to be shifted
            max_delta = min(a[donor] * 0.10, target_sum * 0.001)
            if max_delta <= 1e-12:
                continue

            # try a few trial deltas (full, half, quarter)
            for frac in (1.0, 0.5, 0.25):
                delta = max_delta * frac
                new_max = _max_after_move(a, b, donor, receiver, delta)
                if new_max + 1e-12 < best_local_val:
                    best_local_val = new_max
                    best_local_delta = delta
                    best_pair = (donor, receiver)
                    improved = True

        if improved:
            donor, receiver = best_pair
            delta = best_local_delta
            a, b = _apply_move(a, b, donor, receiver, delta)
            best_val = best_local_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Greedy it={it:5d} move {donor}->{receiver} δ={delta:.6f}"
                  f" new_score={best_val:.6f}")
            no_improve = 0
        else:
            no_improve += 1
            # occasional random jitter to escape flat regions
            if no_improve % 80 == 0:
                rand_donor = rng.integers(0, n)
                rand_recv = rng.integers(0, n)
                if rand_donor != rand_recv and a[rand_donor] > 1e-12:
                    max_delta = min(a[rand_donor] * 0.05, target_sum * 0.001)
                    delta = max_delta * 0.5
                    a, b = _apply_move(a, b, rand_donor, rand_recv, delta)
                    cur_val = _score(a)
                    if cur_val < best_val:
                        best_val = cur_val
                        best_a = a.copy()
                        print(f"[{time.time():.2f}] Random jitter improved score={best_val:.6f}")

        # occasional full recompute to avoid numerical drift
        if it % 5000 == 0:
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # problem size – larger n gives more freedom but costs more time
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # warm‑start from the best known construction if available
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent (≈ 8 % of budget)
    # --------------------------------------------------------------
    exp_deadline = time.time() + 0.08 * (deadline - time.time())
    a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
    best_seq, best_score = a_exp, val_exp
    print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent (≈ 12 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.12 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts (≈ 15 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.15 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – deterministic greedy mass‑transfer refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _greedy_mass_transfer(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After greedy refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
