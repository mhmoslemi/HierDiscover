def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.2

    # Choose n based on budget
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # Build initial candidates
    init_candidates = []
    # warm start from previously stored sequence, resized to n
    if 'height_sequence_1' in globals():
        init = _resize_sequence(np.array(globals()['height_sequence_1'], dtype=np.float64), n)
        init = init * (target_sum / init.sum())
        init_candidates.append(init)

    # power law with several exponents
    for p in [-0.8, -0.7, -0.6, -0.55, -0.5, -0.45]:
        a = np.arange(1, n+1, dtype=np.float64) ** p
        a = a * (target_sum / a.sum())
        init_candidates.append(a)

    # random Dirichlet draws
    rng = np.random.default_rng()
    # allocate some time for random draws
    while time.time() < deadline - 50 and len(init_candidates) < 20: # add maybe 15 random seeds total
        a = rng.gamma(shape=1.0, scale=1.0, size=n)  # gamma=1 => exponential
        a = a * (target_sum / a.sum())
        init_candidates.append(a)

    # Choose best initial by evaluating _score
    best_a = None
    best_val = np.inf
    for a in init_candidates:
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    print(f"[{time.time():.2f}] Best initial score={best_val:.6f}")

    # Phase 1: Adam descent on softmax gradient
    if time.time() < deadline:
        adam_deadline = time.time() + 0.35 * (deadline - time.time())
        best_a, best_val = _adam_descent(best_a, target_sum, adam_deadline)
        print(f"[{time.time():.2f}] After Adam descent best score={best_val:.6f}")

    # Phase 2: Local refinement (hill-climbing)
    if time.time() < deadline:
        local_deadline = time.time() + 0.25 * (deadline - time.time())
        best_a, best_val = _local_refine(best_a, target_sum, local_deadline)
        print(f"[{time.time():.2f}] After local refine best score={best_val:.6f}")

    # Phase 3: Simulated annealing (cheap O(n) updates)
    if time.time() < deadline:
        best_a, best_val = _simulated_annealing(best_a, target_sum, deadline)
        print(f"[{time.time():.2f}] After SA best score={best_val:.6f}")

    # Trim trailing zeros
    arr = np.array(best_a)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Evaluate through external evaluate_sequence (not imported)
    try:
        val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score={val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error {e}")

    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Hybrid search for a low‑autocorrelation non‑negative step function.

The algorithm proceeds in three high‑level stages:

1️⃣  **Warm‑start & random seeding** – builds a small pool of candidate
    vectors (previous best, a few power‑law shapes and many Dirichlet
    draws), evaluates the exact objective and keeps the best one.

2️⃣  **Adam optimisation on a soft‑max surrogate** – minimises a smooth
    approximation of the maximal autocorrelation using the Adam optimiser.
    The temperature β of the soft‑max is increased from 1 to 65536
    (exponential schedule) so that the surrogate gradually approaches the
    true max‑convolution.  After each gradient step the vector is projected
    onto the simplex {x ≥ 0, Σx = √(2n)}.

3️⃣  **Local refinement + simulated annealing** – fast O(n) hill‑climbing
    (deterministic mass‑transfer) followed by a classic SA that moves a
    tiny amount of mass between two indices while updating the convolution
    incrementally.  Both phases respect the remaining time budget.

All phases are time‑aware; the total run never exceeds the supplied
budget (default 1000 s).  The best sequence encountered is returned as a
plain Python list of floats and stored in the global variable
`height_sequence_1` for warm‑starts in subsequent calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²) – the quantity returned by evaluate_sequence."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities (gradient of log‑sum‑exp of the autocorrelation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max (log‑sum‑exp) approximation of max(a*a).

    Returns:
        grad : ndarray of shape a.shape – ∂/∂a  (log‑sum‑exp_{k} β·b_k)
        b    : the full convolution vector (length 2·n−1)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()               # soft‑max probabilities over k

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    grad_full = np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                            np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]

    # the central n entries correspond to the gradient w.r.t. each a_i
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max‑convolution.
    The temperature β grows from 1 to 65536 following an exponential schedule.
    After each update the iterate is projected onto the simplex (fixed ℓ1‑norm).
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # progress fraction in [0,1]
        elapsed = time.time() - start
        total = deadline - start
        progress = elapsed / total if total > 0 else 1.0

        # temperature schedule (geometric from 1 to 65536)
        beta = 1.0 * (65536.0 ** progress)

        # gradient of the soft‑max surrogate
        grad, _ = _gradient_softmax(a, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step size – modestly scaled with the temperature
        base_lr = target_sum * 0.05
        lr = base_lr / (beta + 1.0)

        # Adam update (additive)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the non‑negative simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # evaluate current solution
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:.1f} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # choose a relatively weak receiver (low contribution to max)
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        b_new = b.copy()
        # effect of moving delta from donor to receiver on the convolution
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # Periodic full re‑normalisation to avoid drift
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        # Very occasional global jitter to shake plateaus
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Simple SA that moves tiny mass between two indices, updating the convolution in O(n)."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    # initial temperature – fraction of current excess over 1.0
    T0 = max(0.05, (best_val - 1.0) * 0.5)

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # donor chosen proportionally to its mass (avoids many zero donors)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue
        # distinct receiver uniformly
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.05, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # occasional global jitter to break stagnation
        if (int(elapsed * 100) % 5000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved score={best_val:.6f}")

    return best_a, best_val

    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------ problem size ---------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------ generate initial candidates -------------------
    init_candidates = []

    # warm‑start from previously stored sequence (if any)
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        init = _resize_sequence(seq, n)
        init = init * (target_sum / init.sum())
        init_candidates.append(init)

    # a few deterministic power‑law shapes
    for p in [-0.8, -0.7, -0.6, -0.55, -0.5, -0.45]:
        a = np.arange(1, n + 1, dtype=np.float64) ** p
        a = a * (target_sum / a.sum())
        init_candidates.append(a)

    # random Dirichlet draws – cheap exploration
    rng = np.random.default_rng()
    while time.time() < deadline - 30 and len(init_candidates) < 30:
        a = rng.gamma(shape=1.0, scale=1.0, size=n)          # exponential(1) ~ Dirichlet
        a = a * (target_sum / a.sum())
        init_candidates.append(a)

    # pick the best among the pool using the exact objective
    best_a = None
    best_val = np.inf
    for a in init_candidates:
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    print(f"[{time.time():.2f}] Best initial score={best_val:.6f}")

    # ------------------------ Adam optimisation (soft‑max) ------------------
    if time.time() < deadline:
        adam_deadline = time.time() + 0.38 * (deadline - time.time())
        best_a, best_val = _adam_descent(best_a, target_sum, adam_deadline)
        print(f"[{time.time():.2f}] After Adam descent best score={best_val:.6f}")

    # ------------------------ Exact local refinement -----------------------
    if time.time() < deadline:
        local_deadline = time.time() + 0.27 * (deadline - time.time())
        best_a, best_val = _local_refine(best_a, target_sum, local_deadline)
        print(f"[{time.time():.2f}] After local refinement best score={best_val:.6f}")

    # ------------------------ Simulated annealing -------------------------
    if time.time() < deadline:
        best_a, best_val = _simulated_annealing(best_a, target_sum, deadline)
        print(f"[{time.time():.2f}] After simulated annealing best score={best_val:.6f}")

    # ------------------------ final clean‑up -------------------------------
    arr = np.array(best_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final external evaluation (the official verifier)
    try:
        final_val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
