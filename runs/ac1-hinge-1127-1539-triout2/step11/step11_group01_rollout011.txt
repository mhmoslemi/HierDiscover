"""
Fast convex–concave search for a low‑max‑convolution step function.

The algorithm works in four phases:

1. **Parametric sweep** – a dense grid of simple families
   (power‑law, power‑law+exponential, Gaussian) is evaluated
   and the best candidate is kept.

2. **Projected Adam (soft‑max)** – minimises a smooth approximation
   of the max‑convolution while staying on the simplex Σa = √(2n).

3. **Exact deterministic mass‑transfer** – the classic “peak‑flattening”
   routine that analytically reduces the current maximum to the second
   largest convolution value.

4. **Sub‑gradient descent on the true max** – uses the exact
   sub‑gradient of `max_k (a*a)_k` (binary mask of the active constraints)
   and takes a step of size `Δ / ‖g‖²` where `Δ` is the gap to the
   second‑largest value.  This directly flattens the convolution and
   typically pushes the bound well below the 1.507 … obtained by the
   previous implementation.

All phases respect the global time budget (default 1000 s).  The best
sequence found is trimmed of trailing near‑zero entries and returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact evaluation (equivalent to max‑convolution when Σa = √(2n))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^‑p."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^‑p * exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma_factor: float,
                       target_sum: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ = sigma_factor·n."""
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float) -> Tuple[np.ndarray, np.ndarray]:
    """
    Returns (gradient, convolution) of the soft‑max surrogate
    of `max(b)`, where b = a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> Tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max proxy."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> Tuple[np.ndarray, float]:
    """Several short Adam restarts from jittered starting points."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> Tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner values a_{k‑i}
        partner = a[k - idx]

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (g, conv, max_b) where
    - g is a sub‑gradient of max_k (a*a)_k,
    - conv is the full convolution a*a,
    - max_b is its maximum.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_b = conv.max()

    # binary mask of active constraints (tolerance avoids numerical noise)
    mask = (conv >= max_b - 1e-12).astype(np.float64)
    a_rev = a[::-1]
    size = 1 << ((2 * n - 1).bit_length())
    mask_fft = np.fft.rfft(mask, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(mask_fft * a_rev_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv, max_b


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         tol: float = 1e-13) -> Tuple[np.ndarray, float]:
    """
    Greedy flattening using the exact sub‑gradient of the max‑convolution.
    An analytic step size `Δ / ||g||²` (Δ = gap to 2nd‑largest value) is used.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        grad, conv, max_b = _subgradient_max(a)

        # second‑largest convolution entry
        mask = conv < max_b - 1e-15
        if not np.any(mask):
            break
        second_max = conv[mask].max()
        delta = max_b - second_max
        if delta < tol:
            break

        grad_norm2 = np.dot(grad, grad)
        if grad_norm2 == 0.0:
            break

        step = delta / grad_norm2
        # safety: do not move more than a modest fraction of the total mass
        step = min(step, 0.3 * target_sum)

        a_new = a - step * grad
        a_new = _project_to_simplex(a_new, target_sum)
        cur_score = _score(a_new)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploratory phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> Tuple[np.ndarray, float]:
    """Simple stochastic mass‑transfer to escape shallow plateaus."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a minimal max‑convolution.
    The routine adheres to the given time budget and returns the best
    sequence found (as a plain Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Problem size – larger n gives more flexibility but costs more per iteration
    if budget_s >= 800:
        n = 14000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scale invariant)

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (power‑law, exp‑power, Gaussian)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.30, 0.85, 31)          # 31 values
    q_grid = np.linspace(0.0, 6.0, 31)           # 31 values
    sigma_grid = np.linspace(0.05, 0.30, 13)     # Gaussian width factor

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) Gaussian shapes (centered, symmetric)
    for sigma in sigma_grid:
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (d) include previously found sequence if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_len = prev.shape[0]
            old_x = np.linspace(0.0, 1.0, old_len)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (e) small jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.20
    while time.time() < mt_deadline - 4.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             mt_deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # occasional random moves to escape tiny plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        mt_deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    sg_deadline = time.time() + (deadline - time.time()) * 0.25
    a_sg, val_sg = _subgradient_descent(best_seq,
                                         sg_deadline,
                                         target_sum)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Sub‑gradient descent improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Final polishing with a higher β (soft‑max)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
