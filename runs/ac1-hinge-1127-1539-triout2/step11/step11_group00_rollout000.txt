def _greedy_local_improvement(a, target_sum, deadline):
    n = a.shape[0]
    b = _conv_fft(a)
    best_val = _score(a)
    best_a = a.copy()
    max_iter = 5000  # or until time
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        # Identify peak index
        k_max = int(np.argmax(b))
        # Determine donor index (within convolution contribution)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        # contributions to k_max
        contrib = a[idx] * a[k_max - idx]
        # donor = argmax contrib
        donor = idx[np.argmax(contrib)]
        # receiver = choose index with smallest a (but not donor)
        receiver = np.argmin(a + (np.arange(n)==donor)*np.inf)  # ignore donor
        # set delta limited by donor amount and overall step size
        max_delta = min(0.5 * a[donor], target_sum * 0.0005)  # small step
        if max_delta <= 0:
            break
        delta = rng.random() * max_delta
        # compute new convolution quickly (O(n))
        b_new = b.copy()
        # removal from donor:
        b_new[donor:donor+n] -= 2.0 * delta * a
        # addition to receiver:
        b_new[receiver:receiver+n] += 2.0 * delta * a
        # quadratic adjustments:
        b_new[2*donor] += delta * delta
        b_new[2*receiver] += delta * delta
        if donor != receiver:

        else:

    return best_a, best_val

"""
Improved step‑function construction for the benchmark.

The algorithm consists of four stages:

1️⃣ **Warm‑start & resizing** – reuse the best known construction (global
   `height_sequence_1`) if present, otherwise a simple power‑law.
2️⃣ **Adam optimisation on a soft‑max surrogate** – a fast first‑order
   method that drives the max‑convolution down while keeping the
   ℓ¹‑norm fixed (projection onto the simplex).
3️⃣ **Long simulated‑annealing mass‑transfer** – moves tiny amounts of mass
   from indices that contribute heavily to the current peak towards low
   indices, with a temperature schedule that slowly cools.
4️⃣ **Greedy local refinement** – deterministic pairwise transfers that
   only accept moves that strictly improve the true objective.

All stages respect the time budget (default 1000 s) and keep the
sequence non‑negative. The final sequence is trimmed of trailing zeros
and stored in the global `height_sequence_1` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, power‑of‑2 padding)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective: 2·n·max(a*a) / (sum(a)²).
    The benchmark normalises sum(a) to √(2n); then the score equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Reference: Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used by Adam)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]  # exact convolution

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))                      # avoid overflow
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]

    # the relevant part of the reverse convolution gives the gradient
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase 1 – Adam optimisation on soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Adam optimiser on the soft‑max surrogate of the max‑convolution.
    The iterate is always projected onto the simplex {x≥0, Σx = target_sum}.
    """
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 1.0           # soft‑max temperature
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        # gradient of the surrogate
        g, _ = _softmax_gradient(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size – we scale inversely with β to keep updates stable
        eta = target_sum * 0.025 / (beta + 1.0)

        # Adam update (scalar learning‑rate times the preconditioned direction)
        a -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps sum fixed and non‑negative)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:6d} β={beta:.1f} score={best_val:.6f}")

        # slowly increase the soft‑max temperature to tighten the approximation
        if t % 2500 == 0 and beta < 65536.0:
            beta = min(beta * 2.0, 65536.0)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing mass‑transfer (long run)
# ----------------------------------------------------------------------


def _mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Long‑run annealing that moves a tiny amount of mass from a “donor’’
    (large contribution to the current peak) to a “receiver’’ (small height).
    The temperature starts at roughly 40 % of the current score and cools
    linearly to zero.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.40                   # initial temperature

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – choose proportionally to contribution (if any)
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.04, target_sum * 0.001)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it} improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_full > 9.5:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional tiny jitter to keep dynamics alive
        if it % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – greedy deterministic refinement
# ----------------------------------------------------------------------


def _greedy_local_refinement(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Deterministic pairwise mass transfers that are only accepted when
    they strictly improve the true objective.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_val = _score(a)
    best_a = a.copy()
    rng = np.random.default_rng()

    max_iters = 5000
    it = 0

    while time.time() < deadline and it < max_iters:
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions to the peak
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)]

        # receiver = smallest non‑zero height (avoid donor)
        receiver = np.argmin(np.where(np.arange(n) == donor,
                                      np.full(n, np.inf), a))

        # limit step size
        max_delta = min(0.5 * a[donor], target_sum * 0.0005)
        if max_delta <= 0.0:
            break
        delta = rng.random() * max_delta

        # incremental update of the convolution (same pattern as anneal)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        if new_val < best_val:
            # accept the improvement
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Greedy it={it} improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.3          # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start from the best known construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
    else:
        # simple decreasing power‑law fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation (≈7 % of budget)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + 0.07 * (deadline - time.time())
    a_adam, val_adam = _adam_descent(init, target_sum, phase1_deadline)
    best_seq, best_score = a_adam, val_adam
    print(f"[{time.time():.2f}] After Adam best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long simulated‑annealing mass‑transfer (≈55 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + 0.55 * (deadline - time.time())
        a_anneal, val_anneal = _mass_transfer_anneal(best_seq,
                                                    target_sum,
                                                    phase2_deadline)
        if val_anneal < best_score:
            best_seq, best_score = a_anneal, val_anneal
        print(f"[{time.time():.2f}] After annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – greedy deterministic refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_greedy, val_greedy = _greedy_local_refinement(best_seq,
                                                        target_sum,
                                                        deadline)
        if val_greedy < best_score:
            best_seq, best_score = a_greedy, val_greedy
        print(f"[{time.time():.2f}] After greedy refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    # Trim trailing zeros (keep a tiny epsilon for safety)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Verify with the provided benchmark function (if available)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # Store for future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
