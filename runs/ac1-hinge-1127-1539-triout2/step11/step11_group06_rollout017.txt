def _lp_step(f, M, target_sum):
    # Solve LP: maximize sum(g) s.t. conv(f,g) <= M , g>=0.
    n = f.shape[0]
    # Build sparse matrix A
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        # for i in lo..hi
        count = hi-lo+1
        rows.extend([k]*count)
        cols.extend(list(range(lo, hi+1)))
        data.extend(f[lo:hi+1].tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    # linprog min -sum(g)
    c = -np.ones(n)
    # variable bounds automatically non-negative
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                                 method='highs',
                                 options={'time_limit': max(1, int(deadline - time.time()))})
    if not res.success:
        return None
    g = res.x
    return g
```

"""
Hybrid optimizer with an extra LP‑based alternating step.

The routine follows the structure that achieved the previous best score,
but adds a few new ingredients:

* A quick sweep over power‑law exponents to obtain a better warm‑start.
* A down‑sampling → LP‑based refinement → up‑sampling stage.
  The LP (solved by SciPy HiGHS) maximises the total mass of a
  partner sequence g under the constraint   f∗g ≤ M, where M is the
  current peak of f∗f.  Mixing f with g (a small convex step) provably
  cannot increase the peak and often lowers it.
* A slightly re‑balanced time budget giving the final simulated‑annealing
  phase more time.

The code respects the 1000 s budget, returns a plain list of non‑negative
floats and stores the best sequence in the global variable
``height_sequence_1`` for warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the LP refinement)
# ----------------------------------------------------------------------
try:                     # pragma: no‑cover
    import scipy.sparse as sp
    import scipy.optimize as opt
    _SCIPY_OK = True
except Exception:       # pragma: no‑cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  With the normalisation sum=√(2n) this
    reduces to the peak of the convolution."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _downsample_sum(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Coarse down‑sampling by summing consecutive blocks."""
    old_len = seq.shape[0]
    if new_len >= old_len:
        return seq.copy()
    factor = old_len // new_len
    if factor <= 0:
        return seq.copy()
    trim = old_len - factor * new_len
    if trim > 0:
        seq = seq[:-trim]
    reshaped = seq.reshape(new_len, factor)
    return reshaped.sum(axis=1)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used in the exponential descent phases)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of the max‑convolution entry."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    grad = np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                        np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = 2.0 * grad[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exp‑GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent directly on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with random restarts (momentum version)
# ----------------------------------------------------------------------


def _annealed_gradient_descent_v2(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = a0.copy()
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GDv2 it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent_v2(start,
                                               time.time() + run_budget,
                                               target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – exact local mass‑transfer refinement (deterministic hill‑climb)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        # Choose a donor proportional to its contribution to the peak
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # Choose a receiver that currently contributes little to the peak
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # occasional re‑normalisation
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # tiny random jitter every few seconds
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase E – final simulated annealing (cheap O(n) updates)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Simple SA that moves small mass between two indices, updating the
    convolution in O(n) time. Occasionally accepts worsening moves."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = max(1e-9, deadline - start)

    # initial temperature – a fraction of the current excess over 1.0
    T0 = max(0.05, (best_val - 1.0) * 0.5)

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # donor biased by current mass, receiver uniform
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.05, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) convolution update
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # occasional global jitter to escape plateaus
        if (int(elapsed * 100) % 5000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA global jitter improved score={best_val:.6f}")

    return best_a, best_val

# LP‑based alternating refinement (optional, runs on a down‑sampled problem)

def _build_conv_matrix(f: np.ndarray):
    """Sparse matrix A where (A g)[k] = (f∗g)_k."""
    n = f.size
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        rows.extend([k] * idx.size)
        cols.extend(idx.tolist())
        data.extend(f[idx].tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(f: np.ndarray,
             M: float,
             target_sum: float,
             deadline: float) -> np.ndarray | None:
    """Solve   maximise Σ g_j   s.t.   f∗g ≤ M,   g ≥ 0.
    Returns the optimiser g (as a dense ndarray) or None on failure."""
    if not _SCIPY_OK:               # pragma: no‑cover
        return None
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # maximise Σ g_j  ⇔ minimise −Σ g_j
    bounds = [(0, None)] * n

    # HiGHS is fast; we give it a modest time‑limit via a cheap check
    # on the remaining budget – otherwise we risk overrunning the total deadline.
    remaining = deadline - time.time()
    if remaining < 0.1:
        return None
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs',
                          options={'time_limit': remaining})
    except Exception:               # pragma: no‑cover
        return None
    if not res.success:
        return None
    g = res.x
    # Scale to the required sum (the constraints guarantee Σ g ≤ target_sum)
    if g.sum() == 0:
        return None
    g = g * (target_sum / g.sum())
    return g


def _lp_alternating_refine(f_big: np.ndarray,
                           target_sum_big: float,
                           deadline: float) -> np.ndarray:
    """Down‑sample, run a few LP‑based alternating steps, up‑sample."""
    # 1) down‑sample to a modest size (≈2000)
    n_big = f_big.shape[0]
    n_small = min(2500, max(1500, n_big // 5))
    f_small = _downsample_sum(f_big, n_small)
    target_sum_small = np.sqrt(2.0 * n_small)
    f_small = f_small * (target_sum_small / f_small.sum())

    best = f_small.copy()
    best_val = _score(best)
    it = 0
    while time.time() < deadline - 5.0 and it < 6:
        M = _max_convolution_fft(best)
        g = _lp_step(best, M, target_sum_small, deadline)
        if g is None:
            break
        # Mix a little of g into the current sequence
        t_mix = 0.15
        best = (1 - t_mix) * best + t_mix * g
        best = best * (target_sum_small / best.sum())
        cur_val = _score(best)
        if cur_val < best_val:
            best_val = cur_val
            print(f"[{time.time():.2f}] LP‑refine it={it} score={best_val:.6f}")
        it += 1

    # 2) up‑sample back to the original size
    upsampled = _resize_sequence(best, n_big)
    upsampled = upsampled * (target_sum_big / upsampled.sum())
    return upsampled

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_big = 12000
    elif budget_s >= 500:
        n_big = 8000
    else:
        n_big = 5000

    target_sum_big = np.sqrt(2.0 * n_big)

    # --------------------------------------------------------------
    # warm‑start: either reuse the previous construction or try a few
    # simple power‑law shapes and keep the best.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_big = _resize_sequence(np.array(globals()["height_sequence_1"],
                                            dtype=np.float64), n_big)
    else:
        # test a handful of power‑law exponents
        best_val = np.inf
        best_seq = None
        for p in np.linspace(0.35, 0.85, 9):
            seq = (np.arange(1, n_big + 1, dtype=np.float64) ** (-p))
            seq = seq * (target_sum_big / seq.sum())
            val = _score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq
        init_big = best_seq

    # normalise (just in case)
    init_big = init_big * (target_sum_big / init_big.sum())

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent on soft‑max
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.15 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(init_big, target_sum_big,
                                              exp_deadline)
        best_seq = a_exp
        best_score = val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – sub‑gradient descent on the true max‑convolution entry
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.15 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum_big,
                                              sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.20 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum_big,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – deterministic local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = time.time() + 0.15 * (deadline - time.time())
        a_loc, val_loc = _local_refine(best_seq, target_sum_big,
                                       local_deadline)
        if val_loc < best_score:
            best_seq, best_score = a_loc, val_loc
        print(f"[{time.time():.2f}] After local refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase L – LP‑based alternating refinement on a down‑sampled problem
    # --------------------------------------------------------------
    if time.time() < deadline and _SCIPY_OK:
        lp_deadline = time.time() + 0.10 * (deadline - time.time())
        a_lp = _lp_alternating_refine(best_seq, target_sum_big, lp_deadline)
        val_lp = _score(a_lp)
        if val_lp < best_score:
            best_seq, best_score = a_lp, val_lp
        print(f"[{time.time():.2f}] After LP refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – simulated annealing (final polishing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum_big,
                                            deadline)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
        print(f"[{time.time():.2f}] After simulated annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # final clean‑up (trim trailing near‑zeros)
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final external evaluation (just for logging – the caller will use the
    # returned list for scoring)
    try:                                 # pragma: no‑cover
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                                 # pragma: no‑cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
