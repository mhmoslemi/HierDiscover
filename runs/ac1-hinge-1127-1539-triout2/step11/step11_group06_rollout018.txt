def _mass_transfer_sa(a_init, target_sum, deadline):
    a = a_init.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)   # calls _max_convolution_fft internally, but we could compute from b directly: cur_val = 2*n*b.max()/(target_sum**2)
    cur_val = float(2.0*n*b.max()/(target_sum**2))

    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = max(0.05, (best_val-1.0)*0.5)   # same as before

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed/total))

        # pick donor weighted by a (or contributions)
        donor = rng.choice(n, p=a/a.sum())
        if a[donor] <= 1e-14:
            continue
        # choose receiver uniformly
        receiver = rng.integers(n-1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor]*0.05, target_sum*0.001)
        if max_delta <= 0:
            continue
        delta = rng.random()*max_delta

        # compute new b copy
        b_new = b.copy()
        b_new[donor:donor+n] -= 2.0*delta*a
        b_new[receiver:receiver+n] += 2.0*delta*a
        b_new[2*donor] += delta*delta
        b_new[2*receiver] += delta*delta
        if donor != receiver:
            b_new[donor+receiver] -= 2.0*delta*delta

        new_max = b_new.max()
        new_val = float(2.0*n*new_max/(target_sum**2))

        # acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val)/temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")
        # occasional jitter global
        if (int(elapsed * 100) % 5000) == 0:
            jitter = rng.normal(scale=target_sum*0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = float(2.0*n*b_j.max()/(target_sum**2))
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved score={best_val:.6f}")

    return best_a, best_val

    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = _score(a)
    beta = 1.0
    max_beta = 65536.0
    base_lr = 0.1   # adapt
    rng = np.random.default_rng()
    while time.time() < deadline:
        grad, b = _gradient_softmax(a, beta)
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)

        # compute adaptive lr: maybe base_lr/(sqrt(v_hat)+eps)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = np.maximum(a, 0.0)

"""
Hybrid optimizer for the non‑negative step‑function benchmark.

The routine builds on the previous best construction and adds:
  1. A quick grid search over simple power‑law shapes for a better warm‑start.
  2. A short exponential‑mirror descent phase.
  3. An adaptive Adam optimiser on a soft‑max smooth approximation of the
     max‑convolution.
  4. Several brief simulated‑annealing (SA) restarts with cheap O(n) updates.
  5. One long SA phase that consumes the remaining budget.
  6. An optional final deterministic local refinement (mass‑transfer hill‑climb).

All phases respect the supplied time budget (default 1000 s) and the
resulting sequence is projected onto the simplex (sum fixed to sqrt(2 n))
and clipped to non‑negative values before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used in mirror & Adam phases)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution entry.

    Returns
    -------
    grad : np.ndarray
        Gradient with respect to a (shape n).
    b : np.ndarray
        Full convolution a*a (length 2·n‑1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    grad = np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                        np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = 2.0 * grad[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDesc it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_softmax_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 1.0
    max_beta = 65536.0
    base_lr = 0.12

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = np.maximum(a, 0.0)
        a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamSoftmax it={t:5d} β={beta:6.1f} score={best_val:.6f}")

        # slowly increase the soft‑max sharpness
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.8, max_beta)

        # jitter to escape shallow basins
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – cheap O(n) simulated annealing (mass transfer)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Simple SA that moves small mass between two indices, updating the
    convolution in O(n) time. Accepts occasional worsening moves."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = float(2.0 * n * b.max() / (target_sum ** 2))
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    T0 = max(0.05, (best_val - 1.0) * 0.5)  # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # pick donor weighted by its mass (more likely to influence the max)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue

        # pick a distinct receiver uniformly
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.05, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # compute the updated convolution (copy‑on‑write)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum ** 2))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # occasional global jitter to shake out of plateaus
        if (int(elapsed * 100) % 5000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = float(2.0 * n * b_j.max() / (target_sum ** 2))
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – exact local mass‑transfer refinement (deterministic hill‑climb)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing / simulated‑annealing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = float(2.0 * n * b.max() / (target_sum ** 2))

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04      # initial temperature
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[k_max - idx]

        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # pick a "weak" partner – low contribution to the maximal entry
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # update convolution (copy‑on‑write)
        b_new = b.copy()
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum ** 2))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic re‑normalisation to curb numerical drift
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = float(2.0 * n * b.max() / (target_sum ** 2))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional tiny random shuffle
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = float(2.0 * n * b.max() / (target_sum ** 2))
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Shuffle improved score={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on remaining budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Warm‑start: explore a few simple parametric families
    # ------------------------------------------------------------------
    candidates = []

    # power‑law exponents (coarse + fine)
    coarse_exp = np.linspace(-0.80, -0.30, 11)
    best_exp, best_exp_score = None, np.inf
    for e in coarse_exp:
        seq = (np.arange(1, n + 1, dtype=np.float64) ** e)
        seq = seq * (target_sum / seq.sum())
        candidates.append(seq)
        val = _score(seq)
        if val < best_exp_score:
            best_exp_score = val
            best_exp = e

    # fine search around the best coarse exponent
    if best_exp is not None:
        fine_exp = np.linspace(best_exp - 0.05, best_exp + 0.05, 11)
        for e in fine_exp:
            seq = (np.arange(1, n + 1, dtype=np.float64) ** e)
            seq = seq * (target_sum / seq.sum())
            candidates.append(seq)

    # re‑use the best known construction if available
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=np.float64)
        seq = _resize_sequence(old, n)
        seq = seq * (target_sum / seq.sum())
        candidates.append(seq)

    # keep the three best candidates as starting points
    scored = [(float(_score(c)), c.copy()) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_seqs = scored[:3]

    best_seq = top_seqs[0][1]
    best_score = top_seqs[0][0]
    print(f"[{time.time():.2f}] Initial best score after warm‑start = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Short exponential‑mirror descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + min(0.05 * (deadline - time.time()), 20.0)
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, phase_deadline)
        if val_exp < best_score:
            best_seq, best_score = a_exp, val_exp
            print(f"[{time.time():.2f}] After exponential descent best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam descent on soft‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + min(0.10 * (deadline - time.time()), 60.0)
        a_adam, val_adam = _adam_softmax_descent(best_seq, target_sum, phase_deadline)
        if val_adam < best_score:
            best_seq, best_score = a_adam, val_adam
            print(f"[{time.time():.2f}] After Adam descent best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Several brief SA restarts (exploration)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    num_restarts = 4
    for i in range(num_restarts):
        if time.time() >= deadline:
            break
        # jitter the current best to obtain a fresh start
        rng = np.random.default_rng()
        jitter = rng.normal(scale=target_sum * 0.01, size=best_seq.shape)
        init = best_seq + jitter
        init = np.maximum(init, 0.0)
        init = init * (target_sum / init.sum())

        slice_time = min(30.0, (deadline - time.time()) * 0.2)
        a_sa, val_sa = _simulated_annealing(init, target_sum, time.time() + slice_time)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
            print(f"[{time.time():.2f}] After SA restart {i} best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Long SA phase (intensive exploitation)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
            print(f"[{time.time():.2f}] After final SA best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional final deterministic local refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() + 4.0 < deadline:
        a_loc, val_loc = _local_refine(best_seq, target_sum, deadline - 4.0)
        if val_loc < best_score:
            best_seq, best_score = a_loc, val_loc
            print(f"[{time.time():.2f}] After final local refine best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing zeros, convert to list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Final evaluation (external verifier)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
