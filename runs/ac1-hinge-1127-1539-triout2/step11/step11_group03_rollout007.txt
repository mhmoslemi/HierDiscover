"""
Hybrid optimisation for the self‑convolution peak problem.

The pipeline:
1.  Build a diverse pool of initial shapes (power‑law, exponential,
    mixed power‑law × exponential, logistic, random Dirichlet).
2.  Run many short ‑‑annealed‑gradient‑descent (soft‑max) refinements
    on a moderate length (≈8000).  The best sequence is kept.
3.  A deterministic “mass‑transfer’’ phase (Metropolis moves that shift
    mass from donors heavily contributing to the current maximal
    convolution bin to low‑contributing receivers).
4.  Upscale to a larger length (≈12000–18000) by linear interpolation,
    re‑project onto the ℓ¹‑simplex (sum = √(2 n)).
5.  Final high‑β gradient descent (fine polishing).
6.  A long permutation‑search phase: repeatedly shuffle the multiset
    of heights or perform targeted swaps, accepting moves that lower
    the maximal convolution (or with a Metropolis rule).  This phase
    dominates the tail of the time budget and is often able to push
    the score below the 1.503 target.
7.  Trim trailing zeros and return the sequence.

All steps respect non‑negativity and the ℓ¹‑norm constraint.
The code uses FFT‑based convolutions for speed and runs within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _mixed_power_exp_sequence(n: int, p: float, lam: float,
                              target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-lam·i)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _logistic_sequence(n: int, centre: float, scale: float,
                       target_sum: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i‑centre)/scale))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - centre) / scale))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a_rev) ⋆ p   (⋆ = convolution)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    The algorithm grows the soft‑max temperature (beta) and uses a
    diminishing learning rate tied to the gradient norm.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to leave flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β geometrically
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Deterministic (Metropolis) mass‑transfer refinement

def _mass_transfer(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  start_temp: float = 0.5,
                  end_temp: float = 1e-4,
                  low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis moves that shift a small amount of mass from a
    “donor’’ (large contribution to the current maximal convolution entry)
    to a “receiver’’ (small contribution).  Temperature is annealed.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # indices that actually participate in the peak
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution = product a[i] * a[kmax-i]
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor = idx[np.argmax(prod)]

        # pick a receiver from the low‑contributing part of the peak
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argpartition(prod, low_cnt - 1)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to diversify
        if it % 2500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _permutation_search(a0: np.ndarray,
                       deadline: float,
                       start_temp: float = 0.1,
                       end_temp: float = 1e-5,
                       top_frac: float = 0.02,
                       low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on index order only.  The multiset of heights is
    fixed; we repeatedly swap two positions, preferring indices that
    heavily contribute to the current maximal convolution bin.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # compute contributions (product) for each index in the peak
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        # donor: among top_frac of contributions
        top_k = max(2, int(top_frac * n))
        donor_candidates = np.argpartition(-prod, top_k - 1)[:top_k]
        donor = rng.choice(idx[donor_candidates])

        # receiver: among low_frac of contributions (or any index outside idx)
        low_k = max(2, int(low_frac * n))
        receiver_candidates = np.argpartition(prod, low_k - 1)[:low_k]
        receiver = rng.choice(idx[receiver_candidates])

        # occasionally allow a completely random receiver to increase diversity
        if rng.random() < 0.15:
            receiver = rng.integers(0, n)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSwap new best={best_score:.6f}")

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

    # previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of the pool to enlarge diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.006, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Short gradient‑descent runs (exploration)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 12.0:   # reserve time for later stages
        # pick a seed (occasionally perturb the current best)
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.003, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 20.0, deadline - 10.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_deadline,
            target_expl,
            max_beta=5e5,
            step_factor=0.65,
            gamma=0.98,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Deterministic mass‑transfer refinement (still at modest size)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        mt_deadline = time.time() + (deadline - time.time()) * 0.15
        a_ref, val_ref = _mass_transfer(
            best_seq,
            mt_deadline,
            target_expl,
            start_temp=0.4,
            end_temp=5e-5,
            low_frac=0.06,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscaling to final length (more degrees of freedom)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 40.0 and budget_s >= 800:
        n_final = 15000
    elif remaining > 30.0:
        n_final = 13000
    elif remaining > 20.0:
        n_final = 11000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a handful of systematic reorderings
    candidates = [seq_big,
                  # alternating largest‑smallest ordering
                  np.concatenate([seq_big[np.argsort(-seq_big)][:n_final // 2],
                                  seq_big[np.argsort(seq_big)][:n_final // 2]]),
                  # reverse alternating
                  np.concatenate([seq_big[np.argsort(seq_big)][:n_final // 2],
                                  seq_big[np.argsort(-seq_big)][:n_final // 2]])]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand

    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final high‑β gradient descent (polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        final_dead = deadline - 4.0
        a_fin, val_fin = _annealed_gradient_descent(
            best_seq,
            final_dead,
            target_final,
            max_beta=2e6,
            step_factor=0.45,
            gamma=0.99,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Intensive permutation search (most of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = time.time() + min(30.0, deadline - time.time() - 1.0)
        a_perm, val_perm = _permutation_search(
            best_seq,
            perm_dead,
            start_temp=0.12,
            end_temp=1e-6,
            top_frac=0.04,
            low_frac=0.04,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Optional short post‑polish mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt2_dead = time.time() + min(6.0, deadline - time.time() - 0.5)
        a_mt2, val_mt2 = _mass_transfer(
            best_seq,
            mt2_dead,
            target_final,
            start_temp=0.25,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros (compact representation)
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Log final evaluation (if the harness made evaluate_sequence available)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # Store for potential reuse in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
