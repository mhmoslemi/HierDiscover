    vals = np.sort(seq)[::-1]
    n = len(seq)
    out = np.empty(n)
    idx = 0
    def place(lo, hi):
        nonlocal idx
        if lo > hi:
            return
        mid = (lo + hi) // 2
        out[mid] = vals[idx]; idx += 1
        # fill left, right recursively
        place(lo, mid - 1)
        place(mid + 1, hi)
    place(0, n-1)
    return out

Improved step‑function optimizer for the self‑convolution bound.

Algorithm overview
------------------
1. **Candidate generation** – a diverse pool of power‑law, exponential and
   random ℓ¹‑simplex points for a modest size `n_expl`.
2. **Fast continuous optimisation** – a few short Adam‑style runs on a
   soft‑max approximation of the peak convolution.  The result is projected
   onto the ℓ¹‑simplex after each step.
3. **Upscaling** – the best low‑dimensional vector is interpolated to a larger
   problem size `n_final` (chosen according to remaining time) and several

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    # smallest power of two ≥ 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

    if rho.size == 0:

    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (spread large values uniformly)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Place large values on a cyclic sequence with a step ≈√n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _balanced_order(seq: np.ndarray) -> np.ndarray:
    """Recursive median placement – spreads large coefficients across the domain."""
    vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty(n, dtype=vals.dtype)
    idx = 0

    def place(lo: int, hi: int):
        nonlocal idx
        if lo > hi:
            return
        mid = (lo + hi) // 2
        out[mid] = vals[idx]
        idx += 1
        place(lo, mid - 1)
        place(mid + 1, hi)

    place(0, n - 1)
    return out


def _random_step_order(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random coprime step ordering."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = rng.integers(1, n)
    while math.gcd(step, n) != 1:
        step = rng.integers(1, n)
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max approximation
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2·(a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_projected_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 5e5,
                           step_factor: float = 0.7,
                           beta_schedule: float = 2.0,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam‑style momentum projected GD on the soft‑max surrogate.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = 0.5
    t = 1  # iteration counter for bias correction

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={t:5d} β={beta:.1f} score={best_val:.6f}")

        # increase beta occasionally (soft‑max sharpens)
        if t % 400 == 0 and beta < max_beta:
            beta = min(beta * beta_schedule, max_beta)

        t += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (projected) on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining the max (tolerance is generous)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + eps)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated annealing)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves with Metropolis acceptance and exponential temperature decay.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        # exponential temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The function adheres to the 1000 s time budget and returns the best
    sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – exploratory low‑dimensional optimisation
    # ------------------------------------------------------------------
    # Choose a moderate size for quick experimentation.
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 400:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    # Build an initial pool
    init_pool = []

    for p in np.linspace(0.2, 1.4, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    for lam in np.linspace(0.001, 0.04, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    for _ in range(8):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # Add previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # Jittered copies for diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # Run a few short Adam‑style runs until a fraction of the budget is consumed
    phase1_dead = start_time + 0.30 * (deadline - start_time)
    while time.time() < phase1_dead:
        # pick seed
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 12.0, phase1_dead)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _adam_projected_descent(
            seed, run_dead, target_expl,
            max_beta=5e5, step_factor=0.7, beta_schedule=2.0
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Phase1 new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscaling & reordering
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # Choose final length based on remaining time
    remaining = deadline - time.time()
    if remaining > 120.0:
        n_final = 20000
    elif remaining > 70.0:
        n_final = 16000
    elif remaining > 35.0:
        n_final = 12000
    else:
        n_final = n_expl
    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Evaluate several deterministic reorderings
    reorderings = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
        _balanced_order(seq_big)
    ]

    # Add a few random step reorderings
    for _ in range(3):
        reorderings.append(_random_step_order(seq_big, rng))

    seq_big = min(reorderings, key=_max_convolution_fft)
    best_seq = seq_big
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient refinement
    # ------------------------------------------------------------------
    phase3_dead = deadline - 8.0  # keep a tiny buffer for final polish
    if time.time() < phase3_dead:
        a_sub, val_sub = _subgradient_descent(
            best_seq, phase3_dead, target_final,
            step_factor=0.45, gamma=0.995
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] After SubGrad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final deterministic mass‑transfer polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim negligible tail for compactness and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Store for potential reuse in a later run
    globals()["height_sequence_1"] = final_seq

    # Optional logging of the final evaluated value
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
