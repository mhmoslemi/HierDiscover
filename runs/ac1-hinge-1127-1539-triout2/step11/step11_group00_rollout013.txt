def propose_candidate(budget_s=1000, **kwargs):

    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine n according to budget.
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # Create initialization sequences
    inits = []

    # Warm start if available
    if "height_sequence_1" in globals():
        ws = np.array(globals()["height_sequence_1"], dtype=np.float64)
        ws = _resize_sequence(ws, n)
        inits.append(ws)

    # power laws
    for alpha in np.linspace(0.4, 0.7, 7):
        seq = (np.arange(1, n+1) ** (-alpha))
        inits.append(seq)

    # exponential
    for beta in [0.0005, 0.001, 0.002]:
        seq = np.exp(-beta * np.arange(n))
        inits.append(seq)

    # sine power
    for gamma in [0.5, 1.0, 2.0]:
        seq = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** gamma
        inits.append(seq)

    # piecewise linear (tent shape)
    x = np.linspace(0, 1, n)
    seq = 1 - np.abs(x - 0.5) * 2  # triangle of height 1
    inits.append(seq)

    # Remove duplicates and zero arrays.
    # Normalize each to target sum
    init_normed = []
    for seq in inits:
        if seq is None:
            continue
        seq = seq.astype(np.float64)
        seq = seq.clip(min=0)  # ensure non-negative
        total = seq.sum()
        if total <= 0:
            continue
        seq = seq * (target_sum / total)
        init_normed.append(seq)

    # Deduplicate near identical arrays (not needed).

    best_seq = None
    best_val = np.inf

    # Allocate small fraction of time for quick refinement of each init.
    # We will give each init a slice of time proportional until half of budget used.
    remaining = deadline - time.time()
    per_init_time = min(2.0, remaining * 0.3 / max(1, len(init_normed)))  # at most 2 sec each

    # For each init, run a short subgradient descent or exponential descent
    for seq in init_normed:
        sub_deadline = time.time() + per_init_time
        if time.time() >= deadline:
            break
        # run a few steps of exponential descent (mirror) with softmax but limited.
        a_opt, val_opt = _exponential_descent(seq, target_sum, sub_deadline)
        # after that test with evaluate_sequence
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
        # also try short run of subgradient descent
        if time.time() < deadline:
            sub_deadline2 = time.time() + per_init_time
            a_opt2, val_opt2 = _subgradient_descent(best_seq, target_sum, sub_deadline2)
            if val_opt2 < best_val:
                best_val = val_opt2
                best_seq = a_opt2.copy()
        # Print.
        print(f"[{time.time():.2f}] Init refined best={best_val:.6f}")

    # If none found (unlikely), fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = _score(best_seq)

    # Final annealing
    if time.time() < deadline:
        a_final, val_final = _smart_mass_transfer_anneal(best_seq, target_sum, deadline)
        if val_final < best_val:
            best_seq, best_val = a_final, val_final
        print(f"[{time.time():.2f}] Final anneal best score={best_val:.6f}")

    # Trim trailing zeros
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1]+1]
    final_seq = arr.tolist()

    # Evaluate final
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error: {e}")

    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Improved step‑function search for the max‑self‑convolution benchmark.

Main ideas
----------
* Initialise from a handful of simple analytic shapes (power‑law,
  exponential, sine‑powered, triangular).  If a previous best
  (`height_sequence_1`) exists it is also used.
* Normalise every candidate to the required sum √(2 n).
* For each candidate run a very short mirror‑descent on a soft‑max
  surrogate – this quickly drives the peak down without any heavy
  optimisation machinery.
* The final (and longest) phase is a **deterministic mass‑transfer
  anneal**:
    – locate the current peak kₘₐₓ of the convolution,
    – pick a *donor* index among those that contribute most to the peak,
    – pick a *receiver* that contributes the least (ideally an index that
      does not affect the peak at all),
    – move a small amount of mass Δ from donor to receiver.
  The amount Δ is chosen analytically from the current gap between the
  peak and the second‑largest value, scaled by a linear temperature
  schedule.
* The convolution is updated in O(n) using the exact slice‑wise
  formula from the reference implementation, and a full recomputation
  is performed every 10 s to avoid drift.
* The best sequence found is trimmed, stored globally as
  `height_sequence_1` and returned.

The routine respects the supplied time budget (default 1000 s) and
never returns an empty list – if something goes wrong the caller will
receive `np.inf` from `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear self‑convolution)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective: 2·n·max(a*a) / (sum(a)²).  With sum = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    mx = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * mx / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used for a brief mirror‑descent warm‑up)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Mirror (multiplicative) descent on the soft‑max surrogate.
    Very short – we merely use it as a cheap pre‑processor.
    """
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.03
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Smart mass‑transfer simulated‑annealing
# ----------------------------------------------------------------------


def _smart_mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Deterministic mass‑transfer anneal.
    At each step we move a little mass from the index that contributes most
    to the current peak towards an index that contributes the least.
    The size of the move is chosen from the current gap between the peak
    and the second‑largest convolution value, scaled by a linear temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.30                  # start temperature ≈30 % of current score

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # --- locate the current peak ---------------------------------
        k_max = int(np.argmax(b))
        M = b[k_max]

        # second‑largest entry (used to set a reasonable Δ)
        # (if there are multiple equal maxima the second value may be the same)
        if b.size > 1:
            # fast way to get the second maximum without a full copy
            idx_two = np.argpartition(b, -2)[-2:]
            if idx_two[0] == k_max:
                M2 = b[idx_two[1]]
            elif idx_two[1] == k_max:
                M2 = b[idx_two[0]]
            else:
                M2 = max(b[idx_two[0]], b[idx_two[1]])
        else:
            M2 = 0.0

        # indices that actually participate in this peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partners = k_max - idx                     # partner indices for each idx

        # contributions a[i] * a[partner]
        contrib = a[idx] * a[partners]

        # donor: index with largest contribution to the peak
        donor = idx[np.argmax(contrib)]

        # receiver: index that (almost) does NOT affect the peak
        mask = np.ones(n, dtype=bool)
        mask[idx] = False
        if mask.any():
            receiver = rng.choice(np.flatnonzero(mask))
        else:
            # every index participates – pick one whose partner is smallest
            receiver = idx[np.argmin(a[partners])]

        # partner values for donor / receiver (needed for the effect on the peak)
        p_donor = a[k_max - donor] if 0 <= k_max - donor < n else 0.0
        p_recv = a[k_max - receiver] if 0 <= k_max - receiver < n else 0.0

        # if moving mass from donor to receiver would raise the peak, fall back
        # to a purely random move (the Metropolis test will reject it most of the time)
        if p_donor <= p_recv:
            # random move – keep the same Δ logic as the original code
            donor = idx[np.argmax(contrib)]
            receiver = rng.integers(n)
            if donor == receiver:
                continue

        # maximum feasible transfer this step
        max_delta = min(a[donor] * 0.04, target_sum * 0.001)
        if max_delta <= 0.0:
            continue

        # desired reduction of the peak, based on the current gap
        gap = M - M2
        reduction_per_unit = 2.0 * (p_donor - p_recv)
        if reduction_per_unit > 0.0:
            delta_target = gap / reduction_per_unit
        else:
            delta_target = max_delta

        # combine deterministic target with temperature scaling
        if T0 > 0.0:
            delta = min(max_delta, delta_target) * (temperature / T0)
        else:
            delta = min(max_delta, delta_target)

        if delta <= 0.0:
            continue

        # --- exact O(n) update of the convolution --------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(new_max)               # because Σa = √(2 n)

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SmartAnneal improved score={best_val:.6f}")

        # periodic full recomputation (every ~10 s) to avoid numerical drift
        if time.time() - last_full > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")
            last_full = time.time()

        # occasional tiny jitter to keep dynamics alive
        if it % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.004, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more flexibility, but costs time)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build a small library of initial shapes
    # ------------------------------------------------------------------
    init_shapes = []

    # 1) warm‑start from previous best if available
    if "height_sequence_1" in globals():
        ws = np.array(globals()["height_sequence_1"], dtype=np.float64)
        ws = _resize_sequence(ws, n)
        init_shapes.append(ws)

    # 2) simple decreasing power‑laws
    for alpha in np.linspace(0.45, 0.70, 6):
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
        init_shapes.append(seq)

    # 3) exponential decays
    for beta in (0.0003, 0.0006, 0.0012):
        seq = np.exp(-beta * np.arange(n, dtype=np.float64))
        init_shapes.append(seq)

    # 4) sine‑powered shapes
    x = np.arange(n, dtype=np.float64) + 0.5
    for gamma in (0.5, 1.0, 1.5):
        seq = np.sin(np.pi * x / n) ** gamma
        init_shapes.append(seq)

    # 5) symmetric triangular bump
    t = np.linspace(0.0, 1.0, n)
    tri = 1.0 - np.abs(t - 0.5) * 2.0
    init_shapes.append(tri)

    # ------------------------------------------------------------------
    # Normalise all candidates to the required sum
    # ------------------------------------------------------------------
    candidates = []
    for seq in init_shapes:
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total < 1e-12:
            continue
        seq = seq * (target_sum / total)
        candidates.append(seq)

    # ------------------------------------------------------------------
    # Quick mirror‑descent warm‑up for each candidate (very cheap)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    time_per_init = min(1.5, (deadline - time.time()) * 0.15 / max(1, len(candidates)))

    for seq in candidates:
        if time.time() >= deadline:
            break
        sub_dead = time.time() + time_per_init
        a_opt, val_opt = _exponential_descent(seq, target_sum, sub_dead)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
        print(f"[{time.time():.2f}] Warm‑up best={best_val:.6f}")

    if best_seq is None:                         # fallback – uniform
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_val = _score(best_seq)

    # ------------------------------------------------------------------
    # Main optimisation phase – deterministic mass‑transfer anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, val_final = _smart_mass_transfer_anneal(best_seq, target_sum, deadline)
        if val_final < best_val:
            best_seq, best_val = a_final, val_final
        print(f"[{time.time():.2f}] After smart anneal best score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up, trimming trailing zeros and final evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
