def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5
    # Determine n_expl based on time left
    remaining_budget = deadline - time.time()
    if remaining_budget > 900:
        n_expl = 12000
    elif remaining_budget > 600:
        n_expl = 10000
    else:
        n_expl = 8000
    target_expl = np.sqrt(2.0 * n_expl)
    rng = np.random.default_rng()
    # Build init_pool
    init_pool = []
    # add power law
    for p in np.linspace(0.25, 0.75, 11):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    # exponential:
    for lam in np.linspace(0.0005, 0.02, 11):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    # random Dirichlet
    for _ in range(10):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # linear decreasing
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jitter copy
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # Exploration: run many gradient desc runs
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 30:  # keep 30 sec for later phases
        # choose seed
        if best_seq is not None and rng.random() < 0.7:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.002, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)
        # allocate run time per gradient
        run_dead = min(time.time() + 20.0, deadline - 30.0)
        if run_dead <= time.time():
            break
        a_opt, val_opt = _annealed_gradient_descent(seed, run_dead, target_expl,
                                                    max_beta=3e6,
                                                    step_factor=0.6,
                                                    gamma=0.98)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # deterministic mass transfer refinement
    if best_seq is not None and time.time() < deadline - 20:
        mt_dead = time.time() + (deadline - time.time()) * 0.10
        a_ref, val_ref = _deterministic_mass_transfer(best_seq, mt_dead, target_expl,
                                                      start_temp=0.7, end_temp=1e-5, low_frac=0.05)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] MT refinement best={best_score:.6f}")

    # Upscale
    # decide final n
    remaining = deadline - time.time()
    if remaining > 200 and budget_s >= 800:
        n_final = 24000
    elif remaining > 100:
        n_final = 18000

"""Optimiser for the step‑function benchmark.

The optimiser proceeds in several stages that respect the ℓ¹‑simplex
constraint (non‑negative entries and fixed sum √(2 n)):

1.  Build a diverse pool of initial candidates (power‑law, exponential,
    random Dirichlet, linear, and a jittered copy of the previous best).
2.  Run many short annealed‑gradient‑descent runs on a moderate size
    (≈ 8‑12 k).  The soft‑max approximation of the max‑convolution is used
    together with Euclidean projection onto the simplex.
3.  Apply a cheap deterministic “mass‑transfer’’ refinement that moves
    a small amount of mass from indices that heavily contribute to the
    current peak to indices that contribute little.
4.  Upscale the best sequence to a larger length (≈ 18‑24 k) by linear
    interpolation and try a few systematic reorderings.
5.  A long high‑β gradient descent polish.
6.  A targeted simulated‑annealing permutation search that swaps
    donor/receiver indices based on their contribution to the peak.
7.  A final tiny mass‑transfer pass and trimming of trailing zeros.

The whole pipeline is bounded by the supplied time budget (default
1000 s).  Intermediate scores are printed so the progress can be
monitored.  The resulting sequence is stored in the global variable
``height_sequence_1`` for possible reuse."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a)/(Σa)²  (the benchmark score)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) shape, renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """exp(-lam·i) shape, renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest leftmost, second‑largest rightmost, alternating."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Mirror of _alternating_order."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv) where grad has shape a.shape.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # prevent overflow by subtracting max_b before exponentiation
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)               # soft‑max weights (sum to 1)

    # grad = 2 * (a reversed) ⋆ p   (⋆ = convolution)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Annealed projected gradient descent (soft‑max version)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max surrogate.
    The temperature (beta) is increased geometrically.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:6d} β={beta:6.1f} score={best_val:.6f}")

        # increase beta slowly
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.6, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a small amount of mass from a “donor’’ (large
    contribution to the peak) to a “receiver’’ (small contribution).
    Moves are accepted with a Metropolis rule whose temperature decays
    exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current peak of the convolution
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak (product with its partner)
        w = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        receiver = rng.choice(idx[np.argsort(w)[:low_cnt]])

        if donor == receiver:
            it += 1
            continue

        # move at most half of the donor’s mass
        max_delta = a[donor] * 0.5
        # also respect the disparity of contributions
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:6d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 3000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style re‑ordering of the entries.  The ℓ¹‑sum
    stays unchanged, only the order is shuffled.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # contribution of each position to the current peak
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        # swap
        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        # temperature schedule (log‑linear)
        tau = (time.time() - start_time) / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score
