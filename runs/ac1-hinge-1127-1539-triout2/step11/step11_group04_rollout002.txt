"""
Hybrid search for a low‑peak self‑convolution step function.

The algorithm follows three stages:

1. **Exploration (small n)** – many short annealed‑gradient runs on a
   modest length (≈ 5‑8k).  A diverse pool (power‑law, exponential,
   random simplex and jittered copies of the current best) seeds the
   runs.  The optimizer works on a soft‑max approximation of the
   convolution peak and gradually raises the sharpening parameter β.

2. **Upscaling** – the best small‑scale solution is interpolated to a
   larger length (up to 32 k if time permits).  Several deterministic
   reorderings (alternating, even‑spread, bit‑reversal) are tried and
   the one with the smallest convolution peak is kept.

3. **Polishing (large n)** – a few passes of exact sub‑gradient descent,
   a greedy mass‑transfer refinement and a light Gaussian smoothing are
   applied.  The objective is invariant under scaling, therefore the
   ℓ¹‑norm is kept equal to √(2·n) and the score coincides with the
   maximal convolution entry.

All steps respect non‑negativity and the ℓ¹‑simplex constraint.
The function returns the best sequence found within the allotted
budget (default 1000 s).  The sequence is also stored in the global
variable `height_sequence_1` for reuse in subsequent runs.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Reordering helpers

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime to the length."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _bit_reversal_order(seq: np.ndarray) -> np.ndarray:
    """Place values according to bit‑reversed indices (good uniform spread)."""
    n = seq.shape[0]
    # next power of two for a clean bit‑reversal; if n not power of two we
    # simply reverse the bits modulo the smallest power ≥ n.
    bits = (n - 1).bit_length()
    idx = np.arange(n, dtype=np.int64)
    rev = np.zeros_like(idx)
    for b in range(bits):
        rev = (rev << 1) | ((idx >> b) & 1)
    rev = rev & ((1 << bits) - 1)
    # Since rev may contain indices ≥ n, wrap them back.
    rev = rev % n
    order = np.argsort(rev)  # order of original positions after reversal
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    out[order] = sorted_vals
    return out


def _random_permutation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """A simple random shuffle (kept for occasional diversity)."""
    out = seq.copy()
    rng.shuffle(out)
    return out


# ----------------------------------------------------------------------
# Soft‑max annealed gradient descent (uses the smooth approximation of max)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2·(a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices where the maximum is attained (numerical tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 800 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (Metropolis‐type)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """Greedy mass transfers with a decaying temperature."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        # compute temperature
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        n = a.shape[0]
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if it % 2500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Light Gaussian smoothing (keeps the ℓ¹‑norm)
# ----------------------------------------------------------------------


def _smooth_sequence(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Convolve with a normalized Gaussian kernel of given sigma."""
    if sigma <= 0:
        return seq
    # kernel size ≈ 6σ, ensure it's odd
    radius = int(np.ceil(3 * sigma))
    x = np.arange(-radius, radius + 1)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    # use mode='same' via FFT for speed
    size = 1 << ((len(seq) + len(kernel) - 1).bit_length())
    fseq = np.fft.rfft(seq, n=size)
    fker = np.fft.rfft(kernel, n=size)
    smooth = np.fft.irfft(fseq * fker, n=size)[: len(seq)]
    return smooth


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low self‑convolution peak.
    The routine respects the 1000 s time budget (default) and returns a list
    of floats.  The best sequence is also stored in the global variable
    ``height_sequence_1`` for reuse in later runs.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Choose the exploratory size
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n_expl = 8000
    elif budget_s >= 600:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build an initial pool of candidates
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law curves
    for p in np.linspace(0.2, 1.0, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential curves
    for lam in np.linspace(0.001, 0.02, 6):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random points on the simplex
    for _ in range(8):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # if a previous best exists, include its resized version
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies to enrich diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Exploration with many short annealed‑gradient runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 12.0:   # reserve time for polishing
        # pick a seed (either a jittered copy of current best or a random pool entry)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 18.0, deadline - 10.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_dead, target_expl,
            max_beta=8e5, step_factor=0.68, gamma=0.99)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Brief deterministic mass‑transfer refinement at the small scale
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.12
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_expl,
            start_temp=0.45, end_temp=5e-5, low_frac=0.07)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Post‑MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Upscaling to a larger length (if time permits)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 200.0:
        n_final = 32000
    elif remaining > 120.0:
        n_final = 24000
    elif remaining > 60.0:
        n_final = 16000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try several deterministic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
        _bit_reversal_order(seq_big),
        _random_permutation(seq_big, rng)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)

    best_seq = seq_big
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient polishing on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sub_dead = deadline - 6.0
        a_sub, val_sub = _subgradient_descent(
            best_seq, sub_dead, target_final,
            step_factor=0.45, gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic mass‑transfer + optional smoothing
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # a tiny smoothing pass (helps regularise sharp spikes)
    if time.time() < deadline - 2.0:
        sigma = 1.2  # empirical value; very mild
        smoothed = _smooth_sequence(best_seq, sigma)
        smoothed = _project_to_simplex(smoothed, target_final)
        smooth_score = _score(smoothed)
        if smooth_score < best_score:
            best_score = smooth_score
            best_seq = smoothed.copy()
            print(f"[{time.time():.2f}] Smoothed best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional logging of the true evaluation (if available)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
