"""
Advanced step‑function optimizer.

The routine builds a non‑negative sequence `a` of length *n* (chosen
automatically from the time budget) with Σa = √(2 n).  It proceeds in
three stages:

1. **Warm‑start selection** – try a few power‑law initialisations and
   optionally resize the best known global construction.
2. **Mirror descent on a soft‑max surrogate** – a few hundred
   multiplicative updates quickly reduce the peak of the self‑convolution.
3. **Mass‑transfer simulated annealing** – repeatedly moves a small
   amount of mass from the index that contributes most to the current
   convolution peak to an index with (almost) no contribution.
   The temperature starts high (≈ 0.8·current best) and cools linearly.
   The step size is deliberately larger than in the reference runner
   (up to 50 % of the donor’s mass) to explore the space more aggressively.

The best sequence found before the deadline is returned as a plain list
of floats.  The final sequence is also stored in the global variable
`height_sequence_1` for warm‑starting future runs.
"""

# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in the mirror descent phase)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_anneal_improved(a0: np.ndarray, target_sum: float, deadline: float):
    """Long‑run annealing that moves relatively large chunks of mass
    from the index contributing most to the current peak towards an
    index that contributes little (or nothing) to that peak.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                       # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.80                  # a hotter start than the reference code
    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor = index with largest contribution to the peak
        donor = idx[np.argmax(contrib)]

        # receiver = index outside the overlapping window with smallest height
        mask_outside = np.ones(n, dtype=bool)
        mask_outside[lo:hi + 1] = False
        if mask_outside.any():
            receiver = int(np.argmin(np.where(mask_outside, a, np.inf)))
        else:
            receiver = int(np.argmin(a))

        if donor == receiver:
            continue

        # size of the transfer – significantly larger than before
        max_delta = min(a[donor] * 0.50, target_sum * 0.05)   # up to 50 % of donor or 5 % of total mass
        if max_delta <= 0:
            continue

        # scale with temperature (linear cooling)
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        # --- exact O(n) update of the convolution (copy‑on‑write) ----------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it:6d} score={best_val:.6f}")

        # periodic exact recompute to keep numerical errors in check
        if time.time() - last_full > 5.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")
            last_full = time.time()

        # occasional tiny random jitter to keep dynamics alive
        if it % 3000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.001)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 30000
    elif budget_s >= 600:
        n = 25000
    elif budget_s >= 400:
        n = 18000
    elif budget_s >= 200:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start selection: try a few power‑law exponents and the
    # previously stored construction (if any).  Keep the one with the
    # lowest surrogate score.
    # --------------------------------------------------------------
    candidates = []

    # power‑law candidates
    for p in (0.35, 0.45, 0.55, 0.65, 0.75):
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        seq = seq * (target_sum / seq.sum())
        candidates.append(seq)

    # previously known construction
    if "height_sequence_1" in globals():
        seq = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
        seq = seq * (target_sum / seq.sum())
        candidates.append(seq)

    # pick the best candidate according to the true objective
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    print(f"[{time.time():.2f}] Initial best score after seeding = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – short exponential (mirror) descent on the soft‑max surrogate
    # --------------------------------------------------------------
    phase_a_deadline = time.time() + 0.10 * (deadline - time.time())
    a, val = _exponential_descent(best_seq, target_sum, phase_a_deadline)
    if val < best_val:
        best_val = val
        best_seq = a.copy()
    print(f"[{time.time():.2f}] After exponential descent best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase D – aggressive mass‑transfer simulated annealing (dominant phase)
    # --------------------------------------------------------------
    a, val = _mass_transfer_anneal_improved(best_seq, target_sum, deadline)
    if val < best_val:
        best_val = val
        best_seq = a.copy()
    print(f"[{time.time():.2f}] After annealing best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up: drop trailing near‑zero entries and evaluate
    # --------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    final_seq = best_seq.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
