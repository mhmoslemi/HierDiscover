def best_spread_order(seq):
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    best_seq = seq
    best_val = _score(seq)
    # try step from 1 to maybe int(2*sqrt(n)) inclusive
    max_step = int(2 * np.sqrt(n)) + 1
    for step_candidate in range(1, max_step):
        step = step_candidate
        # ensure coprime
        if np.gcd(step, n) != 1:
            continue
        # construct ordering
        out = np.empty_like(seq)
        for i, v in enumerate(sorted_vals):
            out[(i * step) % n] = v
        val = _score(out)
        if val < best_val:
            best_val = val
            best_seq = out
    return best_seq, best_val

def _simulated_annealing(a0, deadline, target_sum):

    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    # initial temperature maybe 0.01 of target_sum??; we can adapt.
    T0 = max(a) * 0.1  # maybe consider typical scale.
    # compute exponent schedule
    while time.time() < deadline:
        # compute remaining fraction
        frac = (deadline - time.time()) / (deadline - start_time)  # we need start_time param
        # Set temperature decreasing linearly.
        T = T0 * frac
        # propose move
        # choose donor idx with a[i] > epsilon
        donor = rng.choice(np.where(a > 1e-12)[0])
        receiver = rng.integers(len(a))
        if donor == receiver:
            continue
        max_delta = min(a[donor], a[donor] * 0.5)  # up to half donor amount
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # apply
        a[donor] -= delta
        a[receiver] += delta
        # evaluate
        cur_val = _score(a)
        if cur_val < best_val or rng.random() < np.exp(-(cur_val - best_val) / max(T, 1e-12)):
            # accept
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA new best={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
    return best_a, best_val

    a = seq.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    n = a.shape[0]
    # Temperature schedule: start high, linear decay
    T0 = 0.05 * target_sum  # maybe
    while time.time() < deadline:
        # compute fraction of time elapsed
        t_elapsed = time.time() - start_time
        total = deadline - start_time
        frac = 1.0 - t_elapsed / total
        T = T0 * max(frac, 0.01)  # keep minimal
        # propose move: pick indices i, j
        donor = rng.integers(n)
        while a[donor] < 1e-12:

            continue

        if max_delta <= 0:
            continue

        a[receiver] += delta
        # Ensure nonnegativity (should hold)
        cur_val = _score(a)

            best_a = a.copy()

            # Metropolis acceptance
            prob = np.exp(-(cur_val - best_val) / max(T, 1e-12))
            if rng.random() < prob:
                # accept worse
                pass
            else:
                # revert
                a[donor] += delta
                a[receiver] -= delta
        # optional periodic global jitter? not needed.

    return best_a, best_val

  amount of mass between two positions while keeping the ℓ¹‑simplex
  constraint.

All phases respect the non‑negativity and ℓ¹‑simplex constraints required
by ``evaluate_sequence``.  The routine runs for at most the supplied
budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime with the length."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _ordering_sweep(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Try many cyclic spacings of the sorted values and keep the best.
    The step is limited to a few multiples of sqrt(n) to stay cheap.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    best_seq = seq.copy()
    best_val = _score(seq)

    max_step = int(2 * np.sqrt(n)) + 1
    for step in range(1, max_step):
        if math.gcd(step, n) != 1:
            continue
        out = np.empty_like(seq)
        for i, v in enumerate(sorted_vals):
            out[(i * step) % n] = v
        val = _score(out)
        if val < best_val:
            best_val = val
            best_seq = out.copy()
            print(f"[{time.time():.2f}] sweep step={step} new best={best_val:.6f}")
        if time.time() >= deadline:
            break
    return best_seq, best_val


# ----------------------------------------------------------------------
# Annealed gradient descent on the soft‑max approximation
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2·(a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 1e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfers with a simple Metropolis acceptance rule.
    The temperature decays exponentially from start_temp to end_temp.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing move on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(seq: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Redistribute small packets of mass between two positions."""
    a = seq.copy()
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    n = a.shape[0]

    # Temperature scales with the target ℓ¹‑norm
    T0 = 0.03 * target_sum

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = max(0.0, 1.0 - elapsed / total)
        T = max(T0 * frac, 1e-6)

        # pick donor (must have some mass) and receiver
        donor = rng.integers(n)
        while a[donor] < 1e-12:
            donor = rng.integers(n)
        receiver = rng.integers(n)
        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta

        cur_val = _score(a)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SA improved to {best_val:.6f}")
        else:
            prob = np.exp(-(cur_val - best_val) / max(T, 1e-12))
            if rng.random() >= prob:
                # revert
                a[donor] += delta
                a[receiver] -= delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine respects the 1000 second budget and stores the best
    sequence it finds in the global variable ``height_sequence_1``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Small‑scale exploratory phase (several annealed‑GD runs)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)
    rng = np.random.default_rng()

    init_pool = []

    # power‑law family
    for p in np.linspace(0.2, 1.0, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 6):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random simplex points
    for _ in range(6):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # previously discovered sequence, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # Many short GD runs while we have time left
    while time.time() < deadline - 30.0:   # keep ~30 s for later phases
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 18.0, deadline - 12.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seed, run_dead, target_expl,
            max_beta=5e5, step_factor=0.7, gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Upscale to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 180.0:
        n_final = 20000
    elif remaining > 90.0:
        n_final = 16000
    elif remaining > 45.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # ------------------------------------------------------------------
    # Ordering sweep (cheap cyclic spreads)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        sweep_dead = deadline - 20.0
        seq_big, best_score = _ordering_sweep(seq_big, sweep_dead)

    # ------------------------------------------------------------------
    # Sub‑gradient refinement on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        sub_dead = deadline - 15.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_dead, target_final,
            step_factor=0.45, gamma=0.995,
        )
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        mt_dead = deadline - 10.0
        a_mt, val_mt = _deterministic_mass_transfer(
            seq_big, mt_dead, target_final,
            start_temp=0.3, end_temp=1e-5, low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] DMT post‑polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Light simulated‑annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        sa_dead = deadline - 2.0
        a_sa, val_sa = _simulated_annealing(
            seq_big, sa_dead, target_final, start_time=start_time
        )
        if val_sa < best_score:
            best_score = val_sa
            seq_big = a_sa.copy()
            print(f"[{time.time():.2f}] Final SA best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing negligible entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
