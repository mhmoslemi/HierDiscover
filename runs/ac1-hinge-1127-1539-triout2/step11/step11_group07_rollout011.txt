```
def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    best_score = np.inf

    # try to seed if global variable exists
    ...

    schedule = [1500, 3000, 6000, 12000] # increase gradually
    for n in schedule:
        if time.time() >= deadline: break
        target = math.sqrt(2.0 * n)
        # init sequence
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)
        a_init = _project_to_simplex_fast(a_init, target)

        # allocate time
        remaining = deadline - time.time()
        if remaining < 5: break
        opt_time = 0.5 * remaining
        hc_time = 0.4 * remaining
        extra_time = remaining - opt_time - hc_time

        # number of restarts
        n_restarts = max(1, int(opt_time // 10))   # each restart about 10 s
        time_per_restart = opt_time / n_restarts

        local_best = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline: break
            this_deadline = min(deadline - 0.1, time.time() + time_per_restart)

            if r == 0 and best_seq is not None and best_seq.size == n:
                a0 = best_seq.copy()
            else:
                a0 = _random_initial_sequence(n, rng)

            # run Adam smooth max for first half of restart budget
            half_deadline = min(this_deadline, time.time() + time_per_restart * 0.5)
            a_mid = _adam_optimize(a0, target, half_deadline)

            a_opt = _adam_optimize_submax(a_mid, target, this_deadline)

            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best = a_opt.copy()
        # update global best
        if local_best_score < best_score - 1e-12:
            best_seq = local_best.copy()
            best_score = local_best_score
            print(f"Improved at n={n} score={best_score:.6f}")

        # hill climbing
        if time.time() + 0.5 < deadline:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex_fast(a_cur, target)
            b_cur = _full_convolution(a_cur)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, deadline - 0.5)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"Hill climb improved at n={n} score={best_score:.6f}")

        # small random tweaks until time runs out (maybe small step)
        while time.time() + 0.2 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
```

```python
"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence().  The search consists of:

1. Random initialisations from several families (including sparse supports).
2. Projected Adam on a smooth‑max surrogate (log‑sum‑exp).
3. Projected Adam on the true max (sub‑gradient) with a reduced step‑size.
4. Deterministic hill‑climb that moves mass from the most‑contributing
   index to a low‑contributing partner, using an exact O(n) update.
5. A cheap random‑tweak phase (simulated annealing style).
6. A final polishing pass (Adam + hill‑climb).

The routine respects a wall‑clock budget (default 1000 s) and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    The implementation simply clamps to non‑negative and rescales the ℓ¹‑norm.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers (several simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 5:                   # centred Gaussian bump
        sigma = rng.uniform(0.1, 0.3) * n
        centre = rng.integers(0, n)
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)

    elif family == 6:                   # sparse uniform support (≈√n non‑zeros)
        k = max(2, int(np.sqrt(n)))
        chosen = rng.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[chosen] = 1.0

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    # add a tiny jitter for all families (helps escaping degenerate points)
    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev ⋆ p (central n entries)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.12 * target_sum          # a bit softer than the reference
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        # occasional jitter to avoid stagnation
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-5, size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        if t % 250 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is a true sub‑gradient of max(a*a).
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.08 * target_sum          # slightly smaller step for stability
    t = 0

    while time.time() < deadline:
        t += 1

        # full convolution and active mask
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # gradient = 2·(a_rev ⋆ p)  (central n entries)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climb on the exact convolution
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        # indices that contribute to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose a destination j with smallest counterpart
        counterpart = a[M - np.arange(i_low, i_high + 1, dtype=int)]
        # sort indices of the partner side
        order = np.argsort(counterpart)
        j = None
        for cand in np.arange(i_low, i_high + 1, dtype=int)[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # try a relatively aggressive move, then back‑track if necessary
        delta = a[i] * 0.6
        if delta <= 0.0:
            break

        # Helper to compute the new max after a tentative delta
        def _apply_delta(arr_a, arr_b, d, ii, jj):
            # linear‑time update of the convolution (in‑place on a copy)
            b_new = arr_b.copy()
            b_new[ii:ii + n] -= 2.0 * d * arr_a
            b_new[jj:jj + n] += 2.0 * d * arr_a
            k2i = 2 * ii
            k2j = 2 * jj
            kij = ii + jj
            if k2i < b_new.size:
                b_new[k2i] += d * d
            if k2j < b_new.size:
                b_new[k2j] += d * d
            if kij < b_new.size:
                b_new[kij] -= 2.0 * d * d
            return b_new

        b_test = _apply_delta(a, b, delta, i, j)
        new_max = b_test.max()

        if new_max < b.max() - 1e-12:
            # accept the move
            a[i] -= delta
            a[j] += delta
            b = b_test
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            continue

        # otherwise perform a binary‑search on delta
        improved = False
        lo, hi = 0.0, delta
        for _ in range(30):
            mid = (lo + hi) * 0.5
            if mid < 1e-16:
                break
            b_mid = _apply_delta(a, b, mid, i, j)
            if b_mid.max() < b.max() - 1e-12:
                lo = mid
                improved = True
            else:
                hi = mid
        if improved and lo > 1e-16:
            a[i] -= lo
            a[j] += lo
            b = _apply_delta(a, b, lo, i, j)
            cur_max = b.max()
            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_a = a.copy()
            continue

        # fallback: small random jitter to escape a potential plateau
        jitter = rng.normal(scale=1e-6, size=n)
        a += jitter
        a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
        b = _full_convolution(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method combines random families, Adam (smooth‑max and exact‑max),
    an aggressive deterministic hill‑climb and a final polishing stage.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # keep a small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1)  Seed from a previously stored solution if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            target0 = math.sqrt(2.0 * prev.size)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={prev.size}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Length schedule – progressively larger dimensions.
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000, 16000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # initialise: upsample previous best if possible, otherwise random
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex_fast(a_init, target)

        remaining = deadline - time.time()
        if remaining < 15.0:
            break

        # --------------------------------------------------------------
        # 2a)  Time split for this size
        # --------------------------------------------------------------
        opt_budget = remaining * 0.55      # optimisation (Adam)
        hc_budget = remaining * 0.35       # hill‑climb
        tweak_budget = remaining * 0.10    # cheap random tweaks

        # --------------------------------------------------------------
        # 2b)  Adam optimisation phase (multiple restarts)
        # --------------------------------------------------------------
        min_restart = 8.0                  # aim for ~8 s per restart
        n_restarts = max(1, int(opt_budget // min_restart))
        per_restart = opt_budget / n_restarts

        local_best = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)

            # first restart keeps the current best if it already matches n
            if r == 0 and best_seq is not None and best_seq.size == n:
                a_start = best_seq.copy()
            else:
                a_start = _random_initial_sequence(n, rng)

            # smooth‑max Adam (exploratory)
            mid_deadline = min(this_deadline,
                               time.time() + per_restart * 0.45)
            a_mid = _adam_optimize(a_start, target, mid_deadline)

            # exact‑max Adam (refinement)
            a_opt = _adam_optimize_submax(a_mid, target, this_deadline)

            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best = a_opt.copy()

        if local_best is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Improvement after n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2c)  Deterministic hill‑climb (exact max)
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex_fast(a_cur, target)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + hc_budget)
            a_hc, score_hc = _hill_climb_deterministic(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2d)  Tiny random tweaks (simulated‑annealing style)
        # --------------------------------------------------------------
        while time.time() + 0.5 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # use a relatively small delta to allow fine‑grained exploration
            delta = cand[i] * rng.uniform(0.005, 0.02)
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            score = _max_convolution_fft(cand)
            if score < best_score - 1e-12:
                best_seq = cand
                best_score = score
                print(f"[{time.time():.1f}] Random‑tweak improved n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Final polishing on whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 0.5
        best_seq = _adam_optimize_submax(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.2)
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4)  Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # Very unlikely fallback – uniform distribution with n=2000
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5)  Official evaluation report
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```
