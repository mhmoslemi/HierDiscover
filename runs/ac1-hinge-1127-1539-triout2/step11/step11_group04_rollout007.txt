def _simulated_annealing(seq, time_limit):
    n = len(seq)
    target_sum = seq.sum()
    best_seq = seq.copy()
    best_score = _score(seq)
    cur_seq = seq.copy()
    cur_score = best_score
    rng = np.random.default_rng()
    start = time.time()
    end = start + time_limit
    it = 0
    T0 = best_score * 0.1
    while time.time() < end:
        # decide move type
        if rng.random() < 0.7:
            # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # propose swap
            new_seq = cur_seq.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:
            # transfer mass
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            donor = i
            receiver = j
            # amount = random fraction of donor up to 50%
            max_delta = cur_seq[donor] * 0.5
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            new_seq = cur_seq.copy()
            new_seq[donor] -= delta
            new_seq[receiver] += delta

        # evaluate
        new_score = _score(new_seq)
        # temperature decays
        elapsed = time.time() - start
        # linear decay
        T = T0 * (1 - elapsed / time_limit)
        # acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (max(T, 1e-12))):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
        it += 1
    return best_seq, best_score

"""
Hybrid Simulated‑Annealing + Simple Gradient Polishing.

The algorithm works in two scales.

* Small scale (n≈4k) – a fast simulated‑annealing on a
  permutation of a fixed magnitude vector (power‑law / exponential /
  Dirichlet).  Moves are swaps of two positions or a small mass transfer
  between them, accepted with a Metropolis rule.  The best small‑scale
  sequence is kept.

* Upscaling – the small solution is linearly interpolated to a larger
  length (≈12k–16k).  After a brief re‑ordering (alternating / even‑spread)
  a second simulated‑annealing run is performed, now mostly swapping
  entries (the total sum stays constant).  Finally a short projected
  gradient descent (soft‑max approximation) polishes the result.

All steps respect non‑negativity and the ℓ¹‑simplex constraint.
The routine stops automatically before the allotted time budget expires
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple reorderings (helpful after up‑scaling)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Simulated Annealing on a sequence (swap & mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(seq: np.ndarray,
                         time_limit: float,
                         move_fraction: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that respects the ℓ¹‑simplex.
    ``move_fraction`` determines the probability of a mass‑transfer
    move (vs. a pure swap).  The algorithm returns the best sequence
    seen during the allotted time.
    """
    n = seq.shape[0]
    target_sum = seq.sum()
    cur_seq = seq.copy()
    cur_score = _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    end = start + time_limit

    # initial temperature – a fraction of the current score
    T0 = max(cur_score * 0.3, 1e-6)

    it = 0
    while time.time() < end:
        # decide move type
        if rng.random() < move_fraction:
            # ------- mass transfer -------
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            donor, receiver = i, j
            max_delta = cur_seq[donor] * 0.5
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            new_seq = cur_seq.copy()
            new_seq[donor] -= delta
            new_seq[receiver] += delta
        else:
            # ------- swap -------
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_seq = cur_seq.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]

        # projection is not needed – sum stays constant and we never get negatives
        new_score = _score(new_seq)

        # linear temperature decay
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / time_limit)
        if T < 1e-12:
            T = 1e-12

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{time.time():.2f}] SA iter={it:6d} score={best_score:.6f}")

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Soft‑max (approximate) gradient descent (few iterations, polishing)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """Returns gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2·(a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max approximation (tiny polishing step)."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val
