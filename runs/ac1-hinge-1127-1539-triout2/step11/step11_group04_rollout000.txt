"""
Hybrid Adam‑softmax optimizer with reordering & greedy mass‑transfer.

The algorithm works in three stages:

1. **Initialisation** – build a diverse pool of candidates (power‑law,
   exponential, random simplex, previous best) of length `n` and pick the
   one with the smallest convolution peak.

2. **Adam soft‑max gradient descent** – minimise a smooth approximation of
   the max‑convolution (`softmax_β`).  The gradient is obtained via FFT.
   Adam adapts the per‑coordinate step size; the temperature β is doubled
   regularly to sharpen the approximation.  After each update the vector is
   projected onto the ℓ¹‑simplex {x≥0, Σx = √(2 n)}.

3. **Discrete polishing** – a short reordering sweep (alternating,
   even‑spread and a few random shuffles) is performed, followed by a greedy
   mass‑transfer hill‑climb that moves a small amount of mass from the
   “donor’’ (largest contribution to the current peak) to a “receiver’’
   (smallest contribution).  Only improving moves are kept.

The whole routine respects the 1000 s budget; the final sequence is trimmed
to remove trailing near‑zero entries before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear self‑convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    For Σa = √(2n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector `v` onto
    {x | x_i ≥ 0, Σ_i x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), renormalised to Σa = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime to the length."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _random_shuffle_order(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a random permutation of the sequence."""
    perm = rng.permutation(seq.shape[0])
    return seq[perm]


# ----------------------------------------------------------------------
# Soft‑max gradient (exact convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (gradient, full_convolution_array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a_rev * p)  (convolution of reversed a with p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Adam optimiser on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_softmax_opt(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      max_beta: float = 5e5,
                      beta_schedule: int = 400,
                      lr_base: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max‑convolution.
    Returns the best sequence found and its (exact) score.
    """
    rng = np.random.default_rng()
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_base * target_sum / (np.sqrt(v_hat) + eps)
        step = lr * m_hat

            a += jitter
            a = _project_to_simplex(a, target_sum)

        # increase β periodically (soft‑max sharpening)
        if t % beta_schedule == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val

def _reorder_polish(a: np.ndarray,
                    deadline: float,
                    target_sum: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Try a few deterministic and random reorderings,
    keep the best one (according to exact max‑convolution).
    """
    candidates = [a,
                  _alternating_order(a),
                  _even_spread_order(a)]

    # a few random shuffles
    for _ in range(5):
        if time.time() >= deadline:
            break
        candidates.append(_random_shuffle_order(a, rng))

    best_seq = a
    best_val = _score(a)
    for cand in candidates:
        val = _score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
            print(f"[{time.time():.2f}] Reorder improvement: score={best_val:.6f}")

    return best_seq, best_val


def _mass_transfer_hill_climb(a0: np.ndarray,
                              deadline: float,
                              target_sum: float,
                              start_temp: float = 0.3,
                              end_temp: float = 1e-4,
                              frac_move: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer moves (donor → receiver) guided by the current
    peak of the convolution.  Accept only improving moves; a tiny
    simulated‑annealing temperature is kept to allow occasional escapes.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_seq = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    total_time = deadline - start_time

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions to the current peak
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        # pick a receiver among the smallest contributors
        low_cnt = max(1, int(len(idx) * 0.1))
        receiver = rng.choice(idx[np.argsort(prod)[:low_cnt]])

        if donor == receiver:
            continue

        # move a small fraction of mass from donor to receiver
        delta = frac_move * min(a[donor], a[receiver])
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _score(a)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Mass‑transfer new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    Returns a Python list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem dimension based on remaining budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 14000
    else:
        n = 10000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Build a diverse initial pool
    # --------------------------------------------------------------
    pool = []

    # power‑law family (exponents in [0.2, 1.2])
    for p in np.linspace(0.2, 1.2, 7):
        pool.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.015, 6):
        pool.append(_exponential_sequence(n, float(lam), target_sum))

    # random simplex points
    for _ in range(8):
        pool.append(_random_simplex(n, target_sum, rng))

    # include previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        pool.append(_resize_sequence(prev, n))

    # evaluate pool and keep the best seed
    best_seed = None
    best_score = np.inf
    for seq in pool:
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seed = seq.copy()
    print(f"[{time.time():.2f}] Initial best score from pool = {best_score:.6f}")

    # --------------------------------------------------------------
    # Adam based soft‑max optimisation (major part of the budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # reserve ~15% of remaining time for discrete polishing
    opt_deadline = time.time() + remaining * 0.85
    opt_seq, opt_score = _adam_softmax_opt(best_seed, opt_deadline, target_sum,
                                           max_beta=5e5,
                                           beta_schedule=400,
                                           lr_base=0.25)

    # --------------------------------------------------------------
    # Discrete polishing (reordering + mass‑transfer)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        # reordering sweep
        reord_seq, reord_score = _reorder_polish(opt_seq, deadline, target_sum, rng)
        if reord_score < opt_score:
            opt_seq, opt_score = reord_seq, reord_score

    if time.time() < deadline - 1.0:
        # final greedy mass‑transfer with a very short schedule
        mt_seq, mt_score = _mass_transfer_hill_climb(opt_seq, deadline, target_sum,
                                                    start_temp=0.2, end_temp=1e-5,
                                                    frac_move=0.015)
        if mt_score < opt_score:
            opt_seq, opt_score = mt_seq, mt_score

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(opt_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Log final evaluation (purely for information)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for potential later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
