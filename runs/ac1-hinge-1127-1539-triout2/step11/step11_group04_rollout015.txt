    deadline = start + budget_s - 0.5
    # set n_expl = 6000 maybe.

    target_expl = math.sqrt(2.0 * n_expl)
    rng = np.random.default_rng()

    # Build init_pool
    init_pool = []
    # symmetric power law candidates
    for p in np.linspace(0.2, 2.0, 9):

"""
Hybrid search for a low‑max‑self‑convolution step function.

The algorithm proceeds in several phases:

1.  **Diverse seeding** – creates a pool of candidates:
    * symmetric power‑law and exponential families,
    * simple two‑level (high‑edges / low‑center) shapes,
    * random points on the ℓ¹‑simplex.
2.  **Exploration** – short annealed soft‑max gradient runs on a
    modest length (≈6k).  The best seed is kept.
3.  **Optional mass‑transfer polish** on the small scale.
4.  **Upscaling** – linear interpolation to a larger length
    (≈12k‑20k) and a few simple reorderings (alternating, even‑spread)
    to lower the convolution peak.
5.  **Large‑scale optimisation** – a long soft‑max GD pass,
    followed by a projected sub‑gradient descent on the exact max‑convolution,
    and a final deterministic mass‑transfer (Metropolis) pass.
6.  The trailing negligible entries are trimmed and the sequence is
    exposed as ``height_sequence_1`` for reuse.

All updates respect the non‑negativity and ℓ¹‑simplex constraints,
so the evaluation function
``2·n·max(a*a)/(sum(a)²)`` is exactly ``max(a*a)`` when
``sum(a)=√(2n)``.
The routine is timed and automatically adapts the work‑load to the
available budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x≥0, Σx = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _symmetric_power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (dist to nearest edge)^{-p} , symmetric."""
    d = np.minimum(np.arange(n) + 1, n - np.arange(n))
    a = d.astype(np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _symmetric_exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-lam * d.astype(np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _two_level_sequence(n: int, m: int, r: float, target_sum: float) -> np.ndarray:
    """
    Ends (size m on each side) get value r·c_low,
    interior gets value c_low.  r≥1, 0≤2m≤n.
    """
    if 2 * m > n:
        m = n // 2
    low_cnt = n - 2 * m
    # c_low = target / ((2*m)*r + low_cnt)
    c_low = target_sum / ((2 * m) * r + low_cnt)
    c_high = r * c_low
    a = np.full(n, c_low, dtype=np.float64)
    if m > 0:
        a[:m] = c_high
        a[-m:] = c_high
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Gradient / sub‑gradient optimisation helpers
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      max_beta: float = 1e5,
                                      step_factor: float = 0.7,
                                      gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (a.shape[0] - 1))
            hi = min(k, a.shape[0] - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer with exponentially decaying temperature."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step‑function with a low max‑self‑conv."""
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 1. Choose size for the fast exploratory stage
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 6000
    elif budget_s >= 400:
        n_expl = 4000
    else:
        n_expl = 2000
    target_expl = math.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 2. Build a diverse pool of seeds
    # --------------------------------------------------------------
    init_pool = []

    # symmetric power‑law family
    for p in np.linspace(0.2, 2.0, 9):
        init_pool.append(_symmetric_power_law_sequence(n_expl, float(p), target_expl))

    # symmetric exponential family
    for lam in np.linspace(0.001, 0.02, 6):
        init_pool.append(_symmetric_exponential_sequence(n_expl, float(lam), target_expl))

    # two‑level (high‑edges / low‑center) shapes
    for frac in [0.02, 0.05, 0.10, 0.15, 0.20]:
        m = max(1, int(frac * n_expl))
        for r in [2.0, 3.0, 5.0, 10.0]:
            init_pool.append(_two_level_sequence(n_expl, m, r, target_expl))

    # random points on the simplex
    for _ in range(8):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies to increase variability
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # --------------------------------------------------------------
    # 3. Exploration – many short soft‑max GD runs
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    exploration_deadline = start_time + 0.20 * budget_s
    while time.time() < exploration_deadline:
        # choose a seed (biased towards current best)
        if best_seq is not None and rng.random() < 0.4:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.003, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 12.0, exploration_deadline)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seed, run_dead, target_expl,
            max_beta=5e5, step_factor=0.7, gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Small‑scale deterministic polishing (optional)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        mt_dead = time.time() + min(0.08 * (deadline - time.time()), 30.0)
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_expl,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Pre‑upscale MT best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Upscale to a large problem size
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.7, target_expl)

    remaining = deadline - time.time()
    if remaining > 300:
        n_final = 20000
    elif remaining > 150:
        n_final = 15000
    elif remaining > 80:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few reorderings, pick the smallest max‑conv
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)

    best_seq = seq_big
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Upscaled to {n_final}, start score {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Large‑scale optimisation
    # --------------------------------------------------------------
    # 6a) Soft‑max gradient descent (longer)
    if time.time() < deadline - 30.0:
        soft_dead = time.time() + min(120.0, deadline - time.time() - 30.0)
        a_soft, val_soft = _annealed_gradient_descent_v2_ext(
            best_seq, soft_dead, target_final,
            max_beta=1e6, step_factor=0.6, gamma=0.98,
        )
        if val_soft < best_score:
            best_score = val_soft
            best_seq = a_soft.copy()
            print(f"[{time.time():.2f}] Soft‑max GD refined score {best_score:.6f}")

    # 6b) Exact sub‑gradient descent (major chunk of time)
    if time.time() < deadline - 20.0:
        sub_dead = deadline - 20.0
        a_sub, val_sub = _subgradient_descent(
            best_seq, sub_dead, target_final,
            step_factor=0.7, gamma=0.995,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved score {best_score:.6f}")

    # 6c) Final deterministic mass‑transfer pass
    if time.time() < deadline - 5.0:
        mt_dead = deadline - 5.0
        a_final, val_final = _deterministic_mass_transfer(
            best_seq, mt_dead, target_final,
            start_temp=0.2, end_temp=1e-5, low_frac=0.08,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final MT best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Trim negligible tail and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # expose for possible reuse in later runs
    globals()["height_sequence_1"] = final_seq

    # optional final evaluation for logging
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
