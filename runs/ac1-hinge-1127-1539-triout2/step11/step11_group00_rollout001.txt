def _adam_softmax(a0, target_sum, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr=0.05 # etc
    beta=1.0
    max_beta=65536.0
    it=0
    best_a=a.copy()
    best_val=_score(a)
    while time.time()<deadline:
        g,_ = _softmax_gradient(a,beta)
        it+=1
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat)+eps)
        a = a - step
        # projection
        a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it} β={beta:.1f} score={best_val:.6f}")
        if it % 200 == 0:
            # increase beta
            beta = min(beta*2.0, max_beta)
        # optional jitter occasional
        if it % 500 == 0:
            rng=np.random.default_rng()
            jitter = rng.normal(scale=target_sum*0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
    return best_a, best_val

   optimiser eventually focuses on the true maximum.

3. **Mass‑transfer simulated annealing** – a long‑running anneal that
   repeatedly moves a small amount of mass from a “donor’’ index (large
   contribution to the current peak) to a “receiver’’ index (small height),
   accepting moves that improve the objective or that are accepted
   probabilistically according to a linear cooling schedule.

All steps respect the time budget (default 1000 s) and finally return the
best sequence found as a plain list of floats.  The sequence is trimmed
to the last non‑zero entry before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (log‑sum‑exp) gradient
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]  # full conv

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()                                   # soft‑max weights

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1:n - 1 + n]                  # length‑n gradient
    return grad, b


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ----------------------------------------------------------------------


def _adam_softmax(a0: np.ndarray, target_sum: float, deadline: float):
    """Adam optimiser on the log‑sum‑exp surrogate, with simplex projection."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.03                     # base learning rate
    beta = 1.0                    # start smoothing parameter
    max_beta = 65536.0
    it = 0
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        g, _ = _softmax_gradient(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 600 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothing parameter periodically
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent on the true max‑convolution entry."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear schedule for step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """Long‑run annealing that moves tiny amounts of mass from “bad’’ to “good’’."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.35                 # start temperature (~35 % of score)

    last_full = start
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour low heights (inverse‑proportional)
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.07, target_sum * 0.0025)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear contribution: 2·a·Δ
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_full > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # tiny random jitter every few thousand steps
        if it % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 24000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = float(np.sqrt(2.0 * n))

    # --------------------------------------------------------------
    # Warm‑start: previous construction or cheap power‑law scan
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # a quick power‑law scan
    for p in np.linspace(0.30, 1.00, 15):
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a = a * (target_sum / a.sum())
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    # previous best, if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        prev_resized = _resize_sequence(prev, n)
        sc_prev = _score(prev_resized)
        if sc_prev < best_score:
            best_score = sc_prev
            best_seq = prev_resized.copy()

    # --------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (≈15 % of remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.15 * (deadline - time.time())
        a_adam, sc_adam = _adam_softmax(best_seq, target_sum, phase_deadline)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam
        print(f"[{time.time():.2f}] After Adam best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on the true max (≈15 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + 0.15 * (deadline - time.time())
        a_sub, sc_sub = _subgradient_descent(best_seq, target_sum, phase_deadline)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub
        print(f"[{time.time():.2f}] After sub‑gradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – long simulated‑annealing mass transfer (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ann, sc_ann = _mass_transfer_anneal(best_seq, target_sum, deadline)
        if sc_ann < best_score:
            best_score = sc_ann
            best_seq = a_ann
        print(f"[{time.time():.2f}] After annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up: trim trailing zeros, re‑normalise, and evaluate
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
