"""
Improved optimizer for the step‑function benchmark.

Main ideas
----------
* A dense parametric pre‑search (power‑law, exponential‑power‑law,
  broken power‑law) to obtain a very good starting shape fast.
* Repeated cycles of projected Adam optimisation with an accelerated
  soft‑max temperature schedule.  When the temperature becomes large
  the exact sub‑gradient of the max‑convolution is used, which gives a
  precise descent direction.
* Aggressive deterministic mass‑transfer (the “peak‑flattening”
  move from the original solution) with a random tie‑breaker to avoid
  getting stuck.
* A final polishing Adam run with an extremely high temperature
  and a small step‑size.

All steps respect the simplex constraint Σa = √(2 n) and a_i ≥ 0.
The routine adapts its time allocation to the given budget (≤1000 s)
and returns the best sequence found.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Fast convolution utilities (real‑FFT)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score ( = max(a*a) when Σa = √(2 n) )."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the ultra‑dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    conv_full = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()  # soft‑max weights

    # convolution of a_rev with p
    a_rev = a[::-1]
    ar_fft = rfft(a_rev, n=size)
    p_fft = rfft(p, n=size)
    conv_rev = irfft(ar_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (used once β is large)
# ----------------------------------------------------------------------


def _subgradient_maxconv(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) and the full convolution."""
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    # gradient = 2 * a_{k - i} for i in [lo,hi]
    grad[idx] = 2.0 * a[k - idx]
    return grad, conv, max_val, k

# Adam projected gradient descent with temperature schedule
# ----------------------------------------------------------------------

    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.9,
    gamma: float = 0.95,
    beta_inc_iter: int = 100,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max (or exact) gradient of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    # When β becomes large we switch to the exact sub‑gradient.
    use_exact = False

    while time.time() < deadline:
        if not use_exact:
            # soft‑max gradient
            g, conv_full = _gradient_softmax(a, beta)
        else:
            g, conv_full, max_val, k_max = _subgradient_maxconv(a)

        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        # track best
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        # temperature annealing
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
            if beta >= 1e3:
                use_exact = True

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (peak flattening)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
    rng=np.random.default_rng(),
) -> tuple[np.ndarray, float]:
    """Repeatedly move mass from the biggest contributor to the smallest
    contributor for the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        # pick a random maximal index to avoid deterministic loops
        k = int(rng.choice(idx_max))

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of the current peak a_{k-i} * a_i
        contrib = a[k - idx]

        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        # second‑largest value (ignore all maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        denom = 2.0 * (a[k - donor] - a[k - receiver])
        if denom <= 0.0:
            # No improvement possible for this k – try another maximal index
            if len(idx_max) > 1:
                continue
            else:
                break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] DMT it={int(time.time())} score={best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Lightweight global random mass‑transfer (exploratory)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """Very light simulated‑annealing style mass shuffling."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        # pick a fairly large and a fairly small entry
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Random move new best={best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the supplied time budget and returns the best
    sequence found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ---------------------------------------------------------------
    # Choose problem size (larger n → finer resolution, same objective)
    # ---------------------------------------------------------------
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    elif budget_s >= 300:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)  # Σa = √(2 n)

    # ---------------------------------------------------------------
    # Phase 0 – ultra‑dense parametric pre‑search (few seconds)
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    p_grid = np.arange(0.20, 1.21, 0.01)
    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) exponential‑power‑law
    p_grid2 = np.arange(0.20, 1.21, 0.02)
    q_grid = np.arange(0.0, 10.001, 0.2)
    for p in p_grid2:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (2‑segment)
    p1_grid = np.arange(0.20, 0.81, 0.1)
    p2_grid = np.arange(0.80, 2.01, 0.2)
    alpha_grid = np.arange(0.1, 0.91, 0.1)
    for p1 in p1_grid:
        for p2 in p2_grid:
            for alpha in alpha_grid:
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) start from a previously stored solution if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(
        f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}"
    )

    # ---------------------------------------------------------------
    # Phase A – long Adam optimisation (majority of remaining time)
    # ---------------------------------------------------------------
    # allocate ~60 % of the remaining budget to Adam
    remaining = deadline - time.time()
    gd_deadline = time.time() + 0.60 * remaining
    while time.time() < gd_deadline - 5.0:
        # jittered restart around the current best
        rng = np.random.default_rng()
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            gd_deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.9,
            gamma=0.95,
            beta_inc_iter=100,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(
                f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Phase B – deterministic mass‑transfer (core refinement)
    # ---------------------------------------------------------------
    # use the rest of the budget (minus a short final polishing window)
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum, rng=np.random.default_rng()
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(
                f"[{time.time():.2f}] DMT improved best = {best_score:.6f}"
            )

        # occasional global random shuffles to escape tiny plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(
                f"[{time.time():.2f}] Random move improved best = {best_score:.6f}"
            )

        # stop early if near deadline
        if time.time() > deadline - 2.0:
            break

    # ---------------------------------------------------------------
    # Phase C – final polishing (high temperature, tiny steps)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=5e6,
            step_factor=0.5,
            gamma=0.97,
            beta_inc_iter=50,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(
                f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final call to the provided evaluator (harmless if missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
