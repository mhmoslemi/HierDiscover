def _local_hill_climb(a, target_sum, deadline):
    # a: np.ndarray; assume sum = target_sum.
    n = a.shape[0]
    b = _conv_fft(a)
    cur_max = b.max()
    best_a = a.copy()
    best_val = cur_max
    rng = np.random.default_rng()
    it = 0
    # temperature schedule:
    T0 = best_val * 0.05  # initial temp scaling
    while time.time() < deadline:
        elapsed = time.time() - start; But we need start variable. We'll set start = time.time() at start.
        # compute temperature linearly decaying
        temperature = max(0.0, T0 * (1.0 - (time.time() - start) / (deadline - start)))
        # identify k_max
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # calculate contributions
        prod = a[idx] * a[k_max - idx]
        # choose donor: weighted by prod maybe.
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.choice(idx)
        # choose receiver: index with small a (lowest values)
        # some randomness also: pick among low percentile
        low_threshold = np.percentile(a, 20)
        low_idx = np.where(a <= low_threshold)[0]
        if low_idx.size > 0:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)
        if donor == receiver:
            continue
        # delta
        max_delta = min(a[donor] * 0.2, target_sum * 0.002)  # relatively larger.
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta
        # update b_new
        # vector a used for update: need original a before updating
        b_new = b.copy()
        # Apply updates:
        # note that a is unchanged in applying updates; used as original values
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta
        new_max = b_new.max()
        # acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_max - cur_max) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            # apply move
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Hillclimb improved to {best_val:.6f}")
        # occasional jitter to avoid stagnation
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_max = b.max()
        it += 1
    return best_a, best_val

"""
Hybrid optimizer for the non‑negative step‑function benchmark.

The routine works in three stages:
  1.   Warm‑start – reuse a previously found sequence (if any) and
       a very cheap parametric sweep over power‑law and exponential
       shapes.  The best of these candidates is taken as a start point.
  2.   Deterministic “mass‑transfer” hill‑climbing – repeatedly moves a
       small amount of mass from a donor index that contributes most
       to the current maximal convolution entry to a receiver index
       that contributes little.  Acceptance follows a simple linear
       annealing schedule, guaranteeing monotone improvement after the
       temperature has vanished.
  3.   Final clean‑up – re‑normalises the sequence, truncates trailing
       zeros and returns the best sequence found before the time budget
       expires.

All heavy work is performed with FFT‑based convolutions (O(n log n))
and O(n) updates for each hill‑climb move.  The code respects the
supplied budget (default 1000 s) and stores the best sequence in the
global variable ``height_sequence_1`` for warm‑starts in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  Identical to the benchmark score."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to ``new_len`` entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Candidate generation (power‑law / exponential)
# ----------------------------------------------------------------------


def _gen_power_law(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """a_i = (i+1+offset)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64) + offset
    return i ** (-p)


def _gen_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (mass‑transfer) with annealing
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                target_sum: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a tiny amount of mass from a donor (large product
    contribution to the current peak) to a low‑mass receiver.
    Acceptance follows a linear temperature schedule; after the
    temperature reaches zero only improving moves survive.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                 # full convolution
    cur_max = b.max()
    best_a = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # initial temperature – a small fraction of the current excess over 1.0
    T0 = max(0.0, (best_val - 1.0) * 0.06)

    it = 0
    while time.time() < deadline:
        # --- locate the current peak -------------------------------------------------
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contributions a_i * a_{k_max-i}
        prod = a[idx] * a[k_max - idx]

        # donor – weighted by contribution (more likely to be a large contributor)
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:                           # fallback – random index in the support
            donor = rng.choice(idx)

        # receiver – pick among the low‑mass indices (30‑th percentile)
        low_thresh = np.percentile(a, 30)
        low_idx = np.where(a <= low_thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # --- choose a step size ------------------------------------------------------
        # allow a relatively generous move early on; later the temperature will
        # force us to be more cautious.
        max_delta = min(a[donor] * 0.15, target_sum * 0.0015)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # --- update convolution (exact O(n) update) ----------------------------------
        # the formula corresponds to updating a -> a + delta*e_receiver - delta*e_donor
        b_new = b.copy()
        # contribution of the donor (both (donor, *) and (*, donor) terms)
        b_new[donor:donor + n] -= 2.0 * delta * a
        # contribution of the receiver
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # diagonal terms (i=j) need the extra delta^2 corrections
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()

        # --- annealed acceptance ----------------------------------------------------
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))

        accept = False
        if new_max < cur_max:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_max - cur_max) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_max = new_max
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Hill‑climb improved to {best_val:.6f}")

        # occasional global jitter to shake out of plateaus
        if it % 2500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0003, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_max = b.max()
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Jitter improved to {best_val:.6f}")

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
