def _lp_step(f, M, target_sum):
    """
    Solve LP maximize sum(g) subject to conv(f,g) <= M (elementwise), g >= 0.
    Returns (g, sum_g).
    """
    import numpy as np
    from scipy.optimize import linprog
    n = f.shape[0]
    rows = 2*n - 1
    # Build A matrix
    A = np.zeros((rows, n))
    # fill as described:
    # for each column j, rows j:j+n = f
    for j in range(n):
        A[j:j+n, j] = f
    b = np.full(rows, M)
    c = -np.ones(n)  # minimize -sum(g) -> maximize sum(g)
    # bounds are (0, None) for each variable
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    g = res.x
    sum_g = g.sum()
    return g, sum_g

"""

    2 * n * max(a*a) / (sum(a)^2)

Algorithm
---------

* Use a modest problem size (n_small≈2000) for intensive
  exploration where all operations are cheap.
* Dense parametric pre‑search (power‑law families) to obtain a good
  initial shape.
* Repeated cycles on the small problem:
    - Adam‑style projected gradient descent on a soft‑max
      approximation of the max‑convolution.
    - Aggressive deterministic mass‑transfer (exact reduction of the
      current peak).
    - One linear‑programming step: maximise Σg under the
      convolution bound ‖f*g‖_∞ ≤ M, then blend the solution into f.
* Upsample the best small‑size sequence to the target size
  (n_target = 5k‑12k depending on the time budget) by linear
  interpolation and project onto the simplex.
* A short final polishing on the full size (high β Adam, DMT,
  random moves).
* All sequences are kept non‑negative and their sum is forced to
  √(2 n) (the scaling prescribed by the benchmark).

The routine respects the wall‑clock budget (default 1000 s) and

import time

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (full linear convolution)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (sum is assumed to be √(2 n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the dense pre‑search)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(n: int, p1: float, p2: float,
                               alpha: float, target_sum: float) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(a*a)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------
def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 5e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        k = idx_max[0]  # pick the first maximal index

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[k - idx]
        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        denom = 2.0 * (a[k - donor] - a[k - receiver])
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={int(time.time())} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple global random mass‑transfer (exploration)
# ----------------------------------------------------------------------
def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 1.0:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming direction step (optional, uses SciPy)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, target_sum: float) -> tuple[np.ndarray, float]:
    """Solve  max Σg  s.t.  f*g ≤ M (elementwise) , g≥0 .
    Returns the raw solution g and its sum (before scaling)."""
    if not _HAS_SCIPY:
        raise RuntimeError("SciPy not available for LP step")
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A where A[k, j] = f[k-j] (if 0≤k-j<n else 0)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f  # vectorised fill of the j‑th column

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -Σg  → maximise Σg
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed: " + str(res.message))
    g = res.x
    return g, g.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with the minimal benchmark score.
    The routine respects the given time budget and returns the best sequence.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------- problem size selection -------------------
    if budget_s >= 800:
        n_target = 12000
    elif budget_s >= 400:
        n_target = 8000
    else:
        n_target = 5000
    target_sum_target = math.sqrt(2.0 * n_target)

    # a modest size for the heavy optimisation phases
    n_small = min(2000, n_target)
    target_sum_small = math.sqrt(2.0 * n_small)

    # ------------------- initialise best on the small scale -------------------
    best_small = None
    best_score_small = float("inf")
    rng = np.random.default_rng()

    # use any previously discovered sequence if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n_small:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n_small)
            prev_small = np.interp(new_x, old_x, prev)
        else:
            prev_small = prev.copy()
        prev_small = _project_to_simplex(prev_small, target_sum_small)
        val = _score(prev_small)
        best_small, best_score_small = prev_small, val

    # ------------------- dense parametric pre‑search (small n) -------------------
    # (a) pure power‑law
    for p in np.arange(0.20, 1.21, 0.02):
        seq = _power_law_sequence(n_small, float(p), target_sum_small)
        val = _score(seq)
        if val < best_score_small:
            best_score_small, best_small = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in np.arange(0.20, 1.21, 0.05):
        for q in np.arange(0.0, 8.001, 0.5):
            seq = _exp_power_law_sequence(n_small, float(p), float(q),
                                          target_sum_small)
            val = _score(seq)
            if val < best_score_small:
                best_score_small, best_small = val, seq.copy()

    # (c) broken power‑law (two segments)
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(n_small, float(p1), float(p2),
                                                 float(alpha), target_sum_small)
                val = _score(seq)
                if val < best_score_small:
                    best_score_small, best_small = val, seq.copy()

    print(f"[{time.time():.2f}] Small pre‑search best = {best_score_small:.6f}")

    # ------------------- Phase A on the small problem -------------------
    # Allocate roughly 70 % of the overall budget to the intensive small‑scale phase
    small_deadline = start_time + 0.70 * (deadline - start_time)

    while time.time() < small_deadline:
        remaining = small_deadline - time.time()
        if remaining < 5.0:
            break

        # ---- Adam gradient descent restart ----
        gd_time = min(20.0, remaining * 0.5)
        jitter = rng.normal(scale=target_sum_small * 0.015, size=n_small)
        start_vec = best_small + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum_small)

        a_opt, val_opt = _adam_gradient_descent(start_vec,
                                                time.time() + gd_time,
                                                target_sum_small,
                                                max_beta=5e5,
                                                step_factor=0.8,
                                                gamma=0.995)
        if val_opt < best_score_small:
            best_score_small, best_small = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Small GD improved best = {best_score_small:.6f}")

        # ---- Deterministic mass‑transfer ----
        if time.time() >= small_deadline:
            break
        dmt_deadline = time.time() + min(10.0,
                                         (small_deadline - time.time()) * 0.5)
        a_ref, val_ref = _deterministic_mass_transfer(best_small,
                                                      dmt_deadline,
                                                      target_sum_small)
        if val_ref < best_score_small:
            best_score_small, best_small = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Small DMT improved best = {best_score_small:.6f}")

        # ---- Linear‑programming direction step (optional) ----
        if time.time() >= small_deadline:
            break
        try:
            M = _max_conv(best_small)
            g_raw, sum_g_raw = _lp_step(best_small, M, target_sum_small)
            if sum_g_raw > 1e-12:
                blending = 0.20
                g_scaled = g_raw * (target_sum_small / sum_g_raw)
                a_blend = (1.0 - blending) * best_small + blending * g_scaled
                a_blend = np.maximum(a_blend, 0.0)
                a_blend = _project_to_simplex(a_blend, target_sum_small)
                val_blend = _score(a_blend)
                if val_blend < best_score_small:
                    best_score_small, best_small = val_blend, a_blend.copy()
                    print(f"[{time.time():.2f}] Small LP blend improved best = {best_score_small:.6f}")
        except Exception:
            # If SciPy is missing or LP fails, we simply skip this step.
            pass

    # ------------------- Upsample to the target size -------------------
    small_x = np.linspace(0.0, 1.0, n_small)
    target_x = np.linspace(0.0, 1.0, n_target)
    upsampled = np.interp(target_x, small_x, best_small)
    upsampled = _project_to_simplex(upsampled, target_sum_target)

    best_seq = upsampled.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After up‑sampling: score = {best_score:.6f}")

    # ------------------- Phase C: final refinement on full size -------------------
    # Use the remaining time (≈30 % of the budget)
    while time.time() < deadline - 4.0:
        time_left = deadline - time.time()
        if time_left < 5.0:
            break

        # short Adam phase (high β schedule)
        gd_time = min(5.0, time_left * 0.5)
        a_opt, val_opt = _adam_gradient_descent(best_seq,
                                                time.time() + gd_time,
                                                target_sum_target,
                                                max_beta=5e5,
                                                step_factor=0.8,
                                                gamma=0.995)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Final GD improved best = {best_score:.6f}")

        # deterministic mass‑transfer (quick)
        if time.time() >= deadline - 3.0:
            break
        dmt_deadline = time.time() + min(3.0,
                                         deadline - time.time() - 1.0)
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      dmt_deadline,
                                                      target_sum_target)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Final DMT improved best = {best_score:.6f}")

        # occasional global random move
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum_target)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Final random move improved best = {best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # ------------------- Final high‑β polish -------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum_target,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}")

    # ------------------- post‑processing -------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # final evaluation for reporting (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
