def adam_optimize(a0, target_sum, deadline, beta_start=0.5, beta_factor=1.2, beta_max=5e5, base_lr=0.1):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = beta_start
    # schedule: each iteration update, or increase at intervals
    while time.time() < deadline:
        # compute grad
        grad, conv = _gradient_softmax(a, beta)
        t += 1
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size: base_lr * target_sum / (np.sqrt(v_hat) + eps)
        step = base_lr * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        # projection
        a = _project_to_simplex(a, target_sum)

        # clamp? but projection ensures sum and positivity.
        # optionally jitter every certain iters
        if t % 500 == 0:
            # maybe increase beta
            beta = min(beta * beta_factor, beta_max)

        # evaluate
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iter={t} β={beta:.1f} best={best_val:.6f}")

    return best_a, best_val

"""
Propose a step‑function sequence with a low maximum self‑convolution.

The search works in four stages:
1.  Build a large pool of parametric shapes (power‑law, exponential,
    sin‑power, triangular, quadratic, semi‑circle) and jittered copies.
    Each candidate is projected onto the ℓ¹‑simplex {x≥0, Σx = √(2 n)}.
2.  Take the best few candidates and run an Adam‑style optimisation on a
    soft‑max approximation of the max‑convolution (β is annealed up to a
    large value).  After every update the vector is projected back onto the
    simplex.
3.  Refine the current best with a sub‑gradient descent on the true
    max‑convolution.
4.  Perform a greedy deterministic mass‑transfer phase (simulated‑annealing
    style) to move mass from “heavy” contributors of the current peak to
    “light’’ positions.
Finally the tail of near‑zero entries is trimmed and the sequence is stored
in the global variable ``height_sequence_1`` for warm‑starts.

All heavy work uses FFT (O(n log n)) and the routine respects the supplied
time budget (default 1000 s).  The function returns a plain Python list
compatible with ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (unnormalised)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """(i+1)^{-p} for i=0,…,n‑1."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """exp(-λ·i) for i=0,…,n‑1."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    """sin(π·(i+½)/n)^{p}."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth proxy for max)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full self‑convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    # soft‑max weights
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * rev(a) ⋆ p  (restricted to central n entries)
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max proxy
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  beta_start: float = 0.5,
                  beta_factor: float = 1.2,
                  beta_max: float = 5e5,
                  base_lr: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the soft‑max approximation of max‑convolution.
    Returns the best sequence and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t = 0
    beta = beta_start
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        t += 1
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Per‑coordinate step
        step = base_lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # Projection back onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        # anneal beta
        if t % 400 == 0:
            beta = min(beta * beta_factor, beta_max)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        target_sum: float,
                        deadline: float,
                        step_factor: float = 0.5,
                        gamma: float = 0.9,
                        jitter_interval: int = 500,
                        jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent directly on max(conv).  Momentum is used.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient g[i] = 2 * a[kmax‑i] when the index is valid
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (local annealing)
# ----------------------------------------------------------------------


def deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer that reduces the current peak of the self‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index of the current maximal convolution entry
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution w_i = a_i + a_{kmax-i}
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        # Metropolis accept/reject
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to keep exploration alive
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = _project_to_simplex(np.maximum(a + jitter, 0.0), target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximum entry
    of its self‑convolution.  The routine respects the given time budget and
    returns a list ready for `evaluate_sequence`.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------- problem size ---------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # --------------------- build candidate pool ---------------------
    rng = np.random.default_rng()
    candidates = []

    # warm‑start from previous run
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        prev = _project_to_simplex(prev, target_sum)
        candidates.append(prev)

    # sin‑power family
    for p in np.linspace(0.2, 2.0, 61):
        seq = _sin_power_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.001, size=n)
        seq_j = _project_to_simplex(np.maximum(seq + jitter, 0.0), target_sum)
        candidates.append(seq_j)

    # power‑law family
    for p in np.linspace(0.25, 1.2, 50):
        seq = _power_law_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.001, size=n)
        seq_j = _project_to_simplex(np.maximum(seq + jitter, 0.0), target_sum)
        candidates.append(seq_j)

    # exponential decay family
    for lam in np.linspace(5e-4, 0.02, 30):
        seq = _exponential_sequence(n, lam)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.001, size=n)
        seq_j = _project_to_simplex(np.maximum(seq + jitter, 0.0), target_sum)
        candidates.append(seq_j)

    # triangular shape (piecewise linear, peak in the middle)
    tri = np.minimum(np.arange(n, dtype=float) + 1, n - np.arange(n, dtype=float))
    candidates.append(_project_to_simplex(tri, target_sum))

    # quadratic shape (parabolic bump)
    quad = (np.arange(n, dtype=float) + 1) * (n - np.arange(n, dtype=float))
    candidates.append(_project_to_simplex(quad, target_sum))

    # semi‑circular shape (discrete half‑circle)
    x = (2 * np.arange(n, dtype=float) + 1 - n) / n  # ∈(−1,1)
    semi = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    candidates.append(_project_to_simplex(semi, target_sum))

    # -----------------------------------------------------------------
    # Quick evaluation of the pool and pick the best few
    # -----------------------------------------------------------------
    best_val = float("inf")
    best_seq = None
    for seq in candidates:
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best from pool = {best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 1 – Adam optimisation on the top few candidates
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # re‑evaluate and sort to obtain the top performers
        scored = [( _max_convolution_fft(s), s) for s in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = min(5, len(scored))
        per_candidate = remaining * 0.35 / top_k
        for i in range(top_k):
            if time.time() >= deadline - 0.5:
                break
            seq0 = scored[i][1]
            sub_dead = min(deadline, time.time() + per_candidate)
            opt_seq, opt_val = adam_optimize(seq0, target_sum, sub_dead,
                                             beta_start=0.5,
                                             beta_factor=1.2,
                                             beta_max=3e5,
                                             base_lr=0.001)
            if opt_val < best_val:
                best_val, best_seq = opt_val, opt_seq.copy()
                print(f"[{time.time():.2f}] Adam improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 2 – Sub‑gradient refinement
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        sub_dead = time.time() + remaining * 0.20
        sg_seq, sg_val = subgradient_descent(best_seq, target_sum, sub_dead)
        if sg_val < best_val:
            best_val, best_seq = sg_val, sg_seq.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 3 – Deterministic mass transfer
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        mt_dead = time.time() + remaining * 0.30
        mt_seq, mt_val = deterministic_mass_transfer(best_seq, mt_dead,
                                                     target_sum,
                                                     start_temp=0.6,
                                                     end_temp=1e-5,
                                                     low_frac=0.04)
        if mt_val < best_val:
            best_val, best_seq = mt_val, mt_seq.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_val:.6f}")

    # -----------------------------------------------------------------
    # Phase 4 – Final high‑β Adam polishing
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        fin_seq, fin_val = adam_optimize(best_seq, target_sum,
                                         time.time() + remaining,
                                         beta_start=1.0,
                                         beta_factor=1.5,
                                         beta_max=8e5,
                                         base_lr=0.0005)
        if fin_val < best_val:
            best_val, best_seq = fin_val, fin_seq.copy()
            print(f"[{time.time():.2f}] Final polishing improved best to {best_val:.6f}")

    # --------------------- post‑process ---------------------
    arr = best_seq.copy()
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (for logging)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
