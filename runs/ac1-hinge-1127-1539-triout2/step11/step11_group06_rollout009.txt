def _lp_optimize(f, M):
    n = f.shape[0]
    # Build A matrix as described.
    # Use scipy.sparse.diags
    import scipy.sparse as sp
    # reps each diagonal as vector of length n with repeated f[s]
    data = np.vstack([np.full(n, f[s]) for s in range(n)]) # shape (n, n)
    offsets = np.arange(n)

    # objective c = -np.ones(n)
    c = -np.ones(n)

    bounds = [(0, None)] * n

"""
Hybrid LP‑based improvement + final stochastic refinement.

1.  Build a moderate‑size candidate (≈2000 entries) and normalise it to
    √(2 n).  Repeatedly solve a linear programme that maximises the L1‑norm
    of a new vector `g` under the constraint  f * g ≤ M  (where M is the
    current maximal autocorrelation of `f`).  After scaling `g` to the same
    L1‑norm as `f` we mix the two vectors and keep the improvement.
2.  Upsample the best moderate‑size vector to the final length
    (5000 – 12000 depending on the time budget) and re‑normalise.
3.  Apply a deterministic O(n) hill‑climber followed by a cheap O(n)
    simulated‑annealing phase to polish the solution.
4.  Return the best sequence found and store it globally for warm‑starts.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Core convolution utilities (FFT, scoring, simple helpers)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Linear‑program construction (f * g ≤ M)
# ----------------------------------------------------------------------


def _build_ub_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A such that (A @ g)[k] = (f * g)[k] for
    k = 0 … 2·n‑2.  Only the n non‑negative diagonals are non‑zero.
    """
    n = f.shape[0]
    # each diagonal s (0 ≤ s < n) contains the constant value f[s]
    data = [np.full(n, f[s], dtype=np.float64) for s in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_optimize(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j  s.t. 0 ≤ g_j , (f * g)_k ≤ M .
    Returns the optimal g (numpy array) or None on failure.
    """
    n = f.shape[0]
    A = _build_ub_matrix(f)
    c = -np.ones(n, dtype=np.float64)               # maximise Σ g  → minimise -Σ g
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # (f*g)_k ≤ M
    bounds = [(0, None)] * n

    try:
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception as e:  # pragma: no cover
        print(f"linprog exception: {e}")
        return None

    if res.success:
        return res.x
    print(f"linprog failed: {res.message}")
    return None


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (phase D)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """Exact hill‑climbing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.04  # initial temperature for occasional SA steps
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        # pick a donor proportional to its contribution to the peak
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # pick a receiver from indices that contribute little to the peak
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution (see original code)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total_time))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic re‑normalisation to avoid drift
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑normalisation improved score={best_val:.6f}")

        # occasional random jitter to escape plateaus
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap O(n) simulated annealing (phase E)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    """Move small mass between two indices, accepting occasional worsening moves."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    T0 = max(0.05, (best_val - 1.0) * 0.5)  # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # select a donor weighted by its current mass
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue

        # choose a distinct receiver uniformly
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.05, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # occasional global jitter
        if (int(elapsed * 100) % 5000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step sequence.
    The routine combines a small‑scale LP‑based alternating improvement with
    a final stochastic refinement, respecting the given time budget.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # size decisions depending on the time budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        final_n = 12000
    elif budget_s >= 500:
        final_n = 8000
    else:
        final_n = 5000

    target_sum_final = np.sqrt(2.0 * final_n)

    # --------------------------------------------------------------
    # build a moderate‑size starting vector (size suitable for LP)
    # --------------------------------------------------------------
    base_n = min(2000, final_n)          # LP works comfortably up to ≈2000
    base_target = np.sqrt(2.0 * base_n)

    if "height_sequence_1" in globals():
        base_seq = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), base_n)
    else:
        base_seq = (np.arange(1, base_n + 1, dtype=np.float64) ** (-0.55))

    a = base_seq * (base_target / base_seq.sum())
    M = _max_convolution_fft(a)
    best_val = _score(a)
    best_a = a.copy()
    print(f"[{time.time():.2f}] Initial base size {base_n}, score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – LP based alternating improvement
    # --------------------------------------------------------------
    lp_deadline = time.time() + 0.35 * (deadline - time.time())
    lp_iter = 0
    while time.time() < lp_deadline:
        g = _lp_optimize(a, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_scaled = g * (base_target / sum_g)

        # try a few mixing rates – keep the first that yields improvement
        mix = 0.3
        improved = False
        for _ in range(5):
            cand = (1 - mix) * a + mix * g_scaled
            cand_val = _score(cand)
            if cand_val < best_val:
                a = cand
                M = _max_convolution_fft(a)
                best_val = cand_val
                best_a = a.copy()
                improved = True
                print(f"[{time.time():.2f}] LP it={lp_iter} mix={mix:.3f} score={best_val:.6f}")
                break
            mix *= 0.5
        if not improved:
            print(f"[{time.time():.2f}] LP iteration {lp_iter} gave no improvement – stopping LP phase.")
            break
        lp_iter += 1

    # --------------------------------------------------------------
    # Phase 2 – upsample to the final length
    # --------------------------------------------------------------
    a_up = _resize_sequence(best_a, final_n)
    a_up = a_up * (target_sum_final / a_up.sum())
    cur_score = _score(a_up)
    print(f"[{time.time():.2f}] After upsampling score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic hill‑climbing (local refinement)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        local_deadline = time.time() + 0.25 * remaining
        a_up, cur_score = _local_refine(a_up, target_sum_final, local_deadline)
        print(f"[{time.time():.2f}] After local refinement score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – cheap simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_up, cur_score = _simulated_annealing(a_up, target_sum_final, deadline)
        print(f"[{time.time():.2f}] After simulated annealing score={cur_score:.6f}")

    # --------------------------------------------------------------
    # final clean‑up and evaluation
    # --------------------------------------------------------------
    arr = np.array(a_up, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    globals()["height_sequence_1"] = final_seq  # store for warm‑starts

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    return final_seq
