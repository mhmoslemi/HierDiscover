def _search_permutations(seq, num_samples, deadline, rng):
    best_seq = seq
    best_score = _score(seq)
    n = len(seq)
    attempts = 0
    while attempts < num_samples and time.time() < deadline:
        step = rng.integers(1, n)
        while np.gcd(int(step), n) != 1:
            step = rng.integers(1, n)
        offset = rng.integers(0, n)
        perm_seq = _step_permutation(seq, step, offset)
        val = _score(perm_seq)  # max convection after projection? Already sum normalized.
        if val < best_score:
            best_score = val
            best_seq = perm_seq
        attempts += 1
    return best_seq, best_score

def _swap_hill_climbing(seq, deadline, rng, max_iters=5000):
    best_seq = seq.copy()
    best_score = _score(best_seq)
    n = len(seq)
    while time.time() < deadline and max_iters>0:
        i, j = rng.choice(n, size=2, replace=False)
        new_seq = best_seq.copy()
        new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        val = _score(new_seq)
        if val < best_score:
            best_seq = new_seq
            best_score = val
        max_iters -= 1
    return best_seq, best_score

"""
Propose a low‑peak self‑convolution sequence.

The algorithm improves on the previous baseline by adding a
random‑step permutation search and a final hill‑climbing phase.
The overall scheme:

1.  Build a diverse pool of low‑dimensional seeds (power‑law,
    exponential, random simplex).  Run many short annealed gradient
    descents on a modest size (≈ 8 k) and keep the best sequence.
2.  Apply a short deterministic mass‑transfer refinement.
3.  Upscale the best seed to a larger length (12‑20 k depending on
    remaining time) and re‑project onto the ℓ¹‑simplex.
4.  Explore many permutations of the sorted values using a random
    coprime step (spreads large coefficients uniformly).  Keep the
    ordering with the smallest max‑convolution.
5.  Run a long sub‑gradient descent on the up‑scaled sequence.
6.  One more deterministic mass‑transfer pass, then a cheap
    pair‑swap hill‑climbing stage.
7.  Trim trailing near‑zeros and return the sequence.

All steps respect the non‑negativity and ℓ¹‑simplex constraints.
The objective for a sequence `a` with Σa = √(2·n) equals the
peak of the self‑convolution, i.e. `max(a*a)`.  The code uses FFT
convolutions for speed and projects onto the simplex after every
update.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _step_spread_order(seq: np.ndarray, step: int, offset: int = 0) -> np.ndarray:
    """
    Place the sorted values on positions (offset + i*step) mod n.
    `step` should be coprime with n for a full cycle.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(offset + i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed gradient descent on the soft‑max approximation
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2·(a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent_v2_ext(a0: np.ndarray,
                                     deadline: float,
                                     target_sum: float,
                                     max_beta: float = 5e5,
                                     step_factor: float = 0.7,
                                     gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD_ext it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # collect all indices attaining the maximum (numerical tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        # accumulate gradient contributions from every maximal offset
        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (a.shape[0] - 1))
            hi = min(k, a.shape[0] - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift a small amount of mass from a “donor’’
    (large product contribution to the current maximal convolution entry)
    to a “receiver’’ (small product contribution).  Moves are accepted
    with a Metropolis rule whose temperature decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Extra permutation / hill‑climbing utilities
# ----------------------------------------------------------------------


def _search_permutations(seq: np.ndarray,
                         num_samples: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Randomly spread the sorted values using a coprime step and offset.
    Keeps the permutation with the smallest max‑convolution.
    """
    n = seq.shape[0]
    best_seq = seq
    best_val = _score(seq)

    attempts = 0
    while attempts < num_samples and time.time() < deadline:
        step = rng.integers(1, n)
        while math.gcd(int(step), n) != 1:
            step = rng.integers(1, n)
        offset = rng.integers(0, n)
        cand = _step_spread_order(seq, int(step), int(offset))
        val = _score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
        attempts += 1

    return best_seq, best_val


def _swap_hill_climbing(seq: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: repeatedly swap two entries and keep
    the swap if it improves the max‑convolution.
    """
    best_seq = seq.copy()
    best_val = _score(best_seq)
    n = seq.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = rng.choice(n, size=2, replace=False)
        cand = best_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]
        val = _score(cand)
        if val < best_val:
            best_seq = cand
            best_val = val
        it += 1
    return best_seq, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose dimensionality for the exploratory phase
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 400:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family (more granularity)
    for p in np.linspace(0.2, 1.2, 11):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 8):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random simplex points
    for _ in range(10):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # re‑use the previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of everything to boost diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Exploration: many short annealed‑gradient runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # reserve roughly half of the budget for the later stages
    exploration_deadline = start_time + 0.45 * budget_s

    while time.time() < exploration_deadline - 5.0:   # keep a few seconds slack
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 18.0, exploration_deadline - 4.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent_v2_ext(
            seed, run_dead, target_expl,
            max_beta=5e5, step_factor=0.7, gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Brief deterministic mass‑transfer refinement at the small scale
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < exploration_deadline - 2.0:
        mt_dead = time.time() + (exploration_deadline - time.time()) * 0.15
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_expl,
            start_temp=0.4, end_temp=1e-4, low_frac=0.06,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Upscaling to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    # decide final length – longer if we still have a lot of time left
    if remaining > 180.0:
        n_final = 20000
    elif remaining > 120.0:
        n_final = 16000
    elif remaining > 60.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Try a few deterministic reorderings first
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_score)

    best_seq = seq_big
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Random step‑spread permutation search
    # ------------------------------------------------------------------
    perm_dead = time.time() + min(30.0, deadline - time.time() - 10.0)
    if perm_dead > time.time():
        # allow up to a few hundred samples (bounded by time)
        max_samples = 300
        seq_perm, score_perm = _search_permutations(
            best_seq, max_samples, perm_dead, rng)
        if score_perm < best_score:
            best_seq = seq_perm
            best_score = score_perm
            print(f"[{time.time():.2f}] Permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient refinement on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        sub_dead = deadline - 8.0
        a_sub, val_sub = _subgradient_descent(
            best_seq, sub_dead, target_final,
            step_factor=0.45, gamma=0.995,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 2.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final swap‑based hill climbing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        swap_dead = deadline - 1.0
        a_swap, val_swap = _swap_hill_climbing(best_seq, swap_dead, rng)
        if val_swap < best_score:
            best_score = val_swap
            best_seq = a_swap.copy()
            print(f"[{time.time():.2f}] Swap hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing negligible entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional final evaluation (purely for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible reuse in subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
