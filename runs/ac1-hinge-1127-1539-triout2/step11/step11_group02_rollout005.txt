"""
Improved step‑function search.

Main ideas
----------

* Start from the best known construction (global variable `height_sequence_1`);
  if its length differs from the working size, linearly interpolate it.
* A short dense parametric sweep (power‑law / exponential tails) supplies
  an alternative start if it beats the previous best.
* **Phase A – long projected sub‑gradient/Adam descent**  
  – Both a pure sub‑gradient method on the exact max‑convolution and an
    Adam optimiser on the soft‑max approximation are run (restarting from
    jittered copies of the current best).  The majority of the time budget
    (≈ 55 %) is spent here.
* **Phase B – deterministic greedy mass transfer** – the same exact
  peak‑reduction routine used previously, applied repeatedly until the
  deadline.
* **Phase C – final high‑β Adam polish** – a short Adam run with a huge
  soft‑max temperature to tighten the remaining slack.
* The sequence is projected onto the simplex {x≥0, Σx = √(2 n)} after every
  update, guaranteeing a valid candidate for the benchmark.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark score ( 2 n · max(conv) / (sum(a)²) )."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for a quick pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()  # soft‑max weights

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Pure sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    step = target_sum * 0.1  # initial step magnitude
    min_step = target_sum * 1e-8
    iteration = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # active set (indices attaining the maximum up to numerical tolerance)
        tol = max(1e-12, max_val * 1e-12)
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient of max(conv) : average over active indices
        g = np.zeros_like(a)
        n = a.size
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1)
            g[j] += 2.0 * a[k - j]
        if active.size:
            g /= float(active.size)

        # trial step
        a_new = a - step * g
        a_new = _project_to_simplex(a_new, target_sum)

        val_new = _score(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            step *= 1.05  # slightly accelerate when we improve
            print(
                f"[{time.time():.2f}] SubGrad it={iteration:5d} step={step:.6f} score={best_val:.6f}"
            )
        else:
            step *= 0.5  # shrink step when no improvement

        # occasional jitter to avoid stagnation
        if iteration % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a_new += jitter
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        iteration += 1
        if step < min_step:
            step = target_sum * 0.01  # reset to a modest step

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (same as before)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]
        k = idx_max[0]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[k - idx]
        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        denom = 2.0 * (a[k - donor] - a[k - receiver])
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] DMT it={int(time.time())} score={best_score:.6f}"
            )
        # continue

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the benchmark evaluation function.  The routine respects the time budget
    and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the allocated budget.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 650:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)  # Σa = √(2 n)

    # ------------------------------------------------------------------
    # Initial candidate (either the previous best or a parametric sweep).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (1) use the global previous construction if available
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            seq = np.interp(new_x, old_x, prev)
        else:
            seq = prev.copy()
        seq = _project_to_simplex(seq, target_sum)
        val = _score(seq)
        best_seq, best_score = seq, val
        print(f"[{time.time():.2f}] Started from previous best, score={val:.6f}")

    # (2) quick parametric pre‑search (runs only if time permits)
    presearch_deadline = time.time() + 0.10 * (deadline - time.time())
    if time.time() < presearch_deadline:
        # pure power‑law grid
        for p in np.arange(0.20, 1.21, 0.05):
            seq = _power_law_sequence(n, float(p), target_sum)
            val = _score(seq)
            if val < best_score:
                best_seq, best_score = seq, val
                print(f"[{time.time():.2f}] Power‑law p={p:.2f} improved to {val:.6f}")

        # exponential‑power‑law grid (coarse)
        for p in np.arange(0.30, 1.01, 0.10):
            for q in np.arange(0.0, 8.01, 1.0):
                seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
                val = _score(seq)
                if val < best_score:
                    best_seq, best_score = seq, val
                    print(
                        f"[{time.time():.2f}] Exp‑PL p={p:.2f} q={q:.1f} improved to {val:.6f}"
                    )

    # ------------------------------------------------------------------
    # Phase A – long optimisation (sub‑gradient + Adam, with restarts)
    # ------------------------------------------------------------------
    phase_a_deadline = time.time() + 0.55 * (deadline - time.time())
    while time.time() < phase_a_deadline - 5.0:
        # jittered start around the current best
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        # sub‑gradient descent
        a_sub, val_sub = _subgradient_descent(start_vec, phase_a_deadline, target_sum)
        if val_sub < best_score:
            best_seq, best_score = a_sub.copy(), val_sub
            print(
                f"[{time.time():.2f}] Sub‑grad improved best to {best_score:.6f}"
            )
            continue  # try another restart immediately

        # Adam optimisation (soft‑max) as a secondary fallback
        a_adam, val_adam = _adam_gradient_descent(
            start_vec,
            phase_a_deadline,
            target_sum,
            max_beta=5e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val_adam < best_score:
            best_seq, best_score = a_adam.copy(), val_adam
            print(
                f"[{time.time():.2f}] Adam improved best to {best_score:.6f}"
            )
        # loop continues until Phase A time expires

    # ------------------------------------------------------------------
    # Phase B – aggressive deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_mt < best_score - 1e-12:
            best_seq, best_score = a_mt.copy(), val_mt
            print(
                f"[{time.time():.2f}] Deterministic MT improved best to {best_score:.6f}"
            )
        else:
            break  # no further gain; exit early

    # ------------------------------------------------------------------
    # Phase C – final high‑β Adam polish
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_final, val_final = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_final < best_score:
            best_seq, best_score = a_final.copy(), val_final
            print(
                f"[{time.time():.2f}] Final Adam polish improved best to {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.asarray(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional sanity check (harmless if evaluate_sequence is not present)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
