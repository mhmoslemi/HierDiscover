"""
Multi‑start non‑negative step‑function optimizer.

The search proceeds as follows:

1. Choose a problem size `n` based on the time budget (larger budgets
   allow larger `n` for finer granularity).

2. Generate a diverse pool of initial sequences:
   – a resized version of the best known construction (`height_sequence_1`);
   – edge‑heavy power‑law families;
   – centre‑heavy power‑law families;
   – Gaussian blobs with different bandwidths;
   – a few random Dirichlet draws.

3. Normalise each candidate to the required sum √(2 n) and keep the
   three with the smallest objective value.

4. Run a short exponential‑mirror‑descent on every candidate
   (≈ 5 % of the total budget) to obtain quick improvements.

5. Take the best candidate and perform the full optimisation pipeline
   (mirror descent → sub‑gradient descent → projected GD with restarts
   → long simulated‑annealing mass‑transfer).  The annealing phase uses
   an exponential cooling schedule and periodically smoothes the
   sequence with a 3‑point kernel to further reduce the peak of the
   self‑convolution.

6. The best sequence found is trimmed of trailing zeros, evaluated
   with the provided `evaluate_sequence`, stored globally as
   `height_sequence_1` and returned.

All numbers are kept non‑negative, the sum is enforced exactly, and the
implementation respects the given time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used in mirror / GD phases)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 8e-04, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 5e-04, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = _project_to_simplex(a0, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray, target_sum: float, deadline: float):
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start, target_sum,
                                            time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – simulated‑annealing mass‑transfer refinement (with smoothing)
# ----------------------------------------------------------------------


def _mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Long‑run annealing that repeatedly moves tiny amounts of mass from
    high‑contributing indices towards low‑height indices.
    An exponential cooling schedule is used and a 3‑point smoothing
    kernel is applied periodically to further lower the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.30 if best_val > 0 else 0.1   # initial temperature

    smooth_interval = 3000
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        # exponential cooling (temperature -> 0)
        temperature = T0 * np.exp(-5.0 * frac)

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – pick proportionally to contribution
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour low heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.05, target_sum * 0.0015)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        # propose move (in‑place, revert on rejection)
        a[donor] -= delta
        a[receiver] += delta
        b_new = _conv_fft(a)
        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta

        # periodic smoothing
        if it % smooth_interval == 0 and it > 0:
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Smoothing improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate shape families
# ----------------------------------------------------------------------


def _edge_heavy_seq(n: int, p: float) -> np.ndarray:
    """Mass concentrated near the edges: (dist_to_edge+1)^(-p)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    seq = (d + 1.0) ** (-p)
    return seq.astype(np.float64)


def _center_heavy_seq(n: int, p: float) -> np.ndarray:
    """Mass concentrated near the centre: (max_dist - dist + 1)^(-p)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    max_d = d.max()
    seq = (max_d - d + 1.0) ** (-p)
    return seq.astype(np.float64)


def _gaussian_seq(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian blob."""
    x = np.arange(n) - (n - 1) / 2.0
    seq = np.exp(-0.5 * (x / sigma) ** 2)
    return seq.astype(np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 24000
    elif budget_s >= 600:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build a pool of diverse initial sequences
    # --------------------------------------------------------------
    candidates = []

    # warm‑start from the best known construction if present
    if "height_sequence_1" in globals():
        seq = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
        candidates.append(seq)

    # edge‑heavy power‑law family
    for p in (0.45, 0.5, 0.55, 0.6):
        seq = _edge_heavy_seq(n, p)
        candidates.append(seq)

    # centre‑heavy power‑law family
    for p in (0.45, 0.5, 0.55, 0.6):
        seq = _center_heavy_seq(n, p)
        candidates.append(seq)

    # Gaussian blobs
    for sigma in (n / 6.0, n / 4.0, n / 3.0):
        seq = _gaussian_seq(n, sigma)
        candidates.append(seq)

    # a couple of random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(2):
        seq = rng.dirichlet(alpha=np.ones(n), size=1).flatten()
        candidates.append(seq)

    # --------------------------------------------------------------
    # Normalise all candidates to the required sum and compute scores
    # --------------------------------------------------------------
    processed = []
    for seq in candidates:
        seq = np.array(seq, dtype=np.float64)
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0:
            continue
        seq = seq * (target_sum / seq.sum())
        sc = _score(seq)
        processed.append((seq, sc))

    if not processed:
        # fallback: simple decreasing power law
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        seq = seq * (target_sum / seq.sum())
        processed.append((seq, _score(seq)))

    # keep only the three best initial candidates
    processed.sort(key=lambda x: x[1])
    processed = processed[:3]

    # --------------------------------------------------------------
    # Quick mirror‑descent refinement on every candidate (≈ 5 % of budget)
    # --------------------------------------------------------------
    quick_budget = min(0.05 * (deadline - time.time()), 30.0)
    if quick_budget > 5.0:
        per_candidate = quick_budget / len(processed)
        refined = []
        for seq, _ in processed:
            sub_deadline = time.time() + per_candidate
            seq_opt, val_opt = _exponential_descent(seq, target_sum, sub_deadline)
            refined.append((seq_opt, val_opt))
        refined.sort(key=lambda x: x[1])
        best_seq, best_score = refined[0]
    else:
        # no time for quick phase – just take the best pre‑computed one
        best_seq, best_score = processed[0]

    # --------------------------------------------------------------
    # Full optimisation pipeline on the best candidate
    # --------------------------------------------------------------

    # Phase A – exponential (mirror) descent
    if time.time() < deadline:
        phase_deadline = time.time() + 0.10 * (deadline - time.time())
        best_seq, best_score = _exponential_descent(best_seq, target_sum, phase_deadline)

    # Phase B – subgradient descent
    if time.time() < deadline:
        phase_deadline = time.time() + 0.12 * (deadline - time.time())
        best_seq, best_score = _subgradient_descent(best_seq, target_sum, phase_deadline)

    # Phase C – projected GD with restarts
    if time.time() < deadline:
        phase_deadline = time.time() + 0.18 * (deadline - time.time())
        best_seq, best_score = _gradient_descent_with_restarts(best_seq, target_sum, phase_deadline)

    # Phase D – long simulated‑annealing mass‑transfer refinement
    if time.time() < deadline:
        best_seq, best_score = _mass_transfer_anneal(best_seq, target_sum, deadline)

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starting future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
