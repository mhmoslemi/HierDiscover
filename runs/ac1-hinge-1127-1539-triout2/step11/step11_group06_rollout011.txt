"""
Hybrid multi‑scale optimizer for the non‑negative step‑function benchmark.

The routine:
  1. Warm‑starts from the best known construction (or a power‑law).
  2. Performs a short exponential (mirror) descent on a soft‑max
     surrogate of the maximal convolution entry.
  3. Performs a brief sub‑gradient descent directly on the true max.
  4. Executes a long, aggressive simulated‑annealing phase that:
        * chooses donors weighted by their contribution to the current
          maximal convolution entry,
        * selects receivers among indices with little contribution,
        * moves relatively large mass fractions (up to 20 % of the donor),
        * periodically re‑normalises and recomputes the full convolution
          to avoid drift,
        * uses a slow linear cooling schedule.
  5. Returns the best sequence found (trimmed of trailing zeros) and
     stores it globally for future warm‑starts.

All phases respect a total time budget (default 1000 s).  The objective
value printed at the end is the exact score returned by the provided
`evaluate_sequence` function.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate (used in exponential descent)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of the max‑convolution entry."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # grad = 2 * (reverse(a) convolve p)
    a_rev = a[::-1]
    grad = np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                        np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = 2.0 * grad[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on soft‑max
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpGD it={it:5d} β={beta:6.1f} max={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – (multiplicative) sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Multiplicative subgradient descent directly on the maximal convolution entry."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear decay of step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase E – aggressive simulated annealing (O(n) updates)
# ----------------------------------------------------------------------


def _simulated_annealing_improved(a0: np.ndarray,
                                  target_sum: float,
                                  deadline: float) -> tuple[np.ndarray, float]:
    """SA with donor selection based on contribution to the current max entry."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = b.max()
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    # initial temperature (more generous than the original)
    T0 = max(0.1, (best_val - 1.0) * 1.0)

    it = 0
    while time.time() < deadline:
        # --- choose donor (weighted by contribution to current max) ---
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx_range] * a[k_max - idx_range]
        w_sum = contrib.sum()
        if w_sum > 0:
            donor = rng.choice(idx_range, p=contrib / w_sum)
        else:
            donor = rng.choice(idx_range)

        # --- choose receiver (prefer indices with little contribution) ---
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = contrib
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.20, target_sum * 0.005)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # --- update convolution in O(n) (exact) ---
        b_new = b.copy()
        # remove contribution of donor
        b_new[donor:donor + n] -= 2.0 * delta * a
        # add contribution of receiver
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # self‑interaction adjustments
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = b_new.max()

        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved max={best_val:.6f}")

        # periodic re‑normalisation to avoid drift
        if (it % 5000) == 0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = b.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA re‑norm improved max={best_val:.6f}")

        # occasional global jitter to shake out of plateaus
        if (int(elapsed * 100) % 6000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0003, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = b_j.max()
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    # a tiny safety margin at the end
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size – larger n gives a richer search space
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000          # more degrees of freedom
    elif budget_s >= 500:
        n = 14000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # warm‑start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
    else:
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = _project_to_simplex(init, target_sum)

    # ------------------------------------------------------------------
    # Phase A – a short exponential mirror descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    exp_deadline = time.time() + min(20.0, remaining * 0.07)  # ≈7 % of the budget
    best_a, best_val = _exponential_descent(init, target_sum, exp_deadline)
    print(f"[{time.time():.2f}] After exponential descent max={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – a brief sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(20.0, remaining * 0.07)
        a_sub, val_sub = _subgradient_descent(best_a, target_sum, sub_deadline)
        if val_sub < best_val:
            best_a, best_val = a_sub, val_sub
        print(f"[{time.time():.2f}] After sub‑gradient descent max={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – long aggressive simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing_improved(best_a,
                                                     target_sum,
                                                     deadline)
        if val_sa < best_val:
            best_a, best_val = a_sa, val_sa
        print(f"[{time.time():.2f}] After simulated annealing max={best_val:.6f}")

    # ------------------------------------------------------------------
    # final clean‑up
    # ------------------------------------------------------------------
    arr = np.array(best_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final external evaluation (will be shown in the logs)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq

    return final_seq
