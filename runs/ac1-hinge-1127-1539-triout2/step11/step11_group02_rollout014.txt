"""
Hybrid optimizer for the step‑function benchmark.

What it does
------------
1. **Very wide parametric pre‑search** – Exhaustively evaluates several
   low‑dimensional families (power‑law, power‑law+exponential,
   broken power‑law, plateau+exponential, beta‑shape).  The best candidate
   is kept as a warm‑start.

2. **Adam‑style projected gradient descent** – Runs several restarts of
   Adam on a soft‑max approximation of the max–convolution.  The schedule
   for the temperature β is exponential, ending at a very large value
   (≈5·10⁵).  Restarts are performed until the allocated time expires.

3. **Exact sub‑gradient descent on the true max‑convolution** –  Uses the
   exact sub‑gradient (the indicator of the maximal entries convolved
   with the reversed sequence) and a simple diminishing step size.  This
   often squeezes a little more out of the current best shape.

4. **Aggressive deterministic mass transfer** – The same greedy
   “pull mass from the biggest contributor to the smallest” as in the
   reference implementation.

5. **Light random‑move annealing** – Occasionally tries tiny stochastic
   reshuffles to escape plateaus.

6. **Final polishing** – One last Adam run with an even larger soft‑max
   temperature.

All steps respect the simplex constraint Σ a = √(2 n) and a≥0, use FFT‑based
convolutions for speed and stay within the 1000 s budget.  The routine
returns the best sequence found (trailing near‑zero entries are trimmed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (full linear convolution)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (identical to max convolution when Σa = √(2 n))."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _plateau_exponential_sequence(
    n: int, alpha: float, q: float, target_sum: float
) -> np.ndarray:
    """Plateau of constant height (fraction alpha) then exponential tail."""
    K = max(1, int(alpha * n))
    tail_len = n - K
    a = np.empty(n, dtype=np.float64)
    a[:K] = 1.0
    if tail_len > 0:
        i = np.arange(tail_len, dtype=np.float64)
        a[K:] = np.exp(-q * i / tail_len)
    a *= target_sum / a.sum()
    return a


def _beta_shape_sequence(
    n: int, a_exp: float, b_exp: float, target_sum: float
) -> np.ndarray:
    """Shape ~ (i+1)^a_exp * (n-i)^b_exp (Beta‑distribution‑like)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** a_exp) * ((n + 1 - i) ** b_exp)
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12).astype(np.float64)

        # sub‑gradient = 2 * (a_rev * mask) restricted to n entries
        a_rev = a[::-1]
        n = a.size
        size = 1 << ((2 * n - 1).bit_length())
        ar_fft = np.fft.rfft(a_rev, n=size)
        mk_fft = np.fft.rfft(mask, n=size)
        conv_rev = np.fft.irfft(ar_fft * mk_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        norm = np.linalg.norm(grad) + 1e-12
        # Diminishing step – scale so that a typical entry moves by ≈5 % of the target sum
        step = min(target_sum * 0.5 / norm, target_sum * 0.1)

        a_new = a - step * grad
        a_new = _project_to_simplex(a_new, target_sum)

        val = _score(a_new)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] Subgrad step score={best_val:.6f}")

        a = a_new

        # early stop if gradient becomes tiny
        if norm < 1e-8:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]
        k = idx_max[0]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[k - idx]
        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        denom = 2.0 * (a[k - donor] - a[k - receiver])
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={int(time.time())} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 1.0:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step‑function with minimal benchmark score."""
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ---------------------------------------------------------------
    # Problem size – a bit larger when we have enough time.
    # ---------------------------------------------------------------
    if budget_s >= 900:
        n = 16000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)   # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (still very cheap)
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) pure power‑law
    for p in np.arange(0.20, 1.31, 0.02):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail

            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:

    # (c) broken power‑law
    for p1 in np.arange(0.0, 0.81, 0.2):
        for p2 in np.arange(0.8, 4.01, 0.4):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) plateau + exponential tail
    for alpha in np.arange(0.05, 0.51, 0.05):
        for q in np.arange(0.0, 6.001, 0.5):
            seq = _plateau_exponential_sequence(n, float(alpha), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (e) beta‑shape family
    for a_exp in np.arange(0.0, 2.01, 0.5):

            seq = _beta_shape_sequence(n, float(a_exp), float(b_exp), target_sum)
            val = _score(seq)
            if val < best_score:

    # (f) reuse previous global best if present
    if "height_sequence_1" in globals():

        if prev.size != n:

    return final_seq
