def _lp_alternating_step(a, target, t_weight, deadline):
    n = a.size
    M = _max_convolution_fft(a)
    # build convolution matrix A (size (2n-1) x n)
    # Use vectorized method
    k = np.arange(2*n-1)[:, None]
    i = np.arange(n)[None, :]
    idx = k - i
    mask = (idx >= 0) & (idx < n)
    # A = np.zeros((2*n-1, n), dtype=float)
    # A[mask] = a[idx[mask]]
    # Instead allocate and assign, but we can compute directly for linprog 'A_ub' with zeros where mask false: 
    A = np.zeros((2*n-1, n), dtype=float)
    A[mask] = a[idx[mask]]
    # Solve LP
    c = -np.ones(n)  # maximize sum(g)
    b_ub = np.full(2*n-1, M, dtype=float)
    bounds = [(0, None)] * n
    # Try to set time_limit for linprog if available (SciPy 1.10+). Use options dict.
    # Use method 'highs'.
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': max(0, deadline - time.time())})
    except TypeError:
        # older version may not have time_limit
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return unchanged
        return a, False
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 == 0:
        return a, False
    g = g0 * (target / sum_g0)
    a_new = (1 - t_weight) * a + t_weight * g
    # project onto simplex again
    a_new = _project_to_simplex_fast(a_new, target)
    # check improvement
    new_M = _max_convolution_fft(a_new)
    improved = new_M < M - 1e-12
    return a_new, improved
```

    time_left = max(0.0, deadline - time.time())
    options = {}
    if time_left > 0:
        options['time_limit'] = time_left
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if not res.success:
        return a, False
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return a, False
    g = g0 * (target / sum_g0)
    a_new = (1 - t_weight) * a + t_weight * g
    a_new = _project_to_simplex_fast(a_new, target)
    new_M = _max_convolution_fft(a_new)
    return a_new, new_M < M - 1e-12

"""
Optimisation of step‑function using alternating LP + Adam + hill‑climb.
"""

import time, math, numpy as np

# Try to import SciPy; if unavailable we will skip the LP stage.
try:
    import scipy.optimize as opt
    import scipy.sparse as sp
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _project_to_simplex_fast(v, s):
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)

def _max_convolution_fft(a):
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _full_convolution(a):
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _upsample_sequence(a, new_n):
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)

def _random_initial_sequence(n, rng):
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)
    elif family == 1:
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)
    elif family == 2:
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif family == 3:
        a = rng.dirichlet(alpha=np.ones(n)) * target
    elif family == 4:
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
    else:
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b

def _adam_optimize(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _adam_optimize_submax(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.2 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1

        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a

def _hill_climb_deterministic(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        counterpart = np.zeros(n, dtype=float)
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try smaller move
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score

def _conv_matrix_sparse(a):
    """Return sparse matrix A such that A @ g = f * g (full convolution)."""
    n = a.shape[0]
    rows = 2 * n - 1
    # allocate arrays
    total = n * n
    data = np.empty(total, dtype=float)
    row_ind = np.empty(total, dtype=int)
    col_ind = np.empty(total, dtype=int)
    pos = 0
    for i in range(n):
        row_ind[pos:pos + n] = np.arange(i, i + n)
        col_ind[pos:pos + n] = i
        data[pos:pos + n] = a
        pos += n
    return sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

def _lp_alternating_step(a, target, t_weight, deadline):
    """One step of the alternating linear‑programming scheme."""
    n = a.shape[0]
    M = _max_convolution_fft(a)

    if not SCIPY_AVAILABLE:
        return a, False

    # convolution matrix for constraints f * g <= M
    A = _conv_matrix_sparse(a)

    c = -np.ones(n)  # maximise sum(g)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0.0, None)] * n

    time_left = max(0.0, deadline - time.time())
    opts = {}
    if time_left > 0.0:
        opts['time_limit'] = time_left

    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options=opts)

    if not res.success:
        return a, False

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0.0:
        return a, False

    g = g0 * (target / sum_g0)
    a_new = (1.0 - t_weight) * a + t_weight * g
    a_new = _project_to_simplex_fast(a_new, target)
    new_M = _max_convolution_fft(a_new)
    improved = new_M < M - 1e-12
    return a_new, improved

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method combines
        • a small‑scale alternating linear‑programming refinement,
        • Adam optimisation on smooth‑max and exact‑max surrogates,
        • a deterministic hill‑climb,
    and respects a wall‑clock budget (default 1000 s).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # start from a known construction if present
    if 'height_sequence_1' in globals():
        try:
            start = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            start = None
    else:
        start = None

    best_seq = None
    best_val = float('inf')

    # length schedule – we grow the dimension gradually.
    schedule = [256, 512, 1024, 2048]

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        if start is None:
            a = _random_initial_sequence(n, rng)
        else:
            if start.size != n:
                a = _upsample_sequence(start, n)
            else:
                a = start.copy()
        a = _project_to_simplex_fast(a, target)

        # time split for this size
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        lp_budget = remaining * 0.45
        adam_budget = remaining * 0.35
        hc_budget = remaining * 0.20

        # ----- LP phase -------------------------------------------------
        if SCIPY_AVAILABLE:
            # crude estimate of LP cost; allocate a handful of steps.
            est_step = max(0.5, n / 400.0)   # seconds per LP approx.
            max_steps = max(1, int(lp_budget // est_step))
            for step in range(max_steps):
                if time.time() >= deadline:
                    break
                t_weight = 0.4 if step < max_steps / 2 else 0.2
                a, _ = _lp_alternating_step(a, target, t_weight, deadline)

        # ----- Adam phase -----------------------------------------------
        if time.time() < deadline - 0.5:
            # smooth‑max Adam
            a = _adam_optimize(a, target, min(deadline - 0.2,
                                             time.time() + adam_budget * 0.5))
            # exact‑max Adam for the rest of the budget
            a = _adam_optimize_submax(a, target, deadline - 0.2)

        # ----- Deterministic hill‑climb ---------------------------------
        if time.time() < deadline - 0.5:
            b = _full_convolution(a)
            a_hc, _ = _hill_climb_deterministic(a, b, deadline - 0.5)
            a = a_hc

        # ----- Store best ------------------------------------------------
        try:
            val = evaluate_sequence(a.tolist())
        except Exception as exc:
            print(f"evaluate_sequence failed: {exc}")
            val = float('inf')
        print(f"[{time.time():.1f}] n={n}, evaluate={val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            # use this as seed for larger sizes
            start = best_seq.copy()

    # ----- Final polishing on whatever time remains --------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        best_seq = _adam_optimize_submax(best_seq, target,
                                         min(deadline - 0.5,
                                             time.time() + 2.0))
        b = _full_convolution(best_seq)
        best_seq, _ = _hill_climb_deterministic(best_seq, b,

"""
Optimisation of a non‑negative step function for the given evaluation metric.

The algorithm proceeds in four stages for a growing sequence length:
    1.  An alternating linear‑programming refinement (the “LP step” from the
        state‑of‑the‑art description).  It solves
                maximise Σ g_j   s.t.   f * g ≤ M ,   g ≥ 0
        where M is the current maximal convolution value of f.
        The solution g₀ is rescaled to the required ℓ¹‑norm and mixed with f.
    2.  Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
    3.  Adam optimisation on the exact sub‑gradient of the max‑convolution.
    4.  A deterministic hill‑climb that moves mass from the entries that
        create the current worst convolution peak.

A simple length schedule (256 → 512 → 1024 → 2048) is used; the result for a
larger size is obtained by linear interpolation of the best vector from the
previous size.  The whole routine respects a wall‑clock budget (default 1000 s)
and returns the best sequence found.

All helper functions are defined at the top level, no closures are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – required only for the LP phase
# ----------------------------------------------------------------------
try:
    import scipy.optimize as opt
    import scipy.sparse as sp
    SCIPY_AVAILABLE = True
except Exception:          # SciPy may be missing in the execution environment
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the ℓ¹‑simplex {x ≥ 0, Σx = s}.
    Implemented by a simple rescaling after clamping to non‑negative.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of
    max(a*a).  The active set (indices attaining the maximum) is used as
    the probability vector p.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.2 * target_sum          # a slightly larger base step
    t = 0

    while time.time() < deadline:
        t += 1

        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climb on the exact convolution
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        counterpart = np.zeros(n, dtype=float)
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score

# Linear‑programming utilities (only used when SciPy is available)

def _conv_matrix_sparse(a: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A such that (A @ g) = f * g (full linear convolution).
    Shape of A is (2·n‑1, n), where n = len(a).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    total = n * n
    data = np.empty(total, dtype=float)
    row_ind = np.empty(total, dtype=int)
    col_ind = np.empty(total, dtype=int)

    pos = 0
    for i in range(n):
        row_ind[pos:pos + n] = np.arange(i, i + n, dtype=int)
        col_ind[pos:pos + n] = i
        data[pos:pos + n] = a
        pos += n

    return sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))


def _lp_alternating_step(a: np.ndarray, target: float,
                        t_weight: float, deadline: float) -> tuple[np.ndarray, bool]:
    """
    Perform a single alternating LP step:

        * M = max(a * a)
        * maximise Σ g_j  s.t.  a * g ≤ M , g ≥ 0
        * rescale g to have ℓ¹‑norm = target
        * set a ← (1‑t)·a + t·g   and project onto the simplex

    Returns the new vector and a flag indicating whether the max‑convolution
    decreased.
    """
    n = a.shape[0]

    # current maximal convolution value
    M = _max_convolution_fft(a)

    if not SCIPY_AVAILABLE:
        return a, False

    # convolution constraints matrix
    A = _conv_matrix_sparse(a)

    c = -np.ones(n)                     # maximise Σ g_j  ⇒  minimise -Σ g_j
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0.0, None)] * n

    # Solve the LP (HiGHS handles sparse matrices natively)
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs')

    if not res.success:
        return a, False

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0.0:
        return a, False

    # rescale to the required ℓ¹‑norm
    g = g0 * (target / sum_g0)

    # convex combination and projection
    a_new = (1.0 - t_weight) * a + t_weight * g
    a_new = _project_to_simplex_fast(a_new, target)

    new_M = _max_convolution_fft(a_new)
    improved = new_M < M - 1e-12
    return a_new, improved

    """

    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Seed with a previously obtained construction if it exists.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            start = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:

    else:
        start = None

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Length schedule – grow the dimension gradually.
    # ------------------------------------------------------------------
    schedule = [256, 512, 1024, 2048]

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # initialise either from the previous best (upsampled) or at random
        if start is None:
            a = _random_initial_sequence(n, rng)
        else:
            if start.size != n:
                a = _upsample_sequence(start, n)
            else:
                a = start.copy()
        a = _project_to_simplex_fast(a, target)

        # ------------------------------------------------------------------
        # Time allocation for this size
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        lp_budget = remaining * 0.45
        adam_budget = remaining * 0.35
        hc_budget = remaining * 0.20

        # --------------------------------------------------------------
        # 1) Linear‑programming refinement
        # --------------------------------------------------------------
        if SCIPY_AVAILABLE:
            # Rough estimate of the cost of a single LP (seconds)
            est_step = max(0.5, n / 400.0)
            max_steps = min(30, max(1, int(lp_budget // est_step)))
            for step in range(max_steps):
                if time.time() >= deadline:
                    break
                # larger mixing weight early on, smaller later
                t_weight = 0.4 if step < max_steps / 2 else 0.2
                a, _ = _lp_alternating_step(a, target, t_weight, deadline)

        # --------------------------------------------------------------
        # 2) Adam optimisation (smooth max → exact max)
        # --------------------------------------------------------------
        if time.time() < deadline - 0.5:
            # smooth‑max phase – half of the allocated Adam budget
            a = _adam_optimize(
                a, target,
                min(deadline - 0.2, time.time() + adam_budget * 0.5))
            # exact‑max sub‑gradient phase – remaining time
            a = _adam_optimize_submax(a, target, deadline - 0.2)

        # --------------------------------------------------------------
        # 3) Deterministic hill‑climb on the true convolution
        # --------------------------------------------------------------
        if time.time() < deadline - 0.5:
            b = _full_convolution(a)
            a, _ = _hill_climb_deterministic(a, b, deadline - 0.5)

        # --------------------------------------------------------------
        # 4) Record the best sequence seen so far
        # --------------------------------------------------------------
        try:
            val = evaluate_sequence(a.tolist())
        except Exception as exc:
            print(f"evaluate_sequence failed: {exc}")
            val = float('inf')
        print(f"[{time.time():.1f}] n={n}, evaluate={val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            start = best_seq.copy()     # use as seed for the next size

    # ------------------------------------------------------------------
    # Final polishing on whatever time is left
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        best_seq = _adam_optimize_submax(
            best_seq, target,
            min(deadline - 0.5, time.time() + 2.0))
        b = _full_convolution(best_seq)
        best_seq, _ = _hill_climb_deterministic(
            best_seq, b, deadline - 0.5)

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zero entries
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 256
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] final evaluate={final_val:.6f}, "
              f"length={best_seq.size}")
    except Exception as exc:  # pragma: no cover
        print(f"final evaluate failed: {exc}")

    return best_seq.tolist()
