def additive_subgradient_descent(a0, target_sum, deadline):
   a = a0.copy()
   a = a * (target_sum / a.sum())
   best_val = _score(a)
   best = a.copy()

   start = time.time()
   it = 0
   while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        # compute gradient
        lo = max(0, k_max - (len(a)-1))
        hi = min(k_max, len(a)-1)
        idx = np.arange(lo, hi+1)
        # g over all positions
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # step size schedule
        # linear decay: eta = eta0 * (1 - (time_elapsed / total_time)) + eta_min
        elapsed = time.time() - start
        total = deadline - start
        eta0 = 0.2
        eta_min = 0.001
        eta = eta0 * (1 - elapsed/total) + eta_min

        a = a - eta * g
        # enforce non-neg
        a = np.maximum(a, 0.0)
        # renorm sum
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            # reinitialize random
            a = np.random.random_like(a)
            a = a * (target_sum / a.sum())

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] AddSGD it={it} score={best_val:.6f}")
        it += 1

   return best, best_val

"""
Hybrid optimizer for the non‑negative step‑function benchmark (v2).

Main ideas
-----------
* Warm‑start with the best known construction *and* a short sweep over
  power‑law exponents (−0.4 … −0.7). The best of these is kept.
* Phase A – Adam on a soft‑max surrogate of the maximal convolution entry.
  Adam gives fast, well‑conditioned additive updates.
* Phase B – pure additive sub‑gradient descent on the exact max‑convolution
  (the sub‑gradient is 2·a_{k‑i} where k is the current maximal index).
  A linearly decaying step‑size quickly flattens the peak.
* Phase C – deterministic hill‑climbing (exact “mass‑transfer”) as in the
  reference code, but given more time.
* Phase D – cheap O(n) simulated‑annealing moves for final fine‑tuning.
* All phases respect the global time‑budget (default 1000 s) and the best
  sequence found so far is remembered across calls (global variable
  ``height_sequence_1``).

The score for a normalized sequence (∑a = sqrt(2 n)) is exactly the
maximum entry of the self‑convolution, therefore the goal is to make that
maximum as small as possible.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²) – with the canonical normalisation this
    equals ``max(a*a)``."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to ``new_len`` entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient of a soft‑max surrogate (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (grad, conv) where ``grad`` is the gradient of the soft‑max
    approximation of max(a*a) and ``conv`` is the exact convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_full = np.fft.irfft(np.fft.rfft(a_rev, n=size) *
                             np.fft.rfft(p, n=size), n=size)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1:n - 1 + n]   # centre part corresponds to indices 0…n‑1
    return grad, b


# ----------------------------------------------------------------------
# Phase A – Adam on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Additive Adam optimisation of the soft‑max surrogate."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05                        # initial learning rate
    lr_min = 0.005
    beta = 1.0
    max_beta = 65536.0
    start = time.time()
    it = 0

    while time.time() < deadline:
        # gradually increase the soft‑max sharpness
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        g, _ = _gradient_softmax(a, beta)

        it += 1
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (linear decay)
        elapsed = time.time() - start
        total = deadline - start
        lr = lr0 * (1.0 - elapsed / total) + lr_min * (elapsed / total)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0006, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – additive sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _additive_subgradient_descent(a0: np.ndarray,
                                  target_sum: float,
                                  deadline: float):
    """Plain additive sub‑gradient descent on max(a*a)."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # compute convolution and locate the current maximal index
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # linear decay of the step size
        elapsed = time.time() - start
        total = deadline - start
        eta0 = 0.12
        eta_min = 0.001
        eta = eta0 * (1.0 - elapsed / total) + eta_min * (elapsed / total)

        a = a - eta * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.random.random_like(a)
            a = a * (target_sum / a.sum())

        # jitter every few iterations
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – exact local “mass‑transfer’’ hill‑climbing
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  target_sum: float,
                  deadline: float):
    """Deterministic hill‑climbing that moves a tiny amount of mass between two
    indices while updating the convolution in O(n)."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = best_val * 0.04                     # initial temperature for occasional uphill moves
    last_recalc = start

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        # choose a donor proportionally to its contribution to the maximal entry
        if prod.sum() > 0:
            donor = rng.choice(idx, p=prod / prod.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # pick a relatively light receiver (tries to move mass away from the peak)
        weight_all = np.zeros(n, dtype=a.dtype)
        weight_all[lo:hi + 1] = prod
        thresh = np.percentile(weight_all, 30)
        low_idx = np.where(weight_all <= thresh)[0]
        if low_idx.size:
            receiver = rng.choice(low_idx)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.04, target_sum * 0.0008)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # simulated‑annealing acceptance
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Local refine improved score={best_val:.6f}")

        # periodic full re‑normalisation to avoid drift
        if time.time() - last_recalc > 12.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_recalc = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Re‑norm improved score={best_val:.6f}")

        # occasional tiny random swap to break plateaus
        if (int(elapsed * 100) % 2500) == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0004)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Tiny swap improved score={best_val:.6f}")

    return best_a, best_val

# Phase D – cheap O(n) simulated annealing (final fine‑tuning)

def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float):
    """Moves small amounts of mass between two indices; O(n) update of the
    convolution; accepts occasional worsening moves."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    T0 = max(0.05, (best_val - 1.0) * 0.5)   # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # choose a donor weighted by its current mass
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue
        # distinct receiver uniformly
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.05, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved score={best_val:.6f}")

        # occasional global jitter to shake out of plateaus
        if (int(elapsed * 100) % 5000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA global jitter improved score={best_val:.6f}")

    return best_a, best_val

def _make_powerlaw(n: int, exponent: float) -> np.ndarray:
    """Return a decreasing power‑law ``a_i = (i+1)^exponent`` normalised."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** exponent)
    return a


def _pick_best_warm_start(candidates, target_sum):
    """Return the candidate with the lowest score (after normalisation)."""
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        seq = seq.astype(np.float64)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    return best_seq, best_val

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 14000                     # a bit larger than before – more freedom
    elif budget_s >= 500:
        n = 10000
    else:
        n = 6000

    # --------------------------------------------------------------
    # warm‑start: reuse previous best + a few power‑law candidates
    # --------------------------------------------------------------

            _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)

    print(f"[{time.time():.2f}] Warm‑start score = {init_val:.6f}")

    if time.time() < deadline:

        if val_adam < init_val:
