def _simulated_annealing(a0, target_sum, deadline):

    a = a0.copy()
    best = a.copy()
    best_score = _score(best)
    curr_score = best_score

    # compute temperature schedule: start_T maybe (best_score - 1) * something, ensure >0.
    # Use exponential cooling: T = T0 * 0.99**iter
    # Compute number of iterations allowed approx (deadline - time.time()) / per-iteration time (~0.001 sec). We'll just loop until time.

    T0 = max(0.1, (best_score - 1) * 0.2) # maybe small.

    T = T0
    iter = 0
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # choose donor and receiver
        # sample donor weighted by a (so bigger mass more likely)
        donor = rng.choice(n, p=a / a.sum())
        # receiver: choose index with probability proportional to (target_sum - a_i) (capacity)
        cap = target_sum - a
        if cap.sum() <= 0:
            break
        receiver = rng.choice(n, p=cap / cap.sum())
        if donor == receiver:
            continue
        # delta fraction of donor mass (up to some fraction)
        max_delta = a[donor] * 0.3
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # apply move
        a[donor] -= delta
        a[receiver] += delta
        # ensure nonnegative and sum unchanged (thanks to move)
        # compute new score
        new_score = _score(a)
        if new_score < curr_score:
            curr_score = new_score
            # keep this state as current
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"{time.time():.2f} SA new best {best_score:.6f}")
        else:
            # accept with probability exp(-(new_score - curr_score)/T)
            prob = math.exp(-(new_score - curr_score) / T) if T > 0 else 0
            if rng.random() < prob:
                curr_score = new_score
            else:
                # reject: revert move
                a[donor] += delta
                a[receiver] -= delta
        # cooling
        T *= 0.9999  # small decay
        iter +=1

    return best, best_score

"""
Propose a step‑function with a small max‑convolution value.

Algorithmic outline
--------------------
1. **Rich parametric seeding** – generate candidates from several
   families (power‑law, power‑law × exp‑tail, Gaussian, Beta, and a
   mixed power‑law × beta factor).  The best candidate is kept.
2. **Projected Adam on a soft‑max of the convolution** – the usual
   smooth‑max trick makes the objective differentiable.  Several
   restarts with jitter are performed while respecting the time budget.
3. **Deterministic mass‑transfer (exact)** – repeatedly lower the
   current peak of the convolution by moving just enough mass from a
   “donor’’ index (large partner value) to a “receiver’’ index (small
   partner value).  This step drives the solution very close to a
   local optimum.
4. **Simulated annealing** – a cheap stochastic move (transfer a small
   amount of mass between two indices) is accepted according to a
   temperature schedule.  This helps to escape tiny plateaus that the
   deterministic step can get stuck in.
5. **Final polishing** – a last Adam run with a very large soft‑max
   temperature finishes the refinement.

All phases respect the global deadline (default 1000 s) and the
sequence is always projected onto the simplex
{aᵢ ≥ 0, Σaᵢ = √(2 n)} so that the evaluation function reduces to
the maximal convolution entry.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Evaluation: with the chosen normalisation the score equals max(conv)."""
    max_b = _max_convolution_fft(a)
    # sum is forced to sqrt(2n) elsewhere; a tiny safety check
    if a.sum() < 0.01:
        return np.inf
    return max_b


# ----------------------------------------------------------------------
# Projection onto the simplex Σx = s, x ≥ 0 (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i ** 2) / (2.0 * sigma * sigma))
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    # x∈(0,1] – avoid zero to keep a finite value
    x = (np.arange(1, n + 1, dtype=np.float64) / (n + 1))
    a = (np.power(x, alpha - 1.0) *
         np.power(1.0 - x, beta - 1.0))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a *= target_sum / a.sum()
    return a


def _power_law_beta_sequence(n: int, p: float, r: float, target_sum: float) -> np.ndarray:
    """ (i+1)^{-p} * (1 - i/(n-1))^{r}  – extra decay towards the end """
    i = np.arange(n, dtype=np.float64)
    tail = 1.0 - i / (n - 1.0)
    tail = np.maximum(tail, 0.0)  # safety
    a = (i + 1) ** (-p) * (tail ** r)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv) w.r.t. a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max weights

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # central part corresponds to ∂/∂a_i
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimiser with a soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Several short Adam runs with random jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (peak‑reduction)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        # choose donor (index with large partner) among positions that still have mass
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose receiver (index with minimal partner) different from donor
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        # coefficients of donor / receiver in the chosen peak
        partner_donor = partner[donor_local]
        receiver_local = int(np.where(idx == receiver)[0][0])
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        # required amount to reduce the peak to the second‑largest value
        delta_needed = (max_val - second_max) / denom

        # do not move more than 70 % of the donor's mass (stability)
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (global stochastic exploration)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Simple SA that transfers a small amount of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    # initial temperature – a fraction of the current excess over the trivial lower bound (≈1)
    T = max(0.01, (best_score - 1.0) * 0.2)
    it = 0

    while time.time() < deadline:
        # choose donor weighted by its current mass (more mass → more likely)
        donor = rng.choice(n, p=a / a.sum())
        # choose receiver weighted by how much “room’’ it has (target_sum - a_i)
        room = target_sum - a
        if room.sum() <= 0:
            break
        receiver = rng.choice(n, p=room / room.sum())
        if donor == receiver:
            continue

        max_delta = a[donor] * 0.3
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta

        new_score = _score(a)
        improve = new_score < cur_score
        if improve or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA iter={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta

        # geometric cooling
        T *= 0.9999
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (light exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal max‑convolution.
    The routine respects the given time budget and returns the best
    sequence found (as a plain Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a tiny safety margin

    # Choose problem size – larger n gives more degrees of freedom.
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) ⇒ the score equals max(convolution)

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # 0a) previous construction (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _score(prev)
        if val < best_score:
            best_score = val
            best_seq = prev.copy()
            print(f"[{time.time():.2f}] Seed from previous: score={best_score:.6f}")

    # 0b) power‑law grid
    p_grid = np.linspace(0.15, 2.0, 31)
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 0c) power‑law × exponential tail
    q_grid = np.linspace(0.0, 8.0, 31)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # 0d) Gaussian (monotone decreasing)
    sigma_grid = np.linspace(0.02 * n, 0.30 * n, 20)
    for sigma in sigma_grid:
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # 0e) Beta family (flexible shapes)
    alpha_grid = np.linspace(0.3, 5.0, 13)
    beta_grid = np.linspace(0.3, 5.0, 13)
    for al in alpha_grid:
        for be in beta_grid:
            seq = _beta_sequence(n, float(al), float(be), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # 0f) mixed power‑law × beta factor (extra decay)
    r_grid = np.linspace(0.0, 5.0, 11)
    for p in p_grid:
        for r in r_grid:
            seq = _power_law_beta_sequence(n, float(p), float(r), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # 0g) small jittered population around the current best
    for _ in range(12):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 4.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best={best_score:.6f}")

        # occasional random move to break possible stagnation
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best={best_score:.6f}")

        if time.time() > deadline - 4.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Simulated annealing (global stochastic search)
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq,
                                            target_sum,
                                            deadline)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] SA improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Final high‑β Adam polish
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: strip trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
