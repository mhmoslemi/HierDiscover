"""
Hybrid optimizer for the self‑convolution bound.

The method consists of three stages:

1. **Exploration** – many short runs of annealed gradient descent
   (soft‑max approximation of the max‑convolution) on a modest
   dimension (≈6 k).  A diverse pool of power‑law, exponential and
   random simplex seeds is used.  The best sequence found is kept.

2. **Upscaling & polishing** – the best low‑dimensional sequence is
   interpolated to a larger size (≈12–16 k).  A few simple reorderings
   (original, alternating, even‑spread) are tried, then a long
   projected sub‑gradient descent on the exact max‑convolution is run.
   Finally a deterministic “mass‑transfer’’ step (simulated annealing)
   is applied.

3. **Finalisation** – trailing near‑zero entries are trimmed and the
   result is returned as a plain list of floats.

All steps respect the non‑negativity and ℓ¹–simplex constraints.
The algorithm adapts its time budget automatically (total budget
≈ 1000 s) and prints intermediate scores for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Gradient‑descent on the soft‑max approximation
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max‑convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2·(a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e7,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of the
    max‑convolution objective.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the exact max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining the maximum (numerical tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _mass_transfer(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  start_temp: float = 0.4,
                  end_temp: float = 1e-5,
                  low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift a small amount of mass from a “donor’’
    (large product contribution to the current peak) to a “receiver’’
    (small contribution).  Moves are accepted with a Metropolis rule
    whose temperature decays exponentially.
    """
    a = a0.copy()
    best_a = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # location of current maximal convolution entry
        kmax = int(np.argmax(conv))

        n = a.shape[0]
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contributed by each index to the peak (i, kmax‑i)
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        if max_delta <= 0:
            it += 1
            continue

        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to keep exploring
        if it % 3000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    The total wall‑clock time is limited by ``budget_s`` seconds.

    deadline = start_time + budget_s - 0.5  # safety margin

    rng = np.random.default_rng()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n_expl = 6000
    target_expl = np.sqrt(2.0 * n_expl)

    # Build a diverse pool of seeds
    init_pool = []

    for p in np.linspace(0.2, 1.0, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    for lam in np.linspace(0.001, 0.018, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    for _ in range(8):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of the pool to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # Run many short annealed‑gradient runs
    while time.time() < deadline - 30.0:   # keep 30 s for later phases
        # choose seed
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 20.0, deadline - 25.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_dead, target_expl,
            max_beta=5e6, step_factor=0.7, gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Short deterministic mass‑transfer refinement (still small size)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.10
        a_ref, val_ref = _mass_transfer(
            best_seq, mt_dead, target_expl,
            start_temp=0.5, end_temp=1e-4, low_frac=0.06,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After MT refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Upscaling to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 180.0:
        n_final = 16000
    elif remaining > 120.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few simple reorderings and keep the one with smallest max‑conv
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)

    best_seq = seq_big
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Long sub‑gradient descent on the large sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        sub_dead = deadline - 12.0
        a_sub, val_sub = _subgradient_descent(
            best_seq, sub_dead, target_final,
            step_factor=0.45, gamma=0.995,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_final, val_final = _mass_transfer(
            best_seq, mt_dead, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 7. Optional logging of the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for possible reuse in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
