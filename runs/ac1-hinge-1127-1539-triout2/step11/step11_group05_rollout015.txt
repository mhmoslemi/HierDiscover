"""
A fast multi‑phase optimiser for the non‑negative step‑function problem.

Main ideas
----------
1. Initialise with a diverse set of parametric families (power‑law,
   exponential, sine‑power, beta‑like, logistic and two‑piece power).
   All candidates are projected onto the simplex
   Σ a_i = √(2 n).

2. Random‑search phase: while time permits we repeatedly draw a random
   shape from the families above, evaluate its max self‑convolution (FFT)
   and keep the best one.  Whenever a new record is found a short
   soft‑max gradient descent is run to polish it.

3. Refine the current best sequence with three classic phases from the
   reference implementation:
   * annealed soft‑max momentum GD,
   * sub‑gradient descent on the true max,
   * deterministic mass‑transfer (local annealing).

4. Final high‑β polishing and a tiny trailing‑zero trim.

All heavy work uses FFT (O(n log n)) and the whole routine respects the
overall time budget (default 1000 s).  The global variable
`height_sequence_1` is used for warm‑starts across runs.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # only for an optional sanity check, not required

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric shape generators (unnormalised)
# ----------------------------------------------------------------------
def _power_law_seq(n: int, p: float) -> np.ndarray:
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_seq(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    # use log‑space to stay stable for small exponents
    log_vals = (alpha - 1) * np.log(i) + (beta - 1) * np.log(n - i + 1)
    return np.exp(log_vals)


def _logistic_seq(n: int, k: float, m: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return 1.0 / (1.0 + np.exp(k * (i - m)))


def _two_piece_power_seq(n: int, cut: int, p_left: float, p_right: float) -> np.ndarray:
    seq = np.empty(n, dtype=np.float64)
    left_idx = np.arange(1, cut + 1, dtype=np.float64)
    right_idx = np.arange(cut + 1, n + 1, dtype=np.float64)
    seq[:cut] = left_idx ** (-p_left)
    seq[cut:] = right_idx ** (-p_right)
    return seq


# ----------------------------------------------------------------------
# Soft‑max gradient (used for GD)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv). Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – annealed momentum GD on the soft‑max approximation
# ----------------------------------------------------------------------
def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 3e5,
                                   step_factor: float = 0.6,
                                   gamma: float = 0.995) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase X – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic greedy mass transfer (local annealing)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, kmax‑i)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random shape generator for the exploratory phase
# ----------------------------------------------------------------------
def _random_candidate(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    family = rng.integers(0, 6)
    if family == 0:          # power‑law
        p = rng.uniform(0.2, 1.2)
        seq = _power_law_seq(n, p)
    elif family == 1:        # exponential
        lam = rng.uniform(5e-5, 3e-2)
        seq = _exponential_seq(n, lam)
    elif family == 2:        # sine‑power
        p = rng.uniform(0.1, 3.0)
        seq = _sin_power_seq(n, p)
    elif family == 3:        # beta‑like
        alpha = rng.uniform(0.1, 1.5)
        beta = rng.uniform(0.1, 1.5)
        seq = _beta_seq(n, alpha, beta)
    elif family == 4:        # logistic
        k = rng.uniform(0.001, 0.05)
        m = rng.uniform(0.2 * n, 0.8 * n)
        seq = _logistic_seq(n, k, m)
    else:                    # two‑piece power law
        cut = int(rng.uniform(0.2, 0.8) * n)
        p_left = rng.uniform(0.2, 1.5)
        p_right = rng.uniform(0.2, 1.5)
        seq = _two_piece_power_seq(n, cut, p_left, p_right)

    seq = np.maximum(seq, 0.0)
    seq = _project_to_simplex(seq, target_sum)
    return seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximum
    entry of its self‑convolution.  Returns a plain Python list ready for
    `evaluate_sequence`.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        if budget_s >= 800:
            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Helper for fast evaluation
    # --------------------------------------------------------------
    def _eval(a: np.ndarray) -> float:
        return _max_convolution_fft(a)

    # --------------------------------------------------------------
    # Build an initial diversified pool (deterministic families)
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            # simple linear resize
            old_x = np.linspace(0, 1, prev.shape[0])
            new_x = np.linspace(0, 1, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        candidates.append(prev)

    # power‑law grid
    for p in np.linspace(0.25, 1.0, 61):
        seq = _power_law_seq(n, p)
        candidates.append(_project_to_simplex(seq, target_sum))
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        candidates.append(_project_to_simplex(seq_j, target_sum))

    # exponential grid
    for lam in np.linspace(1e-4, 2.5e-2, 50):
        seq = _exponential_seq(n, lam)
        candidates.append(_project_to_simplex(seq, target_sum))
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        candidates.append(_project_to_simplex(seq_j, target_sum))

    # sine‑power grid
    for p in np.linspace(0.2, 3.0, 40):
        seq = _sin_power_seq(n, p)
        candidates.append(_project_to_simplex(seq, target_sum))
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        candidates.append(_project_to_simplex(seq_j, target_sum))

    # beta‑like grid
    for a in np.linspace(0.2, 1.3, 20):
        for b in np.linspace(0.2, 1.3, 20):
            seq = _beta_seq(n, a, b)
            candidates.append(_project_to_simplex(seq, target_sum))

    # --------------------------------------------------------------
    # Quick evaluation of the deterministic pool
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for seq in candidates:
        sc = _eval(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
    print(f"[{time.time():.2f}] Best after deterministic pool: {best_score:.6f}")

    # --------------------------------------------------------------
    # Random‑search phase (large number of cheap evaluations)
    # --------------------------------------------------------------
    random_phase_end = time.time() + (deadline - time.time()) * 0.45
    while time.time() < random_phase_end:
        cand = _random_candidate(n, target_sum, rng)
        sc = _eval(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.2f}] New random record: {best_score:.6f}")

            # quick local polishing with a few GD iterations
            gd_dead = time.time() + min(2.0, random_phase_end - time.time())
            cand_ref, sc_ref = _annealed_gradient_descent_ext(
                best_seq, gd_dead, target_sum,
                max_beta=2e5, step_factor=0.7, gamma=0.99
            )
            if sc_ref < best_score:
                best_score, best_seq = sc_ref, cand_ref.copy()
                print(f"[{time.time():.2f}] Polished random record: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – soft‑max GD on the current best candidate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        gd_dead = time.time() + remaining * 0.30
        best_seq, best_score = _annealed_gradient_descent_ext(
            best_seq, gd_dead, target_sum,
            max_beta=4e5, step_factor=0.6, gamma=0.995
        )
        print(f"[{time.time():.2f}] GD phase result: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase X – sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        sub_dead = time.time() + remaining * 0.25
        best_seq, best_score = _subgradient_descent(
            best_seq, target_sum, sub_dead,
            step_factor=0.55, gamma=0.95
        )
        print(f"[{time.time():.2f}] Sub‑gradient phase result: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – deterministic mass‑transfer
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        mt_dead = time.time() + remaining * 0.30
        best_seq, best_score = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.6, end_temp=1e-5, low_frac=0.04
        )
        print(f"[{time.time():.2f}] Mass‑transfer phase result: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – final high‑β polishing
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        final_dead = time.time() + remaining
        best_seq, best_score = _annealed_gradient_descent_ext(
            best_seq, final_dead, target_sum,
            max_beta=6e5, step_factor=0.5, gamma=0.998
        )
        print(f"[{time.time():.2f}] Final polishing result: {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑process, final evaluation and global store
    # --------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=float)
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]
    final_seq = final_arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
