def project_to_simplex(v, s=1):
    # ensure v>=0 after projection, but algorithm works for any v: sorts descending.
    if s < 0:
        raise ValueError

    # Based on Duchi et al: find rho, theta.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence that (almost) minimizes
the maximum entry of its self‑convolution, i.e. the value returned by
``evaluate_sequence`` (after ℓ¹‑normalisation it equals the peak
convolution).  The optimiser consists of

* random initial families,
* Adam on a smooth‑max (log‑sum‑exp) surrogate,
* Adam on the exact max (sub‑gradient) with a large learning rate,
* a fine‑grained Adam run with a very small step,
* cheap random pair moves,
* a deterministic hill‑climb that moves mass from the worst
  convolution entry.

All phases respect a wall‑clock budget (default 1000 s) and the best
vector seen is returned.  The implementation works with a fixed
length 2000 (the length used by the reference solution) but can be
started from a previously discovered sequence via the global
``height_sequence_1`` variable.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers (several simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex(a, target)


# ----------------------------------------------------------------------
# Gradient of smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    A_fft = np.fft.rfft(a_rev, n=size2)
    P_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(A_fft * P_fft, n=size2)[:conv_len]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam with smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize_smooth(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Beta (smoothness) grows exponentially up to a very large value.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam on the exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float,
                          base_lr_factor: float = 0.2) -> np.ndarray:
    """
    Projected Adam where the gradient is a sub‑gradient of the true max.
    The parameter ``base_lr_factor`` controls the magnitude of the step.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = base_lr_factor * target_sum
    t = 0

    while time.time() < deadline:
        t += 1
        # full convolution and active mask
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:               # safety
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # gradient = 2·(a_rev ⋆ p)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Fine‑grained Adam (tiny learning rate)
# ----------------------------------------------------------------------


def _fine_descent(a0: np.ndarray, target_sum: float,
                  deadline: float) -> np.ndarray:
    """Adam on the exact max with a very small step size."""
    return _adam_optimize_submax(a0, target_sum, deadline, base_lr_factor=0.05)


# ----------------------------------------------------------------------
# Simple random‑pair refinement
# ----------------------------------------------------------------------


def _random_pair_refine(a: np.ndarray, target_sum: float,
                        deadline: float, rng) -> tuple[np.ndarray, float]:
    """
    Cheap Monte‑Carlo moves that shift a small fraction of mass from one
    coordinate to another.  Returns the best vector seen and its score.
    """
    n = a.size
    best_a = a.copy()
    best_score = _max_convolution_fft(best_a)

    while time.time() + 0.5 < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a modest fraction (up to 30 %) of the source mass
        frac = rng.uniform(0.01, 0.30)
        delta = best_a[i] * frac
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, target_sum)
        score = _max_convolution_fft(cand)
        if score < best_score - 1e-12:
            best_a = cand
            best_score = score
            print(f"[{time.time():.1f}] Random‑pair improved score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb (exact max)
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        # indices that actually contribute to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # destination with the smallest counterpart
        counterpart = np.zeros(n, dtype=float)
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # linear‑time update of the convolution array
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # tiny jitter to escape plateaus
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Fixed problem size (the reference uses n = 2000)
    # ------------------------------------------------------------------
    n = 2000
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise – use the previously discovered construction if present,
    # otherwise draw a random one.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            if a0.size != n:
                a0 = _upsample_sequence(a0, n)
            a0 = _project_to_simplex(a0, target_sum)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (score={_max_convolution_fft(a0):.6f})")
        except Exception as exc:
            print(f"Seeding error: {exc}")
            a0 = _random_initial_sequence(n, rng)
    else:
        a0 = _random_initial_sequence(n, rng)

    best_a = a0.copy()
    best_score = _max_convolution_fft(best_a)
    print(f"[{time.time():.1f}] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam (exploratory)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.30)
    a1 = _adam_optimize_smooth(best_a, target_sum, phase1_deadline)
    sc = _max_convolution_fft(a1)
    if sc < best_score - 1e-12:
        best_score, best_a = sc, a1.copy()
        print(f"[{time.time():.1f}] After smooth‑max Adam score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact‑max Adam (sub‑gradient) – aggressive step
    # ------------------------------------------------------------------
    phase2_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.30)
    a2 = _adam_optimize_submax(best_a, target_sum, phase2_deadline, base_lr_factor=0.2)
    sc = _max_convolution_fft(a2)
    if sc < best_score - 1e-12:
        best_score, best_a = sc, a2.copy()
        print(f"[{time.time():.1f}] After sub‑max Adam score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine‑grained Adam (tiny steps)
    # ------------------------------------------------------------------
    phase3_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.15)
    a3 = _fine_descent(best_a, target_sum, phase3_deadline)
    sc = _max_convolution_fft(a3)
    if sc < best_score - 1e-12:
        best_score, best_a = sc, a3.copy()
        print(f"[{time.time():.1f}] After fine‑descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap random‑pair moves
    # ------------------------------------------------------------------
    phase4_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.10)
    a4, sc = _random_pair_refine(best_a, target_sum, phase4_deadline, rng)
    if sc < best_score - 1e-12:
        best_score, best_a = sc, a4.copy()
        print(f"[{time.time():.1f}] After random‑pair refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – deterministic hill‑climb on the exact max
    # ------------------------------------------------------------------
    if time.time() + 0.5 < deadline:
        b_cur = _full_convolution(best_a)
        a5, sc = _hill_climb_deterministic(best_a, b_cur, deadline)
        if sc < best_score - 1e-12:
            best_score, best_a = sc, a5.copy()
            print(f"[{time.time():.1f}] After hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polish (use remaining time, if any)
    # ------------------------------------------------------------------
    if time.time() + 0.5 < deadline:
        a_final = _adam_optimize_submax(best_a, target_sum, deadline, base_lr_factor=0.03)
        sc = _max_convolution_fft(a_final)
        if sc < best_score - 1e-12:
            best_score, best_a = sc, a_final.copy()
            print(f"[{time.time():.1f}] Final Adam polish score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for a tidy output
    # ------------------------------------------------------------------
    nz = np.where(best_a > 1e-12)[0]
    if nz.size:
        best_a = best_a[: nz[-1] + 1]
    else:
        best_a = best_a[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation (for debugging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_a.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_a.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_a.tolist()
