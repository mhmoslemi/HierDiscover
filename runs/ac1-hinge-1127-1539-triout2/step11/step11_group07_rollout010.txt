    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = np.inf
    # attempt seeding from height_sequence_1
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(globals()['height_sequence_1'], dtype=float)
            target0 = math.sqrt(2.0 * prev.size)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(prev)
        except Exception as e:
            print('Failed to seed: ', e)
    # schedule lengths
    schedule = [1000, 2000, 4000, 8000, 16000]
    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)
        # initial vector: either upsample previous or random
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)
        a_init = _project_to_simplex_fast(a_init, target)
        # time allocation
        remaining = deadline - time.time()
        if remaining < 5:
            break
        opt_budget = remaining * 0.55
        hc_budget = remaining * 0.30
        extra_budget = remaining * 0.15
        # number of restarts
        min_restart = 5.0
        n_restarts = max(1, int(opt_budget // min_restart))
        per_restart = opt_budget / n_restarts
        local_best = None
        local_best_score = np.inf
        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            # start vector: use a_init only for first restart, otherwise random
            start_vec = a_init if r == 0 else _random_initial_sequence(n, rng)

            best_seq = local_best.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] New best after n={n} score={best_score:.6f}")
        # hill climb
        if time.time() < deadline - 2:
            a_cur = best_seq.copy()

        # random tweaks after hill climb
        while time.time() + 2 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)

            if score < best_score - 1e-12:
                best_seq = cand
                best_score = score

        # extra post-hc Adam if time
        if time.time() < deadline - 2 and extra_budget > 5:

            a_extra = _adam_optimize_submax(best_seq, target, extra_deadline)

    # final polishing
    if best_seq is not None and time.time() < deadline - 1:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 0.5

def _random_initial_sequence(n, rng):
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 8)  # up to 7 inclusive
    idx = np.arange(n, dtype=float)
    a = np.empty(n)
    if family == 0:  # power law decreasing
        p = rng.uniform(0.1, 1.2)
        a = (idx + 1.0) ** (-p)
    elif family == 1:  # exponential decreasing
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)
    elif family == 2:  # mixed power * exponential
        p = rng.uniform(0.1, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)
    elif family == 3:  # Dirichlet uniform
        a = rng.dirichlet(alpha=np.ones(n)) * target
    elif family == 4:  # piecewise exponential (two segments)
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])
    elif family == 5:  # near uniform jitter
        a = np.full(n, target / n)
    elif family == 6:  # symmetric exponential (edges high)
        lam = rng.exponential(scale=0.02)
        dist = np.minimum(idx, n - 1 - idx)
        a = np.exp(-lam * dist)
    elif family == 7:  # symmetric power law
        p = rng.uniform(0.1, 1.2)
        dist = np.minimum(idx + 1, n - idx)
        a = dist ** (-p)
    else:
        a = np.full(n, target / n)
    # add small jitter
    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

The routine combines several ideas:
1.   Randomly initialise a non‑negative vector on the ℓ¹‑simplex
     Σa = √(2 n).  The families include edge‑heavy (U‑shaped) shapes
     that tend to flatten the self‑convolution.
2.   Two‑phase Adam optimisation:
     – a smooth soft‑max surrogate of the max‑convolution,
     – a true sub‑gradient descent on the exact max.
3.   A deterministic greedy hill‑climb that moves a little mass
     from indices that most contribute to the current peak towards
     indices with the smallest counterpart.
4.   Light simulated‑annealing style random tweaks and a final polish.

The algorithm respects a wall‑clock budget (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}:
    clip to non‑negative and renormalise the ℓ¹‑norm.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # uniform fallback
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation onto a new length, followed by ℓ¹‑simplex projection.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers (several simple families, now including edge‑heavy)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Sample a non‑negative vector on the simplex Σa = √(2 n).

    The families cover decreasing, increasing, mixed, Dirichlet and
    explicitly edge‑heavy (U‑shaped) shapes.
    """
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 8)          # extended to 8 families
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law decreasing
        p = rng.uniform(0.1, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential decreasing
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.1, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 5:                   # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    elif family == 6:                   # symmetric exponential (edges high)
        lam = rng.exponential(scale=0.02)
        dist = np.minimum(idx, n - 1 - idx)          # distance to nearest edge
        a = np.exp(-lam * dist)

    elif family == 7:                   # symmetric power‑law (edges high)
        p = rng.uniform(0.1, 1.2)
        dist = np.minimum(idx + 1, n - idx)          # 1‑based distance to edge
        a = dist ** (-p)

    else:                               # fallback uniform
        a = np.full(n, target / n)

    # small Gaussian jitter to avoid perfect symmetry stagnation
    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp) – phase 1
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth‑max surrogate (1/β)·log Σ exp(β·b),
    where b = a*a (full linear convolution).
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p, keep central n entries
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n - 2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        # slowly increase the smoothness parameter
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Adam optimiser – exact max (sub‑gradient) – phase 2
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of
    max(a*a).  The active set (indices attaining the maximum) forms
    the probability vector p.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.20 * target_sum          # a little larger than phase 1
    t = 0

    while time.time() < deadline:
        t += 1

        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:       # safety
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # sub‑gradient g = 2·(a_rev ⋆ p) (central n entries)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        # which a‑indices contribute to b[M] ?
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # pick a destination index that supplies the smallest counterpart
        counterpart = np.zeros(n, dtype=float)
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart[j_range] = a[M - j_range]

        # select the index with the smallest counterpart (but different from i)
        order = np.argsort(counterpart[j_range])
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # linear‑time update of the convolution after moving delta from i to j
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score
