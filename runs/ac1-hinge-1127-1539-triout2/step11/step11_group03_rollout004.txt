"""
Improved heuristic search for low‑L∞ self‑convolution.
The algorithm proceeds in stages:

1.  Build a diverse pool of candidate step‑functions (power‑law,
    exponential, random Dirichlet, linear, previous best).
2.  Run many short Adam‑based soft‑max gradient‑descent passes on a
    moderate size (≈ 8000).  The soft‑max approximates the max‑convolution
    and yields a smooth gradient; Adam accelerates convergence.
3.  Refine the current best with a Metropolis mass‑transfer annealing that
    repeatedly moves a small amount of mass from “donor’’ indices that
    contribute heavily to the current peak to “receiver’’ indices that
    contribute little.
4.  Upscale the best sequence to a larger length (≈ 10‑12 k) by linear
    interpolation and re‑project to the ℓ¹‑simplex.
5.  Perform a final high‑β Adam pass on the large problem.
6.  Apply a targeted permutation‑search (simulated‑annealing on the
    index order) to further reduce the peak.
7.  One more short mass‑transfer pass and trimming of trailing zeros.

All steps respect non‑negativity and the ℓ¹‑normalisation
∑a_i = √(2 n); consequently the reported score equals the maximum
convolution entry.  The routine adapts its phase lengths to the
remaining time budget (default 1000 s) and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Scoring
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa = √(2n) this equals max(a*a))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest left, 2nd largest right, 3rd left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max approximation of max‑convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(self‑convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_softmax_descent(a0: np.ndarray,
                          deadline: float,
                          target_sum: float,
                          max_beta: float = 1e6,
                          base_lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate of the max‑convolution.
    Keeps the vector on the ℓ¹‑simplex.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    it = 0
    beta = 0.5
    rng = np.random.default_rng()
    step_base = target_sum * base_lr

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        a = a - step_base * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # increase β (sharpen soft‑max) gradually
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic / stochastic mass‑transfer annealing
# ----------------------------------------------------------------------


def _mass_transfer(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   start_temp: float = 0.5,
                   end_temp: float = 1e-4,
                   low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Metropolis annealing that moves a small amount of mass from a donor
    (large contribution to the current peak) to a receiver (small
    contribution).  The temperature decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak at shift kmax
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        # bias delta toward the gap between donor and receiver contributions
        weight_gap = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_gap > 0:
            max_delta = min(max_delta, weight_gap * 0.3)

        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 2500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.12,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.03,
                                 low_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style reorderings that swap a “donor’’ (large
    contribution to the current peak) with a “receiver’’ (small
    contribution).  The ℓ¹‑sum stays unchanged.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # contribution of each position i to the current peak
        partner = kmax - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        if new_score >= best_score:
            # Metropolis acceptance for worsening moves
            if rng.random() >= np.exp(-(new_score - best_score) / max(temp, 1e-12)):
                # revert swap
                a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Phase 1 – many short Adam gradient‑descent runs on the small size
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate roughly half of the total time to this phase
    phase1_dead = min(deadline - 30.0, start_time + 0.45 * (deadline - start_time))

    while time.time() < phase1_dead:
        # pick a seed (jittered best or random pool entry)
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = _project_to_simplex(np.maximum(seed, 0.0), target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 12.0, phase1_dead)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _adam_softmax_descent(
            seed,
            run_dead,
            target_expl,
            max_beta=5e5,
            base_lr=0.7,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1b – short mass‑transfer refinement (still on n_expl)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < phase1_dead:
        mt_dead = min(deadline - 15.0, phase1_dead + 4.0)
        a_ref, val_ref = _mass_transfer(best_seq, mt_dead, target_expl,
                                        start_temp=0.5, end_temp=1e-4, low_frac=0.07)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Post‑exploration MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 40.0:
        n_final = 12000
    elif remaining > 20.0:
        n_final = 10000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings
    cand_variants = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big)
    ]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in cand_variants[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final high‑β Adam polishing on the large size
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_big_opt, val_big_opt = _adam_softmax_descent(
            best_seq,
            deadline - 4.0,
            target_final,
            max_beta=2e6,
            base_lr=0.5,
        )
        if val_big_opt < best_score:
            best_score = val_big_opt
            best_seq = a_big_opt.copy()
            print(f"[{time.time():.2f}] Final Adam polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – targeted permutation search (re‑ordering only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = time.time() + min(12.0, deadline - time.time() - 1.0)
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.12,
            end_temp=1e-5,
            top_frac=0.03,
            low_frac=0.03,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – tiny final mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 0.5)
        a_fin, val_fin = _mass_transfer(best_seq, mt_dead, target_final,
                                        start_temp=0.3, end_temp=1e-5, low_frac=0.07)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros (compact representation) and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # final scoring for logging (optional)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible reuse in a next call
    globals()["height_sequence_1"] = final_seq
    return final_seq
