    # jittered copies

    print(f"Initial best {best_val}")

        if val_opt < best_val:

        if val_fin < best_val:
            best_val, best_seq = val_fin, a_fin.copy()
            print(f"Final polishing improved best to {best_val}")

    if nz.size:

"""
Propose a step‐function (non‑negative heights) that minimizes the
maximum entry of its self‑convolution.

Algorithm Overview
------------------
1. **Problem size** – Choose n (≈12000 for a full 1000 s budget) and set the
   ℓ¹‑norm target `sqrt(2·n)`.  All sequences are projected onto the simplex
   `{x≥0, Σx = target_sum}`.

2. **Candidate pool** – Generate a diverse set of parametric shapes
   (power‑law, exponential, sine‑power, Gaussian‑like, tent) and jittered
   copies.  The previous best sequence (global `height_sequence_1`) is also
   included and resized if necessary.

3. **Quick selection** – Evaluate every candidate via an FFT‑based convolution
   and keep the few best for intensive refinement.

4. **Soft‑max gradient descent** – For each top candidate run a momentum GD on a
   smooth approximation of the max‑convolution (`log‑sum‑exp` with growing β).
   The iterates are projected onto the simplex.

5. **Simulated‑annealing mass transfer** – Starting from the current best,
   repeatedly move a small amount of mass from a donor (large contribution to
   the current peak) to a receiver (small contribution).  Moves are accepted
   according to a temperature schedule, allowing occasional uphill steps.

6. **Final polishing** – One more soft‑max GD with a very large β to fine‑tune
   the solution.

7. **Post‑process** – Trim trailing near‑zero entries, store the result in the
   global `height_sequence_1` for warm‑starts, and return it as a plain list.

All heavy work uses O(n log n) FFTs; the routine respects the 1000 s budget
by checking the wall‑clock time frequently.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x≥0
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (unnormalised)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    return np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)


def _tent_sequence(n: int) -> np.ndarray:
    """A symmetric triangular shape, zero at the ends."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    return np.maximum(mid - np.abs(idx - mid), 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv). Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (rev(a) ⋆ p) restricted to central n entries
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Phase A – Annealed momentum GD on soft‑max
# ----------------------------------------------------------------------
def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 3e5,
                                   step_factor: float = 0.6,
                                   gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Momentum GD on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        # increase β gradually
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing mass transfer
# ----------------------------------------------------------------------
def _simulated_mass_transfer(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             start_temp: float = 0.2,
                             end_temp: float = 1e-6,
                             low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer with SA acceptance, moving mass from high‑ to low‑contrib indices."""
    a = a0.copy()
    n = a.shape[0]

    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    start_time = time.time()

    it = 0
    while time.time() < deadline:
        # temperature schedule (geometric)
        tau = (time.time() - start_time) / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # compute full convolution once per iteration
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # contribution of each index to the current maximal entry
        j = kmax - np.arange(n)
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = a[valid] * a[j[valid]]

        donor = int(np.argmax(contrib))
        # low‑contributing candidates (sorted)
        low_candidates = np.where(valid)[0]
        if low_candidates.size == 0:
            break
        sorted_low = low_candidates[np.argsort(contrib[low_candidates])]
        low_cnt = max(1, int(len(sorted_low) * low_frac))
        receiver = int(rng.choice(sorted_low[:low_cnt]))

        if donor == receiver:
            it += 1
            continue

        # propose a transfer amount
        max_delta = a[donor] * 0.5
        diff = contrib[donor] - contrib[receiver]
        if diff > 0:
            max_delta = min(max_delta, diff * 0.5)
        delta = rng.random() * max_delta
        if delta <= 1e-12:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        # safeguard – numerical drift
        a = np.maximum(a, 0.0)

        new_max = _max_convolution_fft(a)
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            accept = rng.random() < prob

            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()

        else:

            a[donor] += delta
            a[receiver] -= delta

        # occasional jitter to re‑explore
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """

            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

        candidates.append(_project_to_simplex(prev, target_sum))

    # Power‑law family

    # Exponential decay family
    for lam in np.linspace(5e-4, 0.012, 13):

    # Sine‑power family
    for p in np.linspace(0.2, 2.0, 13):

    # Gaussian‑like family (different widths)
    for sigma in [0.04, 0.07, 0.10, 0.16]:
