Implementation of a multi‑stage optimizer for the max‑convolution bound.
Stage 0 – dense parametric sweep (power law + exponential tail).  
Stage 1 – projected Adam (soft‑max) gradient descent with restarts.  
Stage 2 – aggressive deterministic pairwise mass‑transfer that moves the
           maximal safe amount of mass from the “donor’’ index (largest
           contribution to the current peak) to a “receiver’’ index (smallest
           contribution), guaranteeing that the new peak does not exceed the
           previous second‑largest value.  
Stage 3 – final soft‑max polishing.
All steps respect a hard time‑budget (default 1000 s) and return the best
sequence found, trimmed of trailing near‑zero entries.
"""

import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())

def _score(a: np.ndarray) -> float:
    """Objective value – for sum_a = sqrt(2 n) this coincides with max‑convolution."""
    max_b = _max_convolution_fft(a)
    return max_b

# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families used in the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a

# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short Adam restarts starting from jittered versions of *init*."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

# Aggressive deterministic mass‑transfer
# ----------------------------------------------------------------------


def _aggressive_mass_transfer(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    For each peak we compute the *largest* safe amount of mass that can be moved
    from a donor to a receiver without creating a new peak larger than the
    former second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    # parameters controlling how many candidates are examined
    donor_keep = 8
    receiver_keep = 8

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices where the peak occurs
        idx_max = np.where(conv >= max_val - 1e-12)[0]
        # second largest (ignore current peaks)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            # peak already flat enough
            break

        k = idx_max[0]  # pick the first maximal index

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        # select a few best donors
        donor_order = np.argsort(-donor_scores)[:donor_keep]
        donor_candidates = idx[donor_order]

        # choose a few receivers with smallest partner values (but not the same index)
        receiver_order = np.argsort(partner)[:receiver_keep + len(donor_candidates)]
        receiver_candidates = idx[receiver_order]

        # try every donor/receiver pair and keep the move that yields the largest safe delta
        best_delta = 0.0
        best_pair = None

        # pre‑allocate temporary buffers for shifted arrays to avoid repeated allocation
        size_conv = 2 * n - 1
        for di in donor_candidates:
            # shifted view for donor i
            shift_i = np.zeros(size_conv, dtype=a.dtype)
            shift_i[di:di + n] = a

            for rj in receiver_candidates:
                if di == rj:
                    continue
                # shifted view for receiver j
                shift_j = np.zeros(size_conv, dtype=a.dtype)
                shift_j[rj:rj + n] = a

                partner_i = shift_i[k]
                partner_j = shift_j[k]
                denom = 2.0 * (partner_i - partner_j)
                if denom <= 0.0:
                    continue

                delta_needed = (max_val - second_max) / denom

                # safety check: no other entry may exceed second_max after the move
                inc = shift_j - shift_i               # positive => will increase that conv entry
                mask_inc = inc > 0
                if np.any(mask_inc):
                    # delta must satisfy conv + 2*delta*inc <= second_max
                    # => delta <= (second_max - conv) / (2*inc)
                    delta_limits = (second_max - conv[mask_inc]) / (2.0 * inc[mask_inc])
                    # keep only positive limits (otherwise the move would already violate the bound)
                    pos_limits = delta_limits[delta_limits > 0]
                    if pos_limits.size == 0:
                        continue
                    delta_limit = np.min(pos_limits)
                else:
                    delta_limit = np.inf

                delta = min(delta_needed, a[di], delta_limit)
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (di, rj)

        # fallback to the original (more conservative) rule if no pair succeeded
        if best_pair is None or best_delta <= 0.0:
            # original simple rule (donor with max partner, receiver with min partner)
            donor_local = np.argmax(donor_scores)
            donor = idx[donor_local]
            partner_donor = partner[donor_local]

            # receiver: smallest partner that is not the donor
            sorted_loc = np.argsort(partner)
            receiver = None
            for r_loc in sorted_loc:
                if idx[r_loc] != donor:
                    receiver = idx[r_loc]
                    partner_receiver = partner[r_loc]
                    break
            if receiver is None:
                break
            denom = 2.0 * (partner_donor - partner_receiver)
            if denom <= 0.0:
                break
            delta_needed = (max_val - second_max) / denom
            delta = min(a[donor] * 0.7, delta_needed * 0.99)
            if delta <= 0.0:
                break
            best_pair = (donor, receiver)
            best_delta = delta

        donor, receiver = best_pair
        a[donor] -= best_delta
        a[receiver] += best_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Aggressive MT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score

            val = _score(seq)

# Aggressive deterministic mass‑transfer
# ----------------------------------------------------------------------


def _aggressive_mass_transfer(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    For each peak we compute the largest *safe* amount of mass that can be moved
    from a donor to a receiver without creating a new peak larger than the
    former second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    donor_keep = 8
    receiver_keep = 8
    size_conv = 2 * n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                     # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_order = np.argsort(-donor_scores)[:donor_keep]
        donor_candidates = idx[donor_order]

        receiver_order = np.argsort(partner)[:receiver_keep + len(donor_candidates)]
        receiver_candidates = idx[receiver_order]

        best_delta = 0.0
        best_pair = None

        for di in donor_candidates:
            shift_i = np.zeros(size_conv, dtype=a.dtype)
            shift_i[di:di + n] = a

            for rj in receiver_candidates:
                if di == rj:
                    continue
                shift_j = np.zeros(size_conv, dtype=a.dtype)
                shift_j[rj:rj + n] = a

                partner_i = shift_i[k]
                partner_j = shift_j[k]
                denom = 2.0 * (partner_i - partner_j)
                if denom <= 0.0:
                    continue

                delta_needed = (max_val - second_max) / denom

                inc = shift_j - shift_i                     # positive => increase at that index
                mask_inc = inc > 0
                if np.any(mask_inc):
                    delta_limits = (second_max - conv[mask_inc]) / (2.0 * inc[mask_inc])
                    pos_limits = delta_limits[delta_limits > 0]
                    if pos_limits.size == 0:
                        continue
                    delta_limit = np.min(pos_limits)
                else:
                    delta_limit = np.inf

                delta = min(delta_needed, a[di], delta_limit)
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (di, rj)

        # fallback to the original conservative rule if nothing succeeded
        if best_pair is None or best_delta <= 0.0:
            donor_local = np.argmax(donor_scores)
            donor = idx[donor_local]
            partner_donor = partner[donor_local]

            sorted_loc = np.argsort(partner)
            receiver = None
            for r_loc in sorted_loc:
                if idx[r_loc] != donor:
                    receiver = idx[r_loc]
                    partner_receiver = partner[r_loc]
                    break
            if receiver is None:
                break
            denom = 2.0 * (partner_donor - partner_receiver)
            if denom <= 0.0:
                break
            delta_needed = (max_val - second_max) / denom
            delta = min(a[donor] * 0.7, delta_needed * 0.99)
            if delta <= 0.0:
                break
            best_pair = (donor, receiver)
            best_delta = delta

        donor, receiver = best_pair
        a[donor] -= best_delta
        a[receiver] += best_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Aggressive MT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal max‑convolution.
    The routine respects the given time budget and returns the best
    sequence found (as a plain Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – larger sizes give more flexibility but are slower
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scale‑invariant)

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.30, 0.85, 31)
    q_grid = np.linspace(0.0, 6.0, 31)

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) incorporate a previously found sequence if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_len = prev.shape[0]
            old_x = np.linspace(0.0, 1.0, old_len)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (d) jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Aggressive deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _aggressive_mass_transfer(best_seq,
                                                    deadline,
                                                    target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Aggressive MT improved best = {best_score:.6f}")

        # occasional global random moves to shake out of plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Final polishing with a higher β for the soft‑max
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
