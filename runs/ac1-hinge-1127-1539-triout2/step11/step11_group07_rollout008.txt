    # Try seeding from previous if available
    if 'height_sequence_1' in globals():
        try:
            seq = np.asarray(globals()['height_sequence_1'], dtype=float)

    schedule = [2000, 4000, 8000]

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

The algorithm follows three stages:
  1. Randomised warm‑starts using several families of distributions.
  2. Two Adam based optimisation phases:
        – smooth max (log‑sum‑exp) for quick exploration,
        – exact max (sub‑gradient) for aggressive descent.
  3. A deterministic hill‑climb that moves mass from the most
     responsible indices to the least responsible ones.

The search is performed on a schedule of increasing lengths
[2000, 4000, 8000]; each size receives a fraction of the remaining
wall‑clock budget.  All iterates are projected onto the ℓ¹‑simplex
∑a = √(2 n) after every update, guaranteeing that the final score is
the true convolution maximum (the value returned by ``evaluate_sequence``).

The code respects the 1000 s limit, prints progress and finally returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    This is just a scaling after clamping to non‑negative values.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)

# Random initialisers (several simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        split = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(split)),
                            np.exp(-lam2 * np.arange(split, n))])

    elif family == 5:                   # linear ramp (increase)
        a = idx + 1.0

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution of a_rev with p (central n entries)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        # occasional jitter to escape plateaus
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        # increase smoothness parameter
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of max(a*a).
    """
    a = a0.copy()
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.2 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1

        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a

# Deterministic hill‑climb on the exact convolution
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly moves a small chunk of mass away from
    the index that contributes most to the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))                # index of current worst entry

        # indices that influence entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose destination j with the smallest counterpart a[M‑j]
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]
        j = int(np.argmin(counterpart[i_low:i_high + 1]) + i_low)

        if j == i:
            break

        delta = a[i] * 0.6                     # move 60 % of the source mass
        if delta <= 0.0:
            break

        # fast linear‑time update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller step (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # occasional jitter to get unstuck
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1)  Seed from a previously found solution if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            target0 = math.sqrt(2.0 * prev.size)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={prev.size}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Length schedule – gradually increase the dimension.
    # ------------------------------------------------------------------
    schedule = [2000, 4000, 8000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # initialise (upsample previous best if possible)
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex_fast(a_init, target)

        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        # --------------------------------------------------------------
        # Allocate time slices for this length
        # --------------------------------------------------------------
        # 20 % for quick smooth‑max explorations,
        # 60 % for exact‑max Adam,
        # 20 % for deterministic hill‑climb.
        t_phase1 = remaining * 0.20
        t_phase2 = remaining * 0.60
        t_phase3 = remaining * 0.20

        # --------------------------------------------------------------
        # Phase 1 – several short smooth‑max Adam runs
        # --------------------------------------------------------------
        local_best = a_init.copy()
        local_best_score = _max_convolution_fft(local_best)

        # split phase 1 into a number of short restarts
        n_restarts = max(1, int(t_phase1 // 5.0))   # ≈5 s per restart
        time_per_restart = t_phase1 / n_restarts

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            start_vec = local_best.copy() if r == 0 else _random_initial_sequence(n, rng)
            sub_dead = min(deadline - 0.1, time.time() + time_per_restart)
            a_mid = _adam_optimize(start_vec, target, sub_dead)
            sc = _max_convolution_fft(a_mid)
            if sc < local_best_score:
                local_best_score = sc
                local_best = a_mid.copy()
                print(f"[{time.time():.1f}] Phase 1 restart {r} n={n} score={sc:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – exact‑max Adam (sub‑gradient)
        # --------------------------------------------------------------
        sub_dead = min(deadline - 0.1, time.time() + t_phase2)
        a_opt = _adam_optimize_submax(local_best, target, sub_dead)
        sc = _max_convolution_fft(a_opt)
        if sc < local_best_score:
            local_best_score = sc
            local_best = a_opt.copy()
            print(f"[{time.time():.1f}] Phase 2 n={n} score={sc:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – deterministic hill‑climb
        # --------------------------------------------------------------
        if time.time() < deadline - 1.0:
            b_cur = _full_convolution(local_best)
            hill_dead = min(deadline - 0.1, time.time() + t_phase3)
            a_hc, hc_score = _hill_climb_deterministic(local_best, b_cur, hill_dead)
            if hc_score < local_best_score:
                local_best_score = hc_score
                local_best = a_hc.copy()
                print(f"[{time.time():.1f}] Phase 3 hill‑climb n={n} score={hc_score:.6f}")

        # --------------------------------------------------------------
        # Update the global best
        # --------------------------------------------------------------
        if local_best_score < best_score - 1e-12:
            best_score = local_best_score
            best_seq = local_best.copy()
            print(f"[{time.time():.1f}] New global best after n={n} "

        # --------------------------------------------------------------
        # Light random tweaks while time permits
        # --------------------------------------------------------------
        while time.time() + 1.0 < deadline:

            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            sc = _max_convolution_fft(cand)
            if sc < best_score - 1e-12:
                best_seq = cand
                best_score = sc
                print(f"[{time.time():.1f}] Random tweak improvement n={n} "

    # 3) Final polishing (extra sub‑max Adam + hill‑climb) if time permits
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:

        target = math.sqrt(2.0 * n)
        extra_dead = deadline - 1.0
        best_seq = _adam_optimize_submax(best_seq, target, extra_dead)
        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)
        if final_score < best_score - 1e-12:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
