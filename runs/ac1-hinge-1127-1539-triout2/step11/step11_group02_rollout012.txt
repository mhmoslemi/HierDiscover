"""
Optimiser for the step‑function benchmark.

Main ideas
----------

1. **Heavier parametric pre‑search** – a denser grid + random sampling over
   four families:
        a_i ∝ (i+β)^(‑p)                           (shifted power law)
        a_i ∝ (i+β)^(‑p)·exp(‑q·i/n)               (shifted‑exp power law)
        a_i ∝ (i+1)^(‑p)·exp(‑q·i/n)               (plain exp‑power law)
        two‑segment power law (broken power law).

   The best shape from this phase is taken as the start point for the
   subsequent optimisation.

2. **Long Adam‑style projected gradient descent** on the soft‑max
   approximation of the max‑convolution.  The temperature β is increased
   slowly (×1.3 every 200 iterations) so that the gradient stays
   informative for a long time.  Several restarts with Gaussian jitter
   around the current best are performed.

3. **Aggressive deterministic mass transfer** – the same idea as the
   reference code, but run until the deadline.  It repeatedly moves as
   much mass as possible from the “donor’’ (largest contributor to the
   current peak) to the “receiver’’ (smallest contributor).

4. **Fine‑grained random‑move hill‑climb** – a cheap local search that
   repeatedly transfers a tiny amount of mass from a large entry to a
   small entry; the move is kept only if the max‑convolution decreases.
   This is useful for breaking the last few percent of the bound.

All phases respect the simplex constraint Σa = √(2 n) and the
non‑negativity constraint.  The routine monitors the wall‑clock time
and stops before the given budget, returning the best sequence seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities for linear convolution (full length 2·n‑1)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (full linear convolution)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray, target_sum: float) -> float:
    """
    Benchmark score under the fixed normalisation Σa = target_sum.
    Because the score is scale‑invariant, this reduces to the maximal
    convolution entry.
    """
    max_b = _max_convolution(a)
    # 2·n*max_b / (target_sum**2) = max_b because target_sum = sqrt(2·n)
    return max_b


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto the (non‑negative) simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)·exp(‑q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _shifted_power_law_sequence(n: int, p: float, shift: float,
                                target_sum: float) -> np.ndarray:
    """a_i ∝ (i+β)^(‑p)  with β=shift (β≥0)."""
    idx = np.arange(n, dtype=np.float64) + shift + 1.0
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _shifted_exp_power_law_sequence(n: int, p: float, q: float, shift: float,
                                    target_sum: float) -> np.ndarray:
    """a_i ∝ (i+β)^(‑p)·exp(‑q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(n: int, p1: float, p2: float, alpha: float,
                               target_sum: float) -> np.ndarray:
    """
    Two‑segment power law:
        first ⌊α·n⌋ entries use exponent p1,
        remaining entries use exponent p2.
    """
    K = max(1, int(alpha * n))
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in Adam GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, conv_full) where conv_full is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    # avoid overflow – shift by max_b before exponentiation
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()          # soft‑max weights over convolution entries

    # convolution of reversed a with p gives the gradient (see derivation)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # gradient is the central window of length n (each entry appears twice)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Adam‑style projected GD on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  beta_start: float = 0.2,
                  beta_mult: float = 1.30,
                  beta_inc_iter: int = 200,
                  max_beta: float = 1e6,
                  step_factor: float = 0.7,
                  gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth surrogate of max‑convolution.
    Returns the best point encountered and its (true) score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a, target_sum)

    beta = beta_start
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a, target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.2f} "
                  f"score={best_val:.6f}")

        # anneal temperature
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * beta_mult, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Repeated greedy transfer of mass from the biggest contributor to the
    smallest contributor of the current peak.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a, target_sum)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        # pick the first maximal index (any works)
        k = int(idx_max[0])

        # indices that participate in conv[k]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions a_{k-i}
        contrib = a[k - idx]

        donor = int(idx[np.argmax(contrib)])
        receiver = int(idx[np.argmin(contrib)])

        # second biggest value (excluding current maximal entries)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        # denominator in the formula from the paper
        denom = 2.0 * (a[donor] - a[receiver])
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        # move as much as possible but not beyond donor's mass
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a, target_sum)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT move score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random‑move hill climb (post‑processing)
# ----------------------------------------------------------------------


def _local_random_hill_climb(a0: np.ndarray,
                             deadline: float,
                             target_sum: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: transfer a tiny fraction of mass from a large
    entry to a small entry and keep the move only if the peak decreases.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a, target_sum)

    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick donor among the top 5 % and receiver among the bottom 5 %
        top_k = max(1, int(0.05 * n))
        low_k = max(1, int(0.05 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue

        # try a modest delta (up to 3 % of donor mass)
        delta = rng.random() * a[donor] * 0.03
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        # keep the sum correct – we moved mass, so no projection needed
        cur_score = _score(a, target_sum)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Local climb new best={best_score:.6f}")

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step‑function with a minimal benchmark score.
    The routine respects the time budget and returns the best sequence found.

    # ---------------------------------------------------------------

        n = 15000
    elif budget_s >= 600:
        n = 12000

        n = 9000
    else:

    # Phase 0 – intensive parametric pre‑search
    # ---------------------------------------------------------------

    best_score = float("inf")
    rng = np.random.default_rng()

    def _try_candidate(seq: np.ndarray):

            return

        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Pre‑search improved score={best_score:.6f}")

    # (a) grid over plain power‑law
    for p in np.arange(0.20, 1.31, 0.01):
        seq = _power_law_sequence(n, float(p), target_sum)
        _try_candidate(seq)

        for q in np.arange(0.0, 10.001, 0.25):

    # (c) shifted power‑law & shifted‑exp (coarse grid)
