        hi = min(k_max, n-1)
        idx = np.arange(lo, hi+1)
        prod = a[idx] * a[k_max - idx]

"""
Hybrid optimizer for the non‑negative step‑function benchmark.

The algorithm proceeds in three stages:

1. **Warm‑start** – generate a few power‑law sequences
   a_i = (i+1)^(-p) for different exponents p,
   keep the best (lowest max‑convolution) after normalisation.

2. **Targeted greedy descent** – repeatedly identify the
   convolution entry with the current maximum, locate the pair
   (i, k‑i) that contributes most to this entry and move a tiny
   amount of mass from that index to a random other index.
   The convolution is updated in O(n) time using the exact formula,
   and Metropolis acceptance with a linear cooling schedule is used.

3. **Simulated annealing** – a classic SA phase that also updates
   the convolution in O(n) time, allowing many cheap moves.
   The temperature schedule is adapted to the current excess over
   the theoretical lower bound (≈1).

All phases respect the supplied time budget (default 1000 s) and
return the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  For the benchmark sum = √(2n) → score = max."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to new_len entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Warm‑start: quick sweep over power‑law exponents
# ----------------------------------------------------------------------


def _best_powerlaw(n: int, target_sum: float, deadline: float) -> np.ndarray:
    """Return the best of a few power‑law initialisations."""
    best_seq = None
    best_val = np.inf
    # a handful of exponents covering a reasonable range
    exponents = np.linspace(0.35, 1.20, 15)
    for p in exponents:
        if time.time() >= deadline:
            break
        seq = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    # fallback (should never happen)
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) * (target_sum / n)
    print(f"[{time.time():.2f}] Warm‑start best power‑law score={best_val:.6f}")
    return best_seq


# ----------------------------------------------------------------------
# Greedy max‑reduction (targeted mass transfer)
# ----------------------------------------------------------------------


def _greedy_max_reduction(a0: np.ndarray,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly reduces the current maximum convolution entry by moving a
    small amount of mass away from the pair that contributes most to it.
    An annealed Metropolis acceptance rule is used.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                 # full convolution, length 2n‑1
    cur_max = b.max()
    best_a = a.copy()
    best_val = cur_max

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature proportional to the excess over the trivial lower bound 1
    T0 = max(0.02, (best_val - 1.0) * 0.5)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product of the two factors contributing to each term of the max entry
        prod = a[idx] * a[k_max - idx]

        # donor = index that contributes most to the max entry
        donor = idx[np.argmax(prod)]

        # choose a receiver uniformly (reject donor)
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        # magnitude of the move (small, keeping us in a smooth regime)
        max_delta = min(a[donor] * 0.03, target_sum * 0.001)
        if max_delta <= 0:
            # nothing to move – break to avoid infinite loop
            break
        delta = rng.random() * max_delta

        # ----- exact O(n) update of the convolution -----
        b_new = b.copy()
        # contributions of donor and receiver in all positions
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # the squares on the diagonal
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        # the cross term (appears only once because donor != receiver)
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_max - cur_max) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] Greedy it={it:6d} improved score={best_val:.6f}")

        # occasional global jitter to escape plateaus
        if (it % 4000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0003, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = b_j.max()
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_max = val_j
                print(f"[{time.time():.2f}] Greedy jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val

# Simulated annealing (cheap O(n) updates)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Move tiny mass between two indices, updating the convolution in O(n) time."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = b.max()
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    T0 = max(0.03, (best_val - 1.0) * 0.5)   # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # pick donor proportionally to its mass (helps avoid dead donors)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue
        # distinct receiver, uniform
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.05, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) convolution update
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = new_max

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} improved score={best_val:.6f}")

        # occasional global jitter
        if (it % 5000) == 0:
            jitter = rng.normal(scale=target_sum * 0.0002, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = b_j.max()
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val

        greedy_deadline = time.time() + 0.35 * (deadline - time.time())
        best_seq, best_score = _greedy_max_reduction(best_seq, target_sum, greedy_deadline)
        print(f"[{time.time():.2f}] After greedy descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing with cheap O(n) updates
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(best_seq, target_sum, deadline)
        print(f"[{time.time():.2f}] After simulated annealing best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # final external evaluation (the verifier)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starts in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
