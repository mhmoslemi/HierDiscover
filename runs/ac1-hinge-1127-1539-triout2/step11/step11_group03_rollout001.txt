"""
Non‑negative step‑function optimizer.

The search proceeds in three stages:

1. **Exploration** – a pool of parametric families (power law, exponential,
   Gaussian, sine, random Dirichlet) is generated.  Short projected‑gradient
   runs on the smooth soft‑max approximation of the max‑convolution objective
   quickly produce a diverse set of promising shapes.

2. **Polishing** – the best shape is up‑scaled to a larger length and refined
   by a dedicated sub‑gradient method that directly reduces the peak of the
   self‑convolution (gradient of the ℓ∞‑norm is 2·a_{k‑i} for the current
   maximal index *k*).  An optional permutation‑search further lowers the peak
   by re‑ordering the multiset of heights.

3. **Finalisation** – trailing zeros are removed, the sequence is stored in a
   global variable for possible reuse, and the list is returned.

All operations respect the simplex constraint Σa_i = √(2 n) (scale‑invariant)
and the non‑negativity constraint.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric sequence generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (renormalised to target_sum)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam*i) (renormalised)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the interval."""
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _sine_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+1)/(n+1))  (renormalised)."""
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1)))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest ↦ left, second‑largest ↦ right, third‑largest ↦ next left, …."""
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Same as alternating, but start with the largest on the right."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p   (⋆ = convolution)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic (sub‑gradient) refinement of the ℓ∞‑convolution norm
# ----------------------------------------------------------------------


def _subgradient_refine(a0: np.ndarray,
                        deadline: float,
                        target_sum: float,
                        initial_step: float = None) -> tuple[np.ndarray, float]:
    """
    Direct sub‑gradient descent on max_b = max(a*a).

    At each iteration we locate the current maximal convolution index k,
    compute the gradient d max_b / d a_i = 2·a_{k‑i} (where defined) and
    take a small projected step.  A simple exponential step‑size decay and
    occasional random uphill moves help to escape shallow plateaus.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]

    if initial_step is None:
        initial_step = target_sum * 0.2

    step = initial_step
    best_a = a.copy()
    cur_conv = _conv_fft(a)
    best_score = float(2 * n * cur_conv.max() / (target_sum ** 2))
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # current maximal index
        kmax = int(cur_conv.argmax())
        max_b = cur_conv.max()
        # gradient (sub‑gradient of the ℓ∞‑norm)
        grad = np.zeros_like(a)
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        grad[mask] = 2.0 * a[partner[mask]]

        # projected step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        new_conv = _conv_fft(a_new)
        new_max = new_conv.max()
        new_score = float(2 * n * new_max / (target_sum ** 2))

        # accept if improvement, otherwise probabilistic uphill move
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} new best={best_score:.6f}")

        if new_score <= (2 * n * max_b / (target_sum ** 2)) or rng.random() < 0.02:
            a = a_new
            cur_conv = new_conv
        # else keep previous a, cur_conv unchanged

        # step‑size schedule
        it += 1
        if it % 500 == 0:
            step *= 0.85
        if step < target_sum * 1e-5:
            step = target_sum * 1e-5

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing on index order)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style search that swaps indices which contribute heavily
    to the current maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(conv.argmax())

        # contribution of each position to the peak
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        # swap
        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    2. Run many short soft‑max gradient descents (exploration).
    3. Upscale the current best to a larger length if time permits.
    4. High‑β gradient polishing.
    5. Direct sub‑gradient refinement on the true ℓ∞‑norm.
    6. Targeted permutation search.
    7. Final trimming and return.
    """

        init_pool.append(a)

    # linear decreasing shape (for diversity)
    lin = np.linspace(1.0, 0.0, n_base)

        seq_j = _project_to_simplex(seq_j, target_base)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 1. Exploration: many short projected‑gradient runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 12.0:   # keep a reserve for polishing
        # pick a seed (occasionally perturb the current best)
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_base * 0.004, size=n_base)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_base)
        else:
            seed = rng.choice(init_pool)

        # each run gets at most ~20 s (adjusted for remaining budget)
        run_dead = min(time.time() + 20.0, deadline - 8.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_base,
            max_beta=5e5,
            step_factor=0.7,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Upscaling to a larger problem size (if time permits)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_base, 0.55, target_base)

    remaining = deadline - time.time()
    if remaining > 30.0 and budget_s >= 700:
        n_final = 12000
    elif remaining > 15.0:
        n_final = 10000
    else:
        n_final = n_base

    target_final = np.sqrt(2.0 * n_final)

    if n_final != n_base:
        seq_big = _resize_sequence(best_seq, n_final)
        seq_big = _project_to_simplex(seq_big, target_final)
    else:
        seq_big = best_seq.copy()
    best_seq = seq_big
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. High‑β gradient polishing (soft‑max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            deadline - 4.0,
            target_final,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final gradient polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Direct sub‑gradient refinement on the true ℓ∞ objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_dead = min(deadline - 1.0, time.time() + 12.0)
        a_sub, val_sub = _subgradient_refine(best_seq, sub_dead, target_final)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Targeted permutation search (re‑ordering only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = time.time() + min(12.0, deadline - time.time() - 1.0)
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.12,
            end_temp=1e-5,
            top_frac=0.03,
            low_frac=0.03,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 7. Log final evaluation (optional, harmless if evaluate_sequence is unavailable)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose result for possible reuse in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
