```
def gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n - 1]
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()
    # rev(a) convolve with p
    a_rev = a[::-1]
    # compute convolution rev(a) * p via FFT
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[:2*n - 1]
    grad = 2.0 * conv_rev[n-1:n-1 + n]
    return grad, b
```

The gradient is shape n.

def mass_transfer(a0, target_sum, deadline, start_temp=0.6, end_temp=1e-5, low_frac=0.04):
    a = a0.copy()
    n = a.shape[0]
    cur_score = max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    # Comput initial conv
    conv = conv_fft(a)
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau
        # find kmax
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contributions w: a[idx] + a[kmax-idx]
        w = a[idx] + a[kmax - idx]
        # donor (largest contribution)
        donor = idx[np.argmax(w)]
        # low frac for receiver candidates
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)
        if donor == receiver:
            it += 1
            continue
        # determine max delta that doesn't create negative donor and not too large
        max_delta = min(a[donor] * 0.5, (w[np.argmax(w)] - w[np.argmin(w)]) * 0.25)
        if max_delta <= 0:
            it += 1
            continue
        delta = rng.random() * max_delta
        a[donor] -= delta
        a[receiver] += delta
        # project to simplex to correct any slight sum drift
        a = project_to_simplex(a, target_sum)
        conv = conv_fft(a)
        new_score = max_convolution_fft(a)
        # accept if better or with probability exp(-Δ/temp)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = project_to_simplex(a, target_sum)
            conv = conv_fft(a)  # revert conv
        # occasional jitter to explore
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = project_to_simplex(a, target_sum)
            conv = conv_fft(a)
            cur_score = max_convolution_fft(a)
        it += 1
    return best_a, best_score

def gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = b.max()
    # compute softmax probabilities p over conv entries
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()
    # gradient: 2 * reverse(a) convolve p, central slice
    a_rev = a[::-1]
    # compute convolution via FFT
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1 + n]
    return grad, b
```

def adam_optimize(a0, target_sum, beta_start, beta_end, deadline):
    n = a0.shape[0]
    # initialize logits from a0
    eps = 1e-12
    w0 = np.maximum(a0 / target_sum, eps)
    z = np.log(w0)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0
    beta = beta_start
    best_a = a0.copy()
    best_val = max_convolution_fft(a0)
    # set evaluation interval
    eval_interval = 200
    # optional max iterations
    while time.time() < deadline:
        # compute current a
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        w = exp_z / exp_z.sum()
        a = target_sum * w
        # compute gradient approx
        grad_a, _ = gradient_softmax(a, beta)
        # grad_z
        grad_z = target_sum * w * (grad_a - np.dot(w, grad_a))
        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = 0.1 / (np.sqrt(t) + 1.0)  # decaying trainable
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z -= step
        # increase beta
        if t % 500 == 0 and beta < beta_end:
            beta = min(beta * 2.0, beta_end)
        # evaluate occasionally
        if t % eval_interval == 0:
            cur_val = max_convolution_fft(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:.1f} best={best_val:.6f}")
    return best_a, best_val

"""
Optimisation of a non‑negative step function for the max‑self‑convolution
objective.

Approach
--------
1. Build a diverse pool of candidate shapes (power‑law, exponential,
   sinusoidal‑power, Gaussian, random Dirichlet) and project each onto the
   simplex Σa = √(2 n).
2. Keep the best few candidates (by the exact max‑convolution) as start points.
3. Refine every start point with an ADAM optimiser on a soft‑max
   approximation of the max‑convolution.  The parameters a are represented as
   a = S·softmax(z) where S = √(2 n); optimisation is performed on the logits
   `z`.  The soft‑max temperature β is annealed from 0.5 up to 5·10⁵.
4. Run a deterministic mass‑transfer annealing phase that moves mass from
   positions that contribute most to the current peak toward positions that
   contribute least.  This directly reduces the true max‑convolution.
5. Finish with a short high‑β ADAM polishing.
6. Trim trailing zeros, store the result globally for warm‑starts and return it.

All heavy work uses FFT‑based convolutions (O(n log n)).  The algorithm is
self‑contained, respects the 1000 s budget, and does not use any nested
function closures or external I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max gradient of max‑convolution
# ----------------------------------------------------------------------
def gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, conv) where conv is the exact convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)

    # exact convolution
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv.max()
    exps = np.exp(beta * (conv - max_b))
    p = exps / exps.sum()               # soft‑max probabilities

    # gradient = 2 * rev(a) ⊛ p, restricted to the central n entries
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, conv


# ----------------------------------------------------------------------
# ADAM optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray,
                  target_sum: float,
                  beta_start: float,
                  beta_end: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    ADAM optimiser on the soft‑max approximation of max‑convolution.
    The iterates stay on the simplex Σa = target_sum.
    """
    n = a0.shape[0]
    eps = 1e-12

    # initialise logits from a0 (avoid log(0))
    w0 = np.maximum(a0 / target_sum, eps)
    z = np.log(w0)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    t = 0
    beta = beta_start
    best_a = a0.copy()
    best_val = max_convolution_fft(a0)

    eval_interval = 200          # exact evaluation frequency
    while time.time() < deadline:
        # ----- soft‑max reconstruction -----
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        w = exp_z / exp_z.sum()
        a = target_sum * w

        # ----- soft‑max gradient -----
        grad_a, _ = gradient_softmax(a, beta)

        # ----- gradient wrt logits -----
        grad_z = target_sum * w * (grad_a - np.dot(w, grad_a))

        # ----- ADAM update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # decaying learning rate (simple schedule)
        lr = 0.12 / (np.sqrt(t) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z -= step

        # ----- temperature schedule -----
        if t % 500 == 0 and beta < beta_end:
            beta = min(beta * 2.0, beta_end)

        # ----- occasional exact evaluation -----
        if t % eval_interval == 0:
            cur_val = max_convolution_fft(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] ADAM it={t:5d} β={beta:6.1f} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer annealing
# ----------------------------------------------------------------------
def mass_transfer(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  start_temp: float = 0.6,
                  end_temp: float = 1e-5,
                  low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer that reduces the current convolution peak."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # position of the current maximum
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, kmax-i)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # admissible transfer size
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = project_to_simplex(a, target_sum)

        conv = conv_fft(a)
        new_score = max_convolution_fft(a)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = project_to_simplex(a, target_sum)
            conv = conv_fft(a)   # recompute after revert

        # occasional jitter to escape stagnation
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = project_to_simplex(a, target_sum)
            conv = conv_fft(a)
            cur_score = max_convolution_fft(a)

        it += 1

    return best_a, best_score

def resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def generate_candidates(n: int, target_sum: float) -> list[np.ndarray]:
    """Create a diverse set of initial sequences (projected onto the simplex)."""
    rng = np.random.default_rng()
    cand = []

    # warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            prev = resize_sequence(prev, n)
        prev = project_to_simplex(prev, target_sum)
        cand.append(prev)

    # power‑law families
    for p in np.linspace(0.30, 0.90, 31):
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        seq = project_to_simplex(seq, target_sum)
        cand.append(seq)

        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seqj = np.maximum(seq + jitter, 0.0)
        seqj = project_to_simplex(seqj, target_sum)
        cand.append(seqj)

    # exponential decay
    for lam in np.linspace(5e-4, 0.015, 25):
        seq = np.exp(-lam * np.arange(n, dtype=np.float64))
        seq = project_to_simplex(seq, target_sum)
        cand.append(seq)

        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seqj = np.maximum(seq + jitter, 0.0)
        seqj = project_to_simplex(seqj, target_sum)
        cand.append(seqj)

    # sinusoidal power shapes
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    sin_vals = np.sin(theta)
    for p in np.linspace(0.2, 2.0, 15):
        seq = sin_vals ** p
        seq = project_to_simplex(seq, target_sum)
        cand.append(seq)

        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seqj = np.maximum(seq + jitter, 0.0)
        seqj = project_to_simplex(seqj, target_sum)
        cand.append(seqj)

    # Gaussian‑bell shapes
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 4.0, 10):
        seq = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        seq = project_to_simplex(seq, target_sum)
        cand.append(seq)

    # pure Dirichlet (exponential) draws
    for _ in range(10):
        seq = rng.exponential(scale=1.0, size=n)
        seq = project_to_simplex(seq, target_sum)
        cand.append(seq)

    return cand
