"""
step‑function optimizer – “sub‑gradient + greedy mass transfer”

The benchmark score for a non‑negative vector a with Σa = √(2 n) is simply
the maximum entry of the (linear) self‑convolution a*a.
We therefore minimise   max_k (a*a)[k].

Algorithm
---------

1. **Problem size** – Choose n according to the time budget (larger n gives a
   finer discretisation, but costs a little more per evaluation).

2. **Dense parametric pre‑search** – Scan several low‑dimensional families
   (power‑law, power‑law + exponential tail, broken power‑law, sine‑window,
   Hann‑window, Kaiser window).  For each family we sweep a few parameters,
   scale to the required simplex Σa = √(2 n) and keep the best candidate.

3. **Projected sub‑gradient descent** – Repeatedly
      * compute the full convolution,
      * locate the index k of the current maximum,
      * form the exact sub‑gradient of that entry (g_i = 2·a_{k‑i}),
      * move a ← a – η g, where η = (max−second_max) / ‖g‖² (scaled by 0.8
        for safety) and project back onto the simplex.
   This directly reduces the peak while preserving non‑negativity and the sum.

4. **Aggressive deterministic mass‑transfer** – For the same peak k we move
   mass from the index that contributes most to that entry to the index that
   contributes least, but only a fraction (default 0.5) of the amount that
   would bring the peak down to the second‑largest value.  The operation is
   repeated until the time budget expires.

5. **Light stochastic refinement** – Randomly relocate a small fraction of
   mass from a large entry to a small entry; accept the move only if the
   score improves.  This helps to escape tiny plateaus.

6. **Final polishing** – One short run of the Adam‑style optimiser from
   the current best point (very large soft‑max β) to squeeze out any remaining
   slack.

All phases respect the simplex constraint and stop automatically when the
allocation budget (≈1000 s) is exhausted.  The best sequence encountered is
returned (trailing near‑zero entries are removed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (full linear convolution)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Benchmark score.
    When Σa = √(2 n) this equals max(a*a).  The function therefore returns the
    current objective value.
    """
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _sine_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+1)/(n+1))^p   (window that is zero at the ends)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1)) ** p
    a *= target_sum / a.sum()
    return a


def _hann_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (0.5 - 0.5·cos(2π·i/(n-1)))^p   (Hann window)."""
    i = np.arange(n, dtype=np.float64)
    a = (0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a *= target_sum / a.sum()
    return a


def _kaiser_sequence(n: int, beta: float, target_sum: float) -> np.ndarray:
    """Kaiser window – requires scipy.special.i0; fall back to flat if unavailable."""
    try:
        from scipy.special import i0
    except Exception:  # pragma: no cover
        # No SciPy – return a simple cosine window as a placeholder
        return _hann_power_sequence(n, 1.0, target_sum)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    # argument of the Bessel function
    arg = beta * np.sqrt(1 - ((i - alpha) / alpha) ** 2)
    a = i0(arg) / i0(beta)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    safety: float = 0.8,
    min_improve: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Perform a simple projected sub‑gradient descent that directly targets the
    currently maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # compute second‑largest value (strictly smaller)
        mask = conv < max_val - 1e-12
        if mask.any():
            second_max = conv[mask].max()
        else:
            # All entries are (almost) equal – we cannot improve further
            break

        # exact sub‑gradient of the active entry
        n = a.shape[0]
        idx = np.arange(n)
        j = max_idx - idx
        valid = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[j[valid]]
        g_norm2 = float(g @ g)

        if g_norm2 == 0.0:
            break

        # step size that would bring max down to second_max (scaled for safety)
        eta = (max_val - second_max) / g_norm2
        eta *= safety

        a -= eta * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val - min_improve:
            best_val, best_a = cur_val, a.copy()
            print(f"[{time.time():.2f}] sub‑grad iteration score={best_val:.6f}")

        # stop condition if the peak cannot be lowered appreciably
        if max_val - second_max < 1e-12:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (fractional version)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    fraction: float = 0.5,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the peak using a fractional move.
    The amount transferred is fraction·delta_needed (default 0.5) to keep
    other convolution entries from overshooting.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]
        k = int(idx_max[0])                     # pick the first maximal index

        # indices i that appear in conv[k] = Σ a_i·a_{k‑i}
        lo = max(0, k - (a.size - 1))
        hi = min(k, a.size - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions a_{k‑i}
        contrib = a[k - idx]

        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        # second largest convolution entry (ignore all max positions)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        # change of the peak when moving ε mass from donor → receiver:
        # Δmax = -2·(a[donor] - a[receiver])·ε
        diff = a[donor] - a[receiver]
        if diff <= 0:
            # nothing to gain from this peak
            break
        denom = 2.0 * diff
        delta_needed = (max_val - second_max) / denom
        delta = min(a[donor], delta_needed * fraction)
        if delta <= 0:
            break

        a[donor] -= delta
        a[receiver] += delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.2f}] DMT (frac) score={best_score:.6f}")

    return best_a, best_score

# Light stochastic refinement (random mass moves)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    move_fraction: float = 0.05,
    top_ratio: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Very light simulated‑annealing style random moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        top_k = max(1, int(top_ratio * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(top_k)]

        if donor == receiver:
            continue

        delta = rng.random() * a[donor] * move_fraction
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.2f}] random move new best={best_score:.6f}")

    return best_a, best_score

        n = 8000

    target_sum = math.sqrt(2.0 * n)               # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # (a) pure power‑law
    for p in np.arange(0.20, 1.21, 0.02):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in np.arange(0.20, 1.21, 0.04):
        for q in np.arange(0.0, 8.001, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (two‑segment)
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) sine‑window family
    for p in np.arange(0.2, 4.01, 0.2):
        seq = _sine_power_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (e) Hann‑window family
    for p in np.arange(0.2, 4.01, 0.2):
        seq = _hann_power_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (f) Kaiser window (if SciPy is present)
    for beta in np.arange(0.0, 12.01, 1.0):
        seq = _kaiser_sequence(n, float(beta), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (g) seed from previous challenger, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – projected sub‑gradient descent (focused on the peak)
    # ---------------------------------------------------------------
    subgrad_deadline = time.time() + 0.25 * (deadline - time.time())
    a_sg, val_sg = _subgradient_descent(best_seq, subgrad_deadline, target_sum)
    if val_sg < best_score:
        best_score, best_seq = val_sg, a_sg.copy()
        print(f"[{time.time():.2f}] Sub‑gradient phase improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – aggressive deterministic mass transfer (fractional)
    # ---------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            deadline,
            target_sum,
            fraction=0.45,                 # a bit more aggressive than 0.5
        )
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best = {best_score:.6f}")

        # short random move phase to wiggle out of tiny plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # ---------------------------------------------------------------
    # Phase C – final polishing with a high‑β Adam run (optional)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        # Re‑use the Adam implementation from the previous solution (soft‑max).
        # We embed a trimmed‑down version here to avoid extra imports.
        def _adam_gradient_descent(a0, dl, tgt, max_beta=1e6, sf=0.6, gamma=0.995):
            a = _project_to_simplex(a0.copy(), tgt)
            best = a.copy()
            bestval = _score(a)
            beta = 0.5
            rng = np.random.default_rng()
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            eps = 1e-8
            it = 0
            inc_iter = 300
            while time.time() < dl:
                # soft‑max gradient (reuse earlier helper)
                n = a.size
                size = 1 << ((2 * n - 1).bit_length())
                fa = np.fft.rfft(a, n=size)
                conv_full = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
                maxb = conv_full.max()
                exps = np.exp(beta * (conv_full - maxb))
                p = exps / exps.sum()
                # grad = 2*conv_rev[n-1:n-1+n]
                a_rev = a[::-1]
                Ar = np.fft.rfft(a_rev, n=size)
                Pr = np.fft.rfft(p, n=size)
                conv_rev = np.fft.irfft(Ar * Pr, n=size)[: 2 * n - 1]
                grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

                # Adam update
                m = gamma * m + (1 - gamma) * grad
                v = gamma * v + (1 - gamma) * (grad * grad)
                m_hat = m / (1 - gamma ** (it + 1))
                v_hat = v / (1 - gamma ** (it + 1))
                step = sf * tgt / (np.sqrt(v_hat) + eps)
                a -= step * m_hat
                a = _project_to_simplex(a, tgt)

                if it % inc_iter == 0 and beta < max_beta:
                    beta = min(beta * 2.0, max_beta)

                if it % 250 == 0:
                    # tiny jitter to escape plateaus
                    jitter = rng.normal(scale=tgt * 0.001 / (beta + 1.0), size=a.shape)
                    a += jitter
                    a = _project_to_simplex(a, tgt)

                cur = _score(a)
                if cur < bestval:
                    bestval, best = cur, a.copy()
                    print(f"[{time.time():.2f}] Adam polish it={it} β={beta:.1f} score={bestval:.6f}")

                it += 1
            return best, bestval

        a_fin, val_fin = _adam_gradient_descent(best_seq, deadline, target_sum)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries, return list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for logging (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
