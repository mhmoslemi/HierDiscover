def _adam_optimize(a0, target_sum, deadline, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_beta=1e5, beta_schedule_step=200, beta_factor=2.0):

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5
    best = a.copy()
    best_val=_score(a)
    while time.time() < deadline:
        t+=1
        grad,_ = _gradient_softmax(a, beta)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        mhat = m/(1-beta1**t)
        vhat = v/(1-beta2**t)
        step = lr/(np.sqrt(vhat) + eps)
        a -= step * mhat
        a = _project_to_simplex(a,target_sum)
        if t%beta_schedule_step==0 and beta<max_beta:
            beta = min(beta*beta_factor, max_beta)
        val = _score(a)
        if val < best_val:
            best_val=val
            best=a.copy()
        # occasional jitter:

            jitter = np.random.default_rng().normal(scale=target_sum*0.001/(beta+1), size=a.shape)
            a+=jitter
            a=_project_to_simplex(a,target_sum)
    return best,best_val

def _shuffle_hill_climb(a, target_sum, deadline):
    best = a.copy()
    best_val = _score(best)
    cur = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = len(a)
    start = time.time()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # swap only if they differ maybe
        if cur[i] == cur[j]:
            continue
        cand = cur.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_val = _score(cand)
        # update best and accept
        # simple hill-climbing: accept only if improves
        if cand_val < cur_val:
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val
        else:
            # occasional acceptance with probability e^{-delta/T}
            # compute temperature schedule
            elapsed = time.time() - start
            T = max(0.01, (deadline - time.time())/ (deadline - start)) # maybe high early low later
            # Actually maybe set T start high and decay linear
            # We approximate using linear schedule: T = 0.5 * (remaining_fraction)
            if rng.random() < np.exp(- (cand_val - cur_val) / (0.1+0.9*T)):
                cur = cand
                cur_val = cand_val
    return best, best_val

"""
Hybrid optimizer with Adam soft‑max pre‑phase, sub‑gradient
refinement on a larger scale, and a stochastic reordering
(Hill‑Climbing) phase.  The final step is a short deterministic
mass‑transfer pass.  All constraints (non‑negativity, ℓ¹‑norm)
are respected by projection onto the simplex.
"""

import time
import numpy as np

# ---------- Convolution utilities ----------
def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# ---------- Simplex projection ----------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ---------- Sequence factories ----------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# ---------- Gradient of soft‑max ----------
def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# ---------- Adam optimizer on the soft‑max surrogate ----------
def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   lr: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_beta: float = 5e5,
                   beta_step: int = 200,
                   beta_factor: float = 2.0) -> tuple[np.ndarray, float]:
    """Projected Adam on the log‑sum‑exp approximation of max."""
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # beta schedule
        if t % beta_step == 0 and beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val

# ---------- Sub‑gradient descent on the exact max ----------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining the maximum (numeric tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005,
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

# ---------- Deterministic mass‑transfer ----------
def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# ---------- Random reordering hill‑climb ----------
def _shuffle_hill_climb(a0: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing that swaps two entries."""
    cur = a0.copy()
    cur_val = _score(cur)
    best = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    n = cur.shape[0]

    start = time.time()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] == cur[j]:
            continue
        cand = cur.copy()
        cand[i], cand[j] = cur[j], cur[i]
        cand_val = _score(cand)
        # basic greedy acceptance; occasional Metropolis acceptance
        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best, best_val = cand.copy(), cand_val
        else:
            # temperature that decays linearly with remaining time
            elapsed = time.time() - start
            frac = 1.0 - (elapsed / (deadline - start))
            temp = 0.05 * max(frac, 0.0) + 1e-6
            if rng.random() < np.exp(-(cand_val - cur_val) / temp):
                cur, cur_val = cand, cand_val
    return best, best_val

"""
Hybrid optimizer for the self‑convolution bound.

The pipeline consists of four stages:

1. Small‑scale Adam optimisation on a smooth (soft‑max) surrogate of the
   max‑convolution.  This quickly finds a good shape for the coefficients.
2. Upscaling to a larger length (12k‑16k) and exact sub‑gradient descent
   on the true max‑convolution.
3. Stochastic reordering (hill‑climbing) that repeatedly swaps two entries;
   the multiset of heights is unchanged, but the ordering can significantly
   lower the convolution peak.
4. A short deterministic mass‑transfer polish.

All constraints (non‑negativity and ℓ¹‑norm Σa = √(2 n)) are enforced by a
projection onto the simplex after every update.
The function returns the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a)/(Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(–p) then normalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(–lam·i) then normalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of  log‑sum‑exp_β( a*a )  w.r.t. a.
    Returns (grad, full convolution) where grad has length len(a).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2·(a reversed)⋆p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   lr: float = 0.6,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_beta: float = 5e5,
                   beta_step: int = 250,
                   beta_factor: float = 2.0) -> tuple[np.ndarray, float]:
    """Projected Adam on the log‑sum‑exp approximation of max(a*a)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)

        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # increase the soft‑max temperature
        if t % beta_step == 0 and beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

        # occasional jitter to avoid stagnation
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:6.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        # sub‑gradient of the max‑operator
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005,
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.3,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift a small amount of mass from a “donor’’ (large
    product contribution to the current maximal convolution entry) to a
    “receiver’’ (small contribution).  Moves are accepted with a Metropolis
    rule whose temperature decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic reordering (hill‑climbing)
# ----------------------------------------------------------------------


def _shuffle_hill_climb(a0: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that repeatedly swaps two positions.
    The ℓ¹‑norm is unchanged, only the ordering varies.
    """
    cur = a0.copy()
    cur_val = _score(cur)
    best = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    n = cur.shape[0]

    start = time.time()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] == cur[j]:
            continue
        cand = cur.copy()
        cand[i], cand[j] = cur[j], cur[i]
        cand_val = _score(cand)

        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best, best_val = cand.copy(), cand_val
        else:
            # linear cooling schedule for the Metropolis acceptance
            elapsed = time.time() - start
            frac = 1.0 - (elapsed / (deadline - start))
            temp = 0.04 * max(frac, 0.0) + 1e-9
            if rng.random() < np.exp(-(cand_val - cur_val) / temp):
                cur, cur_val = cand, cand_val

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm follows the four stages described in the module docstring.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Small‑scale exploration (Adam on the soft‑max surrogate)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 6000
    elif budget_s >= 400:
        n_expl = 5000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()
    init_pool = []

    # power‑law seeds
    for p in np.linspace(0.2, 1.0, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential seeds
    for lam in np.linspace(0.001, 0.02, 6):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random simplex points
    for _ in range(8):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] >= n_expl:
            init_pool.append(prev[:n_expl].copy())
        else:
            init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_val = np.inf

    while time.time() < deadline - 30.0:
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 15.0, deadline - 20.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _adam_optimize(seed, target_expl, run_dead,
                                        lr=0.6, beta1=0.9, beta2=0.999,
                                        max_beta=5e5, beta_step=250)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Upscaling & sub‑gradient refinement on a larger problem
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.5, target_expl)

    remaining = deadline - time.time()
    if remaining > 200.0:
        n_big = 16000
    elif remaining > 100.0:
        n_big = 12000
    else:
        n_big = n_expl
    target_big = np.sqrt(2.0 * n_big)

    seq_big = _resize_sequence(best_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # Sub‑gradient descent (exact max)
    if time.time() < deadline - 30.0:
        sub_dead = deadline - 10.0
        a_sub, val_sub = _subgradient_descent(seq_big, sub_dead,
                                              target_big,
                                              step_factor=0.45,
                                              gamma=0.995)
        if val_sub < best_val:
            best_val = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Stochastic reordering (hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        shuffle_dead = deadline - 2.0
        a_shuf, val_shuf = _shuffle_hill_climb(seq_big, shuffle_dead)
        if val_shuf < best_val:
            best_val = val_shuf
            seq_big = a_shuf.copy()
            print(f"[{time.time():.2f}] Reordering improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final deterministic mass‑transfer polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(seq_big, mt_dead,
                                                    target_big,
                                                    start_temp=0.25,
                                                    end_temp=1e-5,
                                                    low_frac=0.04)
        if val_mt < best_val:
            best_val = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for logging
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
