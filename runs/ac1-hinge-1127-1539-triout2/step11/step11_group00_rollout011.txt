def _exponential_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.02
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # gradient via softmax
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())
        # jitter occasionally
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it} beta={beta:.1f} score={best_val:.6f}")
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)
        it += 1
    return best_a, best_val

def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    start = time.time()
    it = 0
    while time.time() < deadline:
        b = _conv_fft(a)   # full convolution
        k_max = int(np.argmax(b))
        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # gradient contributions
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]
        # step schedule:
        elapsed = time.time() - start
        total = deadline - start
        eta_start = 3.0
        eta_end = 0.1
        eta = eta_start * (1.0 - elapsed/total) + eta_end * (elapsed/total)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} score={best_val:.6f}")
        it += 1
    return best_a, best_val

def _mass_transfer_anneal(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a) # full conv
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start

    T0 = best_val * 0.5  # initial temperature

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # find max index
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions to the peak
        contrib = a[idx] * a[k_max - idx]
        contrib_sum = contrib.sum()
        if contrib_sum > 0:
            donor = rng.choice(idx, p=contrib / contrib_sum)
        else:
            donor = rng.integers(lo, hi + 1)

        # choose receiver biased to low values
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.0015)
        if max_delta <= 0:
            continue

        if T0 > 0:
            delta = rng.random() * max_delta * (temperature / T0)
        else:
            delta = 0.0
        if delta <= 0:
            continue

        # update b efficiently
        b_new = b.copy()
        # linear part
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        idx2 = 2 * donor
        b_new[idx2] += delta * delta
        idx2 = 2 * receiver
        b_new[idx2] += delta * delta
        if donor != receiver:
            cross = donor + receiver
            b_new[cross] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recompute and optional symmetrization
        if time.time() - last_full > 5.0:
            # enforce sum exactly target_sum (numerical drift)
            a = a * (target_sum / a.sum())
            # optional symmetrize
            a = (a + a[::-1]) / 2.0
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")
        # occasional jitter
        if it % 3000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")
        it += 1

    # final symmetrization
    a = best_a
    a = (a + a[::-1]) / 2.0
    a = a * (target_sum / a.sum())
    best_val = _score(a)
    best_a = a
    return best_a, best_val

def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.2

    # choose n
    if budget_s >= 900:
        n = 40000
    elif budget_s >= 600:
        n = 30000
    elif budget_s >= 300:
        n = 20000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # form candidates
    cand_seqs = []
    # Power law decreasing exponents
    for p in np.linspace(0.4, 0.8, 5):
        seq = np.arange(1, n+1, dtype=np.float64) ** (-p)
        cand_seqs.append(seq)
    # Symmetric Beta (U-shaped) exponents
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n
    for a in np.linspace(0.3, 2.0, 6):
        seq = xs**(a-1) * (1 - xs)**(a-1)
        cand_seqs.append(seq)
    # Gaussian-like shapes
    for sigma in [0.08, 0.12, 0.18]:
        seq = np.exp(-0.5*((xs-0.5)/sigma)**2)
        cand_seqs.append(seq)

    # constant shape
    cand_seqs.append(np.ones(n, dtype=np.float64))

    # also warm start from previous if available
    if "height_sequence_1" in globals():
        seq = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        cand_seqs.append(seq)

    best_val = np.inf
    best_seq = None
    for seq in cand_seqs:
        seq = seq.astype(np.float64)
        # avoid all zeros
        if np.all(seq == 0):
            continue
        seq = seq / seq.sum() * target_sum
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score={best_val:.6f}")

    # Phase A: exponential descent
    if time.time() < deadline:
        exp_deadline = time.time() + 0.07 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_val:
            best_seq, best_val = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best score={best_val:.6f}")

    # Phase B: subgradient descent
    if time.time() < deadline:
        sub_deadline = time.time() + 0.08 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_val:
            best_seq, best_val = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_val:.6f}")

    # Phase D (annealing)
    if time.time() < deadline:
        a_ann, val_ann = _mass_transfer_anneal(best_seq, target_sum, deadline)
        if val_ann < best_val:
            best_seq, best_val = a_ann, val_ann
        print(f"[{time.time():.2f}] After annealing best score={best_val:.6f}")

    # Final symmetrization & scaling
    a = best_seq
    a = (a + a[::-1]) / 2.0
    a = a * (target_sum / a.sum())
    final_score = _score(a)
    print(f"[{time.time():.2f}] Final symmetrized score={final_score:.6f}")

    # Trim trailing zeros
    arr = np.array(a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error: {e}")

    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Step‑function optimizer for the benchmark evaluation.

The algorithm proceeds in three stages:

1. **Warm‑start construction** – a short sweep over a few
   parametric families (decreasing power law, symmetric Beta‑densities
   and Gaussian‑like bumps).  The best sequence (scaled to the required
   sum √(2 n)) seeds the next phases.

2. **Fast mirror / sub‑gradient descent** – a few seconds of
   multiplicative mirror descent on a soft‑max surrogate of the
   max‑convolution, followed by true sub‑gradient descent on the exact
   peak.  Both updates keep the vector non‑negative and renormalise to
   the target sum after every step.

3. **Mass‑transfer simulated annealing** – a long‑running annealing that
   repeatedly moves a tiny amount of mass from indices that contribute
   most to the current convolution peak towards indices that are small.
   The temperature starts at ~50 % of the current objective and cools
   linearly to zero.  A full recompute (with optional symmetrisation)
   is performed every few seconds to avoid drift.

The routine automatically adapts the problem size to the time budget
(up to 40 000 points) and stores the resulting sequence in the global
variable ``height_sequence_1`` for warm‑starts in future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).
    The same expression is used by ``evaluate_sequence`` after clipping,
    but here we avoid the clipping because all generated values stay far
    below the 1000‑limit.
    """
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in the mirror‑descent phase)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(b) where b = a*a.
    Returns (grad, full_convolution) – ``grad`` has the same length as ``a``.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.02
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it} β={beta:.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent on the exact peak of the convolution."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]          # sub‑gradient of max(b) w.r.t. a

        # linear cooling of the step size
        elapsed = time.time() - start
        total = deadline - start
        eta_start, eta_end = 3.0, 0.1
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – long simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def _mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """Annealing that moves tiny mass from “bad’’ to “good’’ indices."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.5                  # start temperature (~50 % of current score)

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – index with large contribution (probability proportional to contrib)
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – index with small height (inverse‑probability sampling)
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by the temperature)
        max_delta = min(a[donor] * 0.08, target_sum * 0.0015)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        # --- O(n) update of the convolution ---------------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (remove from donor, add to receiver)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recomputation (also optional symmetrisation)
        if time.time() - last_full > 5.0:
            a = a * (target_sum / a.sum())
            # enforce (optional) symmetry – this never hurts the max‑convolution
            a = (a + a[::-1]) / 2.0
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional tiny jitter to keep the dynamics alive
        if it % 3000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    # final symmetrisation and normalisation
    a = (best_a + best_a[::-1]) / 2.0
    a = a * (target_sum / a.sum())
    best_val = _score(a)
    best_a = a
    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # Adaptive problem size (more points give a finer discretisation)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 40000
    elif budget_s >= 600:
        n = 30000
    elif budget_s >= 300:
        n = 20000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1️⃣  Build a small library of candidate shapes
    # ------------------------------------------------------------------
    cand = []

    # decreasing power‑law families a_i ∝ i^{-p}
    for p in np.linspace(0.4, 0.8, 5):
        cand.append(np.arange(1, n + 1, dtype=np.float64) ** (-p))

    # symmetric Beta densities (U‑shaped for a<1, bell‑shaped for a>1)
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n
    for a in np.linspace(0.3, 2.0, 6):
        cand.append(xs ** (a - 1) * (1.0 - xs) ** (a - 1))

    # Gaussian‑like bumps centred at 0.5
    for sigma in (0.08, 0.12, 0.18):
        cand.append(np.exp(-0.5 * ((xs - 0.5) / sigma) ** 2))

    # flat (all ones) – a simple baseline
    cand.append(np.ones(n, dtype=np.float64))

    # warm‑start from the best known construction if it exists
    if "height_sequence_1" in globals():
        cand.append(_resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n))

    # ------------------------------------------------------------------
    # 2️⃣  Pick the best among the candidates (fast internal score)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in cand:
        seq = seq.astype(np.float64)
        if np.all(seq == 0):
            continue
        seq = seq * (target_sum / seq.sum())
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Initial best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Phase A – exponential (mirror) descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.07 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(best_seq, target_sum, exp_deadline)
        if val_exp < best_val:
            best_seq, best_val = a_exp, val_exp
        print(f"[{time.time():.2f}] After exponential descent best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Phase B – sub‑gradient descent on the true peak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.08 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_val:
            best_seq, best_val = a_sub, val_sub
        print(f"[{time.time():.2f}] After sub‑gradient descent best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Phase D – long simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = _mass_transfer_anneal(best_seq, target_sum, deadline)
        if val_ann < best_val:
            best_seq, best_val = a_ann, val_ann
        print(f"[{time.time():.2f}] After annealing best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final symmetrisation and clean‑up
    # ------------------------------------------------------------------
    a = best_seq
    a = (a + a[::-1]) / 2.0                 # enforce symmetry (never hurts the max)
    a = a * (target_sum / a.sum())          # renormalise

    final_val = _score(a)
    print(f"[{time.time():.2f}] Final internal score = {final_val:.6f}")

    # Trim trailing zeros – the benchmark allows any length.
    arr = np.array(a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Verify with the official evaluator (includes clipping, sanity checks)
    try:
        verified = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Verified benchmark score = {verified:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for future warm‑starts
    globals()["height_sequence_1"] = final_seq
    return final_seq
