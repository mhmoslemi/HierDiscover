    if s < 0:

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Approach
--------
1.  Initialise a candidate on the ℓ¹‑simplex (Σa = √(2 n)) using several
    random families and, if available, the historic ``height_sequence_1``.
2.  Run a two‑phase Adam optimiser:
        • smooth‑max (log‑sum‑exp) with a temperature β that is doubled
          regularly – this quickly finds a promising region.
        • exact‑max sub‑gradient Adam – drives the true peak down.
3.  Perform a greedy deterministic hill‑climb that repeatedly moves mass
    from the index contributing most to the current worst convolution entry
    to a “quiet’’ index (the same routine used in the reference code).
4.  Finish with a lightweight simulated‑annealing phase that
    proposes random mass moves, accepting them according to a temperature
    schedule and updating the convolution incrementally.
5.  All updates keep the sum exactly constant (mass moves) and after a
    handful of iterations the vector is re‑projected onto the simplex using
    the exact Euclidean projection.
6.  The whole search respects a wall‑clock budget (default 1000 s); the best
    sequence found so far is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # work on a copy
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.full_like(v, s / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a∘a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∘a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(a) to new_n, then simplex proj."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex(a_new, target)

def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, 1.0)

    a += rng.normal(scale=target * 1e-3, size=n)
    return _project_to_simplex(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a∘a.
    Returns g = 2·(a ⋆ p) with p ∝ exp(β·b).
    """
    n = a.shape[0]
    size_fft = _next_pow2(2 * n - 1)

    a_fft = np.fft.rfft(a, n=size_fft)
    b = np.fft.irfft(a_fft * a_fft, n=size_fft)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution a_rev (reversed a) with p
    a_rev = a[::-1]
    size_conv = _next_pow2(len(a_rev) + len(p) - 1)
    A_fft = np.fft.rfft(a_rev, n=size_conv)
    P_fft = np.fft.rfft(p, n=size_conv)
    conv_rev = np.fft.irfft(A_fft * P_fft, n=size_conv)[: n + (2 * n - 1) - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize_softmax(a0: np.ndarray, target_sum: float,
                           deadline: float) -> np.ndarray:
    """Projected Adam on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.size
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    base_lr = 0.15 * target_sum
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """Projected Adam where the gradient is the exact sub‑gradient of max(a∘a)."""
    a = a0.copy()
    n = a.size
    base_lr = 0.20 * target_sum
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

    return a

def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        # pick a destination j whose counterpart a[M‑j] is as small as possible
        counterpart = np.zeros(n, dtype=float)
        counterpart[i_low:i_high + 1] = a[M - np.arange(i_low, i_high + 1)]
        order = np.argsort(counterpart[i_low:i_high + 1])
        j = None
        for cand in np.arange(i_low, i_high + 1)[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        # aggressive move, then back‑track if it does not improve
        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # compute a tentative new convolution (linear‑time update)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
            continue

        # try a smaller move (binary search on the step size)
        improved = False
        delta2 = delta * 0.5
        while delta2 > 1e-14:
            b_test = b.copy()
            b_test[i:i + n] -= 2.0 * delta2 * a
            b_test[j:j + n] += 2.0 * delta2 * a
            if k2i < b_test.size:
                b_test[k2i] += delta2 * delta2
            if k2j < b_test.size:
                b_test[k2j] += delta2 * delta2
            if kij < b_test.size:
                b_test[kij] -= 2.0 * delta2 * delta2
            if b_test.max() < b.max() - 1e-12:
                a[i] -= delta2
                a[j] += delta2
                b = b_test
                if b.max() < best_score - 1e-12:
                    best_score = b.max()
                    best_a = a.copy()
                improved = True
                break
            delta2 *= 0.5

        if not improved:
            # tiny random perturbation to escape possible plateaus
            jitter = rng.normal(scale=1e-6, size=n)
            a += jitter
            a = _project_to_simplex(a, math.sqrt(2.0 * n))
            b = _full_convolution(a)

    return best_a, best_score

def _simulated_annealing(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the exact max.
    Each move transfers a random fraction of mass from i to j.
    The convolution is updated incrementally (O(n)) per move.
    """
    a = a0.copy()
    n = a.size
    target = math.sqrt(2.0 * n)
    b = _full_convolution(a)
    cur_max = b.max()
    best_a = a.copy()
    best_score = cur_max
    rng = np.random.default_rng()

    # temperature schedule – linear decay from T0 to 0
    start_time = time.time()
    total = deadline - start_time
    T0 = max(0.3, cur_max * 0.2)          # a modest initial temperature

    it = 0
    while time.time() < deadline:
        it += 1
        # linear cooling
        remaining = deadline - time.time()
        T = T0 * (remaining / total)

        # choose source i (prefer indices with large a)
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        # choose destination j ≠ i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose a move; delta is a random fraction of a[i] scaled by temperature
        frac = rng.random() * T
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # incremental update of convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        improve = new_max < cur_max
        accept = improve or (rng.random() < math.exp(-(new_max - cur_max) / (T + 1e-12)))
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()

        # occasional re‑projection to correct floating‑point drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target)
            b = _full_convolution(a)
            cur_max = b.max()
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method combines:
        • random families,

    # ------------------------------------------------------------------

              f"(length={best_arr.size})")

        print("Evaluation error:", exc)

    return best_arr.tolist()
