def _greedy_optimize(a, target_sum, deadline):
    n = len(a)
    b = _conv_fft(a)
    current_val = _score(a)
    best_a = a.copy()
    best_val = current_val
    rng = np.random.default_rng()
    last_full = time.time()
    while time.time() < deadline:
        # locate peak
        k_max = int(np.argmax(b))
        # compute contributing region indices:
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        # contributions
        # product of a_i * a_{k_max - i}
        contrib = a[idx] * a[k_max - idx]
        # donor index = idx of max contribution
        max_idx = np.argmax(contrib)
        donor = idx[max_idx]
        # receiver: choose index with smallest a (might be zero)
        # choose random among bottom 5% to add randomness
        sorted_indices = np.argsort(a)
        # take bottom m
        m = max(1, n // 200)  # bottom 0.5%
        candidates = sorted_indices[:m]
        receiver = int(rng.choice(candidates))
        if donor == receiver:
            # pick random other index
            receiver = int(rng.choice(np.setdiff1d(np.arange(n), [donor])))
        # compute delta
        max_delta = min(a[donor] * 0.05, target_sum * 0.002)  # bigger than previous
        if max_delta <= 0:
            # no move possible
            break
        # Set delta adaptively based on current max value (maybe larger)
        delta = max_delta * rng.random()
        # evaluate effect using copy of b
        b_new = b.copy()
        # linear updates:
        # shift donor left: positions donor..donor+n-1 subtract 2*delta*a
        b_new[donor:donor+n] -= 2.0 * delta * a
        b_new[receiver:receiver+n] += 2.0 * delta * a

        b_new[2 * receiver] += delta * delta

        # compute new max
        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))
        # accept if lower or with probability based on temperature
        # Temperature schedule linearly decreasing
        total = deadline - start_time  # we need start_time global maybe pass as param
        elapsed = time.time() - start_time
        # maybe we have a variable T_start = best_val * 0.2 etc.
        # For simplicity accept only if better
        if new_val < current_val:
            # accept
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Greedy improved to {best_val:.6f}")
        else:

    Strategy:
    * Warm‑start from the previous best or a random Dirichlet vector.
    * A short mirror‑descent phase on a soft‑max surrogate of the max‑convolution
      drives the peak down quickly.

      algorithm escape shallow local minima.

"""
Search for a low‑score step‑function sequence.

The algorithm consists of three stages:

1. **Warm‑start** – either reuse the best known construction (`height_sequence_1`)
   or draw a random Dirichlet‑like vector and scale it to the required sum.
2. **Rapid mirror‑descent** – a short exponential (multiplicative) descent on a
   soft‑max surrogate of the max‑convolution drives the peak down quickly.
3. **Aggressive annealing** – the main work is done by a simulated‑annealing
   mass‑transfer loop that repeatedly moves a tiny amount of mass from the
   index that contributes most to the current convolution peak to an index
   with very small height.  Acceptance is based on the *current* objective
   (instead of the best seen) and a linear temperature schedule, which
   enables occasional uphill moves and helps escape shallow local minima.
   When time permits the annealing is restarted from the best solution found
   so far with a small random perturbation.

All heavy work uses FFT based convolution for speed; the final sequence is
validated with the provided `evaluate_sequence` function before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities (used in the short mirror‑descent)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive mass‑transfer annealing (main work horse)
# ----------------------------------------------------------------------


def _improved_mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing mass‑transfer that uses the *current* objective for acceptance.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = cur_val * 0.6                  # higher starting temperature

    last_full = start
    low_idx_cache = None                # cache for low‑height indices
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – index with the largest contribution to the peak
        donor = idx[int(np.argmax(contrib))]

        # receiver – index with a very small height (bottom 0.5 % of values)
        if it % 500 == 0 or low_idx_cache is None:
            low_idx_cache = np.argsort(a)[: max(1, n // 200)]
        receiver = int(rng.choice(low_idx_cache))

        if donor == receiver:
            # pick a completely random receiver if unlucky
            receiver = int(rng.integers(n))
            if receiver == donor:
                continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.08, target_sum * 0.003)
        if max_delta <= 0:
            break
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 1.0)
        if delta <= 0.0:
            delta = max_delta * 0.001  # very small fallback

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance based on *current* value
        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f} it={it}")

        # periodic full recomputation to avoid drift
        if time.time() - last_full > 8.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")
            last_full = time.time()

        # occasional tiny jitter to keep dynamics alive
        if it % 2000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


def _run_annealing_with_restarts(init_seq: np.ndarray, target_sum: float, deadline: float):
    """
    Run several annealing chunks, each starting from the best solution so far
    with a small random perturbation.
    """
    best_seq = init_seq.copy()
    best_val = _score(best_seq)
    rng = np.random.default_rng()

    while True:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        # allocate a chunk (max 300 s, at least 30 s if possible)
        chunk = min(remaining * 0.5, 300.0)
        if chunk < 30.0:
            break

        seq, val = _improved_mass_transfer_anneal(
            best_seq, target_sum, now + chunk
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Restart anneal improved best={best_val:.6f}")

        # small random perturbation before the next restart
        jitter = rng.normal(scale=target_sum * 0.001, size=best_seq.shape)
        best_seq = np.maximum(best_seq + jitter, 0.0)
        best_seq = best_seq * (target_sum / best_seq.sum())

    return best_seq, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 25000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start from the best known construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
    else:
        rng = np.random.default_rng()
        # Gamma‑distributed Dirichlet‑like vector (α < 1 → sparse)
        alpha = 0.3
        raw = rng.gamma(alpha, 1.0, size=n)
        init = raw / raw.sum() * target_sum

    # --------------------------------------------------------------
    # Phase A – short exponential mirror descent (≈ 5 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        exp_deadline = time.time() + 0.05 * (deadline - time.time())
        a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
        best_seq = a_exp
        best_val = val_exp
        print(f"[{time.time():.2f}] After exponential descent best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase B – aggressive annealing with restarts (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_val = _run_annealing_with_restarts(
            best_seq, target_sum, deadline
        )
        print(f"[{time.time():.2f}] After annealing restarts best={best_val:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
