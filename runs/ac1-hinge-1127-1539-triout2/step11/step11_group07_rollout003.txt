"""
Propose a step‑function sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The code combines several complementary heuristics:
  * Random families for diverse starts.
  * Projected Adam on a smooth‑max (log‑sum‑exp) surrogate.
  * Projected Adam on the exact max (sub‑gradient).
  * Spectral flattening – enforce a (nearly) constant magnitude Fourier
    spectrum, which drives the linear convolution towards a flat profile.
  * A cheap linear‑programming “direction” that moves mass while never
    increasing the current worst convolution entry.
  * Deterministic hill‑climbing on the exact convolution.
  * Small random “tweaks” for final fine‑tuning.

All steps respect a wall‑clock budget (default 1000 s) and keep the
vector on the ℓ¹‑simplex Σa = √(2 n) after every update.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    Clamp to non‑negative and renormalise the ℓ¹‑norm.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers (several simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p (full length needed)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate (log‑sum‑exp).
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of max(a*a).
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.20 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1

        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Spectral flattening – force a (nearly) constant magnitude spectrum.
# ----------------------------------------------------------------------


def _spectral_flatten(a: np.ndarray, target_sum: float,
                     deadline: float, max_iter: int = 200) -> np.ndarray:
    """
    Repeatedly replace the magnitude of the FFT of a zero‑padded version
    with a constant (preserving total power) and project back onto the simplex.
    """
    n = a.size
    L = _next_pow2(2 * n)             # length for a circular convolution proxy
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a_pad = np.zeros(L, dtype=float)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        power = np.abs(A) ** 2
        total_power = power.sum()
        const_mag = math.sqrt(total_power / A.size)
        A_new = const_mag * np.exp(1j * np.angle(A))
        a_new_full = np.fft.irfft(A_new, n=L)
        a = a_new_full[:n]
        a = _project_to_simplex_fast(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Linear‑programming step that moves mass while never exceeding the current peak.
# ----------------------------------------------------------------------


def _lp_step(a: np.ndarray, current_max: float, target_sum: float,
             deadline: float, alpha: float = 0.5) -> tuple[np.ndarray, bool]:
    """
    Solve   max Σ g_i   s.t.   a * g ≤ current_max   and   g ≥ 0.
    Return the convex combination (1‑α)·a + α·g_scaled if it improves the
    objective, together with a flag indicating success.
    """
    n = a.size
    m = 2 * n - 1

    # Build sparse matrix for the linear constraints: (a * g)_k ≤ current_max.
    # Each column i contributes a[i] to rows i … i+n‑1.
    A = sp.lil_matrix((m, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a[i]
    A = A.tocsr()

    c = -np.ones(n)                     # maximize Σ g_i  →  minimize –Σ g_i
    bounds = [(0, None)] * n
    b_ub = np.full(m, current_max, dtype=float)

    # Solve with the HiGHS interior‑point method (fast for moderate size).
    # Put a tiny time‑limit so we never exceed the overall deadline.
    remaining = max(0.0, deadline - time.time())
    opts = {"time_limit": remaining}
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options=opts)

    if not res.success:
        return a, False

    g0 = res.x
    sum_g = g0.sum()
    if sum_g == 0.0:
        return a, False

    # Scale g0 to the required ℓ¹‑norm.
    g = g0 * (target_sum / sum_g)

    # Convex combination keeps the sum invariant.
    a_new = (1.0 - alpha) * a + alpha * g

    # Project onto simplex to kill any tiny numerical violations.
    a_new = _project_to_simplex_fast(a_new, target_sum)

    new_max = _max_convolution_fft(a_new)
    if new_max < current_max - 1e-12:
        return a_new, True
    else:
        return a, False


# ----------------------------------------------------------------------
# Deterministic hill‑climb on the exact convolution
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        # indices participating in this entry
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # counterpart values for the selected i
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart = a[M - j_range]

        # pick a destination with the smallest counterpart
        order = np.argsort(counterpart)
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # linear‑time update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small random jitter to escape possible plateaus
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method mixes random restarts, Adam optimisation (smooth and exact),
    spectral flattening, a cheap LP move, deterministic hill‑climbing and
    final random tweaks, while respecting the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously discovered solution, if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            target0 = math.sqrt(2.0 * prev.size)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={prev.size}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # Length schedule – start modest, then try a larger dimension.
    # ------------------------------------------------------------------
    schedule = [2000, 4000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # initialise (upsample previous best if possible)
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng)

        a = _project_to_simplex_fast(a, target)

        remaining = deadline - time.time()
        if remaining < 15.0:
            break

        # ------------------------------------------------------------------
        # 1) Adam – smooth‑max in the first half of the budget for this size.
        # ------------------------------------------------------------------
        phase1_dead = min(deadline - 0.2,
                          time.time() + remaining * 0.45)
        a = _adam_optimize(a, target, phase1_dead)

        # ------------------------------------------------------------------
        # 2) Adam – exact max (sub‑gradient) for the second half of the budget.
        # ------------------------------------------------------------------
        phase2_dead = deadline - 0.2
        a = _adam_optimize_submax(a, target, phase2_dead)

        # ------------------------------------------------------------------
        # Evaluate after Adam phases.
        # ------------------------------------------------------------------
        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_seq = a.copy()
            best_score = cur_score
            print(f"[{time.time():.1f}] After Adam (n={n}) score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 3) Spectral flattening (use a modest slice of the remaining time).
        # ------------------------------------------------------------------
        flat_dead = min(deadline - 0.2,
                        time.time() + (deadline - time.time()) * 0.10)
        a = _spectral_flatten(a, target, flat_dead, max_iter=150)
        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_seq = a.copy()
            best_score = cur_score
            print(f"[{time.time():.1f}] After spectral flatten (n={n}) "
                  f"score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 4) Linear‑programming move (only for modest n to keep memory sane).
        # ------------------------------------------------------------------
        if n <= 4000 and time.time() < deadline - 5.0:
            lp_dead = deadline - 1.0
            while time.time() < lp_dead:
                cur_max = _max_convolution_fft(a)
                a, improved = _lp_step(a, cur_max, target, lp_dead, alpha=0.5)
                if not improved:
                    break
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score - 1e-12:
                    best_seq = a.copy()
                    best_score = cur_score
                    print(f"[{time.time():.1f}] LP step improved (n={n}) "
                          f"score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 5) Deterministic hill‑climb (use a fraction of the remaining budget).
        # ------------------------------------------------------------------
        if time.time() < deadline - 2.0:
            hc_dead = deadline - 0.5
            b = _full_convolution(a)
            a_hc, hc_score = _hill_climb_deterministic(a, b, hc_dead)
            if hc_score < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = hc_score
                print(f"[{time.time():.1f}] Hill‑climb improved (n={n}) "
                      f"score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 6) Tiny random tweaks (simulated‑annealing style) until time runs out.
        # ------------------------------------------------------------------
        while time.time() + 1.5 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            cand_score = _max_convolution_fft(cand)
            if cand_score < best_score - 1e-12:
                best_seq = cand
                best_score = cand_score
                print(f"[{time.time():.1f}] Random tweak improved (n={n}) "
                      f"score={best_score:.6f}")

        # ------------------------------------------------------------------
        # End of this size – report best found for the current n.
        # ------------------------------------------------------------------
        print(f"[{time.time():.1f}] End of n={n} – best score so far: "
              f"{best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final polishing on whatever time is left (short Adam + hill‑climb).
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_dead = deadline - 0.5
        best_seq = _adam_optimize_submax(best_seq, target, final_dead)
        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.2)
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros.
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ----------------------------------------------------------------------
    # Official evaluation report (for debugging / logging).
    # ----------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
