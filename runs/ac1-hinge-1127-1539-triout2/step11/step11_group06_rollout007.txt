def _adam_optimize(a0, target_sum, deadline, beta_schedule=None):
   a = a0.copy()
   a = a * (target_sum / a.sum())
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   best_a = a.copy()
   best_score = _score(a)
   t = 0
   rng = np.random.default_rng()
   while time.time() < deadline:
        t += 1
        # compute gradient (subgradient of max)
        g, max_val, k_max = _max_subgradient(a)
        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # learning rate schedule: decreasing over time
        base_lr = target_sum * 0.1
        lr = base_lr / (1 + 0.001 * t)  # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        # evaluate
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iter {t} score {best_score:.6f}")

        # jitter / restart occasionally
        if t % 500 == 0:
            # jitter small
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            # reset moments to zero to avoid stale momentum
            m.fill(0.0)
            v.fill(0.0)

   return best_a, best_score

"""
Hybrid optimizer with a fast linear‑programming (LP) boost.

Main ideas
----------
* Warm‑start from the previously best sequence (or a simple power‑law).
* Run a few LP iterations on a *small* problem (≈1500 entries).  The LP
  maximises the total mass that can be moved without increasing the
  maximal self‑convolution.  After the LP stage we up‑sample the result
  to the full size.
* Refine the up‑sampled vector with an Adam‑style sub‑gradient descent
  on the exact max‑convolution (the true objective).
* Finish with two cheap but powerful local phases:
      – deterministic “mass‑transfer’’ hill‑climbing,
      – a lightweight simulated‑annealing move set.
* All phases respect the global time budget (default 1000 s).  The best
  sequence found is stored in the global `height_sequence_1` for warm‑starts
  in later calls.

The score returned by ``evaluate_sequence`` equals the maximal entry of
the self‑convolution, so we directly minimise ``max(a*a)`` under the
normalisation ``sum(a)=sqrt(2·n)``.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT, score, projection)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  With the normalisation used below this
    reduces to max(a*a)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:                     # safety – should never happen after normalisation
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to ``new_len`` entries."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------


def _max_subgradient(a: np.ndarray):
    """Return (g, max_val, k_max) where g is a sub‑gradient of max(a*a)."""
    b = _conv_fft(a)
    k_max = int(np.argmax(b))
    max_val = float(b[k_max])
    n = a.shape[0]
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g = np.zeros_like(a)
    # derivative of sum_i a_i a_{k_max-i} w.r.t. a_j is 2*a_{k_max-j}
    g[idx] = 2.0 * a[k_max - idx]
    return g, max_val, k_max


# ----------------------------------------------------------------------
# Adam optimiser on the exact sub‑gradient
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """Adam‑style optimiser on the true max‑convolution."""
    a = a0.copy()
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        g, _, _ = _max_subgradient(a)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule – start relatively aggressive, decay slowly
        base_lr = target_sum * 0.12
        lr = base_lr / (1.0 + 0.0005 * it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} score={best_val:.6f}")

        # occasional jitter & reset of moments to escape plateaus
        if it % 600 == 0:
            jitter = rng.normal(scale=target_sum * 0.0012, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
            m.fill(0.0)
            v.fill(0.0)

    return best_a, best_val

def _build_convolution_matrix(f: np.ndarray):
    """Sparse matrix A with rows = (f * g) for all shifts.  Shape (2n‑1, n)."""
    n = f.shape[0]
    # rows = i + j for i,j∈[0,n)
    row_idx = np.arange(n, dtype=np.int64)[:, None] + np.arange(n, dtype=np.int64)[None, :]
    rows = row_idx.ravel()
    cols = np.repeat(np.arange(n, dtype=np.int64), n)
    data = np.repeat(f, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_step(a: np.ndarray, target_sum: float, M: float):
    """One LP iteration: maximise Σ g subject to (a * g) ≤ M, g≥0."""
    n = a.shape[0]
    A = _build_convolution_matrix(a)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # maximise → minimise –cᵀg

    res = scipy.optimize.linprog(c,
                                 A_ub=A,
                                 b_ub=b_ub,
                                 bounds=[(0, None)] * n,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        # fallback – return the original vector unchanged
        return a, M

    g0 = res.x
    total = g0.sum()
    if total <= 0.0:
        return a, M

    # Scale to satisfy the global normalisation
    g = g0 * (target_sum / total)

    # Small convex combination (t≈0.2) moves us towards the improving direction
    t = 0.20
    new_a = (1.0 - t) * a + t * g
    new_a = np.maximum(new_a, 0.0)
    new_a = new_a * (target_sum / new_a.sum())
    new_M = _conv_fft(new_a).max()
    return new_a, float(new_M)


def _lp_phase(a_init: np.ndarray, target_sum: float, deadline: float):
    """Run a few LP iterations on a reduced‑size problem."""
    # work on a compressed problem (≈1500 entries) for speed
    n_full = a_init.shape[0]
    n_small = min(1500, max(600, n_full // 8))
    a_small = _resize_sequence(a_init, n_small)
    target_small = math.sqrt(2.0 * n_small)
    a_small = a_small * (target_small / a_small.sum())

    it = 0
    while time.time() < deadline:
        it += 1
        M = _conv_fft(a_small).max()
        a_small, M = _lp_step(a_small, target_small, float(M))
        if it % 2 == 0:
            # keep a copy of the best seen on the *full* scale
            a_full = _resize_sequence(a_small, n_full)
            a_full = a_full * (target_sum / a_full.sum())
        # stop after a handful of iterations – LP is expensive
        if it >= 5:
            break

    # final up‑sample to the original size and normalise
    a_up = _resize_sequence(a_small, n_full)
    a_up = a_up * (target_sum / a_up.sum())
    return a_up

def _local_refine(a0: np.ndarray, target_sum: float, deadline: float):
    """Exact hill‑climbing that moves a tiny amount of mass between two indices.
    All updates of the convolution are performed in O(n)."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        # locate the current worst (max) convolution entry
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – the index contributing most to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() == 0:
            donor = rng.integers(lo, hi + 1)
        else:
            donor = rng.choice(idx, p=contrib / contrib.sum())

        # receiver – any index with relatively small value (biased towards zero)
        low = np.where(a <= np.percentile(a, 30))[0]
        if low.size:
            receiver = rng.choice(low)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # magnitude of the move (a few per‑cent of the donor mass)
        max_delta = min(a[donor] * 0.045, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) update of convolution (see original code for the exact formula)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # accept if improvement; otherwise revert
        if new_val < best_val:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Local refine improved to {best_val:.6f}")

        # occasional re‑normalisation to curb drift
        if (time.time() - start) % 11.0 < 0.01:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing (mass moves with O(n) updates)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, target_sum: float, deadline: float):
    """SA that repeatedly transfers a tiny amount of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_a, best_val

    # initial temperature (a fraction of the current excess over 1)
    T0 = max(0.05, (best_val - 1.0) * 0.55)

    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - elapsed / total))

        # donor chosen proportionally to current mass
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            continue
        # receiver uniform and distinct
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = min(a[donor] * 0.06, target_sum * 0.0012)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        # O(n) convolution update
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val < cur_val:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved to {best_val:.6f}")

        # occasional global jitter to escape flat regions
        if (int(elapsed * 50) % 4000) == 0:
            jitter = rng.normal(scale=target_sum * 0.00018, size=n)
            a_j = a + jitter
            a_j = np.maximum(a_j, 0.0)
            a_j = a_j * (target_sum / a_j.sum())
            b_j = _conv_fft(a_j)
            val_j = _score(a_j)
            if val_j < best_val:
                best_val = val_j
                best_a = a_j.copy()
                a = a_j
                b = b_j
                cur_val = val_j
                print(f"[{time.time():.2f}] SA jitter improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low max‑convolution value.
    The routine respects the time budget (seconds) and returns the best
    sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5                     # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more flexibility but costs more)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Warm‑start (reuse previous best if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_full = np.array(globals()['height_sequence_1'], dtype=np.float64)
        if init_full.shape[0] != n:
            init_full = _resize_sequence(init_full, n)
    else:
        # simple decreasing power‑law fallback
        init_full = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init_full = init_full * (target_sum / init_full.sum())

    # ------------------------------------------------------------------
    # Phase 1 – LP boost on a reduced problem
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + min(0.15 * (deadline - time.time()), 30.0)
        a_after_lp = _lp_phase(init_full, target_sum, lp_deadline)
        best_seq = a_after_lp
        best_score = _score(best_seq)
        print(f"[{time.time():.2f}] After LP boost score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimiser on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + 0.30 * (deadline - time.time())
        a_adam, val_adam = _adam_optimize(best_seq, target_sum, adam_deadline)
        if val_adam < best_score:
            best_seq, best_score = a_adam, val_adam
        print(f"[{time.time():.2f}] After Adam optimisation score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic local mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        ref_deadline = time.time() + 0.25 * (deadline - time.time())
        a_ref, val_ref = _local_refine(best_seq, target_sum, ref_deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After local refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Fast simulated‑annealing finetuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, deadline)
        if val_sa < best_score:
            best_seq, best_score = a_sa, val_sa
        print(f"[{time.time():.2f}] After simulated annealing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up – strip trailing near‑zeros and store for warm‑starts
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final external evaluation (the grader will also do this)
    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # keep for the next call
    globals()["height_sequence_1"] = final_seq

    return final_seq
