"""
Improved stochastic search for a low‑peak step function.

The algorithm consists of three stages:

1. **Exploratory short runs on a modest problem size** – many random
   initialisations are refined by a fast “mass‑transfer” annealer.
   The best shape is kept.

2. **Upscaling** – the best small‑size shape is interpolated to the final
   problem size (chosen according to the time budget) and re‑normalised.

3. **Long‑run refinement** – a final annealer with a higher temperature
   and larger transfer steps runs until the time budget expires.

All updates of the self‑convolution are performed in O(n) using the exact
formula from the original benchmark code, while full recomputation via FFT
is done only occasionally to avoid drift.

The returned sequence satisfies the non‑negativity constraints and is
stored in the global variable ``height_sequence_1`` for warm‑starts in
future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²). Small n → fast."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection on {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent (true max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        # gradient of the active max‑convolution entry
        g[idx] = 2.0 * a[k_max - idx]

        # linear schedule for the step size
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.2 * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Mass‑transfer simulated annealing
# ----------------------------------------------------------------------


def _mass_transfer_anneal(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_delta_frac: float = 0.08,
    max_total_frac: float = 0.003,
):
    """
    Simulated annealing that repeatedly moves a small amount of mass from a
    “donor’’ (large contribution to the current peak) to a “receiver’’
    (low height).  The update of the convolution is performed in O(n) time.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val                       # start temperature ≈ current score
    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by the temperature)
        max_delta = min(a[donor] * max_delta_frac, target_sum * max_total_frac)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor : donor + n] += -2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_full > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional tiny jitter
        if it % 2000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _init_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Return a random non‑negative sequence of length n with the required sum."""
    kind = rng.integers(0, 4)
    if kind == 0:
        # exponential random values
        a = rng.exponential(scale=1.0, size=n)
    elif kind == 1:
        # decreasing power‑law
        p = rng.uniform(0.3, 1.0)
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    elif kind == 2:
        # sine shape (zero at the boundaries)
        x = np.arange(1, n + 1, dtype=np.float64)
        a = np.sin(np.pi * x / (n + 1))
    else:
        # simple linear ramp
        a = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # avoid exact zeros
    a += rng.random(n) * 1e-12
    a = a * (target_sum / a.sum())
    return a


# ----------------------------------------------------------------------
# Small‑size exploratory optimisation
# ----------------------------------------------------------------------


def _optimize_small_n(n: int, budget_sec: float, rng: np.random.Generator):
    """
    Run many short annealing runs on a modest problem size.
    Returns the best sequence found and its score.
    """
    target_sum = np.sqrt(2.0 * n)
    start = time.time()
    best_a = None
    best_val = np.inf
    # each short run lasts ≈ 4 s (or the remaining budget)
    while time.time() < start + budget_sec - 2.0:
        run_time = min(4.0, start + budget_sec - time.time())
        a0 = _init_sequence(n, target_sum, rng)
        a_opt, val_opt = _mass_transfer_anneal(
            a0,
            target_sum,
            time.time() + run_time,
            max_delta_frac=0.10,
            max_total_frac=0.004,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] small n={n} new best={best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose final problem size (larger n gives finer granularity)
    # --------------------------------------------------------------
    if budget_s >= 800:
        final_n = 18000
    elif budget_s >= 500:
        final_n = 12000
    else:
        final_n = 8000

    target_sum = np.sqrt(2.0 * final_n)

    # --------------------------------------------------------------
    # Allocate a fraction of the budget to small exploratory runs
    # --------------------------------------------------------------
    total_time = deadline - start
    small_budget = total_time * 0.35
    small_n = max(2000, final_n // 5)      # e.g. 3600 for final_n=18000
    rng = np.random.default_rng()

    best_small_a = None
    best_small_val = np.inf

    if small_budget > 8.0:
        best_small_a, best_small_val = _optimize_small_n(small_n, small_budget, rng)
        print(f"[{time.time():.2f}] Small‑size optimisation best score={best_small_val:.6f}")

    # --------------------------------------------------------------
    # Upscale the best small construction (or fall back to a simple shape)
    # --------------------------------------------------------------
    if best_small_a is not None:
        init_large = _resize_sequence(best_small_a, final_n)
    else:
        # simple decreasing power‑law fallback
        init_large = (np.arange(1, final_n + 1, dtype=np.float64) ** (-0.55))

    init_large = init_large * (target_sum / init_large.sum())

    # --------------------------------------------------------------
    # A short sub‑gradient phase on the full size (helps to escape)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = time.time() + min(8.0, deadline - time.time() - 5.0)
        init_large, sub_val = _subgradient_descent(init_large, target_sum, sub_deadline)
        print(f"[{time.time():.2f}] Subgradient after up‑scaling score={sub_val:.6f}")

    # --------------------------------------------------------------
    # Final long‑run annealing on the full size
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_a, final_val = _mass_transfer_anneal(
            init_large,
            target_sum,
            deadline,
            max_delta_frac=0.08,   # a bit larger than the exploratory phase
            max_total_frac=0.003,
        )
        print(f"[{time.time():.2f}] Final annealing best score={final_val:.6f}")
    else:
        final_a = init_large
        final_val = _score(final_a)

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(final_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_eval = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_eval:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
