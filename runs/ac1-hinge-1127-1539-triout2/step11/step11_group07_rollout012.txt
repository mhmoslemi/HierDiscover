def _lp_refine(a, M, target_sum, t, deadline):
    n = len(a)
    L = 2*n-1
    # build constraint matrix
    A = np.zeros((L, n))
    for i in range(n):

def _lp_refine(a, M, target_sum, step, deadline):
    import scipy.optimize
    n = len(a)
    L = 2 * n - 1
    # build constraint matrix A
    # Build as dense numpy array
    A = np.empty((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = a
    # Solve LP
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # b_ub is vector of M repeated
    b_ub = np.full(L, M, dtype=np.float64)

    # use hi-level linprog
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    if not res.success:
        return a
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return a
    # scale to target sum
    scale = target_sum / sum_g0
    g = g0 * scale
    # combine
    a_new = (1 - step) * a + step * g
    a_new = _project_to_simplex_fast(a_new, target_sum)
    return a_new

"""
Improved step‑function search.

The algorithm combines three ideas that were not used together before:
1. Random families for a diverse start.
2. A linear‑programming “LP‑refine’’ step: given a current vector a,
   we maximise the ℓ¹‑norm of a new vector g while keeping the
   convolution a*g below the current worst autocorrelation M.
   Mixing a with the scaled optimiser g (convex combination) strictly
   reduces the peak in practice.
3. A fast deterministic hill‑climb that moves mass between two
   indices chosen from the current maximal convolution entry.

The LP step is performed a few times on a moderate size (n≈1000) and
again on the final size (n≈2000).  The whole search respects a
wall‑clock budget (default 1000 s) and always returns a feasible,
non‑negative sequence.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # SciPy is allowed by the rules


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}:
    clamp to non‑negative and renormalise the ℓ¹‑norm.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # fall back to uniform distribution
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Maximum entry of a*a using an FFT (fast, O(n log n)).
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (returns the whole array).
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation onto a new length, followed by ℓ¹‑simplex
    projection to keep the prescribed sum.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers (several simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray, M: float, target_sum: float,
               step: float, deadline: float) -> np.ndarray:
    """
    One LP‑refinement iteration.
    Solve   max Σ g_j   s.t.  a * g ≤ M   and  g ≥ 0,
    then mix the scaled optimiser g with a.
    """
    n = a.size
    L = 2 * n - 1

    # Build the convolution matrix (each column is a shifted copy of a)
    A = np.empty((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a

    # Linear program: maximise sum(g)  ==  minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    b_ub = np.full(L, M, dtype=np.float64)

    # Give the solver a tiny time budget so we never exceed the outer deadline
    time_left = max(0.0, deadline - time.time())
    options = {"presolve": True}
    if time_left > 0.0:
        options["time_limit"] = time_left

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options=options)

    if not res.success:
        # fall back to the original vector
        return a

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0.0:
        return a

    # Scale to the required ℓ¹‑norm and mix
    scale = target_sum / sum_g0
    g = g0 * scale
    a_new = (1.0 - step) * a + step * g
    a_new = _project_to_simplex_fast(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
# Deterministic hill‑climb on the exact convolution
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose a destination with the smallest counterpart
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart = a[M - j_range]

        order = np.argsort(counterpart)
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.6          # a more conservative move than 0.8
        if delta <= 0.0:
            break

        # linear‑time update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                # small random jitter to escape flat regions
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method combines
        • random initial families,
        • a few LP‑refinement steps,
        • deterministic hill‑climb,
        • lightweight random tweaks,
    respecting a wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optional seed from a previously found solution
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            target0 = math.sqrt(2.0 * prev.size)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={prev.size}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # --------------------------------------------------------------
    # Length schedule – start small, finish at the target length
    # --------------------------------------------------------------
    schedule = [1000, 2000]   # small (fast LP) then the final length

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ----------------------------------------------------------
        # Initialise (upsample previous best if possible)
        # ----------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng)

        a = _project_to_simplex_fast(a, target)

        # ----------------------------------------------------------
        # Allocate a slice of the remaining budget
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 15.0:
            break

        # 1) a handful of LP refinement steps
        lp_steps = 4 if n == 1000 else 3
        for _ in range(lp_steps):
            if time.time() >= deadline:
                break
            b = _full_convolution(a)
            M = b.max()
            a = _lp_refine(a, M, target, step=0.4, deadline=deadline)

        # 2) Deterministic hill‑climb (exact max)
        if time.time() < deadline - 2.0:
            b = _full_convolution(a)
            a, score = _hill_climb_deterministic(a, b, deadline - 0.5)
            if score < best_score - 1e-12:
                best_seq = a.copy()
                best_score = score
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={best_score:.6f}")

        # 3) Light random tweaks while time permits
        while time.time() + 1.0 < deadline:
            cand = best_seq.copy() if best_seq is not None else a.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            cand_score = _max_convolution_fft(cand)
            if cand_score < best_score - 1e-12:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{time.time():.1f}] Random tweak improvement (n={n}) "
                      f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing (a few extra LP passes + hill‑climb)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        for _ in range(2):
            if time.time() >= deadline:
                break
            b = _full_convolution(best_seq)
            M = b.max()
            best_seq = _lp_refine(best_seq, M, target, step=0.3, deadline=deadline)

        if time.time() < deadline - 1.0:
            b = _full_convolution(best_seq)
            best_seq, final_score = _hill_climb_deterministic(
                best_seq, b, deadline - 0.5)
            if final_score < best_score - 1e-12:
                best_score = final_score
                print(f"[{time.time():.1f}] Final hill‑climb improvement "
                      f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # --------------------------------------------------------------
    # Official evaluation report
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
