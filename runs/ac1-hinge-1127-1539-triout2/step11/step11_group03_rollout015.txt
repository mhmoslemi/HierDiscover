    n = a.shape[0]
    # Build sparse A_ub using diags
    row_count = 2*n - 1
    A_ub = scipy.sparse.diags(a, offsets=range(n), shape=(row_count, n), format='csr')

"""
Propose a low‑score non‑negative step function.
The search consists of:

1.  Build a diverse pool of modest‑size candidates (n≈6‑8 k).
2.  Run many short projected‑gradient (soft‑max) optimisations.
3.  Down‑scale the best candidate to a small size (n≈2000) and
    perform a few linear‑programming refinement steps (LP‑refine).
4.  Upscale to a large final length (up to 20 k if time permits)
    and run a long high‑β gradient polish.
5.  A targeted permutation annealing, followed by a tiny
    deterministic mass‑transfer pass.
All updates keep the ℓ¹‑norm fixed (projected onto the simplex)
so the objective 2·n·max(a*a)/(∑a)² reduces to max(a*a).

The routine respects the 1000 s budget, prints progress and finally
returns the best list of heights found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for all sizes)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (with Σa normalized it is just max)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p), renormalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam*i), renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _linear_decrease_sequence(n: int, target_sum: float) -> np.ndarray:
    """Linear fall from 1 to 0, renormalised."""
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _random_dirichlet_sequence(n: int, target_sum: float, rng) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet with all parameters = 1)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest‑left, second‑largest‑right, third‑largest‑next‑left, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Same as _alternating_order but start placing the biggest on the right."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p   (⋆ = convolution)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max approximation.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (local search)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves from a donor (large contribution to the current peak)
    towards a low‑contribution receiver. Moves are accepted with a
    Metropolis rule whose temperature decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]           # contribution of each index to the peak

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0.0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # rollback
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation search (simulated annealing)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style search that swaps indices which
    contribute heavily to the current maximal convolution entry.
    Only re‑orders the values – the ℓ¹‑sum stays unchanged.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP‑refine)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               deadline: float,
               target_sum: float,
               n_iter: int = 5,
               t: float = 0.15) -> np.ndarray:
    """
    Given a sequence a, iteratively solve
        maximise   Σ b_j
        subject to b_j ≥ 0,  (a * b)_k ≤ M   for all k,
        where M = max(a*a).
    The solution b is rescaled to the required ℓ¹‑norm and mixed
    with a (convex combination with weight t).  After a few
    iterations the maximal convolution typically drops.
    """
    a = a0.copy()
    n = a.shape[0]
    row_cnt = 2 * n - 1

    # Build sparse convolution matrix for the current a:
    # A_ub[i, j] = a[i-j] if i >= j else 0
    # This can be expressed as a set of lower‑diagonals.
    offsets = np.arange(n, dtype=int)          # 0 … n‑1
    # we will rebuild A_ub inside the loop because a changes.
    for it in range(n_iter):
        if time.time() >= deadline:
            break
        M = _max_convolution_fft(a)
        # Sparse matrix with constant diagonals = a[diag_offset]
        A_ub = scipy.sparse.diags(a, offsets, shape=(row_cnt, n), format='csr')
        c = -np.ones(n)                        # maximise Σb  → minimise -Σb
        b_ub = np.full(row_cnt, M, dtype=np.float64)

        res = scipy.optimize.linprog(c,
                                     A_ub=A_ub,
                                     b_ub=b_ub,
                                     bounds=[(0, None)] * n,
                                     method='highs',
                                     options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.2f}] LP iteration {it} failed: {res.message}")
            break
        b0 = res.x
        sum_b0 = b0.sum()
        if sum_b0 <= 0.0:
            break
        b = b0 * (target_sum / sum_b0)
        # convex update
        a = (1.0 - t) * a + t * b
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The function respects the 1000 s budget and returns the best sequence
    found as a plain Python list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin for the harness

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Choose an exploration length (moderate, fast enough for many runs)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seed sequences
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.30, 0.80, 11):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 9):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # linear decreasing
    init_pool.append(_linear_decrease_sequence(n_expl, target_expl))

    # random Dirichlet‑like
    for _ in range(8):
        init_pool.append(_random_dirichlet_sequence(n_expl, target_expl, rng))

    # incorporate any previously saved best
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.006, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Exploration: many short gradient‑descent runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate roughly 30 % of time to this stage
    exploration_deadline = start_time + 0.30 * budget_s
    while time.time() < min(exploration_deadline, deadline - 12.0):
        # pick a seed (occasionally perturb the current best)
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_expl * 0.003, size=n_expl)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 20.0, deadline - 12.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_expl,
            max_beta=8e5,
            step_factor=0.6,
            gamma=0.985,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Small‑scale LP‑refine (on a compact representation)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        # downscale to a size that keeps the LP tractable
        n_lp = 2000
        seq_lp = _resize_sequence(best_seq, n_lp)
        target_lp = np.sqrt(2.0 * n_lp)
        lp_dead = time.time() + min(10.0, deadline - time.time() - 5.0)
        seq_lp = _lp_refine(seq_lp, lp_dead, target_lp, n_iter=5, t=0.12)
        # upscale back
        best_seq = _resize_sequence(seq_lp, n_expl)
        best_seq = _project_to_simplex(best_seq, target_expl)
        best_score = _score(best_seq)
        print(f"[{time.time():.2f}] After LP‑refine best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscale to final problem size (as large as time permits)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    # decide final length: larger gives more freedom but costs a bit more time
    if remaining > 200.0 and budget_s >= 800:
        n_final = 18000
    elif remaining > 120.0:
        n_final = 14000
    elif remaining > 60.0:
        n_final = 10000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a couple of systematic reorderings and keep the best
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final high‑β gradient descent (long polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        final_dead = deadline - 5.0
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            final_dead,
            target_final,
            max_beta=3e6,
            step_factor=0.45,
            gamma=0.99,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] Final polishing best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Permutation‑only annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        perm_dead = time.time() + min(10.0, deadline - time.time() - 2.0)
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.12,
            end_temp=1e-5,
            top_frac=0.03,
            low_frac=0.03,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny deterministic mass‑transfer pass (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(6.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.25,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Post‑polish MT best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros (compact representation)
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 10. Report final evaluation (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # store for possible reuse by a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
