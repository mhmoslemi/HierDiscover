def _generate_candidates(n, target_sum, rng):
    candidates = []
    # warm start
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=np.float64)
        if prev.shape[0]!=n:

        candidates.append(prev)

    # power law
    for p in np.linspace(0.3, 0.95, 31):
        seq = _power_law(n,p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        # jitter
        jitter = rng.normal(scale=target_sum*0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # power law with offset
    for offset in [0.0,0.5,1.0]:

            candidates.append(seq)

    for lam in np.linspace(5e-4, 0.02, 30):
        seq = _exponential(n, lam)

    # sinus
    for p in np.linspace(0.2,2.0,15):

"""
Step‑function optimizer for the self‑convolution upper‑bound problem.

The search proceeds in four stages:

1. **Problem size** – choose a length `n` according to the remaining time.
2. **Rich candidate pool** – generate many parametric families
   (power‑law, exponential, power·exp, sinusoidal, triangular and a few
   random Dirichlet samples).  Each candidate is projected onto the simplex
   `{x≥0, Σx = sqrt(2 n)}` and its maximal self‑convolution (computed by FFT)
   is measured.  The best `top_k` candidates are retained.
3. **Refinement** – for each of the top candidates run a
   *soft‑max gradient descent* with momentum and an annealed inverse‑temperature
   (`β`).  The method works directly on the smooth approximation of
   `max(conv)`.  Afterwards a short *sub‑gradient descent* on the true
   max‑convolution is applied.
4. **Deterministic mass‑transfer** – a greedy simulated‑annealing step moves
   mass from indices that heavily contribute to the current peak to low‑contributing
   indices, often shaving off the last few hundredths.

All FFT work is `O(n log n)`.  Time checks guard the 1000 s budget and the
algorithm stops early if the target bound `1.503` is reached.  The final
sequence is trimmed, stored in the global variable `height_sequence_1`
(for warm‑starts) and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _power_law(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64) + offset
    return i ** (-p)


def _exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


def _power_exp(n: int, p: float, lam: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (-p)) * np.exp(-lam * (i - 1))


def _sin_power(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _triangular(n: int, w_frac: float) -> np.ndarray:
    """Symmetric triangle: height 1 at the centre, zero outside width w_frac·n."""
    mid = (n - 1) / 2.0
    w = w_frac * n
    i = np.arange(n, dtype=np.float64)
    shape = np.maximum(0.0, 1.0 - np.abs(i - mid) / w)
    return shape


def _piecewise_constant_power(n: int, frac: float, p: float) -> np.ndarray:
    """
    Constant block on the left of length frac·n, power‑law tail afterwards.
    """
    block = int(frac * n)
    tail_len = n - block
    tail = _power_law(tail_len, p)
    seq = np.concatenate([np.ones(block, dtype=np.float64), tail])
    return seq


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approx of max(conv))
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # grad = 2 * (rev(a) ⋆ p) restricted to the central n entries
    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum GD on the soft‑max approximation
# ----------------------------------------------------------------------
def _softmax_gradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_beta: float = 5e5,
    step_factor: float = 0.6,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Momentum gradient descent on a smooth max‑convolution surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    step_factor: float = 0.55,
    gamma: float = 0.9,
    jitter_interval: int = 400,
    jitter_scale: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

# Deterministic mass‑transfer (local annealing)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.6,
    end_temp: float = 1e-5,
    low_frac: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer that reduces the current peak."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, kmax‑i)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < np.exp(
            -(new_score - cur_score) / max(temp, 1e-12)
        ):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}"
                )
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score

# Main entry point

def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximum
    entry of its self‑convolution.  Returns a list ready for `evaluate_sequence`.
    """
    start_time = time.time()
    # small safety margin to avoid overrunning the sandbox
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose problem size according to the available time budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 25000
    elif budget_s >= 600:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    elif budget_s >= 250:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Build and evaluate a large pool of candidates
    # --------------------------------------------------------------
    candidates = _generate_candidates(n, target_sum, rng)

    scored = []
    for seq in candidates:
        if time.time() >= deadline - 2.0:
            break
        val = _max_convolution_fft(seq)
        scored.append((val, seq))

    if not scored:
        # fallback – uniform distribution (should never happen)
        uniform = np.full(n, target_sum / n, dtype=np.float64)
        return uniform.tolist()

    scored.sort(key=lambda x: x[0])
    best_seq, best_score = scored[0][1].copy(), scored[0][0]
    print(f"[{time.time():.2f}] Initial best score = {best_score:.6f}")

    # early exit if we already beat the target
    if best_score < 1.503:
        final_arr = best_seq
        nz = np.where(final_arr > 1e-12)[0]
        if nz.size:
            final_arr = final_arr[: nz[-1] + 1]
        final_seq = final_arr.tolist()
        globals()["height_sequence_1"] = final_seq
        return final_seq

    # --------------------------------------------------------------
    # Phase A – soft‑max GD on the most promising candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    top_k = min(6, len(scored))
    if remaining > 8.0:
        gd_budget = remaining * 0.35
        per_candidate = gd_budget / top_k
        for i in range(top_k):
            if time.time() >= deadline - 2.0:
                break
            seq0 = scored[i][1]
            sub_dead = time.time() + min(per_candidate, deadline - time.time() - 1.0)
            a_opt, val_opt = _softmax_gradient_descent(
                seq0,
                target_sum,
                sub_dead,
                max_beta=5e5,
                step_factor=0.6,
                gamma=0.996,
            )
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] GD improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_dead = time.time() + remaining * 0.25
        a_sg, val_sg = _subgradient_descent(
            best_seq, target_sum, sub_dead, step_factor=0.55, gamma=0.95
        )
        if val_sg < best_score:
            best_score, best_seq = val_sg, a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – deterministic mass‑transfer (local annealing)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        mt_dead = time.time() + remaining * 0.30
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_sum,
            start_temp=0.6,
            end_temp=1e-5,
            low_frac=0.04,
        )
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – final polishing with a higher‑β soft‑max GD
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_fin, val_fin = _softmax_gradient_descent(
            best_seq,
            target_sum,
            time.time() + remaining,
            max_beta=8e5,
            step_factor=0.5,
            gamma=0.998,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑process: trim trailing zeros, evaluate, store globally
    # --------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]

    final_seq = final_arr.tolist()
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
