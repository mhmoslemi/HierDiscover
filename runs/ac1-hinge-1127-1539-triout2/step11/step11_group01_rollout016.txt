"""
Propose a step‑function (non‑negative sequence) with a low max‑convolution value.

Algorithm
---------
1. Choose a problem size `n` (larger `n` gives more freedom).
2. Initialise with a parametric family
      a_i ∝ (i+1)^(-p)·exp(-q·i/n)               (p,q ≥ 0)
   Scan a dense grid for (p,q) and keep the best candidate.
   Then perform a long random exploration of the (p,q) space with
   small jitters, always re‑projecting onto the simplex Σa = √(2 n).
3. Sub‑gradient descent on the true non‑smooth objective
      f(a) = max_k (a*a)[k] .
   The sub‑gradient is obtained analytically (sum of shifted copies of `a`
   over all indices attaining the current maximum).  After each step we
   project back onto the simplex.
4. Deterministic “mass‑transfer” refinements (exact version) that pull
   mass from a donor index with large contribution to the peak towards a
   receiver with small contribution – the amount is chosen so that the
   current peak is lowered to the second‑largest convolution entry.
   This step is extremely effective for the remaining few hundredths.
5. Occasionally perform cheap random mass‑transfer moves to escape tiny
   plateaus.
6. A final short Adam‑style projected gradient descent on a soft‑max
   surrogate (large β) is used for polishing.
7. Trim trailing near‑zero entries and return the sequence as a plain
   Python list.

The routine respects a total time budget (default 1000 s) and prints
progress information.  All helper functions are at module level so that
they survive the competition harness environment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective – the maximum convolution entry (the competition metric)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))   # reduces to max_b when Σa = √(2n)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families used in the pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalized to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-q·i/n), normalized to target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max of the max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max weights, length 2n‑1

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # derivative w.r.t. a
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            # small jitter to avoid stagnation
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (improved version)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick a representative peak
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner coefficients a_{k‑i}
        partner = a[k - idx]               # length = hi‑lo+1

        # donor must have positive mass
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver: index with minimal partner value, different from donor
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        # amount needed to bring the peak down to the second‑largest value
        delta_needed = (max_val - second_max) / denom

        # stay within safe fraction of donor's mass
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Cheap random donor‑receiver moves, with periodic simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent directly on the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # accumulate sub‑gradient (sum of shifted copies of a)
        grad = np.zeros_like(a)
        for k in idx_max:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]           # factor 2 is exact for i≠k‑i

        # step size: diminishing schedule
        it += 1
        step_norm = np.linalg.norm(grad) + 1e-12
        eta = 0.25 * target_sum / np.sqrt(it)       # tunable magnitude
        step = eta * grad / step_norm

        a -= step
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        # occasional jitter to hop out of tiny plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal max‑convolution.
    Returns a plain Python list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem size: larger = more flexibility, modest extra cost
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)

    # --------------------------------------------------------------
    # Phase 0 – initialise from previous run (if present)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to new length
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            seq = np.interp(new_x, old_x, prev)
        else:
            seq = prev.copy()
        seq = _project_to_simplex(seq, target_sum)
        val = _score(seq)
        best_seq = seq
        best_score = val
        print(f"[{time.time():.2f}] Init from previous: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – dense parametric grid search (power‑law + exponential tail)
    # --------------------------------------------------------------
    p_grid = np.linspace(0.35, 0.78, 45)         # finer than before
    q_grid = np.linspace(0.0, 6.0, 45)

    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    print(f"[{time.time():.2f}] After grid search: best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – random exploration of (p,q) with jitter
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    # allocate up to ~20 % of the whole budget for this phase
    random_deadline = time.time() + (deadline - time.time()) * 0.20
    while time.time() < random_deadline:
        p = rng.uniform(0.33, 0.85)
        q = rng.uniform(0.0, 6.0)
        seq = _exp_power_law_sequence(n, p, q, target_sum)
        # add a tiny random perturbation
        jitter = rng.normal(scale=target_sum * 0.001, size=n)
        seq = np.maximum(seq + jitter, 0.0)
        seq = _project_to_simplex(seq, target_sum)

        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Random (p,q) improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent on the true objective
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        subgrad_deadline = time.time() + (deadline - time.time()) * 0.25
        a_sgd, val_sgd = _subgradient_descent(best_seq, subgrad_deadline, target_sum)
        if val_sgd < best_score:
            best_score = val_sgd
            best_seq = a_sgd.copy()
            print(f"[{time.time():.2f}] Sub‑gradient descent improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – deterministic mass‑transfer refinements + occasional random moves
    # --------------------------------------------------------------
    while time.time() < deadline - 8.0:
        # deterministic exact mass‑transfer
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq, deadline, target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT improved best={best_score:.6f}")

        # short random exploration to escape possible plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best={best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # --------------------------------------------------------------
    # Phase 5 – final polishing with a high‑β Adam optimizer
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: strip trailing near‑zeros and return as list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluate_sequence score = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.2f}] DONE – best score ≈ {best_score:.6f}, length {len(final_seq)}")
    return final_seq
```

This implementation follows the multi‑phase strategy described above, adds a dedicated sub‑gradient descent on the true max‑convolution objective, and expands the parametric pre‑search with a finer grid and random exploration.  It respects the 1000 s time budget and returns the best non‑negative step‑function it discovers.
