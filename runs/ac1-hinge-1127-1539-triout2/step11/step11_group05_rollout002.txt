def _subgradient_descent_line_search(a0, target_sum, deadline):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        # gradient: g = 2 * a_rev where a_rev[i] = a[kmax-i]
        grad = np.zeros_like(a)
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        i_range = np.arange(lo, hi+1)
        grad[i_range] = 2.0 * a[kmax - i_range]
        # compute step size
        norm = np.linalg.norm(grad)
        if norm == 0:
            break
        # initial step size based on target_sum
        # try decreasing step sizes
        step = target_sum * 0.2 / norm
        # backtracking: try up to 5 reductions
        for _ in range(5):
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new, target_sum)
            val_new = _max_convolution_fft(a_new)
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                a = a_new
                break
            else:
                step *= 0.5
        else:
            # could not improve, add jitter
            jitter = rng.normal(scale=target_sum * 1e-3, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            # continue
        # termination condition maybe 
    return best_a, best_val

def _mass_transfer(a, target_sum, deadline):
    n = len(a)
    cur_val = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_val
    # set initial epsilon proportional to total sum
    init_eps = target_sum * 0.01
    while time.time() < deadline:
        # compute current conv, kmax
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n-1))
        hi = min(kmax, n-1)
        i_idx = np.arange(lo, hi+1)
        partner = a[kmax - i_idx]
        # donor: i with max partner value and a[i] > 0
        donor_candidates = i_idx[a[i_idx] > 0]
        if donor_candidates.size == 0:
            break
        donor = donor_candidates[np.argmax(partner[donor_candidates - lo])]
        # receiver: i with min partner value (could be zero)
        receiver = i_idx[np.argmin(partner)]
        if donor == receiver:
            # maybe random pick different
            continue
        # compute max possible transfer: limited by donor amount and some fraction
        max_transfer = min(a[donor] * 0.5, init_eps * (1 - (time.time() - start_time) / (deadline - start_time)))
        if max_transfer <= 0:
            break
        eps = max_transfer * np.random.rand()
        a[donor] -= eps
        a[receiver] += eps
        # ensure non-neg maybe after rounding
        # projection not needed as sum unchanged and both remain non-negative
        new_val = _max_convolution_fft(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            a[donor] += eps
            a[receiver] -= eps
        # optionally jitter occasionally
    return best_a, best_val

"""
propose_candidate – Heuristic search for a non‑negative step function
===================================================================

Goal
----
Minimise the evaluation function

    2·n·max(a*a) / (sum(a)**2)

which is invariant under scaling, so we fix Σa = sqrt(2·n) and minimise
max(a*a).  The algorithm builds a diverse pool of analytic shapes,
projects them onto the simplex, keeps the best few and refines them by

1. projected sub‑gradient descent on the true ℓ∞‑norm of the self‑convolution,
2. a fast “mass‑transfer” step that moves mass from indices that most
   contribute to the current peak to indices that contribute least,
3. a final polishing pass.

All operations are O(n log n) via FFT, and the routine respects a 1000 s
budget (the caller supplies the time limit).  The best sequence found is
stored globally as ``height_sequence_1`` for warm‑starts in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Unnormalised (i+1)^(-p)."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Unnormalised e^{-lam·i}."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    """Unnormalised (sin(π·(i+0.5)/n))^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent with back‑tracking line search
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(a*a)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        # sub‑gradient g[i] = 2*a[kmax-i] on the active window, else 0
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[kmax - idx]

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        # initial step size (heuristic)
        step = target_sum * 0.3 / norm

        improved = False
        # simple back‑tracking
        for _ in range(5):
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new, target_sum)
            val_new = _max_convolution_fft(a_new)
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            # occasional jitter to escape plateaus
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = _project_to_simplex(a + jitter, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (steepest descent on the active peak)
# ----------------------------------------------------------------------
def _mass_transfer_opt(a0: np.ndarray,
                       target_sum: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from an index that most
    contributes to the current maximal convolution entry to an index
    that contributes least.  Moves are accepted only if they do not
    increase the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_val
    start = time.time()
    rng = np.random.default_rng()

    # initial transfer magnitude (as a fraction of the total ℓ1‑norm)
    init_eps = target_sum * 0.02

    while time.time() < deadline:
        # compute current peak and the indices that affect it
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = a[kmax - idx]                     # derivative components

        # donor: index with largest partner value among those that still have mass
        cand = idx[a[idx] > 0.0]
        if cand.size == 0:
            break
        donor = cand[np.argmax(partner[cand - lo])]
        # receiver: index with smallest partner value (could be zero)
        receiver = idx[np.argmin(partner)]

        if donor == receiver:
            # try a random different receiver
            receiver = rng.choice(idx[idx != donor])
            if receiver == donor:
                continue

        # decreasing transfer budget over time
        progress = (time.time() - start) / (deadline - start)
        eps_max = init_eps * (1.0 - progress)
        eps_max = max(eps_max, target_sum * 1e-5)

        # bound by available mass at donor
        eps_max = min(eps_max, a[donor] * 0.5)
        if eps_max <= 0.0:
            break

        eps = eps_max * rng.random()
        if eps <= 0.0:
            continue

        # apply move (sum unchanged, non‑negativity guaranteed by epsilon bound)
        a[donor] -= eps
        a[receiver] += eps

        new_val = _max_convolution_fft(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # reject move
            a[donor] += eps
            a[receiver] -= eps
            # occasional jitter to shake out of local minima
            if rng.random() < 0.001:
                jitter = rng.normal(scale=target_sum * 2e-4, size=n)
                a = _project_to_simplex(a + jitter, target_sum)

    return best_a, best_val

def _build_candidate_pool(n: int, target_sum: float, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse set of feasible sequences."""
    cand = []

    # Warm‑start if available

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The routine respects a time budget (default 1000 s) and
    returns a Python list of floats ready for ``evaluate_sequence``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = np.array(globals()["height_sequence_1"], dtype=np.float64).shape[0]
    else:
        if budget_s >= 800:
            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build candidate pool
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = _build_candidate_pool(n, target_sum, rng)

    # --------------------------------------------------------------
    # Fast evaluation of the pool
    # --------------------------------------------------------------
    scored = []
    for seq in candidates:
        val = _max_convolution_fft(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])

    # Keep a handful of the best for intensive refinement
    top_k = min(8, len(scored))
    best_seq = scored[0][1].copy()
    best_val = scored[0][0]
    print(f"[{time.time():.2f}] Initial best max‑conv = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Sub‑gradient descent on the top candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        per_candidate = remaining * 0.35 / top_k
        for i in range(top_k):
            if time.time() >= deadline - 0.5:
                break
            seq0 = scored[i][1]
            sg_dead = time.time() + per_candidate
            a_opt, val_opt = _subgradient_descent(seq0, target_sum, sg_dead)
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] Sub‑grad improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Deterministic mass‑transfer refinement
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        mt_dead = time.time() + remaining * 0.40
        a_mt, val_mt = _mass_transfer_opt(best_seq, target_sum, mt_dead)
        if val_mt < best_val:
            best_val, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Final polishing (short sub‑gradient pass)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        fin_dead = time.time() + remaining * 0.25
        a_fin, val_fin = _subgradient_descent(best_seq, target_sum, fin_dead)
        if val_fin < best_val:
            best_val, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Post‑process, evaluate, and store globally
    # --------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=float)

    # Trim trailing near‑zero entries for a compact representation
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]

    final_seq = final_arr.tolist()

    # Report the true objective using the provided evaluator
    try:
        true_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {true_val:.6f}")
    except Exception:
        pass

    # expose for warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
