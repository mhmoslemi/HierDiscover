"""
Search for a non‑negative step function that minimises
``evaluate_sequence``.

The method combines three ingredients:

1.  A fast Adam optimiser that works on a smooth approximation of the
    max‑convolution (log‑sum‑exp) and on the exact sub‑gradient of the
    true max.  This gives a good starting shape.

2.  A linear‑programming refinement that is described in the reference
    paper: for a fixed vector ``f`` we maximise ∑gᵢ subject to
    ``f∗g ≤ M`` (component‑wise) where ``M`` is the current peak of
    ``f∗f``.  The LP pushes mass into entries that are not contributing
    to the peak, flattening the convolution profile.

3.  A short random‑tweak phase and a final Adam polishing.  The whole
    search respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found.
"""

from __future__ import annotations
import time
import math
import numpy as np
import scipy.sparse as sp

# ``cvxpy`` may be unavailable on some runners – the code will still work
# without the LP refinement.
try:
    import cvxpy as cp
except Exception:               # pragma: no cover
    cp = None


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a∗a using an FFT (fast, O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers (simple families – the exponential family works well)
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n).
    The exponential family produces a decreasing shape which is a good
    starting point for the LP step.
    """
    target = math.sqrt(2.0 * n)
    lam = rng.exponential(scale=0.015) + 0.001   # avoid lam=0
    a = np.exp(-lam * np.arange(n))
    # add a tiny jitter so the LP does not get a perfectly rank‑deficient f
    a += rng.normal(scale=target * 1e-4, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of (1/β)·log Σ exp(β·b) where b = a∗a."""
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """Projected Adam on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """Projected Adam where the gradient is the exact sub‑gradient of max(a∗a)."""
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.20 * target_sum
    t = 0

    while time.time() < deadline:
        t += 1
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:       # safety
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP step” from the paper)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve  max ∑gᵢ   s.t.   f∗g ≤ M   (component‑wise) , g ≥ 0.
    Return the scaled solution g (scaled to Σg = target_sum).
    """
    if cp is None:                                 # pragma: no cover
        return None

    n = f.size

    # Build the (2n‑1) × n convolution matrix of f (sparse).
    # row indices are i + j for i,j∈[0,n)
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    col_idx = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    # Linear programme
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, verbose=False)
    except Exception:                              # pragma: no cover
        return None

    if g.value is None:
        return None

    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s <= 0.0:
        return None
    # Scale to the usual ℓ¹‑simplex
    return g0 * (target_sum / s)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step function that minimizes
    ``evaluate_sequence``.  The search consists of:

        * a quick Adam phase on the smooth‑max surrogate,
        * a short exact‑max Adam phase,
        * a few LP‑based refinement steps,
        * a final polishing Adam run and light random tweaks.

    The routine respects the supplied wall‑clock budget and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # 1)  Seed with any previously discovered sequence (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            target0 = math.sqrt(2.0 * prev.size)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)

                  f"(n={prev.size}) score={best_score:.6f}")
        except Exception as e:  # pragma: no cover
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Length schedule – we start modest and try a larger size.
    # ------------------------------------------------------------------
    schedule = [2000, 3000]               # 3000 gives a bit more freedom

    for n in schedule:
        if time.time() >= deadline:

        target = math.sqrt(2.0 * n)

        # initialise (upsample previous best if possible)
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng)

        a = _project_to_simplex_fast(a, target)

        # ----------------------------------------------------------------
        # 2a) Adam on the smooth surrogate (quick exploration)
        # ----------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 15.0:

        phase_deadline = time.time() + remaining * 0.30
        a = _adam_optimize(a, target, phase_deadline)

        # ----------------------------------------------------------------
        # 2b) Exact‑max Adam (sub‑gradient) – brings the peak down faster
        # ----------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 10.0:

        phase_deadline = time.time() + remaining * 0.20
        a = _adam_optimize_submax(a, target, phase_deadline)

        # ----------------------------------------------------------------
        # 2c) LP‑based refinement (the core new ingredient)
        # ----------------------------------------------------------------

                if time.time() >= deadline - 5.0:

                g = _lp_refine(a, M, target)
                if g is None:

                a = (1.0 - t_lp) * a + t_lp * g
                a = _project_to_simplex_fast(a, target)

                if cur_score < best_score - 1e-12:
                    best_seq = a.copy()
                    best_score = cur_score

        # ----------------------------------------------------------------

        # ----------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            phase_deadline = time.time() + min(remaining * 0.15, 8.0)
            a = _adam_optimize_submax(a, target, phase_deadline)

        # ----------------------------------------------------------------
        # 2e) Light random tweaks (simulated‑annealing style)
        # ----------------------------------------------------------------
        while time.time() + 1.5 < deadline:

            i = rng.integers(0, n)
            j = rng.integers(0, n)

            delta = cand[i] * 0.04
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            cur_score = _max_convolution_fft(cand)
            if cur_score < best_score - 1e-12:
                a = cand
                best_seq = cand.copy()
                best_score = cur_score
                print(f"[{time.time():.1f}] Random tweak n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------

        final_deadline = deadline - 0.5
        best_seq = _adam_optimize_submax(best_seq, target, final_deadline)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5) Report final evaluation (optional – does not affect the score)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:                     # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
