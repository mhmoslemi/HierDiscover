"""
Improved search for a step‑function with minimal max‑convolution.

The algorithm builds on the previous approach and adds several
enhancements:

* a broader parametric pre‑search (p ∈ [0.1,1.2], q ∈ [0,10],
  plus a sinusoidal family);
* more aggressive deterministic mass‑transfer (move the full amount
  needed to drop the current peak to the second‑largest value);
* a second‑level projected sub‑gradient descent that works directly
  on the (non‑smooth) max‑convolution using the exact sub‑gradient;
* a simple simulated‑annealing style random walk to escape plateaus;
* a final polishing step with a very large soft‑max β.

All helper functions are defined at top level; the entry point
required by the harness is `propose_candidate`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Evaluation function – identical to evaluate_sequence for a
    normalized sum (the numerator becomes the max‑convolution value)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families used in the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sine_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n). Non‑negative on (0,π)."""
    a = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    a[a < 0] = 0.0
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple short Adam restarts, each starting from a jittered point."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (more aggressive)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_aggressive(a0: np.ndarray,
                                            deadline: float,
                                            target_sum: float,
                                            tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    For each current peak `k`, move mass from the donor with largest
    partner a_{k‑i} to the receiver with smallest partner.
    The full amount required to bring the peak down to the second‑largest
    value (or the donor's remaining mass) is transferred.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner values a_{k-i}
        partner = a[k - idx]

        # donors are indices where a > 0
        donor_mask = a[idx] > 0.0
        if not np.any(donor_mask):
            break

        # choose donor with largest partner
        donor_sel = np.argmax(partner * donor_mask)
        donor = idx[donor_sel]
        partner_d = partner[donor_sel]

        # choose receiver with smallest partner (different from donor)
        # use argsort to efficiently find minimum
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                partner_r = partner[r_loc]
                break
        if receiver is None:
            break

        denom = 2.0 * (partner_d - partner_r)
        if denom <= 0.0:
            # cannot reduce this peak with this pair
            break

        # amount required to bring peak down to second_max
        delta_needed = (max_val - second_max) / denom

        # move at most the donor's remaining mass
        delta = min(delta_needed, a[donor])
        if delta <= 0.0:
            break

        a[donor] -= delta
        a[receiver] += delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Aggressive DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Very simple random walk on the simplex, accepting improving moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _projected_subgradient_descent(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   init_step: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for the exact max‑convolution.
    At each iteration we compute the (possibly multiple) maxima,
    form the sub‑gradient (average of the corresponding linear pieces)
    and take a decreasing step.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    step = init_step

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        max_idx = np.where(conv >= max_val - 1e-12)[0]

        # sub‑gradient = average of 2*a_{k-i} over all maximal k
        grad = np.zeros_like(a)
        for k in max_idx:
            lo = max(0, k - (len(a) - 1))
            hi = min(k, len(a) - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]
        grad = grad / max(1, len(max_idx))

        # simple diminishing step size
        step = init_step / math.sqrt(it + 1)

        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid getting stuck on flat plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric pre‑search (grid + random jitter)
# ----------------------------------------------------------------------


def _parametric_presearch(n: int,
                          target_sum: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Explore a dense grid of (p,q) plus a sinusoidal family."""
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # grid for p and q
    p_grid = np.linspace(0.10, 1.20, 31)          # 31 values
    q_grid = np.linspace(0.0, 10.0, 31)          # 31 values

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) sine family
    seq = _sine_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (d) jittered perturbations around the current best
    if best_seq is not None:
        for _ in range(12):
            jitter = rng.normal(scale=target_sum * 0.004, size=n)
            cand = best_seq + jitter
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, target_sum)
            val = _score(cand)
            if val < best_score:
                best_score = val
                best_seq = cand.copy()

    # (e) use previous global solution if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _score(prev)
        if val < best_score:
            best_score = val
            best_seq = prev.copy()

    print(f"[{time.time():.2f}] Pre‑search best score = {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal max‑convolution.
    The routine respects the given time budget and returns the best sequence.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Choose a problem size. 12000 works well, but we allow a modest increase
    # if the budget is large enough.
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)   # Σa is fixed to this value (scale‑invariant)

    # ------------------------------------------------------------------
    # Phase 0 – parametric pre‑search
    # ------------------------------------------------------------------
    best_seq, best_score = _parametric_presearch(n, target_sum, deadline)

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        # jittered start point
        rng = np.random.default_rng()
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Aggressive deterministic mass‑transfer
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_aggressive(best_seq,
                                                                 deadline,
                                                                 target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Aggressive DMT improved best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Sub‑gradient descent on the exact max‑convolution
        # ------------------------------------------------------------------
        sub_deadline = time.time() + 6.0
        a_sub, val_sub = _projected_subgradient_descent(best_seq,
                                                        sub_deadline,
                                                        target_sum)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Short global random walk to escape possible plateaus
        # ------------------------------------------------------------------
        rnd_deadline = time.time() + 4.0
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        rnd_deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Final polishing with a very large β (soft‑max)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
