"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a))**2

under the evaluator’s clipping (0 ≤ a_i ≤ 1000).  
The algorithm is deliberately simple and fast:

1. Choose a problem size n depending on the time budget.
2. Normalise the ℓ¹‑norm to sqrt(2·n) – the objective is homogeneous,
   so fixing the sum does not affect the optimum (unless clipping occurs).
3. Exhaustively scan two parametric families that are known to be good:
      * exponential a_i = exp(-lam·i)
      * power‑law a_i = (i+1)^(-p)
   The scan uses a tiny FFT‑based evaluation, therefore thousands of
   candidates can be examined in a few seconds.
4. For every candidate try a few deterministic reorderings
   (descending, alternating) – they keep the ℓ¹‑norm unchanged but
   can lower the peak of the convolution.
5. Take the best candidate and polish it with a short projected‑gradient
   descent on a smooth soft‑max approximation of the max‑convolution.
6. A final light‑weight simulated‑annealing permutation search fine‑tunes
   the ordering.
7. Trim trailing zeros and return the list.  The whole pipeline respects
   the 1000 s budget.

The code is self‑contained; helper functions are defined at module level
so that the competition runner can import ``propose_candidate`` directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the (linear) self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())    # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Compute the evaluator’s objective:
        2·n·max(conv(a,a)) / (sum(a))**2
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate sequences
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam·i) normalised to have ℓ¹‑norm = target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p) normalised to have ℓ¹‑norm = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Re‑order by placing the largest value at the leftmost position,
    the second largest at the rightmost, the third largest next to the left,
    etc.
    """
    sorted_vals = np.sort(seq)[::-1]          # descending
    n = seq.shape[0]
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    go_left = True
    for v in sorted_vals:
        if go_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        go_left = not go_left
    return out


# ----------------------------------------------------------------------
# Simple optimisation primitives
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Smooth approximation of the gradient of max(conv(a,a)).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    probs = exps / exps.sum()

    a_rev = a[::-1]
    R = np.fft.rfft(a_rev, n=size)
    P = np.fft.rfft(probs, n=size)
    rev_grad = np.fft.irfft(R * P, n=size)[: 2 * n - 1]
    grad = 2.0 * rev_grad[n - 1: n - 1 + n]
    return grad, conv


def _projected_gradient_descent(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                max_beta: float = 5e6,
                                base_step: float = 0.5,
                                gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max surrogate.
    Runs until the deadline and returns the best vector seen.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * base_step
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _softmax_gradient(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            # print progress (optional)
            # print(f"[{time.time():.2f}] GD it={it} β={beta:.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best, best_val


def _permutation_search(a0: np.ndarray,
                        deadline: float,
                        start_temp: float = 0.12,
                        end_temp: float = 1e-5,
                        top_frac: float = 0.03,
                        low_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the permutation of the entries.
    Only re‑orders; the ℓ¹‑norm stays unchanged.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = np.convolve(a, a)
        kmax = int(np.argmax(conv))

        # contribution of each position to the peak entry
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))
        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)
        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]
        val = _score(a)

        # temperature schedule (linear in log‑time)
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if val < best_val:
            best_val = val
            best = a.copy()
            # print(f"[{time.time():.2f}] Perm best={best_val:.6f}")

        # Metropolis acceptance
        if val < best_val or rng.random() < np.exp(-(val - best_val) / max(temp, 1e-12)):
            # keep swap
            pass
        else:
            # revert
            a[donor], a[receiver] = a[receiver], a[donor]

    return best, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low score.
    The method scans exponential and power‑law families, tries a few
    deterministic reorderings, then refines the best candidate with
    gradient descent and a short permutation simulated‑annealing.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 10000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Exponential family sweep (lam from ~1/n to a few 10⁻³)
    # ------------------------------------------------------------------
    lam_lower = max(1.5 / n, 5e-5)                # ensure support < n
    lam_vals = np.logspace(math.log10(lam_lower), math.log10(5e-3), 40)
    for lam in lam_vals:
        seq = _exponential_sequence(n, float(lam), target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
        # try alternating order (often improves the peak)
        alt = _alternating_order(seq)
        val_alt = _score(alt)
        if val_alt < best_val:
            best_val, best_seq = val_alt, alt.copy()

    # ------------------------------------------------------------------
    # 2) Power‑law family sweep
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.10, 0.90, 41)
    for p in p_vals:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
        # descending order is already the shape; try its reverse
        rev = seq[::-1]
        val_rev = _score(rev)
        if val_rev < best_val:
            best_val, best_seq = val_rev, rev.copy()
        # alternating version
        alt = _alternating_order(seq)
        val_alt = _score(alt)
        if val_alt < best_val:
            best_val, best_seq = val_alt, alt.copy()

    # ------------------------------------------------------------------
    # 3) Random heavy‑tailed attempts (useful when time remains)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline - 5.0:
        # log‑normal heavy tail, then normalise
        a = np.exp(rng.normal(scale=0.6, size=n))
        a = a / a.sum() * target_sum
        a = np.sort(a)[::-1]                     # keep decreasing order
        val = _score(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()

    # ------------------------------------------------------------------
    # 4) Gradient‑descent polish (soft‑max surrogate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        a_opt, v_opt = _projected_gradient_descent(
            best_seq,
            deadline - 1.0,
            target_sum,
            max_beta=5e6,
            base_step=0.6,
            gamma=0.995,
        )
        if v_opt < best_val:
            best_val, best_seq = v_opt, a_opt.copy()

    # ------------------------------------------------------------------
    # 5) Permutation simulated‑annealing (order‑only)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_perm, v_perm = _permutation_search(
            best_seq,
            deadline - 0.5,
            start_temp=0.15,
            end_temp=1e-5,
            top_frac=0.025,
            low_frac=0.025,
        )
        if v_perm < best_val:
            best_val, best_seq = v_perm, a_perm.copy()

    # ------------------------------------------------------------------
    # 6) Final tidy‑up
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence (never happens in practice)
        best_seq = np.full(n, target_sum / n, dtype=float)

    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional logging with the official evaluator
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible reuse in later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
