"""
Propose a step‑function (non‑negative vector) that minimizes the
maximum entry of its self‑convolution.
The implementation follows a multi‑phase scheme:

1️⃣  Build a rich pool of parametric shapes (power‑law, exponential,
    half‑sine, Gaussian, triangular, linear) on a modest length.
    Project each shape onto the ℓ¹‑simplex (∑a = √(2 n)).
2️⃣  Run many short projected‑gradient (soft‑max) refinements on the
    small length, keeping the best few candidates.
3️⃣  Upscale the best candidate to a larger length, try a few
    deterministic reorderings, and polish it with a long high‑β
    gradient descent followed by a deterministic mass‑transfer pass.
4️⃣  Finish with a short simulated‑annealing permutation search that
    only reorders the values (the ℓ¹‑norm stays unchanged).
5️⃣  Trim trailing zeros and return the final list.

All stages respect the non‑negativity and ℓ¹‑sum constraints and
use FFT‑based convolutions for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)².  For ∑a = √(2n) this is just max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (return a vector of length n, **not** yet normalised)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (i = 0,…,n‑1)"""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)"""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


def _half_sine_sequence(n: int) -> np.ndarray:
    """Half‑sine wave: zero at both ends, peak in the middle."""
    i = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (i + 0.5) / n)


def _gaussian_sequence(n: int, sigma_factor: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle.  sigma = sigma_factor·n."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = sigma_factor * n
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)


def _triangular_sequence(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak at centre."""
    i = np.arange(n, dtype=np.float64)
    return np.minimum(i + 1, n - i)


def _linear_decreasing_sequence(n: int) -> np.ndarray:
    """Linear decline from 1 to 0."""
    return np.linspace(1.0, 0.0, n, dtype=np.float64)


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest value left, second largest right, third next left, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Same as alternating but start with the largest on the right."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (reversed a) ★ p    (★ = convolution)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 1e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the soft‑max surrogate of max(conv).
    Returns the best vector seen before the deadline.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase temperature (beta) gradually
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy moves that shift a small amount of mass from a “donor’’
    (large contribution to the current maximal convolution entry) to a
    “receiver’’ (small contribution).  Moves are accepted with a
    Metropolis rule whose temperature decays exponentially.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product contributions to the current peak
        w = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# Targeted permutation search (swap‑only simulated annealing)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.1,
                                 end_temp: float = 1e-4,
                                 top_frac: float = 0.02,
                                 low_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style search that swaps indices which
    contribute heavily to the current maximal convolution entry.
    Only the ordering changes – the ℓ¹‑sum stays unchanged.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # contributions of each index to the peak
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")

        # Metropolis acceptance
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
            pass
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score

    best_score = np.inf

    small_phase_dead = start_time + 0.35 * budget_s  # ~35 % of total time
    while time.time() < small_phase_dead - 2.0:
        # pick a seed – either jittered best or random from pool
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_small * 0.004, size=n_small)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_small)
        else:
            _, seed = best_candidates[rng.integers(len(best_candidates))]

        run_dead = min(time.time() + 7.0, small_phase_dead - 1.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_small,
            max_beta=5e5,
            step_factor=0.7,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Small‑GD new best score={best_score:.6f}")

    # If nothing succeeded (unlikely) fall back to a simple shape
    if best_seq is None:
        best_seq = _project_to_simplex(_power_law_sequence(n_small, 0.55), target_small)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Phase 4 – upscale to a larger length and do a thorough polish
    # ------------------------------------------------------------------
    # decide target length based on remaining time
    remaining = time_left()
    if remaining > 300 and budget_s >= 700:
        n_big = 16000
    elif remaining > 180:
        n_big = 12000
    elif remaining > 80:
        n_big = 10000
    else:
        n_big = n_small

    target_big = np.sqrt(2.0 * n_big)

    seq_big = _resize_sequence(best_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # try a few deterministic reorderings at the big size
    candidates_big = [seq_big,
                      _alternating_order(seq_big),
                      _reversed_alternating(seq_big)]
    best_seq_big = seq_big
    best_score_big = _score(seq_big)
    for cand in candidates_big[1:]:
        sc = _score(cand)
        if sc < best_score_big:
            best_score_big = sc
            best_seq_big = cand.copy()
    print(f"[{time.time():.2f}] Upscaled length={n_big} start score={best_score_big:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – long high‑β gradient descent on the big size
    # ------------------------------------------------------------------
    if time_left() > 20.0:
        big_dead = time.time() + time_left() - 12.0
        a_final, val_final = _annealed_gradient_descent(
            best_seq_big,
            big_dead,
            target_big,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_final < best_score_big:
            best_score_big = val_final
            best_seq_big = a_final.copy()
            print(f"[{time.time():.2f}] Big‑GD polish best={best_score_big:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – deterministic mass‑transfer pass (fine‑tuning)
    # ------------------------------------------------------------------
    if time_left() > 10.0:
        mt_dead = time.time() + min(8.0, time_left() - 5.0)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq_big,
            mt_dead,
            target_big,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt < best_score_big:
            best_score_big = val_mt
            best_seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] MT after big polish best={best_score_big:.6f}")

    # ------------------------------------------------------------------
    # Phase 7 – targeted permutation search (ordering only)
    # ------------------------------------------------------------------
    if time_left() > 5.0:
        perm_dead = time.time() + min(6.0, time_left() - 2.0)
        a_perm, val_perm = _permutation_targeted_search(
            best_seq_big,
            perm_dead,
            start_temp=0.12,
            end_temp=1e-5,
            top_frac=0.03,
            low_frac=0.03,
        )
        if val_perm < best_score_big:
            best_score_big = val_perm
            best_seq_big = a_perm.copy()
            print(f"[{time.time():.2f}] Final permutation best={best_score_big:.6f}")

    # ------------------------------------------------------------------
    # Phase 8 – final trimming of trailing zeros
    # ------------------------------------------------------------------
    arr = np.array(best_seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Optional: evaluate and log the final score (uses the provided function)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
