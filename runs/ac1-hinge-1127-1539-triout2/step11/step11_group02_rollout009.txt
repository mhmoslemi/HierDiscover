def _subgradient_descent(a0, deadline, target_sum, init_step=1.0):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    step = init_step
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # mask for near-maximum entries
        tol = 1e-12
        mask = np.where(conv >= max_val - tol)[0]
        # Build p array (size 2n-1)
        p = np.zeros(conv.shape, dtype=np.float64)
        p[mask] = 1.0 / len(mask)  # average of subgradients across active set
        # Compute gradient g = 2 * conv_rev(p) * reversed a?
        # Actually convolution a_rev (length n) with p (len 2n-1) yields length 2n-1.
        # Then extract middle n elements.
        size = 1 << ((2*a.shape[0] - 1).bit_length())
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * a.shape[0] - 1]
        g = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
        # update with step decaying
        a = a - step * g
        # step schedule: step = init_step / sqrt(it+1)
        it += 1
        step = init_step / math.sqrt(it + 1)
        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} score={best_val:.6f}")
    return best_a, best_val

"""
Improved step‑function optimizer for the *max‑convolution* benchmark.

Main ideas
----------

1. **Larger problem size** – use up to 18 000 coefficients when the full
   time budget (≈1000 s) is available.  A finer discretisation gives a
   strictly larger feasible space and usually yields a lower peak.

2. **Richer parametric pre‑search** – in addition to the power‑law /
   exponential families already used, we also try a symmetric
   triangle, a half‑sine, Hann and Blackman windows.  All candidates are
   projected onto the simplex Σa = √(2 n).

3. **Two complementary gradient phases**  
   • *Soft‑max Adam* (as before) – fast descent while the temperature β
     is raised exponentially.  
   • *Exact sub‑gradient* – after the Adam phase a sub‑gradient descent
     on the true max‑convolution (no soft approximation) is performed
     with a decreasing step size.

4. **Improved deterministic mass‑transfer** – the greedy “peak‑lower‑
   ing” move now scans all maximal convolution indices and chooses the
   donor/receiver pair that can actually reduce the current peak.
   This avoids early termination of the transfer phase.

5. **Random local moves** – light simulated‑annealing steps are kept
   between mass‑transfer rounds to escape tiny plateaus.

6. **Final high‑β polish** – a short Adam run with an extremely large
   β finishes the optimisation.

All steps respect non‑negativity and the simplex constraint and
continue until the 1000 s wall‑clock budget is exhausted.  The best
sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Benchmark score; with Σa = √(2 n) this equals 2 n·max_b / (Σa)²."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) scaled to Σa = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-q·i/n) scaled to Σa = target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    """
    Two‑segment power law:
        first floor(alpha·n) entries use exponent p1,
        remaining entries use exponent p2.
    Scaled to Σa = target_sum.
    """
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _triangle_sequence(n: int, target_sum: float) -> np.ndarray:
    """Symmetric triangle – a_i = min(i+1, n-i)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = np.minimum(i, n - i + 1)
    a *= target_sum / a.sum()
    return a


def _sine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine shape, a_i = sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * i / n)
    a *= target_sum / a.sum()
    return a


def _hann_sequence(n: int, target_sum: float) -> np.ndarray:
    """Hann window (raised cosine)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    a *= target_sum / a.sum()
    return a


def _blackman_sequence(n: int, target_sum: float) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, conv_full).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()          # soft‑max probabilities over convolution indices

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation.
    Stops when the supplied wall‑clock time expires.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake off plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    init_step_factor: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  max(a*a).
    Uses the exact sub‑gradient (average over all active indices).
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)
    it = 0
    tol = 1e-12
    init_step = target_sum * init_step_factor

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_active = np.where(conv >= max_val - tol)[0]
        if idx_active.size == 0:
            break

        # Build weight vector p that averages over active indices
        p = np.zeros_like(conv, dtype=np.float64)
        p[idx_active] = 1.0 / idx_active.size

        # Gradient = 2 * (a_rev * p) restricted to the original n entries
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        it += 1
        step = init_step / math.sqrt(it)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the current peak of a*a.
    At each step we look at all maximal convolution indices and transfer
    mass from the entry that contributes most to the peak to the one
    that contributes least, moving as much as possible (up to the donor's
    mass).  This directly lowers the maximal entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        # second‑largest value (ignore all maximal positions)
        mask = np.ones_like(conv, dtype=bool)
        mask[idx_max] = False
        if not mask.any():
            break
        second_max = conv[mask].max()

        improved = False
        # try each maximal index until a feasible donor/receiver pair is found
        for k in idx_max:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)          # indices that appear in conv[k]
            contrib = a[k - idx]                                 # a_{k-i}

            donor_rel = np.argmax(contrib)
            receiver_rel = np.argmin(contrib)
            donor = idx[donor_rel]
            receiver = idx[receiver_rel]

            diff = contrib[donor_rel] - contrib[receiver_rel]
            if diff <= 0.0:
                continue                  # this k cannot be lowered

            # amount needed to bring the current peak exactly down to second_max
            needed = (max_val - second_max) / (2.0 * diff)
            # move at most the donor's mass (add a tiny safety factor)
            delta = min(a[donor], needed * 0.999)
            if delta <= 0.0:
                continue

            a[donor] -= delta
            a[receiver] += delta
            improved = True
            break        # perform one move per outer loop iteration

        if not improved:
            break          # no feasible move found → stop

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """Very light simulated‑annealing‑style moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 1.0:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with the smallest benchmark
    score.  The routine respects the time budget and returns the best
    sequence found (trailing near‑zero entries are stripped).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives a finer search space)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 18000
    elif budget_s >= 600:
        n = 14000
    else:
        n = 10000

    target_sum = math.sqrt(2.0 * n)                      # Σa is fixed to this value

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) power‑law grid (coarse)
    for p in np.arange(0.20, 1.21, 0.04):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail (coarser than previous run)
    for p in np.arange(0.20, 1.21, 0.10):
        for q in np.arange(0.0, 8.001, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (two‑segment)
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) classic symmetric families
    for maker in (
        _triangle_sequence,
        _sine_sequence,
        _hann_sequence,
        _blackman_sequence,
    ):
        seq = maker(n, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (e) previous global best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(
        f"[{time.time():.2f}] After pre‑search: best score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent (soft‑max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + 0.35 * remaining
    while time.time() < gd_deadline - 5.0:
        # jittered start around current best (90 % chance) or fresh random start (10 %)
        if rng.random() < 0.1:
            start_vec = rng.exponential(scale=1.0, size=n)
            start_vec = _project_to_simplex(start_vec, target_sum)
        else:
            jitter = rng.normal(scale=target_sum * 0.015, size=n)
            start_vec = best_seq + jitter
            start_vec = np.maximum(start_vec, 0.0)
            start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            gd_deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.7,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(
                f"[{time.time():.2f}] Adam GD improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase A2 – Exact sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    sub_deadline = time.time() + 0.20 * remaining
    while time.time() < sub_deadline - 5.0:
        # random start for each sub‑gradient run
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _subgradient_descent(
            start_vec, sub_deadline, target_sum, init_step_factor=0.5
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(
                f"[{time.time():.2f}] Sub‑gradient improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase B – greedy deterministic mass‑transfer + random nudges
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(
                f"[{time.time():.2f}] DMT improved best = {best_score:.6f}"
            )

        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(
                f"[{time.time():.2f}] Random move improved best = {best_score:.6f}"
            )

        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Phase C – final high‑β polish
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(
                f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and convert to list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
