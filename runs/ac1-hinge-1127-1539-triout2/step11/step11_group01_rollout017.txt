def _gradient_max_subgrad(a):
    n = a.shape[0]
    size = 1 << ((3*n - 2).bit_length()) # but we likely need >= n+L-1 (n + 2n-1 -1 = 3n-2)

    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_b = conv.max()
    mask = np.zeros_like(conv)
    mask[abs(conv - max_b) < 1e-12] = 1.0
    # compute gradient
    a_rev = a[::-1]
    f_rev = np.fft.rfft(a_rev, n=size)
    mask_fft = np.fft.rfft(mask, n=size)
    grad_rev = np.fft.irfft(f_rev * mask_fft, n=size)[:3*n-2]  # length n+L-1
    g = 2.0 * grad_rev[n-1:n-1 + n]
    return g, max_b

"""
Propose a step‑function with a low max‑convolution value.

Algorithm
---------
1. Choose a problem size `n` (larger n gives more degrees of freedom).
2. Initialise a candidate shape by an exhaustive grid search over
   power‑law and exponentially‑tilted power‑law families.
3. Refine the best candidate with three optimisation phases
   * (A) Adam‑style projected gradient descent on a soft‑max
         approximation of the max‑convolution (many short restarts).
   * (B) Deterministic “mass‑transfer” that analytically reduces the
         current convolution peak (the exact version used in the
         reference implementation).
   * (C) Adam‑style projected descent on the *hard* max‑convolution
         (sub‑gradient of the true max).  This phase exploits the exact
         gradient of the max entry and is very effective for the final
         polishing.
4. A short random‑mass‑transfer stage is interleaved to escape possible
   plateaus.
5. Trim trailing near‑zero coefficients and return the sequence.

The routine respects a user‑supplied time budget (default 1000 s) and
never exceeds the per‑element cap of 1000 enforced by `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value: 2·n·max(conv)/sum(a)^2."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:          # should never happen, guard for safety
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (i starts from 0)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) * exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by the first Adam phase)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)               # soft‑max weights, sum to 1

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Hard max‑convolution sub‑gradient (used by the final Adam phase)
# ----------------------------------------------------------------------


def _gradient_hard_max(a: np.ndarray):
    """Exact sub‑gradient of max(conv(a,a)). Returns (grad, max_b)."""
    n = a.shape[0]
    # convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    # mask of locations attaining the maximum (within tiny tolerance)
    mask = np.zeros_like(conv)
    mask[np.abs(conv - max_b) < 1e-12] = 1.0

    # gradient = 2 * (a_rev * mask) evaluated on the slice [n‑1 : n‑1+n]
    a_rev = a[::-1]
    size2 = 1 << ((n + (2 * n - 1) - 1).bit_length())   # >= n+L-1
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Mr_fft = np.fft.rfft(mask, n=size2)
    grad_full = np.fft.irfft(Ar_fft * Mr_fft, n=size2)[: 3 * n - 2]
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max version)
# ----------------------------------------------------------------------


def _adam_softmax(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  max_beta: float = 2e5,
                  step_factor: float = 0.8,
                  gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam‑softmax it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Many short Adam‑softmax restarts around the current best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_softmax(start,
                                time.time() + run_budget,
                                target_sum,
                                max_beta=2e5,
                                step_factor=0.8,
                                gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam‑softmax restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (same idea as the reference)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves mass from a donor index (large partner value) to a receiver
    (small partner value) using the analytically required amount to
    bring the current peak down to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index with minimal partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact‑DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploratory phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Small random moves that respect the simplex constraint."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam on the *hard* max‑convolution (final polishing)
# ----------------------------------------------------------------------


def _adam_hard_max(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   step_factor: float = 0.6,
                   gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Adam optimiser that uses the exact sub‑gradient of max(conv)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_hard_max(a)

        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam‑hard‑max it={it} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal max‑convolution.
    The routine respects the given time budget and returns the best
    sequence found (as a plain Python list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # small safety margin

    # ------------ problem size ------------------------------------------------
    # Larger n gives more flexibility, but also higher per‑iteration cost.
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)          # normalisation used by the optimiser

    # ------------ Phase 0 – dense parametric pre‑search -----------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.30, 0.85, 31)
    q_grid = np.linspace(0.0, 6.0, 31)

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) seed from previously found global variable, if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    # (d) jittered population around the current best
    jitter_pop = []
    for _ in range(10):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jitter_pop.append(seq_j)

    for seq in jitter_pop:
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # ------------ Phase A – Adam‑softmax with restarts ------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.35
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Softmax GD improved best = {best_score:.6f}")

    # ------------ Phase B – Exact deterministic mass‑transfer ------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] Exact‑DMT2 improved best = {best_score:.6f}")

        # occasional global random moves to climb out of plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        # break early if we are very close to the deadline
        if time.time() > deadline - 4.0:
            break

    # ------------ Phase C – Adam on the hard max (final polishing) ----------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_hard_max(best_seq,
                                        deadline,
                                        target_sum,
                                        step_factor=0.6,
                                        gamma=0.99)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam‑hard‑max best = {best_score:.6f}")

    # ------------ post‑processing ------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
