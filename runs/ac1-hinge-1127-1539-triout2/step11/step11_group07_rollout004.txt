"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.
The algorithm combines:

* Random families (including a new two‑block shape) to generate diverse starts.
* Projected Adam on a smooth‑max surrogate, then on the exact sub‑gradient.
* An aggressive hill‑climb that moves mass between two indices and uses a
  ternary line‑search to find the optimal transfer amount (this greatly
  reduces the maximum convolution entry).
* A short population‑style random‑tweak phase.
* Progressive up‑sampling: solutions are refined for larger lengths
  (2000 → 4000 → 8000) while respecting the ℓ¹‑simplex Σa = √(2 n).

All operations respect the wall‑clock budget (default 1000 s) and the
sequence is projected onto the simplex after each update, guaranteeing that
``evaluate_sequence`` returns the maximum entry of the convolution (the
objective value).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    This implementation simply clamps negative entries to 0 and rescales
    the ℓ¹‑norm to the target sum.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a computed via FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a larger size followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)


# ----------------------------------------------------------------------
# Random initialisers (several families, now 7)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)          # seven families now
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 5:                   # near‑uniform with tiny jitter
        a = np.full(n, target / n)
        a += rng.normal(scale=target * 0.001, size=n)

    else:                               # double‑block shape (new)
        # two uniform blocks near the edges, random split of the mass
        block_len = rng.integers(int(0.1 * n), int(0.3 * n))
        mass_left = rng.random() * target
        mass_right = target - mass_left
        height_left = mass_left / block_len
        height_right = mass_right / block_len
        a = np.zeros(n)
        a[:block_len] = height_left
        a[-block_len:] = height_right
        # small jitter inside each block
        a[:block_len] += rng.normal(scale=height_left * 0.01, size=block_len)
        a[-block_len:] += rng.normal(scale=height_right * 0.01, size=block_len)

    # final jitter & projection
    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p (central n entries)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.12 * target_sum            # slightly smaller than before
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        # increase smoothing parameter gradually
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of max(a*a).
    """
    a = a0.copy()
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.18 * target_sum           # modestly larger than before
    t = 0

    while time.time() < deadline:
        t += 1
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:       # safety
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # sub‑gradient = 2·(a_rev ⋆ p)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climb with optimal delta (ternary search)
# ----------------------------------------------------------------------


def _hill_climb_optimal_delta(a0: np.ndarray, b0: np.ndarray,
                              deadline: float, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the index i contributing most to
    the current worst convolution entry towards an index j with the smallest
    counterpart.  The transfer amount δ is chosen by a ternary search that
    directly minimises the resulting maximum convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    # stop after a few seconds without improvement
    no_improve_iters = 0
    max_no_improve = 30

    while time.time() < deadline and no_improve_iters < max_no_improve:
        # location of the current worst convolution entry
        M = int(np.argmax(b))

        # indices that actually appear in b[M] = Σ a[i] a[M-i]
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)

        # contribution of each i to b[M]
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        # choose a destination index j with the smallest counterpart a[M-j]
        counterpart = a[M - idx]
        # mask out i itself
        mask = (idx != i)
        if not np.any(mask):
            break
        j = idx[mask][np.argmin(counterpart[mask])]

        # maximum transferable mass (keep a[i] ≥ 0, a[j] ≤ 1000)
        delta_max = min(a[i], 1000.0 - a[j])
        if delta_max <= 0:
            break

        # ternary search on δ ∈ [0, delta_max] to minimise max(b_new)
        lo, hi = 0.0, delta_max
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0

            # compute max after moving m1 and m2
            def _max_after(delta):
                effect = np.zeros_like(b)
                effect[i:i + n] -= 2.0 * delta * a
                effect[j:j + n] -= 2.0 * delta * a
                if 2 * i < effect.size:
                    effect[2 * i] += delta * delta
                if 2 * j < effect.size:
                    effect[2 * j] += delta * delta
                if i + j < effect.size:
                    effect[i + j] -= 2.0 * delta * delta
                return np.max(b + effect)

            f1 = _max_after(m1)
            f2 = _max_after(m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1

        delta_opt = (lo + hi) / 2.0
        new_max = _max_after(delta_opt)

        if new_max < b.max() - tol:
            # Apply the move
            a[i] -= delta_opt
            a[j] += delta_opt
            # update b in place using the same effect formula
            effect = np.zeros_like(b)
            effect[i:i + n] -= 2.0 * delta_opt * a
            effect[j:j + n] -= 2.0 * delta_opt * a
            if 2 * i < effect.size:
                effect[2 * i] += delta_opt * delta_opt
            if 2 * j < effect.size:
                effect[2 * j] += delta_opt * delta_opt
            if i + j < effect.size:
                effect[i + j] -= 2.0 * delta_opt * delta_opt
            b = b + effect
            # store best
            if new_max < best_score - tol:
                best_score = new_max
                best_a = a.copy()
                no_improve_iters = 0
                continue
        # no improvement for this pair – try a random jitter to escape
        no_improve_iters += 1
        jitter = rng.normal(scale=1e-6, size=n)
        a += jitter
        a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
        b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method blends random seeding, Adam optimisation (smooth and exact),
    a powerful hill‑climb with optimal delta, and a brief population‑style
    tweak phase.  It also gradually upsamples solutions to larger lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0) Seed from a previously known solution if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            target0 = math.sqrt(2.0 * prev.size)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = _max_convolution_fft(best_seq)
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={prev.size}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 1) Length schedule – start modest, then up‑sample.
    # ------------------------------------------------------------------
    schedule = [1200, 2000, 4000, 8000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # initialise (upsample previous best if possible)
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex_fast(a_init, target)

        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # ------------------------------------------------------------------
        # 1a) Optimisation restarts (smooth‑max Adam → exact‑max Adam)
        # ------------------------------------------------------------------
        min_restart = 6.0                         # seconds per restart
        n_restarts = max(1, int(remaining * 0.55 // min_restart))
        per_restart = remaining * 0.55 / max(1, n_restarts)

        local_best = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)

            start_vec = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)

            # smooth‑max Adam (exploration)
            mid_deadline = min(this_deadline, time.time() + per_restart * 0.5)
            a_mid = _adam_optimize(start_vec, target, mid_deadline)

            # exact‑max Adam (sub‑gradient)
            a_opt = _adam_optimize_submax(a_mid, target, this_deadline)

            score = _max_convolution_fft(a_opt)
            if score < local_best_score:
                local_best_score = score
                local_best = a_opt.copy()

        if local_best is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Improvement after n={n} "
                  f"score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 1b) Deterministic hill‑climb with optimal delta
        # ------------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex_fast(a_cur, target)
            b_cur = _full_convolution(a_cur)
            hc_deadline = min(deadline - 0.2, time.time() + remaining * 0.30)
            a_hc, score_hc = _hill_climb_optimal_delta(a_cur, b_cur, hc_deadline)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb (optimal‑δ) improved n={n} "
                      f"score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 1c) Small random tweaks (population‑style)
        # ------------------------------------------------------------------
        while time.time() + 1.3 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.07                 # move 7 % of source mass
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            score = _max_convolution_fft(cand)
            if score < best_score - 1e-12:
                best_seq = cand
                best_score = score
                print(f"[{time.time():.1f}] Random tweak improved n={n} "
                      f"score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 1d) Optional extra exact‑max Adam run (if time permits)
        # ------------------------------------------------------------------
        extra_budget = deadline - time.time() - 0.5
        if extra_budget > 3.0:
            extra_deadline = time.time() + extra_budget
            a_ref = _adam_optimize_submax(best_seq, target, extra_deadline)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score - 1e-12:
                best_seq = a_ref.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] Post‑HC Adam improved n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final polishing on the longest length we managed to reach
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 1.0
        best_seq = _adam_optimize_submax(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_optimal_delta(
            best_seq, b_final, deadline - 0.5)

        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Clean‑up: drop trailing near‑zeros
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fallback – uniform distribution over 2000 entries
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 4) Report the final bound (for logging)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
