"""
A multi‑phase optimiser for the step‑function benchmark.

The score of a sequence a (with Σa = √(2 n)) equals the maximal entry
of its self‑convolution.  The optimiser works as follows

1.  A dense parametric pre‑search explores three families:
        * pure power‑law
        * power‑law with exponential tail
        * two‑segment broken power‑law
        * an extra “plateau‑exponential’’ family.
    The best candidate and its parameters are kept.

2.  Starting from that candidate we perform several cycles of
    * Projected Adam optimisation on a soft‑max approximation of the
      max‑convolution (β doubles every few iterations, ending at a very
      large value), and
    * Aggressive deterministic mass‑transfer which moves mass from the
      index that contributes most to the current peak to the index that
      contributes least, lowering the peak as far as possible.
    The cycles are repeated while time permits, always keeping the
    best sequence seen.

3.  A short final Adam run with an extremely large β polishes the
    result.

All steps respect the simplex constraint Σa = √(2 n) and a_i ≥ 0.
The routine trims trailing near‑zero entries before returning the
sequence.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (FFT, fast for all n we use)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score; for Σa = √(2 n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _plateau_exp_sequence(
    n: int, K_frac: float, p: float, q: float, target_sum: float
) -> np.ndarray:
    """Flat plateau (first K entries) followed by exp‑power‑law tail."""
    K = max(1, int(K_frac * n))
    i = np.arange(n, dtype=np.float64)
    tail = (i + 1) ** (-p) * np.exp(-q * i / n)
    plateau_val = tail[K - 1]
    tail[:K] = plateau_val
    tail *= target_sum / tail.sum()
    return tail


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in Adam GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approx. of max(a*a).  Returns (grad, conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    conv_full = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()                # soft‑max weights

    a_rev = a[::-1]
    pr_fft = rfft(p, n=size)
    ar_fft = rfft(a_rev, n=size)
    conv_rev = irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.8,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation of max‑convolution.
    Stops when wall‑clock time reaches *deadline*.
    Returns the best sequence seen and its score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 150   # more frequent β updates

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the current maximal convolution entry.
    At each step mass is moved from the index that contributes most
    to the peak to the index that contributes least, moving the maximal
    amount that brings the peak down to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        # pick first maximal index
        k = idx_max[0]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)        # i such that a_i*a_{k-i} appears
        contrib = a[k - idx]                               # the second factor in each term

        # try a few donor/receiver candidates to avoid degenerate denominator
        sorted_desc = idx[np.argsort(-contrib)]            # biggest contributions first
        sorted_asc = idx[np.argsort(contrib)]              # smallest contributions first

        moved = False
        for donor in sorted_desc[:3]:
            for receiver in sorted_asc[:3]:
                if donor == receiver:
                    continue
                denom = 2.0 * (a[k - donor] - a[k - receiver])
                if denom <= 0.0:
                    continue
                # second‑largest convolution value (ignore all current maxima)
                conv2 = conv.copy()
                conv2[idx_max] = -np.inf
                second_max = conv2.max()

                delta_needed = (max_val - second_max) / denom
                max_delta = min(a[donor], delta_needed * 0.999)
                if max_delta <= 0.0:
                    continue

                a[donor] -= max_delta
                a[receiver] += max_delta
                moved = True
                break
            if moved:
                break

        if not moved:
            # No productive move found – give up
            break

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] DMT iter score={best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """Light simulated‑annealing style moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 1.0:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.07  # larger steps than before
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal benchmark score.
    The routine respects the time budget and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ---------------------------------------------------------------
    # Choose problem size – larger n gives a finer discretisation.
    # ---------------------------------------------------------------
    if budget_s >= 900:
        n = 25000
    elif budget_s >= 600:
        n = 20000
    else:
        n = 12000

    target_sum = math.sqrt(2.0 * n)      # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) pure power‑law grid
    for p in np.arange(0.20, 1.21, 0.01):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail grid (moderately fine)
    for p in np.arange(0.20, 1.21, 0.02):
        for q in np.arange(0.0, 8.001, 0.25):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law grid
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) plateau‑exponential family
    for Kf in np.arange(0.05, 0.51, 0.1):
        for p in np.arange(0.20, 1.21, 0.05):
            for q in np.arange(0.0, 8.001, 0.5):
                seq = _plateau_exp_sequence(n, float(Kf), float(p), float(q), target_sum)
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (e) random local perturbations around the current best (if generated by exp‑law)
    # Simple heuristic: try ~200 random (p,q) around the best exp‑law parameters
    # (we do not know the exact parameters of best_seq; we simply random‑search)
    for _ in range(200):
        p = rng.uniform(0.2, 1.2)
        q = rng.uniform(0.0, 8.0)
        seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (f) include a previously‑found global best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(
        f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}"
    )

    # ---------------------------------------------------------------
    # Phase 1 – long Adam‑style projected gradient descent (soft‑max)
    # ---------------------------------------------------------------
    # Give roughly 35 % of the remaining budget to this phase.
    gd_deadline = time.time() + 0.35 * (deadline - time.time())
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            gd_deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.7,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(
                f"[{time.time():.2f}] GD restart improved best = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Phase 2 – alternating deterministic mass‑transfer & short ADAM
    # ---------------------------------------------------------------
    while time.time() < deadline - 8.0:
        # --- deterministic mass‑transfer (core refinement) ---
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(
                f"[{time.time():.2f}] DMT improved best = {best_score:.6f}"
            )

        # --- a brief random‑move phase (helps escape tiny plateaus) ---
        a_rand, val_rand = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(
                f"[{time.time():.2f}] Random move improved best = {best_score:.6f}"
            )

        # --- short Adam run to polish after the above moves ---
        short_deadline = time.time() + 5.0
        if short_deadline < deadline - 2.0:
            a_tmp, val_tmp = _adam_gradient_descent(
                best_seq,
                short_deadline,
                target_sum,
                max_beta=5e5,
                step_factor=0.6,
                gamma=0.995,
            )
            if val_tmp < best_score:
                best_score, best_seq = val_tmp, a_tmp.copy()
                print(
                    f"[{time.time():.2f}] Short ADAM improved best = {best_score:.6f}"
                )
        else:
            break

    # ---------------------------------------------------------------
    # Phase 3 – final polishing (very high β)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(
                f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries, convert to list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation for reporting (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
