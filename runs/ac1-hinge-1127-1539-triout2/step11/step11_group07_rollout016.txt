"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence().

The search combines several stages, each respecting the wall‑clock budget:
  1. Randomly initialise a distribution on the ℓ¹‑simplex Σa = √(2 n).
  2. Adam optimisation using a smooth max (log‑sum‑exp) surrogate.
  3. Adam with the exact sub‑gradient of the max‑convolution.
  4. Deterministic hill‑climb that moves mass away from the worst
     convolution entry.
  5. A small iterative linear‑programming refinement (LP) inspired by the
     classic “exchange” scheme: maximise the total mass of a vector g under
     the constraint (f ∗ g) ≤ M (current maximal convolution).  Scaling g
     to the target ℓ¹‑norm and mixing it with f (step size ½) provably
     reduces the maximal convolution.
  6. Random local tweaks (simulated‑annealing style) for a final polish.

The algorithm starts from the previously discovered best sequence
(height_sequence_1) if it exists, up‑samples it to larger lengths and
re‑optimises.  All vectors are projected onto the simplex after each
update.  The routine stops before the 1000 s deadline and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex_fast(v: np.ndarray, s: float) -> np.ndarray:
    """
    Fast projection onto the simplex {x ≥ 0, Σx = s}.
    Just rescales after clamping to non‑negative.
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0.0:
        # fallback: uniform distribution
        return np.full_like(v, s / v.shape[0])
    return v * (s / total)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (O(n log n))."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (returns the whole array)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    target = math.sqrt(2.0 * new_n)
    return _project_to_simplex_fast(a_new, target)

# Random initialisers (several simple families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # pure exponential
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate([np.exp(-lam1 * np.arange(b)),
                            np.exp(-lam2 * np.arange(b, n))])

    elif family == 5:                   # Gaussian‑ish bump
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.05 * n, 0.15 * n)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif family == 6:                   # two‑peak mixture
        mu1 = rng.uniform(0, n / 2)
        mu2 = rng.uniform(n / 2, n - 1)
        sigma = rng.uniform(0.05 * n, 0.12 * n)
        a = (np.exp(-0.5 * ((idx - mu1) / sigma) ** 2) +
             np.exp(-0.5 * ((idx - mu2) / sigma) ** 2))

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    return _project_to_simplex_fast(a, target)


# ----------------------------------------------------------------------
# Adam optimiser – smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # convolution a_rev * p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, b


def _adam_optimize(a0: np.ndarray, target_sum: float,
                   deadline: float) -> np.ndarray:
    """
    Projected Adam on the smooth‑max surrogate.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.15 * target_sum
    beta = 2.0
    beta_max = 1e7
    beta_factor = 2.0
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

        # occasional jitter to escape flat regions
        if t % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4 / (beta + 1.0),
                                size=n)
            a += jitter
            a = _project_to_simplex_fast(a, target_sum)

        if t % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Adam optimiser – exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize_submax(a0: np.ndarray, target_sum: float,
                          deadline: float) -> np.ndarray:
    """
    Projected Adam where the gradient is the exact sub‑gradient of
    max(a*a).  The active set (indices attaining the maximum) is used as
    the probability vector p.
    """
    a = a0.copy()
    n = a.size
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.2 * target_sum          # a slightly larger base step
    t = 0

    while time.time() < deadline:
        t += 1

        # full convolution and active mask
        b = _full_convolution(a)
        max_b = b.max()
        mask = (b >= max_b - 1e-12).astype(float)
        if mask.sum() == 0.0:       # numerical safety
            mask = np.ones_like(b) / b.size
        p = mask / mask.sum()

        # gradient = 2·(a_rev ⋆ p)  (central n entries)
        a_rev = a[::-1]
        size = _next_pow2(3 * n - 2)
        A_fft = np.fft.rfft(a_rev, n=size)
        P_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(A_fft * P_fft, n=size)[: 3 * n - 2]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex_fast(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climb on the exact convolution
# ----------------------------------------------------------------------


def _hill_climb_deterministic(a0: np.ndarray, b0: np.ndarray,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that attacks the current worst convolution entry.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    best_a = a.copy()
    best_score = b.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        M = int(np.argmax(b))

        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx] * a[M - idx]
        i = idx[np.argmax(contrib)]

        if a[i] < 1e-14:
            break

        # choose a destination with the smallest counterpart
        j_low, j_high = i_low, i_high
        j_range = np.arange(j_low, j_high + 1, dtype=int)
        counterpart = a[M - j_range]

        order = np.argsort(counterpart)
        j = None
        for cand in j_range[order]:
            if cand != i:
                j = cand
                break
        if j is None:
            break

        delta = a[i] * 0.8
        if delta <= 0.0:
            break

        # update convolution (linear‑time update)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < b.max() - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
        else:
            # try a smaller move (binary search)
            improved = False
            delta2 = delta * 0.5
            while delta2 > 1e-14:
                b_test = b.copy()
                b_test[i:i + n] -= 2.0 * delta2 * a
                b_test[j:j + n] += 2.0 * delta2 * a
                if k2i < b_test.size:
                    b_test[k2i] += delta2 * delta2
                if k2j < b_test.size:
                    b_test[k2j] += delta2 * delta2
                if kij < b_test.size:
                    b_test[kij] -= 2.0 * delta2 * delta2
                if b_test.max() < b.max() - 1e-12:
                    a[i] -= delta2
                    a[j] += delta2
                    b = b_test
                    if b.max() < best_score - 1e-12:
                        best_score = b.max()
                        best_a = a.copy()
                    improved = True
                    break
                delta2 *= 0.5
            if not improved:
                jitter = rng.normal(scale=1e-6, size=n)
                a += jitter
                a = _project_to_simplex_fast(a, math.sqrt(2.0 * n))
                b = _full_convolution(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (iterative exchange)
# ----------------------------------------------------------------------


def _lp_iterative_refine(a: np.ndarray, target_sum: float,
                        deadline: float, max_iter: int = 6) -> np.ndarray:
    """
    Iteratively improve a by solving the LP:
        maximise Σ g_j
        subject to (a ∗ g)[k] ≤ M    for all k
                g_j ≥ 0.
    The solution g is scaled to the target ℓ¹‑norm and mixed with a.
    """
    try:
        from scipy.optimize import linprog
    except Exception:          # SciPy unavailable – skip LP step
        return a

    n = a.size
    a = a.copy()
    eps = 1e-9

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        M = _max_convolution_fft(a)

        # Build the (2n‑1) × n constraint matrix for (a ∗ g) ≤ M.
        rows = 2 * n - 1
        A = np.empty((rows, n), dtype=float)
        # column j contains a[0…n‑1] shifted by j
        for j in range(n):
            A[j:j + n, j] = a  # shape matches because a has length n

        # Solve LP: maximise Σ g  ≡ minimise –Σ g
        c = -np.ones(n)
        b_ub = np.full(rows, M * (1.0 + eps))
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – abort refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= target_sum * (1.0 + 1e-12):
            # cannot increase total mass → no further progress
            break

        # Scale g to the required ℓ¹‑norm
        scale = target_sum / sum_g
        g_scaled = g * scale

        # Mix a and the scaled g (step size ½).  The mixing is projected
        # back onto the simplex to guarantee feasibility.
        t = 0.5
        a_new = (1.0 - t) * a + t * g_scaled
        a_new = _project_to_simplex_fast(a_new, target_sum)
        M_new = _max_convolution_fft(a_new)

        if M_new < M - 1e-12:
            a = a_new
            continue

        # If full mixing does not improve, try smaller steps.
        improved = False
        for t in (0.3, 0.1):
            a_new = (1.0 - t) * a + t * g_scaled
            a_new = _project_to_simplex_fast(a_new, target_sum)
            M_new = _max_convolution_fft(a_new)
            if M_new < M - 1e-12:
                a = a_new
                improved = True
                break
        if not improved:
            break

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method combines random initialisation, Adam optimisation,
    deterministic hill‑climb and an LP‑based exchange step, respecting a
    wall‑clock budget (default 1000 s).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1)  Seed from a previously found construction if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            target0 = math.sqrt(2.0 * prev.size)
            prev = _project_to_simplex_fast(prev, target0)
            best_seq = prev.copy()
            best_score = evaluate_sequence(best_seq.tolist())
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
                  f"(n={prev.size}) score={best_score:.6f}")
        except Exception as e:
            print(f"Seeding error: {e}")

    # ------------------------------------------------------------------
    # 2)  Length schedule – gradually increase the dimension.
    # ------------------------------------------------------------------
    schedule = [2000, 3000, 5000]          # modest growth; LP works up to 3000

    for n in schedule:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # initialise (upsample previous best if possible)
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex_fast(a_init, target)

        remaining = deadline - time.time()
        if remaining < 15.0:
            break

        # --------------------------------------------------------------
        # 2a) Allocation of time for this size
        # --------------------------------------------------------------
        opt_budget = remaining * 0.60      # Adam optimisation
        hc_budget = remaining * 0.25       # hill‑climb
        lp_budget = remaining * 0.15       # LP refinement (only if n≤3000)

        # --------------------------------------------------------------
        # 2b) Adam optimisation (multiple restarts)
        # --------------------------------------------------------------
        min_restart = 7.0                # seconds per restart
        n_restarts = max(1, int(opt_budget // min_restart))
        per_restart = opt_budget / n_restarts

        local_best = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            this_deadline = min(deadline - 0.2, time.time() + per_restart)

            start_vec = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)

            # smooth‑max Adam (quick exploration)
            mid_deadline = min(this_deadline,
                               time.time() + per_restart * 0.5)
            a_mid = _adam_optimize(start_vec, target, mid_deadline)

            # exact‑max Adam (sub‑gradient)
            a_opt = _adam_optimize_submax(a_mid, target, this_deadline)

            score = evaluate_sequence(a_opt.tolist())
            if score < local_best_score:
                local_best_score = score
                local_best = a_opt.copy()

        if local_best is not None and local_best_score < best_score - 1e-12:
            best_seq = local_best.copy()
            best_score = local_best_score
            print(f"[{time.time():.1f}] Adam improvement n={n} "
                  f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2c) Deterministic hill‑climb (exact max)
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_cur = best_seq.copy()
            a_cur = _project_to_simplex_fast(a_cur, target)
            b_cur = _full_convolution(a_cur)
            a_hc, score_hc = _hill_climb_deterministic(
                a_cur, b_cur, deadline - 0.2)
            if score_hc < best_score - 1e-12:
                best_seq = a_hc.copy()
                best_score = score_hc
                print(f"[{time.time():.1f}] Hill‑climb improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2d) LP‑based exchange refinement (only for moderate n)
        # --------------------------------------------------------------
        if n <= 3000 and time.time() < deadline - 1.0:
            a_refined = _lp_iterative_refine(best_seq.copy(),
                                             target,
                                             time.time() + lp_budget,
                                             max_iter=6)
            score_ref = evaluate_sequence(a_refined.tolist())
            if score_ref < best_score - 1e-12:
                best_seq = a_refined.copy()
                best_score = score_ref
                print(f"[{time.time():.1f}] LP refinement improved n={n} "
                      f"score={best_score:.6f}")

        # --------------------------------------------------------------
        # 2e) Small random tweaks (simulated‑annealing style)
        # --------------------------------------------------------------
        while time.time() + 1.5 < deadline:
            cand = best_seq.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cand[i] * 0.05
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex_fast(cand, target)
            score = evaluate_sequence(cand.tolist())
            if score < best_score - 1e-12:
                best_seq = cand
                best_score = score
                print(f"[{time.time():.1f}] Random tweak improved n={n} "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        final_deadline = deadline - 1.0
        best_seq = _adam_optimize_submax(best_seq, target, final_deadline)

        b_final = _full_convolution(best_seq)
        best_seq, final_score = _hill_climb_deterministic(
            best_seq, b_final, deadline - 0.5)

        final_score = evaluate_sequence(best_seq.tolist())
        if final_score < best_score - 1e-12:
            best_score = final_score
            print(f"[{time.time():.1f}] Final polishing improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5) Official evaluation report
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
