def _permutation_annealing(seq, deadline, start_temp=0.5, end_temp=1e-4, max_trials=10000):
    rng = np.random.default_rng()
    cur_seq = seq.copy()
    cur_score = _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    it = 0
    while time.time() < deadline and it < max_trials:
        # temperature schedule
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # propose swap
        i, j = rng.integers(0, len(cur_seq), size=2)
        if i == j:
            continue
        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _score(cur_seq)
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] PermAnneal iter={it} best={best_score:.6f}")
        else:
            # revert swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        it += 1
    return best_seq, best_score

"""
Propose a step‑function with a low self‑convolution peak.

The algorithm consists of four stages:

1. **Exploration** – generate a pool of diverse seeds (power‑law,
   exponential, random‑simplex, previous best) and run several
   short annealed‑gradient (soft‑max) descents on a modest length
   (~8‑10 k).  The best result is kept.

2. **Refinement** – a deterministic mass‑transfer pass (greedy
   redistribution of a tiny amount of mass from a “donor’’ that
   contributes most to the current peak to a “receiver’’ that contributes
   little) is applied to the current best.

3. **Upscaling & polishing** – the sequence is linearly interpolated to
   a larger length (12‑16 k depending on remaining time).  A few
   simple reorderings (alternating, even‑spread) are tested and the best
   is taken.  On this size we run a projected sub‑gradient descent on the
   exact max‑convolution and a short Adam‑style soft‑max descent.

4. **Permutation annealing** – the final candidate is further improved
   by a simulated‑annealing over permutations: random swaps of two
   positions are accepted if they lower the peak or with a probability
   that decays with temperature.  This step is cheap because each
   evaluation uses an FFT‑based convolution.

All updates respect the ℓ¹‑simplex (fixed ℓ¹‑norm) and non‑negativity.
The final sequence is trimmed of trailing near‑zero entries,
stored in the global name ``height_sequence_1`` for reuse and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals maxconv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _equal_spaced_sequence(n: int, k: int, target_sum: float) -> np.ndarray:
    """k equal masses placed evenly on n positions (rest zero)."""
    a = np.zeros(n, dtype=np.float64)
    if k == 0:
        return a
    idx = np.linspace(0, n - 1, k, dtype=int)
    mass = target_sum / k
    a[idx] = mass
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Gradient descent on soft‑max approximation (annealed)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return gradient of soft‑max of a*a and the convolution itself."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.7,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected GD with momentum on the soft‑max of the convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # small jitter every few hundred steps
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices attaining the maximal value (with tolerance)
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """Greedy transfer of a tiny amount of mass from a donor to a receiver."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            it += 1
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based simulated annealing)
# ----------------------------------------------------------------------


def _permutation_annealing(seq: np.ndarray,
                           deadline: float,
                           start_temp: float = 0.5,
                           end_temp: float = 1e-4,
                           max_trials: int = 200_000) -> tuple[np.ndarray, float]:
    """Random swaps of positions, accepting with a Metropolis rule."""
    rng = np.random.default_rng()
    cur_seq = seq.copy()
    cur_score = _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    start_time = time.time()
    it = 0
    n = cur_seq.shape[0]

    while time.time() < deadline and it < max_trials:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # perform swap
        cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
        new_score = _score(cur_seq)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time():.2f}] PermAnneal it={it:6d} best={best_score:.6f}")
        else:
            # revert the swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low self‑convolution peak.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # 0.5 s margin

    # ------------------------------------------------------------------
    # Choose dimensionality for the exploratory phase
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 10000
    elif budget_s >= 400:
        n_expl = 8000
    else:
        n_expl = 6000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a diverse pool of starting points
    # ------------------------------------------------------------------
    init_pool = []

    # power‑law family
    for p in np.linspace(0.2, 1.0, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.001, 0.025, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # equal‑spaced families (few big lumps)
    for k in [3, 5, 9, 15]:
        init_pool.append(_equal_spaced_sequence(n_expl, k, target_expl))

    # random Dirichlet‑like points
    for _ in range(8):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.02, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # Exploration: many short annealed‑gradient runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    while time.time() < deadline - 12.0:   # keep slack for later phases
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_dead = min(time.time() + 22.0, deadline - 10.0)
        if run_dead <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_dead, target_expl,
            max_beta=8e5, step_factor=0.75, gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer refinement (small scale)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        mt_dead = time.time() + (deadline - time.time()) * 0.13
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, mt_dead, target_expl,
            start_temp=0.35, end_temp=5e-5, low_frac=0.07,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Mass‑transfer refined best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Upscaling to a larger size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    # Choose final length based on remaining time
    if remaining > 120.0:
        n_final = 20000
    elif remaining > 80.0:
        n_final = 16000
    elif remaining > 40.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Try simple reorderings; keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)

    best_seq = seq_big
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Upscaled to {n_final} – start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient refinement on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sub_dead = deadline - 6.0
        a_sub, val_sub = _subgradient_descent(
            best_seq, sub_dead, target_final,
            step_factor=0.42, gamma=0.997,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Short Adam‑style soft‑max descent on large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        adam_dead = time.time() + min(4.5, deadline - time.time() - 1.0)
        a_adam, val_adam = _annealed_gradient_descent(
            best_seq, adam_dead, target_final,
            max_beta=1e6, step_factor=0.65, gamma=0.98,
        )
        if val_adam < best_score:
            best_score = val_adam
            best_seq = a_adam.copy()
            print(f"[{time.time():.2f}] Adam‑style GD improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Permutation annealing (swap based) – final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        perm_dead = time.time() + min(5.0, deadline - time.time() - 0.5)
        a_perm, val_perm = _permutation_annealing(
            best_seq, perm_dead,
            start_temp=0.6, end_temp=2e-5,
            max_trials=150_000,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Permutation anneal final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final tidy‑up: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Logging of the final score (purely for information)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # make the sequence globally available for possible warm‑starts
    globals()["height_sequence_1"] = final_seq
    return final_seq
