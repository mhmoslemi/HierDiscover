"""
Propose a step‑function with a low maximal self‑convolution.

The search works in four phases within the time budget:

1. **Rich parametric pre‑search** – a dense grid over a three‑parameter
   family
        a_i ∝ (i+δ)^(-p)·(1-i/(n‑1))^η·exp(-q·i/n)
   (δ,η≥0, p∈[0.45,0.70], q∈[0,10]) is scanned, each candidate is
   normalised to Σa=√(2 n) and the best‑scoring one is kept.

2. **Projected Adam optimisation** – the softened objective
   `softmax_β(conv)` (β starts at 0.5 and is doubled every few hundred
   iterations) is minimised on the simplex {a_i≥0, Σa=S} using Adam with
   Euclidean projection.  Several short restarts are performed.

3. **Greedy deterministic mass transfer** – the index `k` attaining the
   current maximal convolution is identified.  Mass is moved from a
   “donor” index whose derivative 2·a_{k‑i} is large to a “receiver”
   with a small derivative, the amount being the analytically required
   one to bring the peak down to the second‑largest value.  The step is
   repeated until the gap falls below a tiny tolerance.

4. **Final polishing** – a longer Adam run with a very large β (up to
   1 e⁶) finishes the optimisation.

All phases respect the overall time budget (default 1000 s).  The
resulting sequence is trimmed of trailing near‑zero entries before
being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of appropriate size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective value: 2·n·max(conv)/(sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the fast pre‑search)
# ----------------------------------------------------------------------
def _family_sequence(n: int,
                     p: float,
                     delta: float,
                     eta: float,
                     q: float,
                     target_sum: float) -> np.ndarray:
    """
    a_i ∝ (i+δ)^(-p)·(1-i/(n‑1))^η·exp(-q·i/n).
    All parameters are non‑negative.  The result is normalised to
    Σa = target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    # power‑law part, shifted by delta (delta may be zero)
    a = (i + delta) ** (-p)
    if eta != 0.0:
        a *= (1.0 - i / (n - 1)) ** eta
    if q != 0.0:
        a *= np.exp(-q * i / n)
    # guard against all zeros (should never happen)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a *= target_sum / total
    return a


# ----------------------------------------------------------------------
# Soft‑max based gradient (used by Adam)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (grad, conv) where grad is the gradient of
    softmax_β(max(conv)) w.r.t. a and conv is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # soft‑max over the convolution entries
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()          # probabilities, sum to 1

    # grad = 2·(a ⊛ p) restricted to the first n entries
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  max_beta: float = 5e5,
                  step_factor: float = 0.7,
                  gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max objective.
    Returns the best vector seen and its exact score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase soft‑max temperature
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _adam_with_restarts(start_vec: np.ndarray,
                        target_sum: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Run several short Adam restarts starting from jittered copies of
    `start_vec`.  Returns the best vector found.
    """
    best_a = start_vec.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        s = best_a + jitter
        s = np.maximum(s, 0.0)
        s = _project_to_simplex(s, target_sum)

        a, val = _adam_descent(s, time.time() + run_budget,
                               target_sum,
                               max_beta=5e5,
                               step_factor=0.7,
                               gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Restart improved best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (greedy flattening)
# ----------------------------------------------------------------------
def _mass_transfer(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    At each iteration the peak index k is identified and a small amount of
    mass is moved from a donor (large ∂max/∂a_i) to a receiver (small
    ∂max/∂a_i) such that the peak is lowered to the second‑largest value.
    The loop stops when the gap is below `tol` or time expires.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest (ignore all current global maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick a random maximal index to avoid bias
        k = idx_max[np.random.randint(idx_max.size)]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)      # indices i that appear in the k‑th sum
        partners = a[k - idx]                            # derivative coefficients (2·partners)

        # donor: among idx with a[i] > 0 pick the one with largest partner value
        mask_pos = a[idx] > 0
        if not np.any(mask_pos):
            break
        donor_candidates = idx[mask_pos]
        donor_partners = partners[mask_pos]
        donor = donor_candidates[np.argmax(donor_partners)]

        # receiver: among the same window choose index with smallest partner (different from donor)
        receiver_candidates = idx[idx != donor]
        if receiver_candidates.size == 0:
            break
        receiver_partners = partners[idx != donor]
        receiver = receiver_candidates[np.argmin(receiver_partners)]

        # compute analytically the required transfer
        pd = a[k - donor]
        pr = a[k - receiver]
        denom = 2.0 * (pd - pr)
        if denom <= 0.0:
            # cannot lower the peak with this pair – give up on this iteration
            break
        delta_needed = (max_val - second_max) / denom

        # be conservative: never move more than 70 % of donor's mass
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Mass‑transfer it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Light global random moves (exploratory phase)
# ----------------------------------------------------------------------
def _global_random_moves(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        # pick a donor among the largest values
        k_top = max(1, int(top_frac * n))
        donor = np.argsort(-a)[rng.integers(k_top)]

        # pick a receiver among the smallest values
        k_low = max(1, int(low_frac * n))
        receiver = np.argsort(a)[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal max‑convolution.
    The routine respects the given time budget and returns the best
    sequence found as a plain Python list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger size gives more flexibility)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 13000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed (scale‑invariant)

    # ------------------------------------------------------------------
    # Phase 0 – extensive parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # grids – chosen generously but still cheap
    p_grid = np.linspace(0.45, 0.68, 24)          # 24 values
    delta_grid = np.linspace(0.0, 4.0, 11)       # 11 values
    eta_grid = np.linspace(0.0, 3.0, 7)          # 7 values
    q_grid = np.linspace(0.0, 10.0, 11)          # 11 values

    # (a) pure power‑law with optional shift & taper
    for p in p_grid:
        for delta in delta_grid:
            for eta in eta_grid:
                seq = _family_sequence(n, float(p), float(delta), float(eta), 0.0, target_sum)
                val = _score(seq)
                if val < best_score:
                    best_score = val
                    best_seq = seq.copy()

    # (b) add exponential tail
    for p in p_grid:
        for delta in delta_grid:
            for eta in eta_grid:
                for q in q_grid:
                    seq = _family_sequence(n, float(p), float(delta), float(eta), float(q), target_sum)
                    val = _score(seq)
                    if val < best_score:
                        best_score = val
                        best_seq = seq.copy()

    # (c) seed from a previously found construction, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # interpolate to required length
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _score(prev)
        if val < best_score:
            best_score = val
            best_seq = prev.copy()

    # (d) small jittered population around the current best
    rng = np.random.default_rng()
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Projected Adam gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start = best_seq + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a_opt, val_opt = _adam_with_restarts(start, target_sum, gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD phase improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        a_ref, val_ref = _mass_transfer(best_seq, deadline, target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best={best_score:.6f}")

        a_rand, val_rand = _global_random_moves(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Random moves improved best={best_score:.6f}")

        # optional second mass‑transfer pass to exploit any fresh improvements
        a_ref2, val_ref2 = _mass_transfer(best_seq, deadline, target_sum)
        if val_ref2 < best_score:
            best_score = val_ref2
            best_seq = a_ref2.copy()
            print(f"[{time.time():.2f}] Second mass‑transfer improved best={best_score:.6f}")

        if time.time() > deadline - 6.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Final long Adam polishing with very large β
    # ------------------------------------------------------------------
    if time.time() + 4.0 < deadline:
        a_fin, val_fin = _adam_descent(best_seq,
                                        deadline,
                                        target_sum,
                                        max_beta=1e6,
                                        step_factor=0.6,
                                        gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.asarray(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional sanity check (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
