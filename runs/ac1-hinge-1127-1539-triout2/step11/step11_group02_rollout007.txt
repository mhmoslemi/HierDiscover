def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score – with Σa = √(2 n) this equals max(a*a)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return _max_conv(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (used in ADAM)
# ----------------------------------------------------------------------


def _max_subgradient(a: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute a sub‑gradient of  f(a)=max_k (a*a)[k] .
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    # indices which are (almost) maximal – tolerance guards against
    # floating point wiggle.
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution 2*a_{k-i}
        grad[idx] += 2.0 * a[k - idx]
    return grad, conv


# ----------------------------------------------------------------------
# ADAM on the exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _adam_max_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    step_factor: float = 0.8,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """
    Projected ADAM using the exact sub‑gradient of max_k (a*a)[k].
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    # occasional jitter to avoid round‑off plateaux
    jitter_interval = 400

    while time.time() < deadline:
        g, conv = _max_subgradient(a)

        # Adam moments
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # tiny random jitter every few iterations
        if it % jitter_interval == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (np.sqrt(it + 1) + 1.0),
                size=a.shape,
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] ADAM‑max it={it:5d} score={best_val:.6f}"
            )

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy exact reduction of the current maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        # pick the first maximal location
        k = idx_max[0]

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions a_{k-i}
        contrib = a[k - idx]

        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        # second largest value (excluding all maximal entries)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        denom = 2.0 * (a[k - donor] - a[k - receiver])
        if denom <= 0.0:
            # can't improve this peak – give up on this index
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta
        # sum preserved, non‑negativity guaranteed by the bound above

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] DMT it={int(time.time())} score={best_score:.6f}"
            )
        # loop continues
    return best_a, best_score


# ----------------------------------------------------------------------
# Very light global random transfer (helps escape tiny plateaux)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 1.0:
        # pick a relatively large entry and a relatively small entry
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Random‑move new best={best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a minimal benchmark score.
    The routine respects the time budget and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------- problem size ---------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    else:
        n = 10000

    target_sum = math.sqrt(2.0 * n)                # Σa = √(2 n)

    # ------------------- Phase 0: ultra‑dense pre‑search -------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) pure power‑law grid
    for p in np.arange(0.20, 1.21, 0.03):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in np.arange(0.20, 1.21, 0.04):
        for q in np.arange(0.0, 8.001, 0.5):
            seq = _exp_power_law_sequence(
                n, float(p), float(q), target_sum
            )
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (two‑segment)
    for p1 in np.arange(0.20, 0.81, 0.2):
        for p2 in np.arange(0.80, 2.01, 0.3):
            for alpha in np.arange(0.1, 0.91, 0.2):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) random Dirichlet starts (few, cheap)
    for _ in range(5):
        rand_vec = rng.random(n)
        rand_vec = _project_to_simplex(rand_vec, target_sum)
        val = _score(rand_vec)
        if val < best_score:
            best_score, best_seq = val, rand_vec.copy()

    # (e) reuse previous global best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _score(prev)
        if val < best_score:
            best_score, best_seq = val, prev.copy()

    print(
        f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}"
    )

    # ------------------- Phase A: ADAM on exact max‑subgradient -----
    # Allocate roughly 35 % of the remaining budget.
    phase_a_deadline = time.time() + 0.35 * (deadline - time.time())
    while time.time() < phase_a_deadline - 5.0:
        # jittered start around current best
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_max_subgradient_descent(
            start_vec,
            phase_a_deadline,
            target_sum,
            step_factor=0.9,
            gamma=0.992,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(
                f"[{time.time():.2f}] ADAM‑max improved best = {best_score:.6f}"
            )

    # ------------------- Phase B: deterministic mass‑transfer -------
    # Use the bulk of the remaining time (leaving a small buffer).
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(
                f"[{time.time():.2f}] DMT improved best = {best_score:.6f}"
            )

        # occasional random nudges
        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(
                f"[{time.time():.2f}] Random‑move improved best = {best_score:.6f}"
            )

        if time.time() > deadline - 2.0:
            break

    # ------------------- Phase C: final polish (tiny step) ----------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_max_subgradient_descent(
            best_seq,
            deadline,
            target_sum,
            step_factor=0.5,
            gamma=0.998,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(
                f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}"
            )

    # ------------------- post‑processing ---------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final benchmark report (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
