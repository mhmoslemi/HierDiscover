"""
Robust two‑stage optimizer for the max‑auto‑convolution problem.

Stage 1 – Linear‑programming “mass‑expansion”:
    For a fixed non‑negative sequence f with Σf = √(2 n) we set
        M = max(f*f).
    The LP   max Σ g   s.t.   f*g ≤ M ,  g ≥ 0
    yields a candidate g that can carry more total mass without ever
    exceeding the current peak M.  After rescaling g to the required sum
    we mix f←(1‑t) f + t g (t≈0.5).  Repeating the step quickly drives
    the peak down.  The convolution matrix is built once per iteration
    as a sparse Toeplitz matrix; SciPy’s HiGHS solver solves the LP
    reliably for n≈2500 (≈50 MB of non‑zeros).

Stage 2 – Deterministic mass‑transfer + cheap random hill‑climbing:
    A greedy exact‑reduction routine (identical to the “Exact DMT2”
    from the reference code) is applied to polish the LP result.
    Finally a small stochastic local search (random donor/receiver moves)
    finishes the job within the remaining time budget.

Both stages respect the 1000 s global budget, keep all entries
non‑negative, and return the best sequence found (trailing nearly‑zero
entries are stripped).  The final sequence is evaluated with the
provided `evaluate_sequence` for reporting only.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy import sparse


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast and sufficiently accurate)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using the FFT based convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective for a given sequence a.
    When Σa = √(2·n) the score equals max(a*a), otherwise the
    full formula 2·n·max/Σ² is used.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used for final polishing steps
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x≥0, Σx = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming step (maximise mass while respecting the peak)
# ----------------------------------------------------------------------
def _solve_lp_expand(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve   max Σ g   s.t.   f*g ≤ M , g ≥ 0 .
    Returns the optimal g (unscaled) or None on failure.
    """
    n = f.shape[0]
    # Build the (2n‑1)×n Toeplitz matrix for convolution with f.
    # Row index = i + j , column index = j , value = f[i].
    i_rep = np.repeat(np.arange(n, dtype=np.int64), n)          # i repeated n times
    j_rep = np.tile(np.arange(n, dtype=np.int64), n)            # j tiled n times
    rows = i_rep + j_rep
    cols = j_rep
    data = np.repeat(f, n)                                      # f[i] repeated over its whole row

    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)  # maximise Σg  →  minimise –Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # SciPy's HiGHS handles large sparse LPs efficiently.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={"presolve": True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Deterministic exact mass‑transfer (same idea as the reference DMT2)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                tol: float = 1e-12) -> np.ndarray:
    """
    Greedy reduction of the maximal convolution entry.
    Repeatedly moves mass from a “donor” index (large partner value)
    to a “receiver” (small partner value) so that the current peak
    approaches the second‑largest convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick receiver with minimal partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            # optional progress print
            print(f"[{time.time():.2f}] DMT step it={it:5d} score={best_score:.6f}")

        it += 1

    # re‑normalise to the required sum (numerical drift may appear)
    best_a = best_a * (target_sum / best_a.sum())
    return best_a


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climbing (random local moves)
# ----------------------------------------------------------------------
def _random_local_search(a0: np.ndarray,
                         deadline: float,
                         target_sum: float) -> np.ndarray:
    """
    Simple stochastic improvement: repeatedly move a small random amount
    from a donor (chosen proportionally to its mass) to a receiver.
    Accept only moves that lower the FFT‑based score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        if a.sum() == 0:
            break
        donor = rng.choice(n, p=a / a.sum())
        receiver = rng.integers(n)
        if donor == receiver:
            continue

        # move up to 15 % of donor mass
        delta = rng.random() * a[donor] * 0.15
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            # keep the move
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta

    # final normalisation (ensures exact target sum)
    a = a * (target_sum / a.sum())
    return a

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:

    Find a non‑negative step‑function (list of heights) minimising the
    `evaluate_sequence` objective.  The method combines an LP‑based
    expansion, deterministic mass‑transfer, and a short random local
    search, all respecting the global time budget.

    # --------------------------------------------------------------

        if prev.size != n:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            f = np.interp(x_new, x_old, prev)
        else:
            f = prev.copy()
        f = f * (target_sum / f.sum())
    else:
        # a simple power‑law seed
        p = 0.55
        f = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        f = f * (target_sum / f.sum())

    # Ensure feasibility
    f = np.maximum(f, 0.0)

    # --------------------------------------------------------------
    # Phase 1 – LP based mass expansion / mixing
    # --------------------------------------------------------------
    # Allocate ~55 % of the budget to the LP stage.
    lp_deadline = start_time + 0.55 * (deadline - start_time)
    t_mix = 0.5                          # mixing coefficient
    best_f = f.copy()
    best_M = _max_convolution_fft(best_f)

    while time.time() < lp_deadline:
        # solve LP for the current f
        g_raw = _solve_lp_expand(best_f, best_M, target_sum)
        if g_raw is None:
            break

        # rescale g to the required total mass
        sum_g = g_raw.sum()
        if sum_g <= 0:
            break
        g = g_raw * (target_sum / sum_g)

        # convex combination
        f_new = (1.0 - t_mix) * best_f + t_mix * g
        M_new = _max_convolution_fft(f_new)

        # accept if we made progress
        if M_new < best_M * (1.0 - 1e-7):
            best_f = f_new
            best_M = M_new
            print(f"[{time.time():.2f}] LP mix improved M={best_M:.6f}")
            # occasionally increase mixing weight to accelerate if progress is good
            t_mix = min(0.9, t_mix * 1.05)
        else:
            # reduce mixing if we overshoot
            t_mix = max(0.1, t_mix * 0.7)
            # break if mixing has become too tiny
            if t_mix < 0.02:
                break

    # --------------------------------------------------------------
    # Phase 2 – deterministic exact mass‑transfer (greedy reduction)
    # --------------------------------------------------------------
    dmt_deadline = start_time + 0.85 * (deadline - start_time)
    best_f = _deterministic_mass_transfer(best_f, dmt_deadline, target_sum)

    # --------------------------------------------------------------
    # Phase 3 – short stochastic local refinement
    # --------------------------------------------------------------
    rnd_deadline = deadline - 0.25  # leave a few seconds for final reporting
    best_f = _random_local_search(best_f, rnd_deadline, target_sum)

    # --------------------------------------------------------------
    # Final trimming and reporting
    # --------------------------------------------------------------
    arr = np.array(best_f, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
