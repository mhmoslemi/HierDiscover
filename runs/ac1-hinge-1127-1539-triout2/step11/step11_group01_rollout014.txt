"""
Proposes a non‑negative step function (list of heights) that aims to minimise
the max‑convolution value (equivalently the evaluation function).

The search consists of several phases:

1. **Rich parametric pre‑search** – enumerate a variety of simple analytic
   families (pure power‑law, power‑law with exponential tail, and versions
   with an early plateau) and keep the best candidate.  Small random jitters
   around the best are also examined.

2. **Projected Adam gradient descent** – minimise a smooth approximation of
   the max‑convolution (soft‑max) on the simplex Σa = √(2n).  Multiple
   restarts are performed; the soft‑max temperature β is raised slowly.

3. **Exact deterministic mass‑transfer** – repeatedly locate an index where
   the convolution attains its maximum and shift mass from a “donor’’ index
   with large contribution to a “receiver’’ with small contribution, using the
   analytically required amount to drop the peak to the second‑largest value.

4. **Simulated annealing** – random donor/receiver moves are accepted with
   a Metropolis criterion, allowing occasional uphill steps so the algorithm
   can escape shallow local minima.

5. **Final high‑β polishing** – a last short Adam run with a very large β
   and a small step size refines the solution.

All stages respect the global time budget (default 1000 s) and the final
sequence is trimmed of trailing near‑zero entries before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – identical to evaluate_sequence for non‑negative a."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families used in the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _plateau_power_law_sequence(n: int, p: float, L_frac: float,
                                target_sum: float) -> np.ndarray:
    """Plateau up to L = floor(L_frac·n) then power‑law tail."""
    L = max(0, min(n - 1, int(L_frac * n)))
    base = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    plateau_val = base[L]                     # constant level on the plateau
    a = np.maximum(base, plateau_val)
    a *= target_sum / a.sum()
    return a


def _plateau_exp_power_law_sequence(n: int, p: float, q: float,
                                    L_frac: float,
                                    target_sum: float) -> np.ndarray:
    """Plateau + exponential‑weighted power‑law tail."""
    L = max(0, min(n - 1, int(L_frac * n)))
    i = np.arange(n, dtype=np.float64)
    base = (i + 1) ** (-p) * np.exp(-q * i / n)
    plateau_val = base[L]
    a = np.maximum(base, plateau_val)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of β·log‑sum‑exp(β·(conv−max)) w.r.t. a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} "
                  f"score={best_val:.6f}")

        # raise the temperature β gradually
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    For the current peak index k, mass is transferred from a donor index
    with large partner value to a receiver with small partner value,
    using the analytically required amount to bring the peak down to the
    second‑largest convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner coefficients a_{k‑i}
        partner = a[k - idx]               # shape == idx.shape

        # donor must have positive mass
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        # coefficients for the chosen donor / receiver
        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            # cannot reduce this peak using this pair – give up on this iteration
            break

        # amount required to bring the peak down to the second‑largest value
        delta_needed = (max_val - second_max) / denom

        # we never move more than a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (random mass‑transfer with Metropolis acceptance)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         start_time: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Random donor/receiver moves accepted with probability exp(-(Δ)/T).
    Temperature T decays linearly to zero at the end of the allotted time.
    The total mass is preserved, all entries stay non‑negative.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val
    n = a.shape[0]
    rng = np.random.default_rng()
    total_time = deadline - start_time

    # pre‑compute a few index lists to avoid allocating each loop
    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        T = (1.0 - progress) * best_val * 0.02 + 1e-12

        # pick donor from the top 5 % (by value) and receiver from the bottom 5 %
        k_top = max(1, int(0.05 * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(0.05 * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.15
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta

        val = _score(a)
        if val < cur_val:
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_val:.6f}")
        else:
            # metropolis acceptance
            if rng.random() < np.exp(-(val - cur_val) / T):
                cur_val = val
            else:
                # revert
                a[donor] += delta
                a[receiver] -= delta

    return best_a, best_val

    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – larger n gives more flexibility, but must stay fast
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 11000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa is fixed to this value (scale‑invariant)

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (rich family)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # grids
    p_grid = np.linspace(0.20, 1.50, 27)          # 27 values
    q_grid = np.linspace(0.0, 8.0, 17)           # 17 values
    L_frac_grid = np.linspace(0.05, 0.50, 10)    # plateau fraction

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) plateau + power‑law
    for p in p_grid:
        for Lf in L_frac_grid:
            seq = _plateau_power_law_sequence(n, float(p), float(Lf), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (d) plateau + exponential‑weighted power‑law
    for p in p_grid:
        for q in q_grid:
            for Lf in L_frac_grid:
                seq = _plateau_exp_power_law_sequence(
                    n, float(p), float(q), float(Lf), target_sum)
                val = _score(seq)
                if val < best_score:
                    best_score = val
                    best_seq = seq.copy()

    # (e) include previously found sequence if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_len = prev.shape[0]
            old_x = np.linspace(0.0, 1.0, old_len)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (f) jittered population around current best
    for _ in range(10):
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = "
          f"{best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.25
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 15.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # brief random walk to help escape possible plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 10.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Simulated annealing (exploratory)
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_sa, val_sa = _simulated_annealing(best_seq,
                                            target_sum,
                                            start_time,
                                            deadline - 5.0)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Final polishing with a very high β for the soft‑max
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.5,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
