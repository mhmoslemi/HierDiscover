    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv.max()

def simulated_annealing_refine(a, target_sum, deadline):
    cur = a.copy()
    cur_val = max_convolution_fft(cur)
    best = cur.copy()
    best_val = cur_val
    it = 0
    n = len(cur)
    rng = np.random.default_rng()
    # Temperature schedule: start at maybe 0.01*cur_val (or start_temp) and decay.
    start_temp = cur_val * 0.1  # 10% of current objective.
    while time.time() < deadline:
        # Compute temperature linearly decreasing to near zero.
        elapsed = time.time() - start_time_for_this_phase
        t_frac = elapsed / (deadline - start_start)
        temp = start_temp * (1 - t_frac)  # linear
        # donor: choose index weighted by a (prefer large)
        donor = int(rng.choice(n, p=cur/cur.sum())) if cur.sum()>0 else rng.integers(n)
        # Ensure donor has > small mass
        if cur[donor] < 1e-12:
            continue
        # receiver distinct
        receiver = rng.integers(n-1)
        if receiver >= donor:
            receiver += 1
        # delta as a fraction of donor mass
        max_delta = cur[donor] * 0.5
        delta = rng.random() * max_delta
        # create candidate
        cand = cur.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        # maybe small jitter? Not needed.
        # Evaluate max conv.
        val = max_convolution_fft(cand)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / max(temp, 1e-12)):
            cur = cand
            cur_val = val
            if val < best_val:
                best_val = val
                best = cand.copy()
                print(f"[{time.time():.2f}] SA new best {best_val:.6f} it={it}")

    return best, best_val

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a,a)) / (sum(a)^2)

where n = len(a).  The optimum is attained when sum(a) = sqrt(2*n), in which
case the objective equals the maximal entry of the self‑convolution.
The implementation follows a multi‑phase strategy:

1.  Build a large diverse pool of parametric shapes (power‑law,
    exponential, sine‑power, and random convex combinations) and project each
    onto the ℓ¹‑simplex {x≥0, Σx = sqrt(2 n)}.

2.  Keep the best few candidates and refine them with a projected momentum
    gradient descent on a soft‑max approximation of the max‑convolution
    (β is annealed up to 1e6).

3.  Perform a few sub‑gradient steps on the true max‑convolution and a
    deterministic “mass‑transfer” local search that moves mass from high‑
    contributing to low‑contributing positions.

4.  Finally run a simulated‑annealing (random mass‑transfer) phase that
    explores the neighbourhood of the current best solution until the time
    budget expires.

All steps respect the simplex constraint by a Euclidean projection.
The routine returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (unnormalised)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _linear_tent_sequence(n: int) -> np.ndarray:
    """Piecewise linear tent (peak in the centre)."""
    x = np.linspace(-1, 1, n)
    return 1.0 - np.abs(x)


def _random_combination_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random convex combination of two basic shapes."""
    # pick two shape types
    shapes = [_power_law_sequence, _exponential_sequence, _sin_power_sequence]
    f1, f2 = rng.choice(shapes, size=2, replace=False)

    # random parameters
    if f1 is _power_law_sequence:
        p1 = rng.uniform(0.2, 1.5)
        s1 = f1(n, p1)
    elif f1 is _exponential_sequence:
        lam1 = rng.uniform(5e-4, 0.02)
        s1 = f1(n, lam1)
    else:
        p1 = rng.uniform(0.2, 2.5)
        s1 = f1(n, p1)

    if f2 is _power_law_sequence:
        p2 = rng.uniform(0.2, 1.5)
        s2 = f2(n, p2)
    elif f2 is _exponential_sequence:
        lam2 = rng.uniform(5e-4, 0.02)
        s2 = f2(n, lam2)
    else:
        p2 = rng.uniform(0.2, 2.5)
        s2 = f2(n, p2)

    w = rng.random()
    seq = w * s1 + (1 - w) * s2
    return seq


# ----------------------------------------------------------------------
# Soft‑max gradient of max‑convolution (used in GD)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – annealed momentum GD on soft‑max
# ----------------------------------------------------------------------
def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 1e6,
                                   step_factor: float = 0.6,
                                   gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected momentum GD on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase X – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic mass transfer (local annealing)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass transfer that reduces the current peak."""
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, kmax‑i)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase C – Simulated annealing (random mass moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer annealing that explores the neighbourhood of a0."""
    a = a0.copy()
    cur_val = _max_convolution_fft(a)
    best_a = a.copy()
    best_val = cur_val

    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    # start temperature as a fraction of the current objective
    start_temp = cur_val * 0.08 + 5e-3
    it = 0

    while time.time() < deadline:
        # linear temperature decay
        frac = (time.time() - start) / (deadline - start)
        temp = start_temp * (1.0 - frac)

        # choose donor (probability ∝ a) and receiver (uniform)
        if cur_val == 0:
            break
        donor = int(rng.choice(n, p=a / a.sum()))
        if a[donor] < 1e-12:
            # donor is essentially empty, skip
            continue
        # pick a different receiver
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        # move a random fraction of donor's mass
        max_delta = a[donor] * 0.4
        delta = rng.random() * max_delta

        cand = a.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        # No projection needed – sum is preserved and entries stay non‑negative

        cand_val = _max_convolution_fft(cand)

        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / max(temp, 1e-12)):
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the max self‑convolution.
    Returns a Python list suitable for `evaluate_sequence`.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size n based on budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 24000
    elif budget_s >= 500:
        n = 18000
    else:
        n = 12000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Helper for fast evaluation of a candidate
    # --------------------------------------------------------------
    def _eval(arr: np.ndarray) -> float:
        return _max_convolution_fft(arr)

    # --------------------------------------------------------------
    # Build a diverse pool of candidates
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # Warm‑start from previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        candidates.append(prev)

    # Power‑law grid (wide range, fine steps)
    for p in np.linspace(0.2, 1.6, 81):
        seq = _power_law_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        # jittered copy
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Exponential decay grid
    for lam in np.linspace(3e-4, 0.025, 60):
        seq = _exponential_sequence(n, lam)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Sine‑power shapes
    for p in np.linspace(0.2, 2.5, 36):
        seq = _sin_power_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        candidates.append(seq_j)

    # Tent shape (deterministic)
    seq = _linear_tent_sequence(n)
    seq = _project_to_simplex(seq, target_sum)
    candidates.append(seq)

    # Random convex combinations of base shapes
    for _ in range(30):
        seq = _random_combination_sequence(n, rng)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)

    # --------------------------------------------------------------
    # Quick evaluation of the pool
    # --------------------------------------------------------------
    scored = []
    for seq in candidates:
        val = _eval(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])

    best_seq = scored[0][1].copy()
    best_score = scored[0][0]
    print(f"[{time.time():.2f}] Initial best from pool = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – soft‑max GD on the best few candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        top_k = min(8, len(scored))
        gd_time = remaining * 0.30
        gd_deadline = time.time() + gd_time
        per_candidate = gd_time / top_k
        for i in range(top_k):
            if time.time() >= gd_deadline - 0.5:
                break
            seq0 = scored[i][1]
            sub_dead = time.time() + min(per_candidate, gd_deadline - time.time())
            a_opt, val_opt = _annealed_gradient_descent_ext(
                seq0, sub_dead, target_sum,
                max_beta=1e6, step_factor=0.6, gamma=0.995)
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] GD improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase X – sub‑gradient descent on true max‑convolution
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_dead = time.time() + remaining * 0.20
        a_sg, val_sg = _subgradient_descent(best_seq, target_sum, sub_dead,
                                            step_factor=0.55, gamma=0.95)
        if val_sg < best_score:
            best_score, best_seq = val_sg, a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑grad improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – deterministic mass‑transfer
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        mt_dead = time.time() + remaining * 0.25
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                    target_sum,
                                                    start_temp=0.6,
                                                    end_temp=1e-5,
                                                    low_frac=0.04)
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Simulated annealing (random mass transfers)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        sa_dead = time.time() + remaining
        a_sa, val_sa = _simulated_annealing(best_seq, target_sum, sa_dead)
        if val_sa < best_score:
            best_score, best_seq = val_sa, a_sa.copy()
            print(f"[{time.time():.2f}] SA improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – final polish with high‑β soft‑max GD (short)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_fin, val_fin = _annealed_gradient_descent_ext(
            best_seq,
            time.time() + remaining,
            target_sum,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.998,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑process and return
    # --------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=float)
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]

    final_seq = final_arr.tolist()
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
