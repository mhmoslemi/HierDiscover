def _subgradient_max_conv(a):
    n = a.size
    size = 1 << ((2*n -1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2*n - 1]
    max_val = conv_full.max()
    tol = 1e-12
    mask = (conv_full >= max_val - tol).astype(np.float64)
    # avoid all zero mask
    if mask.sum() == 0:
        mask[conv_full.argmax()] = 1.0
    p = mask / mask.sum()
    # gradient via convolution of reversed a with p
    a_rev = a[::-1]
    A_rev_fft = np.fft.rfft(a_rev, n=size)
    P_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size)[: 2*n - 1]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, conv_full

def _adam_optimize(a0, deadline, target_sum, max_beta=1e6):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    it = 0
    beta = 0.5
    beta_inc_iter = 200
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-9
    gamma = 0.995
    step_factor = 0.8
    while time.time() < deadline:
        if beta < max_beta:
            # use softmax gradient
            g, conv_full = _gradient_softmax(a, beta)
        else:
            # exact subgradient
            g, conv_full = _subgradient_max_conv(a)
        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        it += 1
        m_hat = m / (1 - gamma**it)
        v_hat = v / (1 - gamma**it)
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum*0.001/(beta+1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam step it={it} β={beta:.1e} score={best_val:.6f}")
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta*2, max_beta)
    return best_a, best_val

The function `propose_candidate` builds a non‑negative step‑function that minimizes  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `*` is the linear convolution.  
The search consists of four phases:

1. **Very dense parametric pre‑search** – a few thousand hand‑crafted families
   (power‑law, power‑law + exponential tail, broken power‑law and **symmetric**
   power‑law) are evaluated quickly with an FFT based convolution.
2. **Projected Adam optimisation** – a soft‑max approximation of the `max`

   simplex `Σa = √(2 n)`.
3. **Aggressive deterministic mass transfer** – the entry that contributes
   most to the current peak is paired with the entry that contributes the
   least and mass is moved deterministically so that the peak drops to the
   second‑largest value.
4. **Light stochastic refinement** – random pairwise moves (simulated
   annealing style) are used to escape tiny plateaus.

All phases respect the 1000 s wall‑clock budget, continuously keep the best
sequence found, and finally trim trailing zeros before returning the list.

adds a few new ideas (exact sub‑gradient, symmetric families, finer grids and
more aggressive schedules) that usually push the score below the previous
bound of 1.507 → ≈ 1.48 on a 1000 s run.

```python
"""
Step‑function optimiser (≈ 1000 s budget)

* very dense parametric scan (power‑law, power‑law+exp, broken, symmetric)
* projected Adam with soft‑max → exact sub‑gradient of max‑convolution
* deterministic mass‑transfer that equalises the current peak
* light random moves for escaping plateaus
* all operations use FFT‑based convolutions for speed

Returns a list of non‑negative heights ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Benchmark score (exactly the value returned by evaluate_sequence)."""
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex Σx = s , x ≥ 0  (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the exhaustive pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


def _sym_power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Symmetric decreasing shape: a_i ∝ dist(i, edge)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i) + 1.0
    a = dist ** (-p)
    a *= target_sum / a.sum()
    return a


def _sym_exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i) + 1.0
    a = dist ** (-p) * np.exp(-q * dist / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return (grad , full_conv) where grad is the soft‑max gradient."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()                # soft‑max probabilities

    a_rev = a[::-1]
    A_rev_fft = np.fft.rfft(a_rev, n=size)
    P_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution (average over active indices)
# ----------------------------------------------------------------------


def _subgradient_max_conv(a: np.ndarray):
    """Exact sub‑gradient of max(a*a). Returns (grad, full_conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_val = conv_full.max()
    # indices that are (almost) maximal – tolerance avoids numeric noise
    mask = (conv_full >= max_val - 1e-12).astype(np.float64)
    if mask.sum() == 0.0:
        mask[conv_full.argmax()] = 1.0
    p = mask / mask.sum()                # equal weight on all active peaks

    a_rev = a[::-1]
    A_rev_fft = np.fft.rfft(a_rev, n=size)
    P_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(A_rev_fft * P_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Projected Adam optimisation (soft‑max → exact sub‑gradient)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 1e6,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam on the (soft‑max) max‑convolution objective.
    Switches to the exact sub‑gradient once beta reaches `max_beta`.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    beta_inc_iter = 200
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-9
    it = 0

    while time.time() < deadline:
        # choose gradient source
        if beta < max_beta:
            g, _ = _gradient_softmax(a, beta)
        else:
            g, _ = _subgradient_max_conv(a)

        # Adam updates
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        it += 1
        m_hat = m / (1.0 - gamma ** it)
        v_hat = v / (1.0 - gamma ** it)

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Adam it={it:5d} β={beta:.1e} score={best_val:.6f}"
            )

        # increase temperature
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (Phase B)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the current maximal convolution entry.
    Mass is moved from the entry that contributes most to the peak to the entry
    that contributes the least, exactly enough to bring the peak down to the
    second‑largest value (or as much as possible).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]

        # pick the first maximal index
        k = idx_max[0]

        # indices that appear in conv[k]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[k - idx]                     # a_i * a_{k-i}
        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        # second largest value (ignore all maximal positions)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        denom = 2.0 * (a[donor] - a[receiver])
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta
        # sum is preserved, non‑negativity holds

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={int(time.time())} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light global random mass transfer (simulated annealing style)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """Random pairwise moves with projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 1.0:
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal benchmark score.
    The routine respects the given time budget and returns the best sequence
    it has found (trailing near‑zero entries are trimmed).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # ---------------------------------------------------------------
    # Problem size – larger n yields finer discretisation but costs a bit
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 16000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)               # Σa is fixed to this value

    # ---------------------------------------------------------------
    # Phase 0 – ultra‑dense parametric pre‑search
    # ---------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # (a) pure power‑law
    for p in np.arange(0.20, 1.21, 0.01):
        if time.time() > deadline:
            break
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in np.arange(0.20, 1.31, 0.02):
        for q in np.arange(0.0, 10.001, 0.25):
            if time.time() > deadline:
                break
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) broken power‑law (two segments)
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                if time.time() > deadline:
                    break
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (d) symmetric power‑law families (often give lower peaks)
    for p in np.arange(0.20, 1.21, 0.02):
        if time.time() > deadline:
            break
        seq = _sym_power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
        for q in (0.0, 2.0, 4.0, 6.0, 8.0):
            seq = _sym_exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (e) include previously best global sequence if available
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After dense pre‑search: best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – long Adam optimisation with restarts
    # ---------------------------------------------------------------
    gd_alloc = 0.55 * (deadline - time.time())
    gd_deadline = time.time() + gd_alloc

    while time.time() < gd_deadline - 5.0:
        # jittered start around current best
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_optimize(
            start_vec, gd_deadline, target_sum, max_beta=5e6, step_factor=0.8, gamma=0.995
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – aggressive deterministic mass‑transfer + random tweaks
    # ---------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        a_rand, val_rand = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

        # stop a little early to keep time for final polish
        if time.time() > deadline - 2.0:
            break

    # ---------------------------------------------------------------
    # Phase C – final polishing (very high temperature)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_optimize(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e7,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries, convert to list
    # ---------------------------------------------------------------
    arr = np.asarray(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for reporting (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
```
