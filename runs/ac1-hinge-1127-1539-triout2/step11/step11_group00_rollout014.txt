def _subgradient_descent(a0, target_sum, deadline):
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        # compute convolution via FFT
        b = _conv_fft(a)
        k_max = int(np.argmax(b))
        # construct subgradient vector g of size n:
        n = a.shape[0]
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k_max - idx]  # derivative of max entry w.r.t a_i

        # step size schedule: start large and decay
        elapsed = time.time() - start
        total = deadline - start
        eta0 = 2.0
        eta1 = 0.2
        eta = eta0 * (1 - elapsed / total) + eta1 * (elapsed / total)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())
        # jitter occasionally
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target_sum*0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it score {best_val:.6f}")
    return best_a, best_val

"""
Optimising the max‑auto‑correlation of a non‑negative step function.

Approach
--------
1. Choose a problem size n based on the time budget (larger n gives a finer
   discretisation but costs a little more work).
2. Initialise the sequence with a variety of analytic shapes
   (power laws, exponentials, Hann/Hamming windows, a parabolic cap,
   a sine‑window and a random sparse vector).  If a previous best
   construction (`height_sequence_1`) exists it is also considered.
   All candidates are normalised to the required sum √(2n) and the
   best one becomes the warm‑start.
3. A short sub‑gradient descent on the true max‑convolution entry is
   performed (multiplicative updates + jitter) to drive the peak down.
4. The main refinement is a simulated‑annealing “mass‑transfer” loop.
   At each iteration a tiny amount of mass Δ is moved from a donor
   (chosen proportionally to its contribution to the current peak)
   to a receiver (chosen where that contribution is smallest).  The
   convolution is updated in O(n) using the exact formula
   `b' = a'*a' = b - 2Δ a_shifted(donor) + 2Δ a_shifted(receiver) +
   Δ² (self‑terms) – 2Δ² (cross‑term)`.  Acceptance follows a Metropolis
   rule with temperature that linearly cools from ≈ 0.8·current_score.
   Periodic full FFT recomputation removes numerical drift.
5. Finally the sequence is trimmed of trailing zeros, stored globally for
   the next run and returned.

The algorithm respects the 1000 s budget, uses only NumPy, and avoids
any closures or lambda expressions as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Candidate factories (closed‑form shapes)
# ----------------------------------------------------------------------


def _init_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _init_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def _init_sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (i + 0.5) / n)


def _init_hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def _init_hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))


def _init_parabolic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - ((i - centre) / centre) ** 2
    return np.maximum(a, 0.0)


def _init_random_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Place mass on a random subset of indices."""
    m = max(2, int(0.3 * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = rng.random(m)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent on max‑convolution."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]          # derivative of max entry

        # linear schedule for the step size
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.2 * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        # occasional jitter to escape flat regions
        if rng.random() < 0.01:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad score={best_val:.6f}")

    return best_a, best_val

def _mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):

    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = best_val * 0.8                 # initial temperature (≈ 80 % of current score)
    if T0 <= 0:
        T0 = best_val
    last_full = start
    it = 0
    eps = 1e-12

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - frac))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))

        contrib = a[idx] * a[k_max - idx]

        weights = 1.0 / (full_contrib + eps)
        weights[donor] = 0.0                     # cannot receive from itself
        w_sum = weights.sum()
        if w_sum > 0.0:

        else:
            receiver = rng.integers(n)

        if donor == receiver:

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.20, target_sum * 0.002)   # larger than in the original code

    for maker, name in [

    ]:
        a = maker(n)

        val = _score(a)

    except Exception as exc:

    return final_seq
