"""
Step‑function optimizer for the benchmark.

Main ideas
----------

* Choose a problem size n that fits the time budget (larger n → more
  degrees of freedom).  The simplex constraint Σ a_i = √(2 n) is enforced
  exactly by Euclidean projection.

* Phase 0 – ultra‑dense and sparse exploration
    – scan a few classic parametric families (power law, exponential tail,
      broken power law) on a coarse grid.
    – generate many random *sparse* candidates (few hundred non‑zero
      entries, random locations, Dirichlet‑distributed masses).  Sparse
      patterns tend to have a small number of contributions per convolution
      index and therefore a low peak value.
    – keep the best sequence found so far.

* Phase A – long Adam‑style projected gradient descent on a soft‑max
  approximation of the max‑convolution.  The temperature β is increased
  exponentially, the step size is annealed and a small Gaussian jitter is
  added every few iterations to escape plateaus.

* Phase B – sub‑gradient descent on the *exact* max‑convolution.  The active
  set of indices attaining the current maximum is identified and a
  sub‑gradient is computed via a single FFT.  A decaying step size is used
  and the iterate is projected onto the simplex.

* Phase C – greedy deterministic mass‑transfer (the “DMT” step used in the
  reference implementation).  Mass is moved from the entry that contributes
  most to the current peak to the entry that contributes least, reducing the
  peak as fast as possible.

* Phase D – a light global random‑mass‐transfer (simulated‑annealing‑style)
  to jump out of tiny plateaus.

* Phase E – final polishing with a short high‑β Adam run.

All phases respect the wall‑clock deadline (budget_s seconds) and the
best sequence encountered is returned (trailing near‑zero entries are
trimmed).  The routine prints progress messages that are useful for
debugging and for estimating the current bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for all n used)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (full linear convolution)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Benchmark score; because Σa = √(2 n) the expression reduces to max(a*a).
    """
    n = a.shape[0]
    max_b = _max_conv(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the dense pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _broken_power_law_sequence(
    n: int, p1: float, p2: float, alpha: float, target_sum: float
) -> np.ndarray:
    K = max(1, int(alpha * n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.empty_like(idx)
    a[:K] = idx[:K] ** (-p1)
    a[K:] = idx[K:] ** (-p2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Random sparse candidate generator
# ----------------------------------------------------------------------


def _random_sparse_sequence(n: int, k: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Place mass on exactly k distinct positions (Dirichlet‑distributed weights)."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    # random positive proportions, then scale to target_sum
    vals = rng.random(k) + 1e-12
    vals /= vals.sum()
    a[idx] = vals * target_sum
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in Adam GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, conv_full) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = conv_full.max()
    # soft‑max weights over all convolution entries
    exps = np.exp(beta * (conv_full - max_b))
    p = exps / exps.sum()  # probability distribution over indices

    # convolution of a[::-1] with p gives ∂/∂a of Σ p_k (a*a)[k]
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv_full


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 5e5,
    step_factor: float = 0.6,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation of max‑convolution.
    Runs until ``deadline`` is reached.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.1
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # tiny jitter every few hundred iterations
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it} β={beta:.1f} score={best_val:.6f}")

        # anneal β
        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    init_step: float = 0.5,
    step_decay: float = 0.995,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Uses the binary active‑set mask to compute an exact sub‑gradient.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    step = init_step

    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_rev = a[::-1]

    while time.time() < deadline:
        conv = _conv_fft(a)
        M = conv.max()
        # active indices (within a very small tolerance)
        tol = max(M * 1e-12, 1e-12)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0:
            break

        # binary mask for the active set
        mask = np.zeros(2 * n - 1, dtype=np.float64)
        mask[active] = 1.0

        # gradient = 2 * (a_rev * mask) convolution
        a_rev_fft = np.fft.rfft(a_rev, n=size)
        mask_fft = np.fft.rfft(mask, n=size)
        grad_rev = np.fft.irfft(a_rev_fft * mask_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * grad_rev[n - 1 : n - 1 + n]

        # sub‑gradient step
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad step={step:.6f} score={best_val:.6f}")

        step *= step_decay
        # recompute a_rev for the next iteration
        a_rev = a[::-1]

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive deterministic mass‑transfer (Phase C)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Greedy exact reduction of the current maximal convolution entry.
    Moves as much mass as possible from the largest contributor to the
    smallest contributor for the selected peak.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - tol)[0]
        k = idx_max[0]  # pick the first maximal index

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions of candidates to the chosen peak
        contrib = a[k - idx]  # because a_{k-i}
        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        # second largest convolution value (excluding current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        denom = 2.0 * (a[k - donor] - a[k - receiver])
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor], delta_needed * 0.999)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={int(time.time())} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple global random mass‑transfer (exploration, Phase D)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """Light simulated‑annealing style mass moves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline - 0.5:
        # pick a relatively large entry and a relatively small entry
        top_k = max(1, int(0.02 * n))
        low_k = max(1, int(0.02 * n))
        donor = np.argsort(-a)[rng.integers(top_k)]
        receiver = np.argsort(a)[rng.integers(low_k)]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score

    for p in np.arange(0.20, 1.21, 0.02):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # exponential‑power‑law
    for p in np.arange(0.20, 1.21, 0.05):
        for q in np.arange(0.0, 8.01, 0.5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # broken power‑law
    for p1 in np.arange(0.20, 0.81, 0.1):
        for p2 in np.arange(0.80, 2.01, 0.2):
            for alpha in np.arange(0.1, 0.91, 0.1):
                seq = _broken_power_law_sequence(
                    n, float(p1), float(p2), float(alpha), target_sum
                )
                val = _score(seq)
                if val < best_score:
                    best_score, best_seq = val, seq.copy()

    # (b) random sparse candidates
    # try several different sparsities
    sparse_ks = [int(0.03 * n), int(0.05 * n), int(0.08 * n), int(0.12 * n)]
    for k in sparse_ks:
        for _ in range(30):  # 30 random draws per sparsity level
            seq = _random_sparse_sequence(n, k, target_sum, rng)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After Phase 0 pre‑search: best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent (soft‑max)
    # ---------------------------------------------------------------
    # allocate roughly 35 % of the remaining budget
    gd_deadline = time.time() + 0.35 * (deadline - time.time())
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start_vec, gd_deadline, target_sum, max_beta=5e5, step_factor=0.6, gamma=0.995
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase B – Sub‑gradient descent on the exact max
    # ---------------------------------------------------------------
    subgrad_deadline = time.time() + 0.20 * (deadline - time.time())
    while time.time() < subgrad_deadline - 5.0:
        a_opt, val_opt = _subgradient_descent(
            best_seq, subgrad_deadline, target_sum,
            init_step=target_sum * 0.01, step_decay=0.994
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer (core refinement)
    # ---------------------------------------------------------------
    while time.time() < deadline - 10.0:
        a_ref, val_ref = _deterministic_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best = {best_score:.6f}")

        # a short random move phase to get out of tiny plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 4.0:
            break

    # ---------------------------------------------------------------
    # Phase D – Final polishing (high‑β Adam)
    # ---------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries, convert to list
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final external evaluation (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return final_seq
