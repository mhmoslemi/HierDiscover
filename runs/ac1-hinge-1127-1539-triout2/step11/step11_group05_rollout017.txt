"""
propose_candidate – Heuristic search for a non‑negative step‑function that
minimises the maximal entry of its self‑convolution (the value returned by
evaluate_sequence).  The method builds a diverse pool of analytic shapes,
refines the best candidates with soft‑max gradient descent, performs
sub‑gradient descent on the true max‑convolution, then applies a long
deterministic mass‑transfer / simulated‑annealing phase and a final high‑β
soft‑max polishing step.  The code uses only numpy (no external solvers) and
carefully respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used only for the initial pool)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_power_sequence(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    return np.sin(theta) ** p


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (used in GD)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv).
    Returns (grad, full convolution array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    g_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(g_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – momentum GD on soft‑max
# ----------------------------------------------------------------------
def _annealed_gradient_descent_ext(a0: np.ndarray,
                                   deadline: float,
                                   target_sum: float,
                                   max_beta: float = 5e5,
                                   step_factor: float = 0.5,
                                   gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Momentum GD on the soft‑max approximation of max‑convolution.
    The schedule raises β exponentially until `max_beta` and then keeps it.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase X – exact max sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         step_factor: float = 0.55,
                         gamma: float = 0.9,
                         jitter_interval: int = 500,
                         jitter_scale: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of max‑convolution entry
        idx = kmax - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum) + 1e-12
        lr = step_factor * target_sum / norm
        a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * jitter_scale, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic mass‑transfer / simulated annealing
# ----------------------------------------------------------------------
def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.6,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer that reduces the current peak.
    A simple simulated‑annealing acceptance rule is employed.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index where convolution is maximal
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each pair (i, kmax‑i)
        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]
        # choose a receiver among the low‑contributing indices
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        # move a modest amount of mass
        max_delta = a[donor] * 0.5
        # make sure we do not transfer more than needed to lower the peak
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)   # keeps the sum exact

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        # accept if improvement or with Boltzmann probability
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] DMT it={it:5d} new best={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to avoid stagnation
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Optional tiny random hill‑climbing phase (adds a few extra improvements)
# ----------------------------------------------------------------------
def _random_hill_climb(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       start_temp: float = 0.2,
                       end_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Very small‑scale simulated‑annealing move that shifts tiny bits of mass.
    Only used when a few seconds are left.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # weighted choice of donor (more mass ⇒ higher chance)
        donor = rng.choice(n, p=a / target_sum)
        # pick a receiver uniformly among the low‑mass indices
        receiver = rng.choice(n)
        if donor == receiver:
            continue

        # keep delta modest (≤10 % of donor’s mass)
        max_delta = a[donor] * 0.1
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] HillClimb improved to {best_val:.6f}")
        else:
            # Metropolis criterion
            if rng.random() >= math.exp(-(cur - best_val) / max(temp, 1e-12)):
                # reject – revert
                a[donor] += delta
                a[receiver] -= delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises the maximal entry of the
    self‑convolution (the value returned by evaluate_sequence).  The function
    respects the 1000 s time budget, tries several strategies and returns the
    best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = prev_seq.shape[0]
    else:
        # pick a size that is large enough for a good bound but still cheap to
        # handle with the LP‑free methods used here
        if budget_s >= 800:
            n = 12000
        elif budget_s >= 400:
            n = 8000
        else:
            n = 5000

    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Helper for fast evaluation (max‑convolution)
    # ------------------------------------------------------------------
    def _eval(a: np.ndarray) -> float:
        return _max_convolution_fft(a)

    # ------------------------------------------------------------------
    # Build a pool of parametric seeds
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    candidates = []

    # warm‑start using the previous global sequence (if compatible)
    if "height_sequence_1" in globals():
        v = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if v.shape[0] != n:
            v = _resize_sequence(v, n)
        v = _project_to_simplex(v, target_sum)
        candidates.append(v)

    # power‑law family
    for p in np.linspace(0.35, 0.85, 41):
        seq = _power_law_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        # jittered copy
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq2 = np.maximum(seq + jitter, 0.0)
        seq2 = _project_to_simplex(seq2, target_sum)
        candidates.append(seq2)

    # exponential decay family
    for lam in np.linspace(4e-4, 0.012, 30):
        seq = _exponential_sequence(n, lam)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq2 = np.maximum(seq + jitter, 0.0)
        seq2 = _project_to_simplex(seq2, target_sum)
        candidates.append(seq2)

    # sine‑power family
    for p in np.linspace(0.2, 2.0, 20):
        seq = _sin_power_sequence(n, p)
        seq = _project_to_simplex(seq, target_sum)
        candidates.append(seq)
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq2 = np.maximum(seq + jitter, 0.0)
        seq2 = _project_to_simplex(seq2, target_sum)
        candidates.append(seq2)

    # a few random shuffles of the first few seeds (introduce diversity)
    for base in candidates[:4]:
        perm = base.copy()
        rng.shuffle(perm)
        candidates.append(perm)

    # ------------------------------------------------------------------
    # Quick evaluation of the pool
    # ------------------------------------------------------------------
    scored = []
    for seq in candidates:
        val = _eval(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])

    best_seq, best_score = scored[0][1].copy(), scored[0][0]
    print(f"[{time.time():.2f}] Initial best from pool = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – soft‑max momentum GD on the top few candidates
    # ------------------------------------------------------------------
    top_k = min(6, len(scored))
    remaining = deadline - time.time()
    if remaining > 6.0:
        gd_time = remaining * 0.30
        gd_deadline = time.time() + gd_time
        per_candidate = gd_time / top_k
        for i in range(top_k):
            if time.time() >= gd_deadline - 0.3:
                break
            seq0 = scored[i][1]
            sub_dead = time.time() + min(per_candidate, gd_deadline - time.time())
            a_opt, val_opt = _annealed_gradient_descent_ext(
                seq0, sub_dead, target_sum,
                max_beta=5e5,
                step_factor=0.45,
                gamma=0.998
            )
            if val_opt < best_score:
                best_score, best_seq = val_opt, a_opt.copy()
                print(f"[{time.time():.2f}] GD improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase X – sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        sub_dead = time.time() + remaining * 0.25
        a_sg, val_sg = _subgradient_descent(best_seq, target_sum,
                                            sub_dead,
                                            step_factor=0.50,
                                            gamma=0.94)
        if val_sg < best_score:
            best_score, best_seq = val_sg, a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑grad improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – deterministic mass‑transfer / simulated annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        mt_dead = time.time() + remaining * 0.40
        a_mt, val_mt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                    target_sum,
                                                    start_temp=0.7,
                                                    end_temp=1e-6,
                                                    low_frac=0.035)
        if val_mt < best_score:
            best_score, best_seq = val_mt, a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – final polishing with a higher‑β soft‑max GD
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_fin, val_fin = _annealed_gradient_descent_ext(
            best_seq,
            time.time() + remaining,
            target_sum,
            max_beta=1e6,
            step_factor=0.35,
            gamma=0.999,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polishing improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional tiny hill‑climb if a few seconds remain
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        a_hc, val_hc = _random_hill_climb(best_seq, target_sum,
                                          time.time() + remaining)
        if val_hc < best_score:
            best_score, best_seq = val_hc, a_hc.copy()
            print(f"[{time.time():.2f}] Hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final tidy‑up, evaluation and storage
    # ------------------------------------------------------------------
    final_arr = np.array(best_seq, dtype=float)
    nz = np.where(final_arr > 1e-12)[0]
    if nz.size:
        final_arr = final_arr[: nz[-1] + 1]

    final_seq = final_arr.tolist()

    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
