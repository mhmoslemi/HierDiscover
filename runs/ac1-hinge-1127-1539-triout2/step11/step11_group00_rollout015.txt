def solve_lp_for_g(f, M):
    n = f.shape[0]
    # Build A matrix size (2n-1, n)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows indices i to i + n - 1
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')

```
def _lp_step(f, M, target_sum):
    # Build constraint matrix A (size 2n-1 x n)
    n = f.shape[0]
    # Build diagonal matrix using scipy.sparse.diags
    offsets = np.arange(n)
    # data: shape (n, n) each row constant f[i]
    data = np.repeat(f[:, np.newaxis], n, axis=1)
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    # Solve LP
    c = -np.ones(n)  # maximize sum
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = sp.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    # In case of failure, fallback to current f (no change)
    if not res.success:
        return f
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 < 1e-12:
        # fallback to uniform distribution
        g = np.full(n, target_sum / n)
    else:
        g = g0 * (target_sum / sum_g0)
    return g
```

"""
Hybrid step‑function optimizer.

The routine combines several complementary optimisation ideas:

* Phase A – fast exponential (mirror) descent on a soft‑max surrogate of the
  max‑convolution (quickly reduces the peak).

* Phase B – sub‑gradient descent on the true max‑convolution entry (adds
  a more accurate descent direction).

* Phase C – projected gradient descent with random restarts (explores the
  neighbourhood).

* Phase E – alternating linear‑programming refinement.  For a fixed f we
  maximise Σg under the linear constraints (f∗g)≤M, where M=‖f∗f‖∞.
  The optimiser g is then normalised and mixed with f.  Repeating this
  step (with a decaying mixing coefficient) provably does not increase
  the current peak and empirically pushes the solution further.

* Phase D – a long simulated‑annealing “mass‑transfer’’ that moves tiny
  amounts of mass from indices that contribute most to the current
  peak towards low‑height indices.  This final polishing stage removes
  residual spikes.

All phases respect the non‑negativity and sum constraints
(∑a = √(2n)).  The best sequence found before the time budget expires is
returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phases A–C)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Phase A – exponential (mirror) descent on the surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    base_eta = 0.025
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative sub‑gradient descent on the true max‑convolution entry."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        n = a.shape[0]
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        eta_start = 2.0
        eta_end = 0.2
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – projected GD with restarts
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = _project_to_simplex(a0, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.45
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.0015 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray, target_sum: float, deadline: float):
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start, target_sum,
                                            time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase E – alternating linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   subject to   g ≥ 0,   (f ∗ g)_k ≤ M.
    Returns the normalised vector g (∑g = target_sum).
    """
    n = f.shape[0]

    # Build the (2n‑1) × n constraint matrix A where
    # (A g)_k = Σ_i f_i g_{k‑i} = (f ∗ g)_k .
    #   A consists of n constant diagonals (offsets 0 … n‑1) each equal to f_i.
    offsets = np.arange(n)
    # each row of `data` corresponds to one diagonal; repeat f_i n times.
    data = np.repeat(f[:, np.newaxis], n, axis=1)   # shape (n, n)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    # Linear programme (maximise sum(g)  ⇔  minimise –sum(g))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # HiGHS is fast and accepts sparse matrices.
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})

    if not res.success:
        # If the LP fails (should be rare), fall back to the current f.
        return f.copy()

    g0 = res.x
    s = g0.sum()
    if s < 1e-12:                     # degenerate case – use uniform mass
        return np.full(n, target_sum / n)
    return g0 * (target_sum / s)


def _lp_refinement(a0: np.ndarray, target_sum: float, deadline: float):
    """Repeated LP‑based refinements until the time budget is exhausted."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    start = time.time()

    it = 0
    while time.time() < deadline:
        M = _max_convolution(a)

        # LP step
        g = _lp_step(a, M, target_sum)

        # mixing coefficient – starts relatively large and decays
        prog = (time.time() - start) / (deadline - start)
        t_mix = 0.6 * (1.0 - prog) + 0.1      # between 0.6 and 0.1

        a = (1.0 - t_mix) * a + t_mix * g
        a *= target_sum / a.sum()

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] LP it={it} score={best_val:.6f}")

        # occasional tiny jitter to keep dynamics alive
        if it % 15 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – simulated‑annealing mass‑transfer refinement (unchanged)
# ----------------------------------------------------------------------


def _mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """Long‑run annealing that moves tiny amounts of mass from “bad’’ to “good’’ indices."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.30                   # start at ~30 % of the current score

    last_full = start
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – larger contribution ⇒ more likely to be chosen
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – pick an index with relatively small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by the temperature)
        max_delta = min(a[donor] * 0.04, target_sum * 0.001)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        # --- exact O(n) update of the convolution -------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if time.time() - last_full > 10.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional tiny jitter (keeps dynamics alive)
        if it % 2000 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.005, target_sum * 0.0005)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 2500
    elif budget_s >= 500:
        n = 2000
    else:
        n = 1500

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start from the best known construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
    else:
        # simple decreasing power‑law fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent (≈5 % of budget)
    # --------------------------------------------------------------
    exp_deadline = time.time() + 0.05 * (deadline - time.time())
    a_exp, val_exp = _exponential_descent(init, target_sum, exp_deadline)
    best_seq, best_score = a_exp, val_exp
    print(f"[{time.time():.2f}] After exponential descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – subgradient descent (≈10 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.10 * (deadline - time.time())
        a_sub, val_sub = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val_sub < best_score:
            best_seq, best_score = a_sub, val_sub
        print(f"[{time.time():.2f}] After subgradient descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts (≈15 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.15 * (deadline - time.time())
        a_gd, val_gd = _gradient_descent_with_restarts(best_seq,
                                                       target_sum,
                                                       gd_deadline)
        if val_gd < best_score:
            best_seq, best_score = a_gd, val_gd
        print(f"[{time.time():.2f}] After GD restarts best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – alternating LP refinement (≈30 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + 0.30 * (deadline - time.time())
        a_lp, val_lp = _lp_refinement(best_seq, target_sum, lp_deadline)
        if val_lp < best_score:
            best_seq, best_score = a_lp, val_lp
        print(f"[{time.time():.2f}] After LP refinement best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase D – simulated‑annealing mass‑transfer (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, val_ref = _mass_transfer_anneal(best_seq, target_sum, deadline)
        if val_ref < best_score:
            best_seq, best_score = a_ref, val_ref
        print(f"[{time.time():.2f}] After annealing best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
