def propose_candidate(budget_s=1000, **kwargs):
    import time, numpy as np, math
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose n
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 700:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # initial sequence
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(height_sequence_1, dtype=np.float64), n)
    else:
        init = np.arange(1, n+1, dtype=np.float64) ** -0.55
    init = init * (target_sum / init.sum())
    # Make sure sum not zero
    init = np.maximum(init, 0.0)

    # Phase A
    exp_deadline = start + 0.08 * (deadline - start)
    a, best_val = _exponential_descent(init, target_sum, exp_deadline)

    # Phase B
    sub_deadline = time.time() + 0.12 * (deadline - time.time())
    a, val = _subgradient_descent(a, target_sum, sub_deadline)
    if val < best_val:
        best_val = val
        best_seq = a
    else:
        best_seq = a

    # Phase C
    gd_deadline = time.time() + 0.15 * (deadline - time.time())
    a, val = _gradient_descent_with_restarts(best_seq, target_sum, gd_deadline)
    if val < best_val:
        best_val = val
        best_seq = a

    # Phase D
    anneal_deadline = time.time() + 0.55 * (deadline - time.time())
    a, val = _mass_transfer_anneal(best_seq, target_sum, anneal_deadline)
    if val < best_val:
        best_val = val
        best_seq = a

    # final fine-tune subgradient
    final_deadline = deadline
    a, val = _subgradient_descent(best_seq, target_sum, final_deadline)
    if val < best_val:
        best_val = val
        best_seq = a

    # finalize
    arr = np.array(best_seq, dtype=np.float64)
    # truncate trailing zeros
    nz = np.where(arr > 1e-12)[0]
    arr = arr[: nz[-1] + 1] if nz.size else np.array([], dtype=np.float64)
    final_seq = arr.tolist()
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print("Evaluation exception:", exc)
    globals()["height_sequence_1"] = final_seq
    return final_seq
```

"""
Propose a step‑function sequence with a low benchmark score.

The search combines several complementary phases:

* **Phase A – Mirror descent on a soft‑max surrogate** – quickly drives down the
  maximum convolution by multiplicative updates with an exponentially increasing
  temperature (β).  More aggressive jitter and larger step size are used.

* **Phase B – Sub‑gradient descent on the true max‑convolution** – an
  additive‑like update (implemented as a multiplicative step) with a linearly
  decreasing step‑size and stronger jitter.

* **Phase C – Projected gradient descent with random restarts** – optimisation
  on the soft‑max surrogate, now with a larger base learning‑rate and more
  momentum.

* **Phase D – Distance‑biased mass‑transfer annealing** – the dominant phase.
  It repeatedly moves a small amount of mass from indices that contribute most
  to the current convolution peak towards indices that are low and far away.
  The maximum transfer size is enlarged (up to 8 % of the donor height) and the
  temperature schedule is slightly hotter (≈ 40 % of the current score) to allow
  larger exploratory moves.  Periodic full recomputation of the convolution
  guarantees numerical stability.

* **Final fine‑tuning** – another round of sub‑gradient descent using the
  remaining budget.

All phases keep the sequence on the simplex Σa = √(2 n) (the normalisation that
makes the objective equal to the max‑convolution).  The best sequence found is
stored in the global variable ``height_sequence_1`` for warm‑starts in later
runs.

The routine respects the supplied time budget (default 1000 s) and returns a
plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)².  For Σa=√(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used in Phase A and Phase C)
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of the max‑convolution.
    Returns (grad, full_convolution) where grad has the same length as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    grad_fft = np.fft.rfft(a_rev, n=size) * np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(grad_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, b

# Phase A – exponential (mirror) descent on the surrogate
# ----------------------------------------------------------------------


def _exponential_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Multiplicative mirror descent on the soft‑max surrogate."""
    a = a0 * (target_sum / a0.sum())
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    max_beta = 65536.0
    base_eta = 0.035
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        eta = base_eta / (beta + 1.0)
        a = a * np.exp(-eta * g)
        a = a * (target_sum / a.sum())

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0012, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpDescent it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):

def _annealed_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected GD on the soft‑max surrogate with momentum and jitter."""
    a = _project_to_simplex(a0, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 65536.0
    step_base = target_sum * 0.50          # a more aggressive base step
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.93
    it = 0

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.0018 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray, target_sum: float, deadline: float):
    """Run several short GD restarts with random jitter; keep the best."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break
        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _annealed_gradient_descent(start, target_sum,
                                            time.time() + run_budget)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val

# Phase D – distance‑biased mass‑transfer simulated annealing
# ----------------------------------------------------------------------


def _mass_transfer_anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """Long‑run annealing that moves mass from “bad’’ to “good’’ indices."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    T0 = best_val * 0.40                 # slightly hotter start

    idx_array = np.arange(n)
    last_full = start
    it = 0

    # transfer size parameters
    max_delta_factor = 0.08               # up to 8 % of donor height
    max_total_factor = 0.002              # up to 0.2 % of total mass

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total_time
        temperature = max(0.0, T0 * (1.0 - frac))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]

        # donor – chosen proportionally to contribution
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – prefers low height and distance from donor
        inv = 1.0 / (a + 1e-12)
        dist = np.abs(idx_array - donor) + 1.0
        weight = inv * (dist ** -0.7)      # distance bias exponent ≈ 0.7
        prob = weight / weight.sum()
        receiver = rng.choice(n, p=prob)

        if donor == receiver:
            continue

        # size of the transfer
        max_delta = min(a[donor] * max_delta_factor,
                        target_sum * max_total_factor)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0:
            continue

        # --- O(n) update of the convolution ---------------------------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] += -2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic correction
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val < best_val:
            accept = True
        elif temperature > 0.0:
            prob_accept = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < prob_accept:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved score={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if time.time() - last_full > 9.0:
            a = a * (target_sum / a.sum())
            b = _conv_fft(a)
            last_full = time.time()
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved score={best_val:.6f}")

        # occasional tiny jitter to keep dynamics alive
        if it % 2500 == 0:
            i, j = rng.integers(n, size=2)
            if a[i] > 1e-12:
                eps = min(a[i] * 0.01, target_sum * 0.0007)
                a[i] -= eps
                a[j] += eps
                b = _conv_fft(a)
                cur_val = _score(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[{time.time():.2f}] Jitter improved score={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for the evaluator

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but costs time)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 700:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start from the best known construction if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                        dtype=np.float64), n)
    else:
        # simple decreasing power‑law fallback
        init = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))

    init = init * (target_sum / init.sum())
    init = np.maximum(init, 0.0)

    # --------------------------------------------------------------
    # Phase A – exponential (mirror) descent (≈8 % of budget)
    # --------------------------------------------------------------
    exp_deadline = time.time() + 0.08 * (deadline - time.time())
    a, best_val = _exponential_descent(init, target_sum, exp_deadline)
    best_seq = a
    print(f"[{time.time():.2f}] After exponential descent best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase B – sub‑gradient descent (≈12 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.12 * (deadline - time.time())
        a, val = _subgradient_descent(best_seq, target_sum, sub_deadline)
        if val < best_val:
            best_val, best_seq = val, a
        print(f"[{time.time():.2f}] After sub‑gradient descent best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase C – projected GD with restarts (≈18 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = time.time() + 0.18 * (deadline - time.time())
        a, val = _gradient_descent_with_restarts(best_seq,
                                                 target_sum,
                                                 gd_deadline)
        if val < best_val:
            best_val, best_seq = val, a
        print(f"[{time.time():.2f}] After GD restarts best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase D – long simulated‑annealing mass‑transfer (≈55 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a, val = _mass_transfer_anneal(best_seq, target_sum, deadline)
        if val < best_val:
            best_val, best_seq = val, a
        print(f"[{time.time():.2f}] After annealing best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Final fine‑tuning with sub‑gradient descent (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a, val = _subgradient_descent(best_seq, target_sum, deadline)
        if val < best_val:
            best_val, best_seq = val, a
        print(f"[{time.time():.2f}] After final sub‑gradient best score={best_val:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
