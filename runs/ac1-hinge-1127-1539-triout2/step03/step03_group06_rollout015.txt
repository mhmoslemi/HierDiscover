```
def project_onto_simplex(v, s):
    # v array, return projection onto the simplex {x >= 0, sum x = s}
    # Using algorithm from Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Optimizes a non‑negative step sequence to minimise the maximal
self‑convolution (equivalently the external ``evaluate_sequence`` score).
Uses a smoothed max loss (log‑sum‑exp) and Adam‑style projected gradient
descent with many random restarts under a 1000 s wall‑clock budget.

import math
import numpy as np

# optional SciPy for LP refinement (not required)
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (real)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(Fa * Fa, fft_len)[:conv_len]
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of x and y via FFT (real)."""
    n = x.shape[0] + y.shape[0] - 1
    fft_len = _next_pow2(n)
    Fx = np.fft.rfft(x, fft_len)
    Fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(Fx * Fy, fft_len)[:n]
    return conv

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x>=0, sum x = s}.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find the rho (last index where condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero -> uniform distribution
        theta = cssv[-1] / n
    else:
        rho = ind[cond].max()
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _grad_smoothed_max(a: np.ndarray, alpha: float):
    """
    Compute the smoothed max of self‑convolution and its gradient.

    Returns:
        conv   – full self‑convolution (len 2n‑1)
        mmax   – maximum entry of conv
        grad   – ∇_a ( (1/α) log Σ exp(α conv) )
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n-1
    mmax = conv.max()
    # smoothed weights: w_k ∝ exp(α (conv_k - mmax))
    # subtracting mmax improves numeric stability
    exp_w = np.exp(alpha * (conv - mmax))
    w = exp_w / exp_w.sum()                  # shape (2n-1,)

    # Gradient: 2 * correlation of a and w
    a_rev = a[::-1]
    corr_full = _conv_fft(a_rev, w)          # length (n + 2n-1 -1) = 3n-2
    offset = len(w) - 1                      # = 2n-2
    grad = 2.0 * corr_full[offset: offset + n]
    return conv, mmax, grad

def _run_adam(a0: np.ndarray, deadline: float,
              max_iters: int = 5000,
              alpha_max: float = 10.0,
              eta: float = 0.05,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              rng: np.random.Generator = None):
    """
    Adam‑style projected gradient descent on the smoothed max loss.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)           # target ℓ₁‐norm
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_max = float('inf')
    best_vec = a.copy()
    # incremental schedule for smoothing parameter α
    alpha_start = 0.5
    alpha_inc = (alpha_max - alpha_start) / max_iters if max_iters > 0 else 0.0

    for t in range(1, max_iters + 1):
        if time.time() >= deadline - 0.3:
            break

        alpha = min(alpha_max, alpha_start + alpha_inc * t)

        conv, cur_max, grad = _grad_smoothed_max(a, alpha)

        # Adam update (gradient descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a = _project_onto_simplex(a_new, S)

        # Occasionally perturb to escape plateaus
        if rng is not None and (t % 1000 == 0) and (cur_max > best_max * 0.99):
            # add a small zero‑mean Gaussian noise
            noise = rng.normal(scale=a.mean() * 0.02, size=n)
            a = _project_onto_simplex(a + noise, S)

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_vec = a.copy()

    return best_vec, best_max

            best_score = local_max
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        # occasional extra polishing of the current global best
        if (run_id % 5 == 0) and best_seq is not None:
            polish, polish_max = _run_adam(best_seq, deadline,
                                           max_iters=2000, eta=0.03, rng=rng)
            if polish_max < best_score:
                best_score = polish_max
                best_seq = polish.copy()
                print(f"[polish] run={run_id} n={n} new best={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 2)  Final long run on the best candidate (if any) and optional LP.
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_max = _run_adam(best_seq, deadline,
                                         max_iters=10000, eta=0.03, rng=rng)
        best_seq = final_seq
        best_score = final_max

        # optional LP refinement for modest sizes (fast and may improve a little)
        if _SCIPY_AVAILABLE and best_seq.size <= 3000:
            n = best_seq.size
            S = math.sqrt(2.0 * n)
            M = best_score
            # Build H where (H @ g) = conv(best_seq, g)
            H = np.zeros((2 * n - 1, n), dtype=np.float64)
            for i in range(n):
                H[i:i + n, i] = best_seq
            c = -np.ones(n)
            bounds = [(0, None)] * n
            res = linprog(c, A_ub=H, b_ub=np.full(2 * n - 1, M),
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 7})
            if res.success:
                g = res.x
                if g.sum() > 0:
                    g = g * (S / g.sum())
                    t = 0.2
                    best_seq = (1 - t) * best_seq + t * g
                    best_seq = best_seq / best_seq.sum() * S
                    best_score = np.max(_conv_self(best_seq))
                    print(f"[LP polish] improved score to {best_score:.6f}")

        # final external evaluation (the actual scoreboard metric)
        final_eval = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence = {final_eval:.6f}, length = {len(best_seq)}")
        best_score = final_eval
    else:
        # ------------------------------------------------------------------
        # Fallback – a uniform sequence (should never be needed).
        # ------------------------------------------------------------------
        n = 1000
        uniform = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = uniform
        final_eval = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] evaluate_sequence = {final_eval:.6f}")
        best_score = final_eval

    return best_seq.tolist()
