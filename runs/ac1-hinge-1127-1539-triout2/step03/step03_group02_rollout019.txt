    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    base = globals().get('height_sequence_1')
    if isinstance(base, (list, tuple)):
        n = len(base)
    else:
        n = 512   # modest size for fast search
    target_sum = math.sqrt(2.0 * n)

    # start with uniform

    best = cur.copy()

    # Additionally try sparse random start
    for _ in range(5):
        m = max(2, int(0.1 * n))
        idx = np.random.choice(n, m, replace=False)
        sparse = np.zeros(n, dtype=np.float64)
        sparse[idx] = target_sum / m
        score = compute_max_conv(sparse)
        if score < best_score:
            best_score = score
            best = sparse.copy()
    cur = best.copy()

    # Simulated annealing parameters
    T = 0.2
    iteration = 0
    while time.time() < deadline:
        # propose a move
        i, j = np.random.choice(n, 2, replace=False)
        if cur[i] <= 1e-12:

        # random fraction of mass to move
        frac = np.random.rand() * 0.5  # up to 50% of mass at i

        cand[i] -= delta
        cand[j] += delta
        # compute score
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            cur = cand.copy()
            # reset temperature
            T = max(T * 0.99, 1e-4)
            # log progress
            print(f"[{time.time():.2f}] Improved to {best_score:.6f}")

            if np.random.rand() < math.exp(-(cand_score - best_score) / T):

"""
Propose a step‑function (non‑negative coefficient vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

which, because the search enforces sum(a)=√(2 n), is exactly the maximum
value of the linear self‑convolution.  A simple simulated‑annealing /
hill‑climbing loop is used.  The algorithm starts from a few deterministic
initialisations (uniform, a sparse random allocation) and then repeatedly
moves a random amount of mass from one index to another while keeping the
ℓ₁‑norm fixed.  Improvements are always kept; worsening moves are accepted
with a temperature‑controlled probability to escape local minima.
The whole search runs inside the supplied time budget (≈1000 s) and the
best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def compute_max_conv(seq: np.ndarray) -> float:
    """
    Return the maximal entry of the full linear convolution of `seq`
    with itself.  For modest vector lengths (≤ 2048) a direct convolution
    is faster than FFT‑based methods.
    """
    return float(np.convolve(seq, seq).max())


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = target }.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
