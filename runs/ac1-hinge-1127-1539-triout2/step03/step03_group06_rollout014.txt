import time, math, numpy as np
try import scipy.optimize linprog ; set flag.

def full_conv_fft(a):
    n = a.shape[0]
    conv_len = 2*n-1
    fft_len = 1 << (conv_len-1).bit_length()
    Fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(Fa * Fa, fft_len)[:conv_len]
    return conv

def max_conv_and_idx(a):
    conv = full_conv_fft(a)
    max_val = conv.max()
    max_idx = int(np.argmax(conv))
    return conv, max_val, max_idx

def gradient_from_shifts(a, shifts, weights=None):
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    if weights is None:
        weights = np.ones(len(shifts))
    for k, w in zip(shifts, weights):
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        g[mask] += w * 2.0 * a[partner[mask]]
    if weights is not None:
        g /= weights.sum()
    return g

def run_eg(a0, n, deadline, max_iters=300000,
           top_k_start=5, eta0=0.8, beta_mom=0.9,
           decay=0.99997, jitter_prob=0.02,
           smooth_beta=5.0):
    a = a0.copy()
    S = math.sqrt(2.0*n)
    best_a = a.copy()
    best_val = float('inf')
    v = np.zeros_like(a)
    eta = eta0
    it=0
    last_improve=0
    while time.time() < deadline and it < max_iters:
        it+=1
        conv, cur_max, _ = max_conv_and_idx(a)
        # select top_k
        k = min(conv.size, top_k_start)
        if conv.size > top_k_start:
            # get top_k largest indices
            idxs = np.argpartition(conv, -top_k_start)[-top_k_start:]
        else:
            idxs = np.arange(conv.size)
        # compute weights
        # note: we want larger weight for larger conv values.
        raw = conv[idxs] - cur_max
        w = np.exp(smooth_beta * raw)  # raw <=0, so top one gets exp(0)=1, others smaller
        # gradient
        g = gradient_from_shifts(a, idxs, w)
        # momentum
        v = beta_mom * v + (1-beta_mom) * g
        # exponentiated update
        a = a * np.exp(-eta * v)
        # renorm
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)
        # jitter occasionally
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()
        # decay step
        eta *= decay
        # record best
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()
            last_improve = it
        # adapt top_k?
        if it - last_improve > 2000 and top_k_start < 50:
            top_k_start = min(50, top_k_start * 2)
        # early exit if near deadline
        if time.time() >= deadline - 0.2:
            break
    return best_a, best_val

def lp_refine(a, deadline, steps=5, t=0.2):
    if not SCIPY_AVAILABLE or a.shape[0] > 3000:
        return a
    n = a.shape[0]
    S = math.sqrt(2.0*n)
    for _ in range(steps):
        if time.time() >= deadline - 0.5:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        H = np.zeros((2*n-1, n))
        for i in range(n):
            H[i:i+n, i] = a
        # maximize sum g
        c = -np.ones(n)
        res = linprog(c, A_ub=H, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n,
                      method='highs', options={'time_limit': 6})
        if not res.success:
            break
        g = res.x
        s = g.sum()
        if s == 0:
            g = np.full(n, S/n)
        else:
            g = g * (S/s)
        a = (1.0 - t) * a + t * g
        a = a / a.sum() * S
    return a

"""
propose_candidate – search for a non‑negative step function that minimises
evaluate_sequence.  The optimiser is a mirror‑descent (exponentiated‑gradient)
method applied to the exact smooth approximation of  
    f(a) = max_k (a * a)[k] .
The gradient of the smooth max is obtained via a full correlation
computed with FFTs, which makes each iteration O(n log n) and allows
many thousands of steps even for n≈10 000.  Random restarts use a mixture
of uniform, semicircular and exponential initial shapes, occasionally
seeded from the current global best.  Whenever a candidate improves the
best score, a cheap linear‑programming refinement (if SciPy is present and
n≤3000) is applied.  The procedure respects a 1000 s wall‑clock budget
and returns the best sequence found.
"""

import time
import math
import numpy as np

# optional linear‑programming refinement
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using an FFT of length a power of two.
    Returns the convolution array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    L = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(Fa * Fa, L)[:conv_len]
    return conv


def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray,
                         beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation
        φ(a) = log Σ_k exp(beta·(conv[k]-max_conv))
    where conv = a*a.  The gradient equals

        g = 2·corr(a, w) / Σ w

    with w_k = exp(beta·(conv[k]-max_conv)) and corr the full
    linear correlation.  Computation is performed with FFTs.
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp(beta * (conv - max_val))          # weights, longest 2n‑1
    sum_w = w.sum()

    # correlation length = n + (2n‑1) – 1 = 3n‑2
    corr_len = 3 * n - 2
    L = _next_pow2(corr_len)

    # FFTs of a and w with the same zero‑padding length L
    Fa = np.fft.rfft(a, L)
    Fw = np.fft.rfft(w, L)

    # linear correlation a ⋆ w  (full) = ifftr(conj(Fa)·Fw)
    corr_full = np.fft.irfft(np.conj(Fa) * Fw, L)[:corr_len]

    # gradient uses the first n entries of the correlation
    g = 2.0 * corr_full[:n] / sum_w
    return g


def _run_optimize(start: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  eta0: float = 0.7,
                  beta_mom: float = 0.85,
                  decay: float = 0.99996,
                  jitter_prob: float = 0.025,
                  beta_start: float = 1.0,
                  beta_end: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimiser.
    Returns the best vector seen and its maximal self‑convolution entry.
    """
    a = start.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)          # ℓ₁‑norm we keep fixed
    v = np.zeros_like(a)            # momentum buffer
    eta = eta0
    it = 0
    best_a = a.copy()
    best_val = float('inf')
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # self‑convolution (FFT based)
        conv = _conv_self_fft(a)
        cur_max = conv.max()

        # record improvement
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()
            last_improve = it

        # smoothness parameter (annealed from beta_start to beta_end)
        beta = beta_start + (beta_end - beta_start) * min(it / max_iters, 1.0)

        # gradient of the smooth max
        g = _smooth_max_gradient(a, conv, beta)

        # momentum (exponential moving average)
        v = beta_mom * v + (1.0 - beta_mom) * g

        # multiplicative update – guarantees a_i ≥ 0
        a = a * np.exp(-eta * v)

        # renormalise ℓ₁‑norm to S
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to escape tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # decay step size
        eta *= decay

        # safety break
        if time.time() >= deadline - 0.2:
            break

        # optional adaptive increase of jitter probability if stuck
        if it - last_improve > 5000:
            jitter_prob = min(0.1, jitter_prob * 1.3)

    return best_a, best_val


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Dense matrix A where (A @ g)[k] = (f * g)[k].  Used only for LP."""
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _lp_refine(f: np.ndarray, deadline: float,
               t: float = 0.1, max_steps: int = 5) -> np.ndarray:
    """
    Linear‑programming refinement.
    Solve    max Σ g_i    s.t.  g_i ≥ 0  and  (f * g)[k] ≤ M,
    where M = max_k (f * f)[k].  The solution g is mixed with f and
    renormalised.  Repeated a few times can lower the maximal convolution.
    """
    if not _SCIPY_AVAILABLE or f.shape[0] > 3000:
        return f

    n = f.shape[0]
    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σ g_i  ⇔ minimise –Σ g_i
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:                 # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        # convex combination and renormalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f
