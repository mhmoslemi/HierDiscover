def _optimize_adam_smooth(a0, deadline, rng):

    a = a0.copy()
    n = len(a)
    S = np.sqrt(2*n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5  # initial step size

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    best_iter = 0
    iter_cnt = 0
    t = 0
    # hyperparameters
    alpha = 0.5
    alpha_mul = 1.0015
    swap_prob = 0.08
    swap_frac = 0.025
    max_iter = 100000  # huge but will break by time
    while time.time() < deadline and iter_cnt < max_iter:
        iter_cnt += 1
        t += 1
        # compute conv and weights
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)
        # Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection
        a = _project_simplex(a, S)

        # evaluate
        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            best_iter = iter_cnt

        # occasional swap
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_simplex(a_candidate, S)
                cand_max = np.max(_full_convolution(a_candidate))
                # accept if better or with prob
                if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / (cur_max * 0.02 + 1e-12)):
                    a = a_candidate
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        best_iter = iter_cnt

        # schedule updates
        alpha *= alpha_mul

        if iter_cnt - best_iter > 3000:
            break

    return best_a, best_score

def _lp_improve(f, M, rng):
    n = len(f)
    S = np.sqrt(2*n)
    # Build sparse matrix A
    rows = []
    cols = []
    data = []
    for i in range(2*n-1):
        start = max(0, i - n + 1)
        end = min(i, n-1)
        # create arrays for this row
        length = end - start + 1
        rows.extend([i] * length)
        cols.extend(range(start, end+1))
        data.extend(f[start:end+1])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # objective: maximize sum b => minimize -sum b
    c = -np.ones(n)
    ub = np.full(2*n-1, M)
    # solve
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        return f
    b = res.x
    # scale
    if np.sum(b) == 0:
        return f
    scale = S / np.sum(b)
    g = b * scale
    # blend
    t = 0.2
    new_f = (1 - t) * f + t * g
    new_f = _project_simplex(new_f, S)
    return new_f

"""
Search for a non‑negative step function minimizing the maximal entry of its
self‑convolution (the evaluate_sequence score).

Algorithm
---------
* Run many independent restarts with random lengths (≈2000–5000).
* For each restart initialise a non‑negative vector with the prescribed L¹‑norm
  √(2·n) (either from the previous best sequence or from a Dirichlet draw).
* Refine the vector using a projected Adam optimiser on a soft‑max (log‑sum‑exp)
  approximation of the infinity norm of the self‑convolution.
* Occasionally perform a random “mass‑swap’’ move to escape plateaus.
* For moderate sizes (n ≤ 2000) apply a short linear‑programming refinement:
  maximise the ℓ₁‑norm of a new vector g under the current maximum
  convolution bound, then blend g into the current solution.
* Keep the best sequence encountered; at the end evaluate it with the
  black‑box `evaluate_sequence` and return it as a plain list of floats.

The optimiser respects the 1000 s time budget and always returns a feasible
sequence (non‑negative, sum ≈ √(2·n)).  No external I/O is performed.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient term g_j = 2 * Σ_i w_{i+j} * a_i.
    Both a (len n) and w (len 2n-1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reverse for correlation
    conv_len = n + (2 * n - 1) - 1       # = 3n-2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)      # slice giving the required gradient entries
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth max utilities
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights for a convolution vector:
        w_i = exp(alpha * conv_i) / Σ_j exp(alpha * conv_j)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If a base_seq is supplied and long enough, its leading part (with tiny jitter)
    is used; otherwise a Dirichlet/Uniform draw is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # add a minute Gaussian jitter to avoid staying exactly on a previous optimum
        a += rng.normal(scale=max(a.mean(), 1e-6) * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        # Dirichlet(1) – uniform over the simplex
        a = rng.random(n)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Local optimiser (projected Adam on the smooth max)
# ----------------------------------------------------------------------


def _optimize_adam_smooth(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence using projected Adam on a soft‑max surrogate
    of the max‑norm.  Returns the best sequence encountered and its true maximal
    self‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5               # base learning rate

    # smoothing parameter
    alpha = 0.5
    alpha_mul = 1.0015

    # random swap hyper‑parameters
    swap_prob = 0.08
    swap_frac = 0.025

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    best_iter = 0
    t = 0
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1
        t += 1

        # 1. full self‑convolution
        conv = _full_convolution(a)

        # 2. smooth weights (soft‑max)
        w = _smooth_weights(conv, alpha)

        # 3. gradient of the smoothed objective
        grad = _conv_with_weights(a, w)

        # 4. Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # 5. project onto the simplex (keeps sum = √(2n))
        a = _project_simplex(a, S)

        # 6. true max for bookkeeping
        cur_max = np.max(_full_convolution(a))

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            best_iter = iter_cnt

        # 7. occasional random mass‑swap move (simulated‑annealing flavour)
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_simplex(a_candidate, S)
                cand_max = np.max(_full_convolution(a_candidate))
                # accept if improvement or with a Metropolis probability
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.02, 1e-12)):
                    a = a_candidate
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        best_iter = iter_cnt

        # 8. schedule updates
        alpha *= alpha_mul

        # early stopping if stagnated for many iterations
        if iter_cnt - best_iter > 3000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional, used for moderate n)
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray,
               M: float,
               rng: np.random.Generator) -> np.ndarray:
    """
    Given a current sequence f with maximal self‑convolution M,
    solve the LP that maximises Σ b_j under the constraints
      (f * b)_i ≤ M ,   b ≥ 0 .
    The optimiser then blends the LP solution back into f.
    """
    n = f.shape[0]
    S = np.sqrt(2.0 * n)

    # Build the sparse Toeplitz‑type matrix for the convolution constraints:
    #   (f * b)_i = Σ_j f_j * b_{i-j}.
    # For each diagonal offset d = i - j (0 ≤ d ≤ n‑1) the entries are f_j.
    diagonals = [f[: n - d] for d in range(n)]
    offsets = np.arange(n)                     # 0 … n‑1
    A = scipy.sparse.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')

    # Linear programme: minimise - Σ b_j   (i.e. maximise Σ b_j)
    c = -np.ones(n)
    ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast interior‑point / simplex implementation)
    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 5.0}
    )

    if not res.success:
        return f  # fallback – return the original sequence

    b = res.x
    if b.sum() == 0.0:
        return f

    # Rescale to the required ℓ₁‑norm and blend
    scale = S / b.sum()
    g = b * scale
    t = 0.2                                   # modest blending factor
    new_f = (1.0 - t) * f + t * g
    new_f = _project_simplex(new_f, S)
    return new_f

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # Preserve a previously discovered sequence if available
    base_seq = None
    if 'height_sequence_1' in globals():
        try:

        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    while time.time() < deadline:
        attempt += 1

        # -------- problem size -------------------------------------------------
        n = int(rng.integers(1800, 5000))   # length between ~2k and 5k

        # -------- initialise ----------------------------------------------------
        use_base = base_seq is not None and rng.random() < 0.5
        init_base = base_seq if use_base else None
        a0 = _initial_sequence(init_base, n, rng)

        slice_deadline = min(time.time() + 12.0, deadline)   # ~12 s per attempt

    return final_seq
