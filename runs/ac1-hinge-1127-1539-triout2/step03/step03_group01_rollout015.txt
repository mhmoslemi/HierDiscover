def _lp_step(f, max_val):
    n = len(f)
    # Build A matrix
    A = np.empty((2*n-1, n), dtype=float)
    A[:] = 0.0
    for k in range(n):
        A[k:k+n, k] = f
    b = np.full(2*n-1, max_val * (1 + 1e-9))
    # maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a candidate step‑function sequence.

The search combines two complementary ideas:

1. **LP expansion** – given a current feasible vector `f`,
   we solve a linear programme that maximises the ℓ₁‑mass of a new
   vector `g` while keeping the convolution `f*g` bounded by the
   current maximal self‑convolution `M = max(f*f)`.  Mixing a fraction
   of `g` into `f` (followed by projection onto the ℓ₁‑simplex) strictly
   reduces the objective in practice.

2. **Fast projected gradient descent** – after the LP moves we
   fine‑tune the sequence with the soft‑max gradient method from the
   previous state‑of‑the‑art implementation.  This quickly polishes
   the solution to a (locally) minimal maximal self‑convolution entry.

The routine runs many short independent restarts within the supplied
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Initialise a feasible vector of length n with ℓ₁‑norm = √(2·n)
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=max(a.mean(), 1e-12) * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Soft‑max (or hard‑max) gradient step for the ∞‑norm of the
# self‑convolution, followed by projection onto the simplex.
# ----------------------------------------------------------------------
def _softmax_update(a: np.ndarray, tau: float, eta: float) -> tuple[np.ndarray,
                                                                  np.ndarray,
                                                                  float]:
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau <= 1e-12:                     # hard‑max (sub‑gradient)
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:                   # diagonal term appears once
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]
    else:                                # smooth soft‑max
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # weights ≈ soft‑max

        a_rev = a[::-1]
        m = 1 << ((len(w) + n - 1).bit_length())
        Fw = np.fft.rfft(w, n=m)
        Fa_rev = np.fft.rfft(a_rev, n=m)
        conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n - 1]
        g = 2.0 * conv[n - 1:n - 1 + n]

    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    b_new = _full_convolution_fft(a_proj)
    max_new = np.max(b_new)

    return a_proj, b_new, max_new


# ----------------------------------------------------------------------
# Local optimiser (the fast projected‑gradient method from the
# previous state‑of‑the‑art code).  It returns the best vector seen
# and its maximal convolution entry.
# ----------------------------------------------------------------------
def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters (chosen to be aggressive yet stable)
    eta = 0.6
    eta_decay = 0.9994
    move_prob = 0.15
    epsilon_move = 0.05
    temp = cur_max * 0.12
    temp_decay = 0.9995
    jitter_prob = 0.015
    jitter_scale = 0.025
    tau = cur_max
    tau_decay = 0.9994

    while time.time() < deadline:
        # --------------------------------------------------------------
        # random mass move (simulated annealing)
        # --------------------------------------------------------------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = epsilon_move * a[i]
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.clip(cand, 0.0, None)
            cand = _project_to_simplex(cand, S)
            b_cand = _full_convolution_fft(cand)
            cand_max = np.max(b_cand)

            accept = False
            if cand_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                a = cand
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temp *= temp_decay

        else:
            # -----------------------------------------------------------
            # projected soft‑max gradient step
            # -----------------------------------------------------------
            a, b, cur_max = _softmax_update(a, max(tau, 1e-12), eta)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # occasional jitter
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=max(a.mean(), 1e-12) * jitter_scale,
                               size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _full_convolution_fft(a)
            cur_max = np.max(b)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # schedules
        eta *= eta_decay
        tau *= tau_decay
        # safety break if almost no progress
        if time.time() > deadline - 0.1:
            break

    return best_a, best_max


# ----------------------------------------------------------------------
# Linear‑programming step: maximise sum(g) subject to f*g ≤ M.
# Returns the raw solution `g` (non‑negative, not normalised).
# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    n = f.shape[0]
    # construct A such that (f * g)[i] = Σ_j  f[j] * g[i-j]
    # column k contains f shifted by k
    A = np.zeros((2 * n - 1, n), dtype=float)
    for k in range(n):
        A[k:k + n, k] = f
    b = np.full(2 * n - 1, M * (1.0 + 1e-9))   # tiny slack for numeric safety
    c = -np.ones(n)                           # maximise sum(g) ↔ minimise -sum(g)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                      method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
    except Exception:                         # safeguard – linprog may raise
        return None
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# One LP‑descent phase (alternating LP move + projection).
# Returns the best vector and its max‑convolution value.
# ----------------------------------------------------------------------
def _lp_descent(f0: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    f = f0.copy()
    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    best_f = f.copy()
    best_max = np.max(_full_convolution_fft(f))

    mix_factor = 0.3                # initial mixing weight
    mix_decay = 0.96                # slowly reduce if progress stalls

    while time.time() < deadline:
        cur_max = np.max(_full_convolution_fft(f))
        g = _solve_lp(f, cur_max)
        if g is None:
            break

        # Mix a fraction of the LP solution and re‑project onto the simplex
        f_candidate = (1.0 - mix_factor) * f + mix_factor * g
        f_candidate = _project_to_simplex(f_candidate, S)
        cand_max = np.max(_full_convolution_fft(f_candidate))

        if cand_max < best_max - 1e-7:
            best_max = cand_max
            best_f = f_candidate.copy()
            f = f_candidate               # accept improvement
            mix_factor = min(0.5, mix_factor * 1.02)   # be a bit bolder
        else:
            mix_factor = max(0.02, mix_factor * mix_decay)

        # stop early if time is almost exhausted
        if time.time() > deadline - 2.0:
            break

    return best_f, best_max

    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # Choose a problem size.  Wider range improves chance of hitting a
        # better geometry while keeping LP solves affordable.
        if base_seq is not None:
            max_len = max(800, min(5000, int(base_seq.size * 1.3)))
        else:
            max_len = rng.integers(1200, 3001)   # 1200‑3000 roughly
        n = rng.integers(800, max_len + 1)

        # Initialise a feasible vector
        a0 = _initial_sequence(base_seq, n, rng)

        # Allocate a slice of the remaining time for this restart
        slice_deadline = min(time.time() + 35.0, deadline)

        # ------------------------------------------------------------------
        # Phase 1 – LP‑descent (fast, improves the bound in a few steps)
        # ------------------------------------------------------------------
        lp_deadline = slice_deadline - 12.0        # reserve time for the second phase
        if lp_deadline > time.time():
            a_lp, lp_max = _lp_descent(a0, lp_deadline, rng)
        else:
            a_lp, lp_max = a0, np.max(_full_convolution_fft(a0))

        # ------------------------------------------------------------------
        # Phase 2 – fine‑tune with projected gradient descent
        # ------------------------------------------------------------------
        a_opt, opt_max = _optimize_local(a_lp, slice_deadline, rng)

        # ------------------------------------------------------------------
        # Keep the best overall result (using the external black‑box)
        # ------------------------------------------------------------------
        final_seq = a_opt.tolist()
        external_score = evaluate_sequence(final_seq)
        if external_score < best_overall_score:
            best_overall_score = external_score
            best_overall_seq = final_seq
            print(f"[global] attempt={attempt} n={n} "
                  f"new best external={best_overall_score:.6f} "
                  f"internal max={opt_max:.6f}")

        # early exit if we already beat the target
        if best_overall_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found (or a fallback)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = [1.0]

    # expose the result for potential reuse by later runs
    globals()['height_sequence_1'] = best_overall_seq
    print(f"[final] best external score = {best_overall_score:.6f} "
          f"length = {len(best_overall_seq)}")
    return best_overall_seq
