"""
Optimisation of a non‑negative step function (a finite sequence) that
minimises the maximum entry of its self‑convolution after ℓ¹‑normalisation.
The algorithm alternates between a linear‑programming update (the “LP
step” described in the statement) and a short simulated‑annealing
refinement.  The LP step solves

    maximise   Σ b_j
    subject to  (f * b)_i ≤ M   for all i,   b_j ≥ 0 ,

where f is the current sequence, M = max(f*f) and * denotes linear
convolution.  After solving, b is rescaled to the required ℓ¹‑norm and
mixed with f.  Repeating this reduces the maximal self‑convolution.
The whole search respects the 1000 s budget and returns the best sequence
found as a plain Python list.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT implementation)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al., 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming (LP) step
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    Solve the LP
        maximise Σ b_j
        subject to (f * b)_i ≤ M   ∀ i,
                    b_j ≥ 0.
    The convolution matrix for fixed f is built explicitly.
    The returned vector b is rescaled to have ℓ¹‑norm = target_sum.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build the convolution matrix T (size m × n) such that T @ b = f * b
    T = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = f

    # LP variables
    b = cp.Variable(n, nonneg=True)

    # Constraints: (f * b)_i ≤ M  for all i
    constraints = [T @ b <= M + 1e-12]

    # Objective: maximise total mass Σ b_j
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # Try a fast first‑order solver; fall back to SCS if needed
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=2000, verbose=False)
        except Exception:
            # In the very unlikely case of total failure return a uniform vector
            return np.full(n, target_sum / n, dtype=np.float64)

    if b.value is None:
        # Solver failed to produce a solution – fall back to uniform
        return np.full(n, target_sum / n, dtype=np.float64)

    b_opt = np.asarray(b.value, dtype=np.float64).ravel()
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        return np.full(n, target_sum / n, dtype=np.float64)

    # Rescale to the required ℓ¹‑norm
    return b_opt * (target_sum / sum_b)


# ----------------------------------------------------------------------
# Simulated‑annealing (SA) refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that moves a random fraction of mass between two indices.
    Accepts uphill moves with probability exp(-Δ/T).  The temperature
    decays geometrically.  Returns the best sequence encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    # Initialise temperature as a small fraction of the current score
    T = cur_score * 0.02 + 1e-12
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick two distinct indices
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (up to 15 %) of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        # geometric cooling
        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Main refinement routine (LP + optional SA)
# ----------------------------------------------------------------------


def _refine_sequence_lp_sa(f0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Starting from f0 (already ℓ¹‑normalised), repeatedly perform the LP
    update and a convex combination step.  Mixing weight t decays over
    time.  When the deadline is near, a short SA run is performed.
    Returns the best sequence encountered and its maximal self‑convolution.
    """
    f = _project_to_simplex(f0, np.sqrt(2.0 * f0.size))
    n = f.size
    target_sum = np.sqrt(2.0 * n)

    best_f = f.copy()
    best_M = _max_convolution_fft(f)

    # mixing weight parameters
    t = 0.5          # initial convex‑combination weight
    t_min = 0.02
    decay = 0.995

    iteration = 0
    while time.time() < deadline:
        M = _max_convolution_fft(f)

        # update global best if improved
        if M < best_M:
            best_M = M
            best_f = f.copy()
            print(f"[{time.time():.1f}] LP improved best M: {best_M:.6f}")

        # LP step -> new candidate g
        g = _lp_step(f, M, target_sum)

        # convex combination (still respects the ℓ¹‑norm)
        f = (1.0 - t) * f + t * g
        f = _project_to_simplex(f, target_sum)

        # decay mixing weight
        t = max(t_min, t * decay)

        iteration += 1
        # a very cheap stop‑condition: if we have less than 0.5 s left,
        # break to give the SA phase a chance.
        if time.time() + 0.5 >= deadline:
            break

    # Final SA refinement on the best sequence if time permits
    if time.time() < deadline:
        f_sa = _simulated_annealing(best_f, deadline)
        M_sa = _max_convolution_fft(f_sa)
        if M_sa < best_M:
            best_M = M_sa
            best_f = f_sa.copy()
            print(f"[{time.time():.1f}] SA final improved M: {best_M:.6f}")

    return best_f, best_M


# ----------------------------------------------------------------------
# Candidate proposal entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑maximum self‑convolution step‑function.
    The routine runs for at most ``budget_s`` seconds, trying a small
    pool of sequence lengths.  For each length it performs:
        * an ℓ¹‑normalised initialisation (uniform with tiny noise),
        * a series of LP updates with decreasing mixing weight,
        * a final simulated‑annealing polishing step.
    The best sequence found (as a plain Python list) is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_M = np.inf

    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_M
        n = seq.size
        target = np.sqrt(2.0 * n)
        seq = _project_to_simplex(seq, target)
        M = _max_convolution_fft(seq)
        if M < best_M:
            best_M = M
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New overall best M = {best_M:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 1.  Use the globally supplied construction if it exists
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        init = np.asarray(global_seq, dtype=np.float64)
        _maybe_update(init)
        refined, _ = _refine_sequence_lp_sa(init, deadline)
        _maybe_update(refined)

    # ------------------------------------------------------------------
    # 2.  Explore a handful of lengths with fresh starts
    # ------------------------------------------------------------------
    lengths = [800, 1000, 1200, 1500, 1800]   # chosen to balance quality / cost

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # uniform start plus a tiny random perturbation
        f0 = np.full(n, target_sum / n, dtype=np.float64)
        f0 += rng.normal(scale=target_sum * 0.001, size=n)
        f0 = _project_to_simplex(f0, target_sum)

        _maybe_update(f0)

        # run the LP+SA refinement for the remaining budget
        refined_f, _ = _refine_sequence_lp_sa(f0, deadline)
        _maybe_update(refined_f)

    # ------------------------------------------------------------------
    # 3.  Final clean‑up (trim tiny trailing entries)
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best M = {best_M:.6f} length = {best_seq.size if best_seq is not None else 0}")
    return best_seq.tolist() if best_seq is not None else []
