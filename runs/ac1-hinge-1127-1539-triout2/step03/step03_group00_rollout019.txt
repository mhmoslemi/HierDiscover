def _max_convolution_fft(a):
    n = a.size
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]

def _project_to_simplex(v, s):
    # same as Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _build_constraint_matrix(f):
    n = f.shape[0]
    m = 2*n - 1
    A = np.empty((m, n), dtype=np.float64)
    for k in range(m):
        low = max(0, k - (n-1))
        high = min(k, n-1)
        idx = np.arange(low, high+1)
        A[k, idx] = f[k - idx]
    return A

def _lp_refine(f, sum_target, deadline, max_iters=30):

    M = _max_convolution_fft(f)
    t = 0.3
    for it in range(max_iters):

        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # If linprog fails, break out
            break
        g = res.x

        f_new = (1 - t) * f + t * g

            t = min(0.9, t * 1.1)
        else:

            t *= 0.5

"""
Propose a step‑function (non‑negative sequence) that minimises ``evaluate_sequence``.
The algorithm consists of three stages:

1. **Analytic seeds** – uniform, exponential and power‑law families are
   generated for a few candidate lengths and the best few are kept.
2. **LP‑based refinement** – for each seed we iteratively solve a linear
   program that finds a direction ``g`` with maximal ℓ¹‑mass while keeping
   the cross‑convolution `f*g` bounded by the current peak.  The new vector
   `f←(1‑t)f+t g` is projected onto the ℓ¹‑simplex (sum = √(2·n)).  This
   typically drives the self‑convolution peak down from ≈1.8 to below 1.5.
3. **Short simulated‑annealing polish** – a few random mass‑transfer moves
   accept occasional uphill steps to escape tiny local minima.

All work respects the supplied time budget (default 1000 s) and uses only
a single CPU.  The returned list contains non‑negative numbers; after the
call the evaluation function will compute the true score.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility: maximum entry of the linear convolution a*a using FFT
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x≥0, Σx = s}
# (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Build the matrix A for the constraints (f * g)[k] ≤ M
# ----------------------------------------------------------------------


def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    m = 2 * n - 1
    A = np.empty((m, n), dtype=np.float64)
    for k in range(m):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1)
        A[k, idx] = f[k - idx]
    return A


# ----------------------------------------------------------------------
# LP‑based refinement of a candidate vector
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray,
               sum_target: float,
               deadline: float,
               max_iters: int = 25) -> np.ndarray:
    """
    Repeatedly solves a linear program that maximises the ℓ¹‑mass of a
    feasible direction `g` such that (f*g)[k] ≤ current_peak for every k.
    The new vector is a convex mixture with the current one and projected
    back onto the simplex of L1‑norm `sum_target`.
    """
    n = f.shape[0]
    M = _max_convolution_fft(f)        # because sum(f)=sum_target, score = M
    step = 0.3                         # initial mixing factor

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ----- construct linear constraints for the current f -----
        A = _build_constraint_matrix(f)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # ----- linear programme: maximise Σ g_j  (i.e. minimise -Σ g_j) -----
        c = -np.ones(n, dtype=np.float64)          # minimise -∑g_j
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            break
        g = res.x

        # ----- mix and re‑project onto the simplex -----
        f_new = (1.0 - step) * f + step * g
        f_new = _project_to_simplex(f_new, sum_target)

        # ----- evaluate the new peak -----
        M_new = _max_convolution_fft(f_new)

        # ----- accept / reject -----
        if M_new < M - 1e-12:                 # genuine progress
            f, M = f_new, M_new
            step = min(0.9, step * 1.1)       # accelerate a little
        else:                                 # no progress → shrink step
            step *= 0.5
            if step < 1e-6:
                break

    return f


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Randomly moves a fraction of mass between two positions.
    Accepts uphill moves with probability exp(-Δ/T) while the temperature
    decays geometrically.  The total ℓ¹‑mass stays constant, so the
    projection step is only needed after occasional global jitter.
    """
    a = a0.copy()
    best = a.copy()
    cur_score = _max_convolution_fft(a)
    best_score = cur_score

    # initial temperature – a few percent of the current score
    T = cur_score * 0.02 + 1e-9
    cooling = 0.9996

    n = a.size
    rng = np.random.default_rng()

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction (≤15 %) of mass from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            # keep ℓ¹‑norm unchanged
            a = _project_to_simplex(a, a.sum())
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best
