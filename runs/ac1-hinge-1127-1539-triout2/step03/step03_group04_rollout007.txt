"""
A fast multi‑stage optimiser for the “max‑convolution / ℓ¹‑norm’’ benchmark.

Stages
-------
1. **Parametric initialisation** – generate a rich set of candidates from a
   three‑parameter family
        a_i = (i + offset)^(–p) · exp(–q·i)
   for several sequence lengths.  Each candidate is projected onto the simplex
   (ℓ¹‑norm = √(2·n)) and scored by the exact objective.

2. **Sub‑gradient descent on the true max‑convolution** – the active convolution
   index `k* = argmax (a*a)` supplies a sub‑gradient
        g_j = 2·a_{k*‑j}   (if 0 ≤ k*‑j < n, else 0).
   A diminishing step‑size with momentum and Euclidean projection onto the
   simplex drives the sequence to lower peak values.  No smoothing is used,
   therefore the descent follows the exact objective.

3. **Guided hill‑climbing** – after the descent the dominant peak is located.
   Mass is moved from indices that contribute heavily to that peak towards
   indices with low contribution.  Only improving moves are kept; a modest
   random jitter is injected occasionally to escape shallow plateaus.

The routine respects the global 1000 s time budget, keeps the best sequence
found and returns it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # due to finite‑precision we may be off by ≈1e‑12; re‑normalise
    w_sum = w.sum()
    if w_sum > 0:
        w *= s / w_sum
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (exact, via FFT)."""
    return float(_full_convolution_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Fast exact objective for a where Σa = √(2·n):
        score = max_k (a*a)[k] .
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    # The caller normalises the sum to √(2·n); still guard against degenerate case.
    if a.sum() < 0.01:
        return np.inf
    return max_b


# ----------------------------------------------------------------------
# Parametric family for initial candidates
# ----------------------------------------------------------------------


def _parametric_sequence(
    n: int, p: float, q: float, offset: float, target_sum: float
) -> np.ndarray:
    """
    a_i = (i + offset)^(–p) * exp(–q·i),  i = 0 … n‑1,
    subsequently projected onto the simplex of sum = target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) * np.exp(-q * i)
    # normalise to the required ℓ¹‑norm
    a = a * (target_sum / a.sum())
    a = _project_to_simplex(a, target_sum)
    return a


def _generate_initial_candidates(
    lengths: list[int],
    p_vals: np.ndarray,
    q_vals: np.ndarray,
    offset_vals: np.ndarray,
    topk: int,
    deadline: float,
):
    """
    Sweep the parametric family over the supplied lengths and keep the
    `topk` best raw candidates for each length.
    Returns a list of (score, array, n).
    """
    candidates = []
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        best_per_len = []  # temporary list of (score, array)

        for p in p_vals:
            for q in q_vals:
                for off in offset_vals:
                    if time.time() >= deadline:
                        break
                    a = _parametric_sequence(n, p, q, off, target_sum)
                    val = _score(a)
                    best_per_len.append((val, a.copy()))
        # keep the top‑k
        best_per_len.sort(key=lambda x: x[0])
        for val, a in best_per_len[:topk]:
            candidates.append((val, a.copy(), n))
    return candidates


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective:
        minimise max_k (a*a)[k]  with a_i ≥ 0, Σa = √(2·n).

    Uses the active index `k* = argmax (a*a)` to build a sub‑gradient
    g_j = 2·a_{k*‑j} (0 when the shifted index is out of range).
    A diminishing step‑size with momentum is employed; the iterate is
    projected onto the simplex after each step.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    # hyper‑parameters
    step0 = target_sum * 0.4          # base step scale
    gamma = 0.9                       # momentum decay
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # build sub‑gradient (vectorised)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            # diminishing step (1/√t) multiplied by base scale,
            # then combined with momentum.
            step = step0 / (np.sqrt(it + 1) * norm_g)
            momentum = gamma * momentum + step * g
            a = a - momentum

        # enforce feasibility
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 500 == 0 and it > 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] SGD improved to {best_val:.6f} (it={it})")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing improvement
# ----------------------------------------------------------------------


def _guided_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from indices that contribute heavily to
    the current peak towards indices that contribute little.  Only moves
    that improve the objective are retained.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.12  # allow a larger transfer than the original code

    while time.time() < deadline:
        conv = np.convolve(a, a)  # O(n) for the current size (n ≲ 1e4)
        k_max = int(np.argmax(conv))

        # sensitivity vector: contribution of each a_i to conv[k_max]
        sens = np.zeros(n, dtype=np.float64)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        sens[idx] = a[k_max - idx]

        weight = sens * a
        if weight.max() == 0:
            break

        donor = int(np.argmax(weight))
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver or a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] GH improved to {best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the benchmarking harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine consists of:

    1. Parametric grid search over a three‑parameter family for several lengths.
    2. Projected sub‑gradient descent on the exact max‑convolution objective.
    3. Guided hill‑climbing refinement.

    The best sequence discovered before the time budget expires is returned.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Stage 1 – parametric candidate generation
    # ------------------------------------------------------------------
    lengths = [1200, 2400, 4800, 8000, 12000]  # diverse sizes, still tractable
    p_vals = np.linspace(0.38, 0.68, 13)      # step ≈0.025
    q_vals = np.linspace(0.0, 0.0015, 6)     # exponential tail strength
    offset_vals = np.array([0.5, 1.0, 2.0, 4.0])
    topk_per_len = 3

    init_deadline = time.time() + (budget_s * 0.15)
    raw_candidates = _generate_initial_candidates(
        lengths, p_vals, q_vals, offset_vals, topk_per_len, init_deadline
    )

    # Add a simple uniform candidate as a safety net
    for n in lengths:
        target_sum = np.sqrt(2.0 * n)
        a_uni = np.full(n, target_sum / n, dtype=np.float64)
        raw_candidates.append((_score(a_uni), a_uni.copy(), n))

    # keep the globally best few raw candidates
    raw_candidates.sort(key=lambda tup: tup[0])
    shortlisted = raw_candidates[:7]  # a modest number to pass to stage 2

    # ------------------------------------------------------------------
    # Stage 2 – sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for base_score, a0, n in shortlisted:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # give each candidate up to 40 s but never exceed remaining / 2
        sub_deadline = time.time() + min(40.0, remaining / 2.0)

        a_opt, val = _subgradient_descent(a0, sub_deadline)

        # ------------------------------------------------------------------
        # Stage 3 – guided hill‑climbing refinement
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_opt, val = _guided_hill_climb(a_opt, deadline)

        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(
                f"** GLOBAL BEST ** n={n}  start={base_score:.6f}  after={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Return as plain Python list for the external scorer
    return best_seq.tolist()
