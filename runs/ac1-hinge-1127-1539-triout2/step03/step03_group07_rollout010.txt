"""
Propose a candidate step function minimizing the max self‑convolution.

Algorithm:
1. Start from a random (Dirichlet) or inherited sequence with Σa = √(2·n).
2. Iteratively *refine* using a linear program:
   - For the current f compute M = max(f*f).
   - Solve LP: maximise Σ g_j   s.t.    f * g ≤ M  and g ≥ 0.
   - Rescale g to have the required L1‑norm and update f ← (1‑t)f + t g
     with a decreasing step size t.  The update never increases the
     maximum convolution, so M is non‑increasing.
3. After a few LP steps a short stochastic hill‑climbing (simulated‑annealing)
   is applied to fine‑tune the vector.
4. Multiple restarts with different lengths (up to 4000) are performed
   until the time budget (≈1000 s) is exhausted.  The best vector found
   (lowest score) is returned.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse

# ----------------------------------------------------------------------
# Convolution helper – maximum entry of a * a
# ----------------------------------------------------------------------
def conv_max(a: np.ndarray) -> float:
    """Return max entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return float(np.max(conv))


# ----------------------------------------------------------------------
# Small simulated‑annealing hill‑climber (used after the LP phase)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """Simple annealing that moves a random fraction of mass between two indices."""
    a = a_start.copy()
    cur_max = conv_max(a)
    best_a = a.copy()
    best_max = cur_max
    n = a.shape[0]
    temperature = max(0.05 * cur_max, 1e-4)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        frac = rng.uniform(0.05, 0.20)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta

# ----------------------------------------------------------------------
# Linear‑programming refinement of a single vector
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray,
              deadline: float,
              rng: np.random.Generator,
              max_lp_iters: int = 12,
              t0: float = 0.2,
              t_decay: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinements.
    Returns the refined vector and its max‑convolution value.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    # normalise to the exact L1 norm required by the scoring function
    a = a * (S / a.sum())
    M = conv_max(a)
    t = t0

    # pre‑compute sparse pattern (rows, cols) – column i appears in n consecutive rows
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        pos += n
    # shape of the constraint matrix
    shape = (2 * n - 1, n)

    for it in range(max_lp_iters):
        if time.time() > deadline:
            break

        # data = repeat each coefficient a_i exactly n times (the same pattern as rows/cols)
        data = np.repeat(a, n)
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=shape)

        # linear programme: maximise sum(g)  s.t.   A @ g <= M   ,   g >= 0
        g = cp.Variable(n)
        constraints = [A @ g <= M, g >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        # ECOS is a lightweight LP solver that works well for these sizes
        prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)

        if g.value is None:
            # Solver failed – abort the refinement loop
            print("[LP] solver failed, terminating refinement")
            break

        g_val = np.maximum(0.0, np.array(g.value).reshape(-1))
        sum_g = g_val.sum()
        if sum_g == 0:
            break

        # rescale to the required L1 norm
        g_scaled = g_val * (S / sum_g)

        # convex combination
        a_new = (1.0 - t) * a + t * g_scaled
        a_new = a_new * (S / a_new.sum())   # re‑normalise (numeric safety)

        new_M = conv_max(a_new)

        # accept only if the max actually decreased; otherwise shrink the step size
        if new_M < M - 1e-12:
            a = a_new
            M = new_M
            print(f"[LP refine] n={n} iter={it} t={t:.4f} max={new_M:.6f}")
            t *= t_decay          # slowly reduce the mixing weight
        else:
            t *= 0.5               # be more cautious
            if t < 1e-4:
                break

    return a, M

"""
Propose a candidate step‑function with a low evaluation score.

Algorithm
---------
1.  Pick a length n (several values are tried, favouring the larger ones).
2.  Initialise a non‑negative vector a with Σa = √(2·n) – either a random Dirichlet
    sample or the best vector found so far of the same length.
3.  **LP refinement** – repeat a few times:
        * Compute M = max(a*a).
        * Solve the linear programme
              max   Σ g_j
              s.t.  a * g ≤ M      (element‑wise) ,
                    g ≥ 0 .
          The constraint matrix is a sparse Toeplitz matrix that can be built
          in O(n²) time.
        * Scale the solution to the required L₁ norm and update
              a ← (1‑t)·a + t·g
          with a decreasing step size t (the update never increases the
          maximal convolution).
4.  **Short stochastic hill‑climbing** – move a random fraction of mass from
    one index to another; accept improving moves always and worsening moves
    with a temperature‑dependent probability.  This fine‑tunes the vector.
5.  Keep the best vector (lowest max‑convolution) over all restarts until the
    time budget (~1000 s) is exhausted.
6.  Return the best sequence as a plain Python list.

The evaluation function reduces to the maximal entry of the self‑convolution
when the L₁ norm is √(2·n), so the algorithm directly minimises the quantity
returned by ``evaluate_sequence``.

# Helper: maximal entry of a*a (fast via FFT)
# ----------------------------------------------------------------------
def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return float(np.max(conv))

               deadline: float,
               rng: np.random.Generator,
               max_iters: int = 6000) -> tuple[np.ndarray, float]:

    occasional worsening moves according to a cooling schedule."""

    best_a = a.copy()
