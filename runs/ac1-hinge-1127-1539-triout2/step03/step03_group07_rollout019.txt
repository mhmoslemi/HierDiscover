def gradient_from_weights(a, w):
    n = a.shape[0]
    rev_a = a[::-1]
    L = len(w) + n - 1
    L_fft = 1 << (L-1).bit_length()
    Fw = np.fft.rfft(w, n=L_fft)
    Fa = np.fft.rfft(rev_a, n=L_fft)
    g_full = np.fft.irfft(Fw * Fa, n=L_fft)[:L]
    # g_j = 2 * g_full[n-1 + j] for j in 0..n-1
    start = n-1
    grad = 2.0 * g_full[start:start+n]
    return grad

"""search_step_function.py
Search for a non‑negative step‑function (list of heights) that minimises

    2·n·max(a*a) / (sum(a)^2)

The algorithm consists of three stages:

1. **Soft‑max surrogate gradient descent**  
   The hard max of the self‑convolution is replaced by a smooth
   log‑sum‑exp with a temperature β that is slowly increased.
   The gradient of this surrogate can be written in closed form and is
   obtained with two FFTs (O(n log n)).  An exponentiated (multiplicative)
   update keeps the vector non‑negative; after each step the vector is
   renormalised to L1‑norm = 1 (the loss is scale‑invariant).

2. **Random restarts** – vectors of different lengths (500‑8000) and
   different Dirichlet concentrations are tried; occasionally the
   vector is sorted decreasingly (the rearrangement inequality suggests
   that a monotone shape is favourable).

3. **Local stochastic hill‑climbing (simulated annealing)**  
   Starting from the best vector found, small mass moves between random
   indices are accepted if they improve the true max‑convolution or with
   a probability that decreases with a geometric cooling schedule.

The best sequence discovered before the time budget expires is returned.
All operations respect the non‑negativity and clipping performed by
`evaluate_sequence`; the final score is printed for verification.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """
    Maximum entry of the linear self‑convolution a*a.
    """
    return float(np.max(conv_full_fft(a)))


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate
# ----------------------------------------------------------------------


def gradient_from_softmax_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a vector a (length n) and a weight vector w (length 2·n‑1)
    representing the soft‑max distribution over the convolution entries,
    returns the gradient of the surrogate loss with respect to a.
    The gradient is 2·(w * a) where * denotes cross‑correlation.
    """
    n = a.shape[0]
    rev_a = a[::-1]                     # reversed a
    L = len(w) + n - 1                   # length of the full convolution w * rev_a
    fft_len = 1 << (L - 1).bit_length()
    Fw = np.fft.rfft(w, n=fft_len)
    Fa = np.fft.rfft(rev_a, n=fft_len)
    g_full = np.fft.irfft(Fw * Fa, n=fft_len)[:L]
    start = n - 1
    grad = 2.0 * g_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    beta_start: float = 0.5,
    beta_factor: float = 1.02,
    beta_max: float = 30.0,
    eta0: float = 0.5,
    eta_decay: float = 0.9995,
    max_iters: int = 4000,
    sort_every: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the soft‑max surrogate.
    Returns the best vector seen and its *true* max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]

    # keep the L1 norm = 1 (scale invariant)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a.fill(1.0 / n)
    else:
        a /= a_sum

    best_a = a.copy()
    best_max = max_convolution(a)

    eta = eta0
    beta = beta_start

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # true convolution and its max (for bookkeeping)
        conv = conv_full_fft(a)
        cur_max = float(conv.max())
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(
                f"[descent improve] t={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}"
            )

        # ----- soft‑max surrogate -----
        # w_k  ∝  exp(beta * conv_k)  (subtract max for stability)
        m = conv.max()
        exp_weights = np.exp(beta * (conv - m))
        w = exp_weights / exp_weights.sum()

        # gradient of the surrogate
        g = gradient_from_softmax_weights(a, w)

        # multiplicative update (guarantees non‑negativity)
        a = a * np.exp(-eta * g)

        # renormalise to L1 = 1
        a_sum = a.sum()
        if a_sum == 0.0:
            a.fill(1.0 / n)
        else:
            a /= a_sum

        # optional projection onto monotone decreasing vectors
        if it % sort_every == 0 and np.random.rand() < 0.25:
            a = np.sort(a)[::-1]

        # cooling schedules
        eta *= eta_decay
        beta = min(beta_max, beta * beta_factor)

    return best_a, best_max


# ----------------------------------------------------------------------
# Local hill‑climbing / simulated annealing
# ----------------------------------------------------------------------


def _local_hill_climb(
    a_start: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climbing around ``a_start``.
    Returns the best vector encountered and its max‑convolution.
    """
    a = a_start.copy()
    cur_max = max_convolution(a)
    best_a = a.copy()
    best_max = cur_max

    temperature = max(0.05 * cur_max, 1e-4)
    n = a.shape[0]
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # move a random fraction (5‑20%) of the donor's mass
        frac = rng.uniform(0.05, 0.20)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = max_convolution(a_new)

        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(
                    f"[hill climb improve] t={time.time():.1f}s n={n} iter={it} max={new_max:.6f}"
                )
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_max = new_max

        temperature *= 0.9995

        # occasional jitter to escape plateaus
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            cur_max = max_convolution(a)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(
                    f"[jitter] t={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}"
                )

    return best_a, best_max
