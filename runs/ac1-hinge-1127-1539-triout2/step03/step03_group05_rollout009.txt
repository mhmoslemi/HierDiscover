"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(seq)

where the objective equals 2·n·max(conv(seq,seq))/sum(seq)².
The algorithm consists of three parts:

1. **Extensive parametric sweep** over several symmetric “U‑shapes”  
   (linear, quadratic, exponential, power‑law, cosine and β‑type).  
   Each candidate is projected onto the ℓ₁‑simplex  Σaᵢ = √(2 n).

2. **Adam‑style smoothed gradient descent** on a log‑sum‑exp
   approximation of the max‑convolution.  The temperature τ is
   multiplied by a decaying factor, the learning rate follows a simple
   schedule and every iterate is projected back onto the simplex.

3. **Fine‑grained sub‑gradient polishing** and a short stochastic
   “pair‑transfer” hill‑climbing phase.

The whole search respects a 1000 s budget (a small safety margin is
kept).  The best vector found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σx_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # fast path – already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), target):
        return v

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Move a tiny amount of mass from a random index to another one,
    then re‑project onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Families of symmetric “U‑shapes”
# ----------------------------------------------------------------------


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Symmetric shape based on distance from the centre.
    kind ∈ {linear, quadratic, exponential}.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    a = raw * (target_sum / raw.sum())
    return a


def generate_power_U_shape(
    n: int,
    target_sum: float,
    exponent: float = 2.0,
    param: float = 1.0,
) -> np.ndarray:
    """raw = 1 + param * d**exponent  (symmetric)."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = 1.0 + param * (d ** exponent)
    a = raw * (target_sum / raw.sum())
    return a


def generate_cosine_U_shape(
    n: int,
    target_sum: float,
    param: float = 1.0,
) -> np.ndarray:
    """
    raw = 1 + param * (1 - cos(π·d)).
    Gives a smooth “U” with controllable edge amplification.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = 1.0 + param * (1.0 - np.cos(np.pi * d))
    a = raw * (target_sum / raw.sum())
    return a


def generate_beta_U_shape(
    n: int,
    target_sum: float,
    power: float = 2.0,
) -> np.ndarray:
    """
    raw = d**p + (1‑d)**p  (U‑shaped for p>0, more pronounced with larger p).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = (d ** power) + ((1.0 - d) ** power)
    raw = np.clip(raw, 1e-12, None)     # avoid exact zeros
    a = raw * (target_sum / raw.sum())
    return a


# ----------------------------------------------------------------------
# Smoothed gradient (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (grad, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    # temperature proportional to current max
    tau = max(tau_factor * max_conv, 1e-12)

    # soft‑max weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # grad = (2/τ) * (w * a_rev) restricted to the central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)).  Returns (grad, max_conv, idx_max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv, idx_max


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects a ~1000 s time budget
    and always returns the best vector encountered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a problem size.
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_vec = np.array(base, dtype=np.float64)
    else:
        # a size that is large enough for a good “flattening” effect,
        # but still cheap enough for many FFTs.
        n = np.random.randint(8000, 9000)
        base_vec = None

    target_sum = np.sqrt(2.0 * n)     # we will keep the ℓ₁‑norm equal to √(2n)

    # --------------------------------------------------------------
    # Phase 1 – broad parametric sweep.
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)
    best_score = float("inf")

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")

    # ---------- simple families (already used in the reference code) ----------
    for p in np.linspace(0.0, 8.0, 41):                     # linear
        cand = generate_U_shape(n, target_sum, kind="linear", param=p)
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score, best_seq = cur, cand.copy()
            print(f"  new best linear   p={p:.2f} → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    for p in np.linspace(0.0, 6.0, 31):                     # quadratic
        cand = generate_U_shape(n, target_sum, kind="quadratic", param=p)
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score, best_seq = cur, cand.copy()
            print(f"  new best quad     p={p:.2f} → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    for p in np.linspace(0.0, 4.0, 21):                     # exponential
        cand = generate_U_shape(n, target_sum, kind="exponential", param=p)
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score, best_seq = cur, cand.copy()
            print(f"  new best expo     p={p:.2f} → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # ---------- newly added families ----------
    # power‑law U‑shapes: exponent varies, param varies
    for exp in [2, 3, 4, 5]:
        for p in np.linspace(0.0, 6.0, 21):
            cand = generate_power_U_shape(n, target_sum, exponent=exp, param=p)
            cur = conv_self_fft(cand).max()
            if cur < best_score:
                best_score, best_seq = cur, cand.copy()
                print(f"  new best power   exp={exp} p={p:.2f} → {best_score:.6f}")
            if time.time() >= deadline:
                return best_seq.tolist()

    # cosine‑based U‑shape
    for p in np.linspace(0.0, 6.0, 31):
        cand = generate_cosine_U_shape(n, target_sum, param=p)
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score, best_seq = cur, cand.copy()
            print(f"  new best cosine  p={p:.2f} → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # beta‑type U‑shape (d**p + (1-d)**p)
    for p in np.linspace(1.0, 8.0, 31):
        cand = generate_beta_U_shape(n, target_sum, power=p)
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score, best_seq = cur, cand.copy()
            print(f"  new best beta    p={p:.2f} → {best_score:.6f}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # baseline from a previously known construction (if any)
    if base_vec is not None:
        cur = conv_self_fft(base_vec).max()
        if cur < best_score:
            best_score, best_seq = cur, base_vec.copy()
            print(f"  baseline improves → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam‑style smoothed gradient descent.
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Entering Adam‑style descent")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyper‑parameters (they will be decayed)
    lr = 0.5 * (target_sum / n)          # initial learning rate
    tau_factor = 0.2                     # initial temperature factor
    tau_decay = 0.9995                   # per‑iteration decay
    lr_decay = 0.999                     # per‑iteration decay

    t = 1
    while time.time() < deadline:
        grad, cur_max = _smoothed_gradient(a, tau_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # keep the best encountered point
        cur_max = conv_self_fft(a).max()
        if cur_max < best_score - 1e-12:
            best_score, best_seq = cur_max, a.copy()
            print(f"  iter {t:06d} – improved to {best_score:.6f}")

        # schedule updates
        lr *= lr_decay
        tau_factor = max(0.001, tau_factor * tau_decay)

        t += 1
        # early stop if we already beat the target bound
        if best_score < 1.503:
            print(f"  target bound reached at iter {t-1} ({best_score:.6f})")
            break

    # --------------------------------------------------------------
    # Phase 3 – sub‑gradient refinement + random polishing.
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Sub‑gradient refinement")
    eta_sub = 0.3 * (target_sum / n)
    sub_iter = 0
    while time.time() < deadline and sub_iter < 2000:
        grad, cur_max, idx_max = _subgradient_max(a)
        a = a - eta_sub * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        cur_max = conv_self_fft(a).max()
        if cur_max < best_score - 1e-12:
            best_score, best_seq = cur_max, a.copy()
            print(f"  sub‑grad {sub_iter:04d} – improved to {best_score:.6f}")
            # modestly increase step size when we make progress
            eta_sub = min(eta_sub * 1.05, 2.0 * (target_sum / n))
        else:
            if sub_iter % 500 == 0 and sub_iter > 0:
                eta_sub *= 0.9
        sub_iter += 1
        if best_score < 1.503:
            break

    # final stochastic polishing
    print(f"[{time.time():.2f}] Final random polishing")
    for _ in range(80):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.006)
        cand_score = conv_self_fft(cand).max()
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand.copy()
            print(f"  polishing improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Search finished – best value {best_score:.6f}")
    return best_seq.tolist()
