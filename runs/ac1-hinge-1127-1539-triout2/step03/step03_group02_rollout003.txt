"""
Propose a step‑function (non‑negative coefficient vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four phases:

1. **Broad parametric sweep** – generate many symmetric “U‑shaped’’
   sequences (inverse‑power, power, linear, quadratic, exponential)
   with a fine grid of shape parameters.  The best candidate of this
   sweep is used as the starting point.

2. **Adam‑based smoothed descent** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  The gradient is obtained by a
   single FFT‑based convolution, then the Adam optimiser updates the
   vector and a Euclidean projection onto the ℓ₁‑simplex (fixed ℓ₁‑norm)
   keeps the sum constant.  The temperature τ of the smooth max is
   gradually reduced.

3. **Sub‑gradient refinement** – once τ is tiny we switch to the true
   sub‑gradient of the max‑convolution (average over all maximising
   indices) with a decreasing step size.

4. **Polishing** – a short burst of tiny random mass transfers between
   random indices; any improvement is kept.

All phases respect a hard time budget (≈1000 s).  The returned sequence
is a plain Python list of floats; it is already normalised so that
`evaluate_sequence` simply returns the maximal self‑convolution value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, sum_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries become zero – fallback to uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    param controls the shape (exponent for power / inverse, slope for linear…).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-5                      # avoid exact zero
    elif kind == "inverse":
        raw = 1.0 / (1e-5 + d ** param)
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of the full self‑convolution of a."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Smoothed Adam descent
# ----------------------------------------------------------------------


def adam_smoothed_descent(a0: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          base_lr: float = 0.5,
                          tau0_factor: float = 0.15,
                          tau_min: float = 1e-6,
                          tau_decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Minimise a log‑sum‑exp smooth approximation of max(conv(a,a))
    with Adam.  Returns the best vector observed and its max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta1_pow = 1.0
    beta2_pow = 1.0

    # learning‑rate (scaled by a typical entry size)
    lr = base_lr * (target_sum / n)

    # initial temperature (a fraction of current max)
    cur_max = compute_max_conv(a)
    tau = max(tau0_factor * cur_max, tau_min)

    best_a = a.copy()
    best_score = cur_max

    it = 0
    while time.time() < deadline:
        # full convolution and its maximum
        conv = conv_self_fft(a)
        cur_max = conv.max()

        # update temperature
        tau = max(tau * tau_decay, tau_min)

        # soft‑max weights (log‑sum‑exp)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = (2/τ) * (w * a_rev) restricted to central n entries
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # bookkeeping
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam iter {it:07d} – new best {best_score:.6f}")

        # occasional jitter / random transfer to escape plateaus
        if np.random.rand() < 0.008:
            a = random_pair_transfer(a, target_sum, scale=0.006)

        # decay learning‑rate a little every 5000 iterations
        if it % 5000 == 0 and it > 0:
            lr *= 0.97

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient refinement
# ----------------------------------------------------------------------


def subgradient_refinement(a0: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           init_step: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Perform sub‑gradient steps on the true max‑convolution.
    Returns best vector seen and its max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    step = init_step * (target_sum / n)

    best_a = a.copy()
    best_score = compute_max_conv(a)

    it = 0
    while time.time() < deadline:
        conv = conv_self_fft(a)
        cur_max = conv.max()

        # mask of (approximately) maximal entries
        eps = max(1e-12, 1e-9 * cur_max)
        mask = (conv >= cur_max - eps).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = 1.0
            cnt = 1.0

        # gradient = average sub‑gradient over all maximisers
        a_rev = a[::-1]
        L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        M = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
        grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad iter {it:07d} – new best {best_score:.6f}")

        # schedule
        if it % 500 == 0 and it > 0:
            step *= 0.97
        if np.random.rand() < 0.01:
            a = random_pair_transfer(a, target_sum, scale=0.004)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Polishing phase
# ----------------------------------------------------------------------


def polishing(best_a: np.ndarray,
              target_sum: float,
              deadline: float,
              attempts: int = 60,
              scale: float = 0.003) -> tuple[np.ndarray, float]:
    """Tiny random pair transfers; keep any improvement."""
    best_seq = best_a.copy()
    best_score = compute_max_conv(best_seq)

    for k in range(attempts):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=scale)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Polishing #{k} improved to {best_score:.6f}")

    return best_seq, best_score

    param_grids = {
        "linear": np.linspace(0.0, 8.0, 81),          # step 0.1
        "quadratic": np.linspace(0.0, 6.0, 61),      # step 0.1
        "exponential": np.linspace(0.0, 6.0, 61),    # step 0.1
        "power": np.linspace(0.3, 5.0, 48),          # exponent
        "inverse": np.linspace(0.3, 5.0, 48),
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(n, target_sum, kind, p)
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} p={p:.3f} → {best_score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep.")
                return best_seq.tolist()

    # consider supplied baseline if any
    if init_seq is not None:
        score = compute_max_conv(init_seq)
        if score < best_score:
            best_score = score
            best_seq = init_seq.copy()
            print(f"Baseline improves initial score → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam‑based smoothed descent
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Entering Adam smoothed descent")
    a_adam, score_adam = adam_smoothed_descent(
        best_seq, target_sum, deadline,
        base_lr=0.5,          # will be scaled internally
        tau0_factor=0.12,    # slightly smaller temperature for sharper descent
        tau_min=1e-6,
        tau_decay=0.9997
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"After Adam descent → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting sub‑gradient refinement")
    a_sub, score_sub = subgradient_refinement(best_seq, target_sum, deadline,
                                              init_step=0.03)
    if score_sub < best_score:
        best_score = score_sub
        best_seq = a_sub.copy()
        print(f"After sub‑gradient → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final polishing
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Polishing phase")
    best_seq, best_score = polishing(best_seq, target_sum, deadline,
                                     attempts=80, scale=0.0025)

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence with max‑conv {best_score:.6f}")
    # `evaluate_sequence` will clip and normalise automatically, but we
    # keep the exact ℓ₁‑norm for clarity.
    return best_seq.tolist()
