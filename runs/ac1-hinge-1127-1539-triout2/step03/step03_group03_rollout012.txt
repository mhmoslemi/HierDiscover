def _optimize_subgradient(a0, deadline, rng, lr_initial=0.5, t_mix=0.3):

    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iteration = 0
    last_improve = 0
    while time.time() < deadline:
        iteration += 1

        # compute conv
        conv = _full_convolution(a)
        k_max = np.argmax(conv)
        cur_max = conv[k_max]

        # compute gradient
        grad = 2.0 * _shift_vector(a, k_max)  # defined elsewhere

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = lr_initial * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # project
        a_new = _project_simplex(a_new, S)

        # evaluate
        new_conv = _full_convolution(a_new)
        new_max = np.max(new_conv)

        if new_max < best_score:
            best_score = new_max
            best_a = a_new.copy()
            last_improve = iteration

        a = a_new

        # random swap
        if rng.random() < 0.05:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.01 * a[i] * rng.random()
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_simplex(a_candidate, S)
                cand_conv = np.max(_full_convolution(a_candidate))
                if cand_conv < best_score or rng.random() < np.exp(-(cand_conv - new_max) / (new_max * 0.01 + 1e-12)):
                    a = a_candidate
                    if cand_conv < best_score:
                        best_score = cand_conv
                        best_a = a_candidate.copy()
                        last_improve = iteration

        # early stop if stagnated
        if iteration - last_improve > 5000:
            break

    return best_a, best_score

def _refine_lp(a, iterations, rng, t=0.3):
   n = len(a)
   S = np.sqrt(2.0 * n)
   a_cur = a.copy()
   for it in range(iterations):
      conv = _full_convolution(a_cur)
      M = np.max(conv)
      # build A matrix
      col = np.concatenate([a_cur, np.zeros(n-1)])
      row = np.concatenate(([a_cur[0]], np.zeros(n-1)))
      # use scipy.linalg.toeplitz
      A_dense = scipy.linalg.toeplitz(col, row)
      # convert to sparse
      A = sp.csr_matrix(A_dense)
      g = cp.Variable(n)
      constraints = [g >= 0, A @ g <= M * np.ones(2*n-1)]
      prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
      prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
      if g.value is None:
          print("[lp] iteration failed; break")
          break
      g_val = np.maximum(g.value, 0.0)
      # scale to sum S
      sum_g = g_val.sum()
      if sum_g == 0:
          break
      g_scaled = g_val * (S / sum_g)
      # mix
      a_cur = (1 - t) * a_cur + t * g_scaled
      # optional projection to simplex (should already sum to S)
      # evaluate
      cur_max = np.max(_full_convolution(a_cur))
      print(f"[lp] iter {it+1}, max={cur_max:.6f}")
   return a_cur

def _initial_sequence(base_seq, n, rng):

    sum_a = a.sum()

"""
Improved search for a step‑function minimizing the maximum entry of its
self‑convolution (the evaluation score).

Algorithm
---------
1. Work on the probability vector w (non‑negative, Σw = 1) – the
   evaluation of the step function equals 2·n·max(w*w) after the
   standard L¹‑normalisation, so we only need to minimise max(w*w).
2. Perform many independent restarts.  For each restart:
   * Initialise w randomly (or from the previously found best sequence).
   * Run a projected Adam optimiser that follows the sub‑gradient of the
     active convolution entry: for the index k with the largest
     (w*w)[k] the sub‑gradient is 2·shift(w, k).
   * Occasionally execute a small random “mass‑swap’’ move to escape
     plateaus.
   * Keep the best sequence seen during the run.
3. After the time budget for fast gradient runs is exhausted, take the
   overall best sequence and run a few refinement steps by solving the
   linear programme

        maximise   Σ g_i
        subject to  (w * g)_k ≤ M   for all k,
                    g_i ≥ 0,

   where M is the current maximal convolution value.
   The LP solution g is scaled to the proper L¹‑norm and mixed with w
   (Frank‑Wolfe style) to obtain a new w with a (weakly) lower maximum.
4. Return the best sequence found (as a plain Python list).  The routine
   respects the 1000 s time budget, prints progress, and works entirely
   on the CPU.
"""

import time
import numpy as np
import scipy.linalg
import scipy.sparse as sp

# optional imports – the code works without them, only the final LP refinement
# is skipped if they are unavailable.
try:
    import cvxpy as cp
    _cvxpy_available = True
except Exception:          # pragma: no cover
    cp = None
    _cvxpy_available = False


# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _shift_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Return the vector g such that g_i = a_{k-i} if the index is in range,
    otherwise 0.  This is exactly the gradient of the active convolution
    entry (up to the factor 2).
    """
    n = a.shape[0]
    out = np.zeros_like(a)
    start_i = max(0, k - (n - 1))
    end_i = min(n - 1, k)
    if start_i <= end_i:
        idx = np.arange(start_i, end_i + 1)
        out[idx] = a[k - idx]
    return out


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If a base sequence is supplied and long enough, its leading part is used
    (with a tiny random jitter); otherwise a uniform Dirichlet sample is drawn.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # add a very small Gaussian jitter to avoid exact ties
        jitter = rng.normal(scale=abs(a).mean() * 0.001 + 1e-9, size=n)
        a += jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)      # Dirichlet(1) – uniform over the simplex
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Local optimiser – projected Adam on the active‑constraint sub‑gradient
# ----------------------------------------------------------------------


def _optimize_subgradient(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using a projected Adam optimiser that follows the
    sub‑gradient of the maximal convolution entry.  Returns the best sequence
    encountered and its maximal self‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.25
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(conv.argmax())
        cur_max = float(conv[k_max])

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # sub‑gradient of the active entry
        grad = 2.0 * _shift_vector(a, k_max)

        # Adam update (step size decays implicitly via moments)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_proposed = a - step
        a_proposed = _project_simplex(a_proposed, S)

        # evaluate the proposal
        new_max = float(np.max(_full_convolution(a_proposed)))
        if new_max < best_score:
            best_score = new_max
            best_a = a_proposed.copy()
            last_improve = it

        a = a_proposed

        # ---- occasional random “mass‑swap’’ move ---------------------------------
        if rng.random() < 0.07:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = 0.02 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = float(np.max(_full_convolution(cand)))
                # Metropolis style acceptance
                if cand_max <= cur_max or rng.random() < np.exp(
                    -(cand_max - cur_max) / max(cur_max * 0.05, 1e-12)
                ):
                    a = cand
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()
                        last_improve = it

        # early termination if no improvement for a long stretch
        if it - last_improve > 5000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (Frank‑Wolfe style)
# ----------------------------------------------------------------------


def _refine_by_lp(a: np.ndarray,
                  remaining_time: float,
                  rng: np.random.Generator,
                  max_iter: int = 5) -> tuple[np.ndarray, float]:
    """
    Perform a few Frank‑Wolfe style updates using the LP described in the
    statement.  Each iteration solves

        maximise   Σ g_i
        subject to  (a * g)_k ≤ M   (M = maximal entry of a*a)
                    g_i ≥ 0

    The solution g is scaled to the correct L¹‑norm and mixed with a.
    """
    if not _cvxpy_available:
        return a, float(np.max(_full_convolution(a)))

    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    cur = a.copy()
    best_score = float(np.max(_full_convolution(cur)))
    best_seq = cur.copy()

    it = 0
    while time.time() < remaining_time and it < max_iter:
        it += 1
        conv = _full_convolution(cur)
        M = float(conv.max())

        # Build the convolution matrix A such that A @ g = a * g
        col = np.concatenate([cur, np.zeros(n - 1)])
        row = np.concatenate(([cur[0]], np.zeros(n - 1)))
        A_dense = scipy.linalg.toeplitz(col, row)          # (2n‑1) × n
        A = sp.csr_matrix(A_dense)

        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2 * n - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1500, verbose=False)
        except Exception as e:                         # pragma: no cover
            print("[lp] solver exception:", e)
            break

        if g.value is None:
            print("[lp] No solution returned – aborting LP refinement.")
            break

        g_val = np.maximum(g.value, 0.0)
        sum_g = g_val.sum()
        if sum_g == 0.0:
            break

        # mix the new direction (Frank‑Wolfe step)
        t = 0.3  # could be schedule‑based; a constant works well
        g_scaled = g_val * (S / sum_g)
        cur = (1.0 - t) * cur + t * g_scaled

        cur_score = float(np.max(_full_convolution(cur)))
        print(f"[lp] iter {it}/{max_iter} – max conv = {cur_score:.6f}")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()

    return best_seq, best_score
