"""
A smooth‑max based mirror‑descent optimizer for the step‑function problem.

We minimise the maximum entry of the self‑convolution
    max_k (a * a)[k]
subject to a_i ≥ 0 and Σ a_i = √(2 n).
The external score equals this maximum after the above normalisation,
so we can work directly with the convolution.

The algorithm uses a differentiable “soft‑max” surrogate:
    φ(α,a) = (1/α)·log Σ_k exp(α·(a*a)[k])
which converges to the true maximum as α → ∞.
Its gradient can be written in closed form as a correlation,
allowing an O(n log n) implementation via FFT.

We perform exponentiated‑gradient (mirror descent on the simplex)
updates with momentum, an annealing schedule for α and the step‑size,
and occasional jitter to escape tiny plateaus.
Multiple random restarts with different lengths are run until the
time budget expires; the best vector found is returned.
If SciPy is available a tiny LP refinement (as in the reference code)
is applied when the vector length is modest.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional LP refinement (same code as the reference implementation)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Dense matrix H where (H @ g)[k] = (f * g)[k] (linear self‑convolution)."""
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,
               max_steps: int = 5) -> np.ndarray:
    """Small LP refinement (same idea as the reference code)."""
    if not _SCIPY_AVAILABLE or f.shape[0] > 3000:
        return f
    n = f.shape[0]
    S = math.sqrt(2.0 * n)
    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break
        conv = np.convolve(f, f)
        M = conv.max()
        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σ g_i  ⇔ minimise –Σ g_i
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:  # pragma: no cover
            break
        if not res.success:
            break
        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S
    return f


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(Fa * Fa, n=nfft)[: 2 * n - 1]
    return conv


def _gradient_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    g_i = 2·Σ_j a_j·w_{i+j},  i = 0..n‑1.
    Implemented as correlation via FFT (O(n log n)).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    nfft = 1 << ((n + m - 1).bit_length())
    # FFT of w
    Fw = np.fft.rfft(w, n=nfft)
    # FFT of a reversed
    a_rev = a[::-1]
    Fa_rev = np.fft.rfft(a_rev, n=nfft)
    conv = np.fft.irfft(Fw * Fa_rev, n=nfft)   # length nfft
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Core optimizer (smooth‑max mirror descent)
# ----------------------------------------------------------------------
def _smooth_optimize_one_start(a0: np.ndarray, n: int,
                               deadline: float,
                               max_iters: int = 150_000,
                               eta0: float = 0.6,
                               beta: float = 0.9,
                               alpha_start: float = 0.2,
                               alpha_end: float = 8.0,
                               jitter_prob: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with a soft‑max surrogate.
    Returns the best vector seen and its (exact) maximal convolution value.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)
    best_max = float('inf')
    best_seq = a.copy()
    v = np.zeros_like(a)          # momentum buffer
    eta = eta0
    it = 0

    while time.time() < deadline - 0.05 and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()

        # record improvement (exact max is the true objective)
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_seq = a.copy()

        # annealed soft‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        # for stability shift by the current maximum
        exp_vals = np.exp(alpha * (conv - cur_max))
        w = exp_vals / exp_vals.sum()

        # gradient of the surrogate
        g = _gradient_corr(w, a)

        # momentum
        v = beta * v + (1.0 - beta) * g

        # exponentiated update (keeps a_i ≥ 0)
        a = a * np.exp(-eta * v)

        # renormalise to keep ℓ₁‑norm fixed
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # step‑size decay
        eta *= 0.99995

        # occasional jitter to leave tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.005
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

    return best_seq, best_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    ``evaluate_sequence``.  Uses a smooth‑max mirror‑descent optimiser
    with several random restarts and a final LP refinement (if feasible).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1)  If a previous construction is available, evaluate it.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                n0 = cand.size
                S0 = math.sqrt(2.0 * n0)
                cand = cand / cand.sum() * S0
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] height_sequence_1 score = {best_score:.6f}, length = {n0}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 2)  Random‑restart optimisation over a handful of lengths.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000]  # varied problem sizes
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))
        S = math.sqrt(2.0 * n)

        # initialise a starting vector
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.3:
            a0 = best_seq[:n].copy()
        else:
            # Dirichlet(1) – exponential draws give a reasonable spread
            a0 = rng.exponential(scale=1.0, size=n)

        a0 = a0 / a0.sum() * S

        # one optimisation run
        seq, exact_max = _smooth_optimize_one_start(
            a0, n, deadline,
            max_iters=120_000,
            eta0=0.6,
            beta=0.9,
            alpha_start=0.2,
            alpha_end=8.0,
            jitter_prob=0.01)

        # compute the external score (same as exact_max because of normalisation)
        if exact_max < best_score:
            best_score = exact_max
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        # optional LP refinement if we have a promising candidate
        if _SCIPY_AVAILABLE and n <= 3000 and best_score < 1.6:
            refined = _lp_refine(best_seq, deadline)
            refined_max = _conv_self(refined).max()
            if refined_max < best_score:
                best_score = refined_max
                best_seq = refined.copy()
                print(f"[LP refine] run={run_id} n={n} refined score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 3)  Final polishing of the best candidate.
    # ------------------------------------------------------------------
    if best_seq is not None:
        # a short extra mirror‑descent pass with a tighter α schedule
        best_seq, _ = _smooth_optimize_one_start(
            best_seq, best_seq.size, deadline,
            max_iters=80_000,
            eta0=0.4,
            beta=0.95,
            alpha_start=4.0,
            alpha_end=12.0,
            jitter_prob=0.005)

        # final LP refinement (if affordable)
        best_seq = _lp_refine(best_seq, deadline)

        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    else:
        # fallback – uniform distribution (should never happen)
        n = 1000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    return best_seq.tolist()
