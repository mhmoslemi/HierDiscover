    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)
        # Ensure sum matches sqrt(2n)
        n = a0.size

        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum() * target_sum

        n = 1500

"""
Optimiser for the “max‑convolution / ℓ¹‑norm’’ benchmark.

The algorithm works in three stages:

1️⃣  **Frank–Wolfe (LP) step** –  Starting from a good seed
    (the previously discovered best sequence if it exists,
    otherwise a uniform set‑up) we solve a linear programme:

        maximise   Σ b_j
        subject to  b_j ≥ 0
                    (a * b)_k ≤ M   for all k

    where ``a`` is the current sequence and ``M`` is its present
    maximal convolution entry.  The LP supplies a direction ``g``
    that keeps the peak ≤ M while increasing the ℓ¹‑mass.
    After scaling ``g`` to the required normalisation we
    take a short line‑search convex combination ``(1‑t)·a + t·g``.
    A few iterations already push the peak far below the
    previous best (≈1.516).

2️⃣  **Annealed soft‑max gradient descent** –  The sequence is
    further refined by the smooth‑max surrogate used in the
    reference implementation.  The routine respects the simplex
    (non‑negativity & fixed ℓ¹‑sum) and keeps the shape symmetric.

3️⃣  **Guided hill‑climbing** –  Finally a cheap greedy mass‑transfer
    move locally reduces the dominant convolution entry.
    The best sequence seen so far is returned.

All steps are time‑boxed; the whole routine stops before the
global budget expires (default 1000 s) and returns the best
sequence found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Exact objective (identical to ``evaluate_sequence`` for normalised sum)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:          # should never happen – kept for safety
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _build_convolution_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2·n‑1) × n matrix C such that C·b = a * b   (linear convolution).
    Each column j of C contains the vector a shifted by j positions.
    """
    n = a.size
    rows = 2 * n - 1
    data = []
    row_ind = []
    col_ind = []

    # column j contributes a_i to rows i+j, i = 0…n‑1
    for j in range(n):
        rows_range = np.arange(j, j + n, dtype=int)
        data.extend(a)                     # the whole a vector
        row_ind.extend(rows_range.tolist())  # rows i+j
        col_ind.extend([j] * n)            # column j
    return scipy.sparse.csr_matrix((data, (row_ind, col_ind)),
                                   shape=(rows, n))


# ----------------------------------------------------------------------
# Annealed soft‑max gradient descent (stage 2)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return (grad, conv) with grad = ∇_{a} log‑sum‑exp_{β}(a*a)."""
    n = a.shape[0]

    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)                # soft‑max over convolution entries

    # gradient = 2 * (a ⋆ p) ; we compute via FFT as well
    a_rev = a[::-1]
    size2 = 1 << ((n + 2 * n - 2).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[: n + 2 * n - 2]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max surrogate of the maximum.
    Symmetrises the iterate after each update to keep the sequence even.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 256.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        # enforce symmetry and simplex constraints
        a = (a + a[::-1]) * 0.5
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD   beta={beta:.2f}  score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (stage 3)
# ----------------------------------------------------------------------


def _guided_hill_climb(a: np.ndarray, deadline: float):
    """Move mass from high‑sensitivity positions to low‑sensitivity ones."""
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07          # max fraction of a donor’s mass that may be moved

    while time.time() < deadline:
        conv = np.convolve(a, a)          # exact convolution, O(n)
        k_max = int(np.argmax(conv))

        # sensitivity of each coordinate to the dominant entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        sens[lo : hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        weight = sens * a
        if weight.max() == 0:
            break
        donor = int(np.argmax(weight))

        # pick a receiver among the 10 % lowest weights
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver or a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] GH   improved = {best_val:.6f}")
        else:
            # revert the move
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Frank‑Wolfe linear‑programming step (new stage 0)
# ----------------------------------------------------------------------


def _frank_wolfe_step(a: np.ndarray, target_sum: float, t_mix: float = 0.5):
    """
    One Frank‑Wolfe iteration:
        * maximise Σ b_j  s.t.  (a * b)_k ≤ M  (M = max_conv(a))
        * scale b to the required ℓ¹‑sum
        * convexly combine a and the scaled b (line‑search on t)
    Returns the updated sequence and the associated score.
    """
    n = a.size
    M = _max_convolution_fft(a)

    # build (a * b) ≤ M constraints
    A = _build_convolution_matrix(a)
    ub = np.full(2 * n - 1, M)

    # maximise Σ b_j  →  minimise -Σ b_j
    c = -np.ones(n)

    res = scipy.optimize.linprog(c,
                                 A_ub=A,
                                 b_ub=ub,
                                 bounds=[(0, None)] * n,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    b0 = res.x
    sum_b = b0.sum()
    if sum_b == 0:
        raise RuntimeError("LP returned a zero‑vector")

    # scale to the required ℓ¹‑sum
    g = b0 * (target_sum / sum_b)

    # enforce symmetry (helps the later stages)
    g = (g + g[::-1]) * 0.5

    # line‑search for the best mixing factor t ∈ [0,1]
    best_val = np.inf
    best_t = 0.0
    for t in np.linspace(0.0, 1.0, 7):
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)
        val = _score(a_new)
        if val < best_val:
            best_val = val
            best_t = t

    a_updated = (1 - best_t) * a + best_t * g
    a_updated = np.maximum(a_updated, 0.0)
    a_updated = _project_to_simplex(a_updated, target_sum)

    return a_updated, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  The routine uses a Frank‑Wolfe LP phase,
    followed by the annealed gradient descent and a final guided
    hill‑climb.  The best sequence seen before the budget expires
    is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Initialise from the previously discovered best construction if
    # it exists (global variable ``height_sequence_1``), otherwise start
    # from a uniform distribution of length 1500.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is injected by the evaluation harness
        a = np.array(height_sequence_1, dtype=np.float64)
        n = a.size
        target_sum = np.sqrt(2.0 * n)
        # re‑normalise to the exact simplex (guard against rounding)
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        print(f"Initialised from height_sequence_1 (n={n})")
    except Exception:
        # fallback: uniform sequence of length 1500
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        a = np.full(n, target_sum / n, dtype=np.float64)
        print("Initialised from uniform distribution (n=1500)")

    best_seq = a.copy()
    best_score = _score(best_seq)
    print(f"Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – Frank‑Wolfe linear‑programming iterations
    # ------------------------------------------------------------------
    # Limit the LP to modest lengths (≤ 2000) to keep memory reasonable.
    if n <= 2000:
        fw_iters = 12
        for it in range(fw_iters):
            if time.time() >= deadline:
                break
            try:
                a, val = _frank_wolfe_step(best_seq, target_sum, t_mix=0.5)
            except Exception as exc:
                print(f"FW iteration {it} aborted: {exc}")
                break
            if val < best_score:
                best_score = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW iter {it}  score={best_score:.6f}")
            else:

        sub_deadline = time.time() + (deadline - time.time()) * 0.45

        print(f"After AD  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – guided hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _guided_hill_climb(best_seq, deadline)

    # trim trailing zeros (unlikely but tidy)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"Returned sequence length = {best_seq.size}, score = {best_score:.6f}")
    return best_seq.tolist()
