The evaluation function is essentially: given sequence a of length n (list of floats), compute b = convolution of a with a (full convolution length 2n-1) then max_b = max(b). sum_a = sum a. The output is 2 * n * max_b / (sum_a**2). Also they bound each a to [0,1000] and then take max(0, x) (non-negative) and clamp to 1000. For small numbers sum can't be too small: sum_a >=0.01 else inf.

So the objective is to minimize (2 n * max_b)/(sum_a^2). Equivalent to minimize max_b * n / sum_a^2.

We have the constraint a_i >= 0, (with clamping). Equivalent to selecting a nonnegative vector a.

We can also think of scaling: objective is homogeneous: scaling a by factor c multiplies sum by c, max_b (as b = a*a => a_i a_j etc) scales by c^2, so objective p = 2 n * c^2 max_b0 / (c^2 sum_a0^2) = 2 n * max_b0 / sum_a0^2. So objective is invariant under scaling! def _adam_optimize(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    t_step = 0
    cur_max = np.max(_full_convolution_fft(a))
    best_a = a.copy()
    best_max=cur_max

    # schedule for tau: start at cur_max and decay factor maybe 0.99 per step
    tau = cur_max
    tau_decay = 0.995

    while time.time() < deadline:
        t_step += 1
        # compute gradient using softmax approx
        b = _full_convolution_fft(a)
        max_b = np.max(b)
        # compute softmax weights
        diff = b - max_b
        exps = np.exp(diff / max(tau, 1e-12))
        w = exps / np.sum(exps)

        # gradient = 2 * (a cross-correlated with w)
        a_rev = a[::-1]
        # compute convolution of w with a_rev using FFT
        # length of w = 2n-1, a_rev length = n, conv length = 2n-2? Actually both length n? w length 2n-1
        # So we need conv result length len(w)+n-1.
        conv_len = len(w) + n - 1
        fft_len = 1 << ((conv_len - 1).bit_length())
        Fw = np.fft.rfft(w, n=fft_len)
        Fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:conv_len]
        g = 2.0 * conv[n-1:n-1+n]  # this matches earlier _softmax_update

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t_step)
        v_hat = v / (1 - beta2**t_step)
        # learning rate schedule: maybe alpha = 0.3/(np.sqrt(t_step)?) intangible

def generate_window(name, n, param=None):
   if name == 'hann':
        k = np.arange(n)
        w = 0.5*(1 - np.cos(2*np.pi*k/(n-1)))
   elif name == 'hamming':
        w = 0.54 - 0.46 * np.cos(2*np.pi*k/(n-1))
   elif name == 'blackman':
        w = 0.42 - 0.5*np.cos(2*np.pi*k/(n-1)) + 0.08*np.cos(4*np.pi*k/(n-1))
   elif name == 'kaiser':
        from scipy.special import i0
        beta = param or 5.0
        k = np.arange(n)
        alpha = (n-1)/2
        t = (k - alpha) / alpha
        w = i0(beta * np.sqrt(1 - t*t)) / i0(beta)
   elif name == 'gaussian':
        sigma = param or 0.25
        k = np.arange(n)
        center = (n-1)/2
        w = np.exp(-0.5 * ((k - center) / (sigma * n/2))**2)
   elif name == 'uniform':
        w = np.ones(n)
   else: raise

"""
Search for a non‑negative step sequence that minimises the maximal entry of its
self‑convolution (the evaluation metric equals that maximum when the ℓ₁‑norm
is fixed to √(2·n)).  The algorithm proceeds in three phases:

1.  Initialise a pool of candidates using classic window functions
    (uniform, Hann, Hamming, Blackman, Gaussian, Kaiser).  These give good
    starting points because many of them already have a relatively flat
    auto‑convolution.

2.  Run many short local optimisation runs.  Each run uses:
       – Adam‑style adaptive gradient descent on a smooth approximation of the
         ‖a∗a‖∞ by τ·log ∑exp((a∗a)/τ);
       – occasional simulated‑annealing mass‑swap moves;
       – rare Gaussian jitter.
       After every update the vector is projected onto the simplex
       {x≥0, ∑x=√(2·n)}.

3.  Keep the best sequence found overall, evaluate it with the supplied
    ``evaluate_sequence`` function and store it in the global variable
    ``height_sequence_1`` for possible reuse.

The search explores a wide range of lengths (up to ~20 000) and exploits
Adam’s adaptive step‑sizes, which tend to converge faster and escape shallow
local minima compared with plain projected gradient descent.

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (full length) using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Initialise a feasible vector of length n (ℓ₁‑norm = √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with sum √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                                 # Dirichlet(1)
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Gradient of τ·log‑sum‑exp(b/τ) where b = a*a
# ----------------------------------------------------------------------
def _gradient_and_conv(a: np.ndarray, tau: float):
    """Return (gradient g, full convolution b, max(b))."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau <= 1e-12:                     # hard‑max (sub‑gradient)
        k = int(np.argmax(b))
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal term counted twice above – correct to a_i instead of 2*a_i
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]
        return g, b, max_b

    # soft‑max version
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)                      # weights of size 2n‑1

    # g = 2 * (a_rev ⋆ w)  (cross‑correlation)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]

    return g, b, max_b

# Local optimiser – Adam + simulated‑annealing moves + jitter
# ----------------------------------------------------------------------

                    rng: np.random.Generator) -> tuple[np.ndarray, float]:

    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # bookkeeping
    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters for the stochastic parts
    move_prob = 0.09
    epsilon_move = 0.035
    temperature = cur_max * 0.15
    temp_decay = 0.9996
    jitter_prob = 0.008
    jitter_scale = 0.025

    # soft‑max temperature (τ)
    tau = cur_max
    tau_decay = 0.9985

    step = 0
    while time.time() < deadline:
        step += 1

        # --------------------------------------------------------------
        # 1) occasional mass‑swap (simulated‑annealing)
        # --------------------------------------------------------------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n)
                if i != j:
                    delta = epsilon_move * a[i]
                    a_cand = a.copy()
                    a_cand[i] -= delta
                    a_cand[j] += delta
                    cand_max = np.max(_full_convolution_fft(a_cand))
                    accept = False
                    if cand_max <= cur_max:
                        accept = True
                    else:
                        prob = np.exp(-(cand_max - cur_max) / max(temperature, 1e-12))
                        if rng.random() < prob:
                            accept = True
                    if accept:
                        a = a_cand
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                    temperature *= temp_decay

        # --------------------------------------------------------------
        # 2) Adam gradient step on smooth max (τ·log‑sum‑exp)
        # --------------------------------------------------------------
        g, b, max_b = _gradient_and_conv(a, max(tau, 1e-12))

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # learning‑rate schedule (decreasing)
        alpha = 0.6 / (np.sqrt(step) + 1.0)

        a_new = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a_new, S)

        # evaluate after projection
        b = _full_convolution_fft(a)
        cur_max = np.max(b)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # --------------------------------------------------------------
        # 3) occasional jitter
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _full_convolution_fft(a)
            cur_max = np.max(b)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # decay the smooth‑max temperature
        tau = max(tau * tau_decay, 1e-9)

    return best_a, best_max


# ----------------------------------------------------------------------
# Generate a window‑function candidate (non‑negative)
# ----------------------------------------------------------------------
def _window_candidate(shape: str, n: int, rng: np.random.Generator):
    """Return a non‑negative vector of length n for the given window shape."""
    k = np.arange(n, dtype=float)
    if shape == "uniform":
        w = np.ones(n, dtype=float)
    elif shape == "hann":
        w = 0.5 * (1.0 - np.cos(2.0 * np.pi * k / (n - 1)))
    elif shape == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * k / (n - 1))
    elif shape == "blackman":
        w = (0.42 - 0.5 * np.cos(2.0 * np.pi * k / (n - 1))
             + 0.08 * np.cos(4.0 * np.pi * k / (n - 1)))
    elif shape == "gaussian":
        sigma = 0.25                    # fraction of half‑length
        centre = (n - 1) / 2.0
        w = np.exp(-0.5 * ((k - centre) / (sigma * n / 2.0)) ** 2)
    elif shape == "kaiser":
        # use scipy if available, otherwise fall back to Hann
        try:
            from scipy.special import i0
            beta = 6.0
            alpha = (n - 1) / 2.0
            t = (k - alpha) / alpha
            w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
        except Exception:
            w = 0.5 * (1.0 - np.cos(2.0 * np.pi * k / (n - 1)))   # Hann as fallback
    else:
        w = np.ones(n, dtype=float)

    # Normalise to the required ℓ₁‑norm √(2·n)
    S = np.sqrt(2.0 * n)
    w = w.astype(float)
    if w.sum() == 0:
        w = rng.random(n) + 1e-3
    a = w * (S / w.sum())
    # clip tiny negatives that might arise from numerical noise
    a = np.clip(a, 0.0, None)
    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for the best step‑function heights within the given time budget.
    Returns a list of non‑negative floats.  The global variable
    ``height_sequence_1`` is updated with the best found sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – scan a handful of classic windows (different lengths)
    # ------------------------------------------------------------------
    window_shapes = ["uniform", "hann", "hamming", "blackman", "gaussian", "kaiser"]
    lengths_to_try = [512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths_to_try:
        if time.time() >= deadline:
            break
        for shape in window_shapes:
            a = _window_candidate(shape, n, rng)
            cur_max = np.max(_full_convolution_fft(a))
            if cur_max < best_overall_score:
                best_overall_score = cur_max
                best_overall_seq = a.copy()
                print(f"[init] shape={shape:9s} n={n:<5d} max={cur_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many short Adam‑based optimisation restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # Choose a length: explore around the current best length
        if best_overall_seq is not None:
            cur_len = best_overall_seq.size
            min_len = max(800, int(cur_len * 0.8))
            max_len = min(20000, int(cur_len * 1.5))
        else:
            min_len = 800
            max_len = 12000
        n = rng.integers(min_len, max_len + 1)

        # Initialise a feasible vector
        a0 = _initial_sequence(base_seq, n, rng)

        # Allocate a time slice for this restart (≈20 s)
        slice_deadline = min(time.time() + 20.0, deadline)

        # Optimise locally
        a_opt, score = _adam_optimizer(a0, slice_deadline, rng)

        # Record improvement
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best max={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the black‑box function
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        final_seq = best_overall_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_overall_score:.6f}) "
              f"length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq
        return final_seq
    else:
        # Should never happen – fallback to a trivial uniform vector
        n = 1024
        a = np.full(n, np.sqrt(2.0 * n) / n)
        final_seq = a.tolist()
        globals()['height_sequence_1'] = final_seq
        return final_seq
