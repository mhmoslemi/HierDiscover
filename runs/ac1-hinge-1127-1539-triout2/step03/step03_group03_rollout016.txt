"""
Propose a step‑function that minimises the evaluation score
    2·n·max(conv(a,a)) / (sum(a)²)
using a fast projected Adam optimiser and a mathematically motivated
initialisation a_i ≈ 1/√(2·(i+1)).  The optimiser works on a smooth
approximation of the max (soft‑max) and occasionally performs targeted
mass‑swap moves.  Many short independent restarts with random problem
sizes are executed until the time budget (≈1000 s) expires.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x ≥ 0 , Σ x = s }.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT, correctly scaled)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_j = 2 * Σ_i w_{i+j} * a_i.
    ``a`` has length n, ``w`` has length 2n‑1 (the convolution vector).
    """
    n = a.shape[0]
    w_rev = w[::-1]                        # for correlation
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # extract the slice that corresponds to the gradient
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth max (soft‑max) utilities
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights for the smoothed max:
        w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initialiser (inverse‑square‑root shape)
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).

    If a base sequence is supplied it is used (with tiny jitter);
    otherwise the analytically motivated shape a_i = 1/√(2·(i+1))
    is employed, optionally mirrored with probability ½.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        #  a_i = 1 / sqrt(2·(i+1))  (≈ optimal continuous kernel)
        i = np.arange(n, dtype=float)
        a = 1.0 / np.sqrt(2.0 * (i + 1.0))
        # with 50 % probability mirror the shape (mass at the right side)
        if rng.random() < 0.5:
            a = a[::-1]

    # normalise to the required simplex sum
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Local optimiser (projected Adam + occasional targeted swaps)
# ----------------------------------------------------------------------


def _optimize_local_adam(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine ``a0`` using a projected Adam optimiser on the soft‑max
    surrogate of the max‑norm.  Returns the best vector encountered and
    its true maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # hyper‑parameters
    lr0 = 0.25                     # base learning rate
    alpha = 0.8                    # start temperature for soft‑max
    alpha_mul = 1.002              # temperature schedule
    swap_prob = 0.06               # how often we try a targeted mass swap
    swap_frac = 0.02               # max fraction of a source entry that can be moved

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    last_improve = 0

    while time.time() < deadline:
        t_iter += 1

        # ----- 1. forward pass -----
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)

        # ----- 2. gradient (soft‑max surrogate) -----
        grad = _conv_with_weights(a, w)

        # ----- 3. Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new, S)

        # ----- 4. evaluate true max‑norm -----
        cur_max = np.max(_full_convolution(a_new))

        if cur_max < best_score:
            best_score = cur_max
            best_a = a_new.copy()
            last_improve = t_iter

        # ----- 5. occasional targeted mass‑swap move (escape plateaus) -----
        if rng.random() < swap_prob:
            # pick the entry that participates most in the current maximal coefficient
            k_max = np.argmax(conv)                     # index of the current max entry
            # admissible partners i with i + j = k_max
            i_candidates = np.arange(max(0, k_max - (n - 1)), min(n, k_max + 1))
            # pick the partner with largest product contribution
            prod = a[i_candidates] * a[k_max - i_candidates]
            i = i_candidates[np.argmax(prod)]

            # source index i, destination j (random, different)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # accept if improves or with a tiny Metropolis probability
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.04, 1e-12)):
                    a_new = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = cand.copy()
                        last_improve = t_iter

        # ----- 6. schedule updates -----
        alpha *= alpha_mul
        # optional learning‑rate decay (very gentle)
        lr0 *= 0.9999

        a = a_new

        # ----- 7. early termination if stagnated for many iterations -----
        if t_iter - last_improve > 3000:
            break

    return best_a, best_score

        # We sample n in a fairly wide interval.
        # --------------------------------------------------------------
        n = rng.integers(3000, 8001)   # 3k – 8k

        # --------------------------------------------------------------
        # Initialise a candidate – either from the historic best or from the
        # analytic inverse‑sqrt shape.
        # --------------------------------------------------------------
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a slice of time for this optimisation run (≈ 12 s)
        slice_deadline = min(time.time() + 12.0, deadline)

        # --------------------------------------------------------------
        # Run the local optimiser.
        # --------------------------------------------------------------
        a_opt, score = _optimize_local_adam(a0, slice_deadline, rng)

        # --------------------------------------------------------------
        # Keep the global best.
        # --------------------------------------------------------------
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt:3d} n={n:4d} "
                  f"new best score={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check through the supplied black‑box evaluator.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # should never happen – fallback to a trivial feasible vector
        return [1.0]

    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal max = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
