    n = a.shape[0]

    # Prepare FFT length.

    nfft = 1 << (3 * n - 2).bit_length()

    # Compute FFT of a.

    Fa = np.fft.rfft(a, n=nfft)

    # Compute convolution a*a (self) via inverse FFT.

    conv_full = np.fft.irfft(Fa * Fa, n=nfft)[:2 * n - 1]

    # compute max

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The optimizer works in three stages:

1. **Warm‑start generation** – a variety of initial vectors is created
   (random, truncated best, power‑law, exponential decay).  All vectors are
   normalised so that their  ℓ₁‑norm is √(2·n); the external score then equals
   the maximal entry of the self‑convolution.

2. **Smooth‑max exponentiated Adam optimisation** – the non‑smooth
   ``max`` is replaced by a log‑sum‑exp surrogate with a continuation on the
   temperature parameter α.  The resulting gradient can be obtained with a
   single FFT‑based self‑convolution and a single cross‑correlation, which
   makes each iteration O(n log n).  An Adam‑style multiplicative update keeps
   all coefficients non‑negative.

3. **Post‑processing** – a short simulated‑annealing pass and an optional
   linear‑programming refinement (if SciPy is available) try to improve the
   best candidate further.

The whole search stops automatically after the allotted wall‑clock budget
(~1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional linear programming refinement (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray):
    """
    Full self‑convolution of ``a`` using FFT.
    Returns (full array, maximal value, index of the maximum).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    nfft = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=nfft)
    conv_full = np.fft.irfft(Fa * Fa, n=nfft)[:conv_len]
    max_idx = int(np.argmax(conv_full))
    return conv_full, float(conv_full[max_idx]), max_idx


def _smooth_max_and_grad(a: np.ndarray, alpha: float, nfft: int):
    """
    Compute a smooth approximation of ``max(a*a)`` and its gradient.
    The loss is the log‑sum‑exp (temperature ``alpha``) of the full
    self‑convolution.  The gradient is the correlation of ``a`` with the
    softmax weights and equals the sub‑gradient of the smooth loss.
    Returns (loss, gradient, current max of the exact convolution).
    """
    # ----- self‑convolution (exact) ------------------------------------
    Fa = np.fft.rfft(a, n=nfft)
    conv_full = np.fft.irfft(Fa * Fa, n=nfft)[: 2 * a.shape[0] - 1]

    max_conv = conv_full.max()

    # ----- soft‑max weights -------------------------------------------
    # shift by the maximum for numerical stability
    shifted = alpha * (conv_full - max_conv)
    exp_shift = np.exp(shifted)               # safe: shifted ≤ 0
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp                    # soft‑max on the 2n‑1 entries

    # ----- gradient = 2 * correlation(a, w) -------------------------
    w_rev = w[::-1]                           # reverse for correlation
    Fw = np.fft.rfft(w_rev, n=nfft)
    corr_full = np.fft.irfft(Fa * Fw, n=nfft)[: 3 * a.shape[0] - 2]

    offset = 2 * a.shape[0] - 2
    grad = 2.0 * corr_full[offset : offset + a.shape[0]]

    # ----- smooth loss (log‑sum‑exp) ----------------------------------
    loss = max_conv + np.log(sum_exp) / alpha
    return loss, grad, max_conv


# ----------------------------------------------------------------------
# Adam optimiser with multiplicative (exponentiated) updates
# ----------------------------------------------------------------------
def _run_optimize_smooth(
    a0: np.ndarray,
    n: int,
    deadline: float,
    *,
    eta0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    jitter_prob: float = 0.015,
    jitter_scale: float = 0.02,
    alpha0: float = 5.0,
    alpha_max: float = 200.0,
    alpha_factor: float = 1.005,
    decay: float = 0.9999,
    max_iters: int = 30_000,
):
    """
    Smooth‑max Adam optimiser.
    Returns the best sequence seen and the corresponding exact max‑convolution.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)          # target ℓ₁‑norm (kept invariant)
    best_max = float("inf")
    best_seq = a.copy()

    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    eta = eta0
    alpha = alpha0
    nfft = _next_pow2(3 * n - 2)    # large enough for both convolutions
    t = 0                           # iteration counter

    # pre‑compute a small constant for exponent clipping
    EXP_CLIP = 50.0

    while time.time() < deadline and t < max_iters:
        t += 1

        loss, grad, cur_max = _smooth_max_and_grad(a, alpha, nfft)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative update (exponentiated gradient)
        step = eta / (np.sqrt(v_hat) + eps)
        update = -step * m_hat
        # clip exponent to avoid overflow/under‑flow
        np.clip(update, -EXP_CLIP, EXP_CLIP, out=update)
        a = a * np.exp(update)

        # renormalise to keep ℓ₁‑norm fixed
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * jitter_scale
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # record improvement (exact max, not the smooth surrogate)
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_seq = a.copy()

        # step‑size decay and continuation on α
        eta *= decay
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # early exit if time is almost exhausted
        if time.time() >= deadline - 0.2:
            break

    return best_seq, best_max

# LP refinement (same idea as the reference implementation)

def _build_conv_matrix(f: np.ndarray):
    """
    Build the dense matrix H (size (2n‑1) × n) such that H @ g = f * g.
    Used only for modest values of n (≤ 3000) where the LP stays tractable.
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i : i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2, max_steps: int = 4):
    """
    Linear‑programming refinement: maximise Σ gᵢ under the convolution bound.
    If SciPy is unavailable or n is large the function returns ``f`` unchanged.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:  # keep the LP size reasonable
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.6:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σ g  ⇔ minimise –Σ g
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": 8},
            )
        except Exception:  # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        # convex combination and renormalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Very cheap simulated annealing that moves a small amount of mass between
    two randomly chosen positions.  The acceptance criterion follows the
    Metropolis rule with a slowly decreasing temperature.
    """
    a = seq.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # current exact max
    _, cur_max, _ = _conv_max(a)

    best = a.copy()
    best_max = cur_max

    # temperature schedule
    T = 0.02 * cur_max + 1e-4
    decay = 0.99995

    while time.time() < deadline - 0.5:
        i, j = rng.choice(n, size=2, replace=False)

        # move at most 10 % of the source mass
        max_delta = a[i] * 0.1
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # maintain the ℓ₁‑norm (the operation already does)
        _, new_max, _ = _conv_max(a_new)

        dE = new_max - cur_max
        if dE < 0 or rng.random() < math.exp(-dE / T):
            a = a_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best = a.copy()

        T *= decay

    return best, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the time budget, prints progress, and finally returns
    the best sequence (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin

    best_seq = None
    best_score = float("inf")   # here ``best_score`` stores the exact max‑conv value

    # ------------------------------------------------------------------
    # 0) use the supplied baseline if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                n0 = cand.size
                S0 = math.sqrt(2.0 * n0)
                cand = cand / cand.sum() * S0
                score = _conv_max(cand)[1]  # max entry of the self‑convolution
                best_seq = cand.copy()
                best_score = float(score)
                print(f"[baseline] length={n0} score={best_score:.6f}")
        except Exception:  # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1) generic search over several lengths
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000, 12000, 16000, 20000]
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # ------- initial vector -------------------------------------------------
        use_trunc = best_seq is not None and best_seq.size >= n and rng.random() < 0.25
        if use_trunc:
            a0 = best_seq[:n].copy()
        else:
            typ = rng.random()
            if typ < 0.5:          # uniform random
                a0 = rng.random(n)
            elif typ < 0.8:        # power‑law decay
                p = rng.uniform(0.0, 2.0)
                a0 = (np.arange(1, n + 1) ** (-p)).astype(float)
            else:                  # exponential decay
                lam = rng.uniform(0.0, 0.015)
                a0 = np.exp(-lam * np.arange(n))

        # normalise to √(2 n)
        S = math.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # ------- optimisation run ----------------------------------------------
        # iteration budget grows when n is small to keep total work roughly bounded
        max_iters = max(1000, int(6e5 / n))
        seq, max_val = _run_optimize_smooth(
            a0,
            n,
            deadline,
            eta0=0.6,
            alpha0=5.0,
            alpha_max=250.0,
            alpha_factor=1.008,
            max_iters=max_iters,
        )

        # ------- record improvement ---------------------------------------------
        if max_val < best_score - 1e-12:
            best_score = max_val
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        # ------- optional LP refinement -----------------------------------------
        if _SCIPY_AVAILABLE and n <= 3000 and max_val < best_score * 0.98:
            refined = _lp_refine(best_seq, deadline)
            _, refined_max, _ = _conv_max(refined)
            if refined_max < best_score:
                best_score = refined_max
                best_seq = refined.copy()
                print(f"[LP refine] run={run_id} n={n} refined score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 2) final polishing: extra smooth‑max optimisation + annealing + LP
    # ------------------------------------------------------------------
    if best_seq is not None:
        # a few more optimisation steps at the final length
        best_seq, best_score = _run_optimize_smooth(
            best_seq,
            best_seq.size,
            deadline,
            eta0=0.4,
            alpha0=15.0,
            alpha_max=400.0,
            alpha_factor=1.010,
            max_iters=30_000,
        )

        # short simulated‑annealing pass
        best_seq, best_score = _anneal(best_seq, deadline, rng)

        # final LP refinement (if affordable)
        best_seq = _lp_refine(best_seq, deadline)

        # recompute the exact external score for reporting
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best score = {final_score:.6f} (length = {len(best_seq)})")
    else:
        # fallback – uniform vector (should never happen)
        n = 1000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {final_score:.6f}")

    return best_seq.tolist()
