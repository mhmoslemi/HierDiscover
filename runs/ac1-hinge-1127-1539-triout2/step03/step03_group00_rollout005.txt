```python
def _adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyperparameters
    lr = 0.1  # base learning rate, will be scaled
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = 0.5
    max_beta = 2048.0
    # schedule: every 200 iterations multiply by 1.5 up to max_beta

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # adapt step size: we need scale such that sum constraint is satisfied after projection
        # Use step size = lr / (np.sqrt(v_hat) + eps)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        # beta schedule
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        # occasional jitter
        if t % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum*0.001/(beta+1), size=n)
            a += jitter
            a = _project_to_simplex(np.maximum(a,0), target_sum)
    return a
```

The objective is  

\[
\min_{a\ge 0,\;\sum a = \sqrt{2n}} \max_k (a*a)_k
\]

where `*` is the linear convolution.  
After normalisation the objective equals the maximal entry of the
self‑convolution, therefore we only have to minimise that maximum.

The implementation follows a *multi‑stage stochastic optimisation*:

1. **Diverse random initialisations** – power‑law, exponential,
   mixed, Dirichlet (both uniform and very sparse) and explicit sparse
   “spike’’ patterns.  
2. **Smooth‑max gradient descent** – we replace the hard maximum by a
   log‑sum‑exp with an annealing parameter `β`.  
   The gradient of this smooth surrogate can be written as a simple
   convolution, which we evaluate with FFTs.  
   An Adam optimiser (adaptive moments) performs the descent while the
   iterate is projected onto the ℓ¹‑simplex after each step.
3. **Simulated annealing refinement** – after gradient descent a fast
   SA routine moves random amounts of mass between two positions,
   accepting uphill moves with a temperature schedule.  Large jumps,
   occasional global jitter and a higher initial temperature
   increase the chance to escape local minima.
4. **Length sweep & time budget handling** – several sequence lengths
   (1500 … 10000) are tried in a round‑robin fashion; every candidate
   receives a bounded amount of time (gradient → SA).  The best sequence
   found before the deadline is returned.

All heavy work is done with NumPy‑FFT (`O(n log n)`) and the code obeys
the 1000 s time limit.

```python
"""Propose a non‑negative step function that minimises `evaluate_sequence`.

The algorithm mixes many random starts, an Adam optimiser on a
log‑sum‑exp smooth approximation of the max self‑convolution,
and a simulated‑annealing refinement.  Different sequence lengths are
tried until the time budget expires; the best sequence seen is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = s}
    (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation – simply the maximal convolution entry."""
    # a is already on the simplex (sum = sqrt(2·n))
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate of the maximum
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, b) where
        b = a*a (full convolution, length 2·n‑1)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
             = 2·(a ⋆ p)  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # b = a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities p
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))               # stable
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed maximum
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser on the log‑sum‑exp surrogate of max(a*a).
    The iterate is kept on the ℓ¹‑simplex (sum = sqrt(2·n)).
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    beta = 0.5
    max_beta = 4096.0

    rng = np.random.default_rng()
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

        # anneal β – fast increase to focus on the true maximum
        if t % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.6, max_beta)

        # occasional jitter to avoid plateaus
        if t % 500 == 0:
            jitter = rng.normal(
                scale=target * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a

def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that transfers a random fraction of mass between two indices.
    The temperature starts a few percent of the current score and is
    geometrically cooled.
    """
    a = a0.copy()
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score
    n = a.size
    rng = np.random.default_rng()

    # temperature proportional to the current objective
    T = cur_score * 0.05 + 1e-6
    cooling = 0.9995

    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-15:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 30 %) of mass from i to j
        delta = a[i] * rng.random() * 0.30
        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        if dE <= 0.0 or rng.random() < np.exp(-dE / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.002, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

def _random_initial_sequence_v2(n: int, rng) -> np.ndarray:
    """Create a diverse non‑negative sequence on the ℓ¹‑simplex."""
    target = np.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)
    choice = rng.integers(0, 8)

    if choice == 0:                     # pure power‑law
        p = rng.uniform(0.3, 0.8)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                   # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif choice == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.3, 0.8)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                   # Dirichlet, uniform concentration
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif choice == 4:                   # Dirichlet, very sparse (α < 1)
        c = rng.uniform(0.05, 0.3)
        a = rng.dirichlet(alpha=np.full(n, c)) * target

    elif choice == 5:                   # explicit sparse “spikes’’
        K = max(2, int(np.sqrt(n)))
        a = np.zeros(n)
        pos = rng.choice(n, size=K, replace=False)
        masses = rng.random(K)
        masses = masses / masses.sum() * target
        a[pos] = masses

    elif choice == 6:                   # uniform + tiny Gaussian perturbation
        a = np.full(n, target / n)
        a += rng.normal(scale=target * 0.001, size=n)

    else:                               # hybrid Dirichlet + extra spikes
        a = rng.dirichlet(alpha=np.full(n, 0.2)) * target
        K = max(2, int(np.sqrt(n) / 2))
        pos = rng.choice(n, size=K, replace=False)
        a[pos] += target * 0.1 * rng.random(K)

    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    `evaluate_sequence`.  The routine runs at most ``budget_s`` seconds,
    trying several lengths, diverse random starts, a short Adam descent
    and a simulated‑annealing refinement.  The best sequence encountered
    is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional: start from an already‑known construction
    # ------------------------------------------------------------------
    init_from_global = globals().get("height_sequence_1")
    if isinstance(init_from_global, (list, tuple, np.ndarray)):
        a = np.asarray(init_from_global, dtype=float)
        n0 = a.size
        target0 = np.sqrt(2.0 * n0)
        a = _project_to_simplex(a, target0)
        score0 = _score(a)
        best_seq = a.copy()
        best_score = score0
        print(f"[{time.time():.1f}] Starting from height_sequence_1 (n={n0}) "
              f"score={score0:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more degrees of freedom
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 5000, 8000, 10000]

    # round‑robin over the lengths while time permits
    length_idx = 0
    while time.time() < deadline:
        n = lengths[length_idx % len(lengths)]
        length_idx += 1
        target = np.sqrt(2.0 * n)

        # ----------- random start ------------------------------------
        a0 = _random_initial_sequence_v2(n, rng)
        init_sc = _score(a0)
        if init_sc < best_score:
            best_score = init_sc
            best_seq = a0.copy()
            print(f"[{time.time():.1f}] New global best (rand) n={n} score={best_score:.6f}")

        # ----------- allocate time for optimisation ------------------
        remaining = deadline - time.time()
        if remaining < 6.0:
            break

        # split remaining time: 60 % for Adam, the rest for SA
        gd_dead = time.time() + min(12.0, remaining * 0.6)
        a_opt = _adam_optimize(a0, gd_dead)

        # SA uses whatever time is left after the gradient phase
        if time.time() < deadline:
            a_ref = _simulated_annealing(a_opt, deadline)
        else:
            a_ref = a_opt

        sc = _score(a_ref)
        if sc < best_score:
            best_score = sc
            best_seq = a_ref.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback: uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_seq = a
        best_score = _score(a)
        print(f"[{time.time():.1f}] Fallback uniform n={n} score={best_score:.6f}")

    # Trim insignificant trailing zeros for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")

    return best_seq.tolist()
```
