"""
Propose a non‑negative step function (vector of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Broad parametric sweep** – generate many “U‑shaped’’ families
   (linear, quadratic, exponential, power‑law, exp‑power) with a wide
   range of parameters, normalise each to the ℓ₁‑simplex
   Σa = √(2 n) and keep the best one.

2. **Smoothed Adam optimisation** – minimise a
   log‑sum‑exp smoothed version of the maximum convolution while staying
   on the simplex.  An Adam‑style adaptive step plus a decreasing
   smoothing temperature τ allow fast progress; occasional random pair
   transfers and a modest acceptance‑of‑worse‑step rule help escape
   plateaus.

3. **Sub‑gradient polishing & simulated‑annealing final pass** – use the
   exact sub‑gradient of the (non‑smoothed) max‑convolution to drive the
   objective down, followed by many tiny random mass‑transfer moves
   (annealed temperature) to polish the solution.

The routine respects a 1000‑second wall‑clock budget and always returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of a vector using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
        {x | x_i ≥ 0, Σ x_i = target}
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small amount of mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls edge enhancement.
    The result is projected onto the ℓ₁‑simplex of total mass `target_sum`.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_power_shape(
    n: int,
    target_sum: float,
    param: float,
    exponent: float,
) -> np.ndarray:
    """
    Valley shape based on (1 + param * d) ** exponent,
    where d = |i - centre| / centre ∈ [0,1].
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = np.power(1.0 + param * d, exponent, where=1 + param * d > 0)
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_exp_power_shape(
    n: int,
    target_sum: float,
    param: float,
    alpha: float,
) -> np.ndarray:
    """
    Valley shape based on exp(param * d**alpha), d as above.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = np.exp(param * np.power(d, alpha))
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (gradient, current max convolution value).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)           # length 2n‑1
    max_conv = conv.max()
    # smoothing temperature (never too small)
    tau = max(tau_factor * max_conv, 1e-12)

    # stable soft‑max weights over the convolution entries
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of the weight vector with the reversed a
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]

    # central slice corresponds to the gradient of max‑conv w.r.t. a
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a, a)).
    Returns (gradient, current max, index of the maximal entry).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())

    # indices i that actually appear in the maximal entry
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)

    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv, idx_max

    # incorporate previous construction if present
    if base_vec is not None:

    # ------------------------------------------------------------------

        if best_score < 1.503:

    return best_seq.tolist()
