import time
import numpy as np

def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = S}."""
    # Reference: Duchi et al. (2008)
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), S):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # fallback to uniform distribution
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full (non‑circular) convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with ℓ₁‑sum = 1."""
    S = 1.0
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=max(a.mean(), 1e-12) * 1e-3, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)  # Dirichlet(1) → uniform on simplex before projection
    a = _project_to_simplex(a, S)
    return a


def _softmax_update(a: np.ndarray, tau: float, eta: float) -> tuple[np.ndarray,
                                                                  np.ndarray,
                                                                  float]:
    """
    One projected descent step on the smooth (log‑sum‑exp) approximation of max.
    Returns the projected vector, its convolution, and the new maximum.
    """
    n = a.shape[0]
    S = np.sum(a)  # keep current ℓ₁‑sum (normally 1)

    # current convolution
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau <= 1e-12:
        # hard‑max sub‑gradient (identical to the sub‑gradient branch below)
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]
    else:
        # smooth soft‑max gradient
        diff = b - max_b
        exps = np.exp(diff / tau - np.max(diff / tau))  # stability
        w = exps / np.sum(exps)  # weights, sum to 1

        # cross‑correlation a ⋆ w
        a_rev = a[::-1]
        m = 1 << ((len(w) + n - 1).bit_length())
        Fw = np.fft.rfft(w, n=m)
        Fa_rev = np.fft.rfft(a_rev, n=m)
        conv = np.fft.irfft(Fw * Fa_rev, n=m)[: len(w) + n - 1]
        g = 2.0 * conv[n - 1: n - 1 + n]

    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    b_new = _full_convolution_fft(a_proj)
    max_new = np.max(b_new)

    return a_proj, b_new, max_new


def _grad_max(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the entry of the self‑convolution that attains the maximum."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    k = int(np.argmax(b))
    idx = np.arange(n)
    partner = k - idx
    mask = (partner >= 0) & (partner < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[partner[mask]]
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]          # diagonal term counted only once
    return g, k, np.max(b)


def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mixed optimiser mixing:
      – projected soft‑max gradient descent,
      – occasional simulated‑annealing mass swaps,
      – occasional jitter,
      – hard‑max momentum steps when the temperature is low.
    Returns the best vector found and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sum(a)

    # best record
    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters (tuned for speed vs quality)
    eta = 0.6
    eta_decay = 0.9998
    move_prob = 0.13
    epsilon_move = 0.05
    temp = cur_max * 0.2
    temp_decay = 0.9995
    jitter_prob = 0.015
    jitter_scale = 0.025
    tau = cur_max
    tau_decay = 0.9994

    # momentum for the hard‑max phase
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.time() < deadline:
        # ---- 1) simulated‑annealing mass move ---------------------------------
        if rng.random() < move_prob:
            # choose source index weighted by current contribution to the max entry
            g, k_max, _ = _grad_max(a)
            probs = np.abs(g) + 1e-12
            probs /= probs.sum()
            i = rng.choice(n, p=probs)
            if a[i] <= 1e-12:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = epsilon_move * a[i]
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            b_candidate = _full_convolution_fft(a_candidate)
            cand_max = np.max(b_candidate)

            accept = False
            if cand_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                a = a_candidate
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temp *= temp_decay
            continue

        # ---- 2) gradient step -------------------------------------------------
        if tau > 1e-9:
            a, b, cur_max = _softmax_update(a, tau, eta)
        else:
            # hard‑max sub‑gradient with momentum
            g, _, cur_max = _grad_max(a)
            momentum = beta * momentum + (1.0 - beta) * g
            a_candidate = a - eta * momentum
            a = _project_to_simplex(a_candidate, S)
            cur_max = np.max(_full_convolution_fft(a))

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # ---- 3) occasional jitter ---------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=max(a.mean(), 1e-12) * jitter_scale,
                               size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            cur_max = np.max(_full_convolution_fft(a))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # decay schedules
        eta *= eta_decay
        tau *= tau_decay

    return best_a, best_max


def _refine_sequence(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fine‑grained deterministic refinement using a small constant step and
    momentum on the hard‑max sub‑gradient. Stops when time expires.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sum(a)

    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # small constant step size, momentum parameters
    eta = 0.02
    beta = 0.9
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        g, _, cur_max = _grad_max(a)
        momentum = beta * momentum + (1.0 - beta) * g
        a_candidate = a - eta * momentum
        a_candidate = _project_to_simplex(a_candidate, S)
        b_candidate = _full_convolution_fft(a_candidate)
        cand_max = np.max(b_candidate)

        if cand_max < cur_max:
            a = a_candidate
            cur_max = cand_max
            if cand_max < best_max:
                best_max = cand_max
                best_a = a.copy()
        else:
            # occasional uphill move to escape tiny plateaus
            if rng.random() < 0.005:
                a = a_candidate
                cur_max = cand_max

        # tiny jitter every now and then
        if rng.random() < 0.003:
            a = np.clip(a + rng.normal(scale=max(a.mean(),1e-12)*0.008,
                                       size=n), 0.0, None)
            a = _project_to_simplex(a, S)
            cur_max = np.max(_full_convolution_fft(a))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a, best_max


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function height vector that minimises
    evaluate_sequence(...).
    The algorithm runs many independent restarts of a mixed optimiser
    (soft‑max gradient descent + simulated annealing) and finishes with a
    fine‑grained momentum refinement.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # try to reuse a previously discovered construction
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    while time.time() < deadline:
        attempt += 1
        # pick a length – we explore a fairly wide interval
        if base_seq is not None:
            max_len = max(1000, min(12000, int(base_seq.size * 1.3)))
        else:
            max_len = 8000
        n = rng.integers(800, max_len + 1)

        a0 = _initial_sequence(base_seq, n, rng)

        # give each restart ~30 s (or the remaining budget, whichever smaller)

        score = 2.0 * n * max_b / (np.sum(a_opt) ** 2)
        if score < best_score:

"""
Search for a step‑function height vector that minimizes
evaluate_sequence(...).

The algorithm:
  * many independent restarts,
  * each restart runs a mixed optimiser that combines
        – projected soft‑max gradient descent,
        – simulated‑annealing mass swaps,
        – occasional jitter,
        – hard‑max sub‑gradient steps with momentum once the
          soft‑max temperature becomes tiny,
  * the best candidate is refined with a fine‑grained momentum
    optimiser that works solely on the hard‑max sub‑gradient,
  * the final sequence is returned and stored in the global
    variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), S):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:                     # fallback to uniform distribution
        return np.full_like(v, S / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Initialise a feasible vector of length n (ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with ℓ₁‑sum = 1."""
    S = 1.0
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=max(a.mean(), 1e-12) * 1e-3, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                     # Dirichlet(1) before projection
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# One soft‑max (or hard‑max) gradient step followed by projection
# ----------------------------------------------------------------------
def _softmax_update(a: np.ndarray, tau: float, eta: float) -> tuple[np.ndarray,
                                                                  np.ndarray,
                                                                  float]:
    """
    Perform a single step:
      – compute b = a*a,
      – compute soft‑max weights w_k ∝ exp(b_k / τ),
      – evaluate the gradient g_i = 2 Σ_j w_{i+j} a_j,
      – take an additive step a ← a – η·g,
      – project onto the ℓ₁‑simplex.
    Returns the new vector, its convolution, and the new maximal entry.
    """
    n = a.shape[0]
    S = np.sum(a)

    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau <= 1e-12:                     # treat as hard max (sub‑gradient)
        k = int(np.argmax(b))
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]
    else:                                 # smooth soft‑max gradient
        diff = b - max_b
        # stable soft‑max
        exp_arg = diff / tau - np.max(diff / tau)
        exps = np.exp(exp_arg)
        w = exps / np.sum(exps)

        a_rev = a[::-1]
        m = 1 << ((len(w) + n - 1).bit_length())
        Fw = np.fft.rfft(w, n=m)
        Fa_rev = np.fft.rfft(a_rev, n=m)
        conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n - 1]
        g = 2.0 * conv[n - 1:n - 1 + n]

    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    b_new = _full_convolution_fft(a_proj)
    max_new = np.max(b_new)

    return a_proj, b_new, max_new


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _grad_max(a: np.ndarray) -> tuple[np.ndarray, int, float]:
    """Return sub‑gradient g, location k of the maximal entry, and that maximum."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    k = int(np.argmax(b))
    idx = np.arange(n)
    partner = k - idx
    mask = (partner >= 0) & (partner < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[partner[mask]]
    if k % 2 == 0:
        d = k // 2
        if 0 <= d < n:
            g[d] = a[d]                # diagonal counted once
    return g, k, np.max(b)


# ----------------------------------------------------------------------
# Local optimiser (one restart)
# ----------------------------------------------------------------------
def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mixed optimisation:
      – projected soft‑max gradient descent,
      – occasional simulated‑annealing mass swaps,
      – occasional jitter,
      – hard‑max momentum steps when τ becomes tiny.
    Returns the best vector encountered and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sum(a)

    b = _full_convolution_fft(a)
    cur_max = np.max(b)
    best_a = a.copy()
    best_max = cur_max

    # ------------------- hyper‑parameters -------------------
    eta = 0.6
    eta_decay = 0.9998
    move_prob = 0.13
    epsilon_move = 0.05
    temp = cur_max * 0.2
    temp_decay = 0.9995
    jitter_prob = 0.015
    jitter_scale = 0.025
    tau = cur_max                      # start temperature for soft‑max
    tau_decay = 0.9994

    # momentum for the hard‑max phase
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) random mass swap (simulated annealing)
        # --------------------------------------------------------------
        if rng.random() < move_prob:
            g, _, _ = _grad_max(a)
            probs = np.abs(g) + 1e-12
            probs /= probs.sum()
            i = rng.choice(n, p=probs)
            if a[i] <= 1e-12:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = epsilon_move * a[i]
            a_cand = a.copy()
            a_cand[i] -= delta
            a_cand[j] += delta
            b_cand = _full_convolution_fft(a_cand)
            cand_max = np.max(b_cand)

            accept = False
            if cand_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                a = a_cand
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temp *= temp_decay
            continue

        # --------------------------------------------------------------
        # 2) projected gradient step
        # --------------------------------------------------------------
        if tau > 1e-9:
            a, _, cur_max = _softmax_update(a, tau, eta)
        else:
            # hard‑max sub‑gradient with momentum
            g, _, cur_max = _grad_max(a)
            momentum = beta * momentum + (1.0 - beta) * g
            a_cand = a - eta * momentum
            a = _project_to_simplex(a_cand, S)
            cur_max = np.max(_full_convolution_fft(a))

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # --------------------------------------------------------------
        # 3) occasional jitter
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=max(a.mean(), 1e-12) * jitter_scale,
                               size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            cur_max = np.max(_full_convolution_fft(a))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # decay schedules
        eta *= eta_decay
        tau *= tau_decay

    return best_a, best_max


# ----------------------------------------------------------------------
# Fine‑grained refinement (hard‑max with momentum)
# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Deterministic refinement that uses a small constant step size,
    momentum on the hard‑max sub‑gradient and tiny jitter.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sum(a)

    cur_max = np.max(_full_convolution_fft(a))
    best_a = a.copy()
    best_max = cur_max

    eta = 0.02
    beta = 0.9
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        g, _, cur_max = _grad_max(a)
        momentum = beta * momentum + (1.0 - beta) * g
        a_cand = a - eta * momentum
        a_cand = _project_to_simplex(a_cand, S)
        cand_max = np.max(_full_convolution_fft(a_cand))

        if cand_max < cur_max:
            a = a_cand
            cur_max = cand_max
            if cand_max < best_max:
                best_max = cand_max
                best_a = a.copy()
        else:
            if rng.random() < 0.005:          # occasional uphill move
                a = a_cand
                cur_max = cand_max

        # tiny jitter to avoid stagnation
        if rng.random() < 0.003:
            a = np.clip(a + rng.normal(scale=max(a.mean(), 1e-12) * 0.008,
                                       size=n), 0.0, None)
            a = _project_to_simplex(a, S)
            cur_max = np.max(_full_convolution_fft(a))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a, best_max
