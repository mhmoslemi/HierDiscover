def _adam_optimize(a0, deadline):
   a = a0.copy()
   n = a.shape[0]
   target_sum = np.sqrt(2.0*n)
   best_a = a.copy()
   best_score = _score(a)
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   t = 0
   beta = 1.0
   max_beta = 512.0
   lr0 = target_sum * 0.2   # base
   rng = np.random.default_rng()
   while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = lr0 / (np.sqrt(v_hat) + eps)  # use vector lr
        a = a - lr * m_hat
        # Projection
        a = _project_to_simplex(a, target_sum)
        # Occasionally add jitter
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum*0.001/(beta+1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target_sum)
        # Score
        val = _score(a)
        if val < best_score:
            best_score = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] ADAM score={best_score:.6f} beta={beta:.2f}")
        # Increase beta slowly.
        if t % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)
   return best_a, best_score

def _hill_climb_v2(a, deadline):
   a = a.copy()
   n = a.shape[0]
   best_a = a.copy()
   best_val = _score(a)
   target_sum = np.sqrt(2.0*n)
   rng = np.random.default_rng()
   while time.time()<deadline:
        # compute convolution via fft once? Use np.convolve for speed (cost O(n^2) maybe heavy). For n up to maybe 5000, O(n^2) maybe too heavy in many iterations.
        # So use FFT for conv to compute both conv and maybe sensitivity quickly.
        conv = _max_convolution_fft_full(a)  # compute full conv values
        k_max = int(np.argmax(conv))
        # compute sensitivity vector that tells contribution of each a_i to conv[k_max], i.e., a_{k_max-i} if in range.
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # indices i that contribute: i from lo to hi inclusive. a[i] multiplies with a[k_max - i].
        sens[lo:hi+1] = a[k_max - np.arange(lo, hi+1)]
        # compute weight = a * sens (contribution to peak)
        weight = a * sens
        # Choose donor index with high weight
        donor = int(np.argmax(weight))
        # Choose receiver index with low weight (lowest 10% values).
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size==0:
            break
        receiver = int(rng.choice(candidates))
        if donor==receiver or a[donor]<=1e-12:
            continue
        # move portion: compute maximal delta such that donor stays >=0 and sum unchanged.
        max_delta = a[donor]  # can move all
        delta = max_delta * rng.random() * 0.2 # up to 20%
        a[donor] -= delta
        a[receiver] += delta
        # projection not needed because sum preserved and nonneg.
        new_val = _score(a)
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] HC improved={best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta
   return best_a, best_val

def _full_convolution_fft(a):
    size = 1 << ((2*a.size-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*a.size-1]

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Candidate generation** – for several lengths *n* create a diverse set of
   initial sequences:
   * power‑law a_i ∝ (i+1)^(-p)
   * exponential a_i ∝ exp(-λ·i)
   * shifted inverse linear a_i ∝ 1/(i+α)
   * a few random Dirichlet samples.
   All candidates are projected onto the simplex
   Σa_i = √(2·n) (the normalisation that makes the objective equal the
   maximal convolution entry).

2. **Coarse filtering** – keep the best few (default 2) candidates per length
   according to the fast score ``_score`` (exact maximal convolution when the
   sum is normalised).

3. **Adam optimisation** – starting from each kept candidate run a projected
   Adam optimiser on the *soft‑max* smooth approximation of the max‑convolution.
   The temperature β is annealed (1 → 512).  After each gradient step the
   vector is projected back onto the simplex.  Occasionally a small Gaussian
   jitter is added to escape shallow plateaus.

4. **Guided hill‑climbing** – with the refined vector, repeatedly move a small
   amount of mass from indices that contribute most to the current peak
   (high sensitivity) to indices that contribute least.  A move is kept only
   if it lowers the true objective (max‑convolution).  This fine‑tunes the
   solution without any gradient approximations.

5. **Best‑of‑all** – the globally best sequence found before the time budget
   expires is returned (trailing zeroes are trimmed).

The whole routine respects the supplied ``budget_s`` (default 1000 s) and
periodically prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation: with Σa = √(2·n) the score equals max(conv)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:   # should never happen after projection
        return np.inf
    return 2.0 * n * max_b / (s * s)   # equals max_b when s=√(2n)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Generate exp(-lam·i) sequence and normalise."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _inverse_linear_sequence(n: int, shift: float, target_sum: float) -> np.ndarray:
    """Generate (i+1+shift)^(-1) sequence and normalise."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx + shift)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (g, b) where b = a*a (full convolution) and
    g = ∇_a  log‑sum‑exp_{β}(b)  = 2·(a ⋆ p)
    with p_i = exp(β(b_i‑max_b))/∑exp.
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # shape (2·n‑1,)

    # convolution a_rev * p  (needed for gradient)
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]       # keep the central n entries
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float):
    """Projected Adam on the soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    soft_beta = 1.0
    max_soft_beta = 512.0
    lr0 = target_sum * 0.20
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, soft_beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # element‑wise learning rate
        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # keep feasible
        a = _project_to_simplex(a, target_sum)

        # occasional stochastic jitter
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (soft_beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] ADAM  score={best_val:.6f}  β={soft_beta:.2f}")

        # anneal the temperature
        if t % 500 == 0 and soft_beta < max_soft_beta:
            soft_beta = min(soft_beta * 1.5, max_soft_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (mass‑transfer)
# ----------------------------------------------------------------------


def _hill_climb_v2(a: np.ndarray, deadline: float):
    """Move mass from high‑sensitivity positions to low‑sensitivity ones."""
    a = a.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07                     # max fraction of a donor’s mass to move

    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # sensitivity vector: contribution of each a_i to conv[k_max]
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        sens[lo:hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        weight = sens * a                 # contribution of each coordinate to the peak
        if weight.max() == 0:
            break

        donor = int(np.argmax(weight))
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver or a[donor] <= 1e-12:
            continue

        # transfer a random fraction of the donor’s mass
        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] HC    improved={best_val:.6f}")
        else:
            # revert the move
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine creates a diverse set of initial shapes, refines them with a
    projected Adam optimiser and finishes with a guided hill‑climbing phase.
    The best sequence before the time budget expires is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1. Generate a broad pool of initial candidates
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 1800, 2100, 2400, 3000, 4000]
    all_candidates = []                     # (base_score, array, n)

    for n in lengths:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)

        # power‑law
        for p in np.linspace(0.40, 0.75, 8):
            a = _power_law_sequence(n, p, target)
            all_candidates.append((_score(a), a, n))

        # exponential
        for lam in np.linspace(0.001, 0.020, 8):
            a = _exponential_sequence(n, lam, target)
            all_candidates.append((_score(a), a, n))

        # shifted inverse linear
        for shift in [0.5, 1.0, 2.0, 5.0]:
            a = _inverse_linear_sequence(n, shift, target)
            all_candidates.append((_score(a), a, n))

        # a few random Dirichlet‑like samples
        for _ in range(3):
            x = rng.random(n)
            a = x / x.sum() * target
            all_candidates.append((_score(a), a, n))

    # keep only the most promising few
    all_candidates.sort(key=lambda t: t[0])
    top_candidates = all_candidates[:12]        # at most 12 sequences

    # ------------------------------------------------------------------
    # 2. Refine each candidate (Adam → hill‑climbing)
    # ------------------------------------------------------------------
    for base_score, a0, n in top_candidates:
        if time.time() >= deadline:
            break

        # slice a sensible amount of remaining budget for this candidate
        remaining = deadline - time.time()
        adam_allowed = min(25.0, remaining * 0.6)          # up to 25 s for Adam
        adam_deadline = time.time() + adam_allowed

        a_opt, val = _adam_optimize(a0, adam_deadline)

        # hill‑climbing for the rest of the time budget
        if time.time() < deadline:
            a_opt, val = _hill_climb_v2(a_opt, deadline)

        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"*** GLOBAL BEST ***  n={n}  base={base_score:.6f}  final={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fallback – use a known construction if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        try:
            # ``height_sequence_1`` may be supplied by the environment
            best_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
            n = best_seq.size
            best_seq = _project_to_simplex(best_seq, np.sqrt(2.0 * n))
        except Exception:
            # simple uniform fallback
            n = 1000
            best_seq = np.ones(n, dtype=np.float64)
            best_seq *= np.sqrt(2.0 * n) / n

    # Trim trailing zeros (they do not affect the objective)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
