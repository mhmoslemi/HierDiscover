"""
Propose a non‑negative step‑function (vector a) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The evaluation routine normalises the sum to √(2 n), therefore the
objective equals the maximum entry of the self‑convolution.
The algorithm consists of four phases:

1. **Warm‑start** – generate many random points on the ℓ₁‑simplex
   (Dirichlet draws, simple U‑shapes, a few asymmetric sketches) and keep the best.
2. **Smoothed Adam descent** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp).  The gradient is obtained by a single
   FFT‑based convolution; an Adam optimiser (first/second moment
   adaptation) drives the iterates while the smoothing temperature is
   gradually reduced.
3. **Sub‑gradient refinement** – switches to the true sub‑gradient of the
   max‑convolution with a decreasing step‑size.
4. **Polishing** – random pair‑mass transfers accepted only when they
   improve the maximum convolution.

All updates are projected onto the simplex {x≥0, Σx=S} where
S=√(2 n).  The routine respects a 1000 s time‑budget and returns the
best sequence seen so far.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative)."""
    return float(conv_self_fft(a).max())


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x_i ≥ 0, Σx_i = target}.
    O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """Transfer a small amount of mass from a random index i to j."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_u_shape(n: int, target_sum: float,
                     kind: str, param: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ sequences.
    kind ∈ {'linear','quadratic','exponential','power','inverse'}.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def subgradient_step(a: np.ndarray, target_sum: float,
                     step: float) -> tuple[np.ndarray, float]:
    """
    One sub‑gradient step on the true max‑convolution.
    Returns (new_a, max_before_step).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    eps = max(1e-12, 1e-9 * max_conv)
    mask = (conv >= max_conv - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        mask[conv.argmax()] = 1.0
        cnt = 1.0

    a_rev = a[::-1]
    L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
    grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, max_conv

        tau = max(tau_factor * cur_max, 1e-12)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max (FFT‑based)
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)

        a = a_new

        # keep best solution seen so far
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {it:07d} – improved to {best_score:.6f}")
        else:
            no_improve += 1

        # adaptive schedules
        if it % 800 == 0:
            base_lr *= 0.97
            tau_factor = max(0.001, tau_factor * 0.96)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.012:
            jitter = np.random.randn(n) * (target_sum / n) * 0.0004
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)

        # random pair transfer accepted only if it improves
        if np.random.rand() < 0.01:
            cand = random_pair_transfer(a.copy(), target_sum, scale=0.008)
            cand_score = compute_max_conv(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand.copy()
                print(f"  Transfer improved to {best_score:.6f}")

        # long stagnation → stronger jitter
        if no_improve >= 5000:
            jitter = np.random.randn(n) * (target_sum / n) * 0.001
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {it:07d} – heavy jitter applied")

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement (true max)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Adam phase finished; best score {best_score:.6f}")
    sub_step = 0.03 * (target_sum / n)
    sub_it = 0
    while time.time() < deadline:
        sub_it += 1
        a, cur_max = subgradient_step(a, target_sum, sub_step)
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"Sub‑grad {sub_it:06d} – improved to {best_score:.6f}")

        if sub_it % 500 == 0:
            sub_step *= 0.95

        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.004)

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 3 – final polishing (random transfers)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Sub‑gradient phase finished; best score {best_score:.6f}")

    for _ in range(60):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.003)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Returning best sequence with score {best_score:.6f}")
    return best_seq.tolist()
