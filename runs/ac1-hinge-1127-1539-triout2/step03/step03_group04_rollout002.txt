"""
Optimisation of non‑negative step functions for the max‑convolution benchmark.

Strategy:
1. Initialise many candidate sequences for several lengths using
   - power‑law with a shift,
   - exponential decay,
   - random Dirichlet draws,
   - the provided `height_sequence_1` if available.

   `max(a*a)` (the sub‑gradient at the current maximal index is
   g_i = 2 a_{k‑i}).  After each step the vector is projected onto the
   simplex {x ≥ 0, Σ x = √(2 n)}.  Step size is adapted by a simple
   decay schedule.

3. Refine with two complementary local searches:
   * Guided hill‑climb that moves mass from indices that contribute most
     to the current peak to low‑contributing indices.
   * Randomised Monte‑Carlo mass exchanges with a tiny Metropolis
     acceptance rule to escape shallow minima.

4. Keep the best sequence seen before the outer time budget expires.
"""

import time
import numpy as np
from math import sqrt

# helper functions ---------------------------------------------------------

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a computed via FFT."""
    return float(_full_convolution_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact competition objective for a (assuming sum > 0)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _powerlaw_shifted(n: int, p: float, shift: float, target_sum: float) -> np.ndarray:
    """Generate a shifted power‑law sequence and normalise to target_sum."""
    idx = np.arange(1, n + 1, dtype=np.float64) + shift
    # avoid division by zero or negative bases
    idx = np.where(idx <= 0, 1e-12, idx)
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_decay(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Generate a simple exponential decay and normalise."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    a *= target_sum / a.sum()
    return a


def _random_dirichlet(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using a Dirichlet(1,…,1) distribution."""
    x = rng.gamma(1.0, 1.0, size=n)
    a = x * target_sum / x.sum()
    return a

def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the max‑convolution objective.
    Uses the exact sub‑gradient at the current maximal index.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    best = a.copy()
    best_val = _score(a)

    step_base = target_sum * 0.20   # initial magnitude
    decay = 0.9995                  # geometric decay of step size
    step = step_base

    it = 0
    while time.time() < deadline:
        # fast convolution and argmax
        conv = _full_convolution_fft(a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # update and project
        if np.linalg.norm(g) > 0:
            a = a - step * g
        a = _project_to_simplex(a, target_sum)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            # optional logging
            # print(f"[{time.time():.1f}] SG  val={val:.6f} it={it}")

        # step‑size schedule
        step = max(step * decay, 1e-8)
        it += 1

    return best, best_val


def _guided_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Move mass from indices that contribute heavily to the peak to lighter ones.
    Works on the current maximal convolution entry.
    """
    n = a.size
    best = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07  # max proportion of donor mass to shift in one move

    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        k_max = int(np.argmax(conv))

        # contribution of each coordinate to conv[k_max] is a[k_max-i]
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        sens[lo:hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        weight = sens * a
        if weight.max() == 0:
            break
        donor = int(np.argmax(weight))
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver or a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
                # print(f"[{time.time():.1f}] GH  improved={best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta

    return best, best_val


def _random_monte_carlo(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑style Monte‑Carlo mass‑exchange, useful for escaping
    shallow minima after deterministic refinements.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    # maximum proportion of a donor that may be moved in a single proposal
    max_move = 0.15
    temperature = 1e-3

    while time.time() < deadline:
        # pick donor proportional to current mass (more mass → more likely)
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-12:
            continue
        receiver = rng.integers(0, n)
        if receiver == donor:
            continue
        # propose moving a fraction
        frac = rng.random() * max_move
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < best_val:
            best_val = new_val
            best = a.copy()
            # print(f"[{time.time():.1f}] MC  improved={best_val:.6f}")
            cur_val = new_val
        else:
            # Metropolis acceptance
            accept_prob = np.exp(-(new_val - best_val) / temperature)
            if rng.random() < accept_prob:
                cur_val = new_val
            else:
                # revert
                a[receiver] -= delta
                a[donor] += delta

        # simple cooling schedule
        temperature *= 0.9999
        if temperature < 1e-9:
            temperature = 1e-9

    return best, best_val

            best_seq = a_opt.copy()

    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Return as plain Python list (required by the evaluation harness)
    return best_seq.tolist()

"""
Optimisation of non‑negative step functions for the max‑convolution benchmark.

Strategy
========
1. **Generate diverse initial candidates** for several vector lengths using
   - shifted power‑law shapes,
   - exponential decays,
   - random Dirichlet draws,
   - the provided `height_sequence_1` if it exists.

   All vectors are normalised to Σa_i = √(2·n) and projected onto the simplex.

2. **Sub‑gradient descent** on the exact objective.
   For the current maximal convolution index k the sub‑gradient is
       g_i = 2·a_{k‑i}   (zero when the index is out of range).
   After each gradient step the vector is projected back onto the simplex.
   A simple geometric decay controls the step size.

3. **Guided hill‑climbing**:
   Move mass from indices that contribute most to the current peak to
   indices that contribute least.  Each accepted move reduces the score.

4. **Monte‑Carlo fine‑tuning**:
   Random mass exchanges are accepted if they improve the score or with a
   tiny Metropolis probability to escape shallow local minima.

5. Keep the best sequence observed before the time budget expires.
"""

import time

# Core utilities (convolution, scoring, projection)

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_full_convolution_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact competition objective."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
