import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    conv_expr = cp.conv(cp.Constant(f), g)  # returns length 2n-1
    constraints = [conv_expr <= M]
    # note: ext constraints linear but conv_expr shape maybe (2n-1,) vector.
    obj = cp.Maximize(cp.sum(g))  # or cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLPK, glpk={'msg_lev':'GLP_MSG_OFF'})
    return g.value

"""
Search algorithm for minimizing the maximal self‑convolution of a
non‑negative step function.

Strategy
--------
1. Choose a problem size n (use the length of a previously found
   construction if available, otherwise 8000 – 9000).  The ℓ₁–norm is
   fixed to √(2 n), which makes the evaluation function equal to the
   maximal convolution value.

2. Perform several random restarts.  For each restart:
   * Initialise a on the simplex (either the previous best construction
     or a random Dirichlet vector).
   * Run an Adam optimiser on a *smooth* surrogate of the max‑convolution
     (log‑sum‑exp smoothing).  The gradient is obtained via FFTs and the
     iterate is projected back onto the simplex after each step.
   * After the Adam phase, execute a cheap deterministic “max‑reduction”
     step that moves a small amount of mass from the index pair that
     contributes most to the current maximal convolution entry towards the
     globally smallest entry.
   * Finish with a short random‑pair‑transfer polishing pass.

3. Keep the best sequence seen over all restarts.  The whole routine stops
   after the allotted time budget (≈ 1000 s) and always returns a feasible
   sequence.

All helper functions are top‑level; the code uses only numpy and the
standard library to stay lightweight.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to `x`."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using the FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
        {x | x_i ≥ 0, ∑_i x_i = target }.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then project back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (gradient, max_conv).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    # smoothing temperature (never zero)
    tau = max(tau_factor * max_conv, 1e-12)

    # softmax weights over the convolution entries
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w * a_rev) restricted to the central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]

    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv
