def _run_softmax_descent(a0, n, deadline, rng, max_iters=50000):
    a = a0.copy()
    S = np.sqrt(2.0 * n)
    # hyperparams
    eta = 0.5
    beta1=0.9
    beta2=0.999
    epsilon=1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lam = 1.0  # initial smoothing
    lam_factor = 1.02
    lam_max = 40.0
    # best tracking
    best_score = np.inf
    best_a = a.copy()
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute conv
        conv = _conv_full_fft(a)   # returns np.ndarray length 2n-1
        cur_max = np.max(conv)
        # compute objective value: eval = 2 * n * cur_max / (S**2) = cur_max (since S**2=2n)
        cur_score = cur_max   # because normalized sum
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[softmax descent improve] time={time.time():.1f}s n={n} iter={t} score={cur_score:.6f}")

        # Compute smoothed weights
        # to avoid overflow, subtract max for stability
        maxc = cur_max
        exp_vals = np.exp(lam * (conv - maxc))
        w = exp_vals / np.sum(exp_vals)

        # gradient
        g = _gradient_from_weights(w, a)  # shape (n,)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        # multiplicative update: a = a * exp(-eta * m_hat / (sqrt(v_hat)+epsilon))
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + epsilon))

        # renormalize sum to S
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # update lambda
        lam = min(lam * lam_factor, lam_max)

        # optional jitter
        if rng.random() < 0.01:
            noise = rng.normal(scale=a.mean()*0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()
        # enforce positivity: already positive due to multiplicative update.

        # maybe break early if lam reached lam_max and improvement minimal
        # not necessary.

    # final evaluation also compute real score
    final_conv = _conv_full_fft(best_a)
    final_score = np.max(final_conv)
    return best_a, final_score

def _conv_full_fft(a):
    n = a.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _gradient_from_weights(w, a):
    # compute g_i = 2 * sum_{j} w_{i+j} * a_j
    n = a.shape[0]
    a_rev = a[::-1]
    # convolve w (len 2n-1) with a_rev (len n)
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(W * A_rev, n=fft_len)[:conv_len]
    # extract indices from n-1 to n-1 + n -1 inclusive.
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g

"""
search for a non‑negative step‑function that minimises
        2·n·max(a*a) / (sum(a)^2)

The previous implementation used a top‑K sub‑gradient and a simple
exponential update.  Here we improve the optimisation in three ways:

1. **Smooth max‑approximation** – we replace the hard maximum of the
   self‑convolution by a temperature‑controlled softmax
   (log‑sum‑exp).  The gradient of this surrogate is a true
   convolution of the weight vector with the reversed current
   sequence, which can be computed in O(n log n) with FFTs.  This gives a
   much more accurate direction than a noisy top‑K sub‑gradient.

2. **Adam‑style multiplicative updates** – after the gradient we apply
   an Adam‑like adaptive step (on the logarithm of the variables) and
   renormalise to the constant L₁‑norm  √(2 n).  The updates are always
   non‑negative.

3. **Annealing schedule** – the softmax temperature λ is increased
   slowly (starting from a smooth objective and ending at a hard max),
   while a small learning‑rate decay and occasional jitter help to
   escape plateaux.  The result of each run is refined with a cheap
   simulated‑annealing hill‑climb.

All phases respect the 1000 s budget, keep every coefficient inside the
allowed range, and keep track of the globally best sequence found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())    # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(np.max(_full_convolution(a)))


def _gradient_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate with respect to a.
    For a fixed weight vector w (length 2·n‑1, sum=1) the i‑th component
    is 2·∑_j w_{i+j}·a_j.
    Computed via a convolution of w with the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = w.shape[0] + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(W * A_rev, n=fft_len)[:conv_len]
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient descent with Adam‑style multiplicative
# updates.  The returned value is the best sequence and its *max‑conv*
# (which equals the external score because we keep Σa = √(2·n)).
# ----------------------------------------------------------------------


def _run_softmax_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 30000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    S = np.sqrt(2.0 * n)                      # target L1‑norm

    # Adam parameters (operating on the log‑space, i.e. multiplicative update)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.35                               # base step size

    # temperature schedule for the soft‑max
    lam = 0.5                                 # start smooth
    lam_factor = 1.06
    lam_max = 30.0

    best_a = a.copy()
    best_score = np.inf
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _full_convolution(a)
        cur_max = float(conv.max())          # max‑conv (our current score)
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            print(f"[softmax] time={time.time():.1f}s n={n} iter={it} score={cur_max:.6f}")

        # ---------- soft‑max weights ----------
        #   w_k ∝ exp(lam·(conv_k – max(conv)))   (stable version)
        exp_vals = np.exp(lam * (conv - cur_max))
        w = exp_vals / np.sum(exp_vals)

        # ---------- gradient ----------
        g = _gradient_from_weights(w, a)      # shape (n,)

        # ---------- Adam on the log‑space ----------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # multiplicative update (guarantees non‑negativity)
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # re‑normalise to the required L1‑norm
        total = a.sum()
        if total <= 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # ----- schedules -----
        lam = min(lam * lam_factor, lam_max)      # sharpen the max
        eta *= 0.9999                              # gentle decay

        # occasional jitter to escape shallow plateaus
        if rng.random() < 0.015:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

    # final score of the best vector found
    final_score = _max_convolution(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing hill‑climb for final polishing
# ----------------------------------------------------------------------


def _local_hill_climb(a_start: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 20000) -> tuple[np.ndarray, float]:
    a = a_start.copy()
    cur_max = _max_convolution(a)
    best_a = a.copy()
    best_max = cur_max

    # temperature is a fraction of the current objective
    temperature = max(0.05 * cur_max, 1e-4)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # move a random fraction (5‑20 %) of the donor’s mass
        frac = rng.uniform(0.05, 0.20)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new)

        # accept if better, otherwise with Metropolis probability
        if new_max < cur_max - 1e-12:
            a, cur_max = a_new, new_max
            if new_max < best_max:
                best_max, best_a = new_max, a_new.copy()
                print(f"[hill‑climb] time={time.time():.1f}s n={n} iter={it} max={new_max:.6f}")
        else:
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a, cur_max = a_new, new_max

        temperature *= 0.9995  # geometric cooling

        # occasional jitter to diversify search
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= np.sqrt(2.0 * n) / a.sum()
            cur_max = _max_convolution(a)
            if cur_max < best_max:
                best_max, best_a = cur_max, a.copy()
                print(f"[jitter] time={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}")

    return best_a, best_max
