def _zigzag(arr):
    n = len(arr)
    out = np.empty(n, dtype=arr.dtype)
    left = 0
    right = n-1
    for idx, val in enumerate(arr):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out
```

"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises the evaluation score:

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The search consists of
1. Random restarts with different lengths.
2. A projected Adam‑style gradient descent on a smooth
   approximation of the infinity‑norm (soft‑max of the full self‑convolution).
3. A permutation‑refinement phase that re‑orders the obtained magnitudes
   (zig‑zag, random shuffles, hill‑climbing swaps) to further reduce the
   maximum convolution entry.
All steps respect the 1000 s budget; the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term g_j = 2 * Σ_i w_{i+j} * a_i.
    a has length n, w has length 2n‑1.
    """
    n = a.shape[0]
    w_rev = w[::-1]                       # reverse for correlation
    conv_len = n + (2 * n - 1) - 1        # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)        # mapping to gradient indices
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth max utilities
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If a base sequence is supplied it is used as a seed (with tiny jitter);
    otherwise a Dirichlet‑(1) sample is drawn.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # tiny Gaussian jitter (≈0.1 % of the mean magnitude)
        sigma = 0.001 * (a.mean() if a.mean() > 0 else 1.0)
        a += rng.normal(scale=sigma, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)               # uniform (Dirichlet‑1) sample

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Local optimiser (Adam‑style projected gradient descent)
# ----------------------------------------------------------------------


def _optimize_local_adam(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence using a projected Adam optimiser on the
    smoothed max‑norm. Returns the best sequence encountered and its true
    maximal self‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    t = 0
    max_iter = 20000

    while time.time() < deadline and t < max_iter:
        t += 1

        # --- 1. full self‑convolution ---------------------------------
        conv = _full_convolution(a)

        # --- 2. smooth weights (soft‑max) ----------------------------
        # linear increase of α from 0 to ~10 over the run
        alpha = min(10.0, 0.0005 * t)
        w = _smooth_weights(conv, alpha)

        # --- 3. gradient ------------------------------------------------
        grad = _conv_with_weights(a, w)

        # --- 4. Adam update -------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # --- 5. projection onto simplex --------------------------------
        a = _project_simplex(a_new, S)

        # --- 6. evaluation ---------------------------------------------
        cur_score = np.max(_full_convolution(a))

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # --- 7. occasional tiny mass‑swap (simulated‑annealing flavour) -----
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = 0.01 * a[i] * rng.random()
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = _project_simplex(a_tmp, S)
                cand_score = np.max(_full_convolution(a_tmp))
                # simple Metropolis acceptance
                if cand_score <= cur_score or rng.random() < 0.2:
                    a = a_tmp
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation improvement utilities
# ----------------------------------------------------------------------


def _zigzag(arr: np.ndarray) -> np.ndarray:
    """
    Interleave a sorted array by placing the largest, then the second largest
    at the opposite end, then the third at the leftmost free slot, etc.
    """
    n = arr.shape[0]
    out = np.empty_like(arr)
    left, right = 0, n - 1
    for idx, val in enumerate(arr):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out


def _permutation_optimize(a: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Given a magnitude vector `a`, try different permutations (zig‑zag,
    a few random shuffles and hill‑climbing swaps) to lower the maximal
    self‑convolution entry.
    """
    best = a.copy()
    best_score = np.max(_full_convolution(best))

    # ---- 1. zig‑zag ordering -----------------------------------------
    sorted_vals = np.sort(best)[::-1]          # descending
    zig = _zigzag(sorted_vals)
    zig_score = np.max(_full_convolution(zig))
    if zig_score < best_score:
        best = zig
        best_score = zig_score

    # ---- 2. a handful of random shuffles ------------------------------
    for _ in range(5):
        perm = sorted_vals.copy()
        rng.shuffle(perm)
        perm_score = np.max(_full_convolution(perm))
        if perm_score < best_score:
            best = perm
            best_score = perm_score

    # ---- 3. hill‑climbing swaps ---------------------------------------
    n = best.shape[0]
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best = cand
            best_score = cand_score

    return best, best_score
