def _conv_max(a: np.ndarray):
    """
    Linear self‑convolution via FFT.
    Returns (full convolution array, maximal entry, index of the maximum).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray):
    """
    Sub‑gradient of the sum of the selected convolution entries.
    For each shift k we add 2·a_{k‑i} to the i‑th component whenever the
    index lies inside the vector.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g

def _build_conv_matrix(f: np.ndarray):
    """
    Construct the (2n‑1) × n matrix H where (H @ g)[k] = (f * g)[k].
    This routine is used only in the LP refinement and therefore builds
    a dense matrix – the dimensions are kept modest (n ≤ 3000).

    # each column i contributes the vector f aligned at rows i … i+n‑1
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,

    """
    Linear‑programming refinement (Frank‑Wolfe / cutting‑plane style).

    Given the current vector ``f`` we solve:
        maximise   Σ g_i

    The optimal ``g`` is then mixed with ``f`` (step size ``t``) and
    renormalised.  A few such iterations often lower the maximal
    convolution entry.

    The method is invoked only when SciPy is available and n is small
    enough (≤ 3000) to keep the LP tractable.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:          # avoid gigantic dense LPs
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg  ⇔ minimise –Σg
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})

            break

        if not res.success:

        else:
            g = g0 * (S / s0)

        # convex combination and renormalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f
