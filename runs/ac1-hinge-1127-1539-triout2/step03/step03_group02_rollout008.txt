def smooth_gradient_step_adam(a, target_sum, eta, tau_factor, m, v, t, beta1=0.9, beta2=0.999, eps=1e-8):
    conv = conv_self_fft(a)
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1: n - 1 + n]
    # Adam update
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, m, v, max_conv

"""
Search for a non‑negative step function that minimises the
maximum self‑convolution (equivalently the value returned by
`evaluate_sequence`).  

The algorithm consists of three stages:

1. **Broad parametric sweep** – generate many simple symmetric
   “U‑shapes’’ (linear, quadratic, exponential, power) and a
   piece‑wise edge‑heavy family.  Every candidate is projected onto the
   ℓ₁‑simplex {x ≥ 0, Σx = √(2 n)} and the best (smallest max‑convolution)
   is kept.

2. **Adam‑based smoothed‑max gradient descent** – minimise the
   log‑sum‑exp smoothing of the max‑convolution.
   The gradient is obtained by a single FFT‑based convolution.
   An Adam optimiser (first‑ and second‑moment adaptation) together
   with a decreasing learning‑rate / temperature schedule refines the
   candidate while each iterate is projected back onto the simplex.

3. **Polishing** – occasional random pair‑mass transfers and tiny
   jitter are applied; any improvement is stored.

The whole search respects a time‑budget (default 1000 s) and returns
the best sequence observed before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0.0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / v.size)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small amount of mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    param ≥ 0 adjusts the shape.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4            # small epsilon avoids exact zero
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)    # peak in the centre (opposite U)
    else:
        raise ValueError("unknown shape kind")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_piecewise_edges(n: int, target_sum: float,
                            edge_width: int, ratio: float) -> np.ndarray:
    """
    Simple two‑level shape:
        - first/last `edge_width` entries have value = `ratio`
        - remaining entries have value = 1
    Afterwards the vector is projected onto the simplex.
    """
    if edge_width * 2 >= n:
        edge_width = n // 2 - 1
    arr = np.ones(n, dtype=np.float64)
    if edge_width > 0:
        arr[:edge_width] = ratio
        arr[-edge_width:] = ratio
    return project_onto_simplex(arr, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximum entry of the self‑convolution of a."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Smoothed‑max gradient step with Adam momentum
# ----------------------------------------------------------------------


def adam_gradient_step(a: np.ndarray, target_sum: float,
                       eta: float, tau_factor: float,
                       m: np.ndarray, v: np.ndarray,
                       t_iter: int,
                       beta1: float = 0.9, beta2: float = 0.999,
                       eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray, float]:
    """
    One Adam iteration on the log‑sum‑exp smooth approximation of max.
    Returns (new_a, new_m, new_v, max_convolution_before_projection).
    """
    n = a.shape[0]

    # linear self‑convolution and its current maximum
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # smoothing weight (log‑sum‑exp)
    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w * a_rev)   – convolution in frequency domain
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

    # Adam moment updates
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t_iter)
    v_hat = v / (1.0 - beta2 ** t_iter)

    # parameter update
    a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)

    return a_new, m, v, max_conv


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine respects a time budget (~1000 s)
    and returns the best sequence observed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        init_seq = np.array(base, dtype=np.float64)
    else:
        # Fast power‑of‑two size; 8192 is a good compromise.
        n = 8192
        init_seq = None

    target_sum = math.sqrt(2.0 * n)          # Σa = √(2 n)

    # --------------------------------------------------------------
    # Phase 0 – broad parametric sweep
    # --------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)    # uniform start
    best_score = compute_max_conv(best_seq)

    if init_seq is not None:
        sc = compute_max_conv(init_seq)
        if sc < best_score:
            best_seq = init_seq.copy()
            best_score = sc

    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")
    shape_kinds = ("linear", "quadratic", "exponential", "power", "inverse")
    param_grids = {
        "linear": np.linspace(0.0, 20.0, 101),          # step 0.2
        "quadratic": np.linspace(0.0, 12.0, 61),       # step 0.2
        "exponential": np.linspace(0.0, 8.0, 41),     # step 0.2
        "power": np.linspace(0.5, 12.0, 24),           # exponent
        "inverse": np.linspace(0.5, 12.0, 24),
    }

    # simple U‑shapes
    for kind in shape_kinds:
        for p in param_grids[kind]:
            if time.time() >= deadline:
                print("Time exhausted during parametric sweep.")
                return best_seq.tolist()
            cand = generate_shape(n, target_sum, kind, float(p))
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {best_score:.6f}")

    # piece‑wise edge‑heavy family
    edge_widths = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    ratios = [1.5, 2, 3, 5, 8, 12, 20, 30]
    for w in edge_widths:
        for r in ratios:
            if time.time() >= deadline:
                print("Time exhausted during piece‑wise sweep.")
                return best_seq.tolist()
            cand = generate_piecewise_edges(n, target_sum, w, float(r))
            score = compute_max_conv(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"  New best piecewise (w={w}, r={r}) → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam‑based smoothed‑max gradient descent
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 1

    # hyper‑parameters (they will be decayed on the fly)
    eta = 0.6 * (target_sum / n)          # initial learning rate
    tau_factor = 0.07                     # τ = τ_factor * max_conv
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    no_improve = 0
    iteration = 0

    print(f"[{time.time():.2f}] Entering Adam‑based descent")
    while time.time() < deadline:
        a, m, v, cur_max = adam_gradient_step(
            a, target_sum, eta, tau_factor, m, v, t_iter, beta1, beta2, eps
        )
        iteration += 1
        t_iter += 1

        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iteration:07d} – improved to {best_score:.6f}")
        else:
            no_improve += 1

        # Adaptive schedules
        if iteration % 800 == 0:
            eta *= 0.94
            tau_factor = max(0.001, tau_factor * 0.92)

        # occasional stochastic mass transfer
        if np.random.rand() < 0.015:
            a = random_pair_transfer(a, target_sum, scale=0.009)

        # jitter after long stagnation
        if no_improve >= 5000:
            jitter = np.random.randn(n) * 0.0005
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {iteration:07d} – jitter applied")

    # --------------------------------------------------------------
    # Phase 2 – final polishing (random pair transfers)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Descent finished; best score {best_score:.6f}")

    polish_steps = 70
    for _ in range(polish_steps):
        if time.time() >= deadline:
            break
        candidate = random_pair_transfer(best_seq.copy(), target_sum,
                                         scale=0.004 * (1.0 - _ / polish_steps))
        cand_score = compute_max_conv(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            print(f"Polish improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence with score {best_score:.6f}")
    return best_seq.tolist()
