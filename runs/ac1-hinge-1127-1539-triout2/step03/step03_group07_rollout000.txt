def hill_climb(a_start, deadline, rng, max_iters=5000):
    a = a_start.copy()
    n = a.shape[0]
    cur_score = conv_max(a)
    best_a = a.copy()
    best_score = cur_score
    T = max(cur_score*0.1, 1e-4)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # Propose move: random donor i > 0, random recipient j != i.
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        # choose fraction
        frac = rng.uniform(0.1, 0.5)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalize not needed because sum unchanged.
        new_score = conv_max(a_new)
        if new_score < cur_score or rng.random() < np.exp(-(new_score-cur_score)/T):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        T *= 0.9995
        # occasional jitter
        if rng.random() < 0.001:
            # add small Gaussian noise and renorm
            noise = rng.normal(scale=a.mean()*0.01, size=n)
            a = np.clip(a + noise, 0, None)
            a *= np.sqrt(2.0*n)/a.sum()
            cur_score = conv_max(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score
```

"""
search for a non‑negative step function that minimises

    score = 2·n·max(a*a) / (sum(a)^2)

The code explores a variety of parametric families (exponential,
power‑law, sinusoidal, linear) and random Dirichlet vectors.
Promising candidates are refined by a simulated‑annealing hill‑climber
that moves a random fraction of mass between two positions.
All vectors are renormalised to Σa = sqrt(2·n) so that the external
score equals the maximum entry of the self‑convolution.  The routine
runs until the time budget (≈1000 s) expires and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (cached FFT length per n)
# ----------------------------------------------------------------------
_fft_len_cache = {}

def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two length for an n‑point linear convolution."""
    if n not in _fft_len_cache:
        conv_len = 2 * n - 1
        _fft_len_cache[n] = 1 << (conv_len - 1).bit_length()
    return _fft_len_cache[n]

def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using an FFT."""
    n = a.shape[0]
    fft_len = _fft_len_for_n(n)
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return float(conv.max())

def _normalize(a: np.ndarray) -> np.ndarray:
    """Rescale so that Σa = sqrt(2·len(a))."""
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, target / n, dtype=float)
    else:
        a = a * (target / s)
    return a

# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _exp_vec(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i) , i = 0..n‑1."""
    return np.exp(-lam * np.arange(n, dtype=float))

def _exp_rev_vec(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * (n‑1‑i))."""
    return np.exp(-lam * np.arange(n - 1, -1, -1, dtype=float))

def _pow_vec(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    return (np.arange(1, n + 1, dtype=float) ** (-p))

def _pow_rev_vec(n: int, p: float) -> np.ndarray:
    """a_i = (n-i)^{-p}."""
    return (np.arange(n, 0, -1, dtype=float) ** (-p))

def _sin_vec(n: int) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n)  (positive sinusoid)."""
    i = np.arange(n, dtype=float)
    return np.sin(math.pi * (i + 0.5) / n)

def _cos_vec(n: int) -> np.ndarray:
    """Raised cosine: a_i = 0.5·(1‑cos(2π·i/(n‑1))) ."""
    if n == 1:
        return np.ones(1, dtype=float)
    i = np.arange(n, dtype=float)
    return 0.5 * (1.0 - np.cos(2.0 * math.pi * i / (n - 1)))

def _linear_vec(n: int) -> np.ndarray:
    """Linearly decreasing: a_i = n-i."""
    return np.arange(n, 0, -1, dtype=float)

def _linear_rev_vec(n: int) -> np.ndarray:
    """Linearly increasing: a_i = i+1."""
    return np.arange(1, n + 1, dtype=float)

# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climber
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that moves a random fraction of mass
    from one index to another.  Accepts worsening moves with a temperature
    that decays geometrically.  Returns the best vector found and its
    max‑convolution.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_score = conv_max(a)
    best_a = a.copy()
    best_score = cur_score

    # initialise temperature as a modest fraction of the current peak
    temperature = max(0.02 * cur_score, 1e-5)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # pick donor and recipient
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # transfer a random fraction (5‑30 %) of donor's mass
        frac = rng.uniform(0.05, 0.30)
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = conv_max(a_new)

        # Metropolis accept
        accept = False
        if new_score < cur_score - 1e-12:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[hill improve] t={time.time():.1f}s n={n} iter={it} score={new_score:.6f}")

        # geometric cooling
        temperature *= 0.9993

        # occasional jitter to escape plateaus
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _normalize(a)
            cur_score = conv_max(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[jitter] t={time.time():.1f}s n={n} iter={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget, mixes random starts, parametric
    families, and a hill‑climber.  Returns a Python list of heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")
    run_id = 0

    # ------------------------------------------------------------------
    # If a previous construction is present, record it as a baseline.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                cand = _normalize(cand)
                sc = conv_max(cand)
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score={sc:.6f} len={cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 8000]   # candidate lengths
    family_funcs = [
        ("exp", _exp_vec),
        ("exp_rev", _exp_rev_vec),
        ("pow", _pow_vec),
        ("pow_rev", _pow_rev_vec),
        ("sin", _sin_vec),
        ("cos", _cos_vec),
        ("lin", _linear_vec),
        ("lin_rev", _linear_rev_vec),
    ]

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = int(rng.choice(lengths,
                           p=np.array([0.15, 0.20, 0.25, 0.25, 0.15])))
        # decide between a parametric family or a Dirichlet start
        use_param = rng.random() < 0.65

        if use_param:
            # pick a family
            fam_name, fam_fun = family_funcs[rng.integers(len(family_funcs))]
            # draw its parameter(s)
            if fam_name in ("exp", "exp_rev"):
                lam = 10 ** rng.uniform(-4.0, 0.0)          # lam in [1e‑4, 1]
                a = fam_fun(n, lam)
            elif fam_name in ("pow", "pow_rev"):
                p = rng.uniform(0.1, 3.0)
                a = fam_fun(n, p)
            else:
                a = fam_fun(n)
        else:
            # Dirichlet start – concentration controls sparsity
            conc = rng.uniform(0.05, 2.0)
            a = rng.dirichlet(np.full(n, conc))

        a = _normalize(a)
        cur_score = conv_max(a)

        # record raw candidate
        if cur_score < best_score - 1e-12:
            best_seq = a.copy()
            best_score = cur_score
            print(f"[raw improve] run={run_id} n={n} type={fam_name if use_param else 'dir'} "
                  f"score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # Local refinement on promising vectors
        # ------------------------------------------------------------------
        refine = False
        # always give a small chance to refine, but favour low scores
        if cur_score < best_score * 0.98:
            refine = True
        elif rng.random() < 0.03:
            refine = True

        if refine:
            refined_a, refined_score = hill_climb(a, deadline, rng,
                                                 max_iters=2500)
            if refined_score < best_score - 1e-12:
                best_seq = refined_a.copy()
                best_score = refined_score
                print(f"[refine improve] run={run_id} n={n} after_hc score={refined_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Final intensive polishing of the best found sequence (time permitting)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        best_seq, best_score = hill_climb(best_seq, deadline, rng,
                                          max_iters=5000)
        print(f"[final hill climb] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)
        best_score = conv_max(best_seq)
        print("[fallback] using uniform default")

    # ------------------------------------------------------------------
    # Final verification (the external evaluator will also be called)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f} length = {len(best_seq)}")
    return best_seq.tolist()
