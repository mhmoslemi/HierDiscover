def _adam_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    beta = 0.5
    max_beta = 4096.0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        t_iter += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)

        # step size schedule: base step size targeted to maintain sum
        step = (target_sum * 0.1) * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # beta annealing
        if t_iter % 200 == 0:
            beta = min(beta * 1.5, max_beta)

        # possibly add jitter
        if t_iter % 1000 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
    return a

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The algorithm blends several ideas:

* a diverse family of random initialisers (power‑law, exponential,
  Gaussian, arcsine‑like, Dirichlet, uniform + jitter);
* a projected Adam optimiser on a smooth log‑sum‑exp surrogate of the
  max convolution; the optimiser adapts step‑sizes and anneals the
  smoothing parameter β;
* a long simulated‑annealing (pairwise mass‑transfer) refinement that
  accepts occasional uphill moves.

The routine respects the 1000 s budget, explores a few different
vector lengths (more entries give more flexibility) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation: max(conv(a,a)).
    The caller must ensure that ``a`` already lies on the simplex
    Σa = sqrt(2·n) and a ≥ 0.
    """
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx_i = s,  x_i ≥ 0
# (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) of the convolution and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where

        conv = a * a   (full linear convolution, length 2·n‑1)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·conv_k) )
              = 2·(a ⋆ p)   with p_k ∝ exp(β·conv_k)

    The implementation uses FFTs and works in O(n log n).
    """
    n = a.shape[0]

    # ---- convolution a*a ----------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities ---------------------------------------
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))      # stability trick
    p = exps / np.sum(exps)

    # ---- gradient = 2·(a ⋆ p) ----------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    tmp = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * tmp[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed max
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam on the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best vector seen before ``deadline``.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # β for the smooth approximation (annealed)
    beta = 0.5
    max_beta = 2048.0

    # bookkeeping
    it = 0
    best = a.copy()
    best_score = _score(best)

    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of smoothed max
        g, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size – scaled with target_sum to keep numbers moderate
        step = (target_sum * 0.02) * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # keep feasible (non‑negative & simplex)
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional evaluation & record best
        if it % 200 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                print(f"[{time.time():.1f}] Adam improved: {best_score:.6f}")

        # anneal the smoothing parameter
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        # jitter to escape plateaus
        if it % 1200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (pairwise mass transfers)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly moves a random fraction of mass from one
    index to another.  It keeps the best sequence seen and returns it.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _score(best)

    cur_score = best_score

    # initialise temperature from current score
    T = best_score * 0.02 + 1e-6
    cooling = 0.9997

    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick two distinct indices with non‑zero source mass
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a random fraction (up to 20 %) of the source mass
        frac = rng.random() * 0.20
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score

        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert the move
            a[j] -= delta
            a[i] += delta

        # geometric cooling
        T *= cooling

        # occasional global jitter to escape deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a non‑negative vector on the ℓ¹‑simplex (sum = sqrt(2·n)).
    Several families are mixed to increase diversity:
        * pure power‑law,
        * exponential,
        * mixed power‑law & exponential,
        * Gaussian bump,
        * arcsine‑like (high at the ends),
        * Dirichlet (uniform simplex),
        * uniform + small jitter.
    """
    target = np.sqrt(2.0 * n)
    choice = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if choice == 0:                                   # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                                 # exponential decay
        lam = rng.exponential(scale=0.008) + 0.001
        a = np.exp(-lam * idx)

    elif choice == 2:                                 # mixed power‑law + exp
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.008) + 0.001
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                                 # Gaussian bump
        centre = rng.uniform(0.3, 0.7) * n
        sigma = rng.uniform(0.1, 0.35) * n
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)

    elif choice == 4:                                 # arcsine‑like
        # a_i ∝ 1 / sqrt((i+0.5)*(n-i-0.5))
        a = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))

    elif choice == 5:                                 # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    else:                                             # uniform + jitter
        a = np.full(n, target / n)

    # add a touch of Gaussian noise and re‑project onto the simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, explores several vector
    lengths, uses random restarts, a projected‑Adam optimiser and a
    simulated‑annealing refinement, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # safety margin before the hard deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previous construction is present, seed from it
    # ------------------------------------------------------------------
    init_from_global = globals().get("height_sequence_1")
    if isinstance(init_from_global, (list, tuple, np.ndarray)):
        a = np.asarray(init_from_global, dtype=float)
        n0 = a.size
        target0 = np.sqrt(2.0 * n0)
        a = _project_to_simplex(a, target0)
        sc = _score(a)
        best_seq = a.copy()
        best_score = sc
        print(f"[{time.time():.1f}] Starting from global height_sequence_1 "
              f"(n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to investigate – larger n gives more flexibility
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 5000, 8000, 12000]

    # keep a small pool of promising candidates for deeper work
    top_candidates = []  # list of (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Phase A – quick random sampling
        # --------------------------------------------------------------
        phase_end = min(deadline, time.time() + 10.0)   # 10 s per length
        while time.time() < phase_end:
            a0 = _random_initial_sequence(n, rng)
            sc = _score(a0)

            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} "
                      f"score={best_score:.6f}")

            # keep a few best random seeds
            if len(top_candidates) < 7 or sc < top_candidates[-1][0]:
                top_candidates.append((sc, a0.copy()))
                top_candidates.sort(key=lambda x: x[0])
                top_candidates = top_candidates[:7]

        # --------------------------------------------------------------
        # Phase B – Adam optimisation on the elite seeds
        # --------------------------------------------------------------
        for _, seed_vec in top_candidates:
            if time.time() >= deadline:
                break

            # allocate at most 25 s for each refinement (Adam + SA)
            sub_dead = min(deadline, time.time() + 25.0)

            # Adam optimisation (projected) ---------------------------------
            a_adam = _adam_opt(seed_vec, sub_dead)

            # Simulated annealing uses whatever time remains
            a_sa = _simulated_annealing(a_adam, deadline)

            sc = _score(a_sa)
            if sc < best_score:
                best_score = sc
                best_seq = a_sa.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} "
                      f"score={best_score:.6f}")

        # clear the pool before the next length
        top_candidates.clear()

    # ------------------------------------------------------------------
    # Fallback to a simple uniform sequence (should never be taken)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_seq = a
        best_score = _score(a)

    # Trim trailing near‑zero entries for a tidy output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
