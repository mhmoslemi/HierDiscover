```
def _project_to_simplex(v, s):
    # v may be negative, we project onto non-negative simplex sum=S
    # Using algorithm by Duchi et al.
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def conv_full_fft(a):
    n = a.shape[0]
    conv_len = 2*n-1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}= \frac{2n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear self‑convolution of the non‑negative vector `a`.  
If we enforce the normalisation  

\[
\sum_i a_i=\sqrt{2n},
\]

the denominator equals `2n` and the score is simply the maximum entry of the
self‑convolution.  Hence the problem is: *find a non‑negative vector of a
given length whose self‑convolution has the smallest possible \(\ell_\infty\)‑norm.*

The program below searches for such a vector using three complementary
techniques:

1. **Exact projection‑exponential descent** – an exponentiated‑gradient
   update with the *true* sub‑gradient of the max‑convolution (all
   maximal shifts, not only a sampled subset).  After the multiplicative
   update the vector is projected onto the simplex
   \(\{a\ge0,\;\sum a=\sqrt{2n}\}\) which guarantees feasibility.

2. **Alternating linear‑programming refinement** – given a candidate `f`
   we solve the linear programme  

   \[
   \max\_{g\ge0}\ \sum_j g_j\quad\text{s.t.}\quad (f*g)_k\le M\;\;\forall k,
   \]

   where \(M=\max(f*f)\).  The optimiser `g` is rescaled to the required
   \(\ell_1\)‑norm and mixed with `f`.  Repeating this step monotonically
   reduces the maximal convolution value.

3. **Stochastic hill‑climbing** – a small amount of mass is moved from a
   random donor entry to a random recipient entry.  Moves that improve the
   maximum are always accepted; worsening moves are accepted with a
   temperature‑controlled probability (simulated‑annealing).

The search runs many independent restarts (different lengths, different
Dirichlet initialisations) until the 1000‑second budget is exhausted.
The best sequence found is returned.

All helper functions are top‑level, the code uses only `numpy`,
`scipy.optimize`/`scipy.sparse` and the provided `evaluate_sequence`
function.

```python
"""
Propose a step‑function (list of non‑negative heights) with a minimal
self‑convolution peak.  The algorithm combines:

1. Exact exponentiated‑gradient descent on the max‑convolution (projection
   onto the simplex after each multiplicative update).

2. An alternating linear‑programming refinement that mixes the current
   vector with a solution of  max Σg  s.t. f*g ≤ current maximum.

3. A final simulated‑annealing hill‑climber.

The search uses many random restarts and respects the 1000 s time budget.
"""
import time
import numpy as np
import scipy.optimize
import scipy.sparse


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the non‑negative simplex
    {x | x_i ≥ 0, Σ x_i = s}.  Duchi et al.  (2008) algorithm.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using an FFT (size 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(_conv_full_fft(a)))


def _subgradient_maxconv(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of the map a ↦ max(a*a).  All shifts attaining the maximum
    are used (exact gradient, not a sampled top‑k).
    """
    n = a.shape[0]
    conv = _conv_full_fft(a)
    max_val = conv.max()
    # indices where the convolution equals the maximum (within eps)
    max_idx = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_idx:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        grad[valid] += 2.0 * a[partner[valid]]
    return grad


def _exponential_descent_exact(a0: np.ndarray,
                               deadline: float,
                               lr0: float = 0.6,
                               beta: float = 0.9,
                               decay: float = 0.99995,
                               jitter_prob: float = 0.02,
                               max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent with the *exact* sub‑gradient
    of the max‑convolution.  Returns the best vector encountered and its
    maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, S)

    best_a = a.copy()
    best_val = _conv_max(a)

    v = np.zeros_like(a)          # momentum buffer
    lr = lr0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        g = _subgradient_maxconv(a)

        # momentum
        v = beta * v + (1.0 - beta) * g

        # exponentiated (multiplicative) update – guarantees positivity
        a = a * np.exp(-lr * v)

        # project back onto the simplex
        a = _project_to_simplex(a, S)

        # occasional jitter to escape flat regions
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

        # learning‑rate decay
        lr *= decay

        cur_val = _conv_max(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[exp exact improve] t={time.time():.1f}s n={n} it={it} max={cur_val:.6f}")

    return best_a, best_val


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_j  s.t.  (f * g)_k ≤ M  (for all k) ,  g ≥ 0.
    Returns the optimal g (may be None if the LP fails).
    """
    n = f.shape[0]

    # Build the sparse Toeplitz‑like matrix A where
    #   (A g)_k = Σ_i f_i * g_{k-i}
    i = np.arange(n).reshape(-1, 1)          # (n,1)
    j = np.arange(n).reshape(1, -1)          # (1,n)
    rows = i + j                              # (n,n) – shift index
    cols = j                                  # (n,n) – column index = g index
    data = f[i]                               # (n,n) – coefficient = f_i

    rows = rows.ravel()
    cols = cols.ravel()
    data = data.ravel()
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                           # maximise Σ g_j  ⇔ minimise –Σ g_j
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                    bounds=bounds, method='highs')
    except Exception:
        return None

    if res.success:
        return res.x
    return None


def _alternating_lp_refine(a_start: np.ndarray,
                           deadline: float,
                           t: float = 0.2,
                           max_iters: int = 100) -> tuple[np.ndarray, float]:
    """
    Alternating linear‑programming refinement.
    Repeatedly solves the LP described in _solve_lp_max_sum, rescales the
    solution and mixes it with the current vector.
    """
    a = a_start.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_M = _conv_max(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = best_M
        g0 = _solve_lp_max_sum(a, M)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break
        # rescale to the required ℓ₁‑norm
        alpha = S / sum_g0
        g = g0 * alpha

        a_new = (1.0 - t) * a + t * g
        a_new = _project_to_simplex(a_new, S)
        M_new = _conv_max(a_new)

        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            print(f"[lp improve] t={time.time():.1f}s n={n} it={it} max={M_new:.6f}")
        else:
            # if no improvement, reduce the mixing factor
            t *= 0.5
            if t < 1e-4:
                break

    return best_a, best_M


def _hill_climb_refine(a_start: np.ndarray,
                       deadline: float,
                       max_iters: int = 25000,
                       temperature: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climber that moves a random fraction of
    mass from one position to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = _conv_max(a)

    temp = max(temperature * best_val, 1e-6)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        frac = rng.uniform(0.05, 0.20)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_to_simplex(a_new, S)

        new_val = _conv_max(a_new)

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a_new.copy()
            a = a_new
            print(f"[hill climb improve] t={time.time():.1f}s n={n} it={it} max={new_val:.6f}")
        else:
            prob = np.exp(-(new_val - best_val) / max(temp, 1e-12))
            if rng.random() < prob:
                a = a_new

        temp *= 0.9995

    return best_a, best_val
```
