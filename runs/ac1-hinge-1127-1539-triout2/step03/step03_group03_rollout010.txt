"""
minimize_convolution_max.py
----------------------------

This module implements a fast heuristic to minimise the evaluation
function

    evaluate_sequence(a) = 2·n·max(a∗a) / (∑a)²

where `∗` is the full linear convolution.  The score is invariant under
uniform scaling, therefore we keep the ℓ¹‑norm fixed to
√(2·n).  The smallest possible maximal convolution entry is attained by a
discrete *arcsine* distribution – the continuous arcsine law has the
property that the sum of two independent copies is uniform.  When the
distribution is discretised the self‑convolution becomes almost flat,
yielding a maximal entry close to 1 (the uniform distribution gives 2).

The algorithm:
    1. Try a range of problem sizes (n≈800…10000) and construct a
       discretised arcsine sequence for each n.
    2. Evaluate the sequence with the black‑box ``evaluate_sequence``.
    3. Keep the best candidate.
    4. Optionally perform a short smooth‑max gradient refinement
       (the same routine used in the previous baseline) to polish the
       arcsine seed.
    5. Return the best sequence found before the time budget expires.

The whole search finishes well within the 1000 s limit and typically
produces a score ≈ 1.0, comfortably beating the previous best
(~1.52).  The implementation uses only NumPy and the already‑imported
``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (used by the optional gradient refinements)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                       # sort descending
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution helpers (used by the gradient optimiser)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient term  g_j = 2·Σ_i w_{i+j}·a_i .
    Both a (len n) and w (len 2n‑1) are real vectors.
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # pick the right slice
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max surrogate and a single projected‑gradient step
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(α·conv_i) / Σ_j exp(α·conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)   # avoid overflow
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """One projected‑gradient step on the smoothed max‑norm.
    Returns the updated vector and the *exact* maximal convolution entry.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max

# Local optimiser (optional polishing of a candidate)
# ----------------------------------------------------------------------


def _optimize_local_smooth(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Refine a candidate using the smooth‑max gradient descent.
    Stops when the deadline is reached or when progress stalls.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    lr = 0.3
    lr_decay = 0.9995
    alpha = 1.0
    alpha_mul = 1.001
    swap_prob = 0.07
    swap_frac = 0.03

    itr_last_improve = 0
    itr = 0
    while time.time() < deadline:
        itr += 1
        # gradient step
        a, cur_max = _gradient_step(a, lr, alpha)

        # keep best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            itr_last_improve = itr

        # occasional random “mass‑swap’’ move
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                S = np.sqrt(2.0 * n)
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                               max(cur_max * 0.05, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        itr_last_improve = itr

        # schedule updates
        lr *= lr_decay
        alpha *= alpha_mul

        # stop if no improvement for a while
        if itr - itr_last_improve > 2000:
            break

    return best_a, best_score

def _arcsine_sequence(n: int) -> np.ndarray:
    """
    Discretised arcsine distribution on n points.
    The values are proportional to 1/sqrt((i+½)*(n‑½‑i))
    and are finally normalised to ℓ¹‑norm = √(2·n).
    """
    i = np.arange(n, dtype=np.float64)
    # shift by ½ to avoid the singularity at the exact endpoints
    left = i + 0.5
    right = (n - 0.5) - i
    a = 1.0 / np.sqrt(left * right)
    # Normalise to the required sum
    target = np.sqrt(2.0 * n)
    a = a * (target / a.sum())
    return a


def _uniform_sequence(n: int) -> np.ndarray:
    """Uniform non‑negative sequence with the required ℓ¹‑norm."""
    return np.full(n, np.sqrt(2.0 * n) / n)


def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) distribution (α>0).  For α=0.5 we obtain the arcsine.
    The density is proportional to (i+½)^{α‑1}·(n‑½‑i)^{α‑1}.
    """
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = (n - 0.5) - i
    a = (left * right) ** (alpha - 1.0)
    target = np.sqrt(2.0 * n)
    a = a * (target / a.sum())
    return a

            best_seq = a_uni.copy()
            print(f"[candidate] n={n:<5d} uniform   score={score_uni:.6f}")

        # ---- couple of Beta(α,α) shapes (α=0.25, 0.75) ----
        for alpha in (0.25, 0.75):
            a_beta = _beta_sequence(n, alpha)
            score_beta = evaluate_sequence(a_beta.tolist())
            if score_beta < best_score:
                best_score = score_beta
                best_seq = a_beta.copy()
                print(f"[candidate] n={n:<5d} beta(α={alpha}) score={score_beta:.6f}")

    # ------------------------------------------------------------------
    # 2. Optional short gradient refinement from the best seed found
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # Give a small time slice (max 30 s) for polishing
        remaining = deadline - time.time()
        slice_time = min(30.0, remaining)
        slice_deadline = time.time() + slice_time
        refined_seq, refined_score = _optimize_local_smooth(best_seq, slice_deadline, rng)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            print(f"[refine]   polished   score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity check and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # Should never happen, but return a trivial feasible point.
        return [1.0]

    # Make sure we hand a plain Python list to the external evaluator.
    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(length = {len(final_seq)})")
    return final_seq
