def _adam_optimize(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta = 1.0
    max_beta = 2048.0
    step = target_sum * 0.1   # base step size
    t = 1
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute gradient
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size adaptation: reduce with beta
        lr = step / (np.sqrt(v_hat) + eps) * (1.0/(beta+1.0))
        # vector update
        a = a - lr
        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum*0.001/(beta+1.0), size=n)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)

        # evaluate
        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam beta={beta:.2f} val={best_val:.6f}")

        # schedule beta increase

            beta = min(beta*1.5, max_beta)

def _hill_climb(a0, deadline):
    a = a0.copy()
    n = a.size
    cur_val = _score(a)
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    # parameters
    max_iter = 1000000
    start_temp = 0.01 * cur_val  # maybe
    end_temp = 1e-6
    # linear cooling based on time
    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # compute temperature
        t_frac = elapsed / (deadline - start_time)
        temp = start_temp * (1 - t_frac) + end_temp * t_frac

        # propose move
        donor = rng.integers(n)
        if a[donor] < 1e-12:
            continue
        # choose receiver weighted by small a? maybe uniformly
        receiver = rng.integers(n)
        if donor == receiver:
            continue
        # transfer fraction
        frac = rng.random() * 0.05 # up to 5%
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta
        # evaluate
        new_val = _score(a)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            if temp > 0:
                prob = np.exp(-(new_val - cur_val) / temp)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] Hill climb improved to {best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta
    return best_a, best_val
```

"""
search_optimizer.py

A multi‑restart, momentum‑based optimizer for the “max‑convolution / ℓ¹‑norm’’
benchmark.  The algorithm works as follows:

1.  A set of candidate lengths (1500 … 20000) is prepared.
2.  For each length a few diverse initial sequences are generated
    (power‑law, exponential, Gaussian, sparse random, uniform and the
    previously‑found best sequence if it exists).
3.  Each candidate is refined with an Adam‑style projected gradient descent
    on a soft‑max approximation of the convolution maximum.  The temperature
    β is annealed from 1 to a large value (≈2048) while the learning rate
    adapts automatically.
4.  The result is then polished by a fast “guided hill‑climb’’ that moves
    mass from positions that contribute most to the current peak of the
    convolution to positions that contribute the least.
5.  The best sequence encountered before the time budget expires is returned.
    All vectors are projected onto the simplex {x≥0, Σx = √(2·n)} after every
    update, which guarantees scale‑invariance of the objective.

The code runs entirely in‑memory, uses only NumPy and the standard library,
and respects the 1000‑second time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Fast exact evaluation of ``evaluate_sequence`` for a vector that already
    satisfies the normalisation Σa = √(2·n).  The returned value is therefore
    equal to the maximal convolution entry.
    """
    return _max_convolution_fft(a)


def _simplex_fix(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Very fast “project onto the simplex’’: clip to non‑negativity and
    renormalise to the desired ℓ¹‑norm.
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        # uniform fallback – should never happen
        a.fill(target_sum / a.size)
    else:
        a *= target_sum / s
    return a


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(conv)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where ``b`` is the full convolution a*a and ``g`` is
    ∇_{a} log‑sum‑exp_{β}(b).  The gradient formula is
    g = 2·(a_rev ⋆ p) with p the soft‑max probabilities over ``b``.
    """
    n = a.shape[0]

    # full convolution a*a (via FFT)
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # shape (2·n‑1,)

    # gradient: 2·(a_rev ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]

    return g, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smooth objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Adam optimiser with a temperature schedule.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    beta = 1.0               # soft‑max temperature
    max_beta = 2048.0
    base_lr = target_sum * 0.12  # raw learning‑rate scale
    rng = np.random.default_rng()

    best_a = a.copy()
    best_val = _score(a)

    it = 1
    while time.time() < deadline:
        # gradient of the smooth surrogate
        g, _ = _gradient_softmax(a, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate that shrinks with the temperature
        lr = base_lr / (beta + 1.0)

        # parameter update and projection onto the simplex
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _simplex_fix(a, target_sum)

        # occasional random jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = _simplex_fix(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam  β={beta:.1f}  score={best_val:.6f}")

        # anneal temperature
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.6, max_beta)

        it += 1
        if time.time() >= deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (mass‑transfer based refinement)
# ----------------------------------------------------------------------


def _guided_hill_climb(a0: np.ndarray, deadline: float):
    """
    Repeatedly moves a small fraction of mass from positions that strongly
    contribute to the current maximal convolution entry to positions that
    contribute the least.  Only improving moves are kept.
    """
    a = a0.copy()
    n = a.size
    target_sum = a.sum()
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07   # at most 7 % of a donor’s mass is moved per step

    while time.time() < deadline:
        # compute convolution (FFT – fast for all n we consider)
        size1 = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, n=size1)
        b = np.fft.irfft(fa * fa, n=size1)[: 2 * n - 1]

        k_max = int(np.argmax(b))

        # sensitivity vector: contribution of a[i] to the peak at k_max
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        sens[lo : hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        # weight = a[i] * sensitivity – total contribution of index i
        weight = sens * a
        if weight.max() == 0.0:
            break

        donor = int(np.argmax(weight))

        # receiver among the 10 % lowest weighted positions (excluding donor)
        low_thr = np.percentile(weight, 10)
        candidates = np.where((weight <= low_thr) & (np.arange(n) != donor))[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))

        # transfer a random fraction of donor’s mass
        frac = rng.random() * max_frac
        delta = a[donor] * frac
        if delta <= 0.0:
            continue
        a[donor] -= delta
        a[receiver] += delta

        # re‑project (keeps ℓ¹‑norm exact)
        a = _simplex_fix(a, target_sum)

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] HillClimb improved to {best_val:.6f}")
        else:
            # revert the move
            a[receiver] -= delta
            a[donor] += delta
            a = _simplex_fix(a, target_sum)

    return best_a, best_val

def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.35, 0.75)
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a = a * (np.sqrt(2.0 * n) / a.sum())
    return a


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    decay = rng.uniform(0.0005, 0.01)
    a = np.exp(-decay * np.arange(n, dtype=np.float64))
    a = a * (np.sqrt(2.0 * n) / a.sum())
    return a


def _init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = n / 2.0 + rng.normal(scale=n * 0.1)
    sigma = n / 5.0
    a = np.exp(-0.5 * ((np.arange(n, dtype=np.float64) - mu) / sigma) ** 2)
    a = a * (np.sqrt(2.0 * n) / a.sum())
    return a


def _init_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    # 0.5 % – 2 % non‑zero entries
    frac = rng.uniform(0.005, 0.02)
    k = max(2, int(frac * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a = a * (np.sqrt(2.0 * n) / a.sum())
    return a


def _init_uniform(n: int) -> np.ndarray:
    a = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine performs many short random restarts, each refined by an
    Adam‑projected‑gradient descent followed by a guided hill‑climb.
    The best sequence discovered before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Candidate lengths (larger lengths usually give a lower maximal peak)
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 5000, 8000, 12000, 16000, 20000]

    # collection of initialisers
    init_fns = [_init_powerlaw, _init_exponential,
                _init_gaussian, _init_sparse, _init_uniform]

    # ------------------------------------------------------------------
    # Main search loop – keep generating candidates until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = int(rng.choice(lengths))
        target_sum = np.sqrt(2.0 * n)

        # pick a random initialiser
        init_fn = rng.choice(init_fns)
        a0 = init_fn(n, rng) if init_fn.__code__.co_argcount == 2 else init_fn(n)

        # occasionally inject the previously found best sequence (if any)
        if best_seq is not None and rng.random() < 0.15:
            old = np.array(best_seq, dtype=np.float64)
            if len(old) >= n:
                a0 = old[:n].copy()
            else:
                a0[: len(old)] = old
                # fill the rest with a tiny uniform background
                a0[len(old):] = target_sum * 1e-4
            a0 = _simplex_fix(a0, target_sum)

        # ---------- Adam optimisation ----------
        a_opt, val = _adam_optimize(a0, target_sum, deadline)

        # ---------- Guided hill‑climb ----------
        if time.time() < deadline:
            a_opt, val = _guided_hill_climb(a_opt, deadline)

        # keep the overall best
        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] ** GLOBAL BEST **  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = _init_uniform(n)

    # Trim any trailing zeros that may have appeared due to rounding
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"Search finished – best score ≈ {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
