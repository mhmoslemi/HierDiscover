def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    step = target * 0.05
    best_a = a.copy()
    best_score = _score(a)
    t=0
    while time.time() < deadline:
        # compute conv
        conv = _conv_full_fft(a) # compute full conv with similar method to gradient
        max_val = conv.max()
        # find indices of max
        ks = np.where(conv == max_val)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for k in ks:
            start = max(0, k - (n-1))
            end = min(n-1, k)
            # for i in start..end: grad[i] += 2*a[k-i]
            # we can vectorize:
            i_idx = np.arange(start, end+1)
            grad[i_idx] += 2*a[k - i_idx]
        # average over ks
        grad = grad / len(ks)
        # step size decaying
        step = target * 0.2 / (t+1)**0.5
        a -= step * grad
        a = _project_to_simplex(a, target)
        t+=1
        if t % 100 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] subgrad iter {t} best={sc:.6f}")

"""
Improved search for a non‑negative step‑function that minimises
evaluate_sequence(...).

The algorithm works in three stages:

1. **Broad random sampling** – for several lengths (1500 … 6000) a few
   candidates are drawn from a rich collection of families (power‑law,
   exponential, mixed, Dirichlet, sparse binary).  The best candidate
   overall is kept.

2. **Long smoothed‑gradient optimisation** – starting from the best
   candidate a projected Adam‑style optimiser minimises a
   log‑sum‑exp approximation of the maximum convolution entry.
   The softmax temperature β is gradually increased so that the
   optimisation approaches the true ℓ∞‑objective.

3. **Exact sub‑gradient polishing** – after the soft optimisation the
   true max‑convolution is used, its (sub‑)gradient is taken and a few
   more Adam steps are performed.  The best sequence seen during the
   whole run is returned (trimmed of trailing zeros).

All updates stay on the ℓ¹‑simplex {x_i≥0, Σx_i=√(2·n)} so the
objective returned by `evaluate_sequence` equals the maximum entry of
the self‑convolution of the normalised vector.  The routine respects
the 1000 s budget and prints intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution a*a using an FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Objective used internally: after ℓ¹‑normalisation
    Σa = √(2·n) the value equals max(conv(a,a)).
    """
    max_b = _max_convolution_fft(a)
    return max_b


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    For a vector a and a temperature beta return
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
    where b = a*a (full convolution).  The gradient can be written as
        g = 2·(a ⋆ p)
    with p_k ∝ exp(β·b_k) (soft‑max over the convolution entries).

    Returns a tuple (g, b) where b is the full convolution.
    """
    n = a.shape[0]

    # ----- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities p ------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))               # stable
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random non‑negative sequence on the ℓ¹‑simplex."""
    target_sum = np.sqrt(2.0 * n)
    choice = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if choice == 0:                     # pure power‑law
        p = rng.uniform(0.15, 1.0)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif choice == 2:                   # mixed power‑law + exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                   # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target_sum

    elif choice == 4:                   # sparse binary pattern
        # size roughly √(2n) to keep the ℓ¹‑norm reasonable
        m = max(2, int(np.sqrt(2.0 * n)))
        a = np.zeros(n, dtype=float)
        sel = rng.choice(n, size=m, replace=False)
        a[sel] = 1.0
        a = a * target_sum / a.sum()

    else:                               # uniform plus tiny jitter
        a = np.full(n, target_sum / n)

    # add a little Gaussian noise and re‑project onto the simplex
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed maximum
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    Projected Adam optimiser on the log‑sum‑exp smoothed max‑convolution.
    The temperature beta is annealed from 1 up to a large value.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    beta = 1.0          # softmax temperature
    max_beta = 4096.0
    best = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        t += 1
        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Step size – the scale `target` keeps updates roughly in the
        # feasible region irrespective of n.
        step = (target * 0.15) * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_to_simplex(a, target)

        # Anneal beta every few hundred iterations
        if t % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.3, max_beta)

        # Occasionally inject a tiny random jitter to escape plateaus
        if t % 1200 == 0:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # Keep track of the best (real) objective seen so far
        if t % 250 == 0:
            cur = _score(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()
                print(f"[{time.time():.1f}] Adam iter {t} new best {best_score:.6f}")

        # Early stop if we already beat the target
        if best_score <= 1.503:
            break

    return best


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray,
                          deadline: float) -> np.ndarray:
    """
    Uses the true max‑convolution (no smoothing) to compute a
    sub‑gradient and applies a few Adam steps.  This stage fine‑tunes
    the solution obtained by the smoothed optimiser.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # Adam parameters (a second short Adam run)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    best = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        # Full convolution (FFT) – reuse same routine as above
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_val = conv.max()
        max_idx = np.where(conv == max_val)[0]   # could be several

        # Sub‑gradient: average of the gradients belonging to all maximal indices
        grad = np.zeros_like(a)
        for k in max_idx:
            # i ranges where 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            grad[i_range] += 2.0 * a[k - i_range]
        grad /= max_idx.size

        t += 1
        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = (target * 0.07) * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target)

        if t % 150 == 0:
            cur = _score(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad iter {t} new best {best_score:.6f}")

        if best_score <= 1.503:
            break

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    competition objective.  The routine respects the supplied time budget
    (default 1000 s) and returns a plain Python list of numbers.
    """
    rng = np.random.default_rng(seed)

    # Small safety margin before the hard deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Broad random sampling over several lengths
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 4500, 6000]
    random_per_len = 8   # number of random starts per length

    for n in lengths:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)

        for _ in range(random_per_len):
            if time.time() >= deadline:
                break
            a0 = _random_initial_sequence(n, rng)
            sc = _score(a0)
            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()
                print(f"[{time.time():.1f}] Random n={n} score={sc:.6f} (global best)")

            # Very early exit if we already reached the target bound
            if best_score <= 1.503:
                break

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2) Deep optimisation of the current global best
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of modest length
        n = 2000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Allocate the remaining budget to the optimiser
    opt_deadline = deadline
    # First the smoothed Adam optimisation
    a_opt = _adam_optimize(best_seq, opt_deadline, rng)

    # Second the exact sub‑gradient polishing (use whatever time is left)
    if time.time() < deadline:
        a_opt = _subgradient_optimize(a_opt, deadline)

    final_score = _score(a_opt)
    if final_score < best_score:
        best_score = final_score
        best_seq = a_opt.copy()
        print(f"[{time.time():.1f}] FINAL refined score={final_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Trim tiny trailing values for neatness and return
    # ------------------------------------------------------------------
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] RETURN best score={best_score:.6f} length={best_seq.size}")
    # expose the sequence under a conventional name for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
