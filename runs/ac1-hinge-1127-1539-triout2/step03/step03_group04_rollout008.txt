def lp_refine(f, target_sum, max_iter, deadline):
    best_f = f.copy()
    best_val = _score(f)
    for iter in range(max_iter):
        if time.time()>deadline: break
        M = _max_convolution_fft(f)  # max of f*f
        # Build convolution matrix
        A = _conv_matrix(f)  # shape (2n-1, n)
        c = -np.ones(f.size)  # minimize -sum => maximize sum
        bounds = [(0, None)] * f.size
        # Solve LP
        res = linprog(c, A_ub=A, b_ub=np.full(2*f.size-1, M), bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # Scale down to target_sum
        if sum_g > target_sum:
            g = g * (target_sum / sum_g)
        else:
            # not enough sum, cannot improve further
            break
        # Mix
        t = 0.5
        f_new = (1 - t) * f + t * g
        # Normalize may be required due to rounding
        # Ensure sum close to target_sum: project onto simplex
        f_new = _project_to_simplex(f_new, target_sum)
        val = _score(f_new)
        if val < best_val:
            best_val = val
            best_f = f_new.copy()
            print(f"[{time.time():.1f}] LP iter={iter} val={val:.6f}")
        # set f for next iter
        f = f_new
        # optional early stop if improvement low
    return best_f, best_val
```

"""
Propose a step‑function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm is a multi‑stage search:

1. **Diverse random families** – power‑law, exponential, linear‑decay,
   piecewise‑constant and a few pure random vectors are generated for several
   lengths.  All candidates are projected onto the simplex
   {a_i≥0, Σa_i = √(2·n)} and scored with a fast FFT‑based evaluator.

2. **Linear‑program refinement** – for the best few candidates we solve a
   linear program

        max   Σ g_i
        s.t.  (f * g)_k ≤ M          for every convolution index k
              g_i ≥ 0

   where `f` is the current sequence and `M = max(f*f)`.  The optimal `g`
   is scaled down to the target sum and mixed with `f`.  A couple of such
   LP steps quickly lower the peak convolution.

3. **Annealed soft‑max gradient descent** – the mixed sequence is refined
   using the smooth max approximation and projected gradient descent
   (temperature β is annealed).

4. **Guided hill‑climbing** – mass is transferred from positions that
   contribute heavily to the current maximal convolution entry to positions
   that contribute little.  Only improving moves are kept.

All stages are time‑boxed; the best sequence found before the global budget
expires is returned.  The implementation uses only NumPy, SciPy and the
provided ``evaluate_sequence`` for the final check.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core utilities (FFT‑based max‑convolution, score, simplex projection)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast version of the benchmark objective (exact when Σa = √(2·n))."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:           # degenerate – should never happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient descent on the soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return (g, b) where b = a*a and g = ∇ softmax_β(max(b))."""
    n = a.shape[0]

    # full convolution a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over the 2·n‑1 entries
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # shape (2·n‑1,)

    # gradient = 2·(a ⋆ p)  (convolution of a with p, mirrored a)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1               # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]        # the central n components
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """Projected gradient descent on the soft‑max approximation, with annealing."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 256.0
    step_base = target_sum * 0.20
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD   beta={beta:.2f}  score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (mass transfer away from the current peak)
# ----------------------------------------------------------------------


def _guided_hill_climb(a: np.ndarray, deadline: float):
    """Iteratively relocate mass from high‑sensitivity to low‑sensitivity indices."""
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07          # maximum fraction of a donor’s mass to move

    while time.time() < deadline:
        conv = np.convolve(a, a)          # O(n) for moderate n
        k_max = int(np.argmax(conv))

        # sensitivity vector: contribution of a[i] to conv[k_max]
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        sens[lo : hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        weight = sens * a
        if weight.max() == 0:
            break

        donor = int(np.argmax(weight))

        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver or a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] GH   improved = {best_val:.6f}")
        else:
            # revert the move
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Create an exponential decay sequence exp(-lam·i) and normalise."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _linear_sequence(n: int, target_sum: float) -> np.ndarray:
    """Triangular (linearly decreasing) shape, clipped to non‑negative."""
    a = np.maximum(0.0, (n - np.arange(n, dtype=np.float64)))
    a *= target_sum / a.sum()
    return a


def _piecewise_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Piecewise constant then power‑law tail."""
    K = rng.integers(5, n // 2)
    const = rng.uniform(0.5, 1.5)
    tail = (np.arange(1, n - K + 1, dtype=np.float64) ** (-rng.uniform(0.3, 0.8)))
    a = np.concatenate([np.full(K, const), tail])
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑program refinement (LP step described in the reference)
# ----------------------------------------------------------------------


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that A @ g = f * g  (full linear convolution).
    Row k, column j contains f[k‑j] when k‑j∈[0,n‑1], otherwise 0.
    """
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f[j]
    return A


def _lp_refine(f: np.ndarray, target_sum: float, deadline: float,
               max_iters: int = 3) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinement steps:
        maximize Σ g_i  s.t. (f * g)_k ≤ M, g_i ≥ 0
    Then mix the solution into f.
    Returns the refined sequence and its score.
    """
    n = f.size
    best_f = f.copy()
    best_val = _score(f)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = _max_convolution_fft(f)
        A = _build_convolution_matrix(f)
        c = -np.ones(n)                     # maximize Σ g_i  <=> minimize -Σ g_i
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        # solve the LP (using HiGHS, which is fast for dense problems)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Scale down so that Σ g_i = target_sum (the LP gives a vector with Σ g_i ≥ target_sum)
        if sum_g > target_sum:
            g = g * (target_sum / sum_g)
        else:
            # cannot reach the target sum without violating constraints – stop
            break

        # Mix with the current sequence (t = 0.5 works well)
        t = 0.5
        f_new = (1.0 - t) * f + t * g
        f_new = _project_to_simplex(f_new, target_sum)

        val_new = _score(f_new)
        if val_new < best_val:
            best_val = val_new
            best_f = f_new.copy()
            print(f"[{time.time():.1f}] LP   n={n} iter={it} score={best_val:.6f}")

        f = f_new

    return best_f, best_val

# Main entry point required by the competition

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine builds a diverse pool of candidates, refines the most
    promising ones with an LP step, followed by annealed gradient descent
    and guided hill‑climbing.  The best sequence found before the time
    budget expires is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Stage 1 – generate a diverse set of candidates
    # ------------------------------------------------------------------
    candidates = []   # each entry: (score, array, n)

    # maybe reuse the previous best if it exists
    if 'height_sequence_1' in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = seq.size
        target = np.sqrt(2.0 * n)
        seq = _project_to_simplex(seq, target)
        candidates.append((np.inf, seq, n))

    # Choose a few lengths (larger n often yields lower peaks)
    lengths = [1500, 1800, 2100]   # modest size – keeps LP solvable
    for n in lengths:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)

        # --- power‑law family -------------------------------------------------
        for _ in range(30):
            p = rng.uniform(0.35, 0.75)
            a = _power_law_sequence(n, p, target)
            candidates.append((np.inf, a, n))

        # --- exponential family -----------------------------------------------
        for _ in range(30):
            lam = rng.uniform(1e-4, 0.02)
            a = _exponential_sequence(n, lam, target)
            candidates.append((np.inf, a, n))

        # --- linear (triangular) shape ----------------------------------------
        for _ in range(10):
            a = _linear_sequence(n, target)
            # small random jitter to break perfect symmetry
            a += rng.normal(scale=target * 1e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            candidates.append((np.inf, a, n))

        # --- piecewise‑constant + tail ----------------------------------------
        for _ in range(20):
            a = _piecewise_sequence(n, target, rng)
            a += rng.normal(scale=target * 1e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            candidates.append((np.inf, a, n))

        # --- pure random (simplex‑projected) ----------------------------------
        for _ in range(15):
            a = rng.random(n)
            a = _project_to_simplex(a, target)
            candidates.append((np.inf, a, n))

    # Evaluate all candidates quickly (FFT based)
    for idx, (_, a, n) in enumerate(candidates):
        if time.time() >= deadline:
            break
        val = _score(a)
        candidates[idx] = (val, a, n)

    # Keep the top few for deeper optimisation
    candidates.sort(key=lambda t: t[0])
    topk = 8
    candidates = candidates[:topk]

    # ------------------------------------------------------------------
    # Stage 2 – LP refinement + gradient descent + hill climbing
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for base_val, a0, n in candidates:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)

        # ---- LP refinement ------------------------------------------------
        lp_deadline = time.time() + min(20.0, (deadline - time.time()) / 2.0)
        a_lp, val_lp = _lp_refine(a0, target, lp_deadline, max_iters=3)

        # ---- Annealed gradient descent ------------------------------------
        gd_deadline = time.time() + min(30.0, deadline - time.time())
        a_gd, val_gd = _annealed_gradient_descent(a_lp, gd_deadline)

        # ---- Guided hill climbing -----------------------------------------
        if time.time() < deadline:
            a_hc, val_hc = _guided_hill_climb(a_gd, deadline)
        else:
            a_hc, val_hc = a_gd, val_gd

        # Update global best
        if val_hc < best_score:
            best_score = val_hc
            best_seq = a_hc.copy()
            print(f"** GLOBAL BEST **  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)

    # Trim trailing zeros (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Return as plain Python list – the caller will pass it to evaluate_sequence
    return best_seq.tolist()
