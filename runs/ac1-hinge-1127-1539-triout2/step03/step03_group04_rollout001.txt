def _adam_softmax_descent(a0, deadline):

    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2*n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = target * 0.1  # base learning rate

    best_a = a.copy()
    best_score = _score(a)

    beta = 2.0
    max_beta = 1024.0
    step = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        step += 1
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        # dynamic learning rate: decaying with beta?
        lr = lr0 / (beta + 1)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = _project_to_simplex(a, target)

        # jitter occasionally
        if step % 500 == 0:
            rng = np.random.default_rng()
            a += rng.normal(scale=target * 0.001 / (beta + 1), size=n)
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target)

        # increase beta gradually
        if step % 400 == 0 and beta < max_beta:
            beta = min(beta * 2, max_beta)

        val = _score(a)
        if val < best_score:
            best_score = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam beta={beta:.2f} score={best_score:.6f}")

    return best_a, best_score

def _simulated_annealing(a0, deadline):

    a = a0.copy()
    n = a.size
    target = np.sqrt(2*n)
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    cur_a = a.copy()
    start = time.time()
    rng = np.random.default_rng()
    # set initial temperature based on best_score - maybe use percentage.
    T0 = best_score * 0.1  # initial temperature
    # schedule: linear cooling.
    while time.time() < deadline:
        elapsed = time.time() - start
        # fraction of time used
        frac = elapsed / (deadline - start)
        T = T0 * (1 - frac)  # linearly reduce
        if T < 1e-6:
            T = 1e-6
        # propose move
        # select donor i with probability proportional to a_i * weight maybe
        # Use sensitivity weight: pick i where a_i * (peak_contrib) is high, else random.
        # Simpler: random i, random j
        i = rng.integers(n)
        if cur_a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        delta_frac = rng.random() * 0.05  # up to 5% of donor mass
        delta = cur_a[i] * delta_frac
        a_candidate = cur_a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # compute score
        cand_score = _score(a_candidate)
        # accept if better or with probability
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / T):
            cur_a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                print(f"[{time.time():.1f}] SA improved to {best_score:.6f}")

    return best_a, best_score

"""
Improved optimizer for the “max‑convolution / ℓ¹‑norm’’ benchmark.

The algorithm proceeds in three phases:

1. **Initialisation** – start from the best known construction
   (`height_sequence_1` if available) and a few jittered copies.
   If none are present a simple power‑law seed is created.

2. **Adam on a smooth max** – minimise a soft‑max approximation of the
   maximal convolution entry with an Adam optimiser.  The temperature
   (`β`) controlling the smoothness is annealed (geometrically) up to
   large values (≈2048) so the optimisation gradually focuses on the
   true max.  After each step the vector is projected onto the simplex
   {x≥0, Σx = √(2 n)}.

3. **Simulated‑annealing refinement** – a cheap random walk that moves
   a small fraction of mass from a donor index to a receiver index.
   Moves that improve the exact max‑convolution are always accepted,
   otherwise they are accepted with a probability decreasing with a
   linear temperature schedule.

Both phases respect the non‑negativity and sum constraints, and the
best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Exact competition objective for a (the result equals max(conv))."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:               # safety – should never happen after projection
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where b = a*a (full convolution) and
    g = ∇ (log‑sum‑exp_{β}(b)).
    """
    n = a.shape[0]

    # full convolution b = a*a (via FFT)
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # shape (2·n‑1,)

    # gradient: g = 2·(a_rev ⋆ p) restricted to first n entries
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1              # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]

    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_softmax_descent(a0: np.ndarray, deadline: float):
    """
    Projected Adam optimiser on the smooth max (log‑sum‑exp) of the convolution.
    The temperature β is increased geometrically up to a large value.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # learning‑rate base (scaled with the target sum)
    lr_base = target_sum * 0.20

    best_a = a.copy()
    best_val = _score(a)

    beta = 2.0
    max_beta = 2048.0
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size scaled down as β grows
        lr = lr_base / (beta + 1.0)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam β={beta:.1f}  score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (exact max‑convolution)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Random walk that moves a small fraction of mass between two indices.
    Accepts worsening moves with a linearly decreasing temperature.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val
    cur_a = a.copy()

    start = time.time()
    rng = np.random.default_rng()

    # initial temperature proportional to the current score
    T0 = best_val * 0.10 if best_val > 0 else 0.01

    while time.time() < deadline:
        elapsed = time.time() - start
        # linear cooling
        frac = elapsed / (deadline - start)
        T = max(T0 * (1.0 - frac), 1e-7)

        # pick donor and receiver
        i = rng.integers(n)
        if cur_a[i] < 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # move a random fraction (up to 5 %) of donor's mass
        delta = cur_a[i] * (rng.random() * 0.05)
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # projection is not needed – the move preserves the sum and positivity
        cand_val = _score(cand)

        # Metropolis acceptance
        if cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) / T):
            cur_a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
                print(f"[{time.time():.1f}] SA improved to {best_val:.6f}")

    # final projection to guarantee feasibility (tiny numerical drift)
    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_val
