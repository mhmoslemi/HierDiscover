"""
Propose a step‑function (non‑negative coefficient vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.  The algorithm consists of
four stages and runs within the 1000 s time budget:

1. **Initial candidates** – a short parametric sweep over several
   symmetric *U‑shapes* (linear, quadratic, exponential, power,
   inverse) and a few *skewed* shapes (left‑/right‑biased linear and
   exponential).  Random Dirichlet vectors are also tried.  All vectors
   are projected onto the ℓ₁‑simplex
   `{x_i≥0, Σx_i = √(2 n)}`.  The best candidate so far (lowest
   max‑self‑convolution) is kept.

2. **Smoothed gradient descent** – minimise a log‑sum‑exp smooth
   surrogate of the maximum using momentum.  The smoothing parameter
   τ = τ_factor·max_conv starts relatively large (τ_factor≈0.15) and is
   reduced geometrically; the step size η decays slowly.  Every few
   thousand iterations a tiny random mass transfer and a small jitter
   are applied to avoid stagnation.

3. **Sub‑gradient refinement** – after the smooth phase we switch to
   the true nonsmooth objective (the maximum of the self‑convolution).
   The sub‑gradient is the average of gradients at all (approximately)
   maximal lags.  A decreasing step size together with occasional random
   pair‑transfers is used.

4. **Polishing** – a final burst of many tiny random pair‑transfers
   (≈ few hundred) keeps any improvement found.

Throughout the run the best vector found is stored, and at the end it
is returned as a plain Python list.  The code obeys the 1000 s budget
by partitioning the available time into the four stages.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of `a` with itself using an FFT of length
    equal to the next power of two.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a (possibly asymmetric) “U‑shaped’’ or biased sequence of length n.
    `kind` may be one of:
        linear, quadratic, exponential, power, inverse,
        linear_left, linear_right, exponential_left, exponential_right.
    The parameter `param` controls the shape (exponent, rate, etc.).
    The result is projected onto the ℓ₁‑simplex with the required sum.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    elif kind == "linear_left":
        # decreasing from left to right (param ∈ [0,1))
        raw = 1.0 - param * idx / (n - 1)
    elif kind == "linear_right":
        raw = 1.0 - param * (n - 1 - idx) / (n - 1)
    elif kind == "exponential_left":
        raw = np.exp(-param * idx)
    elif kind == "exponential_right":
        raw = np.exp(-param * (n - 1 - idx))
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def compute_max_conv(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (a is assumed non‑negative)."""
    return float(conv_self_fft(a).max())
