"""
Propose a non‑negative step function (a vector of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The code follows a multi‑stage heuristic:

1.  **Broad parametric sweep** – generate many symmetric “U‑shapes’’
    (linear, quadratic, exponential, power, inverse‑power and a β‑distribution
    based shape).  Each candidate is projected onto the ℓ₁‑simplex
    `sum = sqrt(2 n)` and the best is kept.

2.  **Progressive up‑sampling** – start with a modest size (≈4096),
    optimise it, then linearly interpolate to a larger power‑of‑two size
    (8192 → 16384 → 32768) re‑optimising each time.  Larger n gives the
    optimiser more freedom to lower the peak of the self‑convolution.

3.  **Adam‑based smoothed gradient descent** – the max‑convolution is
    approximated by a log‑sum‑exp (temperature `τ`).  The gradient is
    obtained by a single FFT‑based convolution.  Adam’s per‑coordinate
    step‑size adaptation together with a decreasing learning‑rate and
    temperature schedule yields fast, stable convergence.

4.  **Simulated annealing** – after the gradient phase a cheap
    random‑pair‑transfer move is explored with a Metropolis acceptance
    rule.  The temperature is slowly cooled, allowing occasional uphill
    moves that escape local minima.

5.  **Polishing** – a final burst of tiny jitter / pair transfers
    keeps any remaining improvement.

All heavy work uses FFT‑based convolutions (O(n log n)).  The routine
runs until the 1000 s time budget expires and returns the best vector
found.  It also re‑uses an existing `height_sequence_1` (if supplied)
as a warm‑start.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = target} (Duchi et al. 2008)."""
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random mass‑transfer move
# ----------------------------------------------------------------------


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Move a small amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Shape generators for the parametric sweep
# ----------------------------------------------------------------------


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """Symmetric U‑shaped sequence (linear, quadratic, exponential, power, inverse)."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-6
    elif kind == "inverse":
        raw = 1.0 / (1e-6 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_beta_shape(n: int, target_sum: float, alpha: float, beta: float) -> np.ndarray:
    """U‑shape obtained by inverting a beta‑distribution density."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # beta density ∝ x^{α‑1} (1‑x)^{β‑1}; invert to make edges large
    eps = 1e-6
    dens = (x + eps) ** (alpha - 1) * ((1.0 - x) + eps) ** (beta - 1)
    raw = 1.0 / (dens + eps)
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def interpolate_to_length(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence onto a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    interp = np.interp(x_new, x_old, seq)
    return np.maximum(interp, 0.0)


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------


def gradient_smooth(a: np.ndarray, tau_factor: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad is the gradient of
    log‑sum‑exp approximation of max(conv(a,a)).
    """
    conv = conv_self_fft(a)
    max_c = conv.max()
    tau = max(tau_factor * max_c, 1e-12)

    w = np.exp((conv - max_c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]

    # pick the n entries that correspond to the convolution of a with a
    grad = (2.0 / tau) * conv_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]
    return grad, max_c


# ----------------------------------------------------------------------
# Adam optimiser using the smoothed gradient
# ----------------------------------------------------------------------


def adam_optimize(initial_a: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  init_lr: float = 0.02,
                  init_tau: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the max‑self‑convolution.
    Stops when the deadline is reached.
    Returns (best_sequence, best_max_conv).
    """
    a = initial_a.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    lr = init_lr * (target_sum / n)      # scale with typical magnitude
    tau_factor = init_tau

    iteration = 0
    best_a = a.copy()
    best_score = compute_max_conv(a)

    print(f"  Adam start – n={n}, init LR={lr:.4e}, init τ={tau_factor:.4e}")

    while time.time() < deadline:
        iteration += 1
        grad, cur_max = gradient_smooth(a, tau_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # evaluate newly obtained max‑conv
        cur_max = compute_max_conv(a)

        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"    Adam iter {iteration:07d} – new best {best_score:.6f}")

        # schedules
        if iteration % 2000 == 0:
            lr *= 0.90                # slowly decay learning rate
            tau_factor = max(0.001, tau_factor * 0.90)

        # occasional exploratory mass transfer
        if np.random.rand() < 0.008:
            a = random_pair_transfer(a, target_sum, scale=0.006)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing phase
# ----------------------------------------------------------------------


def simulated_annealing(start_a: np.ndarray,
                        target_sum: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis annealing using random pair transfers.
    Returns the best sequence seen and its max‑convolution.
    """
    a = start_a.copy()
    cur_score = compute_max_conv(a)
    best_a = a.copy()
    best_score = cur_score

    T0 = cur_score * 0.15          # initialise temperature as a fraction of the score
    t = 0

    print(f"  Simulated annealing start – init score {cur_score:.6f}")

    while time.time() < deadline:
        t += 1
        T = T0 * (0.9998 ** t)     # exponential cooling

        # propose a tiny pair transfer
        cand = random_pair_transfer(a.copy(), target_sum, scale=0.004)
        cand_score = compute_max_conv(cand)
        delta = cand_score - cur_score

        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"    SA step {t:07d} – improved to {best_score:.6f}")

        # occasional larger jitter to shake the system
        if t % 5000 == 0:
            jitter = np.random.randn(a.size) * 0.0006
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            cur_score = compute_max_conv(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine respects the given time budget
    (≈1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.4   # small safety margin

    # ------------------------------------------------------------------
    # Determine problem size(s).  We will climb a ladder of powers of two.
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        # reuse the supplied length as the final target size
        size_hierarchy = [len(base)]
    else:
        size_hierarchy = [4096, 8192, 16384, 32768]   # increase while time permits

    best_overall_seq = None
    best_overall_score = math.inf

    for n in size_hierarchy:
        # stop early if we run out of time
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this size
        # --------------------------------------------------------------
        if best_overall_seq is not None and best_overall_seq.shape[0] != n:
            # upscale the current best to the new length
            init_seq = interpolate_to_length(best_overall_seq, n)
            init_seq = project_onto_simplex(init_seq, target_sum)
        elif best_overall_seq is not None:
            init_seq = best_overall_seq.copy()
        else:
            # fresh uniform start
            init_seq = np.full(n, target_sum / n, dtype=np.float64)

        # --------------------------------------------------------------
        # Stage 0 – parametric sweep (lightweight)
        # --------------------------------------------------------------
        best_seq = init_seq.copy()
        best_score = compute_max_conv(best_seq)

        print(f"[{time.time():.2f}] Size {n} – parametric sweep start (current best {best_score:.6f})")

        shape_kinds = ("linear", "quadratic", "exponential", "power", "inverse")
        param_grid = {
            "linear": np.linspace(0.0, 12.0, 61),            # more aggressive range
            "quadratic": np.linspace(0.0, 8.0, 41),
            "exponential": np.linspace(0.0, 10.0, 51),
            "power": np.linspace(0.2, 8.0, 40),
            "inverse": np.linspace(0.2, 8.0, 40),
        }

        for kind in shape_kinds:
            for p in param_grid[kind]:
                cand = generate_shape(n, target_sum, kind, p)
                cand_score = compute_max_conv(cand)
                if cand_score + 1e-12 < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(f"    sweep {kind} p={p:.3f} → {best_score:.6f}")
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

        # additional beta‑shape sweep
        alphas = np.linspace(0.5, 5.0, 20)
        betas = np.linspace(0.5, 5.0, 20)
        for a in alphas:
            for b in betas:
                cand = generate_beta_shape(n, target_sum, a, b)
                cand_score = compute_max_conv(cand)
                if cand_score + 1e-12 < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(f"    beta shape α={a:.2f} β={b:.2f} → {best_score:.6f}")
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

        # --------------------------------------------------------------
        # Stage 1 – Adam based smoothed gradient descent
        # --------------------------------------------------------------
        sub_deadline = deadline - 0.2   # leave some room for later phases
        best_seq, best_score = adam_optimize(best_seq, target_sum, sub_deadline)

        # --------------------------------------------------------------
        # Stage 2 – Simulated annealing fine‑tuning
        # --------------------------------------------------------------
        if time.time() < deadline:
            best_seq, best_score = simulated_annealing(best_seq, target_sum, deadline)

        # --------------------------------------------------------------
        # Polishing – tiny random pair transfers
        # --------------------------------------------------------------
        while time.time() < deadline:
            cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.002)
            cand_score = compute_max_conv(cand)
            if cand_score + 1e-12 < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"    polish → {best_score:.6f}")

        # Record the best for the whole run
        if best_score + 1e-12 < best_overall_score:
            best_overall_score = best_score
            best_overall_seq = best_seq.copy()
            print(f"[{time.time():.2f}] New overall best (n={n}) → {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – uniform sequence (should never happen)
        n = size_hierarchy[-1]
        target_sum = math.sqrt(2.0 * n)
        best_overall_seq = np.full(n, target_sum / n, dtype=np.float64)

    print(f"[{time.time():.2f}] Returning best sequence; final max‑conv = {best_overall_score:.6f}")
    return best_overall_seq.tolist()
