def _fft_convolve(x, y):
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:n]
    return conv

"""
A multi‑phase heuristic for minimizing the L∞‑norm of a self‑convolution
under a fixed ℓ₁‑norm (∑a_i = √(2·n)).  The algorithm works as follows:

1.  If a previously discovered sequence is available (global variable
    ``height_sequence_1``) it is refined with a dedicated optimiser.
2.  The remaining time budget is spent on many independent restarts.
    Each restart picks a length n (moderate or large), builds a feasible
    starting vector (either a random Dirichlet draw or a scaled copy of the
    best known sequence) and runs a local optimiser.
3.  The local optimiser mixes several moves:
        – a soft‑max approximation of the ∞‑norm (smooth gradient descent),
        – an exact sub‑gradient step on the max‑convolution entry with a
          simple line‑search,
        – random mass‑swap moves (simulated‑annealing style),
        – occasional Gaussian jitter.
    All updates are projected onto the simplex
        {x ≥ 0, ∑x = √(2·n)}.
4.  The best sequence discovered overall is returned.

The implementation relies on FFT‑based convolutions (O(n log n)) and
uses only NumPy – no external solvers are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: nearest power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the simplex with total mass S."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]               # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution of two real vectors
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y using a real FFT."""
    n = len(x) + len(y) - 1
    f_len = _next_pow2(n)
    X = np.fft.rfft(x, n=f_len)
    Y = np.fft.rfft(y, n=f_len)
    conv = np.fft.irfft(X * Y, n=f_len)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Linear scaling / interpolation of a sequence to a new length
# ----------------------------------------------------------------------
def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to ``new_n`` by linear interpolation."""
    old_n = len(seq)
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Initialise a feasible vector (ℓ₁‑norm = √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with sum √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size > 0:
        a = _scale_sequence(base_seq, n).astype(float)
        a += rng.normal(scale=max(a.mean(), 1e-12) * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)                     # Dirichlet(1)
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Soft‑max (smooth) gradient of the ∞‑norm of a*a
# ----------------------------------------------------------------------
def _softmax_update(a: np.ndarray, tau: float, eta: float):
    """
    One additive step using a τ‑soft‑max approximation of max(b),
    where b = a*a.  Returns the projected vector, the new convolution
    and its maximum entry.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau <= 1e-12:                      # hard max → sub‑gradient (fallback)
        # treat as hard max by using the exact sub‑gradient (handled elsewhere)
        g = np.zeros_like(a)
    else:
        # smooth soft‑max weights
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)           # shape (2n‑1,)

        # cross‑correlation a ⋆ w = Σ_j a_j w_{i+j}
        a_rev = a[::-1]
        conv_len = len(w) + n - 1
        f_len = _next_pow2(conv_len)
        Fw = np.fft.rfft(w, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]

    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    b_new = _full_convolution_fft(a_proj)
    max_new = b_new.max()
    return a_proj, b_new, max_new


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(b) and the corresponding max value
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, eps: float = 0.0):
    """
    Compute the sub‑gradient of max(a*a) and the maximal convolution entry.
    ``eps`` allows a small tolerance so that all entries within
    max‑eps are treated as active.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    if eps > 0.0:
        mask = (b >= max_b - eps).astype(float)
    else:
        mask = (b == max_b).astype(float)

    a_rev = a[::-1]
    conv = _fft_convolve(mask, a_rev)            # length 3n‑2
    g = 2.0 * conv[n - 1:n - 1 + n]               # slice length n
    return g, max_b


# ----------------------------------------------------------------------
# Local optimiser (one restart)
# ----------------------------------------------------------------------
def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Hybrid optimiser mixing soft‑max steps, hard sub‑gradient steps,
    simulated‑annealing swaps and jitter.  Returns the best vector seen
    and its maximal self‑convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise record
    b = _full_convolution_fft(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters (tuned empirically)
    eta = 0.7                 # additive step size for gradient updates
    eta_decay = 0.9997

    move_prob = 0.13          # probability of a random mass move
    eps_move = 0.04           # fraction of mass moved when a move is tried
    temperature = cur_max * 0.12
    temp_decay = 0.9996

    jitter_prob = 0.015
    jitter_scale = 0.015

    use_softmax = True
    tau = cur_max
    tau_decay = 0.9995

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) choose an update type
        # --------------------------------------------------------------
        r = rng.random()

        if use_softmax and r < 0.55:
            # ---- smooth soft‑max gradient step ----
            a, b, cur_max = _softmax_update(a, max(tau, 1e-12), eta)
            tau *= tau_decay

        elif r < 0.80:
            # ---- exact hard‑max sub‑gradient step with tiny line‑search ----
            g, cur_max = _subgradient_max(a)
            # try a few step‑size scalings
            improved = False
            for factor in (0.5, 1.0, 1.5):
                a_cand = a - eta * factor * g
                a_cand = _project_to_simplex(a_cand, S)
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max - 1e-9:
                    a = a_cand
                    cur_max = max_cand
                    improved = True
                    break
            if not improved:
                # accept worse move with simulated‑annealing probability
                a_cand = a - eta * g
                a_cand = _project_to_simplex(a_cand, S)
                max_cand = _full_convolution_fft(a_cand).max()
                if max_cand < cur_max or rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                    a = a_cand
                    cur_max = max_cand
                temperature *= temp_decay

        elif r < 0.95:
            # ---- random mass swap (simulated annealing) ----
            i = rng.integers(n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = eps_move * a[i]
            a_cand = a.copy()
            a_cand[i] -= delta
            a_cand[j] += delta
            max_cand = _full_convolution_fft(a_cand).max()
            if max_cand < cur_max or rng.random() < np.exp(-(max_cand - cur_max) / max(temperature, 1e-12)):
                a = a_cand
                cur_max = max_cand
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
                temperature *= temp_decay

        else:
            # ---- Gaussian jitter ----
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a_cand = np.clip(a + noise, 0.0, None)
            a_cand = _project_to_simplex(a_cand, S)
            max_cand = _full_convolution_fft(a_cand).max()
            if max_cand < cur_max:
                a = a_cand
                cur_max = max_cand

        # --------------------------------------------------------------
        # 2) bookkeeping
        # --------------------------------------------------------------
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # decay of the additive step size
        eta *= eta_decay

        # occasional toggle of the soft‑max mode (helps escape plateaus)
        if rng.random() < 0.01:
            use_softmax = not use_softmax

    return best_a, best_max

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Run a time‑limited search for a non‑negative step sequence that
    minimises the evaluation function.  The function returns the best
    sequence discovered within the budget.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0          # 5‑second safety margin

    # ------------------------------------------------------------------
    # Retrieve a previously stored construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – polish the existing construction (if available)
    # ------------------------------------------------------------------
    if base_seq is not None and base_seq.size > 0:
        n0 = int(base_seq.shape[0])
        a0 = _project_to_simplex(base_seq, np.sqrt(2.0 * n0))
        phase_deadline = min(time.time() + 180.0, end_time)   # up to 3 min
        seq_opt, score_opt = _local_optimize(a0, phase_deadline, rng)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[phase 1] refined base length={n0} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many independent restarts with varied lengths
    # ------------------------------------------------------------------
    while time.time() < end_time:
        attempt += 1

        # pick a length: mix moderate and large sizes

    if best_seq is None:

        globals()['height_sequence_1'] = final_seq

    return final_seq
