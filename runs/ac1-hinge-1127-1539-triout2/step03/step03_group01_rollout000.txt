def _full_conv(a):
    n = len(a)
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len-1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa*Fa, n=fft_len)[:conv_len]
    return conv

def _gradient(a, tau):
    n = a.shape[0]
    b = _full_conv(a)
    max_b = b.max()
    if tau <= 1e-12:
        # subgradient at max index
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:
            d = k//2
            if 0 <= d < n:
                g[d] = a[d]  # diagonal counted once
        return g, max_b
    else:
        # compute weights w = exp(b/tau) / sum
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        # compute gradient g = 2 * conv(a_rev, w) slice
        a_rev = a[::-1]
        m = 1 << ((len(w) + n - 1).bit_length())
        Fw = np.fft.rfft(w, n=m)
        Fa_rev = np.fft.rfft(a_rev, n=m)
        conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n -1]
        g = 2.0 * conv[n-1:n-1+n]
        return g, max_b

def _adam_optimize(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0*n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    # initial tau
    b = _full_conv(a)
    max_b = b.max()
    tau = max_b * 0.5  # maybe start at 0.5*max? Not sure.
    tau_decay = 0.9995
    # also schedule lr
    base_lr = 0.5
    # store best
    best_a = a.copy()
    best_max = max_b
    while time.time() < deadline:
        step += 1
        # gradient
        g, max_b = _gradient(a, tau)
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1-beta1**step)
        v_hat = v / (1-beta2**step)
        lr = base_lr / (np.sqrt(v_hat) + eps)
        # elementwise lr? typical Adam uses division each coordinate: a - lr * m_hat / (sqrt(v_hat)+eps)
        # We'll compute new a
        a = a - lr * m_hat
        # project
        a = _project_to_simplex(a, S)
        # compute new max
        b = _full_conv(a)
        cur_max = b.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        # update tau
        tau = max(tau * tau_decay, 1e-8)
        # occasional random jitter
        if rng.random() < 0.005:
            noise = rng.normal(scale=a.mean()*0.01, size=n)
            a = np.clip(a + noise, 0, None)
            a = _project_to_simplex(a, S)
        # (optional) occasional large perturbation if stuck - maybe if no improvement for many steps.
        # we can keep track of stagnation but ignore for now.
    return best_a, best_max

"""
Projected Adam optimisation on a smooth (log‑sum‑exp) approximation of the
∞‑norm of the self‑convolution.

The algorithm:
 1. Initialise a feasible non‑negative vector a of length n with
    ℓ₁‑norm S = √(2·n) (simplex projection).
 2. Repeatedly perform either
      – a stochastic mass‑move (simulated‑annealing style), or
      – an Adam gradient step on the smooth surrogate
        τ·log Σ exp((a*a)_k/τ),
    followed by projection onto the simplex.
 3. The temperature τ (soft‑max temperature) and the annealing temperature
    are exponentially decayed.
 4. Occasionally add Gaussian jitter to escape shallow minima.
 5. Run many independent restarts (different lengths, random seeds) within
    the given time budget and keep the best vector (lowest max convolution).

The returned list is evaluated by the provided `evaluate_sequence`,
which for feasible inputs equals the maximal entry of a*a.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]               # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Full convolution a * a using a real FFT (returns length 2n‑1)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

# ----------------------------------------------------------------------
# Gradient of τ·log Σ exp(b/τ) w.r.t. a (b = a*a)
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, b: np.ndarray, tau: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the smooth approximation of max(b) and return
    the gradient vector and the current maximum of b.
    """
    n = a.shape[0]
    max_b = b.max()

    if tau <= 1e-12:
        # Hard max (sub‑gradient)
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]   # diagonal counted once
        return g, max_b

    # Smooth case: soft‑max weights
    diff = b - max_b                     # for numerical stability
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)              # shape (2n‑1,)

    # Gradient: g_i = 2 Σ_j a_j w_{i+j}
    a_rev = a[::-1]
    m = 1 << ((len(w) + n - 1).bit_length())
    Fw = np.fft.rfft(w, n=m)
    Fa_rev = np.fft.rfft(a_rev, n=m)
    conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n - 1]
    g = 2.0 * conv[n - 1:n - 1 + n]       # slice of length n
    return g, max_b

# ----------------------------------------------------------------------
# Local optimiser (single restart)
# ----------------------------------------------------------------------
def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mixed optimisation:
      – Adam projected gradient steps on the smooth surrogate,
      – occasional simulated‑annealing mass swaps,
      – rare Gaussian jitter.
    Returns the best vector discovered and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initial statistics
    b = _full_convolution_fft(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # Adam bookkeeping
    m_adam = np.zeros_like(a)
    v_adam = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5
    step = 0

    # annealing parameters
    temp = cur_max * 0.1
    temp_decay = 0.9995
    move_prob = 0.07
    epsilon_move = 0.05

    jitter_prob = 0.01
    jitter_scale = 0.02

    # soft‑max temperature schedule
    tau = cur_max
    tau_decay = 0.9997

    while time.time() < deadline:
        step += 1

        if rng.random() < move_prob:
            # ---- random mass move (simulated annealing) ----
            i = rng.integers(n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = epsilon_move * a[i]
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # ensure non‑negativity (tiny rounding errors)
            if a_candidate[i] < 0.0:
                continue
            b_candidate = _full_convolution_fft(a_candidate)
            cand_max = b_candidate.max()
            accept = False
            if cand_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                a = a_candidate
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temp *= temp_decay

        else:
            # ---- Adam gradient step on smooth surrogate ----
            b = _full_convolution_fft(a)
            cur_max = b.max()
            g, _ = _gradient(a, b, tau)

            # Adam update
            m_adam = beta1 * m_adam + (1.0 - beta1) * g
            v_adam = beta2 * v_adam + (1.0 - beta2) * (g ** 2)
            m_hat = m_adam / (1.0 - beta1 ** step)
            v_hat = v_adam / (1.0 - beta2 ** step)

            a = a - base_lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto the ℓ₁‑simplex
            a = _project_to_simplex(a, S)

            # recompute statistics after the projection
            b = _full_convolution_fft(a)
            cur_max = b.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

            # decay the soft‑max temperature
            tau = max(tau * tau_decay, 1e-12)

        # occasional jitter to perturb stagnated solutions
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _full_convolution_fft(a)
            cur_max = b.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a, best_max

    MAX_N = 20000
