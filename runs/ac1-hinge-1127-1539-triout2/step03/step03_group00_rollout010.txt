def _simulated_annealing(a0, deadline):
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score
    T = best_score * 0.05 + 1e-6  # start temperature
    cooling = 0.9999
    rng = np.random.default_rng()
    # Precompute cumulative sum for weighted sampling
    while time.time() < deadline:
        # source index weighted by a (probability mass)
        prob = a / a.sum()
        i = rng.choice(n, p=prob)
        # target index: choose low mass with 1/(a+epsilon) weight
        inv_prob = 1.0 / (a + 1e-12)
        inv_prob = inv_prob / inv_prob.sum()
        j = rng.choice(n, p=inv_prob)
        if i == j:
            continue
        # transfer fraction upto say 0.3 of source mass
        frac = rng.random() * 0.3
        delta = a[i] * frac
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        new_score = _score(a)
        dE = new_score - cur_score
        if dE <= 0 or (T > 0 and rng.random() < np.exp(-dE / T)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling
        if rng.random() < 0.001:
            # occasional global jitter
            jitter = rng.normal(scale=a.sum()*0.001, size=n)
            a = np.maximum(a + jitter, 0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best

def _adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    beta = 0.5
    max_beta = 2048.0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # learning rate schedule: start with 0.5 * target_sum / (beta+1)
        lr = 0.5 * target_sum / (beta + 1.0)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)
        # anneal beta
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.4, max_beta)
        # occasional jitter to escape local minima
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
    return a

"""
Improved step‑function search.

The algorithm works in three stages:

1. **Broad random search** – for a set of candidate lengths we draw a
   large variety of random non‑negative vectors (power‑law, exponential,
   Dirichlet, log‑normal, beta‑shaped, sinusoidal, …).  Each vector is
   projected onto the ℓ¹‑simplex with Σa = √(2·n) so that the evaluation
   objective equals the maximum entry of the self‑convolution.

2. **Annealed Adam optimisation** – the best few random seeds for each
   length are refined by an Adam‑style projected gradient descent on a
   smooth log‑sum‑exp approximation of the maximum convolution.  The
   smoothing parameter β is increased slowly (annealing) and a tiny
   jitter is added periodically to escape plateaus.

3. **Simulated‑annealing + deterministic local polishing** – the Adam‑
   refined candidates are further improved by a classic SA routine that
   moves a fraction of mass between a source and a low‑mass target.
   After the time budget is almost exhausted a cheap deterministic local
   refinement focuses on the current maximal convolution entry and
   transfers a small amount of mass from the most responsible position
   to a low‑mass one.

All steps respect the non‑negativity and ℓ¹‑simplex constraints, and the
best sequence found before the deadline is returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective after ℓ¹‑normalisation (simply max convolution)."""
    return _max_convolution_fft(a)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth maximum (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where
        conv = a*a (full linear convolution)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·conv_k) )
              = 2·(a ⋆ p)   with p_k ∝ exp(β·conv_k)
    """
    n = a.shape[0]

    # ---- full convolution a*a ---------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ------------------------------------
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)

    # ---- gradient g = 2·(a ⋆ p) ------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser (projected on the simplex) with β‑annealing
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed Adam‑style projected gradient descent on the smoothed max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5               # start smoothing
    max_beta = 2048.0        # final smoothing
    m = np.zeros_like(a)     # first moment
    v = np.zeros_like(a)     # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam update (per‑coordinate)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # scalar base learning rate that shrinks with β
        lr = 0.5 * target_sum / (beta + 1.0)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.4, max_beta)

        # occasional jitter to shake out of plateaus
        if t % 500 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Random mass‑transfer SA (accepts uphill moves with a decaying temperature)."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    # initialise temperature as a modest fraction of the current score
    T = cur_score * 0.05 + 1e-9
    cooling = 0.9999
    rng = np.random.default_rng()

    while time.time() < deadline:
        # source proportional to current mass, target inversely proportional
        src = rng.choice(n, p=a / a.sum())
        inv_weights = 1.0 / (a + 1e-12)
        inv_weights /= inv_weights.sum()
        tgt = rng.choice(n, p=inv_weights)
        if src == tgt:
            continue

        # move up to 30 % of the source mass
        frac = rng.random() * 0.30
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        a[src] -= delta
        a[tgt] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        if dE <= 0.0 or rng.random() < np.exp(-dE / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert
            a[tgt] -= delta
            a[src] += delta

        T *= cooling

        # very occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

    return best


# ----------------------------------------------------------------------
# Deterministic local polishing (focus on the current maximal entry)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """A cheap directional descent that repeatedly reduces the dominant peak."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # compute full convolution once
        size = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, n=size)
        conv_full = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
        k = int(np.argmax(conv_full))

        # contributions of each index to the maximal entry
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        src = int(np.argmax(contrib))
        # pick a low‑mass target that is not the source
        tgt_candidates = np.where(np.arange(n) != src)[0]
        if tgt_candidates.size == 0:
            break
        tgt = int(tgt_candidates[np.argmin(a[tgt_candidates])])

        delta = a[src] * 0.05  # move 5 % of source mass
        if delta <= 0.0:
            break

        a[src] -= delta
        a[tgt] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # stop if improvement stalls
        if delta < 1e-12:
            break

    return best


# ----------------------------------------------------------------------
# Random initialisers – a richer palette than in the reference code
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative vector on the ℓ¹‑simplex."""
    target = np.sqrt(2.0 * n)
    choice = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if choice == 0:                         # pure power‑law
        p = rng.uniform(0.1, 2.0)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                       # exponential decay
        lam = rng.exponential(scale=0.015) + 1e-6
        a = np.exp(-lam * idx)

    elif choice == 2:                       # mixed power‑law + exponential
        p = rng.uniform(0.1, 2.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                       # Dirichlet with random concentration
        alpha = rng.uniform(0.1, 5.0, size=n)
        a = rng.gamma(shape=alpha, scale=1.0)

    elif choice == 4:                       # log‑normal
        mu = rng.uniform(-2.0, 2.0)
        sigma = rng.uniform(0.2, 1.0)
        a = rng.lognormal(mean=mu, sigma=sigma, size=n)

    elif choice == 5:                       # beta‑shaped (peaked in the centre)
        alpha = rng.uniform(0.5, 5.0)
        beta = rng.uniform(0.5, 5.0)
        a = (idx + 1) ** alpha * (n - idx) ** beta

    else:                                   # sinusoidal squared plus tiny floor
        a = np.sin(np.pi * (idx + 1) / (n + 1)) ** 2 + 0.01

    # add a little Gaussian jitter and re‑project onto the simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, explores several vector lengths,
    refines the most promising seeds with Adam optimisation and simulated
    annealing, and finally performs a deterministic polishing step.
    The best sequence found (as a Python list) is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a global construction is already present, start from it
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, tuple, np.ndarray)):
        a = np.asarray(global_seq, dtype=float)
        n0 = a.size
        a = _project_to_simplex(a, np.sqrt(2.0 * n0))
        sc = _score(a)
        best_seq = a.copy()
        best_score = sc
        print(f"[{time.time():.1f}] Starting from global height_sequence_1 "
              f"(n={n0}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – broad random sampling over a range of lengths
    # ------------------------------------------------------------------
    lengths = [
        500, 800, 1200, 1500, 1800, 2100, 2500,
        3000, 3500, 4000, 4500, 5000, 6000,
        8000, 10000, 15000, 20000,
    ]

    # allocate ~30 % of the total budget to random sampling
    random_budget = budget_s * 0.30
    per_len_budget = max(2.0, random_budget / len(lengths))

    # store the top few seeds per length for later refinement
    top_per_len: dict[int, list[tuple[float, np.ndarray]]] = {}

    for n in lengths:
        if time.time() >= deadline:
            break
        phase_end = min(deadline, time.time() + per_len_budget)
        target_sum = np.sqrt(2.0 * n)
        best_local = []
        while time.time() < phase_end:
            a0 = _random_initial_sequence(n, rng)
            sc = _score(a0)
            # keep global best
            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} score={sc:.6f}")

            # maintain a tiny heap of the best seeds for this length
            if len(best_local) < 5:
                best_local.append((sc, a0.copy()))
                best_local.sort(key=lambda x: x[0])
            else:
                if sc < best_local[-1][0]:
                    best_local[-1] = (sc, a0.copy())
                    best_local.sort(key=lambda x: x[0])

        if best_local:
            top_per_len[n] = best_local

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation + simulated annealing on the elite seeds
    # ------------------------------------------------------------------
    # flatten the elite pool and sort by their random‑search score
    elite_pool: list[tuple[float, np.ndarray, int]] = []
    for n, seeds in top_per_len.items():
        for sc, arr in seeds:
            elite_pool.append((sc, arr, n))
    elite_pool.sort(key=lambda x: x[0])

    # give each elite candidate about 15–20 s (or the remaining budget)
    while elite_pool and time.time() < deadline - 5.0:
        sc0, a_seed, n = elite_pool.pop(0)
        sub_dead = min(deadline, time.time() + 15.0)

        # Adam refinement
        a_adam = _adam_optimize(a_seed, sub_dead)

        # Simulated annealing (uses whatever time is still left)
        a_sa = _simulated_annealing(a_adam, deadline)

        sc_final = _score(a_sa)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final deterministic polishing on the current global best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        a_refined = _local_refine(best_seq, deadline)
        sc_ref = _score(a_refined)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_refined.copy()
            print(f"[{time.time():.1f}] FINAL POLISHING improved to {sc_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety net – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_score = _score(best_seq)

    # Trim negligible trailing zeros for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Final sanity check with the provided evaluator
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        if np.isinf(final_val) or np.isnan(final_val):

    except Exception:  # fallback to uniform if something went wrong

        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        final_val = evaluate_sequence(best_seq.tolist())
        best_score = _max_convolution_fft(best_seq)

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
