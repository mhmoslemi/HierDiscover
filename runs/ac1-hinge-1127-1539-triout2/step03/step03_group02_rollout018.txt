The objective  

\[
\operatorname{score}(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{(\sum_i a_i)^2}
\]

is *scale invariant* – multiplying the whole vector by a constant does **not**
change the score.  Consequently we may fix the ℓ₁‑norm to the convenient value  

\[
\sum_i a_i = \sqrt{2\,n}\;.
\]

Under this normalisation the score is simply the maximal entry of the self‑
convolution, i.e. we have to **minimise the maximum convolution value**.

The previous implementation explored only *symmetric* “U‑shapes”.  
A far richer family is obtained by **monotone decreasing shapes**, e.g.

\[
a_i \propto (1-i/(n-1))^{p}\qquad\text{or}\qquad
a_i \propto e^{-c\,i/n}.
\]

These shapes pair a large value with a small one in the centre of the
convolution, which often lowers the peak.  We first sweep a few exponents
(`p`) and decay factors (`c`) to obtain a good initial vector, then polish it
with the same smoothed‑gradient / sub‑gradient routine that was used before.
Random “pair‑transfer’’ moves keep the search stochastic and help escape
plateaus.

The whole method respects the 1000 s budget, stores the best candidate seen
so far and finally returns it as a plain Python list.

```python
"""
search for a step‑function (non‑negative coefficient vector) with a low
max‑self‑convolution.  The search proceeds in three phases:

1. Parametric sweep over monotone decreasing shapes (power‑law and exponential)
   – the best shape becomes the starting point.
2. Smoothed gradient descent with momentum (log‑sum‑exp smoothing) while
   staying on the ℓ₁‑simplex (sum = √(2 n)).
3. Sub‑gradient refinement + a short random‑pair‑transfer polishing phase.

All heavy work uses FFT‑based convolutions, runs well within the 1000 s budget
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (non‑negative a)."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Simple projections on the ℓ₁‑simplex
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto
        {x | x_i ≥ 0, Σ_i x_i = target }.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small random perturbations (pair transfers)
# ----------------------------------------------------------------------


def random_pair_transfer(a: np.ndarray,
                         target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """Transfer a tiny amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Shape generators (monotone decreasing)
# ----------------------------------------------------------------------


def shape_power_decreasing(n: int,
                           target_sum: float,
                           exponent: float) -> np.ndarray:
    """
    a_i ∝ (1 - i/(n-1))^exponent  (decreasing power‑law shape).
    The vector is scaled to have the required ℓ₁‑norm.
    """
    idx = np.arange(n, dtype=np.float64)
    raw = (1.0 - idx / (n - 1)) ** exponent
    raw_sum = raw.sum()
    return raw * (target_sum / raw_sum)


def shape_exponential_decreasing(n: int,
                                 target_sum: float,
                                 decay: float) -> np.ndarray:
    """
    a_i ∝ exp( - decay * i / n )  (decreasing exponential shape).
    The decay parameter is a positive number; larger decay ⇒ faster drop.
    """
    idx = np.arange(n, dtype=np.float64)
    raw = np.exp(-decay * idx / n)
    raw_sum = raw.sum()
    return raw * (target_sum / raw_sum)


# ----------------------------------------------------------------------
# Smoothed gradient descent with momentum (log‑sum‑exp approx.)
# ----------------------------------------------------------------------


def smooth_gradient_step(a: np.ndarray,
                         target_sum: float,
                         eta: float,
                         tau_factor: float,
                         momentum: np.ndarray,
                         beta: float):
    """
    One iteration of smoothed gradient descent with momentum.
    Returns (new_a, new_momentum, current_max_conv).
    """
    n = a.shape[0]

    # --- convolution and smoothing weights ---
    conv = conv_self_fft(a)
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)

    w = np.exp((conv - max_conv) / tau)          # log‑sum‑exp weights
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # --- gradient (convolution of w with reversed a) ---
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

    # --- momentum / projection ---
    momentum = beta * momentum - eta * grad
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)

    return a_new, momentum, max_conv


# ----------------------------------------------------------------------
# Sub‑gradient step (exact max)
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray,
                     target_sum: float,
                     step: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient step on the true max‑convolution.
    Returns (new_a, max_conv_before_step).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()

    # indices attaining (approximately) the maximum
    eps = max(1e-12, 1e-9 * max_conv)
    mask = (conv >= max_conv - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        mask[conv.argmax()] = 1.0
        cnt = 1.0

    a_rev = a[::-1]
    L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
    grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, max_conv
```
