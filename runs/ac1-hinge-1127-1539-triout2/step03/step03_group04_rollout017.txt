```python
"""
Optimizes non‑negative step‑function heights to minimise the
evaluation function.  The algorithm consists of:

1. Coarse grid search over power‑law and exponential families for several
   lengths and exponents, keeping the best few candidates.
2. Projected Adam optimisation on a soft‑max approximation of the maximum
   convolution entry, with an annealed temperature parameter β.
3. Simulated‑annealing hill‑climbing that moves a small amount of mass
   from positions that contribute heavily to the current peak to positions
   that contribute little.  Worse moves are occasionally accepted to
   escape local minima.

All steps respect the simplex constraint Σ a_i = √(2 n) by Euclidean
projection onto the simplex.  The routine returns the best sequence found
before the time budget expires.
"""

import time
import numpy as np

def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _score(a: np.ndarray) -> float:
    """Exact objective for a sequence with sum fixed to √(2 n)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    # sum is already √(2 n) after projection, but keep for safety
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x>=0, sum x = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _gradient_softmax(a: np.ndarray, beta: float):
    """Soft‑max gradient of the maximum convolution entry."""
    n = a.shape[0]
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)
    # gradient = 2 * (a ⋆ p) where ⋆ denotes correlation (reverse convolution)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3 n - 2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b

def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected Adam optimisation on the soft‑max surrogate."""
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate scale (relative to target sum)
    base_lr = target_sum * 0.05     # empirically good for our scale

    beta = 1.0
    max_beta = 2048.0
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # occasional jitter to prevent stagnation
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a = a + jitter

        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD   it={it} beta={beta:.1f}  score={best_val:.6f}")

        # increase the soft‑max temperature gradually
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

    return best_a, best_val

def _simulated_annealing_hill_climb(a0: np.ndarray, deadline: float):
    """Move a small amount of mass from high‑contribution positions to low‑contribution ones.
       Accepts worsening moves with a temperature schedule."""
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    T = 0.02          # initial temperature
    T_min = 5e-6
    decay = 0.9995    # geometric decay per iteration

    while time.time() < deadline and T > T_min:
        # compute full convolution (small enough for typical n)
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # contribution weights for each index that participates in the current peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = k_max - idx
        weight = a[idx] * a[partner]          # contribution to conv[k_max]

        # donor – the index that contributes the most
        donor = int(idx[np.argmax(weight)])

        # receiver – pick a low‑contributing index (bottom 12 % of the weights)
        thr = np.percentile(weight, 12)
        low_idxs = idx[weight <= thr]
        if low_idxs.size == 0:
            # fallback: a random index different from donor
            receiver = int(rng.choice(np.delete(np.arange(n), donor)))
        else:
            receiver = int(rng.choice(low_idxs))

        if donor == receiver or a[donor] < 1e-12:
            # nothing sensible to move; try a random jitter instead
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, a.sum())   # keep the sum unchanged
            continue

        # move a random fraction of the donor's mass
        frac = rng.random() * 0.12          # up to 12 % per move
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        dE = new_val - cur_val
        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA   T={T:.6f}  score={best_val:.6f}")
        else:
            # revert the move
            a[receiver] -= delta
            a[donor] += delta

        # temperature decay
        T = max(T * decay, T_min)

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine performs a multi‑stage optimisation (grid search → Adam →
    simulated‑annealing hill‑climbing) and returns the best sequence found
    before the time budget expires."""
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Stage 1 – coarse exploration over lengths and families
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 1800, 2100, 2500, 3000, 3500, 4000, 5000]
    # parameter grids
    p_grid = np.linspace(0.35, 0.80, 46)    # step 0.01
    lam_grid = np.linspace(0.0005, 0.0040, 36)   # for exponential family

    # keep the top‑k candidates per length
    topk = 4
    candidates = []   # list of (score, array, n)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # power‑law family
        for p in p_grid:
            a = _power_law_sequence(n, p, target_sum)
            # small random jitter to break exact symmetry
            a += rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            val = _score(a)
            candidates.append((val, a.copy(), n))

        # exponential family
        for lam in lam_grid:
            a = _exponential_sequence(n, lam, target_sum)
            a += rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            val = _score(a)
            candidates.append((val, a.copy(), n))

        # a purely random start on the simplex
        rand_vec = rng.exponential(scale=1.0, size=n)
        rand_vec = _project_to_simplex(rand_vec, target_sum)
        val = _score(rand_vec)
        candidates.append((val, rand_vec.copy(), n))

    # keep only the most promising candidates
    candidates.sort(key=lambda x: x[0])
    candidates = candidates[: topk * len(lengths)]

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the shortlisted candidates
    # ------------------------------------------------------------------
    for base_score, a0, n in candidates:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # allocate a slice of the remaining budget (max 25 s per candidate)
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(25.0, remaining / 3.0)

        a_opt, val = _adam_optimize(a0, target_sum, sub_deadline)

        # ------------------------------------------------------------------
        # Stage 3 – simulated‑annealing hill‑climbing refinement
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_opt, val = _simulated_annealing_hill_climb(a_opt, deadline)

        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"** GLOBAL BEST **  n={n}  base_score={base_score:.6f}  "
                  f"final_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – simple constant sequence (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # Trim trailing zeros for compactness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Return the sequence as a plain python list
    return best_seq.tolist()

"""
Multi‑stage optimiser for the “max‑convolution / ℓ¹‑norm’’ benchmark.

The routine works in three phases:

1. **Coarse exploration** – for several lengths `n` we generate
   – power‑law sequences,
   – exponential decays,
   – a single random simplex point,
   – a greedy Sidon‑set construction.
   All candidates are normalised to Σ a = √(2 n) and the best few
   (according to the exact objective) are kept.

2. **Projected Adam optimisation** – on each retained candidate we run a
   projected Adam optimiser that minimises a soft‑max approximation of the
   maximal convolution entry.  The temperature `β` of the soft‑max is
   annealed, the iterate is projected onto the simplex after every step
   and a tiny random jitter is added occasionally to avoid stagnation.

3. **Simulated‑annealing hill climbing** – the best sequence from phase 2
   is refined by repeatedly moving a small fraction of mass from a
   “high‑contribution’’ index (the one that contributes most to the current
   peak of the convolution) to a low‑contribution one.  Worse moves are
   accepted with probability `exp(-Δ/ T)` where the temperature `T` is
   slowly cooled.

All steps respect the non‑negativity and fixed‑sum constraints.
The function returns the best sequence found before the time budget expires.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _max_convolution_fft_and_index(a: np.ndarray):
    """Return (max value, index) of the convolution a*a (FFT implementation)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _score(a: np.ndarray) -> float:
    """Exact objective for a sequence whose sum is (approximately) √(2 n)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of initial sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (normalised to target_sum)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i = exp(-lam·i)  (normalised to target_sum)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _sidon_sequence(n: int, target_sum: float, max_size: int = None) -> np.ndarray:
    """
    Greedy construction of a Sidon set (all unordered pairwise sums distinct).
    Equal weight is given to the selected positions.
    """
    sums = set()
    indices = []
    for i in range(n):
        if max_size is not None and len(indices) >= max_size:
            break
        conflict = False
        for j in indices:
            if i + j in sums:
                conflict = True
                break
        if not conflict:
            for j in indices:
                sums.add(i + j)
            indices.append(i)

    m = len(indices)
    if m == 0:
        return np.zeros(n, dtype=np.float64)

    a = np.zeros(n, dtype=np.float64)
    a[indices] = target_sum / m
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the maximal convolution entry
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return g = ∇[log‑sum‑exp_β(conv)] (shape n) and the full convolution."""
    n = a.shape[0]

    # full convolution a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over the convolution entries
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                # shape (2 n‑1,)

    # gradient g = 2·(a ⋆ p)   (⋆ = correlation = reverse‑convolution)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected Adam optimiser (soft‑max surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected Adam on the soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = _score(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate scale (relative to the simplex size)
    base_lr = target_sum * 0.04

    beta = 1.0                # soft‑max temperature
    max_beta = 2048.0
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # occasional jitter to shake the iterate
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = a + jitter

        # projection onto the simplex (keeps Σ a = target_sum)
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD   it={it} β={beta:.1f}  score={best_val:.6f}")

        # anneal the soft‑max temperature
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------


def _simulated_annealing_hill_climb(a0: np.ndarray, deadline: float):
    """
    Move a small fraction of mass from a high‑contributing index to a
    low‑contributing one.  Worsening moves are occasionally accepted
    according to a geometric cooling schedule.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    T = 0.02                 # initial temperature
    T_min = 5e-6
    decay = 0.9995

    while time.time() < deadline and T > T_min:
        # locate the current peak of the convolution (FFT version)
        max_val, k_max = _max_convolution_fft_and_index(a)

        # indices that actually take part in this peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = k_max - idx
        contrib = a[idx] * a[partner]                # contribution of each idx

        # donor – the index with the largest contribution
        donor = int(idx[np.argmax(contrib)])

        # receiver – a low‑contributing index (bottom ~12 % of the contributions)
        thr = np.percentile(contrib, 12)
        low = idx[contrib <= thr]
        if low.size == 0:                 # fallback: any index ≠ donor
            receiver = int(rng.choice(np.delete(np.arange(n), donor)))
        else:
            receiver = int(rng.choice(low))

        if donor == receiver or a[donor] < 1e-12:
            # nothing sensible to move; inject a small random jitter instead
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
            continue

        # move a random fraction (up to 12 %) of the donor's mass
        frac = rng.random() * 0.12
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        dE = new_val - cur_val
        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA   T={T:.6f}  score={best_val:.6f}")
        else:
            # reject – revert the move
            a[receiver] -= delta
            a[donor] += delta

        T = max(T * decay, T_min)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs a short grid search, refines the most promising
    candidates with projected Adam optimisation and finishes with a
    simulated‑annealing hill‑climbing phase.  The best sequence found
    before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Phase 1 – coarse grid over a few families and lengths
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 1800, 2100, 2500, 3000, 3500, 4000, 5000]
    p_grid = np.linspace(0.35, 0.80, 46)          # step ≈0.01
    lam_grid = np.linspace(0.0005, 0.0040, 36)   # exponential rates
    candidates = []                               # (score, array, n)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # power‑law family
        for p in p_grid:
            a = _power_law_sequence(n, p, target_sum)
            a += rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            val = _score(a)
            candidates.append((val, a.copy(), n))

        # exponential family
        for lam in lam_grid:
            a = _exponential_sequence(n, lam, target_sum)
            a += rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            val = _score(a)
            candidates.append((val, a.copy(), n))

        # a random simplex point
        rnd = rng.exponential(scale=1.0, size=n)
        rnd = _project_to_simplex(rnd, target_sum)
        candidates.append((_score(rnd), rnd.copy(), n))

        # greedy Sidon set (moderate size)
        sid = _sidon_sequence(n, target_sum, max_size=int(0.6 * np.sqrt(2 * n)))
        candidates.append((_score(sid), sid.copy(), n))

    # retain only the globally best few candidates
    candidates.sort(key=lambda x: x[0])
    top_global = 12                       # keep at most 12 candidates total
    candidates = candidates[:top_global]

    # ------------------------------------------------------------------
    # Phase 2 – projected Adam optimisation on the shortlisted candidates
    # ------------------------------------------------------------------
    for base_score, a0, n in candidates:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        remaining = deadline - time.time()
        # give each candidate up to 20 s but never more than a third of the rest
        sub_deadline = time.time() + min(20.0, remaining * 0.33)

        a_opt, val = _adam_optimize(a0, target_sum, sub_deadline)

        if val < best_score:
            best_score = val
            best_seq = a_opt.copy()
            print(f"** GLOBAL BEST (after Adam) **  n={n}  base={base_score:.6f}  "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill climbing on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _simulated_annealing_hill_climb(best_seq, deadline)
        print(f"** GLOBAL BEST (final) **  len={len(best_seq)}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – simple constant sequence (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # Trim trailing zeros for compactness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
