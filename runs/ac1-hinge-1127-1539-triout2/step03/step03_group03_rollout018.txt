def exponential_sequence(n, k_factor):
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    # compute c using formula to ensure sum = S
    # safe compute using expm1
    one_minus_q = -np.expm1(-lam) # = 1 - exp(-lam)
    one_minus_qn = -np.expm1(-k_factor)  # = 1 - exp(-k_factor)
    c = S * one_minus_q / one_minus_qn
    # compute a_i
    i = np.arange(n)
    a = c * np.power(q, i)
    # correct potential error in sum
    sum_a = a.sum()
    if sum_a != 0:
        a *= (S / sum_a)
    return a

def _exponential_sequence(n, k_factor, rng):
    """
    Generate a decreasing exponential-like sequence of length n,
    with parameter k_factor controlling the decay.
    The sequence is normalized so that its L1 norm equals sqrt(2*n).
    """
    S = np.sqrt(2.0 * n)
    lam = k_factor / n
    q = np.exp(-lam)
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - np.exp(-k_factor)  # since q**n = e^{-k_fact}
    c = S * one_minus_q / one_minus_qn
    # small jitter to avoid exact duplicates
    a = c * (q ** np.arange(n))
    # Re-normalize exactly to S
    sum_a = a.sum()
    if sum_a != 0.0:
        a *= (S / sum_a)
    return a

    max_no_improve = 3000
    no_improve_counter = 0

    while time.time() < deadline:
        conv = _full_convolution(a)
        cur_max = conv.max()
        # get indices of max (maybe several)
        k_idxs = np.flatnonzero(conv >= cur_max * 0.999999)  # close to max
        # pick random one among them
        k = rng.choice(k_idxs)

        # compute contributions for this k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        contributions = a[i_range] * a[j_range]

        # Identify source index (i_high) with largest contribution and nonzero a_i
        idx_high = np.argmax(contributions)
        i_high = i_range[idx_high]

        # Identify target index (i_low) with smallest contribution
        idx_low = np.argmin(contributions)
        i_low = i_range[idx_low]

        if a[i_high] <= 0:
            # none to move, skip
            continue

        delta = eps * a[i_high] * rng.random()
        if delta <= 0:
            continue

        a_candidate = a.copy()
        a_candidate[i_high] -= delta
        a_candidate[i_low] += delta
        # Ensure non-negativity and project onto simplex
        a_candidate = _project_simplex(a_candidate, S)

        new_max = np.max(_full_convolution(a_candidate))

        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(cur_max * 0.02, 1e-12)):
            a = a_candidate
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                no_improve_counter = 0
                continue

        no_improve_counter += 1
        if no_improve_counter > max_no_improve:
            break

    return best_a, best_score

    The gradient of the smoothed objective can be obtained efficiently
    by FFTs.  After each step the vector is projected back onto the
    simplex.

3.  **Mass‑swap moves** – a mixture of random swaps (to explore) and a
    cheap targeted move that explicitly reduces the current maximal
    convolution entry.

The algorithm runs for the whole time budget (≤1000 s), keeps the
overall best sequence and finally returns it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the rho index
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    # next power‑of‑two for speed
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_j = 2 Σ_i w_{i+j}·a_i.
    a has length n, w has length 2·n‑1.
    """
    n = a.shape[0]
    # correlation a ⋆ w
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # slice that corresponds to Σ_i w_{i+j}·a_i
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max helpers
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights from a convolution vector:
        w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """
    One projected gradient step on the smoothed max‑norm.
    Returns the updated sequence and the *exact* maximal convolution entry.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    w = _smooth_weights(conv, alpha)
    grad = _conv_with_weights(a, w)
    a_new = a - lr * grad
    # project onto the simplex Σa = √(2·n)
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


# ----------------------------------------------------------------------
# Exponential warm‑start
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, k_factor: float) -> np.ndarray:
    """
    Generate a decreasing exponential sequence of length n.
    k_factor controls the decay (larger → faster decay).
    The result is projected onto the simplex Σa = √(2·n).
    """
    S = np.sqrt(2.0 * n)
    lam = k_factor / n                     # λ = k_factor / n
    q = np.exp(-lam)                       # q = e^{-λ}
    # exact normalisation (avoids loss of precision for tiny λ)
    one_minus_q = 1.0 - q
    one_minus_qn = 1.0 - np.exp(-k_factor)   # = 1 - q**n
    c = S * one_minus_q / one_minus_qn
    a = c * (q ** np.arange(n))
    # final projection guarantees the exact sum
    a = _project_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Local optimiser (gradient + targeted swaps)
# ----------------------------------------------------------------------


def _local_optimize(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using a mixture of smooth‑max gradient steps
    and inexpensive targeted mass‑swap moves that explicitly lower the
    current maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # best ever seen during the run
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters (chosen empirically)
    lr = 0.5               # initial learning rate
    lr_decay = 0.9999
    alpha = 0.5            # temperature of the soft‑max
    alpha_mul = 1.005      # speed of annealing
    swap_prob = 0.15       # probability of a random swap per iteration
    swap_frac = 0.04       # fraction of source mass moved in a swap
    target_prob = 0.2      # chance to try the cheap targeted move
    eps_target = 0.03      # mass fraction used in the targeted move

    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- gradient descent step ----
        a, cur_max = _gradient_step(a, lr, alpha)

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # ---- occasional random mass‑swap (simulated‑annealing flavour) ----
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                # Metropolis acceptance
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                             max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- cheap targeted move aimed at the current maximal index ----
        if rng.random() < target_prob:
            conv = _full_convolution(a)
            k_max = int(np.argmax(conv))

            # indices that contribute to conv[k_max]
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contributions = a[i_range] * a[j_range]

            # source = index with the largest contribution
            src_idx = np.argmax(contributions)
            i_src = i_range[src_idx]

            # target = index with the smallest contribution
            tgt_idx = np.argmin(contributions)
            i_tgt = i_range[tgt_idx]

            if a[i_src] > 0 and i_src != i_tgt:
                delta = eps_target * a[i_src] * rng.random()
                cand = a.copy()
                cand[i_src] -= delta
                cand[i_tgt] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max <= cur_max or rng.random() < np.exp(-(cand_max - cur_max) /
                                                             max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # ---- schedule updates ----
        lr *= lr_decay
        alpha *= alpha_mul

        # early stop if stagnated for a long time
        if it - last_improve > 5000:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a sequence that minimises evaluate_sequence().
    The routine performs a short exponential warm‑start followed by a
    long projected‑gradient optimisation with occasional swaps.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1) initialise with a cheap exponential warm‑start (few random tries)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # try a handful of different lengths / decay rates quickly
    for n in (800, 1200, 2000, 3000, 4000):
        if time.time() >= deadline:
            break
        for k_factor in np.linspace(0.6, 2.5, 10):
            a = _exponential_sequence(n, k_factor)
            cur_score = np.max(_full_convolution(a))   # exact score, sum is already √(2·n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[warm] n={n:4d} k={k_factor:4.2f} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2) if a previous solution exists, bring it into the pool
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            if prev.size > 0:
                # project onto the correct simplex (might have a different length)
                n = prev.size
                S = np.sqrt(2.0 * n)
                prev = _project_simplex(prev, S)
                cur = np.max(_full_convolution(prev))
                if cur < best_score:
                    best_score = cur
                    best_seq = prev.copy()
                    print(f"[prev] using height_sequence_1, score={cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) main optimisation phase – run until the budget runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (biased towards lengths that have performed well)
        if best_seq is not None and rng.random() < 0.6:
            n = best_seq.size
        else:
            n = rng.integers(800, 5001)

        # initialise either from the current best (with a tiny jitter) or
        # from a fresh exponential sample
        if best_seq is not None and best_seq.size == n and rng.random() < 0.5:
            a0 = best_seq.copy()
            jitter = rng.normal(scale=0.001 * a0.mean(), size=n)
            a0 = np.clip(a0 + jitter, 0.0, None)
            a0 = _project_simplex(a0, np.sqrt(2.0 * n))
        else:
            k_factor = rng.uniform(0.6, 2.5)
            a0 = _exponential_sequence(n, k_factor)

        # give the optimiser a time slice (≈ 30 s or the remaining time)
        slice_deadline = min(deadline, time.time() + 30.0)

        a_opt, score_opt = _local_optimize(a0, slice_deadline, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[global] new best n={n:4d} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final sanity check via the external black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of length 1000
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  length = {len(final_seq)}")
    return final_seq
