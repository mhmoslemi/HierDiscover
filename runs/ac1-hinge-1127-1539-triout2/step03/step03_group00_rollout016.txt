    target = np.sqrt(2*len(a))
    it = 0
    while time.time() < deadline:
        b = convolve_fft(a) # returns full conv
        k = int(np.argmax(b))
        # compute gradient g
        g = np.zeros_like(a)
        # j indices where idx - j in [0,n-1]
        lo = max(0, k - (len(a)-1))
        hi = min(len(a)-1, k)
        # for j from lo to hi: g[j] = 2 * a[k - j]

"""
search_step_function.py

Multi‑stage optimizer for a non‑negative step‑function that minimizes

    evaluate_sequence(a) = 2·n·max(a∗a) / (sum(a)^2)

The ℓ¹‑normalisation sum(a)=√(2·n) makes the objective exactly
max(a∗a).  The algorithm proceeds in four stages:

1.  Random initialisation on several parametric families.
2.  Annealed projected‑gradient descent on a smooth (log‑sum‑exp)
    approximation of the max‑convolution.  The “temperature’’ β is
    increased up to ~1e5, providing a close approximation of the true
    maximum.
3.  Sub‑gradient descent on the exact max‑convolution (a simple step
    in the direction that reduces the current worst entry).
4.  Optional simulated‑annealing refinement that swaps small amounts
    of mass between coordinates.

All updates are projected onto the ℓ¹‑simplex
{ x≥0 , Σx = √(2·n) } (Duchi et al. 2008).  The routine runs for at most
``budget_s`` seconds and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s }.
    (Duchi et al. 2008)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth max‑convolution utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, conv) where
        conv = a*a (full convolution, length 2·n‑1)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·conv_k) )
             = 2·(a ⋆ p)   with p_k ∝ exp(β·conv_k)
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))          # stable exponentials
    p = exps / np.sum(exps)

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1                # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


def _gradient_descent_opt(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          max_beta: float = 1e5,
                          max_iter: int = 10_000) -> np.ndarray:
    """
    Annealed projected‑gradient descent on the smooth log‑sum‑exp
    approximation of max(a∗a).  Momentum, adaptive step size and occasional
    jitter are used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    step_base = target_sum * 0.22          # a bit larger than the original
    momentum = np.zeros_like(a)
    rho = 0.9                               # momentum decay

    it = 0
    while time.time() < deadline and it < max_iter:
        grad, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         max_iter: int = 5_000) -> np.ndarray:
    """
    Sub‑gradient descent on the true max‑convolution.
    At each step the index k* attaining the maximum is identified and
    the gradient of b_k* (=2·shift of a) is followed in a decreasing step
    size.  Projection onto the simplex maintains feasibility.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    it = 0
    cur_score = _max_convolution_fft(a)

    while time.time() < deadline and it < max_iter:
        # full convolution (FFT)
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        k = int(np.argmax(conv))

        # gradient of b_k  :  g_j = 2·a_{k-j} (zero outside range)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break

        # step size decreasing like 1/√t
        eta = target * 0.12 / (norm_g * np.sqrt(it + 1))

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target)

        new_score = _max_convolution_fft(a_new)

        # accept if we improve; otherwise keep the current point
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
        else:
            # occasional aggressive move when stuck
            if it % 400 == 0:
                a = a_new
                cur_score = new_score

        it += 1
    return a


# ----------------------------------------------------------------------
# Simulated annealing (light‑weight)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iter: int = 200_000) -> np.ndarray:
    """
    Simple mass‑transfer SA that keeps the best solution seen.
    Moves a random fraction of mass from i to j; uphill moves are
    accepted with probability exp(-Δ/T).  Temperature decays geometrically.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_fft(a)

    # initialise temperature as a small fraction of current score
    T = best_score * 0.02 + 1e-8
    cooling = 0.9997

    n = a.size
    it = 0
    while time.time() < deadline and it < max_iter:
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            it += 1
            continue
        j = rng.integers(0, n)
        if i == j:
            it += 1
            continue

        frac = rng.random() * 0.13        # up to 13 % of a[i]
        delta = a[i] * frac
        if delta <= 0.0:
            it += 1
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _max_convolution_fft(a)
        dE = new_score - best_score

        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_score:.6f}")
        else:
            # revert move
            a[j] -= delta
            a[i] += delta

        T *= cooling
        it += 1

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur = _max_convolution_fft(a)
            if cur < best_score:
                best_score = cur
                best = a.copy()
                print(f"[{time.time():.1f}] SA global jitter: {best_score:.6f}")

    return best

# Random initialiser (various families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random non‑negative vector on the ℓ¹‑simplex."""
    target = np.sqrt(2.0 * n)
    choice = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if choice == 0:                         # pure power‑law
        p = rng.uniform(0.12, 1.2)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:                       # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif choice == 2:                       # mixed power‑law + exponential
        p = rng.uniform(0.12, 1.2)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:                       # two‑exponential mixture
        lam1 = rng.exponential(scale=0.015)
        lam2 = rng.exponential(scale=0.040)
        w = rng.random()
        a = w * np.exp(-lam1 * idx) + (1.0 - w) * np.exp(-lam2 * idx)

    elif choice == 4:                       # Dirichlet (uniform simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    else:                                   # uniform plus tiny noise
        a = np.full(n, target / n)

    # small Gaussian jitter and re‑project
    a += rng.normal(scale=target * 0.0015, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Upscaling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upscale(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a vector to a larger length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Main entry point

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    competition objective.  The routine runs for at most ``budget_s``
    seconds, exploring random starts, a long annealed gradient phase,
    sub‑gradient polishing, and a final short simulated‑annealing sweep.
    The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # safety margin before hard deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a global construction exists, use it as a warm start (up‑scaled)
    # ------------------------------------------------------------------
    init_global = globals().get("height_sequence_1")
    base_n = 8000                     # main working length
    target_sum = np.sqrt(2.0 * base_n)

    if isinstance(init_global, (list, tuple, np.ndarray)):
        a = np.asarray(init_global, dtype=float)
        a = _upscale(a, base_n)
        score = _max_convolution_fft(a)
        best_seq = a.copy()
        best_score = score
        print(f"[{time.time():.1f}] Starting from global up‑scaled "
              f"length={base_n} score={score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – random restarts with short gradient runs
    # ------------------------------------------------------------------
    restarts = 30
    for r in range(restarts):
        if time.time() >= deadline:
            break
        a0 = _random_initial_sequence(base_n, rng)
        # short gradient run (few hundred iterations)
        a_gd = _gradient_descent_opt(a0, deadline, rng, max_iter=800)
        sc = _max_convolution_fft(a_gd)
        if sc < best_score:
            best_score = sc
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] New global best (rand) r={r} "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long annealed gradient descent starting from best so far
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_long = _gradient_descent_opt(best_seq.copy(), deadline, rng,
                                       max_iter=12_000)
        sc = _max_convolution_fft(a_long)
        if sc < best_score:
            best_score = sc
            best_seq = a_long.copy()
            print(f"[{time.time():.1f}] After long GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – sub‑gradient refinement on the true maximum
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_sub = _subgradient_descent(best_seq.copy(), deadline,
                                     max_iter=8_000)
        sc = _max_convolution_fft(a_sub)
        if sc < best_score:
            best_score = sc
            best_seq = a_sub.copy()
            print(f"[{time.time():.1f}] After sub‑grad best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final simulated‑annealing polish (short, if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_sa = _simulated_annealing(best_seq.copy(), deadline, rng,
                                    max_iter=150_000)
        sc = _max_convolution_fft(a_sa)
        if sc < best_score:
            best_score = sc
            best_seq = a_sa.copy()
            print(f"[{time.time():.1f}] Final SA improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_score = _max_convolution_fft(best_seq)

    # Trim negligible trailing zeros for neatness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f} "
          f"length = {best_seq.size}")
    return best_seq.tolist()
