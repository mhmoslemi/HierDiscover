def _generate_sequence(n, rng):
    shape_type = rng.choice(shape_types)
    if shape_type == 'power':
        p = rng.uniform(0.2, 1.5)
        a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
    elif shape_type == 'exp':
        lam = rng.uniform(0.0005, 0.01) # small lambda
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
    elif shape_type == 'linear':
        m = rng.integers(int(n*0.3), n+1) # support length
        slope = 1.0 / m # will be normalized later
        a = np.maximum(0.0, 1.0 - np.arange(n)/m) # linear decreasing to zero at m
    elif shape_type == 'quadratic':
        m = rng.integers(int(n*0.3), n+1)
        x = np.arange(n, dtype=np.float64) / m
        a = np.maximum(0.0, 1.0 - x**2)
    elif shape_type == 'beta':
        alpha = rng.uniform(-0.5, 3.0)
        beta = rng.uniform(-0.5, 3.0)
        i = np.arange(1, n+1, dtype=np.float64)
        a = (i**alpha) * ((n - i + 1)**beta)
    elif shape_type == 'gauss':
        mu = rng.uniform(0, n*0.5)  # shift left side maybe
        sigma = rng.uniform(n*0.05, n*0.3)
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((i - mu)/sigma)**2)
    else:  # fallback uniform
        a = np.ones(n)
    # Clip negative zeros
    a = np.maximum(a, 0.0)
    # Normalize to target sum via projection onto simplex (Euclidean)
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    a += rng.normal(scale=target_sum*1e-4, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a

"""
A fast multi‑stage stochastic optimizer for the “max‑convolution / ℓ¹‑norm’’
benchmark.

Stage 0 – Random shape pool.
    Generate many short candidate sequences of various analytic shapes
    (power‑law, exponential, linear, quadratic, beta‑like, Gaussian) with
    random parameters, project them onto the ℓ¹‑simplex
    Σa_i = √(2·n) and keep the best few.

Stage 1 – Sub‑gradient descent on the exact ℓ∞‑objective.
    Using the current maximal convolution entry k* we form the exact
    sub‑gradient g_i = 2·a_{k*‑i} (zero outside the valid range), take a
    modest step, then Euclidean‑project back onto the simplex.  The step
    size is annealed automatically.

Stage 2 – Soft‑max (log‑sum‑exp) gradient descent.
    A smooth approximation of the maximum is minimised by projected Adam‑type
    updates; this gives a more global search and helps escape shallow basins.

Stage 3 – Guided hill‑climbing.
    Identify the “sensitivity’’ s_i = a_{k*‑i} of each coordinate to the peak,
    move a small fraction of mass from high‑sensitivity (large s_i·a_i)
    positions to low‑sensitivity ones, and keep improving moves.

All stages are time‑boxed so that the routine always returns the best
sequence found before the overall budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of the linear convolution a*a (FFT version)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real output)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast version of the competition objective (exact after normalisation)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Random shape generator (Stage 0)
# ----------------------------------------------------------------------


def _random_shape_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random non‑negative sequence of length n from a family of shapes."""
    target = np.sqrt(2.0 * n)

    shape = rng.choice(
        ["power", "exp", "linear", "quadratic", "beta", "gauss", "uniform"]
    )
    i = np.arange(n, dtype=np.float64)

    if shape == "power":
        p = rng.uniform(0.25, 1.6)            # typical optimum around 0.5‑0.7
        a = (i + 1.0) ** (-p)

    elif shape == "exp":
        lam = rng.uniform(0.0004, 0.02)      # slower decay → larger support
        a = np.exp(-lam * i)

    elif shape == "linear":
        # linear decay to zero over a random support length m
        m = rng.integers(int(0.3 * n), n + 1)
        a = np.maximum(0.0, 1.0 - i / m)

    elif shape == "quadratic":
        m = rng.integers(int(0.3 * n), n + 1)
        x = i / m
        a = np.maximum(0.0, 1.0 - x ** 2)

    elif shape == "beta":
        # beta‑type shape: (i+1)^α * (n‑i)^β   (α,β ≥ 0)
        alpha = rng.uniform(0.0, 3.0)
        beta = rng.uniform(0.0, 3.0)
        a = (i + 1.0) ** alpha * (n - i) ** beta

    elif shape == "gauss":
        mu = rng.uniform(0.0, 0.5 * n)            # centre biased towards the left
        sigma = rng.uniform(0.05 * n, 0.3 * n)
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)

    else:  # uniform (fallback)
        a = np.ones(n, dtype=np.float64)

    # make sure we stay non‑negative
    a = np.maximum(a, 0.0)

    # project onto the ℓ¹‑simplex (this also normalises the sum)
    a = _project_to_simplex(a, target)

    # a tiny jitter to break possible symmetries, then re‑project
    a += rng.normal(scale=target * 1e-4, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)

    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (Stage 1)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the ℓ∞‑convolution objective.
    Uses the exact location of the maximum to form a sub‑gradient.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    # initial step size – a fraction of the total mass
    step_factor = target_sum * 0.6
    rng = np.random.default_rng()

    while time.time() < deadline:
        # full convolution (FFT) to locate the peak index
        conv = _full_convolution_fft(a)
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2 * a_{k - i}  (zero outside support)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start <= i_end:
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            idx_j = k - idx_i
            g[idx_i] = 2.0 * a[idx_j]

        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_factor / norm_g
            a -= lr * g

        # project back onto the simplex (keeps Σa = √(2n))
        a = _project_to_simplex(a, target_sum)

        # occasional tiny random jitter to avoid stagnation
        if rng.random() < 0.01:
            a += rng.normal(scale=target_sum * 0.001, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        current_val = _score(a)
        if current_val < best_val:
            best_val = current_val
            best_a = a.copy()
            # slowly shrink the step factor when we improve
            step_factor *= 0.93
            print(f"[{time.time():.1f}] SUB  new best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient descent (Stage 2)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of  log‑sum‑exp_{β}(a*a) .
    Returns (g, conv) where conv is the full convolution.
    """
    n = a.shape[0]

    # full convolution via FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)                     # shape (2·n‑1,)

    # gradient = 2·(a ⋆ p)   (⋆ = cross‑correlation)
    a_rev = a[::-1]
    size2 = 1 << ((n + (2 * n - 1) - 1).bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    corr_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[: 3 * n - 2]

    g = 2.0 * corr_rev[n - 1 : n - 1 + n]
    return g, conv


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected Adam‑style descent on the soft‑max approximation.
    The temperature β is increased (annealed) while the step size adapts.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    beta = 0.5
    beta_max = 256.0
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = 0.4 * target / (np.sqrt(v_hat) + eps)   # adaptive per‑coordinate step

        a -= lr * m_hat

        # projection onto the simplex
        a = _project_to_simplex(a, target)

        # occasional jitter to avoid plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.1f}] AD   β={beta:.2f}  best={best_val:.6f}")

        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * 1.45, beta_max)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (Stage 3)
# ----------------------------------------------------------------------


def _guided_hill_climb(a: np.ndarray, deadline: float):
    """Mass‑redistribution hill‑climbing that focuses on the current peak."""
    n = a.size
    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.12          # up to 12 % of a donor’s mass can be moved per step

    while time.time() < deadline:
        conv = np.convolve(a, a)                # O(n²) but n stays modest
        k_max = int(np.argmax(conv))

        # sensitivity s_i = a_{k_max‑i} (0 if out of range)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        sens[lo : hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        # weight = contribution of each coordinate to the peak
        weight = sens * a
        if weight.max() == 0.0:
            break

        donor = int(np.argmax(weight))

        # pick a receiver from the 15 % lowest weighted positions
        low_thr = np.percentile(weight, 15)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            break
        receiver = int(rng.choice(candidates))
        if donor == receiver or a[donor] < 1e-14:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] HC   improved {best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine builds a diverse pool of analytically‑generated candidates,
    refines each one with sub‑gradient → soft‑max → hill‑climb phases and
    returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0        # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Stage 0 – generate a pool of random shapes
    # ------------------------------------------------------------------
    lengths = [800, 1500, 2500, 3500, 5000, 8000]
    pool = []                               # (score, array, n)
    max_pool = 30

    for n in lengths:
        target = np.sqrt(2.0 * n)
        # number of random draws per length (adaptable to remaining time)
        draws = 30
        for _ in range(draws):
            if time.time() >= deadline:
                break
            a = _random_shape_sequence(n, rng)
            sc = _score(a)
            if len(pool) < max_pool:
                pool.append((sc, a, n))
            else:
                # keep only the best ``max_pool`` candidates overall
                worst_idx = np.argmax([p[0] for p in pool])
                if sc < pool[worst_idx][0]:
                    pool[worst_idx] = (sc, a, n)

    # sort pool by current score (ascending)
    pool.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Stage 1‑3 – iterative refinement of the most promising candidates
    # ------------------------------------------------------------------
    # we allocate a bounded amount of time to each candidate; the first few
    # get a larger share.
    for rank, (base_score, a0, n) in enumerate(pool):
        if time.time() >= deadline:
            break

        # allocate time proportionally (more for higher‑ranked candidates)
        remaining = deadline - time.time()
        # are we at the very beginning? give 45 s, otherwise split the rest.
        if rank == 0:
            cand_deadline = time.time() + min(45.0, remaining * 0.5)
        else:
            cand_deadline = time.time() + min(20.0, remaining * 0.3)

        # Stage 1 – exact sub‑gradient descent
        a_sub, val_sub = _subgradient_descent(a0, cand_deadline)
        if time.time() >= deadline:
            # time ran out, keep what we have
            final_a, final_val = a_sub, val_sub
        else:
            # Stage 2 – annealed soft‑max descent
            a_soft, val_soft = _annealed_gradient_descent(a_sub, deadline)
            # Stage 3 – guided hill‑climb (use whatever time remains)
            a_final, val_final = _guided_hill_climb(a_soft, deadline)

            final_a, final_val = a_final, val_final

        if final_val < best_score:
            best_score = final_val
            best_seq = final_a.copy()
            print(
                f"** GLOBAL BEST **  n={n}  init={base_score:.6f}  best={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.ones(n, dtype=np.float64)
        a *= np.sqrt(2.0 * n) / n
        best_seq = a

    # Trim trailing zeros for a tidy output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
