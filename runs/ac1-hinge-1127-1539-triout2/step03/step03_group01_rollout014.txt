def adam_optimizer(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 0.5  # base learning rate
    tau = None
    cur_max = np.max(full_convolution_fft(a))
    step = 0
    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        step += 1
        # compute conv
        b = full_convolution_fft(a)
        cur_max = np.max(b)
        # update tau
        if tau is None:
            tau = cur_max
        else:
            tau *= 0.995  # or schedule
            if tau < 1e-6:
                tau = 1e-6

        # compute softmax weights
        max_b = cur_max
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)

        # gradient: g_i = 2 * (w * a)_i via cross-correlation
        # compute cross-correlation using FFT
        a_rev = a[::-1]
        mlen = len(w) + n - 1
        fft_len = 1 << ((mlen - 1).bit_length())
        Fw = np.fft.rfft(w, n=fft_len)
        Fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:mlen]
        g = 2.0 * conv[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        lr = eta0 / (np.sqrt(v_hat) + eps)
        a_new = a - lr * m_hat
        # project
        a = project_simplex(a_new, S)

        # recompute max after projection
        b = full_convolution_fft(a)
        cur_max = np.max(b)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # occasional random mass swap (targeted)
        if rng.random() < 0.08:
            # pick i with largest a_i * a_{k*-i}
            k_star = np.argmax(b)
            # compute contributions for each i: c_i = a_i * a_{k*-i} if valid
            idx = np.arange(n)
            partner = k_star - idx
            valid = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(a)
            contrib[valid] = a[valid] * a[partner[valid]]
            # also if i==partner (self) add a_i**2
            # already included because a_i*a_i.
            # Choose source i proportional to contrib (more heavy)
            if contrib.sum() > 0:
                probs = contrib / contrib.sum()
                i = rng.choice(n, p=probs)
            else:
                i = rng.integers(n)
            # choose destination j random not i
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            delta = 0.05 * a[i]  # move 5% of mass
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # re-project to simplex (almost unchanged)
            a_candidate = project_simplex(a_candidate, S)

            b_candidate = full_convolution_fft(a_candidate)
            cand_max = np.max(b_candidate)
            # accept if improves or with SA
            if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max)/ (0.01 * cur_max)):
                a = a_candidate
                cur_max = cand_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
        # occasionally jitter
        if rng.random() < 0.01:
            a = project_simplex(a + rng.normal(scale=0.01 * a.mean(), size=n), S)

    return best_a, best_max

"""
Propose a step‑function sequence that minimizes
   2·n·max(a*a) / (sum(a)**2)

The optimizer is a two‑phase method:

1)  A long Adam‑based projected gradient descent on a smooth
    approximation of the ℓ∞‑norm of the self‑convolution.
    The loss is τ·log ∑exp(b/τ) where b = a*a.
    After each gradient step we Euclidean‑project onto the
    simplex {a≥0, Σa = √(2n)}.

2)  An optional refinement using a linear program:
    for the current vector f we maximise Σg_j under the
    constraint f*g ≤ M (where M = max(f*f)).  If the optimum
    has a larger ℓ₁‑norm than √(2n) we scale it back and mix
    it with f.  This is precisely the “LP‑step” described in the
    statement and can shave a few hundredths off the bound.

Both phases respect the time budget (default 1000 s).  The best
sequence found is stored in the global variable `height_sequence_1`
so that a later run can start from it.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum(x)=S}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full (linear) convolution using a real FFT – returns length 2n‑1 array
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Initialise a feasible vector (non‑negative, ℓ₁ = √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    S = math.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # tiny jitter so the restart is not identical
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)            # Dirichlet(1) – uniform on simplex before projection
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      b: np.ndarray,
                      tau: float) -> np.ndarray:
    """Return sub‑gradient of τ·log∑exp(b/τ) w.r.t. a."""
    n = a.shape[0]
    max_b = np.max(b)

    if tau <= 1e-12:                # hard‑max (sub‑gradient)
        k = np.argmax(b)           # index of the maximal entry
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # the diagonal term appears only once, the expression above already counts it twice;
        # we correct by halving it when k is even (i = k/2)
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]        # because the product a_d*a_d appears only once
        return g

    # ----- smooth case -----
    diff = b - max_b                # for numerical stability
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)         # soft‑max weights, shape (2n‑1,)

    # gradient = 2 * cross‑correlation of w with a
    a_rev = a[::-1]                # reverse a for cross‑correlation via convolution
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]    # slice of length n
    return g


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the soft‑max loss
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run Adam until the supplied deadline. Return the best a seen and
    its maximal self‑convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.85
    beta2 = 0.995
    eps = 1e-8
    base_lr = 0.8                     # base learning rate (will be scaled by the Adam denominator)

    # temperature for the soft‑max (τ)
    tau = None
    # keep a record of the best vector found
    best_a = a.copy()
    best_val = np.max(_full_convolution_fft(a))

    step = 0
    while time.time() < deadline:
        step += 1

        # convolution and current max
        b = _full_convolution_fft(a)
        cur_max = np.max(b)

        # initialise / update τ
        if tau is None:
            tau = cur_max
        else:
            tau = max(tau * 0.997, 1e-6)       # exponential cooling, never goes below 1e‑6

        # gradient of the smooth ℓ∞‑approximation
        g = _softmax_gradient(a, b, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        step_size = base_lr / (np.sqrt(v_hat) + eps)
        a_new = a - step_size * m_hat
        a = _project_to_simplex(a_new, S)

        # record best
        new_val = np.max(_full_convolution_fft(a))
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()

        # --------------------------------------------------------------
        # occasional *targeted* mass‑swap (simulated annealing)
        # --------------------------------------------------------------
        if rng.random() < 0.07:
            # choose a source index proportional to its contribution to the peak
            k_star = np.argmax(b)
            idx = np.arange(n)
            partner = k_star - idx
            valid = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(a)
            contrib[valid] = a[valid] * a[partner[valid]]
            if contrib.sum() > 0:
                probs = contrib / contrib.sum()
                i = rng.choice(n, p=probs)
            else:
                i = rng.integers(n)

            # destination – pick a random different index
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

            delta = 0.06 * a[i]          # move 6 % of the source mass
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, S)

            cand_max = np.max(_full_convolution_fft(cand))
            accept = False
            if cand_max <= cur_max:
                accept = True
            else:
                # annealing temperature proportional to current value
                temp = cur_max * 0.08
                if rng.random() < math.exp(-(cand_max - cur_max) / max(temp, 1e-12)):
                    accept = True
            if accept:
                a = cand
                cur_max = cand_max
                if cur_max < best_val:
                    best_val = cur_max
                    best_a = a.copy()

        # --------------------------------------------------------------
        # occasional Gaussian jitter
        # --------------------------------------------------------------
        if rng.random() < 0.012:
            jitter = rng.normal(scale=0.015 * a.mean(), size=n)
            a = _project_to_simplex(a + jitter, S)
            cur_max = np.max(_full_convolution_fft(a))
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP step” from the description)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_steps: int = 4) -> tuple[np.ndarray, float]:
    """Given a vector a, repeatedly solve
        max Σ b_j   s.t.  a*b ≤ M, b ≥ 0,
    scale and mix the result back into a.
    Returns the improved vector and its max‑convolution."""
    n = a.shape[0]
    S = math.sqrt(2.0 * n)
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = np.max(_full_convolution_fft(cur_a))
    mix = 0.5                       # convex combination weight

    for _ in range(max_steps):
        if time.time() >= deadline:
            break

        conv = _full_convolution_fft(cur_a)
        M = np.max(conv)

        # build Toeplitz matrix for the full convolution a * b
        # column = a padded with zeros, row = a[0] followed by zeros
        col = np.concatenate([cur_a, np.zeros(n - 1, dtype=float)])
        row = np.concatenate([cur_a[:1], np.zeros(n - 1, dtype=float)])
        A = toeplitz(col, row)                     # shape (2n‑1) × n

        # linear program:
        #   maximise Σ b_j        ⇔ minimise –Σ b_j
        c = -np.ones(n)
        bounds = [(0, None) for _ in range(n)]
        ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= S * (1.0 + 1e-9):   # no useful improvement
            break

        # scale so that the ℓ₁‑norm matches S and mix
        scale = S / sum_b
        g = b * scale
        cur_a = (1.0 - mix) * cur_a + mix * g
        # mixing preserves the ℓ₁‑norm already
        cur_val = np.max(_full_convolution_fft(cur_a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = cur_a.copy()
        # gradually reduce mixing factor
        mix *= 0.85

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point – called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step sequence that minimises the
    evaluation function.  The routine uses a long Adam optimisation
    (phase 1) followed by an optional LP refinement (phase 2).  The
    best sequence discovered before the time budget expires is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # give ourselves a few seconds cushion for the final print / return
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Retrieve a previously found construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = math.inf
    attempt = 0

    # ------------------------------------------------------------------
    # We run a *single* long optimisation (the most time‑consuming
    # part) and then, if time permits, a few short restarts.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate at most 30 s for a short restart; otherwise use everything left
        slice_budget = min(30.0, remaining)

        # Choose a length.  If we have a previous construction we tend
        # to stay in its neighbourhood, otherwise we explore a range.
        if base_seq is not None:
            n = int(rng.integers(max(500, base_seq.size // 2),
                                 int(min(3000, base_seq.size * 1.5)) + 1))
        else:
            n = int(rng.integers(800, 2500 + 1))

        a0 = _initial_sequence(base_seq, n, rng)
        slice_deadline = time.time() + slice_budget

        # -------- Phase 1 – Adam gradient descent --------------------
        a_opt, max_conv = _adam_optimize(a0, slice_deadline, rng)

        # -------- Phase 2 – optional LP refinement (only for modest n) --
        if n <= 1500 and time.time() < deadline:
            a_opt, max_conv = _lp_refine(a_opt, deadline, rng, max_steps=3)

        # evaluate with the black‑box
        seq = a_opt.tolist()
        external_score = evaluate_sequence(seq)

        attempt += 1
        if external_score < best_score:
            best_score = external_score
            best_seq = seq
            print(f"[global] attempt={attempt} n={n} new best score={best_score:.6f}")

        # If we still have a healthy time budget we try another short run,
        # otherwise we break out – the loop condition will catch the timeout.
        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0]
        best_score = evaluate_sequence(best_seq)

    # store for possible reuse by a later run
    globals()['height_sequence_1'] = best_seq
    print(f"[final] best external score = {best_score:.6f} length = {len(best_seq)}")
    return best_seq
