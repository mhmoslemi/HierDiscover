```
def _solve_lp(a, M):
    import scipy.optimize as opt
    import scipy.linalg as la
    n = a.shape[0]
    # form matrix A (2n-1 x n)
    col = np.concatenate([a, np.zeros(n-1)])
    row = np.r_[a[0], np.zeros(n-1)]
    A = la.toeplitz(col, row)  # dense
    # Set bound slightly above M to account for numeric tolerance
    b_ub = np.full(2*n-1, M * (1.0 + 1e-9))
    # objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # solve
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fallback: return a
        return None

"""
A two‑phase optimizer for the max‑self‑convolution problem.

Phase 1 – Linear‑programming “balancing”:
    For a current non‑negative vector f (ℓ₁‑norm = √(2 n)),
    we maximise Σ b_j subject to the convolution bound
        f * b ≤ M  (M = max(f*f)).
    The LP always has a feasible solution (e.g. b = f).  After solving,
    we rescale b to the same ℓ₁‑norm and move a little towards it.
    This step tends to flatten the autocorrelation.

Phase 2 – Simple stochastic refinement:
    Interleave occasional random mass‑swaps, Gaussian jitter and a
    tiny projected‑gradient step (soft‑max approximation of the ∞‑norm)
    to escape possible plateaus.

Multiple restarts with different lengths are tried while the total
budget (≈1000 s) allows.  The best sequence found is returned.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.linalg as _la

# ----------------------------------------------------------------------
# Helper: projection onto the ℓ₁‑simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    return float(_full_convolution_fft(a).max())


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine
# ----------------------------------------------------------------------
def _solve_lp_balancing(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ b_j  s.t.  a * b ≤ M  (entry‑wise),  b ≥ 0.
    Returns the (unscaled) solution b, or None if LP fails.
    """
    n = a.shape[0]

    # avoid exact zeros – otherwise some columns become free → unbounded LP
    eps = 1e-12
    a_eps = np.maximum(a, eps)

    # Build dense Toeplitz matrix A such that (A @ b)[k] = Σ_i a_i b_{k-i}
    col = np.concatenate([a_eps, np.zeros(n - 1, dtype=a_eps.dtype)])
    row = np.r_[a_eps[0], np.zeros(n - 1, dtype=a_eps.dtype)]
    A = _la.toeplitz(col, row)                     # shape (2n‑1, n)

    # Slight numerical slack on the bound
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-9), dtype=np.float64)

    # maximise Σ b_j  → minimise –Σ b_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Use the HiGHS solver (fast for moderate sized dense problems)
    res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                       method='highs', options={'presolve': True})
    if res.success:
        return res.x
    # on failure we simply give up on this LP step
    return None


# ----------------------------------------------------------------------
# A tiny soft‑max based gradient step (same spirit as the previous code)
# ----------------------------------------------------------------------
def _softmax_update(a: np.ndarray, tau: float, eta: float) -> tuple[np.ndarray, float]:
    """
    One projected gradient step on the smooth approximation of max(a*a).
    Returns the new vector and its maximal self‑convolution entry.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # convolution of a with itself
    b = _full_convolution_fft(a)
    max_b = b.max()

    # soft‑max weights
    if tau <= 1e-12:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / exps.sum()

    # compute gradient = 2 * (a ⋆ w)
    a_rev = a[::-1]
    m = 1 << ((len(w) + n - 1).bit_length())
    Fw = np.fft.rfft(w, n=m)
    Fa_rev = np.fft.rfft(a_rev, n=m)
    conv = np.fft.irfft(Fw * Fa_rev, n=m)[: len(w) + n - 1]
    g = 2.0 * conv[n - 1 : n - 1 + n]

    # gradient step + projection onto the ℓ₁‑simplex
    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    new_max = _max_convolution(a_proj)
    return a_proj, new_max


# ----------------------------------------------------------------------
# Main optimizer for a single length
# ----------------------------------------------------------------------
def _optimize_one_length(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Combine LP “balancing” steps with occasional stochastic moves.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution(a)

    # parameters
    t = 0.4                     # convex‑combination weight
    t_min = 0.02
    move_prob = 0.12            # chance of a random mass swap
    jitter_prob = 0.04
    tau = best_val              # temperature for soft‑max step
    eta = 0.5                   # step size for gradient step

    while time.time() < deadline:
        cur_max = _max_convolution(a)

        # --------------------------------------------------------------
        # 1) LP balancing step
        # --------------------------------------------------------------
        b_lp = _solve_lp_balancing(a, cur_max)
        if b_lp is not None:
            # scale to the correct ℓ₁‑norm
            scale = S / b_lp.sum()
            g = b_lp * scale
            # blend a little towards g
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_to_simplex(a_new, S)
        else:
            a_new = a.copy()   # fallback – keep current

        # --------------------------------------------------------------
        # 2) occasional random mass‑swap (simulated annealing flavour)
        # --------------------------------------------------------------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a_new[i] > 0:
                j = rng.integers(n)
                if i != j:
                    delta = 0.05 * a_new[i]               # move 5 % of mass
                    cand = a_new.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = _project_to_simplex(cand, S)
                    cand_max = _max_convolution(cand)
                    # accept if improvement, otherwise Metropolis rule
                    if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / (0.01 * cur_max)):
                        a_new = cand
                        cur_max = cand_max

        # --------------------------------------------------------------
        # 3) tiny projected gradient step (soft‑max)
        # --------------------------------------------------------------
        a_new, new_max = _softmax_update(a_new, max(tau, 1e-12), eta)

        # --------------------------------------------------------------
        # 4) occasional Gaussian jitter
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            jitter = rng.normal(scale=0.01 * a_new.mean(), size=n)
            a_new = np.clip(a_new + jitter, 0.0, None)
            a_new = _project_to_simplex(a_new, S)
            new_max = _max_convolution(a_new)

        # keep the best seen so far
        if new_max < best_val:
            best_val = new_max
            best_a = a_new.copy()
            print(f"[iter] n={n} best_max={best_val:.6f}")

        a = a_new

        # schedule updates
        t = max(t_min, t * 0.96)
        tau = max(1e-4, tau * 0.96)
        eta = max(0.05, eta * 0.97)

    return best_a, best_val


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine tries several lengths,
    starting from any previously discovered construction, and
    applies the LP‑balancing + stochastic refinement loop.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin

    # ------------------------------------------------------------------
    # Prepare a list of problem sizes to explore.
    # If we already have a construction, reuse its length as a baseline.
    # ------------------------------------------------------------------
    lengths = []
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        base_len = max(800, min(2500, int(prev.size * 1.0)))
        # try a few neighbouring sizes
        lengths = [base_len, base_len + 200, base_len - 200]
        lengths = [max(800, min(3000, L)) for L in lengths]
    else:
        lengths = [1000, 1500, 2000]

    best_seq = None
    best_score = float("inf")
    attempt = 0

    for n in lengths:
        if time.time() >= deadline:
            break
        attempt += 1

        # ------------------------------------------------------------------
        # Initialise a feasible vector of length n (ℓ₁‑norm = √(2·n)).
        # Prefer the existing construction if it is long enough.
        # ------------------------------------------------------------------
        S = np.sqrt(2.0 * n)
        if "height_sequence_1" in globals():
            base = np.asarray(globals()["height_sequence_1"], dtype=float)
            if base.size >= n:
                a0 = base[:n].copy()
            else:
                a0 = np.concatenate([base, rng.random(n - base.size)])
        else:
            a0 = rng.random(n)

        a0 = np.clip(a0, 0.0, None)
        a0 = _project_to_simplex(a0, S)

        # ------------------------------------------------------------------
        # Run the local optimiser on this length, respecting the global deadline.
        # ------------------------------------------------------------------
        slice_deadline = min(time.time() + 40.0, deadline)   # at most 40 s per length
        a_opt, score = _optimize_one_length(a0, slice_deadline, rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.array([1.0], dtype=float)

    # Final verification through the black‑box evaluator
    final_list = best_seq.tolist()
    external_score = evaluate_sequence(final_list)
    print(f"[final] external_score={external_score:.6f} internal_max={best_score:.6f} length={len(final_list)}")

    # store for possible reuse
    globals()["height_sequence_1"] = final_list
    return final_list
