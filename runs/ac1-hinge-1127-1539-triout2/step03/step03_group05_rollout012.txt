"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation builds on three ideas that differ from the previous
submission:

1. **Asymmetric start** – use a few random Dirichlet vectors and keep the
   one with the smallest convolution maximum (the sum is normalised to
   sqrt(2 n) so the objective equals the raw max‑convolution).

2. **Targeted mass transfer** – at every iteration we locate the index that
   contributes most to the current maximal convolution entry and move a
   fraction of its mass to the smallest entry of the whole vector.  This
   strictly reduces the value of that max entry, unless the move creates a
   larger entry elsewhere.

3. **Simulated‑annealing style acceptance** – occasionally we accept a
   non‑improving move (controlled by a temperature that is slowly cooled)
   to escape plateaus, and we inject tiny random perturbations every few
   thousand steps.

All steps respect the 1000 s time budget.  The best vector seen during the
search is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex
        {x | x_i ≥ 0, Σ x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    """
    Produce a random point on the L¹‑simplex of size n (exponential sampling).
    """
    w = np.random.exponential(size=n)
    return w / w.sum() * target_sum


def max_contrib_index(a: np.ndarray, idx_max: int) -> tuple[int, np.ndarray]:
    """
    For the convolution entry at idx_max, return the index i that
    maximises a[i] * a[idx_max - i] (i.e. the largest contribution) and the
    full contribution vector for the valid range.
    """
    n = a.shape[0]
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    i_range = np.arange(low, high + 1)
    partner = idx_max - i_range
    contrib = a[i_range] * a[partner]
    i_star = i_range[np.argmax(contrib)]
    return int(i_star), contrib

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function minimizing `evaluate_sequence`.
    The search runs for roughly `budget_s` seconds and returns the
    best vector encountered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse previous construction if available
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_vec = np.array(base, dtype=np.float64)
    else:
        # a comfortable size for this budget
        n = np.random.randint(7500, 8700)
        base_vec = None

    target_sum = np.sqrt(2.0 * n)      # with this normalisation the objective = max(conv)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = np.full(n, target_sum / n)   # fallback uniform
    best_score = float("inf")

    def evaluate_and_update(cand: np.ndarray, label: str = ""):
        nonlocal best_seq, best_score
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            if label:
                print(f"[{time.time():.2f}] New best ({label}) → {best_score:.6f}")

    print(f"[{time.time():.2f}] Starting random initialisation (n={n})")
    # uniform start
    evaluate_and_update(best_seq, "uniform")

    # try the previous base if given
    if base_vec is not None:

    # a handful of Dirichlet samples
    for k in range(8):
        cand = random_dirichlet(n, target_sum)
        evaluate_and_update(cand, f"rand{k}")
        if time.time() >= deadline:
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 1 – targeted mass‑transfer with simulated annealing
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_max = best_score

    # annealing schedule
    temperature = cur_max * 0.04      # start relatively warm
    temp_cool_factor = 0.9995        # multiplicative cooling per iteration
    move_factor = 0.22               # fraction of the donor's mass to move
    iter_cnt = 0
    last_proj = 0

    print(f"[{time.time():.2f}] Beginning targeted annealing (init score {cur_max:.6f})")

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # compute current convolution and the index of its maximum
        # ------------------------------------------------------------------
        conv = conv_self_fft(a)
        cur_max = conv.max()
        idx_max = int(conv.argmax())

        # ------------------------------------------------------------------
        # identify the index that contributes most to this maximal entry
        # ------------------------------------------------------------------
        i_star, _ = max_contrib_index(a, idx_max)

        # pick a recipient with very small mass (preferably zero)
        # we use the global minimum; if it coincides with i_star, take the second min
        j_star = int(np.argmin(a))
        if j_star == i_star:

            tmp = a.copy()
            tmp[i_star] = np.inf
            j_star = int(np.argmin(tmp))

        eps = move_factor * a[i_star]          # amount to transfer
        if eps <= 0:
            # nothing to move – break early
            break

        # perform the transfer
        a_new = a.copy()
        a_new[i_star] -= eps
        a_new[j_star] += eps

        # safeguard against tiny negative numbers caused by floating point
        if a_new[i_star] < 0:
            a_new[i_star] = 0.0

        # (optional) re‑project onto the simplex every few thousand steps
        if (iter_cnt - last_proj) >= 3000:
            a_new = project_onto_simplex(a_new, target_sum)
            last_proj = iter_cnt

        # evaluate new max convolution
        new_max = conv_self_fft(a_new).max()

        # ------------------------------------------------------------------
        # acceptance test (always accept improvements; otherwise Metropolis)
        # ------------------------------------------------------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:

            if temperature > 1e-12:

                if np.random.rand() < prob:
                    accept = True

        if accept:

            cur_max = new_max
            if cur_max < best_score - 1e-12:

                if cur_max < best_score:
                    best_score = cur_max
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] Jitter step – new best {best_score:.6f}")

        if best_score < 1.503:
            print(f"[{time.time():.2f}] Target bound reached ({best_score:.6f}) – stopping early")
            break

    # Return the best sequence discovered (plain Python list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Search finished – final best score {best_score:.6f}")
    return best_seq.tolist()
