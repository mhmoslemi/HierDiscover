def _full_conv(a, b):
    n = a.size + b.size - 1
    L = 1 << ((n - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)[:n]
    return conv

def _smooth_optimize(a0, n, deadline,
                     max_iters=50000,
                     alpha=0.1,
                     theta_start=2.0,
                     theta_end=30.0):
    a = a0.copy()
    S = math.sqrt(2.0 * n)
    # initialize y
    eps = 1e-12
    y = np.log(a + eps)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    best_seq = a.copy()
    best_max = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute a from y
        maxy = np.max(y)  # for numeric stability
        exp_y = np.exp(y - maxy)  # subtract max to avoid overflow
        a = exp_y
        a_sum = a.sum()
        a = a * (S / a_sum)   # scaled to sum S
        # Compute self-convolution
        conv = _full_conv(a, a)   # length 2n-1
        # compute theta - linear schedule
        theta = theta_start + (theta_end - theta_start) * it / max_iters
        # compute softmax weights
        # compute max for numerical stability
        theta_conv = theta * conv
        max_tc = np.max(theta_conv)
        w = np.exp(theta_conv - max_tc)
        w_sum = w.sum()
        w /= w_sum  # now sum to 1
        # gradient wrt a
        a_rev = a[::-1]
        conv_rev = _full_conv(a_rev, w)
        g = 2.0 * conv_rev[n-1:n-1+n]
        # compute mu
        mu = np.dot(g, a) / S
        grad_y = a * (g - mu)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
        y = y - step
        # record improvement on true max
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
        # maybe occasionally jitter
        if it % 5000 == 0:
            # small random jitter to escape
            jitter = np.random.randn(n) * a.mean() * 0.005
            a_jit = np.clip(a + jitter, 0, None)
            a_jit_sum = a_jit.sum()
            if a_jit_sum > 0:
                a_jit = a_jit * (S / a_jit_sum)
                # update y correspondingly
                y = np.log(a_jit + eps)
                # reset moments
                m.fill(0)
                v.fill(0)
        # early exit if near deadline
        if time.time() >= deadline - 0.5:
            break
    return best_seq, best_max
```

"""
Solution optimizer for the step‑function bound.

The algorithm searches over several vector lengths.  For each length a
log‑softmax parametrisation (`y` such that a = S·softmax(y) ) is used;
this guarantees non‑negativity and a fixed ℓ₁‑norm `S = sqrt(2·n)`.
A smooth approximation of the max‑convolution,
   f_θ(a) = (1/θ)·log Σ_k exp(θ·(a*a)[k]),
is minimised by (Adam‑style) gradient descent.
The temperature `θ` is increased from a low value (smooth landscape) to a
high value (focus on the true maximum).  After a run the best vector
found is kept.  Good candidates are up‑sampled to larger lengths and
optimised again, allowing the construction to approach the asymptotic
limit.  If SciPy is available a tiny LP refinement (as in the reference
code) is optionally applied at the end.

The function `propose_candidate` returns a list of non‑negative heights
that (hopefully) yields a score lower than the current best bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for an LP refinement (high‑precision polishing)

# ----------------------------------------------------------------------
# Helper: full linear convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * b (full length) using real FFT."""
    size = a.size + b.size - 1
    # next power of two for efficient FFT
    L = 1 << ((size - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)[:size]
    return conv


# ----------------------------------------------------------------------
# Smooth max‑convolution optimiser (soft‑max temperature annealing)
# ----------------------------------------------------------------------
def _smooth_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    *,
    max_iters: int = 5000,
    alpha: float = 0.5,
    theta_start: float = 2.0,
    theta_end: float = 30.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smooth max‑convolution.
    Returns the best vector seen and its (true) max‑convolution value.
    """
    S = math.sqrt(2.0 * n)                     # fixed ℓ₁‑norm
    # initialise y = log(a)  (soft‑max parameters)
    y = np.log(a0 + eps)

    m = np.zeros_like(y)                       # first moment estimate
    v = np.zeros_like(y)                       # second moment estimate

    best_seq = None
    best_val = float('inf')
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- a = S·softmax(y) ---------------------------------------
        # Stabilise the exponentials
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        a = exp_y
        a_sum = a.sum()
        a = a * (S / a_sum)                     # enforce ℓ₁‑norm

        # ----- self‑convolution ---------------------------------------
        conv = _full_conv(a, a)                 # length 2·n‑1

        # ----- temperature schedule ------------------------------------
        theta = theta_start + (theta_end - theta_start) * (it / max_iters)

        # ----- soft‑max weights over convolution entries -------------
        # w_k ∝ exp(θ·conv_k)  (numerically stable)
        tc = theta * conv
        tc_max = np.max(tc)
        w = np.exp(tc - tc_max)
        w /= w.sum()                            # Σ w = 1

        # ----- gradient w.r.t. a --------------------------------------
        # g_i = Σ_k w_k ∂conv_k/∂a_i = 2· Σ_k w_k a_{k-i}
        a_rev = a[::-1]
        tmp = _full_conv(a_rev, w)              # length 3·n‑2
        g = 2.0 * tmp[n - 1 : n - 1 + n]        # extract the n‑length slice

        # ----- convert gradient to y‑space (soft‑max Jacobian) -------
        mu = np.dot(g, a) / S                    # scalar u in the formula
        grad_y = a * (g - mu)                   # shape (n,)

        # ----- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        y -= step

        # ----- keep best according to the exact (non‑smooth) score ---
        cur_max = conv.max()
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_seq = a.copy()

        # ----- occasional jitter to escape plateaus -------------------
        if it % max(500, max_iters // 10) == 0:
            jitter = np.random.randn(n) * a.mean() * 0.005
            a_jit = np.clip(a + jitter, 0.0, None)
            if a_jit.sum() > 0.0:
                a_jit = a_jit * (S / a_jit.sum())
                y = np.log(a_jit + eps)
                m.fill(0.0)
                v.fill(0.0)

        # ----- early‑out if very close to deadline --------------------
        if time.time() >= deadline - 0.5:
            break

    # In the unlikely case no improvement was ever recorded
    if best_seq is None:
        best_seq = a.copy()
        best_val = conv.max()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Heuristic to decide the number of iterations for a given length
# ----------------------------------------------------------------------
def _calc_max_iters(n: int) -> int:
    """Return a reasonable iteration budget, larger for smaller n."""
    base = 8000                 # target total iterations at n≈4000
    iters = int(base * 4000 / max(1, n))
    return max(500, min(iters, 12000))


# ----------------------------------------------------------------------
# Linear‑programming refinement (identical to the reference implementation)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the dense matrix H with (H @ g)[k] = (f * g)[k] (full convolution)."""
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i : i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,
               max_steps: int = 5) -> np.ndarray:
    """Optional LP polishing step (requires SciPy)."""
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:                     # keep the LP tractable
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break
        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σ g_i  → minimise –Σ g_i
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:  # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        # convex combination and re‑normalise
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f

            print(f"[final] final score = {best_score:.6f}, length = {len(best_seq)}")
    else:
        # ------------------------------------------------------------------
        # Fallback – uniform sequence (should never be required).
        # ------------------------------------------------------------------
        n = 2000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq.tolist()
