def generate_sidon_set(n, rng):
    chosen = []
    sums = set()
    for i in range(n):
        ok = True
        for j in chosen:
            s = i + j
            if s in sums:
                ok = False
                break
        if ok:
            # also check sum i+i
            if i + i in sums:
                ok=False
            else:
                pass
        if ok:
            # add
            for j in chosen:
                sums.add(i + j)
            sums.add(i + i)
            chosen.append(i)

"""
Hybrid optimizer for the step‑function bound.

The objective (after normalising the ℓ₁‑norm to √(2·n)) is to minimise the
maximal entry of the self‑convolution a∗a.  The algorithm combines

* a hard‑max sub‑gradient step with Euclidean projection onto the ℓ₁‑simplex,
* a targeted simulated‑annealing mass‑move that specifically reduces the
  contribution to the currently maximal convolution entry,
* occasional random swaps of two positions,
* small Gaussian jitter.

Multiple restarts with different initialisations (uniform, Dirichlet,
monotonically decreasing, and a deterministic Sidon‑set construction) are
performed until the time budget expires.  The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Full convolution via real FFT (fast O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Generate a deterministic Sidon set (B2 sequence) inside [0, n).
# Greedy construction – fast and provides a useful non‑uniform start.
# ----------------------------------------------------------------------
def _sidon_set_greedy(n: int) -> np.ndarray:
    """Return a Sidon set of indices (no duplicate unordered sums)."""
    chosen = []
    sums = set()
    for i in range(n):
        ok = True
        # check i+i
        if (i + i) in sums:
            ok = False
        if ok:
            for j in chosen:
                if (i + j) in sums:
                    ok = False
                    break
        if ok:
            # add i and update sums
            for j in chosen:
                sums.add(i + j)
            sums.add(i + i)
            chosen.append(i)
    return np.asarray(chosen, dtype=int)


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient step (projected) – one sweep.
# ----------------------------------------------------------------------
def _hardmax_step(a: np.ndarray, S: float) -> tuple[np.ndarray, float]:
    """Perform one sub‑gradient descent step for the current maximal lag."""
    b = _full_convolution_fft(a)
    k = int(b.argmax())
    n = a.shape[0]

    # gradient of the max‑lag entry
    grad = np.zeros_like(a)
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)
    grad[valid] = 2.0 * a[partner[valid]]

    # sub‑gradient descent with a fixed step (will be decayed by outer loop)
    step = 0.6  # initial step size – outer code scales this
    a_new = a - step * grad
    a_proj = _project_to_simplex(a_new, S)

    new_max = float(_full_convolution_fft(a_proj).max())
    return a_proj, new_max


# ----------------------------------------------------------------------
# Targeted simulated‑annealing move that reduces the current maximal lag.
# ----------------------------------------------------------------------
def _targeted_move(a: np.ndarray,
                   k_max: int,
                   cur_max: float,
                   S: float,
                   rng: np.random.Generator,
                   temp: float,
                   epsilon: float = 0.10) -> tuple[np.ndarray, float, bool]:
    """
    Move a fraction ε of the mass from an index that heavily participates
    in the maximal lag to an index that does not.
    Returns the candidate vector, its new maximal convolution entry and
    a boolean indicating whether the move was accepted.
    """
    n = a.shape[0]
    idx = np.arange(n)
    partner = k_max - idx
    valid = (partner >= 0) & (partner < n)

    # contribution of each index to the maximal lag (zero where not involved)
    contrib = np.zeros(n)
    contrib[valid] = a[valid] * a[partner[valid]]

    if np.sum(contrib) == 0.0:
        return a, cur_max, False

    probs = contrib / contrib.sum()
    i = rng.choice(n, p=probs)          # source with large contribution
    if a[i] < 1e-12:
        return a, cur_max, False

    delta = min(epsilon * a[i], a[i])

    # choose a destination that does *not* belong to the maximal lag
    zero_contrib = np.where(contrib == 0.0)[0]
    if zero_contrib.size > 0:
        j = rng.choice(zero_contrib)
    else:
        # fallback: random index different from i
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)

    a_cand = a.copy()
    a_cand[i] -= delta
    a_cand[j] += delta

    # evaluate
    new_max = float(_full_convolution_fft(a_cand).max())

    # Metropolis acceptance
    if new_max <= cur_max:
        accept = True
    else:
        prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
        accept = rng.random() < prob

    return a_cand, new_max, accept


# ----------------------------------------------------------------------
# One optimisation run (a local search) – time‑limited.
# ----------------------------------------------------------------------
def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hybrid local optimiser:
      – hard‑max sub‑gradient steps,
      – targeted annealing mass moves,
      – occasional random swaps,
      – small Gaussian jitter.
    Returns the best vector discovered and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise best record
    b = _full_convolution_fft(a)
    cur_max = float(b.max())
    best_a = a.copy()
    best_max = cur_max

    # hyper‑parameters
    eta = 0.6               # step size for gradient step (will be decayed)
    eta_decay = 0.9997
    move_prob = 0.25        # probability of attempting a targeted mass move
    swap_prob = 0.10        # probability of attempting a random swap
    jitter_prob = 0.02
    jitter_scale = 0.02
    epsilon_move = 0.12     # fraction of mass moved when a targeted move is tried
    temp = cur_max * 0.2    # temperature for simulated‑annealing moves
    temp_decay = 0.9998

    while time.time() < deadline:
        # recompute convolution and maximal lag
        b = _full_convolution_fft(a)
        cur_max = float(b.max())
        k_max = int(b.argmax())

        rnd = rng.random()

        if rnd < move_prob:
            # ----- targeted annealing move -----
            a_cand, new_max, accepted = _targeted_move(
                a, k_max, cur_max, S, rng, temp, epsilon=epsilon_move)
            if accepted:
                a = a_cand
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
            temp *= temp_decay

        elif rnd < move_prob + swap_prob:
            # ----- random swap of two positions -----
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                a_cand = a.copy()
                a_cand[i], a_cand[j] = a_cand[j], a_cand[i]
                cand_max = float(_full_convolution_fft(a_cand).max())
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(temp, 1e-12)):
                    a = a_cand
                    cur_max = cand_max
                    if cur_max < best_max:
                        best_max = cur_max
                        best_a = a.copy()
                temp *= temp_decay

        else:
            # ----- hard‑max gradient descent step -----
            a, new_max = _hardmax_step(a, S)
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # occasional jitter to escape plateaus
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

        # decay schedules
        eta *= eta_decay
        temp *= temp_decay

    return best_a, best_max


# ----------------------------------------------------------------------
# Outer orchestration – many restarts with different lengths/initialisations.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Run a series of short hybrid optimisation restarts within the time
    budget and return the sequence that yields the smallest maximal
    self‑convolution entry (i.e. the smallest score from `evaluate_sequence`).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # reuse a previously found good construction if it exists
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    while time.time() < deadline:
        attempt += 1

        # --------------------------------------------------------------
        # Choose a length.  A broad range (800‑4000) works well in practice.
        # --------------------------------------------------------------
        n = rng.integers(800, 3501)

        # --------------------------------------------------------------
        # Build a collection of diverse initial vectors.
        # --------------------------------------------------------------
        inits = []

        # 1) Uniform Dirichlet(1) – pure random.
        inits.append(rng.random(n))

        # 2) Dirichlet with many small components (more spread).
        inits.append(rng.dirichlet(np.ones(n) * 0.5))

        # 3) Monotonically decreasing exponential shape.
        exp_shape = np.exp(-np.linspace(0, 5, n))
        inits.append(exp_shape)

        # 4) Sidon‑set based start (few non‑zero equal entries).
        sidon = _sidon_set_greedy(n)
        if sidon.size > 0:
            a = np.zeros(n)
            a[sidon] = 1.0
            inits.append(a)

        # 5) Re‑use the best known construction (cropped or padded).
        if base_seq is not None:
            old = np.asarray(base_seq, dtype=float)
            if old.size >= n:
                inits.append(old[:n])
            else:
                a = np.zeros(n)
                a[:old.size] = old
                inits.append(a)

        # --------------------------------------------------------------
        # Run a short optimisation for each initialisation.
        # --------------------------------------------------------------
        for init in inits:
            # Project to the feasible simplex (ℓ₁‑norm = √(2·n))

            slice_deadline = min(time.time() + 30.0, deadline)

            a_opt, cur_score = _optimize_local(a0, slice_deadline, rng)

            if cur_score < best_overall_score:
                best_overall_score = cur_score
                best_overall_seq = a_opt.copy()

            if time.time() >= deadline:

    # Final evaluation through the provided black‑box function.
    # ------------------------------------------------------------------
    if best_overall_seq is not None:

              f"(internal max = {best_overall_score:.6f}) "
              f"length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq
        return final_seq
    else:
        # Should never happen – return a trivial feasible vector.
        return [1.0]
