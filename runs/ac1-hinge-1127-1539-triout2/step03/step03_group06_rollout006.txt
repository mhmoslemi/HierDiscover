def _fft_convolve(a,b):
    n = a.shape[0] + b.shape[0] - 1
    fft_len = 1 << (n-1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Fa*Fb, n=fft_len)[:n]
    return conv

def _smooth_grad(a, alpha):
    conv = _fft_convolve(a, a)  # full conv
    maxc = conv.max()
    # compute scaled weights
    # subtract maxc for numeric stability
    ws = np.exp(alpha * (conv - maxc))
    w = ws / ws.sum()

    grad = 2.0 * conv2[len(a)-1 : len(a)-1 + len(a)]  # slice n elements starting at n-1
    return grad, conv.max()

def _run_optimize_smooth(a0, n, deadline,
                         eta0=0.5, decay=0.9999,
                         beta=0.9,
                         jitter_prob=0.01,
                         max_iters=200000,
                         alpha0=0.1,
                         alpha_max=1000.0,
                         alpha_scale=1.05):
    a = a0.copy()
    S = math.sqrt(2.0 * n)
    best_max = float('inf')
    best_seq = a.copy()
    v = np.zeros_like(a)  # momentum
    eta = eta0
    alpha = alpha0
    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max = _smooth_grad(a, alpha)
        # momentum
        v = beta * v + (1-beta) * grad
        # exponentiated update
        a = a * np.exp(-eta * v)
        # clip maybe small epsilon to avoid zero
        a = np.maximum(a, 0.0)
        # renorm
        total = a.sum()
        if total == 0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # jitter
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # update alpha gradually
        if it % 2000 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_scale, alpha_max)

        eta *= decay

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                # maybe reset momentum or increase alpha
                alpha = min(alpha * 1.5, alpha_max)
                no_improve = 0

        # early exit if time near deadline
        if time.time() >= deadline - 0.1:
            break

    return best_seq, best_max
```

Its gradient can be written as a convolution of the soft‑max weights
with the (reversed) current vector, which is computed with FFTs.
A multiplicative (exponentiated) update together with a projection onto
the ℓ1‑sphere (∑a = √(2·n)) guarantees non‑negativity.
A simple momentum term, step‑size decay, random jitter and an
increasing temperature (α) help to escape plateaus.
Several independent restarts with different lengths are performed
until the 1000 s wall‑clock budget expires.  When a promising vector is
found a tiny linear‑programming refinement (SciPy) is applied, and a
final polishing run is executed on the current best sequence.

The returned list contains non‑negative heights (floats) and should
achieve a score lower than the previous bound (~1.518).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional linear‑programming refinement (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a power‑of‑two FFT."""
    n = x.shape[0] + y.shape[0] - 1
    fft_len = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)[:n]
    return conv


def _smooth_grad(a: np.ndarray, alpha: float):
    """
    Compute the gradient of the soft‑max surrogate of the self‑convolution.

    Returns
    -------
    grad : np.ndarray (shape a)
        The gradient (2 * (w ⋆ a_rev) on the relevant slice).
    cur_max : float
        The true maximum entry of the convolution (the external score).
    """
    # self‑convolution a∗a (full length 2n‑1)
    conv = _fft_convolve(a, a)
    cur_max = float(conv.max())

    # soft‑max weights, numerically stabilised
    # w_k ∝ exp(alpha * (conv_k - cur_max))
    w = np.exp(alpha * (conv - cur_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (w ⋆ a_rev) restricted to the n relevant positions
    a_rev = a[::-1]
    conv2 = _fft_convolve(w, a_rev)               # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv2[n - 1:n - 1 + n]           # slice i ↦ Σ_j w_{i+j} a_j
    return grad, cur_max


def _run_optimize_smooth(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         eta0: float = 0.6,
                         decay: float = 0.99996,
                         beta: float = 0.9,
                         jitter_prob: float = 0.015,
                         max_iters: int = 250_000,
                         alpha0: float = 0.05,
                         alpha_max: float = 200.0,
                         alpha_inc: float = 1.06) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the soft‑max surrogate.
    Returns the best vector (as a NumPy array) seen during the run and its
    true maximal convolution value.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)                # target ℓ₁‑norm
    best_val = float('inf')
    best_seq = a.copy()
    v = np.zeros_like(a)                  # momentum buffer

    eta = eta0
    alpha = alpha0
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max = _smooth_grad(a, alpha)

        # momentum (exponential moving average)
        v = beta * v + (1.0 - beta) * grad

        # multiplicative (exponentiated) update – guarantees non‑negativity
        a = a * np.exp(-eta * v)

        # re‑normalise to keep the ℓ₁‑norm fixed
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to break tiny plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.015
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # adapt the temperature (α) – slowly sharpen the approximation
        if it % 1500 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_inc, alpha_max)

        # step‑size decay
        eta *= decay

        # keep the best seen vector
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 7_000:
                # if we stall, raise α a bit to focus on the true max
                alpha = min(alpha * 1.5, alpha_max)
                no_improve = 0

        # early exit if we are out of time
        if time.time() >= deadline - 0.1:
            break

    return best_seq, best_val


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct H ∈ ℝ^{(2n‑1)×n} with (H·g)[k] = (f∗g)[k].
    Used only for the optional LP refinement (n ≤ 3000).
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, mix: float = 0.22,
               max_steps: int = 4) -> np.ndarray:
    """
    Small linear‑programming refinement (Frank‑Wolfe style) as in the
    original baseline.  It solves
        max Σ g_i   s.t.   0 ≤ g_i , (f∗g)[k] ≤ M .
    The optimal g is mixed with f and renormalised.
    Skipped when SciPy is unavailable or n > 3000.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = float(conv.max())

        H = _build_conv_matrix(f)
        c = -np.ones(n)                      # maximise Σg  ⇔ minimise –Σg
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 7})
        except Exception:                     # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        # convex combination and renormalisation
        f = (1.0 - mix) * f + mix * g
        f = f / f.sum() * S

    return f


def propose_candidate(seed: int = 887, budget_s: int = 1000) -> list[float]:
    """
    Entry point for the evaluation harness.
    Returns a list of non‑negative heights that (hopefully) achieves a
    lower ``evaluate_sequence`` score than the previous best bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # small safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1)  Use the existing construction if it exists (baseline).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                # normalise to the ℓ₁‑norm √(2 n); the external score is scale‑invariant
                n0 = cand.size
                S0 = math.sqrt(2.0 * n0)
                cand = cand / cand.sum() * S0
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] height_sequence_1 score = {best_score:.6f}, length = {n0}")
        except Exception:                    # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 2)  Randomised multi‑restart optimisation.
    # ------------------------------------------------------------------
    # a modest set of lengths – larger lengths give more freedom but cost more time
    lengths = [1500, 3000, 4000, 6000, 8000, 12000, 16000, 20000]
    run_id = 0

    while time.time() < deadline:
        n = int(rng.choice(lengths))

        # initialise a starting vector
        if best_seq is not None and best_seq.size >= n and rng.random() < 0.4:
            a0 = best_seq[:n].copy()
        else:
            # exponential distribution tends to give a few larger entries,
            # which is often a good seed for this problem
            a0 = rng.exponential(scale=1.0, size=n)

        # normalise to ℓ₁‑norm √(2 n)
        S = math.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # one optimisation run
        seq, conv_max = _run_optimize_smooth(a0, n, deadline)

        # compute the external score (identical to conv_max because of the normalisation)
        score = float(conv_max)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        # optional LP refinement when a noticeable improvement occurs
        if score < best_score * 0.97:
            refined = _lp_refine(best_seq, deadline)
            # recompute the true max after refinement
            refined_max = _fft_convolve(refined, refined).max()
            if refined_max < best_score:
                best_score = refined_max
                best_seq = refined.copy()
                print(f"[LP refine] run={run_id} n={n} refined score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 3)  Final polishing of the best candidate.
    # ------------------------------------------------------------------
    if best_seq is not None:
        # a few extra gradient‑descent steps with a slightly larger step‑size
        best_seq, final_max = _run_optimize_smooth(
            best_seq,
            best_seq.size,
            deadline,
            eta0=0.45,
            decay=0.99998,
            jitter_prob=0.02,
            max_iters=120_000,
            alpha0=0.03,
            alpha_max=300.0)

        # final LP pass (if affordable)
        best_seq = _lp_refine(best_seq, deadline)

        # evaluate the exact score for reporting
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    else:
        # fallback – a simple uniform vector (should never be taken)
        n = 2000
        a = np.full(n, math.sqrt(2.0 * n) / n)
        best_seq = a
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score = {best_score:.6f}")

    return best_seq.tolist()
