   total_sum = target_sum
   # compute softmax of z0 yields distribution, we will have a = total_sum * softmax(z). We'll just set z0 = np.log(a0 + eps), use eps to avoid log(0).

   epsilon = 1e-12
   z = np.log(a0 + epsilon)
   m = np.zeros_like(z)
   v = np.zeros_like(z)
   beta1=0.9; beta2=0.999; eps=1e-8
   lr = 0.1  # default

   step = 0
   best_a = a0.copy()
   best_score = compute_score(best_a)

   beta = init_beta
   while time.time() < deadline:
       # compute a
       a = total_sum * _softmax(z)   # softmax returns distribution sum to 1

       # compute convolution via FFT
       # compute b = a*a (full conv)
       # compute softmax probabilities p = exp(beta*(b - max(b))) normalized.

       # compute grad_z

       grad_z = a * (g_a - dot)   # shape (n,)

       # Adam updates
       step += 1
       m = beta1 * m + (1 - beta1) * grad_z
       v = beta2 * v + (1 - beta2) * (grad_z ** 2)
       m_hat = m / (1 - beta1**step)
       v_hat = v / (1 - beta2**step)
       z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

       # occasional jitter: maybe add small noise to z.

       # evaluate
       a = total_sum * _softmax(z)
       cur_score = compute_score(a)
       if cur_score < best_score:
           best_score = cur_score
           best_a = a.copy()
           print(f"[{time.time():.1f}] Adam improved score {best_score:.6f} n={n} beta={beta:.2f}")

       # schedule beta increase
       if step % 200 == 0 and beta < max_beta:
           beta = min(beta * 1.5, max_beta)

       # optionally reduce lr?

"""
Propose a high‑quality step‑function sequence.
The algorithm has three stages:

1. **Broad random generation** – for several lengths we create many
   candidates from simple parametric families (power‑law, exponential,
   mixtures).  All candidates are normalised so that
   Σ a_i = √(2·n); for such sequences the evaluation reduces to
   ``max(a*a)`` (the maximal convolution entry).

2. **Smooth‑max Adam optimisation** – each shortlisted candidate is
   re‑parameterised as a soft‑max of free variables `z`.
   We minimise a smooth approximation of the maximum,
   ``f_β(a)= (1/β)·log∑exp(β·(a*a))``,
   using the Adam optimiser and an annealing schedule for the
   temperature β (β grows from 1 to a few hundred).  The gradient of
   ``f_β`` is computed analytically via FFT‑based convolutions.

3. **Guided hill‑climbing** – after the Adam phase we locally transfer
   mass from indices that contribute heavily to the current peak of the
   convolution to indices that contribute little, accepting only
   improving moves.  This quickly eliminates remaining spikes.

All stages are time‑boxed; the routine returns the best sequence found
before the global deadline.  The implementation avoids file I/O,
uses only NumPy and the standard library, and works with any
pre‑imported ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (FFT‑based convolution, scoring, simplex handling)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Fast objective for a normalised sequence: max(a*a)."""
    return _max_convolution_fft(a)


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (sums to 1)."""
    e = np.exp(z - np.max(z))
    return e / e.sum()


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (smooth‑max) of the convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a ( 1/β·log∑exp(β·b) )
    The gradient equals 2·(a ⋆ p) where p are the soft‑max
    probabilities of the entries of β·b.
    """
    n = a.shape[0]

    # full convolution a*a via FFT (size a power of two)
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over the convolution entries
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)  # shape (2·n‑1,)

    # gradient = 2·(a ⋆ p)  – convolution of a (reversed) with p
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]   # keep only the n central entries
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (smooth‑max) objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a0 (already normalised to sum=√(2·n)) using Adam on the
    smooth‑max objective.  The variables are the unconstrained log‑weights
    `z` such that a = S·softmax(z).  Returns the best a seen and its score.
    """
    n = a0.size
    target_sum = np.sqrt(2.0 * n)

    # initialise z from a0 (add tiny epsilon to avoid log(0))
    eps_init = 1e-12
    z = np.log(a0 + eps_init)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.2                     # base learning rate (works well in practice)
    step = 0

    # smooth‑max temperature schedule
    beta = 1.0
    max_beta = 512.0

    best_a = a0.copy()
    best_val = _score(best_a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        # current sequence from soft‑max re‑parameterisation
        a = target_sum * _softmax(z)

        # gradient of the smooth‑max w.r.t a
        g_a, _ = _gradient_softmax(a, beta)

        # chain rule: ∂f/∂z = a * (∂f/∂a - (∂f/∂a)·a)
        dot = float(np.dot(g_a, a))
        grad_z = a * (g_a - dot)

        # Adam update
        step += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional jitter to escape plateaus
        if step % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            z += jitter

        # evaluate
        a_now = target_sum * _softmax(z)
        cur_val = _score(a_now)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_now.copy()
            print(f"[{time.time():.1f}] Adam best={best_val:.6f} n={n} β={beta:.2f}")

        # anneal temperature
        if step % 100 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (pairwise mass transfer)
# ----------------------------------------------------------------------


def _guided_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy local search: repeatedly move a fraction of mass from a
    “donor’’ index (large contribution to the current peak) to a
    “receiver’’ index (small contribution).  Only improving moves are kept.
    """
    n = a.size
    best_a = a.copy()
    best_val = _score(best_a)
    cur_val = best_val

    rng = np.random.default_rng()
    max_frac = 0.07            # at most 7 % of a donor’s mass per move

    while time.time() < deadline:
        conv = np.convolve(a, a)          # O(n) full convolution
        k_max = int(np.argmax(conv))

        # compute sensitivity vector for the peak entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # indices i∈[lo,hi] have partner j = k_max‑i
        sens[lo : hi + 1] = a[k_max - np.arange(lo, hi + 1)]

        # weight = contribution (a_i * sens_i)
        weight = sens * a
        if weight.max() == 0.0:
            break

        donor = int(np.argmax(weight))

        # pick a receiver among the low‑weight tail (≈10 %)
        low_thr = np.percentile(weight, 10)
        cand = np.where(weight <= low_thr)[0]
        if cand.size == 0:
            break
        receiver = int(rng.choice(cand))
        if donor == receiver or a[donor] <= 1e-12:
            continue

        # propose a move
        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:                     # accept only improving
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] Hill‑climb improved={best_val:.6f}")
        else:
            # revert
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val

# Candidate generation (parametric families)
# ----------------------------------------------------------------------


def _make_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random candidate of length n from a simple parametric family."""
    target_sum = np.sqrt(2.0 * n)

    shape_type = rng.choice(["power", "exp", "mix"])
    if shape_type == "power":
        p = rng.uniform(0.2, 1.2)
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    elif shape_type == "exp":
        alpha = rng.uniform(0.0005, 0.02)
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
    else:  # mixture of two power‑laws
        p1 = rng.uniform(0.2, 1.2)
        p2 = rng.uniform(0.2, 1.2)
        w = rng.uniform(0.0, 1.0)
        a1 = (np.arange(1, n + 1, dtype=np.float64) ** (-p1))
        a2 = (np.arange(1, n + 1, dtype=np.float64) ** (-p2))
        a = w * a1 + (1.0 - w) * a2

    # add a little noise, enforce non‑negativity and normalise
    noise = rng.normal(scale=0.02 * a.mean(), size=n)
    a += noise
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.ones(n, dtype=np.float64) * target_sum / n
    else:
        a *= target_sum / a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines a broad random sampling, Adam optimisation on a
    smooth maximum, and a final guided hill‑climbing phase.  It returns the
    best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Stage 1 – generate a pool of diverse candidates
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 4500, 6000, 8000, 12000]
    per_len = 12                     # how many raw candidates per length
    raw_candidates = []

    for n in lengths:
        if time.time() >= deadline:
            break
        for _ in range(per_len):
            a = _make_candidate(n, rng)
            sc = _score(a)
            raw_candidates.append((sc, a.copy(), n))

    # optional: add the previously discovered construction if available
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n0 = prev.size
        target = np.sqrt(2.0 * n0)
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev * (target / prev.sum())
            raw_candidates.append((_score(prev), prev.copy(), n0))
    except Exception:
        pass

    # keep the globally best few raw candidates for deeper optimisation
    raw_candidates.sort(key=lambda x: x[0])
    top_k = 8
    candidates = raw_candidates[:top_k]

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation (smooth‑max) on each shortlisted candidate
    # ------------------------------------------------------------------
    for base_score, a0, n in candidates:
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget (max 55 s per candidate)
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(55.0, remaining * 0.6)

        a_opt, val_opt = _adam_optimize(a0, sub_deadline)

        # ------------------------------------------------------------------
        # Stage 3 – guided hill‑climbing refinement
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_opt, val_opt = _guided_hill_climb(a_opt, deadline)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"*** GLOBAL BEST *** n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final local polishing on the best sequence (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _guided_hill_climb(best_seq, deadline)

    # fallback in the extremely unlikely case that nothing was found
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)

    # trim trailing numerical zeros for a cleaner output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    return best_seq.tolist()
