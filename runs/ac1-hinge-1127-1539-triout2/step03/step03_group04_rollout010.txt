"""
advanced_optimize.py

Search for a non‑negative step‑function (sequence of heights) that minimises
``evaluate_sequence``.  The objective is homogeneous, so after each update the
sequence is projected onto the simplex {x_i ≥ 0, Σ x_i = √(2·n)}.  The
algorithm consists of three stages:

1. **Coarse power‑law init** – a short grid over the exponent *p* for a
   length *n* (large, 5·10⁴) is examined and the best exponent is kept.
2. **Adam optimiser on a soft‑max approximation of the max‑convolution**
   – the smooth proxy ``log‑sum‑exp_β(a*a)`` is minimised while the temperature
   β is annealed.  Adam (adaptive moments) respects the simplex via a
   Euclidean projection after each step.
3. **Deterministic sub‑gradient descent** – the exact sub‑gradient of the
   (non‑smooth) maximum convolution entry is used, again with projection.
   Finally a lightweight hill‑climbing routine shuffles a tiny amount of mass
   from high‑contributing to low‑contributing indices.

All stages are time‑boxed; the best sequence found before the global deadline
is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Fast proxy for the competition objective.
    For a vector projected onto the simplex Σa = sqrt(2·n) the score equals
    the maximum autocorrelation entry.
    """
    return _max_convolution_fft(a)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i≥0, Σx_i=s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the maximum convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where b = a*a (full convolution) and
    g = ∇ (log‑sum‑exp_{β}(b)), i.e. the gradient of the smooth max.
    """
    n = a.shape[0]

    # convolution a*a (full)
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)            # shape (2·n‑1,)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]    # keep only the n relevant entries
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smooth maximum
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float):
    """
    Adam optimiser on the soft‑max approximation.
    The temperature β is slowly increased while the parameters stay on the
    simplex Σa = √(2·n).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    beta = 1.0
    max_beta = 512.0

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # gradient of the smooth max (log‑sum‑exp)
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size: a modest base scaled inversely with √it
        lr = 0.10 * target_sum / np.sqrt(it)

        a = a - lr * (m_hat / (np.sqrt(v_hat) + eps))

        # projection onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional temperature increase
        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 1.2, max_beta)

        # keep track of the best
        if it % 200 == 0:
            cur = _score(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[{time.time():.2f}] Adam it={it} beta={beta:.2f} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution entry
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Deterministic sub‑gradient descent on the (non‑smooth) maximum convolution
    entry.  The sub‑gradient is 2·a_rev restricted to the active index.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    step0 = target_sum * 0.05
    it = 0

    while time.time() < deadline:
        it += 1
        # full convolution (FFT, O(n log n))
        size = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

        k = int(np.argmax(conv))

        # compute sub‑gradient for the active index k
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # derivative of (a*a)_k w.r.t a_i is 2·a_{k-i}
        g[idx] = 2.0 * a[k - idx]

        # diminishing step size
        lr = step0 / np.sqrt(it)
        a = a - lr * g
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            cur = _score(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[{time.time():.2f}] SubGrad it={it} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing (mass shuffling) – works with FFT‑based conv.
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray, deadline: float):
    """
    Local search that repeatedly moves a small fraction of mass from the
    indices that contribute most to the current peak towards indices that
    contribute the least.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)
    cur_val = best_val

    rng = np.random.default_rng()

    max_frac = 0.07          # maximal fraction of donor mass to relocate

    while time.time() < deadline:
        # full convolution via FFT (fast)
        size = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

        k_max = int(np.argmax(conv))

        # contribution weights w_i = a_i * a_{k_max-i} (zero out‑of‑range)
        weight = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        weight[idx] = a[idx] * a[k_max - idx]

        if weight.max() == 0.0:
            break

        # donor = index with largest contribution
        donor = int(np.argmax(weight))

        # receiver = a low‑contribution index (bottom 10 % of weights)
        low_thr = np.percentile(weight, 10)
        candidates = np.where(weight <= low_thr)[0]
        if candidates.size == 0:
            # fallback to a random different index
            candidates = np.arange(n)
        receiver = int(rng.choice(candidates))
        if donor == receiver:
            continue
        if a[donor] <= 1e-12:
            continue

        frac = rng.random() * max_frac
        delta = a[donor] * frac
        a[donor] -= delta
        a[receiver] += delta

        new_val = _score(a)
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] HillClimb improved={best_val:.6f}")
        else:
            # revert move
            a[receiver] -= delta
            a[donor] += delta

    return best_a, best_val


# ----------------------------------------------------------------------
# Power‑law generator (used for the coarse initialization)
# ----------------------------------------------------------------------


def _power_law_seq(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create (i+1)^(-p) of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for the best non‑negative step‑function within the given time budget.
    The routine:
      1. scans a small exponent grid for a long power‑law,
      2. refines the best exponent with Adam + sub‑gradient descent,
      3. finishes with a fast hill‑climbing phase.
    The sequence with the smallest evaluation score seen before the deadline is
    returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Global time management
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length – large values give a tighter bound.
    # We use n = 50 000 which still keeps FFT sizes modest.
    # ------------------------------------------------------------------
    n = 50000
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Stage 1 – quick scan over power‑law exponents
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.55, 0.65, 21)   # fine enough around the theoretical optimum
    best_p = None
    best_score = np.inf
    best_seq = None

    for p in p_grid:
        a = _power_law_seq(n, p, target_sum)
        # tiny jitter to break perfect symmetry (helps later optimisation)
        a += rng.normal(scale=target_sum * 1e-4, size=n)
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_p = p
            best_seq = a
    print(f"[{time.time():.2f}] Power‑law scan finished – best p={best_p:.4f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimiser on the smooth max (soft‑max temperature annealed)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + min(300.0, (deadline - time.time()) * 0.5)
        best_seq, best_score = _adam_optimize(best_seq, sub_dead)

    # ------------------------------------------------------------------
    # Stage 3 – sub‑gradient descent on the true max entry
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + min(200.0, (deadline - time.time()) * 0.3)
        best_seq, best_score = _subgradient_descent(best_seq, sub_dead)

    # ------------------------------------------------------------------
    # Stage 4 – final hill‑climbing refinement (mass shuffling)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, deadline)

    # ------------------------------------------------------------------
    # Safety fallback – constant sequence (never needed unless everything fails)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.ones(n, dtype=np.float64) * np.sqrt(2.0 * n) / n

    # Trim trailing zeros (mainly for aesthetic output)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.2f}] Final best score ≈ {best_score:.6f} (length={best_seq.size})")
    return best_seq.tolist()
