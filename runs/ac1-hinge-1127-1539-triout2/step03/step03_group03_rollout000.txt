"""
Hybrid gradient‑plus‑greedy optimizer for the max‑self‑convolution problem.

The score returned by ``evaluate_sequence`` is      2·n·max(b) / (∑a)².
If we keep the ℓ¹‑norm fixed to √(2·n) the score equals max(b),
so the task is to minimise the largest entry of the auto‑convolution.

Algorithm
---------
1. Choose a random length n (2000‑12000) and initialise a non‑negative
   vector with the required ℓ¹‑norm (Dirichlet sample or an up‑sampled
   best sequence from previous runs).

2. Iteratively improve the vector:
   * **Smooth‑max gradient descent** – a soft‑max over the convolution
     entries yields differentiable weights wᵢ.  The gradient
     gⱼ = 2·∑ᵢ wᵢ₊ⱼ·aᵢ is computed by an FFT‑based correlation,
     a ← Π\_simplex(a – η·g) where Π projects onto the simplex
     {x≥0,∑x=√(2n)}.
   * **Greedy mass‑move** (probability p_swap) – locate the current
     maximal convolution index k, pick the pair (i,j) with i+j=k that
     contributes most (aᵢ·aⱼ).  Transfer a small amount of mass δ from
     the larger of the two to a random far index, re‑project to the
     simplex and accept if the true max is reduced (or with a
     Metropolis probability).

   Learning‑rate η decays, the soft‑max temperature α grows,
   and swap‑probability is modest (≈0.15).  Early stopping triggers
   after a long streak without improvement.

3. Run many short independent attempts within the time budget,
   always remembering the globally best sequence.

The implementation uses only NumPy (FFT) and pure Python, stays inside
the 1000 s limit, and prints progress.  The final sequence is returned
as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σ x = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute gradient g_j = 2 * Σ_i w_{i+j} * a_i
    using FFT based correlation.
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_len = _next_pow2(conv_len)
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # slice indices 2n-2 - j
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights: w_i = exp(alpha * conv_i) / Σ_j exp(alpha * conv_j)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    return exp_shifted / exp_shifted.sum()


# ----------------------------------------------------------------------
# One gradient‑descent step (returns updated a and true max)
# ----------------------------------------------------------------------


def _gradient_step(a: np.ndarray, lr: float, alpha: float) -> tuple[np.ndarray, float]:
    """
    Perform a projected gradient descent step on the smooth max norm.
    Returns the updated sequence and the *exact* maximal convolution entry.
    """
    n = a.shape[0]
    # 1. convolution of a with itself
    conv = _full_convolution(a)

    # 2. soft‑max weights
    w = _softmax_weights(conv, alpha)

    # 3. gradient w.r.t. a
    grad = _grad_from_weights(a, w)

    # 4. take a step and project onto simplex Σ a = √(2n)
    a_new = a - lr * grad
    S = np.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)

    # 5. evaluate true max for bookkeeping
    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max


# ----------------------------------------------------------------------
# Greedy mass‑move (helps escape plateaus)
# ----------------------------------------------------------------------


def _greedy_mass_move(a: np.ndarray, rng: np.random.Generator,
                     max_conv: np.ndarray, S: float) -> np.ndarray:
    """
    Identify the current maximal convolution entry and shift a small
    amount of mass away from the pair that contributes most to it.
    Returns a new candidate vector (projected onto the simplex).
    """
    n = a.shape[0]
    k = int(np.argmax(max_conv))            # index of max entry
    # valid i such that 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_vals = np.arange(i_min, i_max + 1)
    j_vals = k - i_vals
    contributions = a[i_vals] * a[j_vals]
    # pair with the largest contribution
    idx = int(np.argmax(contributions))
    i_star = i_vals[idx]
    j_star = j_vals[idx]

    # choose the larger of the two as source
    src = i_star if a[i_star] >= a[j_star] else j_star
    if a[src] <= 0.0:
        return a  # nothing to move

    # take a modest fraction of the source mass
    delta = min(a[src] * 0.08, a[src])

    # choose a destination far from src (to spread mass)
    dest = rng.integers(0, n)
    # ensure dest not identical to src; if too close, shift further
    if dest == src or abs(dest - src) < n // 10:
        dest = (src + n // 2) % n

    a_candidate = a.copy()
    a_candidate[src] -= delta
    a_candidate[dest] += delta
    # re‑project to keep the exact ℓ¹‑norm
    a_candidate = _project_simplex(a_candidate, S)
    return a_candidate

# Local optimiser (gradient + occasional greedy moves)
# ----------------------------------------------------------------------


def _optimize_local(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence using smooth‑max gradient descent
    together with stochastic greedy mass moves.
    Returns best sequence seen and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # bookkeeping
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters
    lr = 0.35                # learning rate
    lr_decay = 0.9994
    alpha = 1.0              # temperature for soft‑max
    alpha_mul = 1.0015
    swap_prob = 0.15         # probability of a greedy mass move per iteration
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # --- gradient descent step ---
        a, cur_max = _gradient_step(a, lr, alpha)

        # keep best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        if rng.random() < swap_prob:

            conv_now = _full_convolution(a)
            cand = _greedy_mass_move(a, rng, conv_now, S)
            cand_max = np.max(_full_convolution(cand))
            # accept if improves, otherwise Metropolis acceptance
            if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.04, 1e-12)):
                a = cand
                cur_max = cand_max
                if cur_max < best_score:
                    best_score = cur_max
                    best_a = a.copy()
                    last_improve = it

        # --- schedule updates ---
        lr *= lr_decay
        alpha *= alpha_mul

        # early stop after many iterations without improvement
        if it - last_improve > 2500:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the maximal self‑convolution entry.  The routine runs many independent
    restarts, each performing a hybrid gradient / greedy optimisation,
    and returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # If a previous best exists, keep it for up‑sampling
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempts = 0

    while time.time() < deadline:
        attempts += 1

        # --------------------------------------------------------------
        # Choose problem size (length).  Occasionally sample very large n.
        # --------------------------------------------------------------
        if rng.random() < 0.2:               # bias a bit towards larger sizes
            n = int(rng.integers(8000, 15000))
        else:
            n = int(rng.integers(2000, 10000))

        # --------------------------------------------------------------
        # Initialise a candidate vector
        # --------------------------------------------------------------
        S = np.sqrt(2.0 * n)
        if base_seq is not None and base_seq.size >= n // 2:
            # Upsample the known good shape by linear interpolation
            positions = np.linspace(0, base_seq.size - 1, n)
            a0 = np.interp(positions,
                           np.arange(base_seq.size),
                           base_seq).astype(float)
        else:
            a0 = rng.random(n)

        # enforce non‑negativity and normalise to required ℓ¹‑norm
        a0 = np.clip(a0, 0.0, None)
        a0_sum = a0.sum()
        if a0_sum == 0.0:
            a0 = np.full(n, 1.0)
            a0_sum = a0.sum()
        a0 = a0 / a0_sum * S

        # --------------------------------------------------------------
        # Allocate a small time slice for this attempt (≈ 12 s)
        # --------------------------------------------------------------
        slice_deadline = min(time.time() + 12.0, deadline)

        # --------------------------------------------------------------
        # Optimise locally
        # --------------------------------------------------------------
        a_opt, score = _optimize_local(a0, slice_deadline, rng)

        # --------------------------------------------------------------
        # Update global best
        # --------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempts} n={n} "
                  f"new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the provided black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen
        return [1.0]

    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f} "
          f"(internal max = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
