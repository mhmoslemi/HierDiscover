"""
Propose a non‑negative step‑function (vector of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The evaluation reduces to `max(conv(a,a))` when the ℓ₁‑norm of `a`
is fixed to √(2 n).  The algorithm is a pure projected sub‑gradient
method on the simplex:

1. Initialise a candidate (the previously found best, a parametrised
   “U‑shape”, or a random vector) and project it onto the ℓ₁‑simplex
   Σa = √(2 n).

2. Repeatedly
   * compute the full linear self‑convolution (FFT),
   * locate every index where the convolution attains its maximum,
   * form the exact sub‑gradient of `max(conv)` (the gradient of the
     active convolution entries),
   * take a decreasing step in the opposite direction,
   * project back onto the simplex.

   Step‑size = η₀ / sqrt(t) with η₀ = 0.5·√(2 n) / n.  The schedule
   guarantees ever smaller steps, letting the iteration settle.

3. If progress stalls for many iterations, inject a small random
   perturbation (or restart from a fresh U‑shape) to escape plateaus.

4. After the time budget expires, perform a short random‑pair‑transfer
   polishing pass.

The routine respects a 1000 s budget, prints occasional progress
messages and finally returns the best sequence encountered as a plain
Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a real vector using FFT.
    Returns the full length‑2·n‑1 result."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex
       {x | x_i ≥ 0, Σ x_i = target} (Duchi et al., 2008)."""
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a small amount of mass from a random index i to a random index j,
    then project back onto the simplex."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """Symmetric U‑shaped profile.
    kind ∈ {linear, quadratic, exponential}.
    `param` controls how strongly the ends are lifted."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError(f"unknown shape kind {kind}")

    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def subgradient_max(a: np.ndarray, conv: np.ndarray) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient of `max(conv(a,a))`.
    Returns (gradient, current maximal convolution value)."""
    n = a.shape[0]
    max_val = conv.max()
    # all positions attaining the maximum (could be more than one)
    active = np.flatnonzero(conv == max_val)

    grad = np.zeros_like(a)

    for k in active:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i = np.arange(low, high + 1, dtype=int)
        j = k - i
        eq = i == j
        # non‑diagonal contributions: ∂/∂a_i a_i a_j = a_j
        grad[i[~eq]] += a[j[~eq]]
        # diagonal contributions: ∂/∂a_i a_i² = 2 a_i
        grad[i[eq]] += 2.0 * a[i[eq]]

    # average over all active indices – improves stability
    grad /= max(1, len(active))
    return grad, max_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function minimising the evaluation function.
    Returns the best sequence found before the time budget expires."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    # --------------------------------------------------------------
    # Problem size – use previously discovered construction if present
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        target_sum = math.sqrt(2.0 * n)
        a = np.array(base, dtype=np.float64)
        a = project_onto_simplex(a, target_sum)
        print(f"[{time.time():.2f}] Using supplied base of length {n}")
    else:
        # a comfortable size that gives a rich search space
        n = np.random.randint(7500, 8700)
        target_sum = math.sqrt(2.0 * n)
        # start from a mildly lifted linear U‑shape
        a = generate_U_shape(n, target_sum, kind="linear", param=2.0)
        print(f"[{time.time():.2f}] No base found – created fresh size {n}")

    # initial evaluation
    conv = conv_self_fft(a)
    best_score = conv.max()
    best_seq = a.copy()
    print(f"[{time.time():.2f}] Initial max‑convolution {best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient descent on the simplex
    # --------------------------------------------------------------
    iteration = 0
    no_improve = 0
    # initial step size (≈ 0.5·Σa / n)
    eta0 = 0.5 * target_sum / n
    min_eta = 1e-7

    while time.time() < deadline:
        # compute sub‑gradient of the current max‑convolution
        grad, cur_max = subgradient_max(a, conv)

        # keep the best vector seen so far
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iteration:06d} – new best {best_score:.6f}")
        else:
            no_improve += 1

        # step‑size schedule – diminishing with √t
        eta = eta0 / math.sqrt(iteration + 1)
        eta = max(eta, min_eta)

        # projected gradient step (simple Euclidean projection)
        a_candidate = a - eta * grad
        a_candidate = project_onto_simplex(np.maximum(a_candidate, 0.0), target_sum)

        # prepare next iteration
        a = a_candidate
        conv = conv_self_fft(a)          # recompute the convolution
        iteration += 1

        # occasional stochastic perturbation to prevent getting stuck
        if np.random.rand() < 0.008:
            a = random_pair_transfer(a, target_sum, scale=0.015)
            conv = conv_self_fft(a)

        # if we have been idle for a long time, shake the current best
        if no_improve >= 5000:
            # small Gaussian jitter around the best known vector
            jitter = np.random.randn(n) * 5e-4
            a = best_seq + jitter
            a = project_onto_simplex(np.maximum(a, 0.0), target_sum)
            conv = conv_self_fft(a)
            no_improve = 0
            print(f"Iter {iteration:06d} – jitter applied")

        # early exit when the target bound is achieved
        if best_score < 1.503:
            print(f"[{time.time():.2f}] Target bound reached ({best_score:.6f}) – stopping")
            break

    # --------------------------------------------------------------
    # Final polishing – a short random‑pair‑transfer pass
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting final polishing (best {best_score:.6f})")
    for _ in range(200):
        if time.time() >= deadline:

            best_score = cand_score
            best_seq = cand.copy()
            print(f"  Polishing improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Search finished – final best {best_score:.6f}")
    return best_seq.tolist()
