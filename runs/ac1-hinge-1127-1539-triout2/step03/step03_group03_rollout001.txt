"""
Propose a candidate step‑function that (hopefully) beats the previous
score ~1.516.  The algorithm combines three ideas:

1. **Simplex projection** – the L¹ norm is kept fixed to √(2·n) after
   every update, because the evaluation score simplifies to the maximal
   entry of the self‑convolution when this normalization holds.

2. **Two‑phase gradient descent** – first a *soft‑max* approximation of the
   infinity‑norm is optimized, gradually sharpening the temperature
   (parameter α).  When α becomes large (≈ 30) the soft‑max turns into the
   exact sub‑gradient of the max‑convolution, and we continue with that
   exact sub‑gradient.

3. **Guided random mass moves** – occasionally a small amount of mass is
   moved from an index that participates heavily in the currently maximal
   convolution lag to a random index.  The move is accepted if it improves
   the max‑convolution or with a Metropolis probability that decays with
   the worsening amount.

Many short independent restarts with random problem sizes (n≈2000–8000)
are performed until the time budget expires.  The best sequence seen
throughout the run (measured by the true max‑convolution) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Geometry: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using an FFT (real‑to‑real).
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient term g_j = 2·Σ_i w_{i+j}·a_i.
    a has length n, w has length 2n‑1.
    """
    n = a.shape[0]
    # correlation a * w (w reversed) via FFT
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # pick the slice that corresponds to the required indices
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max on the convolution vector.
    w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)
    """
    if alpha <= 0:
        # Degenerate case – uniform weights
        return np.full_like(conv, 1.0 / conv.size, dtype=float)

    scaled = alpha * conv
    # Numerical stability: shift by max
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _subgradient_weights(conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑norm: distribute weight uniformly among
    all entries attaining the maximum.
    """
    max_val = np.max(conv)
    mask = (conv == max_val)
    w = mask.astype(float)
    cnt = w.sum()
    if cnt == 0:
        # should not happen
        return np.full_like(conv, 1.0 / conv.size, dtype=float)
    w /= cnt
    return w


# ----------------------------------------------------------------------
# One optimisation step (gradient or sub‑gradient)
# ----------------------------------------------------------------------


def _step(a: np.ndarray,
          lr: float,
          alpha: float,
          use_soft: bool) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Perform a single update and return the new vector, its true maximal
    convolution entry, and the convolution vector of the *old* a (used for
    guided random moves).
    """
    conv = _full_convolution(a)

    if use_soft:
        w = _softmax_weights(conv, alpha)
    else:
        w = _subgradient_weights(conv)

    grad = _conv_with_weights(a, w)

    a_new = a - lr * grad
    # projection back onto the simplex Σa = √(2·n)
    n = a.shape[0]
    S = math.sqrt(2.0 * n)
    a_new = _project_simplex(a_new, S)

    true_max = np.max(_full_convolution(a_new))
    return a_new, true_max, conv


# ----------------------------------------------------------------------
# Local optimiser (one restart)
# ----------------------------------------------------------------------


def _optimise_one_restart(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using the two‑phase gradient/descent scheme and
    occasional guided random moves.  Returns the best vector encountered
    and its maximal self‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # best so far inside this restart
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters
    lr = 0.5                     # initial learning rate
    lr_decay = 0.9995
    alpha = 0.5                  # start temperature
    alpha_mul = 1.04             # temperature growth per iteration
    alpha_switch = 30.0          # when to switch to exact sub‑gradient
    swap_prob = 0.06             # chance of a guided random move per iter
    swap_frac = 0.07             # fraction of source mass moved
    last_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # -------- gradient / sub‑gradient step ----------
        use_soft = (alpha < alpha_switch)
        a, cur_max, conv = _step(a, lr, alpha, use_soft)

        # keep global best of this restart
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            last_improve = it

        # -------- guided random mass move ----------
        if rng.random() < swap_prob:
            # pick the lag that is currently maximal
            k = int(np.argmax(conv))                       # 0 … 2n‑2
            # Determine the range of i that contributes to this lag
            if k < n:
                i_min, i_max = 0, k
            else:
                i_min, i_max = k - n + 1, n - 1
            i = rng.integers(i_min, i_max + 1)              # source index
            j = rng.integers(0, n)                         # destination
            if i != j and a[i] > 0:
                delta = swap_frac * a[i] * rng.random()
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_simplex(a_candidate, S)
                cand_max = np.max(_full_convolution(a_candidate))
                # accept if improvement or with Metropolis probability
                if cand_max <= cur_max or rng.random() < math.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.04, 1e-12)):
                    a = a_candidate
                    cur_max = cand_max
                    conv = _full_convolution(a)
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        last_improve = it

        # -------- schedule updates ----------
        lr *= lr_decay
        if use_soft:
            alpha *= alpha_mul

        # early stop if no improvement for many iterations
        if it - last_improve > 2500:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Initial sequence generator
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).

    - If a base_seq is supplied and long enough, its leading part is used
      (with a tiny jitter).
    - Otherwise an exponential decay profile a_i ∝ exp(-λ·i) is sampled,
      where λ is drawn randomly.
    """
    S = math.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        a += rng.normal(scale=a.mean() * 0.001, size=n)   # tiny jitter
        a = np.clip(a, 0.0, None)
    else:
        # random exponential decay
        lam = rng.uniform(0.0005, 0.004)                 # shape parameter
        a = np.exp(-lam * np.arange(n, dtype=float))

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimizes the evaluation score.
    The function runs many short independent restarts, each consisting of
    a two‑phase gradient descent with occasional guided random mass moves.
    The best sequence found (as a plain list of floats) is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 4.0   # safety margin

    # Use the previously discovered best sequence if present
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = float('inf')
    attempt = 0

    while time.time() < deadline:
        attempt += 1

        # Problem size – explore a wide, but reasonable range
        n = rng.integers(2000, 8001)          # 2000 ≤ n ≤ 8000

        # Initialise
        a0 = _initial_sequence(base_seq, n, rng)

        # Allocate a slice of the remaining time (≈ 12 s per restart)
        slice_deadline = min(time.time() + 12.0, deadline)

        # Optimise this restart
        a_opt, score = _optimise_one_restart(a0, slice_deadline, rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} "
                  f"new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – a trivial uniform sequence
        n_fallback = 3000
        S_fallback = math.sqrt(2.0 * n_fallback)
        best_seq = np.full(n_fallback, S_fallback / n_fallback)

    final_seq = best_seq.tolist()
    external_score = evaluate_sequence(final_seq)   # black‑box
    print(f"[final] external score = {external_score:.6f} "

    return final_seq
