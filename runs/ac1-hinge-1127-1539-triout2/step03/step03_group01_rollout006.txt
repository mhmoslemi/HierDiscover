def compute_gradient(a, tau):
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)  # shape (2n-1)
    # compute cross-correlation: (a * w) [where cross-correlation mirrored]. Equivalent to convolution of a reversed with w:
    a_rev = a[::-1]
    # Use FFT len as next power of two for length len(w)+n-1
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    # cross-correlation result same length as w + n -1; extract positions from n-1 to n-1+n:
    g = 2.0 * conv[n-1:n-1+n]
    return g, b, max_b.

    a = a_initial.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    t = 0
    best_a = a.copy()
    best_max = np.max(_full_convolution_fft(a))
    # set initial tau maybe as best_max (like start from '+ large')
    tau = best_max

    while time.time() < time_end:
        t += 1
        # compute gradient and current max
        g, b, cur_max = _compute_gradient(a, tau)  # need to implement
        # Adam update (raw a)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new, S)
        # recompute max for new a
        b_new = _full_convolution_fft(a_new)
        cur_max_new = np.max(b_new)
        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()
        a = a_new
        # schedule: reduce tau slowly
        tau = max(tau * 0.9995, 1e-6)  # similar to previous but slower
        # optionally jitter:
        if rng.random() < 0.01:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0, None)
            a = _project_to_simplex(a, S)
        # optionally reduce lr over time
        lr *= 0.9999
    return best_a, best_max

def random_distribution(n, rng):
    typ = rng.choice(['uniform', 'exp', 'symexp', 'gaussian', 'triangular', 'power', 'cosine'])
    if typ == 'uniform':
        a = rng.random(n)
    elif typ == 'exp':
        lam = rng.uniform(0.001, 0.2)  # lambda
        a = np.exp(-lam * np.arange(n))
    elif typ == 'symexp':
        lam = rng.uniform(0.001, 0.2)
        idx = np.arange(n)
        center = (n-1)/2
        a = np.exp(-lam * np.abs(idx - center))
    elif typ == 'gaussian':
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n-1)/2
        a = np.exp(-0.5 * ((np.arange(n) - mu)/sigma)**2)
    elif typ == 'triangular':
        # ramp up then down linearly
        half = n // 2
        if n % 2 == 0:
            left = np.linspace(1, half, half)
            right = np.linspace(half, 1, half)
        else:
            left = np.linspace(1, half+1, half+1)
            right = np.linspace(half, 1, half)
        a = np.concatenate([left, right])
    elif typ == 'power':
        p = rng.uniform(0.0, 2.0)
        a = (np.arange(1, n+1) ** -p)
    elif typ == 'cosine':
        # raised cosine from 0 to pi
        a = (1 + np.cos(np.pi * (np.arange(n) - (n-1)/2) / (n/2))) / 2
        # Might produce negatives near edges, clip
        a = np.clip(a, 0, None)
    else:
        a = rng.random(n)
    return a

"""
Simple evolutionary search + Adam‑projected gradient refinement.

The program proceeds in two phases within the allotted time budget:

1️⃣  Randomly generate a large number of candidate height vectors.
    Each candidate is built from a cheap parametric family
    (exponential, symmetric‑exponential, Gaussian, triangular,
    power‑law, cosine, or uniform Dirichlet) with random parameters and
    random length.  After projection onto the ℓ₁‑simplex
    (∑a = √(2·n)) the maximal self‑convolution entry
    max(a∗a) (the true optimisation objective) is evaluated.
    The best candidate seen so far is retained.

2️⃣  The best candidate is refined by an Adam optimiser that minimises a
    smooth approximation of the ∞‑norm (soft‑max of the convolution
    entries).  After each Adam update the vector is projected back onto
    the simplex.  The temperature of the soft‑max and the learning‑rate
    are annealed.  The refinement runs until the remaining time budget
    is exhausted.

The final sequence is returned and also stored in the global variable
`height_sequence_1` for possible reuse by subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the ℓ₁‑simplex of sum S."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via a real FFT (O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution entry
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, tau: float) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Compute gradient of τ·log∑exp(b/τ) where b = a*a,
    together with the raw convolution b and its current maximum.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau < 1e-12:          # hard max ⇒ sub‑gradient
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal contribution (i=j) occurs only once -> gradient = 2*a_i
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = 2.0 * a[d]
        return g, b, max_b

    # soft‑max weights w_k ∝ exp(b_k / τ) (stable version)
    diff = b - max_b
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)          # shape (2n‑1,)

    # cross‑correlation a ⋆ w = Σ_j a_j w_{i+j}
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa_rev = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(Fa_rev * Fw, n=fft_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]   # length n

    return g, b, max_b


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max approximation of max(a*a) using Adam.
    Returns the best vector found and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6                      # initial learning rate
    t = 0

    # initialise temperature for soft‑max
    _, _, cur_max = _gradient_softmax(a, tau=0.0)   # hard max just to get a scale
    tau = cur_max * 0.5 + 1e-8

    best_a = a.copy()
    best_max = cur_max

    while time.time() < deadline:
        t += 1
        # ----- gradient step -------------------------------------------------
        g, b, cur_max = _gradient_softmax(a, tau)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, S)

        # ----- evaluation ----------------------------------------------------
        b_new = _full_convolution_fft(a_new)
        cur_max_new = np.max(b_new)

        if cur_max_new < best_max:
            best_max = cur_max_new
            best_a = a_new.copy()

        a = a_new

        # ----- schedules -----------------------------------------------------
        tau = max(tau * 0.9995, 1e-7)           # cooling of the soft‑max temperature
        lr *= 0.9999                           # gradual learning‑rate decay

        # occasional tiny Gaussian jitter to escape plateaus
        if rng.random() < 0.008:
            noise = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)

    return best_a, best_max


# ----------------------------------------------------------------------
# Generate a random parametric distribution of length n
# ----------------------------------------------------------------------
def _random_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create an unnormalised non‑negative vector using a random family."""
    typ = rng.choice(
        ["uniform", "exp", "symexp", "gaussian", "triangular", "power", "cosine"]
    )
    idx = np.arange(n, dtype=float)

    if typ == "uniform":
        a = rng.random(n)

    elif typ == "exp":
        lam = rng.uniform(0.001, 0.2)
        a = np.exp(-lam * idx)

    elif typ == "symexp":
        lam = rng.uniform(0.001, 0.2)
        centre = (n - 1) / 2.0
        a = np.exp(-lam * np.abs(idx - centre))

    elif typ == "gaussian":
        sigma = rng.uniform(0.05, 0.3) * n
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif typ == "triangular":
        half = (n + 1) // 2
        inc = np.linspace(1, half, half)
        if n % 2 == 0:
            dec = inc[::-1]
        else:
            dec = inc[:-1][::-1]
        a = np.concatenate([inc, dec])

    elif typ == "power":
        p = rng.uniform(0.0, 2.0)
        a = (idx + 1) ** (-p)

    elif typ == "cosine":
        # raised cosine – non‑negative on the interval
        a = (1.0 + np.cos(np.pi * (idx - (n - 1) / 2.0) / (n / 2.0))) / 2.0
        a = np.clip(a, 0.0, None)

    else:
        a = rng.random(n)

    # small random jitter before projection (helps diversity)
    if rng.random() < 0.3:
        a += rng.normal(scale=a.mean() * 0.01, size=n)
        a = np.clip(a, 0.0, None)

    return a
