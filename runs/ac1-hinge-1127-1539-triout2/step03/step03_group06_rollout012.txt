def conv_self(a):
    n = a.size
    nfft = 1 << ((2*n-1 - 1).bit_length())
    Fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(Fa * Fa, n=nfft)[:2*n-1]

def grad_from_weights(w, a):
    # compute g = 2 * conv(w, a_rev)[n-1 : 2*n-1]
    n = a.size
    a_rev = a[::-1]
    len_conv = 3*n - 2

    A_rev = np.fft.rfft(a_rev, n=nfft)
    conv = np.fft.irfft(W * A_rev, n=nfft)[:len_conv]
    g = 2.0 * conv[n-1:2*n-1]

def optimize_step_function(n, deadline):
    rng = np.random.default_rng()
    S = math.sqrt(2.0 * n)
    # initial random distribution
    a = rng.random(n)
    a = a / a.sum() * S

    best_max = float('inf')
    best_a = a.copy()
    # hyperparams
    lr = 0.5  # initial learning rate magnitude for exponentiated update exponent factor
    gamma = 0.5 # smoothness
    gamma_inc = 1.05  # increase each iteration
    min_lr = 1e-5
    # Use momentum maybe
    v = np.zeros_like(a)
    beta = 0.9
    iter = 0

    while time.time() < deadline - 0.2:
        iter += 1

        conv = conv_self(a)
        cur_max = conv.max()
        if cur_max < best_max - 1e-9:
            best_max = cur_max
            best_a = a.copy()

        # compute softmax weights
        # shift to max for numeric stability
        w = np.exp(gamma * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            # all underflow, reduce gamma
            gamma = max(gamma * 0.5, 1e-3)
            continue
        w = w / w_sum

        # compute gradient
        g = grad_from_weights(w, a)

        # momentum
        v = beta * v + (1 - beta) * g

        # exponentiated update
        a = a * np.exp(-lr * v)

        # renormalize L1 sum to S
        total = a.sum()
        if total > 0:
            a = a * (S / total)
        else:
            a = np.full_like(a, S / n)

        # anneal parameters
        lr = max(lr * 0.9999, min_lr)
        gamma = min(gamma * gamma_inc, 200.0)  # avoid overflow

        # jitter
        if rng.random() < 0.01:
            noise = rng.standard_normal(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0, None)
            total = a.sum()
            if total > 0:
                a = a * (S / total)

    return best_a, best_max

"""
Improved step‑function search.

The external score returned by ``evaluate_sequence`` equals the
maximum entry of the self‑convolution once the vector is normalised to
ℓ₁‑norm √(2 n).  We therefore minimise this maximum directly.

Our core optimiser works on a *smooth max* surrogate:
    Lγ(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k) .

For γ≈0 the surrogate is almost the average, for large γ it approaches
the true max.  We start with a small γ and increase it slowly (annealing).
The gradient of Lγ is a simple convolution, which we compute with FFTs
so that each iteration is O(n log n).  An exponentiated‑gradient update
keeps the iterates non‑negative; a renormalisation step enforces the
ℓ₁‑norm constraint.  Momentum, occasional jitter and a learning‑rate
decay help escape plateaus.

Several random restarts with different lengths are performed while the
global 1000 s budget allows.  The best sequence found is optionally
refined by a tiny linear‑program (if SciPy is available) that mimics the
alternating‑LP scheme described in the statement.  Finally a short
polish with the original EG optimiser is run before returning the
candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming refinement (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog

    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the (2n‑1)×n matrix H with (H·g)[k] = (f∗g)[k]."""
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i : i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2, max_steps: int = 5) -> np.ndarray:
    """One‑step LP refinement (Frank‑Wolfe style)."""
    if not _SCIPY_AVAILABLE or f.shape[0] > 3000:
        return f
    n = f.shape[0]
    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)  # maximise Σ g_i  ⇔ minimise –Σ g_i
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": 8},
            )
        except Exception:  # pragma: no cover
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of two real vectors using FFT."""
    L = x.shape[0] + y.shape[0] - 1
    nfft = 1 << (L - 1).bit_length()
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(X * Y, n=nfft)[:L]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a∗a (full length 2n‑1) via FFT."""
    return _full_convolve_fft(a, a)


def _grad_from_weights_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k w.r.t. a.
    Implements  g = 2·(w ∗ a_rev)[n‑1 : 2n‑1] .
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = _full_convolve_fft(w, a_rev)          # length = 3n‑2
    return 2.0 * conv[n - 1 : 2 * n - 1]          # n entries


# ----------------------------------------------------------------------
# Smooth‑max optimiser (exponentiated gradient with momentum)
# ----------------------------------------------------------------------
def _smooth_max_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
    lr_init: float = 0.6,
    gamma_init: float = 0.3,
    gamma_factor: float = 1.04,
    beta: float = 0.9,
    jitter_prob: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Returns (best_sequence, best_max_conv) starting from a0.
    The algorithm minimises the smooth‑max surrogate of the max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    best_seq = a.copy()
    best_max = float("inf")

    v = np.zeros_like(a)  # momentum buffer
    lr = lr_init
    gamma = gamma_init
    rng = np.random.default_rng()

    for it in range(max_iters):
        if time.time() >= deadline - 0.1:
            break

        # --- convolution and current objective ---
        conv = _conv_self_fft(a)
        cur_max = conv.max()

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_seq = a.copy()

        # --- smooth‑max weights (softmax) ---
        # shift by cur_max for numerical stability
        exp_arg = gamma * (conv - cur_max)
        # clip to avoid overflow in extreme cases
        exp_arg = np.clip(exp_arg, -700, 700)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # numerical trouble – back off gamma
            gamma = max(gamma * 0.5, 1e-3)
            continue
        w /= w_sum

        # --- gradient of the surrogate ---
        g = _grad_from_weights_fft(w, a)

        # --- momentum and multiplicative update ---
        v = beta * v + (1.0 - beta) * g
        a = a * np.exp(-lr * v)

        # --- maintain positivity (exp‑update may under‑flow) ---
        a = np.maximum(a, 1e-15)

        # --- renormalise to the required ℓ₁‑norm ---
        total = a.sum()
        if total > 0.0:
            a *= (S / total)
        else:
            a.fill(S / n)

        # --- schedule ---
        lr *= 0.9999
        gamma = min(gamma * gamma_factor, 200.0)

        # --- occasional jitter to escape tiny plateaus ---
        if rng.random() < jitter_prob:
            noise = rng.standard_normal(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            total = a.sum()
            if total > 0.0:
                a *= (S / total)

    return best_seq, best_max


# ----------------------------------------------------------------------
# A lightweight EG optimiser (used for final polishing)
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray):
    """Return (full conv, max entry, index of max) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_from_shifts(a: np.ndarray, shifts: np.ndarray):
    """Sub‑gradient for the sum of selected convolution entries."""
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in shifts:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g


def _run_optimize_one_start(
    a0: np.ndarray,
    n: int,
    deadline: float,
    top_k_start: int = 3,
    eta0: float = 0.8,
    beta: float = 0.9,
    decay: float = 0.99995,
    jitter_prob: float = 0.02,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent (with momentum) on the *max* convolution.
    Returns the best vector seen and its surrogate value (= max‑conv).
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)                # required ℓ₁‑norm
    best_local_max = float("inf")
    best_local_seq = a.copy()

    v = np.zeros_like(a)                 # momentum
    eta = eta0
    last_improve = 0
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        it += 1
        conv, cur_max, _ = _conv_max(a)

        # adaptive active set size
        top_k = top_k_start
        if it - last_improve > 2_000 and top_k < 30:
            top_k = min(30, top_k * 2)

        if conv.size <= top_k:
            shifts = np.arange(conv.size)
        else:
            shifts = np.argpartition(conv, -top_k)[-top_k:]

        g = _gradient_from_shifts(a, shifts)

        # momentum + multiplicative update
        v = beta * v + (1.0 - beta) * g
        a = a * np.exp(-eta * v)

        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # jitter
        if rng.random() < jitter_prob:
            noise = rng.standard_normal(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        eta *= decay

        if cur_max < best_local_max - 1e-12:
            best_local_max = cur_max
            best_local_seq = a.copy()
            last_improve = it

        if time.time() >= deadline - 0.1:
            break

    return best_local_seq, best_local_max

    else:

        return a.tolist()
