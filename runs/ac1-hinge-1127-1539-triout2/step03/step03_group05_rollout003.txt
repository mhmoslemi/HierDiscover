Goal: minimize 2 * n * max(conv) / (sum(a)^2). Since we can scale sequence arbitrarily (except for the clamp at 1000). If we scale a sequence by factor c>0, then sum scales by c, max(conv) scales by c^2 because convolution self: conv(c*a,c*a) = c^2 conv(a,a). So objective = 2n * max(conv) / (sum^2) = 2n * (c^2 max(conv))/ (c^2 sum^2) = 2n * max(conv)/sum^2 unaffected by scaling. Thus the objective is invariant under scaling. """
Propose a non‑negative step‑function (vector of heights) that minimizes  

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Broad parametric sweep** – try many simple families of “U‑shapes”
   (linear, quadratic, exponential, power‑law and cosine‑based) together
   with a few random Dirichlet vectors.  The best candidate is kept.

2. **Adam‑accelerated smoothed descent** – minimise a log‑sum‑exp
   approximation of the max‑convolution while staying on the ℓ₁‑simplex
   (`sum = sqrt(2 n)`).  The temperature (smoothing) is decreased
   progressively, and the learning‑rate follows an Adam schedule.
   Occasional random pair‑mass transfers and jitter are applied to
   escape plateaus.

3. **Fine‑grained sub‑gradient polishing** – once the temperature is
   tiny we switch to the exact sub‑gradient of the max convolution and
   finish with a short random‑perturbation pass.

All steps respect a 1000 s wall‑clock budget and the best sequence found
is returned as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of *a* using FFT.
    Returns an array of length 2*len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex {x | x_i ≥ 0, Σ x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index to another,
    then re‑project onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Symmetric “U‑shaped’’ sequence of length *n*.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls the edge enhancement.
    The result is projected onto the ℓ₁‑simplex (sum = target_sum).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_power_shape(
    n: int,
    target_sum: float,
    param: float = 1.0,
    exponent: float = 2.0,
) -> np.ndarray:
    """
    Power‑law U‑shape: 1 + param * d**exponent.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = 1.0 + param * (d ** exponent)
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def generate_cosine_shape(
    n: int,
    target_sum: float,
    param: float = 0.0,
) -> np.ndarray:
    """
    Cosine‑based U‑shape: 1 + param * (1 - cos(pi * d)).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid
    raw = 1.0 + param * (1.0 - np.cos(np.pi * d))
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (gradient, max_conv) where max_conv = max(conv(a,a)).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    tau = max(tau_factor * max_conv, 1e-12)

    # stable log‑sum‑exp weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w * a_rev) restricted to the central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a, a)).
    Returns (gradient, max_conv, idx_max).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv, idx_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The search respects a ~budget_s second wall‑clock limit and returns
    the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    # ----------- Problem size -------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_vec = np.array(base, dtype=np.float64)
    else:
        # Choose a length that balances search speed and quality
        n = np.random.randint(8000, 10000)
        base_vec = None

    target_sum = np.sqrt(2.0 * n)   # normalisation makes objective = max(conv)

    # ----------- Phase 0 – broad parametric sweep ----------------------------
    best_seq = np.full(n, target_sum / n)   # placeholder (uniform)
    best_score = float("inf")
    print(f"[{time.time():.2f}] Starting parametric sweep (n={n})")

    # Helper to test a candidate
    def test_candidate(cand: np.ndarray, label: str):
        nonlocal best_seq, best_score
        cur = conv_self_fft(cand).max()
        if cur < best_score:
            best_score = cur
            best_seq = cand.copy()
            print(f"  New best {label} → {best_score:.6f}")

    # Linear, quadratic, exponential U‑shapes
    for p in np.linspace(0.0, 5.0, 51):
        cand = generate_U_shape(n, target_sum, kind="linear", param=p)
        test_candidate(cand, f"linear(p={p:.2f})")
        if time.time() >= deadline:
            return best_seq.tolist()

    for p in np.linspace(0.0, 4.0, 41):
        cand = generate_U_shape(n, target_sum, kind="quadratic", param=p)
        test_candidate(cand, f"quadratic(p={p:.2f})")
        if time.time() >= deadline:
            return best_seq.tolist()

    for p in np.linspace(0.0, 2.5, 26):
        cand = generate_U_shape(n, target_sum, kind="exponential", param=p)
        test_candidate(cand, f"exp(p={p:.2f})")
        if time.time() >= deadline:
            return best_seq.tolist()

    # Power‑law shapes (different exponents)
    for exp in [1.5, 2.0, 3.0, 4.0]:
        for p in np.linspace(0.0, 6.0, 31):
            cand = generate_power_shape(n, target_sum, param=p, exponent=exp)
            test_candidate(cand, f"power(exp={exp},p={p:.2f})")
            if time.time() >= deadline:
                return best_seq.tolist()

    # Cosine‑based shapes
    for p in np.linspace(0.0, 5.0, 51):
        cand = generate_cosine_shape(n, target_sum, param=p)
        test_candidate(cand, f"cosine(p={p:.2f})")
        if time.time() >= deadline:
            return best_seq.tolist()

    # Random Dirichlet draws (few)
    for _ in range(12):
        raw = np.random.dirichlet(np.ones(n))
        cand = raw * target_sum
        test_candidate(cand, "dirichlet")
        if time.time() >= deadline:
            return best_seq.tolist()

    # Baseline from previous run if available
    if base_vec is not None:
        test_candidate(base_vec, "baseline")
        if time.time() >= deadline:
            return best_seq.tolist()

    # ----------- Phase 1 – Adam‑accelerated smoothed descent -----------------
    a = best_seq.copy()
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lr = 0.5 * (target_sum / n)   # initial learning rate
    tau_factor = 0.3              # start with relatively high temperature
    no_improve = 0

    print(f"[{time.time():.2f}] Entering Adam smoothed descent (init lr={lr:.4g})")

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smoothed_gradient(a, tau_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
        a = a_new

        # Exact evaluation (max of convolution)
        cur_score = conv_self_fft(a).max()
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {t:07d} – improved to {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedules
        if t % 400 == 0:
            lr *= 0.94
            tau_factor = max(0.001, tau_factor * 0.92)
            beta1 = max(0.5, beta1 * 0.995)

        # Random tiny mass transfer (exploration)
        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.015)

        # Jitter if stuck for a while

            jitter = np.random.randn(n) * 8e-4
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            no_improve = 0
            print(f"Iter {t:07d} – jitter applied")

        # Early exit if we already beat the target bound
        if best_score < 1.503:
            print(f"Target bound reached ({best_score:.6f}); stopping early.")
            break

    # ----------- Phase 2 – sub‑gradient refinement ---------------------------
    print(f"[{time.time():.2f}] Switching to sub‑gradient refinement (best {best_score:.6f})")
    eta_sub = 0.6 * (target_sum / n)
    sub_iter = 0
    while time.time() < deadline and sub_iter < 3500:
        sub_iter += 1
        grad, cur_max, _ = _subgradient_max(a)
        a = a - eta_sub * grad
        a = project_onto_simplex(np.maximum(a, 0.0), target_sum)

        cur_score = conv_self_fft(a).max()
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Sub‑grad {sub_iter:04d} – improved to {best_score:.6f}")
            # modestly increase step size when we make progress
            eta_sub = min(1.2 * eta_sub, 2.0 * (target_sum / n))
        else:
            if sub_iter % 600 == 0:
                eta_sub *= 0.88

        if best_score < 1.503:
            break

    # ----------- Phase 3 – final random polishing ---------------------------
    print(f"[{time.time():.2f}] Final polishing")
    for _ in range(80):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.006)
        cand_score = conv_self_fft(cand).max()
        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"  Polishing improved to {best_score:.6f}")

    print(f"[{time.time():.2f}] Search finished – final best score {best_score:.6f}")
    return best_seq.tolist()
