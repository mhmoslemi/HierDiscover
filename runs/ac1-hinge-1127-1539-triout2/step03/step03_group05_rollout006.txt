"""
Propose a non‑negative step‑function (vector of heights) that minimizes  

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.  

The algorithm works in three stages:

1. **Multiple random restarts** – each restart starts from a random point
   on the ℓ₁‑simplex (sum = √(2 n)).  The best sequence discovered so far
   is also used as a starting point.

2. **Adam optimisation of a log‑sum‑exp smoothing of the max‑convolution**.
   For a smoothing temperature τ the smoothed objective is  

        f_τ(a) = τ·log( Σ_i exp( (a∗a)_i / τ ) ).

   Its gradient can be written in closed form and is used with the
   Adam adaptive‑step optimiser.  τ is initialised as a small fraction of
   the current maximum convolution value and is decreased slowly,
   gradually turning the smoothed problem into the original one.

3. **Sub‑gradient refinement + random polishing** – after the Adam phase
   we perform a pure sub‑gradient descent on the true max‑convolution,
   followed by a short random‑pair‑mass‑transfer polishing loop.

All updates are projected back onto the ℓ₁‑simplex to keep the sum
fixed (the evaluation function is invariant under scaling).  The code
stops after the supplied time budget (≈ 1000 s) and returns the best
vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padded FFT)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x)
    m = len(y)
    L = _next_pow2(n + m - 1)
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[: n + m - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another,
    then re‑project onto the simplex to restore the ℓ₁‑norm.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def gradient_smoothed(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (grad, current_max, τ).
    """
    n = a.shape[0]
    conv = conv_fft(a, a)                     # length 2n‑1
    cur_max = conv.max()
    # smoothing temperature (never zero)
    tau = max(tau_factor * cur_max, 1e-12)

    # stable soft‑max weights
    w = np.exp((conv - cur_max) / tau)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (w * rev(a))   →   2 * conv(w, rev(a))[n‑1 : n‑1+n]
    a_rev = a[::-1]
    grad_full = conv_fft(w, a_rev)          # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, cur_max, tau


def adam_optimize(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  global_best: float,
                  max_iter: int = 20000):
    """
    Adam optimiser on the smoothed objective.
    Stops early if the time budget is exhausted or the objective falls
    below 1.5 (well under the current best known bound).
    Returns the best vector seen during this run and its (max‑conv) score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # step‑size relative to the typical magnitude of a_i
    base_lr = 0.3 * target_sum / n
    lr = base_lr

    # smoothing schedule
    tau_factor = 0.25

    best_a = a.copy()
    best_score = conv_fft(a, a).max()

    # bias‑correction powers
    beta1_pow = beta1
    beta2_pow = beta2

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad, cur_max, tau = gradient_smoothed(a, tau_factor)

        # Adam update (biased‑corrected version)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # bookkeeping
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            # early stop if we cross a promising bound
            if best_score < 1.5:
                break

        # decay schedules
        if it % 400 == 0:
            tau_factor = max(0.005, tau_factor * 0.96)
            lr *= 0.96

        # update bias‑correction powers
        beta1_pow *= beta1
        beta2_pow *= beta2

    return best_a, best_score


def subgradient_descent(a0: np.ndarray,
                        target_sum: float,
                        deadline: float,
                        max_iter: int = 3000):
    """
    Pure sub‑gradient descent on the true max‑convolution.
    Returns the best vector encountered and its max‑conv value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = conv_fft(a, a).max()

    step = 0.7 * target_sum / n  # relatively aggressive

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = conv_fft(a, a)
        cur_max = conv.max()
        idx_max = int(conv.argmax())

        # sub‑gradient for the maximal entry
        low = max(0, idx_max - (n - 1))
        high = min(idx_max, n - 1)
        idx_range = np.arange(low, high + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx_range] = 2.0 * a[idx_max - idx_range]

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # keep the best
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            # slowly increase step size when we are making progress
            step = min(step * 1.02, target_sum / n)

        # if we stall, shrink the step
        if it % 500 == 0:
            step *= 0.9

        # early exit if we have already reached a very good bound
        if best_score < 1.5:
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the given time budget and returns the
    best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        best_seq = np.array(base, dtype=np.float64)
    else:
        # a comfortable size – large enough to allow a good shape,
        # but small enough for fast FFTs.
        n = np.random.randint(7500, 8700)
        best_seq = None

    target_sum = np.sqrt(2.0 * n)          # the normalisation used by the evaluator

    # --------------------------------------------------------------
    # Initial candidate (uniform if nothing else is available)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # initial score (max of the self‑convolution)
    best_score = conv_fft(best_seq, best_seq).max()
    print(f"[{time.time():.2f}] Initial best score {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – multiple random restarts with Adam optimisation
    # --------------------------------------------------------------
    restarts = 6
    for r in range(restarts):
        if time.time() >= deadline:
            break

        # start from the current global best for the first restart,
        # random Dirichlet points for the others
        if r == 0:
            a0 = best_seq.copy()
        else:
            a0 = np.random.rand(n)
            a0 = project_onto_simplex(a0, target_sum)

        a_opt, score_opt = adam_optimize(a0, target_sum, deadline, best_score)

        if score_opt < best_score - 1e-12:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Restart {r} improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting sub‑gradient refinement (score {best_score:.6f})")
    best_seq, best_score = subgradient_descent(best_seq, target_sum, deadline)

    # --------------------------------------------------------------
    # Phase 3 – final random polishing (pair‑mass transfers)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Polishing phase (score {best_score:.6f})")
    while time.time() < deadline:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.006)
        cand_score = conv_fft(cand, cand).max()
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Polishing improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence encountered
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Search finished – final best score {best_score:.6f}")
    return best_seq.tolist()
