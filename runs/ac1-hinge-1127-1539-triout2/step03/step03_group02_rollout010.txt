def optimize_candidate(a0, target_sum, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr0 = 0.5 * target_sum / a.shape[0]   # approx step size
    tau_factor = 0.4   # initial
    t = 0
    best_seq = a.copy()
    best_score = compute_max_conv(a)
    while time.time() < deadline:
        conv = conv_self_fft(a)
        max_conv = conv.max()
        # compute gradient
        tau = max(tau_factor * max_conv, 1e-12)   # avoid zero
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # compute gradient via convolution of weights w with reversed a
        a_rev = a[::-1]
        L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
        grad = (2.0 / tau) * conv_full[a.shape[0]-1 : a.shape[0]-1 + a.shape[0]]
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = lr0 / (np.sqrt(t) + 1)   # maybe extra decay
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        # ensure non-negative
        a = np.maximum(a, 0.0)
        # projection onto simplex
        a = project_onto_simplex(a, target_sum)

        cur_max = compute_max_conv(a)
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            # optionally print progress
            print(f"Iter {t:07d} – best max_conv {best_score:.6f}")

        # adapt schedule
        if t % 2000 == 0:
            tau_factor = max(0.001, tau_factor * 0.8)
            lr0 *= 0.95

        # random jitter possibility
        if np.random.rand() < 0.01:
            jitter_scale = 0.001 * target_sum / a.shape[0]
            jitter = np.random.randn(a.shape[0]) * jitter_scale
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)

        # exit if max iterations or time close to deadline
        if t >= 20000:
            # might be enough
            break

    # Subgradient refinement
    # Use step size decreasing schedule
    sub_step = 0.02 * target_sum / a.shape[0]
    sub_iter = 0
    while time.time() < deadline and sub_iter < 8000:
        sub_iter += 1
        a, cur_max = subgradient_step(a, target_sum, sub_step)
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            print(f"Subgrad {sub_iter:06d} – improved max_conv {best_score:.6f}")
        if sub_iter % 500 == 0:
            sub_step *= 0.95
        if np.random.rand() < 0.02:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    # final polishing via hill climbing
    polishing_iters = 0
    while time.time() < deadline and polishing_iters < 5000:

                best_score = cand_score
                best_seq = candidate.copy()
                print(f"Polish big {polishing_iters} – improved to {best_score:.6f}")

    return best_seq, best_score

"""
Optimisation of the step‑function bound

The code searches for a non‑negative coefficient vector `a` of length *n*
(normalised to sum = √(2 n)) that minimises

    2 n·max(conv(a,a)) / (sum(a)²)   =   max(conv(a,a))

The strategy combines

* a quick parametric sweep (symmetric “U‑shapes” + monotone decays)
* a small set of random Dirichlet starts,
* a projected Adam optimiser that works on a smooth log‑sum‑exp
  approximation of the max‑convolution,
* a true sub‑gradient phase,
* a final hill‑climbing polishing with random pair‑transfers.

The optimiser respects a 1000 s wall‑time budget and always returns the
best sequence seen so far.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core linear‑algebra helpers
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of `a` using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all will be zero – fall back to uniform distribution
        return np.full_like(v, target / v.size)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full linear self‑convolution of `a`."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Simple stochastic moves
# ----------------------------------------------------------------------


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """
    Move a tiny amount of mass from one random index to another,
    then re‑project onto the ℓ₁‑simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Shape generators (used for the parametric sweep)
# ----------------------------------------------------------------------


def generate_symmetric_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ about the centre.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # ∈[0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    return project_onto_simplex(raw, target_sum)


def generate_monotonic_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Decreasing shape (large at the left, small at the right).
    kind ∈ {linear, exponential, power}.
    """
    idx = np.arange(n, dtype=np.float64)
    t = 1.0 - idx / (n - 1)                         # ∈[1,0]

    if kind == "linear":
        raw = 1.0 + param * t
    elif kind == "exponential":
        raw = np.exp(param * t)
    elif kind == "power":
        raw = t ** param + 1e-4
    else:
        raise ValueError("unknown monotonic kind")
    return project_onto_simplex(raw, target_sum)


# ----------------------------------------------------------------------
# Sub‑gradient step for the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray, target_sum: float, step: float) -> tuple[np.ndarray, float]:
    """
    One sub‑gradient descent step on `max(conv(a,a))`.
    Returns the updated vector and the pre‑step max‑convolution value.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()

    eps = max(1e-12, 1e-9 * max_conv)
    mask = (conv >= max_conv - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        mask[conv.argmax()] = 1.0
        cnt = 1.0

    a_rev = a[::-1]
    L = 1 << ((mask.size + a_rev.size - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(M * A_rev, n=L)[: mask.size + a_rev.size - 1]
    grad = (2.0 / cnt) * conv_full[n - 1 : n - 1 + n]

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, max_conv


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises a smooth approximation
    of `max(conv(a,a))`.  Returns the best vector seen and its
    max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.5 * target_sum / n          # base learning rate
    tau_factor = 0.4                    # initial smoothing factor

    best_seq = a.copy()
    best_score = compute_max_conv(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # self‑convolution and its maximum
        conv = conv_self_fft(a)
        max_conv = conv.max()

        # --- smoothed gradient (log‑sum‑exp) -------------------------
        tau = max(tau_factor * max_conv, 1e-12)
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (central n entries)
        a_rev = a[::-1]
        L = 1 << ((w.size + a_rev.size - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_full = np.fft.irfft(W * A_rev, n=L)[: w.size + a_rev.size - 1]
        grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]

        # --- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / (np.sqrt(t) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        cur_score = compute_max_conv(a)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam] iter {t:07d} – best max_conv {best_score:.6f}")

        # --- simple schedule tweaks ----------------------------------
        if t % 2000 == 0:
            tau_factor = max(0.001, tau_factor * 0.8)
            lr0 *= 0.95

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * (0.0004 * target_sum / n)
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single start vector
# ----------------------------------------------------------------------


def optimise_candidate(
    start_vec: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Runs the Adam phase, a sub‑gradient refinement and a final
    random‑pair polishing.  Returns the best vector found.
    """
    # ---- Phase 1: Adam with smooth approximation ----
    seq, best = adam_optimize(start_vec, target_sum, deadline, max_iters=20000)

    # ---- Phase 2: true sub‑gradient descent ----
    step = 0.02 * target_sum / seq.size
    sub_iter = 0
    while time.time() < deadline and sub_iter < 8000:
        sub_iter += 1
        seq, cur = subgradient_step(seq, target_sum, step)
        if cur + 1e-12 < best:
            best = cur
            seq = project_onto_simplex(seq, target_sum)
            print(f"[Subgrad] iter {sub_iter:06d} – new best {best:.6f}")

        if sub_iter % 500 == 0:
            step *= 0.95
        if np.random.rand() < 0.02:
            seq = random_pair_transfer(seq, target_sum, scale=0.004)

    # ---- Phase 3: hill‑climbing polishing ----
    polish_iter = 0
    while time.time() < deadline and polish_iter < 5000:
        polish_iter += 1
        cand = random_pair_transfer(seq.copy(), target_sum, scale=0.003)
        cand_score = compute_max_conv(cand)
        if cand_score + 1e-12 < best:
            seq = cand
            best = cand_score
            print(f"[Polish] iter {polish_iter:05d} – improved to {best:.6f}")

        # every few hundred steps try a slightly larger transfer
        if polish_iter % 500 == 0 and np.random.rand() < 0.3:
            cand = random_pair_transfer(seq.copy(), target_sum, scale=0.01)
            cand_score = compute_max_conv(cand)
            if cand_score + 1e-12 < best:
                seq = cand
                best = cand_score
                print(f"[Polish‑big] iter {polish_iter:05d} – improved to {best:.6f}")

    return seq, best

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The algorithm respects a ~1000 s time
    budget and returns the best sequence observed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        target_sum = math.sqrt(2.0 * n)
        init_from_base = np.array(base, dtype=np.float64)
    else:
        # pick a power‑of‑two around 8192 (FFT‑friendly)
        n_candidates = [4096, 8192, 16384]
        n = int(np.random.choice(n_candidates))
        target_sum = math.sqrt(2.0 * n)
        init_from_base = None

    # --------------------------------------------------------------
    # Phase 0 – generate a pool of starting vectors
    # --------------------------------------------------------------
    starters = []

    # uniform start (always useful)
    starters.append(np.full(n, target_sum / n))

    # symmetric sweep (U‑shapes)
    sym_kinds = ("linear", "quadratic", "exponential", "power", "inverse")
    sym_grids = {
        "linear": np.linspace(0.0, 8.0, 41),
        "quadratic": np.linspace(0.0, 6.0, 31),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 10),
        "inverse": np.linspace(0.5, 5.0, 10),
    }
    for kind in sym_kinds:
        for p in sym_grids[kind]:
            cand = generate_symmetric_shape(n, target_sum, kind, p)
            starters.append(cand)

    # monotone decreasing shapes
    mono_kinds = ("linear", "exponential", "power")
    mono_grids = {
        "linear": np.linspace(0.0, 8.0, 41),
        "exponential": np.linspace(0.0, 6.0, 31),
        "power": np.linspace(0.5, 5.0, 10),
    }
    for kind in mono_kinds:
        for p in mono_grids[kind]:
            cand = generate_monotonic_shape(n, target_sum, kind, p)
            starters.append(cand)

    # a few random Dirichlet‑type vectors
    for _ in range(5):
        rand = np.random.exponential(scale=1.0, size=n)
        rand = project_onto_simplex(rand, target_sum)
        starters.append(rand)

    # if a baseline is provided, add it as well
    if init_from_base is not None:
        starters.append(init_from_base)

    # --------------------------------------------------------------
    # Evaluate all starters (quick sweep) and keep the top few
    # --------------------------------------------------------------
    scores = []
    for i, v in enumerate(starters):
        sc = compute_max_conv(v)
        scores.append(sc)
        if time.time() >= deadline:
            break
    # pick the 5 best starters
    best_indices = np.argsort(scores)[:5]
    top_starters = [starters[i] for i in best_indices]

    # --------------------------------------------------------------
    # Phase 1 – optimise each starter, keep the overall best
    # --------------------------------------------------------------
    overall_best_seq = None
    overall_best_score = float("inf")

    for idx, start in enumerate(top_starters, 1):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.2f}] Optimising starter {idx}/{len(top_starters)}")
        seq, sc = optimise_candidate(start, target_sum, deadline)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** New global best = {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # fallback: uniform distribution (should never happen)
        overall_best_seq = np.full(n, target_sum / n)

    print(f"[{time.time():.2f}] Returning best sequence with max_conv {overall_best_score:.6f}")
    return overall_best_seq.tolist()
