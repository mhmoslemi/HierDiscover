"""
Improved step‑function search.

The algorithm repeatedly:
  * picks a length n and an initial vector (uniform, random, exponential,
    cosine‑shaped);
  * runs a fast exponentiated‑gradient descent on the maximal entry of the
    self‑convolution (the external score after normalising the ℓ₁‑norm to
    √(2 n));
    – the update uses a weighted sub‑gradient built from the *k* largest
      convolution entries (k≈0.5 % · n);
    – momentum (β≈0.9) and an exponential decay of the step‑size are
      employed;
    – occasional Gaussian jitter helps to escape plateaus;
  * keeps the best vector seen globally;
  * optionally refines a promising candidate with a tiny linear‑programming
    pass (SciPy’s high‑s performance, only for n≤3000).

When the time budget (≈ 1000 s) is almost exhausted a final polishing run
with smaller step‑size and a final LP refinement is performed and the
resulting sequence is returned.
"""

import time
import math
import numpy as np

# optional import – the LP refinement is only used when SciPy is available
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _conv_max(a: np.ndarray):
    """Return the full self‑convolution, its maximal entry and the index."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_from_shifts_weighted(a: np.ndarray, conv: np.ndarray,
                                  shifts: np.ndarray, cur_max: float):
    """
    Weighted sub‑gradient of the sum of the selected convolution entries.
    The weight of shift k is conv[k]/cur_max (∈(0,1]).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    # compute weights
    weights = conv[shifts] / cur_max
    for k, w in zip(shifts, weights):
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * w * a[partner[valid]]
    return g


def _run_optimize_one_start(a0: np.ndarray, n: int, deadline: float,
                            eta0: float = 0.9,
                            decay: float = 0.99995,
                            beta: float = 0.9,
                            top_k_frac: float = 0.005,
                            jitter_prob: float = 0.015,
                            jitter_scale: float = 0.02,
                            max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with momentum on the maximal self‑convolution.
    Returns the best vector seen (still ℓ₁‑norm = √(2 n)) and the corresponding
    maximal convolution value.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)         # target ℓ₁‑norm
    v = np.zeros_like(a)          # momentum buffer
    best_seq = a.copy()
    best_val = float('inf')
    it = 0
    eta = eta0

    # minimum number of active shifts – at least three to give a non‑zero
    # direction even when the convolution is already almost flat.
    min_shifts = 3

    while time.time() < deadline and it < max_iters:
        it += 1

        conv, cur_max, _ = _conv_max(a)

        # record best ever seen in this run
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()

        # --- active set -------------------------------------------------
        # number of shifts proportional to n, but never larger than the
        # whole convolution and never smaller than ``min_shifts``.
        k = int(max(min_shifts, min(conv.size, int(top_k_frac * conv.size))))
        # pick the k largest entries (argpartition is O(n))
        shifts = np.argpartition(conv, -k)[-k:]

        # --- weighted sub‑gradient ---------------------------------------
        g = _gradient_from_shifts_weighted(a, conv, shifts, cur_max)

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # multiplicative (exponentiated) update – guarantees non‑negativity
        a = a * np.exp(-eta * v)

        # renormalise to keep ℓ₁‑norm exactly S
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * jitter_scale
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # step‑size decay
        eta *= decay

        # early exit if we run out of time
        if time.time() >= deadline - 0.15:
            break

    return best_seq, best_val


def _build_conv_matrix(f: np.ndarray):
    """
    Build the dense matrix H (size (2n‑1)×n) such that H @ g = f * g.
    Used only for the LP refinement when n is modest.
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,
               max_steps: int = 5) -> np.ndarray:
    """
    Small linear‑programming refinement (highs solver).  Works only when
    SciPy is available and n≤3000.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg_i  ⇔ minimise –Σg_i
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 == 0.0:
            g = np.full(n, S / n)
        else:
            g = g0 * (S / s0)

        # convex combination and renormalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f

The evaluation function  

\[
E(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum a_i\bigr)^2}
\]

is *scale‑invariant*: multiplying the whole vector by a constant leaves the
score unchanged.  
Consequently we can **fix the ℓ₁‑norm** to  

\[
\|a\|_1 = \sqrt{2n}
\]

and minimise only the maximal entry of the self‑convolution.
For a fixed norm the external score equals this maximal entry, therefore the
search works directly on the surrogate loss `cur_max = max(a*a)`.

The optimiser is an **exponentiated‑gradient descent with momentum** :

* the gradient of a single convolution entry  
  \( (a*a)_k = \sum_i a_i a_{k-i} \) is  

  \[
  \frac{\partial (a*a)_k}{\partial a_i}=2\,a_{k-i}
  \]

  (zero when the index is out of range).  
  For the *max* loss we use a *weighted* sub‑gradient built from the
  `k` largest convolution entries – the larger a entry, the larger its
  weight.
* the update `a ← a · exp(‑η·v)` (element‑wise) keeps all components
  non‑negative.
* after each step the vector is renormalised to the required ℓ₁‑norm.
* occasional Gaussian jitter and an exponentially decaying step size help
  to escape plateaus.
* for short vectors ( ≤ 3000 ) a tiny **LP refinement** (SciPy‑highs)
  solves  

  \[
  \max\sum_j g_j\quad\text{s.t.}\; g_j\ge0,\;(a*g)_k\le M
  \]

  (with `M = max(a*a)`).  
  Mixing the LP solution back into the current vector often reduces `M`
  further.

The overall search:

1. **Baseline** – use the provided `height_sequence_1` if any.
2. **Stochastic restarts** – repeatedly pick a length `n` (2000…24000) and an
   initial shape (uniform, random, exponential, half‑sine).  
   Run the optimiser with a length‑dependent budget of iterations.
3. Keep the best vector seen so far, apply LP refinement when affordable.
4. **Final polishing** – a longer low‑step‑size run on the best vector,
   followed by a final LP pass.
5. Return the best sequence as a plain `list[float]`.

The whole routine respects the 1000 s wall‑clock limit, prints progress,
and works with only a single CPU.

```python
"""
Improved step‑function search.

The algorithm repeatedly:
  * picks a length n and an initial vector (uniform, random, exponential,
    cosine‑shaped);
  * runs a fast exponentiated‑gradient descent on the maximal entry of the
    self‑convolution (the external score after normalising the ℓ₁‑norm to
    √(2 n));
    – the update uses a weighted sub‑gradient built from the *k* largest
      convolution entries (k≈0.5 %·n);
    – momentum (β≈0.9) and an exponential decay of the step‑size are
      employed;
    – occasional Gaussian jitter helps to escape plateaus;
  * keeps the best vector seen globally;
  * optionally refines a promising candidate with a tiny linear‑programming
    pass (SciPy’s high‑s performance, only for n≤3000).

When the time budget (≈ 1000 s) is almost exhausted a final polishing run
with smaller step‑size and a final LP refinement is performed and the
resulting sequence is returned.
"""

import time
import math
import numpy as np

# optional import – the LP refinement is only used when SciPy is available
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _conv_max(a: np.ndarray):
    """Return the full self‑convolution, its maximal entry and the index."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    return conv, float(conv[max_idx]), max_idx


def _gradient_from_shifts_weighted(a: np.ndarray, conv: np.ndarray,
                                  shifts: np.ndarray, cur_max: float):
    """
    Weighted sub‑gradient of the sum of the selected convolution entries.
    The weight of shift k is conv[k]/cur_max (∈(0,1]).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = np.arange(n)
    weights = conv[shifts] / cur_max
    for k, w in zip(shifts, weights):
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * w * a[partner[valid]]
    return g


def _run_optimize_one_start(a0: np.ndarray, n: int, deadline: float,
                            eta0: float = 0.9,
                            decay: float = 0.99995,
                            beta: float = 0.9,
                            top_k_frac: float = 0.005,
                            jitter_prob: float = 0.015,
                            jitter_scale: float = 0.02,
                            max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with momentum on the maximal self‑convolution.
    Returns the best vector seen (still ℓ₁‑norm = √(2 n)) and the corresponding
    maximal convolution value.
    """
    a = a0.copy()
    S = math.sqrt(2.0 * n)           # target ℓ₁‑norm
    v = np.zeros_like(a)            # momentum buffer
    best_seq = a.copy()
    best_val = float('inf')
    it = 0
    eta = eta0
    min_shifts = 3

    while time.time() < deadline and it < max_iters:
        it += 1

        conv, cur_max, _ = _conv_max(a)

        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()

        # ---- active set -------------------------------------------------
        k = int(max(min_shifts,
                    min(conv.size, int(top_k_frac * conv.size))))
        shifts = np.argpartition(conv, -k)[-k:]

        # ---- weighted sub‑gradient ---------------------------------------
        g = _gradient_from_shifts_weighted(a, conv, shifts, cur_max)

        # momentum
        v = beta * v + (1.0 - beta) * g

        # multiplicative update (guarantees non‑negativity)
        a = a * np.exp(-eta * v)

        # renormalise to keep the ℓ₁‑norm exactly S
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional jitter
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * jitter_scale
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # step‑size decay
        eta *= decay

        # early stop if we are about to exceed the deadline
        if time.time() >= deadline - 0.15:
            break

    return best_seq, best_val


def _build_conv_matrix(f: np.ndarray):
    """
    Build the dense matrix H (size (2n‑1)×n) such that H @ g = f * g.
    Used only for the LP refinement when n is modest.
    """
    n = f.shape[0]
    m = 2 * n - 1
    H = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f
    return H


def _lp_refine(f: np.ndarray, deadline: float, t: float = 0.2,
               max_steps: int = 5) -> np.ndarray:
    """
    Small linear‑programming refinement (highs solver).  Works only when
    SciPy is available and n≤3000.
    """
    if not _SCIPY_AVAILABLE:
        return f
    n = f.shape[0]
    if n > 3000:
        return f

    S = math.sqrt(2.0 * n)

    for _ in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        H = _build_conv_matrix(f)
        c = -np.ones(n)                     # maximise Σg_i  ⇔ minimise –Σg_i
        A_ub = H
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:
            break

        if not res.success:
            break

        g0 = res.x
        s0 = g0.sum()
        g = np.full(n, S / n) if s0 == 0.0 else g0 * (S / s0)

        # convex combination and renormalisation
        f = (1.0 - t) * f + t * g
        f = f / f.sum() * S

    return f
```
