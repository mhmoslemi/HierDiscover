def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum(x) = S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]          # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution a * a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Initialise a feasible vector (non‑negative, ℓ₁‑norm = √(2·n))
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create an initial vector of length n with sum √(2·n)."""
    S = np.sqrt(2.0 * n)
    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # tiny jitter to break symmetry
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
    else:
        a = rng.random(n)        # Dirichlet(1) (uniform on simplex before projection)
    a = _project_to_simplex(a, S)
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on τ·log‑sum‑exp(b/τ)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a single optimisation run.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                # initial learning rate
    lr_decay = 0.9997

    # state for Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # bookkeeping of the best we have seen
    b = _full_convolution_fft(a)
    best_max = np.max(b)
    best_a = a.copy()

    # occasional simulated‑annealing temperature for mass swaps
    temp = best_max * 0.15
    temp_decay = 0.9995

    while time.time() < deadline:
        # --------------------------------------------------------------
        # compute convolution and its soft‑max weights
        # --------------------------------------------------------------
        b = _full_convolution_fft(a)
        max_b = np.max(b)

        # soft‑max temperature – we start with a fraction of the current max
        tau = max_b * 0.12
        if tau < 1e-9:
            tau = 1e-9

        # stable soft‑max
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)                # shape (2n‑1,)

        # --------------------------------------------------------------
        # gradient of the smoothed ∞‑norm:  g = 2 * (a ⋆ w)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        conv_len = len(w) + n - 1
        fft_len = 1 << ((conv_len - 1).bit_length())
        Fw = np.fft.rfft(w, n=fft_len)
        Fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:len(w) + n - 1]
        g = 2.0 * conv[n - 1:n - 1 + n]          # same length as a

        # --------------------------------------------------------------
        # Adam update + projection
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a = _project_to_simplex(a_new, S)

        # --------------------------------------------------------------
        # occasional explorations
        # --------------------------------------------------------------
        # random mass‑swap (simulated‑annealing)
        if rng.random() < 0.07:
            i = rng.integers(n)
            if a[i] > 0.0:
                j = rng.integers(n)
                if i != j:
                    delta = 0.05 * a[i] * rng.random()
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    cand_max = np.max(_full_convolution_fft(a_candidate))
                    accept = False
                    if cand_max <= max_b:
                        accept = True
                    else:
                        prob = np.exp(-(cand_max - max_b) / max(temp, 1e-12))
                        if rng.random() < prob:
                            accept = True
                    if accept:
                        a = a_candidate
                        max_b = cand_max
                        temp *= temp_decay

        # Gaussian jitter (tiny)
        if rng.random() < 0.015:
            jitter = rng.normal(scale=a.mean() * 0.015, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = _project_to_simplex(a, S)
            max_b = np.max(_full_convolution_fft(a))

        # decay learning rate
        lr *= lr_decay

        # --------------------------------------------------------------
        # record best
        # --------------------------------------------------------------
        cur_max = np.max(_full_convolution_fft(a))
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function with a very small maximal
    self‑convolution entry.  A collection of short Adam‑optimisation runs
    with random restarts (different lengths, random seeds) is performed
    within the allotted time; the best sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # try to re‑use a previously discovered construction
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # pick a length – explore a wide interval (800 … 5000)
        max_len = 5000
        if base_seq is not None:
            # allow a bit larger than previous best, but not too large
            max_len = max(max_len, int(base_seq.size * 1.5))
        n = rng.integers(800, max_len + 1)

        # initialise feasible vector
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a time slice for this restart (≈30 s, adapt to remaining)
        remaining = deadline - time.time()
        slice_time = min(30.0, remaining * 0.9)
        slice_deadline = time.time() + slice_time

        # optimise
        a_opt, local_max = _adam_optimize(a0, slice_deadline, rng)

        # update global best (the evaluation score is exactly the max entry)
        if local_max < best_overall_score:
            best_overall_score = local_max
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} "
                  f"new best max={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity check – evaluate via the provided black‑box
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        final_seq = best_overall_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_overall_score:.6f}) "
              f"length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq
        return final_seq
    else:
        # should never happen – fallback to a trivial sequence
        return [1.0]
