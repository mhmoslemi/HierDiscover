"""
Optimisation of a non‑negative step‑function (vector of heights) that
minimises 2 n·max(conv(a,a))/∑a².  Because the search keeps the ℓ₁‑norm
∑a = √(2 n), the objective reduces to minimising max(self‑convolution).

Algorithm
---------

1. **Initialisation**
   * If a previous construction `height_sequence_1` exists, start from it;
     otherwise draw a random symmetric “U‑shape” (linear / quadratic /
     exponential) and project onto the ℓ₁‑simplex.

2. **Adam on a smoothed objective**
   * The non‑smooth max is approximated by a log‑sum‑exp (temperature τ).
   * Gradient of the smoothed objective is computed analytically via FFT.
   * An Adam optimiser (adaptive moments) updates the vector, each step
     projected back onto the simplex.
   * τ and the learning‑rate decay gradually; occasional jitter and a
     mixture‑restart help escape plateaus.

3. **Sub‑gradient fine‑tuning**
   * Switch to the exact sub‑gradient of the max‑convolution entry.
   * A simple decreasing step size (with occasional increase when progress
     is made) further reduces the peak.

4. **Random polishing**
   * Small random pair‑mass transfers are accepted if they improve the
     maximum convolution value.

The whole search respects a 1000 s budget and always returns the best
vector encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x ≥ 0, Σx = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index to another,
    then re‑project onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_U_shape(
    n: int,
    target_sum: float,
    kind: str = "linear",
    param: float = 0.0,
) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length n.
    kind ∈ {linear, quadratic, exponential}.
    param ≥ 0 controls edge enhancement.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


def _smoothed_gradient(a: np.ndarray, tau_factor: float):
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    Returns (gradient, current max value).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)                     # length 2n‑1
    max_conv = conv.max()
    # smoothing temperature (never zero)
    tau = max(tau_factor * max_conv, 1e-12)

    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (2/τ) * (w * a_rev) restricted to central n entries
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[n - 1 : n - 1 + n]
    return grad, max_conv


def _subgradient_max(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)).
    Returns (gradient, max value, index of maximal entry).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_conv = conv.max()
    idx_max = int(conv.argmax())
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    idx_range = np.arange(low, high + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx_range] = 2.0 * a[idx_max - idx_range]
    return grad, max_conv, idx_max


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation metric.
    Returns the best vector found within the time budget.
    """
    np.random.seed(seed)
    # Small safety margin to avoid overrunning the allotted time
    deadline = time.time() + budget_s - 0.3

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, (list, tuple)) and len(base) > 0:
        n = len(base)
        base_vec = np.array(base, dtype=np.float64)
    else:
        # Use a size that is large enough to give a good “shape space’’
        n = np.random.randint(9000, 12001)
        base_vec = None

    target_sum = np.sqrt(2.0 * n)       # enforce ∑a = √(2 n) → objective = max(conv)

    # ------------------------------------------------------------------
    # Initial vector
    # ------------------------------------------------------------------
    if base_vec is not None:
        a = base_vec.copy()
    else:
        # Random symmetric U‑shape as a start
        kind = np.random.choice(["linear", "quadratic", "exponential"])
        param = np.random.uniform(0.0, 4.0) if kind != "exponential" else np.random.uniform(0.0, 2.0)
        a = generate_U_shape(n, target_sum, kind=kind, param=param)

    a = project_onto_simplex(np.maximum(a, 0.0), target_sum)

    # Track the best solution seen so far
    best_seq = a.copy()
    best_score = conv_self_fft(best_seq).max()
    print(f"[{time.time():.2f}] Init – n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimiser on the smoothed objective
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12 * (target_sum / n)          # base learning rate (≈ 1‑2 % of typical entry)
    tau_factor = 0.25                       # initial smoothing factor
    min_tau_factor = 1e-5
    stagnation = 0
    iter_idx = 0
    max_iter = 500_000     # generous upper bound; loop ends by time budget

    while time.time() < deadline and iter_idx < max_iter:
        # Gradient of the smoothed max‑convolution
        grad, cur_max = _smoothed_gradient(a, tau_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (iter_idx + 1))
        v_hat = v / (1.0 - beta2 ** (iter_idx + 1))

        lr = lr0 * (0.99 ** (iter_idx // 500))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # Evaluate the new max‑convolution
        max_val = conv_self_fft(a).max()
        iter_idx += 1

        # Record improvement
        if max_val + 1e-12 < best_score:
            best_score = max_val
            best_seq = a.copy()
            stagnation = 0
            # tighten smoothing as we approach a better point
            tau_factor = max(min_tau_factor, tau_factor * 0.95)
            print(f"Iter {iter_idx:06d} – new best {best_score:.6f} (τ={tau_factor:.6f})")
        else:
            stagnation += 1

        # Occasionally inject jitter to escape flat regions
        if stagnation > 2000 and np.random.rand() < 0.02:
            jitter = np.random.randn(n) * (target_sum / n) * 0.05
            a = project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)
            stagnation = 0
            print(f"Iter {iter_idx:06d} – jitter applied")

        # If we have been stuck for a while, restart from a mixture
        if stagnation > 5000:
            cand = generate_U_shape(
                n,
                target_sum,
                kind=np.random.choice(["linear", "quadratic", "exponential"]),
                param=np.random.uniform(0.0, 3.0),
            )
            a = 0.7 * best_seq + 0.3 * cand
            a = project_onto_simplex(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0
            print(f"Iter {iter_idx:06d} – restart with mixture of best & random")

        # Global schedules
        if iter_idx % 1000 == 0:
            tau_factor = max(min_tau_factor, tau_factor * 0.97)

        # Early exit if target bound is already reached
        if best_score < 1.503:
            print(f"Target bound reached {best_score:.6f} at iter {iter_idx}")
            break

    # ------------------------------------------------------------------
    # Phase 2 – Sub‑gradient refinement (exact max entry)
    # ------------------------------------------------------------------
    eta_sub = 0.6 * (target_sum / n)
    sub_iter = 0
    while time.time() < deadline and sub_iter < 5000:
        grad, cur_max, _ = _subgradient_max(a)
        a = a - eta_sub * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        max_val = conv_self_fft(a).max()
        sub_iter += 1

        if max_val + 1e-12 < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"Sub‑grad iter {sub_iter:05d} – new best {best_score:.6f}")
            # When improving, slightly increase step size
            eta_sub = min(eta_sub * 1.02, target_sum / n)
        else:
            if sub_iter % 500 == 0:
                eta_sub *= 0.9

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 3 – Final random polishing
    # ------------------------------------------------------------------
    polish_iter = 0
    while time.time() < deadline and polish_iter < 200:
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.003)
        cand_score = conv_self_fft(cand).max()
        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"Polish {polish_iter:03d} – improved to {best_score:.6f}")
        polish_iter += 1

    return best_seq.tolist()
