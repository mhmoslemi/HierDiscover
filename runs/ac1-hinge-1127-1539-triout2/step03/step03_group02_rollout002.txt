"""
Propose a non‑negative step‑function (vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

which, after the normalisation sum(a)=√(2 n), is simply the
maximum entry of the linear self‑convolution.
The code combines

* a quick parametric sweep over several symmetric “U‑shapes’’,
* a large set of random Dirichlet initialisations,
* an Adam optimiser on a smoothed log‑sum‑exp approximation of the
  max‑convolution, and
* a short sub‑gradient polishing phase.

All updates are projected onto the ℓ₁‑simplex
{ x ≥ 0 , Σ x = √(2 n) }.
The routine respects the 1000 s time budget and returns the best
vector found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())            # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return float(conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the ℓ₁‑simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = target}."""
    if target <= 0:
        return np.zeros_like(v)

    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # extremely unlikely – fall back to uniform distribution
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple stochastic mass‑transfer operator
# ----------------------------------------------------------------------


def random_pair_transfer(a: np.ndarray, target_sum: float, scale: float = 0.01) -> np.ndarray:
    """Transfer a small fraction of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Parametric “U‑shapes’’ – symmetric functions useful as deterministic starts
# ----------------------------------------------------------------------


def generate_shape(n: int, target_sum: float, kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric shape of length n.
    kind ∈ {linear, quadratic, exponential, power, inverse}.
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed log‑sum‑exp approximation of max(conv)
# ----------------------------------------------------------------------


def smooth_logsumexp_grad(a: np.ndarray, max_conv: float, tau: float,
                         L: int) -> np.ndarray:
    """
    Compute the gradient of τ·log ∑ exp(conv/τ) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]

    # conv = a * a (linear convolution)
    conv = conv_self_fft(a)

    # soft‑max weights
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # grad = (2/τ) * (w * a_rev) where * denotes linear convolution
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * full[n - 1 : n - 1 + n]
    return grad

    while time.time() < deadline:
        conv = conv_self_fft(a)
        cur_max = float(conv.max())
        mask = (conv >= cur_max - 1e-9).astype(np.float64)
        cnt = mask.sum()
        if cnt == 0:
            mask[conv.argmax()] = 1.0
            cnt = 1.0
        a_rev = a[::-1]
        M = np.fft.rfft(mask, n=L_fft)
        A_rev = np.fft.rfft(a_rev, n=L_fft)
        full = np.fft.irfft(M * A_rev, n=L_fft)[: len(mask) + len(a_rev) - 1]
        grad = (2.0 / cnt) * full[n - 1 : n - 1 + n]

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        sub_iter += 1
        if cur_max + 1e-12 < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f" Sub‑grad {sub_iter:06d} – improved to {best_score:.6f}")

        if sub_iter % 400 == 0:
            step *= 0.96

        if np.random.rand() < 0.012:
            a = random_pair_transfer(a, target_sum, scale=0.004)

    # --------------------------------------------------------------
    # Phase 3 – final polishing via tiny random pair transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Phase 3 – polishing")
    for _ in range(60):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_seq.copy(), target_sum, scale=0.003)
        cand_score = compute_max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"  Polished → {best_score:.6f}")

    print(f"[{time.time():.2f}] Finished – best score {best_score:.6f}")
    return best_seq.tolist()
