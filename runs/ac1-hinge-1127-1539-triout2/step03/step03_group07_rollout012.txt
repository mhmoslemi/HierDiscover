"""
Implementation of a softmax-smooth gradient descent (Adam) with
continuation on the temperature parameter, followed by a stochastic
hill‑climbing refinement. The algorithm works for any length `n`,
uses only NumPy/FFT operations, and respects the non‑negative
constraints described by `evaluate_sequence`. It is designed to find
step‑functions with lower `2·n·max(a*a)/(sum(a)^2)` than the previous
state‑of‑the‑art method.

import time
import numpy as np

def conv_full(a):
    n = a.shape[0]
    conv_len = 2 * n - 1
    L = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[:conv_len]
    return conv

def conv_max(a):
    return float(np.max(conv_full(a)))

def correlation(a, w):
    n = a.shape[0]
    m = w.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(Fa * np.conj(Fw), n=L)
    offset = m - 1
    return corr[offset:offset + n]

def adam_descent(a0, deadline, max_iters=20000,
                 lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8,
                 alpha_start=0.5, alpha_inc=0.0005, alpha_max=12.0):
    a = a0.copy()
    n = a.shape[0]
    S = a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_max = conv_max(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute self‑convolution
        conv = conv_full(a)
        cur_max = float(conv.max())
        # update soft‑max temperature
        alpha = min(alpha_max, alpha_start + alpha_inc * it)
        # compute weights w_i = exp(alpha*(conv_i - cur_max))  (stable)
        w = np.exp(alpha * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / conv.shape[0])
        else:
            w /= w_sum
        # gradient (2 * correlation of a and w)
        grad = 2.0 * correlation(a, w)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # enforce non‑negativity

        if total == 0.0:

        # record best

            best_max = cur_max
            best_a = a.copy()
            print(f"[adam improve] t={time.time():.1f}s n={n} iter={it} max={cur_max:.6f}")
    return best_a, best_max

"""
Improved search for a non‑negative step function that minimises

    2·n·max(a*a) / (sum(a)^2)

The algorithm consists of two phases:

1. **Adam optimisation with a smooth approximation of the max**
   – The self‑convolution is obtained with an FFT.
   – A soft‑max (log‑sum‑exp) with a temperature that is slowly increased
     provides a dense sub‑gradient.
   – Adam (first‑ and second‑moment estimates) updates the vector while
     keeping it non‑negative and re‑normalising the L¹‑norm (the objective
     is scale‑invariant, any constant sum works).

2. **Local stochastic hill‑climbing (simulated annealing)**
   – Randomly moves a fraction of mass from one index to another.
   – Accepts improving moves always and worsening moves with a temperature
     that cools geometrically.

Both phases respect the security checks of ``evaluate_sequence``.  The
search runs as many independent restarts as time permits (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·n‑1) using an FFT.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    # FFT length large enough for the convolution and the later correlation
    L = 1 << ((3 * n - 2).bit_length())          # next power of two ≥ 3n
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return float(np.max(_conv_full(a)))


def _correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ w)[k] = Σ_j a_j · w_{j+k},
    for k = 0 … n‑1.  Returns a length‑n real vector.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = 1 << ((n + m - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(Fa * np.conj(Fw), n=L)
    offset = m - 1                # index that represents shift = 0
    return corr[offset:offset + n]


def _adam_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 30000,
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 0.3,
    alpha_inc: float = 0.0007,
    alpha_max: float = 14.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of max(a*a) using a smooth soft‑max.
    Returns the best vector (in terms of max‑convolution) and that maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = a.sum()                     # keep the L1‑norm fixed
    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment
    best_a = a.copy()
    best_max = _conv_max(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution ------------------------------------------------
        conv = _conv_full(a)
        cur_max = float(conv.max())

        # ----- soft‑max weights (smooth approximation of the max) ------------
        alpha = min(alpha_max, alpha_start + alpha_inc * it)
        # stability: shift by current maximum (≤0 after the shift)
        w = np.exp(alpha * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient = 2·(a ⋆ w) -------------------------------------------
        grad = 2.0 * _correlation(a, w)

        # ----- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- enforce non‑negativity & renormalise ---------------------------
        a = np.clip(a, 0.0, None)
        tot = a.sum()
        if tot == 0.0:
            a.fill(target_sum / n)
        else:
            a *= (target_sum / tot)

        # ----- record the best ------------------------------------------------
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            print(
                f"[adam improve]  t={time.time():.1f}s  n={n}  iter={it}  max={cur_max:.6f}"
            )

    return best_a, best_max


def _hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_iters: int = 25000,
    rng: np.random.Generator | None = None,
) -> tuple[np.ndarray, float]:

    Simple simulated‑annealing hill‑climbing around ``a_start``.
    Returns the best vector encountered and its max‑convolution.
    """
    if rng is None:

    a = a_start.copy()
