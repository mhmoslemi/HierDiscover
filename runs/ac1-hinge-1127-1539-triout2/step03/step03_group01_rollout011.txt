def _softmax_gradient(a, tau):
    b = _full_convolution_fft(a)
    max_b = np.max(b)
    if tau <= 1e-12:
        # hard gradient; choose max index k (largest)
        k = np.argmax(b)
        idx = np.arange(len(a))
        partner = k - idx
        valid = (partner >= 0) & (partner < len(a))
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        if k % 2 == 0:
            d = k // 2
            if d >= 0 and d < len(a):
                g[d] = a[d]
        return g
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        # cross-correlation a ⋆ w
        a_rev = a[::-1]
        m = 1 << ((len(w) + len(a) - 1).bit_length())
        Fw = np.fft.rfft(w, n=m)
        Fa_rev = np.fft.rfft(a_rev, n=m)
        conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + len(a) - 1]
        g = 2.0 * conv[len(a)-1:len(a)-1+len(a)]
        return g
```

def _optimize_adam(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    b = _full_convolution_fft(a)
    best_max = np.max(b)
    # Adam params
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iter = 0
    eta = 0.5
    eta_decay = 0.9995
    tau = best_max
    tau_decay = 0.9995
    while time.time() < deadline:
        # compute gradient
        g = _softmax_gradient(a, max(tau, 1e-12))
        iter += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** iter)
        v_hat = v / (1 - beta2 ** iter)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_proj = _project_to_simplex(a_new, S)
        a = a_proj
        # compute new max
        b = _full_convolution_fft(a)
        cur_max = np.max(b)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        # occasionally random swap to escape
        if rng.random() < 0.01:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 0:
                delta = min(epsilon_move * a[i], a[i])
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_to_simplex(a_candidate, S)
                b_candidate = _full_convolution_fft(a_candidate)
                cand_max = np.max(b_candidate)
                # accept if better, else accept with small probability
                if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / (0.1 * cur_max + 1e-12)):
                    a = a_candidate
                    cur_max = cand_max
                    if cur_max < best_max:
                        best_max = cur_max
                        best_a = a.copy()
        # schedule decay
        eta *= eta_decay
        tau *= tau_decay
        # optional jitter
        if rng.random() < 0.001:
            a += rng.normal(scale = a.mean() * 0.01, size=n)
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a, S)
    return best_a, best_max

"""
Propose a step‑function (non‑negative sequence) with a low
self‑convolution maximum.
The algorithm mixes several ideas:

* Random initialisations (dense Dirichlet‑like or sparse “spike” vectors).
* An Adam optimiser that minimises a smooth soft‑max approximation of the
  ∞‑norm of the self‑convolution.
* Occasional simulated‑annealing mass‑swaps to escape plateaus.
* Tiny Gaussian jitteres to avoid getting stuck in exact saddle points.

All vectors are projected onto the ℓ₁‑simplex
    {x ≥ 0,  Σ x = √(2·n)},
so the evaluation function reduces to the maximum convolution entry.
The routine runs many independent restarts within the time budget
(≈1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Simplex projection (Euclidean, O(n log n))
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto {x≥0, Σx = S}."""
    if S <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Full linear (non‑circular) convolution via a real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution) using FFTs."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# 3.  Gradient of the soft‑max approximation of max(b)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute ∇_a  τ·log Σ_k exp( (a*a)_k / τ ).
    Returns a vector of size n.
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau <= 1e-12:                # hard max – sub‑gradient at the peak
        k = np.argmax(b)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal term occurs only once
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]
        return g

    # smooth case – soft‑max weights
    diff = b - max_b                # for numerical stability
    exps = np.exp(diff / tau)
    w = exps / np.sum(exps)         # shape (2n-1,)

    # cross‑correlation a ⋆ w  (equivalent to Σ_j w_{i+j} a_j )
    a_rev = a[::-1]
    m = 1 << ((len(w) + n - 1).bit_length())
    Fw = np.fft.rfft(w, n=m)
    Fa_rev = np.fft.rfft(a_rev, n=m)
    conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n - 1]
    g = 2.0 * conv[n - 1:n - 1 + n]   # keep the central n entries
    return g


# ----------------------------------------------------------------------
# 4.  Local optimiser – Adam + occasional mass‑swaps
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth max‑convolution loss.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # bookkeeping of the best solution
    b = _full_convolution_fft(a)
    best_max = np.max(b)
    best_a = a.copy()

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # step‑size schedule (tuned for the problem)
    eta = 0.8
    eta_decay = 0.9997

    # temperature for the soft‑max (τ) – start at current max
    tau = best_max
    tau_decay = 0.9995

    # simulated‑annealing move parameters
    move_prob = 0.09
    eps_move = 0.06          # fraction of mass moved when a move is attempted
    temp_factor = 0.12       # initial temperature as a fraction of current max
    temp = best_max * temp_factor
    temp_decay = 0.9998

    # jitter parameters
    jitter_prob = 0.008
    jitter_scale = 0.015

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Adam gradient step on the soft‑max loss
        # --------------------------------------------------------------
        g = _softmax_gradient(a, max(tau, 1e-12))
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a = _project_to_simplex(a_new, S)

        # --------------------------------------------------------------
        # 2) Evaluate current max
        # --------------------------------------------------------------
        b = _full_convolution_fft(a)
        cur_max = np.max(b)

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # --------------------------------------------------------------
        # 3) Occasional simulated‑annealing mass‑swap
        # --------------------------------------------------------------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] > 0:
                j = rng.integers(n)
                if i != j:
                    delta = eps_move * a[i]
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = _project_to_simplex(a_candidate, S)

                    b_cand = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_cand)

                    accept = False
                    if cand_max <= cur_max:
                        accept = True
                    else:
                        prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                        if rng.random() < prob:
                            accept = True

                    if accept:
                        a = a_candidate
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                    temp *= temp_decay

        # --------------------------------------------------------------
        # 4) Tiny Gaussian jitter (helps to escape exact plateaus)
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _full_convolution_fft(a)
            cur_max = np.max(b)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # --------------------------------------------------------------
        # 5) Decay schedules
        # --------------------------------------------------------------
        eta *= eta_decay
        tau *= tau_decay

    return best_a, best_max


# ----------------------------------------------------------------------
# 5.  Random initialisation helpers
# ----------------------------------------------------------------------
def _initial_dense(rng: np.random.Generator, n: int) -> np.ndarray:
    """Dense Dirichlet‑like start (all entries >0)."""
    S = np.sqrt(2.0 * n)
    # Gamma(1) is equivalent to exponential → Dirichlet(1,…,1)
    draws = rng.gamma(1.0, size=n)
    a = draws / draws.sum() * S
    return a


def _initial_sparse(rng: np.random.Generator, n: int) -> np.ndarray:
    """Sparse start – a few spikes of equal height."""
    S = np.sqrt(2.0 * n)

    # typical number of spikes is O(sqrt(n)), but we allow a moderate range
    min_k = max(2, int(np.sqrt(n) * 0.8))
    max_k = max(min_k + 1, int(np.sqrt(n) * 2.2))
    K = rng.integers(min_k, max_k + 1)

    a = np.zeros(n)
    idx = rng.choice(n, size=K, replace=False)
    a[idx] = S / K
    return a


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a feasible vector of length n (ℓ₁‑norm = √(2·n)).
    If a base sequence is supplied we recycle a prefix, otherwise we draw
    a random dense or sparse start.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # tiny jitter to avoid perfect copies
        a += rng.normal(scale=a.mean() * 0.001, size=n)
        a = np.clip(a, 0.0, None)
        a = _project_to_simplex(a, S)
        return a

    # randomly decide between dense and sparse seeds
    if rng.random() < 0.55:
        a = _initial_dense(rng, n)
    else:
        a = _initial_sparse(rng, n)
    return a
