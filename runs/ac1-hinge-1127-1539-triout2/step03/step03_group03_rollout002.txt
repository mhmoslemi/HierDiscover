def _optimize_adam(a0, deadline, rng):
  a = a0.copy()
  n = len(a)
  S = np.sqrt(2.0 * n)
  m = np.zeros_like(a)
  v = np.zeros_like(a)
  beta1=0.9
  beta2=0.999
  eps = 1e-8
  lr=0.5
  lr_decay=0.9997 # slowly
  alpha = 0.5
  alpha_mul = 1.01
  max_alpha = 200.0
  best_a = a.copy()
  best_score = np.max(_full_convolution(a))
  best_iter = 0
  t = 0
  swap_prob = 0.07
  swap_frac = 0.02

  while time.time() < deadline:
    t += 1
    # conv
    conv = _full_convolution(a)

    # soft-max
    if alpha < 50:
        w = _smooth_weights(conv, alpha)
    else:
        idx = np.argmax(conv)
        w = np.zeros_like(conv)
        w[idx] = 1.0

    # grad
    grad = _conv_with_weights(a, w)

    # adam update
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step

    # project
    a = _project_simplex(a_new, S)

    # optionally clamp small negative due to numerics
    # compute true max and maybe track
    if t % 10 == 0:
        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            best_iter = t

    # random swap
    if rng.random() < swap_prob:
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i != j and a[i] > 0:
            delta = swap_frac * a[i] * rng.random()
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand, S)
            cand_max = np.max(_full_convolution(cand))
            if cand_max < cur_max or rng.random() < np.exp(-(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                a = cand
                cur_max = cand_max
                if cur_max < best_score:
                    best_score = cur_max
                    best_a = a.copy()

    # schedule
    lr *= lr_decay
    alpha = min(alpha * alpha_mul, max_alpha)

    if t - best_iter > 10000: # no improvement for many iter
        break

  return best_a, best_score

"""
Propose a candidate step‑function minimizing the maximal self‑convolution entry.

Algorithm
---------
1. Keep the L¹‑norm fixed to √(2·n) by Euclidean projection onto the simplex.
2. Optimise the smoothed max‑norm using the Adam optimiser.
   – The smoothing weight w_i = exp(α·c_i) / Σ_j exp(α·c_j) (c = a∗a).
   – α starts small (smooth landscape) and grows exponentially (≈ 1 → 200)
     so the optimisation gradually focuses on the true max‑norm.
3. Occasionally perform a random “mass‑swap’’ move (simulated‑annealing
   flavour) to escape plateaus.
4. Run many independent restarts with random lengths (≈ 1.5k–7k) and
   random initialisations (uniform Dirichlet or exponential).  The best
   sequence over all attempts is kept.
5. After the time budget is exhausted the best sequence is returned and
   checked with the black‑box ``evaluate_sequence`` function.

The whole optimisation respects the 1000 s budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient term g_j = 2 * Σ_i w_i * a_{i-j}.
    a has length n, w has length 2n‑1.
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # gradient slice: indices (2n‑2)‑j
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max utilities
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute soft‑max weights from a convolution vector.
    w_i = exp(alpha*conv_i) / Σ_j exp(alpha*conv_j)
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with L¹‑norm √(2·n).
    If a base_seq is supplied and long enough, its leading part (with tiny jitter)
    is used; otherwise a random Dirichlet/Exponential sample is taken.
    """
    S = np.sqrt(2.0 * n)

    if base_seq is not None and base_seq.size >= n:
        a = base_seq[:n].astype(float).copy()
        # tiny Gaussian jitter to avoid exact replicas
        a += rng.normal(scale=0.001 * a.mean() if a.mean() > 0 else 1e-6,
                         size=n)
        a = np.clip(a, 0.0, None)
    else:
        # randomly choose a distribution type
        mode = rng.integers(0, 3)
        if mode == 0:
            a = rng.random(n)                     # Dirichlet(1)
        elif mode == 1:
            a = rng.exponential(scale=1.0, size=n)  # exponential
        else:
            a = np.abs(rng.normal(loc=0.0, scale=1.0, size=n))  # half‑normal

    # normalise to the required L¹‑norm
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Local optimiser – Adam + soft‑max + random swaps
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence using Adam on the smoothed max‑norm.
    Returns the best sequence found and its true maximal self‑convolution
    value (the objective to be minimised).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyper‑parameters
    lr = 0.5
    lr_decay = 0.9997
    alpha = 0.5          # soft‑max temperature (low → smooth)
    alpha_mul = 1.015
    max_alpha = 250.0

    # random‑swap schedule
    swap_prob = 0.07
    swap_frac = 0.03

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    best_iter = 0
    t = 0

    while time.time() < deadline:
        t += 1

        # ----- 1. full self‑convolution -----
        conv = _full_convolution(a)

        # ----- 2. soft‑max (or hard‑max when α is large) -----
        if alpha < 80.0:
            w = _smooth_weights(conv, alpha)
        else:
            # approximate hard max: one‑hot at the maximal index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ----- 3. gradient of the smoothed max -----
        grad = _conv_with_weights(a, w)

        # ----- 4. Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # ----- 5. projection onto the simplex (keeps Σa = √(2n)) -----
        a = _project_simplex(a_new, S)

        # ----- 6. occasional random mass‑swap move -----
        if rng.random() < swap_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = swap_frac * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                cur_max = np.max(_full_convolution(a))
                # Metropolis‑like acceptance
                if cand_max <= cur_max or rng.random() < np.exp(
                        -(cand_max - cur_max) / max(cur_max * 0.02, 1e-12)):
                    a = cand
                    cur_max = cand_max

        # ----- 7. bookkeeping -----
        if t % 10 == 0:
            cur_max = np.max(_full_convolution(a))
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                best_iter = t

        # ----- 8. schedule updates -----
        lr *= lr_decay
        alpha = min(alpha * alpha_mul, max_alpha)

        # ----- 9. early stop if no progress for long -----
        if t - best_iter > 15000:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Runs many independent Adam‑based refinements (with random restarts)
    inside the given time budget and returns the best step‑function
    discovered (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 4.0   # a small safety margin

    # try to reuse a previously found construction, if any
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Repeated short‑to‑medium optimisation attempts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempt += 1

        # Choose a problem size.  A wide range gives the optimiser flexibility.
        n = rng.integers(1500, 7000)

        # Initialise a candidate (either from the previous best or at random)
        a0 = _initial_sequence(base_seq, n, rng)

        # Allocate a time slice for this attempt (≈ 30‑45 s)
        remaining = deadline - time.time()
        slice_time = min(45.0, remaining)
        slice_deadline = time.time() + slice_time

        a_opt, score = _optimize_adam(a0, slice_deadline, rng)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best score={best_overall_score:.6f}")

        # If we already beat the target we can stop early.
        if best_overall_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # Should never happen, but provide a fallback.
        return [1.0]

    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal max = {best_overall_score:.6f}) "
          f"length = {len(final_seq)}")
    return final_seq
