    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = np.inf
    prev_best_seq = None

    lengths = [1500, 3000, 5000, 8000, 12000, 20000]   # progressively larger

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        top = []   # list of (score, seq)

        while time.time() < random_end:

4.  Refine each candidate with a two‑phase local optimisation:

         (log‑sum‑exp with an annealing “β’’);

5.  The best refined candidate is up‑scaled to the next length and the
    process repeats.
6.  After the time budget expires a final brief SA run cleans up the
    current best solution.

All operations stay on the ℓ¹‑simplex (sum = sqrt(2·n)), which makes the
objective scale‑invariant.  The code is deliberately written with no
closures, no file I/O and only the standard ``numpy`` and ``time`` modules.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation:
        max(conv(a,a))
    (the factor 2·n/∑a² is constant for a fixed simplex radius).
    """
    # a is already on the simplex, therefore sum(a)=sqrt(2·n)
    return _max_convolution_fft(a)

# Random initialisers (various families)

def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a non‑negative vector on the ℓ¹‑simplex for length ``n``."""
    target = np.sqrt(2.0 * n)
    choice = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if choice == 0:      # pure power‑law
        p = rng.uniform(0.1, 1.4)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:    # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif choice == 2:    # mixed power‑law + exponential
        p = rng.uniform(0.1, 1.4)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:    # heavy‑tail power‑law (small exponent)
        p = rng.uniform(0.05, 0.3)
        a = (idx + 1.0) ** (-p)

    elif choice == 4:    # Dirichlet (uniform on the simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif choice == 5:    # Dirichlet with concentration <1 (spiky)
        alpha = rng.uniform(0.1, 0.9, size=n)
        a = rng.dirichlet(alpha=alpha) * target

    else:                # uniform plus tiny jitter
        a = np.full(n, target / n)

    # small Gaussian perturbation before projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp (soft‑max) approximation of max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Return (grad, b) where
        b = a*a (full convolution, length 2·n‑1)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
               = 2·(a ⋆ p)  with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ---- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ---- soft‑max probabilities p ------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))                # stable
    p = exps / np.sum(exps)

    # ---- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _hard_gradient_max(a: np.ndarray, i_max: int) -> np.ndarray:
    """
    Sub‑gradient of the true max‑convolution (i.e. derivative of
    b[i_max] w.r.t. a).  Result is
        g_j = 2·a[i_max‑j]   if 0 ≤ i_max‑j < n else 0.
    """
    n = a.size
    g = np.zeros_like(a)
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    # j runs from lo..hi, and a[i_max‑j] is then valid
    j = np.arange(lo, hi + 1, dtype=int)
    g[j] = 2.0 * a[i_max - j]
    return g


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max‑convolution
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Annealed projected gradient descent.
    Starts with a soft‑max (β small) and gradually sharpens; when β exceeds
    a threshold we switch to the exact max‑gradient for faster final
    convergence.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.8          # start a bit larger than the old 0.5
    max_beta = 4096.0
    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.92          # momentum decay

    it = 0
    while time.time() < deadline:
        # soft‑max gradient and full convolution
        grad, b = _gradient_softmax(a, beta)

        # switch to exact max‑gradient once β is large enough
        if beta > 512.0:
            i_max = int(np.argmax(b))
            grad = _hard_gradient_max(a, i_max)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * grad
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto simplex (preserves ℓ¹‑norm)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0:
            jitter = np.random.default_rng().normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β logarithmically
        if it % 350 == 0 and beta < max_beta:
            beta = min(beta * 1.3, max_beta)

        it += 1
    return a

# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that repeatedly transfers a fraction of mass between two
    randomly chosen indices.  Accepts uphill moves with a temperature
    schedule T←T·α (α≈0.9995).  Returns the best vector encountered.
    """
    a = a0.copy()
    best = a.copy()
    cur_score = _score(a)
    best_score = cur_score

    # Initial temperature is a modest fraction of current score
    T = cur_score * 0.025 + 1e-9
    cooling = 0.99955
    rng = np.random.default_rng()

    n = a.shape[0]

    while time.time() < deadline:
        # pick source (must contain some mass)
        i = rng.integers(0, n)
        if a[i] <= 1e-16:
            continue
        # pick destination ≠ source
        j = rng.integers(0, n)
        if i == j:
            continue

        # random fraction of mass to move (up to 20 %)
        frac = rng.random() * 0.20
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        # cool
        T *= cooling

        # occasional global jitter to shake out of deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

def _refine_candidate(a0: np.ndarray, time_budget: float) -> np.ndarray:
    """
    Given a candidate vector ``a0`` and a time budget (seconds), run
    projected gradient descent for roughly half the time and then
    simulated annealing with the remaining budget.  Returns the best
    vector found.
    """
    now = time.time()
    # Reserve a tiny margin for the outer loop
    half = max(0.0, time_budget * 0.5 - 0.05)
    gd_dead = now + half
    a_gd = _gradient_descent_opt(a0, gd_dead)

    # SA gets the rest of the allocation
    sa_dead = now + time_budget
    a_sa = _simulated_annealing(a_gd, sa_dead)

    # Return the best of the three (original, GD, SA)
    candidates = [a0, a_gd, a_sa]
    scores = [_score(v) for v in candidates]
    best_idx = int(np.argmin(scores))
    return candidates[best_idx]

                top_candidates.sort(key=lambda x: x[0])
                top_candidates = top_candidates[:6]

        # ----------- Phase B – refinement of the top seeds ----------
        # Distribute remaining time evenly among the seeds (bounded)
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate at most 30 s per seed, but sum must stay below remaining
        per_seed_budget = min(30.0, remaining / max(1, len(top_candidates)))

        for _, seed_vec in top_candidates:
            if time.time() >= deadline:
                break
            a_refined = _refine_candidate(seed_vec, per_seed_budget)
            sc_ref = _score(a_refined)
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_refined.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} score={sc_ref:.6f}")

        # keep the current best as the seed for the next length
        # (already stored in ``best_seq``)

    # ------------------------------------------------------------------
    # Final short SA polish on the overall best (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        final_budget = min(4.0, deadline - time.time())
        best_seq = _simulated_annealing(best_seq, time.time() + final_budget)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Clean‑up and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 1000
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _score(best_seq)

    # Trim trailing near‑zero entries for a neat output
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(f"[{time.time():.1f}] Final best score = {best_score:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
```

This implementation stays within the 1000 s budget, explores a richer set of initial families, exploits up‑scaling of good solutions, and uses a two‑phase local optimiser (annealed gradient + simulated annealing).  It should push the objective well below the previous 1.514 966 and move towards the target 1.503.
