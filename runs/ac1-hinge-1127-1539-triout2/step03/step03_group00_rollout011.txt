def upsample_seq(seq, new_n):
    old_n = len(seq)
    # use linear interpolation
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    seq_new = np.interp(x_new, x_old, seq)

    seq_new = np.maximum(seq_new, 0)
    # project to sum sqrt(2*new_n)
    target = np.sqrt(2*new_n)
    seq_new = _project_to_simplex(seq_new, target)

def _random_initial_sequence_extended(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0: # power-law
        p = rng.uniform(0.05, 1.5)
        idx = np.arange(n) + 1
        a = idx**(-p)
    elif choice == 1: # exponential
        lam = rng.uniform(0.001, 0.05)
        idx = np.arange(n)
        a = np.exp(-lam*idx)
    elif choice == 2: # mixed
        p = rng.uniform(0.05, 1.5)
        lam = rng.uniform(0.001, 0.05)
        idx = np.arange(n)
        a = (idx+1)**(-p) * np.exp(-lam*idx)
    elif choice == 3: # dirichlet
        a = rng.dirichlet(alpha=np.ones(n))
    elif choice == 4: # hann window
        i = np.arange(n)
        a = 0.5 - 0.5*np.cos(2*np.pi*i/(n-1))
    elif choice == 5: # hamming
        i = np.arange(n)
        a = 0.54 - 0.46*np.cos(2*np.pi*i/(n-1))
    elif choice == 6: # blackman
        i = np.arange(n)
        a = 0.42 - 0.5*np.cos(2*np.pi*i/(n-1)) + 0.08*np.cos(4*np.pi*i/(n-1))
    else:
        # gaussian
        sigma = rng.uniform(0.1, 0.4) * n
        mu = rng.uniform(0.0, 1.0) * n
        i = np.arange(n)
        a = np.exp(-0.5*((i-mu)/sigma)**2)
    # add minor noise to avoid zeros
    a += rng.normal(scale=1e-3, size=n)
    a = np.maximum(a, 0.0)
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)
    return a

def _adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0*n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_scale = target * 0.5  # base step factor

    beta = 0.5
    max_beta = 2048.0
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        g, _ = _gradient_softmax(a, beta)
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        # Adam step
        step = step_scale / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        # ensure nonnegativity along with projection to simplex
        a = _project_to_simplex(a, target)

        # schedule beta
        if iteration % 300 == 0 and beta < max_beta:
            beta = min(beta*1.5, max_beta)

        # jitter occasionally
        if iteration % 1000 == 0:
            jitter = np.random.default_rng().normal(scale=target*0.0005/(beta+1), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
    return a

"""
A multi‑stage search for a non‑negative step function that minimises
evaluate_sequence(...).  The algorithm works as follows

1.  Fast random sampling on several short lengths (≈800‑1500) to obtain a
    small pool of promising “shapes”.
2.  Upsample these shapes to larger lengths (2500‑8000) and add a few
    additional random seeds.
3.  For each seed run an Adam‑style projected gradient descent on a
    log‑sum‑exp smooth approximation of the ℓ∞‑norm of the self‑convolution.
4.  If time permits a short simulated‑annealing (SA) refinement is applied.
5.  The best sequence seen during the whole run is returned.

All vectors are projected onto the ℓ¹‑simplex with
∑a_i = √(2·n), because with this normalisation the evaluation
objective equals max(a*a).  The code contains several families of
initialisers (power‑law, exponential, mixed, Dirichlet, Hann,
Hamming, Blackman, Gaussian) and the optimisation parameters are
chosen to make good use of the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x | x_i ≥ 0, Σ x_i = s } (Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """
    Objective after ℓ¹‑normalisation:  max(a*a).  a is assumed to have
    sum(a) = √(2·len(a)).
    """
    return _max_convolution_fft(a)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution, length 2·n‑1)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p)   with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # ----- b = a*a ----------------------------------------------------
    size1 = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # ----- soft‑max probabilities p ------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))               # numeric stability
    p = exps / np.sum(exps)

    # ----- gradient g = 2·(a ⋆ p) --------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Random initialisers (a variety of families)
# ----------------------------------------------------------------------


def _random_initial_sequence_extended(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative sequence on the ℓ¹‑simplex.
    Several families are mixed to increase diversity.
    """
    target_sum = np.sqrt(2.0 * n)
    choice = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if choice == 0:  # pure power‑law
        p = rng.uniform(0.05, 1.5)
        a = (idx + 1.0) ** (-p)

    elif choice == 1:  # exponential decay
        lam = rng.uniform(0.001, 0.05)
        a = np.exp(-lam * idx)

    elif choice == 2:  # mixed power‑law + exponential
        p = rng.uniform(0.05, 1.5)
        lam = rng.uniform(0.001, 0.05)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif choice == 3:  # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif choice == 4:  # Hann window
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))

    elif choice == 5:  # Hamming window
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))

    elif choice == 6:  # Blackman window
        a = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1))
        )

    else:  # Gaussian bump
        sigma = rng.uniform(0.1, 0.4) * n
        mu = rng.uniform(0.0, 1.0) * n
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    # small Gaussian perturbation to avoid exact zeros
    a += rng.normal(scale=target_sum * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


def _upsample_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence to a larger length.
    The output is projected onto the correct simplex afterwards.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    seq_new = _project_to_simplex(seq_new, target)
    return seq_new

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser on the log‑sum‑exp smooth maximum.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # step scaling – empirically works well across lengths
    step_scale = target * 0.30

    beta = 0.5
    max_beta = 2048.0
    it = 0

    while time.time() < deadline:
        it += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # adaptive step
        step = step_scale / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto simplex
        a = _project_to_simplex(a, target)

        # anneal beta (smoothness) periodically
        if it % 300 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            jitter = np.random.default_rng().normal(
                scale=target * 0.0008 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return a


def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple SA that transfers a random fraction of mass between two
    indices; uphill moves are accepted with prob. exp(-Δ/T).  The
    temperature decays geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    T = cur_score * 0.02 + 1e-6
    cooling = 0.9997
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick two distinct indices (i has mass, j receives)
        i = rng.integers(0, n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 15 %) from i to j
        frac = rng.random() * 0.15
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a)
        dE = new_score - cur_score
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA improved: {best_score:.6f}")
        else:
            # revert
            a[j] -= delta
            a[i] += delta

        T *= cooling

        # occasional global jitter to shake out of deep minima
        if rng.random() < 0.001:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, a.sum())
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] SA jitter improved: {best_score:.6f}")

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    evaluate_sequence(...).  The routine respects the time budget,
    tries many vector lengths, uses diverse random initialisers,
    a projected Adam optimiser and a short simulated‑annealing
    refinement.  The best sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    # safety margin before hard deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a baseline construction is already present, use it
    # ------------------------------------------------------------------
    baseline = globals().get("height_sequence_1")
    if isinstance(baseline, (list, tuple, np.ndarray)):
        a0 = np.asarray(baseline, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        sc0 = _score(a0)
        best_seq = a0.copy()
        best_score = sc0
        print(
            f"[{time.time():.1f}] baseline from height_sequence_1 "
            f"n={n0} score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase A – short random sampling on short lengths to collect seeds
    # ------------------------------------------------------------------
    base_lengths = [800, 1200, 1500]
    top_per_len = 5
    seed_pool = []  # (score, array, length)

    for n in base_lengths:
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)
        phase_end = min(deadline, time.time() + 8.0)  # ~8 s per short length
        while time.time() < phase_end:
            a = _random_initial_sequence_extended(n, rng)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] New global best (rand short) "
                    f"n={n} score={best_score:.6f}"
                )
            # keep a few of the best candidates for up‑sampling later
            if len(seed_pool) < top_per_len or sc < seed_pool[-1][0]:
                seed_pool.append((sc, a.copy(), n))
                seed_pool.sort(key=lambda x: x[0])
                seed_pool = seed_pool[:top_per_len]

    # ------------------------------------------------------------------
    # Phase B – generate candidates for larger lengths
    # ------------------------------------------------------------------
    large_lengths = [2500, 4000, 6000, 8000]
    candidates = []

    for n in large_lengths:
        target = np.sqrt(2.0 * n)

        # up‑sample promising short seeds
        for sc, a_small, n_small in seed_pool:
            a_up = _upsample_seq(a_small, n)
            # re‑project to guarantee exact sum
            a_up = _project_to_simplex(a_up, target)
            candidates.append((sc, a_up.copy(), n))

        # a few fresh random seeds for the current length
        for _ in range(3):
            if time.time() >= deadline:
                break
            a_rand = _random_initial_sequence_extended(n, rng)
            candidates.append((_score(a_rand), a_rand.copy(), n))

    # ------------------------------------------------------------------
    # Phase C – optimise each candidate (Adam + optional SA)
    # ------------------------------------------------------------------
    # shuffle to avoid pathological ordering
    rng.shuffle(candidates)

    for orig_sc, a_start, n in candidates:
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate a generous chunk for Adam (up to 30 s or half of the left‑over)
        adam_time = min(30.0, remaining * 0.55)
        sub_dead = time.time() + adam_time

        a_adam = _adam_optimize(a_start, sub_dead)
        sc_adam = _score(a_adam)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
            print(
                f"[{time.time():.1f}] New best after Adam n={n} score={best_score:.6f}"
            )

        # If time permits, run a short SA refinement (≈5 s)
        if time.time() < deadline:
            sa_dead = min(deadline, time.time() + 5.0)
            a_sa = _simulated_annealing(a_adam, sa_dead)
            sc_sa = _score(a_sa)
            if sc_sa < best_score:
                best_score = sc_sa
                best_seq = a_sa.copy()
                print(
                    f"[{time.time():.1f}] New best after SA n={n} score={best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
        best_seq = a
        best_score = _score(a)
        print(f"[{time.time():.1f}] fallback uniform sequence, score={best_score:.6f}")

    # Trim trailing near‑zero entries for tidiness
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    print(
        f"[{time.time():.1f}] FINAL BEST score={best_score:.6f}, length={best_seq.size}"
    )
    return best_seq.tolist()
