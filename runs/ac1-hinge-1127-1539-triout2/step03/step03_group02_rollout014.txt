```
def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for i from i_start to i_end
        # j = k - i
        # column index = j
        # value = a[i]
        # That is, for each i, column = k - i.
        # So we can compute vector of columns: ks - i_range
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        rows.extend([k]*len(i_range))
        cols.extend(j_range.tolist())
        data.extend(a[i_range].tolist())
    T = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return T
```

"""
Propose a step‑function (non‑negative coefficient vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

which, after normalising the ℓ₁‑norm to √(2 n), is simply the maximum
self‑convolution.  The algorithm works on two resolutions:

1. **Small scale (≈1024)** – a fast parametric sweep, smoothed gradient
   descent and a *Frank–Wolfe* style linear‑programming step that
   maximises the ℓ₁‑mass under the current peak‑convolution bound.
   This yields a high‑quality U‑shape.

2. **Large scale (≈8192)** – the small‑scale shape is up‑sampled,
   re‑normalised and refined with smoothed gradient descent,
   sub‑gradient descent and a final polishing by random mass transfers.

All operations respect the simplex ∑aᵢ = √(2 n) and aᵢ ≥ 0.  The code
checks the remaining time budget and returns the best sequence found
before the 1000 s deadline.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic convolution / projection utilities
# ----------------------------------------------------------------------


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of `a` using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of `a`."""
    return float(conv_self_fft(a).max())


def project_onto_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
        {x | x_i ≥ 0, Σ_i x_i = target}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_pair_transfer(a: np.ndarray, target_sum: float,
                         scale: float = 0.01) -> np.ndarray:
    """Move a tiny amount of mass from a random index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * a[i]
    if delta <= 0:
        return a
    a[i] -= delta
    a[j] += delta
    return project_onto_simplex(a, target_sum)


def generate_shape(n: int, target_sum: float,
                   kind: str, param: float) -> np.ndarray:
    """
    Produce a symmetric “U‑shaped’’ sequence of length `n`.
    `kind` ∈ {linear, quadratic, exponential, power, inverse}.
    `param` controls the shape (exponent for power/inverse).
    """
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    d = np.abs(idx - mid) / mid                     # distance ∈ [0,1]

    if kind == "linear":
        raw = 1.0 + param * d
    elif kind == "quadratic":
        raw = 1.0 + param * d ** 2
    elif kind == "exponential":
        raw = np.exp(param * d)
    elif kind == "power":
        raw = d ** param + 1e-4
    elif kind == "inverse":
        raw = 1.0 / (1e-4 + d ** param)
    else:
        raise ValueError("unknown shape kind")
    raw = np.clip(raw, 0.0, None)
    return project_onto_simplex(raw, target_sum)


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp smoothing) and sub‑gradient
# ----------------------------------------------------------------------


def smooth_gradient_step(a: np.ndarray, target_sum: float,
                         eta: float, tau_factor: float,
                         momentum: np.ndarray, beta: float):
    """
    One iteration of smoothed gradient descent with momentum.
    Returns (new_a, new_momentum, max_conv).
    """
    conv = conv_self_fft(a)
    max_conv = conv.max()
    # log‑sum‑exp smoothing
    tau = max(tau_factor * max_conv, 1e-12)
    w = np.exp((conv - max_conv) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient = (2/τ) * conv(w, reverse(a))
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(W * A_rev, n=L)[: len(w) + len(a_rev) - 1]
    grad = (2.0 / tau) * conv_full[len(a) - 1: len(a) - 1 + len(a)]

    momentum = beta * momentum - eta * grad
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, momentum, max_conv


def subgradient_step(a: np.ndarray, target_sum: float,
                     step: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient step on the true max‑convolution.
    Returns (new_a, max_conv_before_step).
    """
    conv = conv_self_fft(a)
    max_conv = conv.max()
    eps = max(1e-12, 1e-9 * max_conv)
    mask = (conv >= max_conv - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt == 0:
        mask[conv.argmax()] = 1.0
        cnt = 1.0

    a_rev = a[::-1]
    L = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
    M = np.fft.rfft(mask, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(M * A_rev, n=L)[: len(mask) + len(a_rev) - 1]
    grad = (2.0 / cnt) * conv_full[len(a) - 1: len(a) - 1 + len(a)]

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_onto_simplex(a_new, target_sum)
    return a_new, max_conv


# ----------------------------------------------------------------------
# Linear‑programming direction (Frank–Wolfe sub‑problem)
# ----------------------------------------------------------------------


def solve_lp_direction(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        max Σ b_j
        s.t. b ≥ 0,
             b ≤ M,
             (a * b)[k] ≤ M   for all k,
    where * is linear convolution (a fixed, b variable).
    Returns the optimal `b` (as a dense ndarray) or None on failure.
    The full convolution matrix is built as a sparse Toeplitz matrix.
    """
    n = a.shape[0]
    # build sparse matrix representing the linear map b → a ∗ b
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        rows.extend([k] * len(i_range))
        cols.extend(j_range.tolist())
        data.extend(a[i_range].tolist())
    T = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = cp.Variable(n)
    constraints = [
        b >= 0,
        b <= M,
        T @ b <= M
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=False, max_iters=2000)
    except Exception:  # pragma: no cover
        return None
    if b.value is None:
        return None
    return np.array(b.value, dtype=np.float64)

        "inverse": np.linspace(0.5, 12.0, 25),
    }

    print(f"[{time.time():.2f}] Starting parametric sweep (n={small_n})")
    for kind in shape_kinds:
        for p in param_grids[kind]:
            cand = generate_shape(small_n, target_small, kind, float(p))
            score = compute_max_conv(cand)
            if score < best_small_score:
                best_small_score = score
                best_small = cand.copy()
                print(f"  New best {kind} (p={p:.3f}) → {score:.6f}")
            if time.time() >= deadline:
                print("Time exhausted during sweep (small).")
                return best_small.tolist()

    # --------------------------------------------------------------
    # Phase 1 – smoothed gradient descent on the small problem
    # --------------------------------------------------------------
    a_small = best_small.copy()
    momentum_small = np.zeros_like(a_small)
    eta = 0.8 * (target_small / small_n)   # relatively aggressive
    beta = 0.9
    tau_factor = 0.10
    iter_small = 0
    print(f"[{time.time():.2f}] Starting smoothed GD (small)")

    while time.time() < deadline - 30:   # keep ~30 s for later stages
        a_small, momentum_small, cur_max = smooth_gradient_step(
            a_small, target_small, eta, tau_factor, momentum_small, beta
        )
        iter_small += 1
        if cur_max < best_small_score:
            best_small_score = cur_max
            best_small = a_small.copy()
            # occasional logging
            print(f"Small GD iter {iter_small:06d} – new best {best_small_score:.6f}")
        if iter_small % 500 == 0:
            eta *= 0.97
            tau_factor = max(0.001, tau_factor * 0.96)
        if np.random.rand() < 0.012:
            a_small = random_pair_transfer(a_small, target_small, scale=0.015)

    # --------------------------------------------------------------
    # Phase 2 – Frank–Wolfe / LP refinement on the small problem
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting LP refinement (small)")
    lp_iter = 0
    max_lp_iters = 6
    while time.time() < deadline - 30 and lp_iter < max_lp_iters:
        M = compute_max_conv(a_small)
        b_dir = solve_lp_direction(a_small, M)
        if b_dir is None:
            print("  LP solver failed – breaking LP stage")
            break
        sum_b = b_dir.sum()
        if sum_b <= 0:
            break
        # scale to the ℓ₁‑target
        g = b_dir * (target_small / sum_b)
        # convex combination (step size α)
        alpha = 0.5
        a_new = (1 - alpha) * a_small + alpha * g
        a_new = project_onto_simplex(a_new, target_small)
        new_score = compute_max_conv(a_new)
        if new_score < best_small_score:
            best_small_score = new_score
            best_small = a_new.copy()
            a_small = a_new.copy()
            print(f"  LP iter {lp_iter+1} – improved to {new_score:.6f}")
        else:
            # try a smaller step if no improvement
            alpha = 0.2
            a_new = (1 - alpha) * a_small + alpha * g
            a_new = project_onto_simplex(a_new, target_small)
            new_score = compute_max_conv(a_new)
            if new_score < best_small_score:
                best_small_score = new_score
                best_small = a_new.copy()
                a_small = a_new.copy()
                print(f"  LP iter {lp_iter+1} (small α) – improved to {new_score:.6f}")
            else:
                print(f"  LP iter {lp_iter+1} – no improvement")
                break
        lp_iter += 1

    # --------------------------------------------------------------
    # Phase 3 – upscale to the big problem size
    # --------------------------------------------------------------
    factor = big_n // small_n
    a_big = np.repeat(best_small, factor)
    remainder = big_n - a_big.shape[0]
    if remainder > 0:
        a_big = np.concatenate([a_big, best_small[:remainder]])
    a_big = project_onto_simplex(a_big, target_big)
    best_big = a_big.copy()
    best_big_score = compute_max_conv(best_big)
    print(f"[{time.time():.2f}] Upscaled – initial big score {best_big_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – smoothed gradient descent on the big problem
    # --------------------------------------------------------------
    a = best_big.copy()
    momentum = np.zeros_like(a)
    eta = 0.3 * (target_big / big_n)
    beta = 0.9
    tau_factor = 0.06
    iter_big = 0
    print(f"[{time.time():.2f}] Starting smoothed GD (big)")

    while time.time() < deadline - 10:
        a, momentum, cur_max = smooth_gradient_step(
            a, target_big, eta, tau_factor, momentum, beta
        )
        iter_big += 1
        if cur_max < best_big_score:
            best_big_score = cur_max
            best_big = a.copy()
            print(f"Big GD iter {iter_big:07d} – new best {best_big_score:.6f}")
        if iter_big % 2000 == 0:
            eta *= 0.97
            tau_factor = max(0.001, tau_factor * 0.96)
        if np.random.rand() < 0.01:
            a = random_pair_transfer(a, target_big, scale=0.005)

    # --------------------------------------------------------------
    # Phase 5 – sub‑gradient refinement (big)
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Starting sub‑gradient refinement")
    step = 0.02 * (target_big / big_n)
    sub_iter = 0
    while time.time() < deadline - 2:
        a, cur_max = subgradient_step(a, target_big, step)
        sub_iter += 1
        if cur_max < best_big_score:
            best_big_score = cur_max
            best_big = a.copy()
            print(f"Sub‑grad iter {sub_iter:06d} – improved to {best_big_score:.6f}")
        if sub_iter % 500 == 0:
            step *= 0.97
        if np.random.rand() < 0.01:
            a = random_pair_transfer(a, target_big, scale=0.003)

    # --------------------------------------------------------------
    # Phase 6 – final polishing with tiny random transfers
    # --------------------------------------------------------------
    print(f"[{time.time():.2f}] Polishing phase")
    for _ in range(80):
        if time.time() >= deadline:
            break
        cand = random_pair_transfer(best_big.copy(), target_big, scale=0.002)
        cand_score = compute_max_conv(cand)
        if cand_score < best_big_score:
            best_big_score = cand_score
            best_big = cand.copy()
            print(f"Polish improved to {best_big_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.2f}] Returning best sequence, score {best_big_score:.6f}")
    return best_big.tolist()
