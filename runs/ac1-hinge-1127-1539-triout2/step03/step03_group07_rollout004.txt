def _run_exp_full(a0, n, deadline, eta0=0.5, decay=0.9999, beta=0.9, max_iters=50000, jitter_prob=0.02):
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    # renorm a to ensure sum = S
    total = a.sum()
    a *= S / total
    best_a = a.copy()
    conv = full_convolution(a)
    best_max = np.max(conv)
    v = np.zeros_like(a)
    eta = eta0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # Compute conv and max
        conv = full_convolution(a)
        M = np.max(conv)
        if M < best_max - 1e-12:
            best_max = M
            best_a = a.copy()
            print(f"[exp full improve] time={time.time():.1f}s n={n} iter={it} max={M:.6f}")
        # Compute gradient
        g, _ = max_shift_gradient(a, conv)
        # Momentum
        v = beta * v + (1.0 - beta) * g
        # Exponentiated update
        a = a * np.exp(-eta * v)
        # Ensure non-negative (should be)
        a = np.clip(a, 0.0, None)
        # Renormalize to sum S
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= S / total
        # jitter
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * a.mean() * 0.01
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()
        # decay learning rate
        eta *= decay
    return best_a, best_max

"""
Propose a candidate step‑function (list of non‑negative heights) that minimises

    2·n·max(a*a) / (sum(a)^2)

Because the evaluation function normalises the L1‑norm to √(2·n), the
objective is equivalent to minimise the maximum entry of the self‑convolution
under the constraint Σa = √(2·n) and a ≥ 0.

The search consists of two complementary phases that run in parallel
until the time budget (≈1000 s) is exhausted:

1. **Exact exponentiated (multiplicative) gradient descent**  
   – The sub‑gradient of the max‑convolution is computed *exactly* from the
     shift indices where the convolution attains its current maximum.  
   – An exponential update (mirror descent with KL‑divergence) together
     with momentum guarantees non‑negativity.  
   – After each step the vector is projected onto the simplex
     (Σa = √(2·n)).  The learning‑rate decays slowly.  Random jitter is
     occasionally added to escape flat plateaus.

2. **Gradient‑guided stochastic hill‑climbing**  
   – Uses the same exact gradient to pick a donor index (large contribution
     to the maximal shift) and a recipient index (small contribution) and
     moves a fraction of mass.  
   – Moves that improve the objective are always accepted; worsening moves
     are accepted with a temperature‑controlled probability (simulated
     annealing).  Small random jitter is injected occasionally.

Both phases are repeatedly restarted from random Dirichlet initialisations
with varying concentration and length (500 … 4000).  The best vector found
so far (lowest max‑convolution) is kept globally.  At the end the vector
is returned as a plain Python list – the external evaluator will compute
the final score.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear self‑convolution a*a using an FFT.
    Returns the full length (2·n‑1) array.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())          # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def max_shift_gradient(a: np.ndarray, conv: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute the sub‑gradient of the max‑convolution at the current point.
    Returns (g, M) where g_j = 2·∑_{k∈Kmax} a_{k‑j} and
    M = max(conv).
    """
    M = float(np.max(conv))
    # indices where conv is (practically) maximal
    tol = max(1e-12, 1e-12 * abs(M))
    max_inds = np.where(conv >= M - tol)[0]

    n = a.shape[0]
    idx = np.arange(n)
    g = np.zeros_like(a)

    # each maximal shift contributes 2·a_{k‑j}
    for k in max_inds:
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g[valid] += 2.0 * a[partner[valid]]
    return g, M


# ----------------------------------------------------------------------
# Exact exponentiated gradient descent (multiplicative updates)
# ----------------------------------------------------------------------


def _run_exp_full(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  eta0: float = 0.5,
                  decay: float = 0.9999,
                  beta: float = 0.9,
                  jitter_prob: float = 0.02,
                  max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """
    Exponentiated gradient descent that uses the *exact* sub‑gradient of the
    max‑convolution.  Returns the best vector encountered and its max‑conv.
    """
    S = np.sqrt(2.0 * n)                     # target L1‑norm
    a = a0.copy()
    a *= S / a.sum()                         # enforce Σa = S

    best_a = a.copy()
    conv = full_convolution(a)
    best_max = float(np.max(conv))

    v = np.zeros_like(a)                     # momentum buffer
    eta = eta0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution and exact gradient
        conv = full_convolution(a)
        g, M = max_shift_gradient(a, conv)

        # record improvement
        if M < best_max - 1e-12:
            best_max = M
            best_a = a.copy()
            print(f"[exp full improve] t={time.time():.1f}s n={n} it={it} max={M:.6f}")

        # momentum
        v = beta * v + (1.0 - beta) * g

        # exponentiated (multiplicative) update
        a = a * np.exp(-eta * v)

        # keep non‑negative and renormalise to the simplex
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a.fill(S / n)
        else:
            a *= (S / total)

        # occasional random jitter to escape flat regions
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()

        # learning‑rate decay
        eta *= decay

    return best_a, best_max


# ----------------------------------------------------------------------
# Gradient‑guided stochastic hill climbing / simulated annealing
# ----------------------------------------------------------------------


def _grad_hill_climb(start_vec: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that moves mass from indices with large
    gradient contribution to those with small contribution.
    Returns the best vector found and its max‑convolution value.
    """
    S = np.sqrt(2.0 * n)
    a = start_vec.copy()
    cur_max = float(np.max(full_convolution(a)))
    best_a = a.copy()
    best_max = cur_max

    temperature = max(0.05 * cur_max, 1e-4)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        conv = full_convolution(a)
        g, M = max_shift_gradient(a, conv)

        # donor = index with large (a·g) product, recipient = smallest g
        donor = int(np.argmax(g * a))
        recipient = int(np.argmin(g))

        # safeguard against degenerate choice
        if donor == recipient:
            donor = rng.integers(n)
            recipient = rng.integers(n)
            while recipient == donor:
                recipient = rng.integers(n)

        # move a random fraction of the donor's mass
        frac = rng.uniform(0.05, 0.20)
        delta = a[donor] * frac

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta

        new_max = float(np.max(full_convolution(a_new)))

        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(f"[hill improve] t={time.time():.1f}s n={n} it={it} max={new_max:.6f}")
        else:
            # simulated‑annealing acceptance
            prob = np.exp(-(new_max - cur_max) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_max = new_max

        # geometric cooling
        temperature *= 0.9995

        # occasional jitter for diversification
        if rng.random() < 0.001:
            noise = rng.normal(scale=a.mean() * 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a *= S / a.sum()
            cur_max = float(np.max(full_convolution(a)))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[hill jitter] t={time.time():.1f}s n={n} it={it} max={cur_max:.6f}")

    return best_a, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function of arbitrary length that yields a
    low score for ``evaluate_sequence``.  The function respects the time
    budget, runs many independent restarts of the exact exponentiated
    gradient descent and finishes with a gradient‑guided hill‑climbing
    refinement.  Returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 3.0      # small safety margin

    best_seq = None
    best_score = float("inf")                    # external score = max‑conv

    # ------------------------------------------------------------------
    # If a previously discovered construction is present, record it.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                sc = evaluate_sequence(cand.tolist())
                best_seq = cand.copy()
                best_score = sc
                print(f"[baseline] existing height_sequence_1 score={sc:.6f} n={cand.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter space for restarts
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 2000, 3000, 4000]
    probs = np.array([0.12, 0.15, 0.18, 0.25, 0.18, 0.12])
    probs /= probs.sum()

    run_id = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (bias towards the middle range)
        n = int(rng.choice(lengths, p=probs))

        # ------------------------------------------------------------------
        # Phase 1 – exponentiated gradient descent (majority of time)
        # ------------------------------------------------------------------
        if rng.random() < 0.75:
            conc = rng.uniform(0.05, 1.3)               # Dirichlet concentration
            a0 = rng.dirichlet(np.full(n, conc))
            a0 = a0 * np.sqrt(2.0 * n)                  # enforce Σa = √(2·n)

            # randomise hyper‑parameters a little
            eta0 = rng.uniform(0.25, 0.65)
            decay = rng.uniform(0.99980, 0.99996)
            max_iters = 50000

            a_opt, max_conv = _run_exp_full(
                a0, n, deadline, rng,
                eta0=eta0,
                decay=decay,
                beta=0.9,
                jitter_prob=0.02,
                max_iters=max_iters,
            )
            score = max_conv

        # ------------------------------------------------------------------
        # Phase 2 – gradient‑guided hill climbing (occasional)
        # ------------------------------------------------------------------
        else:
            if best_seq is not None and best_seq.size == n:
                start_vec = best_seq.copy()
            else:
                start_vec = rng.dirichlet(np.full(n, 1.0))
                start_vec = start_vec * np.sqrt(2.0 * n)

            a_opt, max_conv = _grad_hill_climb(
                start_vec, n, deadline, rng, max_iters=25000
            )
            score = max_conv

        # ------------------------------------------------------------------
        # Record global improvement
        # ------------------------------------------------------------------
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[global improve] run={run_id} n={n} score={best_score:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # Fallback (very unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification (the platform will also evaluate)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
