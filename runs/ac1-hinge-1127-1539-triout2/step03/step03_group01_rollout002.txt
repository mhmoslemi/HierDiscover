"""
A hybrid optimizer for the step‑function self‑convolution problem.

The algorithm mixes three ideas:
1. Projected soft‑max gradient descent on the max entry of a*a
   (the same technique used in the previous state‑of‑the‑art).
2. Occasional simulated‑annealing random mass moves (as before).
3. An LP‑based “active‑set” update: for the current vector a,
   solve a linear program that finds a vector g with larger ℓ₁‑norm
   while keeping the convolution a∗g bounded by the current maximum.
   Mixing a with g (convex combination) provably reduces the
   maximal self‑convolution entry, accelerating convergence.

The code performs many short restarts with different lengths (800‑8000)
and keeps the best sequence found before the time budget expires.
If a previous sequence is stored in the global variable
`height_sequence_1`, it is used as a warm‑start.
"""

import time
import numpy as np
from scipy import optimize, linalg

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=S}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=S}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - S
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, S / n)
    rho = np.where(cond)[0][-1]               # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# One soft‑max (or hard‑max) gradient step followed by projection
# ----------------------------------------------------------------------
def _softmax_update(a: np.ndarray, tau: float, eta: float):
    """
    Perform a single update:
      – compute b = a*a,
      – compute soft‑max weights w_k ∝ exp(b_k / τ),
      – evaluate the gradient g_i = 2 Σ_j w_{i+j} a_j,
      – take an additive step a ← a – η·g,
      – project onto the ℓ₁‑simplex.
    Returns the new vector, its convolution, and the new max entry.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # convolution and its current maximum
    b = _full_convolution_fft(a)
    max_b = np.max(b)

    if tau <= 1e-12:                      # treat as hard max (sub‑gradient)
        k = np.argmax(b)                 # index of the maximal entry
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]
        # diagonal contribution appears only once
        if k % 2 == 0:
            d = k // 2
            if 0 <= d < n:
                g[d] = a[d]
    else:                                 # smooth soft‑max gradient
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)

        # cross‑correlation a ⋆ w = Σ_j a_j w_{i+j}
        a_rev = a[::-1]
        m = 1 << ((len(w) + n - 1).bit_length())
        Fw = np.fft.rfft(w, n=m)
        Fa_rev = np.fft.rfft(a_rev, n=m)
        conv = np.fft.irfft(Fw * Fa_rev, n=m)[:len(w) + n - 1]
        g = 2.0 * conv[n - 1:n - 1 + n]

    # additive update and projection
    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)

    # recompute convolution after projection
    b_new = _full_convolution_fft(a_proj)
    max_new = np.max(b_new)

    return a_proj, b_new, max_new


# ----------------------------------------------------------------------
# Linear‑programming active‑set update
# ----------------------------------------------------------------------
def _lp_active_update(a: np.ndarray, cur_max: float, S: float, rng: np.random.Generator):
    """
    Solve max Σ g_j subject to g ≥ 0 and (a * g) ≤ cur_max.
    The feasible point a itself guarantees Σ g = S, therefore the
    optimum has Σ g ≥ S.  The resulting vector is scaled back to
    ℓ₁‑norm S and mixed with the current a.
    Returns a new candidate vector (or None on failure).
    """
    n = a.shape[0]
    # Build the convolution matrix A (shape (2n‑1) × n).  Entry (i,j) = a_{i‑j}
    # for i ≥ j and 0 otherwise – a Toeplitz matrix with first column a padded
    # with zeros and first row [a[0], 0, …, 0].
    first_col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    first_row = np.concatenate([np.array([a[0]], dtype=a.dtype), np.zeros(n - 1, dtype=a.dtype)])
    A = linalg.toeplitz(first_col, first_row)          # dense, shape (2n‑1, n)

    # Linear program: maximise Σ g  ⇔ minimise –Σ g
    c = -np.ones(n, dtype=a.dtype)
    b_ub = np.full(2 * n - 1, cur_max * (1.0 + 1e-9), dtype=a.dtype)   # tiny slack for numerics
    bounds = [(0.0, None)] * n

    # Solve with HiGHS (fast interior‑point).  If it fails, silently skip.
    res = optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                           method='highs', options={'presolve': True})
    if not res.success:
        return None

    g = res.x
    sum_g = np.sum(g)
    if sum_g <= 0:
        return None
    # Scale down so that Σ g = S (the LP solution typically has Σ g > S)
    scale = S / sum_g
    g_scaled = g * scale

    # Mix a and g_scaled with a modest step size (t = 0.2 works well)
    t = 0.2
    a_new = (1.0 - t) * a + t * g_scaled
    # Re‑project to the simplex to avoid drift from round‑off
    a_new = _project_to_simplex(a_new, S)
    return a_new

    tau = cur_max
    tau_decay = 0.9995
    lp_every = 20                     # do an LP update every 20 iterations
    it = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) random mass swap (simulated annealing)
        # --------------------------------------------------------------
        if rng.random() < move_prob:
            i = rng.integers(n)
            if a[i] > 0:
                j = rng.integers(n)
                if i != j:
                    delta = epsilon_move * a[i]
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    b_candidate = _full_convolution_fft(a_candidate)
                    cand_max = np.max(b_candidate)

                    accept = False
                    if cand_max <= cur_max:
                        accept = True
                    else:
                        prob = np.exp(-(cand_max - cur_max) / max(temp, 1e-12))
                        if rng.random() < prob:
                            accept = True
                    if accept:
                        a = a_candidate
                        cur_max = cand_max
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                    temp *= temp_decay

        # --------------------------------------------------------------
        # 2) projected soft‑max gradient step
        # --------------------------------------------------------------
        else:
            a, _, cur_max = _softmax_update(a, max(tau, 1e-12), eta)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # --------------------------------------------------------------
        # 3) occasional jitter
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            noise = rng.normal(scale=a.mean() * jitter_scale, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_to_simplex(a, S)
            b = _full_convolution_fft(a)
            cur_max = np.max(b)
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # --------------------------------------------------------------
        # 4) periodic LP active‑set refinement
        # --------------------------------------------------------------
        if it % lp_every == 0:
            lp_candidate = _lp_active_update(a, cur_max, S, rng)
            if lp_candidate is not None:
                a = lp_candidate
                b = _full_convolution_fft(a)
                cur_max = np.max(b)
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()

        # decay schedules
        eta *= eta_decay
        tau *= tau_decay
        it += 1

    return best_a, best_max

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Run many independent restarts of the hybrid optimiser within the
    allotted time budget and return the sequence with the smallest
    maximal self‑convolution entry (i.e. the smallest score returned by
    `evaluate_sequence`).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # reuse a previously discovered construction if available
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempt = 0

    # helper to create a feasible vector of length n
    def _initial_sequence(base_seq, n):
        S = np.sqrt(2.0 * n)
        if base_seq is not None and base_seq.size >= n:
            a = base_seq[:n].astype(float).copy()
            a += rng.normal(scale=max(a.mean(), 1e-6) * 0.001, size=n)
            a = np.clip(a, 0.0, None)
        else:
            a = rng.random(n)  # Dirichlet(1) – uniform on the simplex before projection
        a = _project_to_simplex(a, S)
        return a

    while time.time() < deadline:
        attempt += 1

        # pick a length – explore a wide interval, favouring moderate sizes
        if base_seq is not None:
            max_len = max(800, min(8000, int(base_seq.size * 1.5)))
        else:
            max_len = 8000
        n = rng.integers(800, max_len + 1)

        a0 = _initial_sequence(base_seq, n)

        # allocate a slice of the remaining budget for this restart
        slice_deadline = min(time.time() + 45.0, deadline)   # a bit longer than before
        a_opt, inner_max = _optimize_local(a0, slice_deadline, rng)

        if inner_max < best_overall_score:
            best_overall_score = inner_max
            best_overall_seq = a_opt.copy()
            print(f"[global] attempt={attempt} n={n} new best max={best_overall_score:.6f}")

    # final evaluation using the black‑box function
    if best_overall_seq is not None:
        final_seq = best_overall_seq.tolist()
        external_score = evaluate_sequence(final_seq)
        print(f"[final] best external score = {external_score:.6f} "
              f"(internal max = {best_overall_score:.6f}) "
              f"length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq
        return final_seq
    else:
        # fallback – should never happen
        return [1.0]
